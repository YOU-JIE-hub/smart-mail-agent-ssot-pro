#!/usr/bin/env bash
set -Eeuo pipefail
set -o errtrace
trap 'echo "[FAIL] line=$LINENO cmd=$BASH_COMMAND code=$?" >&2' ERR

log(){ printf '%s\n' "$*"; }
fail(){ printf '[FAIL] %s\n' "$*" >&2; exit 96; }

TS="$(date +%Y%m%dT%H%M%S)"
MAIN="src/ai_rpa/main.py"
[[ -f "$MAIN" ]] || fail "缺少 $MAIN"
cp -f "$MAIN" "$MAIN.bak.$TS"
log "[BACKUP] $MAIN -> $MAIN.bak.$TS"

python3 - "$MAIN" <<'PY'
from pathlib import Path
import re, sys

p = Path(sys.argv[1])
s = p.read_text(encoding="utf-8")

changed = False

# (1) 修正 NameError：將 if output_path: 統一換成 getattr(ns, 'output', None)
s2 = re.sub(r'\bif\s+output_path\b', "if getattr(ns, 'output', None)", s)
if s2 != s:
    s = s2
    changed = True

# (2) 先移除所有先前誤插的 print(payload)，避免重複輸出與縮排錯
s2 = re.sub(r'(?m)^\s*print\(\s*payload\s*\)\s*\n', '', s)
if s2 != s:
    s = s2
    changed = True

# (3) 在「payload = json.dumps(out, ensure_ascii=False)」當行之後、同一縮排層級，插入唯一的 print(payload)
pat = re.compile(r'(?m)^(\s*)payload\s*=\s*json\s*\.\s*dumps\s*\(\s*out\s*,\s*ensure_ascii\s*=\s*False\s*\)\s*$')

def _inject_print_once(text: str) -> str:
    m = pat.search(text)
    if not m:
        return text  # 找不到就不動，避免破壞；之後也不會報縮排錯
    indent = m.group(1)
    # 檢查接下來是否已有 print(payload)
    tail = text[m.end(): m.end()+200]
    if re.search(r'(?m)^\s*print\(\s*payload\s*\)\s*$', tail):
        return text  # 已存在
    return text[:m.end()] + f"\n{indent}print(payload)" + text[m.end():]

s2 = _inject_print_once(s)
if s2 != s:
    s = s2
    changed = True

# (4) 避免空的 else 導致縮排錯：如果 else: 之後緊接著 comment/空行/另一個 if，就補一個 pass
s2 = re.sub(r'(\n\s*else:\s*\n)(\s*)(?=(#.*\n|\s*\n|\s*if\b))', r'\1\2pass  # keep else non-empty\n', s)
if s2 != s:
    s = s2
    changed = True

if changed:
    p.write_text(s, encoding="utf-8")
    print("[PATCH] main.py: 修正 output 判斷 + 單一 stdout 輸出完成")
else:
    print("[SKIP] main.py 無需修補")
PY

# (5) 修 shim：若存在，確保使用相對匯入
SHIM="src/smart_mail_agent/spam/inference_classifier.py"
if [[ -f "$SHIM" ]]; then
  cp -f "$SHIM" "$SHIM.bak.$TS"
  python3 - "$SHIM" <<'PY'
from pathlib import Path
p = Path(__import__('sys').argv[1])
s = p.read_text(encoding='utf-8')
s2 = s.replace(
    "from src.smart_mail_agent.inference_classifier import (",
    "from ..inference_classifier import ("
)
if s2 != s:
    p.write_text(s2, encoding='utf-8')
    print("[PATCH] 修正 shim 匯入路徑")
else:
    print("[SKIP] shim 無需修補")
PY
fi

# (6) 最小煙囪：主程式必須吐出 JSON 到 stdout
mkdir -p samples
printf "合作與退款測試\n" > samples/in.txt
export OFFLINE="${OFFLINE:-1}"
out="$(python -m ai_rpa.main --tasks "nlp,mailguard,actions" --input-path samples/in.txt --dry-run)"
head -c 200 <<<"$out"; echo
[[ -n "$out" ]] || fail "未見 stdout JSON"
log "[OK] hotfix 完成"
