   219	        o.body.accept(self)
   220	        for i in range(len(o.types)):
   221	            tp = o.types[i]
   222	            if tp is not None:
   223	                tp.accept(self)
   224	            o.handlers[i].accept(self)
   225	        for v in o.vars:
   226	            if v is not None:
   227	                v.accept(self)
   228	        if o.else_body is not None:
   229	            o.else_body.accept(self)
   230	        if o.finally_body is not None:
   231	            o.finally_body.accept(self)

   237	            if targ is not None:
   238	                targ.accept(self)
   239	        o.body.accept(self)

   245	            guard = o.guards[i]
   246	            if guard is not None:
   247	                guard.accept(self)
   248	            o.bodies[i].accept(self)
