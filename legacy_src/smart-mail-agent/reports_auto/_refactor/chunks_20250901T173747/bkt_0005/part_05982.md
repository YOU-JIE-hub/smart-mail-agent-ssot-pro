  108	                )
   109
   110	    # Prepare implicit attribute accessors as needed if an attribute overrides a property.
   111	    for module, cdef in classes:
   112	        class_ir = mapper.type_to_ir[cdef.info]
   113	        if class_ir.is_ext_class:
   114	            prepare_implicit_property_accessors(cdef.info, class_ir, module.fullname, mapper)
   115
   116	    # Collect all the functions also. We collect from the symbol table
   117	    # so that we can easily pick out the right copy of a function that
   118	    # is conditionally defined.
   119	    for module in modules:
   120	        for func in get_module_func_defs(module):
   121	            prepare_func_def(module.fullname, None, func, mapper, options)
   122	            # TODO: what else?
   123
   124	    # Check for incompatible attribute definitions that were not
   125	    # flagged by mypy but can't be supported when compiling.
   126	    for module, cdef in classes:
   127	        class_ir = mapper.type_to_ir[cdef.info]
   128	        for attr in class_ir.attributes:
   129	            for base_ir in class_ir.mro[1:]:
   130	                if attr in base_ir.attributes:
   131	                    if not is_same_type(class_ir.attributes[attr], base_ir.attributes[attr]):
   132	                        node = cdef.info.names[attr].node
   133	                        assert node is not None
   134	                        kind = "trait" if base_ir.is_trait else "class"
   135	                        errors.error(
   136	                            f'Type of "{attr}" is incompatible with '
   137	                            f'definition in {kind} "{base_ir.name}"',
   138	                            module.path,
   139	                            node.line,
   140	                        )

   176	    module_name: str,

   193	def prepare_method_def(
   194	    ir: ClassIR,
   195	    module_name: str,
   196	    cdef: ClassDef,
   197	    mapper: Mapper,
   198	    node: FuncDef | Decorator,
   199	    options: CompilerOptions,
   200	) -> None:
   201	    if isinstance(node, FuncDef):
   202	        ir.method_decls[node.name] = prepare_func_def(
   203	            module_name, cdef.name, node, mapper, options
   204	        )
   205	    elif isinstance(node, Decorator):
   206	        # TODO: do something about abstract methods here. Currently, they are handled just like
   207	        # normal methods.
   208	        decl = prepare_func_def(module_name, cdef.name, node.func, mapper, options)
   209	        if not node.decorators:
   210	            ir.method_decls[node.name] = decl
   211	        elif isinstance(node.decorators[0], MemberExpr) and node.decorators[0].name == "setter":
   212	            # Make property setter name different than getter name so there are no
   213	            # name clashes when generating C code, and property lookup at the IR level
   214	            # works correctly.
   215	            decl.name = PROPSET_PREFIX + decl.name
   216	            decl.is_prop_setter = True
   217	            # Making the argument implicitly positional-only avoids unnecessary glue methods
   218	            decl.sig.args[1].pos_only = True
   219	            ir.method_decls[PROPSET_PREFIX + node.name] = decl
   220
   221	        if node.func.is_property:
   222	            assert node.func.type, f"Expected return type annotation for property '{node.name}'"
   223	            decl.is_prop_getter = True
   224	            ir.property_types[node.name] = decl.sig.ret_type

   227	def is_valid_multipart_property_def(prop: OverloadedFuncDef) -> bool:
   228	    # Checks to ensure supported property decorator semantics
   229	    if len(prop.items) != 2:
   230	        return False
   231
   232	    getter = prop.items[0]
   233	    setter = prop.items[1]
   234
   235	    return (
   236	        isinstance(getter, Decorator)
   237	        and isinstance(setter, Decorator)
   238	        and getter.func.is_property
   239	        and len(setter.decorators) == 1
   240	        and isinstance(setter.decorators[0], MemberExpr)
   241	        and setter.decorators[0].name == "setter"
   242	    )

   260	def prepare_class_def(
   261	    path: str,
   262	    module_name: str,
   263	    cdef: ClassDef,
   264	    errors: Errors,
   265	    mapper: Mapper,
   266	    options: CompilerOptions,
   267	) -> None:
   268	    """Populate the interface-level information in a class IR.
   269
   270	    This includes attribute and method declarations, and the MRO, among other things, but
   271	    method bodies are generated in a later pass.
   272	    """
   273
   274	    ir = mapper.type_to_ir[cdef.info]
   275	    info = cdef.info
   276
   277	    attrs = get_mypyc_attrs(cdef)
   278	    if attrs.get("allow_interpreted_subclasses") is True:
   279	        ir.allow_interpreted_subclasses = True
   280	    if attrs.get("serializable") is True:
   281	        # Supports copy.copy and pickle (including subclasses)
   282	        ir._serializable = True
   283
   284	    # Check for subclassing from builtin types
   285	    for cls in info.mro:
   286	        # Special case exceptions and dicts
   287	        # XXX: How do we handle *other* things??
   288	        if cls.fullname == "builtins.BaseException":
   289	            ir.builtin_base = "PyBaseExceptionObject"
   290	        elif cls.fullname == "builtins.dict":
   291	            ir.builtin_base = "PyDictObject"
   292	        elif cls.fullname.startswith("builtins."):
   293	            if not can_subclass_builtin(cls.fullname):
   294	                # Note that if we try to subclass a C extension class that
   295	                # isn't in builtins, bad things will happen and we won't
   296	                # catch it here! But this should catch a lot of the most
   297	                # common pitfalls.
   298	                errors.error(
   299	                    "Inheriting from most builtin types is unimplemented", path, cdef.line
   300	                )
   301	                errors.note(
   302	                    "Potential workaround: @mypy_extensions.mypyc_attr(native_class=False)",
   303	                    path,
   304	                    cdef.line,
   305	                )
   306	                errors.note(
   307	                    "https://mypyc.readthedocs.io/en/stable/native_classes.html#defining-non-native-classes",
   308	                    path,
   309	                    cdef.line,
   310	                )
   311
   312	    # Set up the parent class
   313	    bases = [mapper.type_to_ir[base.type] for base in info.bases if base.type in mapper.type_to_ir]
   314	    if len(bases) > 1 and any(not c.is_trait for c in bases) and bases[0].is_trait:
   315	        # If the first base is a non-trait, don't ever error here. While it is correct
   316	        # to error if a trait comes before the next non-trait base (e.g. non-trait, trait,
   317	        # non-trait), it's pointless, confusing noise from the bigger issue: multiple
   318	        # inheritance is *not* supported.
   319	        errors.error("Non-trait base must appear first in parent list", path, cdef.line)
   320	    ir.traits = [c for c in bases if c.is_trait]
   321
   322	    mro = []  # All mypyc base classes
   323	    base_mro = []  # Non-trait mypyc base classes
   324	    for cls in info.mro:
   325	        if cls not in mapper.type_to_ir:
   326	            if cls.fullname != "builtins.object":
   327	                ir.inherits_python = True
   328	            continue
   329	        base_ir = mapper.type_to_ir[cls]
   330	        if not base_ir.is_trait:
   331	            base_mro.append(base_ir)
   332	        mro.append(base_ir)
   333
   334	        if cls.defn.removed_base_type_exprs or not base_ir.is_ext_class:
   335	            ir.inherits_python = True
   336
   337	    base_idx = 1 if not ir.is_trait else 0
   338	    if len(base_mro) > base_idx:
   339	        ir.base = base_mro[base_idx]
   340	    ir.mro = mro
   341	    ir.base_mro = base_mro
   342
   343	    prepare_methods_and_attributes(cdef, ir, path, module_name, errors, mapper, options)
   344	    prepare_init_method(cdef, ir, module_name, mapper)
   345
   346	    for base in bases:
   347	        if base.children is not None:
   348	            base.children.append(ir)
   349
   350	    if is_dataclass(cdef):
   351	        ir.is_augmented = True
   352

   354	def prepare_methods_and_attributes(
   355	    cdef: ClassDef,
   356	    ir: ClassIR,
   357	    path: str,
   358	    module_name: str,
   359	    errors: Errors,
   360	    mapper: Mapper,
   361	    options: CompilerOptions,
   362	) -> None:
   363	    """Populate attribute and method declarations."""
   364	    info = cdef.info
   365	    for name, node in info.names.items():
   366	        # Currently all plugin generated methods are dummies and not included.
   367	        if node.plugin_generated:
   368	            continue
   369
   370	        if isinstance(node.node, Var):
   371	            assert node.node.type, "Class member %s missing type" % name
   372	            if not node.node.is_classvar and name not in ("__slots__", "__deletable__"):
   373	                attr_rtype = mapper.type_to_rtype(node.node.type)
   374	                if ir.is_trait and attr_rtype.error_overlap:
   375	                    # Traits don't have attribute definedness bitmaps, so use
   376	                    # property accessor methods to access attributes that need them.
   377	                    # We will generate accessor implementations that use the class bitmap
   378	                    # for any concrete subclasses.
   379	                    add_getter_declaration(ir, name, attr_rtype, module_name)
   380	                    add_setter_declaration(ir, name, attr_rtype, module_name)
   381	                ir.attributes[name] = attr_rtype
   382	        elif isinstance(node.node, (FuncDef, Decorator)):
   383	            prepare_method_def(ir, module_name, cdef, mapper, node.node, options)
   384	        elif isinstance(node.node, OverloadedFuncDef):
   385	            # Handle case for property with both a getter and a setter
   386	            if node.node.is_property:
   387	                if is_valid_multipart_property_def(node.node):
   388	                    for item in node.node.items:
   389	                        prepare_method_def(ir, module_name, cdef, mapper, item, options)
   390	                else:
   391	                    errors.error("Unsupported property decorator semantics", path, cdef.line)
   392
   393	            # Handle case for regular function overload
   394	            else:
   395	                if not node.node.impl:
   396	                    errors.error(
   397	                        "Overloads without implementation are not supported", path, cdef.line
   398	                    )
   399	                else:
   400	                    prepare_method_def(ir, module_name, cdef, mapper, node.node.impl, options)
   401
   402	    if ir.builtin_base:
   403	        ir.attributes.clear()
   404

   406	def prepare_implicit_property_accessors(
   407	    info: TypeInfo, ir: ClassIR, module_name: str, mapper: Mapper
   408	) -> None:
   409	    concrete_attributes = set()
   410	    for base in ir.base_mro:
   411	        for name, attr_rtype in base.attributes.items():
   412	            concrete_attributes.add(name)
   413	            add_property_methods_for_attribute_if_needed(
   414	                info, ir, name, attr_rtype, module_name, mapper
   415	            )
   416	    for base in ir.mro[1:]:
   417	        if base.is_trait:
   418	            for name, attr_rtype in base.attributes.items():
   419	                if name not in concrete_attributes:
   420	                    add_property_methods_for_attribute_if_needed(
   421	                        info, ir, name, attr_rtype, module_name, mapper
   422	                    )

   425	def add_property_methods_for_attribute_if_needed(
   426	    info: TypeInfo,
   427	    ir: ClassIR,
   428	    attr_name: str,
   429	    attr_rtype: RType,
   430	    module_name: str,
   431	    mapper: Mapper,
   432	) -> None:
   433	    """Add getter and/or setter for attribute if defined as property in a base class.
   434
   435	    Only add declarations. The body IR will be synthesized later during irbuild.
   436	    """
   437	    for base in info.mro[1:]:
   438	        if base in mapper.type_to_ir:
   439	            base_ir = mapper.type_to_ir[base]
   440	            n = base.names.get(attr_name)
   441	            if n is None:
   442	                continue
   443	            node = n.node
   444	            if isinstance(node, Decorator) and node.name not in ir.method_decls:
   445	                # Defined as a read-only property in base class/trait
   446	                add_getter_declaration(ir, attr_name, attr_rtype, module_name)
   447	            elif isinstance(node, OverloadedFuncDef) and is_valid_multipart_property_def(node):
   448	                # Defined as a read-write property in base class/trait
   449	                add_getter_declaration(ir, attr_name, attr_rtype, module_name)
   450	                add_setter_declaration(ir, attr_name, attr_rtype, module_name)
   451	            elif base_ir.is_trait and attr_rtype.error_overlap:
   452	                add_getter_declaration(ir, attr_name, attr_rtype, module_name)
   453	                add_setter_declaration(ir, attr_name, attr_rtype, module_name)

   456	def add_getter_declaration(
   457	    ir: ClassIR, attr_name: str, attr_rtype: RType, module_name: str
   458	) -> None:
   459	    self_arg = RuntimeArg("self", RInstance(ir), pos_only=True)
   460	    sig = FuncSignature([self_arg], attr_rtype)
   461	    decl = FuncDecl(attr_name, ir.name, module_name, sig, FUNC_NORMAL)
   462	    decl.is_prop_getter = True
   463	    decl.implicit = True  # Triggers synthesization
   464	    ir.method_decls[attr_name] = decl
   465	    ir.property_types[attr_name] = attr_rtype  # TODO: Needed??

   469	    ir: ClassIR, attr_name: str, attr_rtype: RType, module_name: str
   470	) -> None:
   471	    self_arg = RuntimeArg("self", RInstance(ir), pos_only=True)
   472	    value_arg = RuntimeArg("value", attr_rtype, pos_only=True)

   474	    setter_name = PROPSET_PREFIX + attr_name

   481	def prepare_init_method(cdef: ClassDef, ir: ClassIR, module_name: str, mapper: Mapper) -> None:
   482	    # Set up a constructor decl
   483	    init_node = cdef.info["__init__"].node
   484	    if not ir.is_trait and not ir.builtin_base and isinstance(init_node, FuncDef):
   485	        init_sig = mapper.fdef_to_sig(init_node, True)
   486
   487	        defining_ir = mapper.type_to_ir.get(init_node.info)
   488	        # If there is a nontrivial __init__ that wasn't defined in an
   489	        # extension class, we need to make the constructor take *args,
   490	        # **kwargs so it can call tp_init.
   491	        if (
   492	            defining_ir is None
   493	            or not defining_ir.is_ext_class
   494	            or cdef.info["__init__"].plugin_generated
   495	        ) and init_node.info.fullname != "builtins.object":
   496	            init_sig = FuncSignature(
   497	                [
   498	                    init_sig.args[0],
   499	                    RuntimeArg("args", tuple_rprimitive, ARG_STAR),
   500	                    RuntimeArg("kwargs", dict_rprimitive, ARG_STAR2),
   501	                ],
   502	                init_sig.ret_type,
   503	            )
   504
   505	        last_arg = len(init_sig.args) - init_sig.num_bitmap_args
   506	        ctor_sig = FuncSignature(init_sig.args[1:last_arg], RInstance(ir))
   507	        ir.ctor = FuncDecl(cdef.name, None, module_name, ctor_sig)
   508	        mapper.func_to_decl[cdef.info] = ir.ctor

   550	def find_singledispatch_register_impls(
   551	    modules: list[MypyFile], errors: Errors
   552	) -> SingledispatchInfo:
   553	    visitor = SingledispatchVisitor(errors)
   554	    for module in modules:
   555	        visitor.current_path = module.path
   556	        module.accept(visitor)
   557	    return SingledispatchInfo(visitor.singledispatch_impls, visitor.decorators_to_remove)

   560	class SingledispatchVisitor(TraverserVisitor):
   561	    current_path: str
   562
   563	    def __init__(self, errors: Errors) -> None:
   564	        super().__init__()
   565
   566	        # Map of main singledispatch function to list of registered implementations
   567	        self.singledispatch_impls: defaultdict[FuncDef, list[RegisterImplInfo]] = defaultdict(list)
   568
   569	        # Map of decorated function to the indices of any decorators to remove
   570	        self.decorators_to_remove: dict[FuncDef, list[int]] = {}
   571
   572	        self.errors: Errors = errors
   573
   574	    def visit_decorator(self, dec: Decorator) -> None:
   575	        if dec.decorators:
   576	            decorators_to_store = dec.decorators.copy()
   577	            decorators_to_remove: list[int] = []
   578	            # the index of the last non-register decorator before finding a register decorator
   579	            # when going through decorators from top to bottom
   580	            last_non_register: int | None = None
   581	            for i, d in enumerate(decorators_to_store):
   582	                impl = get_singledispatch_register_call_info(d, dec.func)
   583	                if impl is not None:
   584	                    self.singledispatch_impls[impl.singledispatch_func].append(
   585	                        (impl.dispatch_type, dec.func)
   586	                    )
   587	                    decorators_to_remove.append(i)
   588	                    if last_non_register is not None:
   589	                        # found a register decorator after a non-register decorator, which we
   590	                        # don't support because we'd have to make a copy of the function before
   591	                        # calling the decorator so that we can call it later, which complicates
   592	                        # the implementation for something that is probably not commonly used
   593	                        self.errors.error(
   594	                            "Calling decorator after registering function not supported",
   595	                            self.current_path,
   596	                            decorators_to_store[last_non_register].line,
   597	                        )
   598	                else:
   599	                    if refers_to_fullname(d, "functools.singledispatch"):
   600	                        decorators_to_remove.append(i)
   601	                        # make sure that we still treat the function as a singledispatch function
   602	                        # even if we don't find any registered implementations (which might happen
   603	                        # if all registered implementations are registered dynamically)
   604	                        self.singledispatch_impls.setdefault(dec.func, [])
   605	                    last_non_register = i
   606
   607	            if decorators_to_remove:
   608	                # calling register on a function that tries to dispatch based on type annotations
   609	                # raises a TypeError because compiled functions don't have an __annotations__
   610	                # attribute
   611	                self.decorators_to_remove[dec.func] = decorators_to_remove
   612
   613	        super().visit_decorator(dec)

   621	def get_singledispatch_register_call_info(
   622	    decorator: Expression, func: FuncDef

   830	def truncate_literal(value: Value, rtype: RPrimitive) -> Value:
   831	    """If value is an integer literal value, truncate it to given native int rtype.
   832
   833	    For example, truncate 256 into 0 if rtype is u8.
   834	    """
   835	    if not isinstance(value, Integer):
   836	        return value  # Not a literal, nothing to do

   841	        # Adjust to make it a negative value

   178	        builder.get_assignment_target(first_lvalue, stmt.line)
   179	        return
   180
   181	    # Special case multiple assignments like 'x, y = e1, e2'.
   182	    if (
   183	        isinstance(first_lvalue, (TupleExpr, ListExpr))
   184	        and isinstance(stmt.rvalue, (TupleExpr, ListExpr))
   185	        and len(first_lvalue.items) == len(stmt.rvalue.items)
   186	        and all(is_simple_lvalue(item) for item in first_lvalue.items)
   187	        and len(lvalues) == 1
   188	    ):
   189	        temps = []
   190	        for right in stmt.rvalue.items:
   191	            rvalue_reg = builder.accept(right)
   192	            temp = Register(rvalue_reg.type)
   193	            builder.assign(temp, rvalue_reg, stmt.line)
   194	            temps.append(temp)
   195	        for left, temp in zip(first_lvalue.items, temps):
   196	            assignment_target = builder.get_assignment_target(left)
   197	            builder.assign(assignment_target, temp, stmt.line)
   198	        builder.flush_keep_alives()
   199	        return
   200
   201	    line = stmt.rvalue.line
   202	    rvalue_reg = builder.accept(stmt.rvalue)
   203
   204	    if builder.non_function_scope() and stmt.is_final_def:
   205	        builder.init_final_static(first_lvalue, rvalue_reg)
   206
   207	    # Special-case multiple assignments like 'x, y = expr' to reduce refcount ops.
   208	    if (
   209	        isinstance(first_lvalue, (TupleExpr, ListExpr))
   210	        and isinstance(rvalue_reg.type, RTuple)
   211	        and len(rvalue_reg.type.types) == len(first_lvalue.items)
   212	        and len(lvalues) == 1
   213	        and all(is_simple_lvalue(item) for item in first_lvalue.items)
   214	        and any(t.is_refcounted for t in rvalue_reg.type.types)
   215	    ):
   216	        n = len(first_lvalue.items)
   217	        borrows = [builder.add(TupleGet(rvalue_reg, i, borrow=True)) for i in range(n)]
   218	        builder.builder.keep_alive([rvalue_reg], steal=True)
   219	        for lvalue_item, rvalue_item in zip(first_lvalue.items, borrows):
   220	            rvalue_item = builder.add(Unborrow(rvalue_item))
   221	            builder.assign(builder.get_assignment_target(lvalue_item), rvalue_item, line)
   222	        builder.flush_keep_alives()
   223	        return
   224
   225	    for lvalue in lvalues:
   226	        target = builder.get_assignment_target(lvalue)
   227	        builder.assign(target, rvalue_reg, line)
   228	        builder.flush_keep_alives()
   229

   236	    """Operator assignment statement such as x += 1"""
   237	    builder.disallow_class_assignments([stmt.lvalue], stmt.line)
   238	    if (
   239	        is_tagged(builder.node_type(stmt.lvalue))
   240	        and is_tagged(builder.node_type(stmt.rvalue))
   241	        and stmt.op in int_borrow_friendly_op
   242	    ):
   243	        can_borrow = is_borrow_friendly_expr(builder, stmt.rvalue) and is_borrow_friendly_expr(
   244	            builder, stmt.lvalue
   245	        )

   247	        can_borrow = False

   256	    builder.assign(target, res, res.line)
   257	    builder.flush_keep_alives()

   260	def import_globals_id_and_name(module_id: str, as_name: str | None) -> tuple[str, str]:
   261	    """Compute names for updating the globals dict with the appropriate module.
   262
   263	    * For 'import foo.bar as baz' we add 'foo.bar' with the name 'baz'
   264	    * For 'import foo.bar' we add 'foo' with the name 'foo'

   266	    Typically we then ignore these entries and access things directly
   267	    via the module static, but we will use the globals version for
   268	    modules that mypy couldn't find, since it doesn't analyze module
   269	    references from those properly."""
   270	    if as_name:
   271	        globals_id = module_id
   272	        globals_name = as_name
   273	    else:
   274	        globals_id = globals_name = module_id.split(".")[0]

   276	    return globals_id, globals_name

   279	def transform_import(builder: IRBuilder, node: Import) -> None:
   280	    if node.is_mypy_only:
   281	        return
   282
   283	    # Imports (not from imports!) are processed in an odd way so they can be
   284	    # table-driven and compact. Here's how it works:
   285	    #
   286	    # Import nodes are divided in groups (in the prebuild visitor). Each group
   287	    # consists of consecutive Import nodes:
   288	    #
   289	    #   import mod         <| group #1
   290	    #   import mod2         |
   291	    #
   292	    #   def foo() -> None:
   293	    #       import mod3    <- group #2 (*)
   294	    #
   295	    #   import mod4        <| group #3
   296	    #   import mod5         |
   297	    #
   298	    # Every time we encounter the first import of a group, build IR to call a
   299	    # helper function that will perform all of the group's imports in one go.
   300	    if not node.is_top_level:
   301	        # (*) Unless the import is within a function. In that case, prioritize
   302	        # speed over codesize when generating IR.
   303	        globals = builder.load_globals_dict()
   304	        for mod_id, as_name in node.ids:
   305	            builder.gen_import(mod_id, node.line)
   306	            globals_id, globals_name = import_globals_id_and_name(mod_id, as_name)
   307	            builder.gen_method_call(
   308	                globals,
   309	                "__setitem__",
   310	                [builder.load_str(globals_name), builder.get_module(globals_id, node.line)],
   311	                result_type=None,
   312	                line=node.line,
   313	            )
   314	        return
   315
   316	    if node not in builder.module_import_groups:
   317	        return
   318
   319	    modules = []
   320	    static_ptrs = []
   321	    # To show the right line number on failure, we have to add the traceback
   322	    # entry within the helper function (which is admittedly ugly). To drive
   323	    # this, we need the line number corresponding to each module.
   324	    mod_lines = []
   325	    for import_node in builder.module_import_groups[node]:
   326	        for mod_id, as_name in import_node.ids:
   327	            builder.imports[mod_id] = None
   328	            modules.append((mod_id, *import_globals_id_and_name(mod_id, as_name)))
   329	            mod_static = LoadStatic(object_rprimitive, mod_id, namespace=NAMESPACE_MODULE)
   330	            static_ptrs.append(builder.add(LoadAddress(object_pointer_rprimitive, mod_static)))
   331	            mod_lines.append(Integer(import_node.line, c_pyssize_t_rprimitive))
   332
   333	    static_array_ptr = builder.builder.setup_rarray(object_pointer_rprimitive, static_ptrs)
   334	    import_line_ptr = builder.builder.setup_rarray(c_pyssize_t_rprimitive, mod_lines)
   335	    builder.call_c(
   336	        import_many_op,
   337	        [
   338	            builder.add(LoadLiteral(tuple(modules), object_rprimitive)),
   339	            static_array_ptr,
   340	            builder.load_globals_dict(),
   341	            builder.load_str(builder.module_path),
   342	            builder.load_str(builder.fn_info.name),
   343	            import_line_ptr,
   344	        ],
   345	        NO_TRACEBACK_LINE_NO,
   346	    )

   349	def transform_import_from(builder: IRBuilder, node: ImportFrom) -> None:
   350	    if node.is_mypy_only:
   351	        return
   352
   353	    module_state = builder.graph[builder.module_name]
   354	    if builder.module_path.endswith("__init__.py"):
   355	        module_package = builder.module_name
   356	    elif module_state.ancestors is not None and module_state.ancestors:
   357	        module_package = module_state.ancestors[0]
   358	    else:
   359	        module_package = ""
   360
   361	    id = importlib.util.resolve_name("." * node.relative + node.id, module_package)
   362	    builder.imports[id] = None
   363
   364	    names = [name for name, _ in node.names]
   365	    as_names = [as_name or name for name, as_name in node.names]
   366	    names_literal = builder.add(LoadLiteral(tuple(names), object_rprimitive))
   367	    if as_names == names:
   368	        # Reuse names tuple to reduce verbosity.
   369	        as_names_literal = names_literal
   370	    else:
   371	        as_names_literal = builder.add(LoadLiteral(tuple(as_names), object_rprimitive))

   373	    # since that case *shouldn't* load everything into the globals dict.

   376	        import_from_many_op,
   377	        [builder.load_str(id), names_literal, as_names_literal, builder.load_globals_dict()],
   378	        node.line,

   407	def transform_while_stmt(builder: IRBuilder, s: WhileStmt) -> None:
   408	    body, next, top, else_block = BasicBlock(), BasicBlock(), BasicBlock(), BasicBlock()
   409	    normal_loop_exit = else_block if s.else_body is not None else next

   411	    builder.push_loop_stack(top, next)
   412
   413	    # Split block so that we get a handle to the top of the loop.

   416
   417	    builder.activate_block(body)
   418	    builder.accept(s.body)
   419	    # Add branch to the top at the end of the body.
   420	    builder.goto(top)
   421
   422	    builder.pop_loop_stack()

   424	    if s.else_body is not None:
   425	        builder.activate_block(else_block)
   426	        builder.accept(s.else_body)
   427	        builder.goto(next)
   428
   429	    builder.activate_block(next)

   464	def transform_try_except(
   465	    builder: IRBuilder,
   466	    body: GenFunc,
   467	    handlers: Sequence[tuple[tuple[ValueGenFunc, int] | None, Expression | None, GenFunc]],
   468	    else_body: GenFunc | None,
   469	    line: int,
   470	) -> None:
   471	    """Generalized try/except/else handling that takes functions to gen the bodies.
   472
   473	    The point of this is to also be able to support with."""
   474	    assert handlers, "try needs except"
   475
   476	    except_entry, exit_block, cleanup_block = BasicBlock(), BasicBlock(), BasicBlock()
   477	    double_except_block = BasicBlock()
   478	    # If there is an else block, jump there after the try, otherwise just leave
   479	    else_block = BasicBlock() if else_body else exit_block
   480
   481	    # Compile the try block with an error handler
   482	    builder.builder.push_error_handler(except_entry)
   483	    builder.goto_and_activate(BasicBlock())
   484	    body()
   485	    builder.goto(else_block)
   486	    builder.builder.pop_error_handler()
   487
   488	    # The error handler catches the error and then checks it
   489	    # against the except clauses. We compile the error handler
   490	    # itself with an error handler so that it can properly restore
   491	    # the *old* exc_info if an exception occurs.
   492	    # The exception chaining will be done automatically when the
   493	    # exception is raised, based on the exception in exc_info.
   494	    builder.builder.push_error_handler(double_except_block)
   495	    builder.activate_block(except_entry)
   496	    old_exc = builder.maybe_spill(builder.call_c(error_catch_op, [], line))
   497	    # Compile the except blocks with the nonlocal control flow overridden to clear exc_info
   498	    builder.nonlocal_control.append(ExceptNonlocalControl(builder.nonlocal_control[-1], old_exc))
   499
   500	    # Process the bodies
   501	    for type, var, handler_body in handlers:
   502	        next_block = None
   503	        if type:
   504	            type_f, type_line = type
   505	            next_block, body_block = BasicBlock(), BasicBlock()
   506	            matches = builder.call_c(exc_matches_op, [type_f()], type_line)
   507	            builder.add(Branch(matches, body_block, next_block, Branch.BOOL))
   508	            builder.activate_block(body_block)
   509	        if var:
   510	            target = builder.get_assignment_target(var)
   511	            builder.assign(target, builder.call_c(get_exc_value_op, [], var.line), var.line)
   512	        handler_body()
   513	        builder.goto(cleanup_block)
   514	        if next_block:
   515	            builder.activate_block(next_block)
   516
   517	    # Reraise the exception if needed
   518	    if next_block:
   519	        builder.call_c(reraise_exception_op, [], NO_TRACEBACK_LINE_NO)
   520	        builder.add(Unreachable())
   521
   522	    builder.nonlocal_control.pop()
   523	    builder.builder.pop_error_handler()
   524
   525	    # Cleanup for if we leave except through normal control flow:
   526	    # restore the saved exc_info information and continue propagating
   527	    # the exception if it exists.
   528	    builder.activate_block(cleanup_block)
   529	    builder.call_c(restore_exc_info_op, [builder.read(old_exc)], line)
   530	    builder.goto(exit_block)
   531
   532	    # Cleanup for if we leave except through a raised exception:
   533	    # restore the saved exc_info information and continue propagating
   534	    # the exception.
   535	    builder.activate_block(double_except_block)
   536	    builder.call_c(restore_exc_info_op, [builder.read(old_exc)], line)
   537	    builder.call_c(keep_propagating_op, [], NO_TRACEBACK_LINE_NO)
   538	    builder.add(Unreachable())
   539
   540	    # If present, compile the else body in the obvious way
   541	    if else_body:
   542	        builder.activate_block(else_block)
   543	        else_body()
   544	        builder.goto(exit_block)
   545
   546	    builder.activate_block(exit_block)
   547

   549	def transform_try_except_stmt(builder: IRBuilder, t: TryStmt) -> None:
   550	    def body() -> None:
   551	        builder.accept(t.body)
   552
   553	    # Work around scoping woes
   554	    def make_handler(body: Block) -> GenFunc:
   555	        return lambda: builder.accept(body)
   556
   557	    def make_entry(type: Expression) -> tuple[ValueGenFunc, int]:
   558	        return (lambda: builder.accept(type), type.line)
   559
   560	    handlers = [
   561	        (make_entry(type) if type else None, var, make_handler(body))
   562	        for type, var, body in zip(t.types, t.vars, t.handlers)
   563	    ]
   564	    else_body = (lambda: builder.accept(t.else_body)) if t.else_body else None
   565	    transform_try_except(builder, body, handlers, else_body, t.line)

   569	    builder: IRBuilder,
   570	    err_handler: BasicBlock,
   571	    return_entry: BasicBlock,
   572	    main_entry: BasicBlock,

   581	    try_body()

   589	def try_finally_entry_blocks(
   590	    builder: IRBuilder,
   591	    err_handler: BasicBlock,
   592	    return_entry: BasicBlock,
   593	    main_entry: BasicBlock,
   594	    finally_block: BasicBlock,
   595	    ret_reg: Register | AssignmentTarget | None,
   596	) -> Value:
   597	    old_exc = Register(exc_rtuple)
   598
   599	    # Entry block for non-exceptional flow
   600	    builder.activate_block(main_entry)
   601	    if ret_reg:
   602	        builder.assign(ret_reg, builder.add(LoadErrorValue(builder.ret_types[-1])), -1)
   603	    builder.goto(return_entry)
   604
   605	    builder.activate_block(return_entry)
   606	    builder.add(Assign(old_exc, builder.add(LoadErrorValue(exc_rtuple))))
   607	    builder.goto(finally_block)
   608
   609	    # Entry block for errors
   610	    builder.activate_block(err_handler)
   611	    if ret_reg:
   612	        builder.assign(ret_reg, builder.add(LoadErrorValue(builder.ret_types[-1])), -1)
   613	    builder.add(Assign(old_exc, builder.call_c(error_catch_op, [], -1)))
   614	    builder.goto(finally_block)
   615
   616	    return old_exc

   619	def try_finally_body(
   620	    builder: IRBuilder, finally_block: BasicBlock, finally_body: GenFunc, old_exc: Value
   621	) -> tuple[BasicBlock, FinallyNonlocalControl]:
   622	    cleanup_block = BasicBlock()
   623	    # Compile the finally block with the nonlocal control flow overridden to restore exc_info
   624	    builder.builder.push_error_handler(cleanup_block)
   625	    finally_control = FinallyNonlocalControl(builder.nonlocal_control[-1], old_exc)
   626	    builder.nonlocal_control.append(finally_control)
   627	    builder.activate_block(finally_block)
   628	    finally_body()
   629	    builder.nonlocal_control.pop()
   630
   631	    return cleanup_block, finally_control

   634	def try_finally_resolve_control(
   635	    builder: IRBuilder,
   636	    cleanup_block: BasicBlock,
   637	    finally_control: FinallyNonlocalControl,
   638	    old_exc: Value,
   639	    ret_reg: Register | AssignmentTarget | None,
   640	) -> BasicBlock:
   641	    """Resolve the control flow out of a finally block.
   642
   643	    This means returning if there was a return, propagating
   644	    exceptions, break/continue (soon), or just continuing on.
   645	    """
   646	    reraise, rest = BasicBlock(), BasicBlock()
   647	    builder.add(Branch(old_exc, rest, reraise, Branch.IS_ERROR))
   648
   649	    # Reraise the exception if there was one
   650	    builder.activate_block(reraise)
   651	    builder.call_c(reraise_exception_op, [], NO_TRACEBACK_LINE_NO)
   652	    builder.add(Unreachable())
   653	    builder.builder.pop_error_handler()
   654
   655	    # If there was a return, keep returning
   656	    if ret_reg:
   657	        builder.activate_block(rest)
   658	        return_block, rest = BasicBlock(), BasicBlock()
   659	        # For spill targets in try/finally, use nullable read to avoid AttributeError
   660	        if isinstance(ret_reg, AssignmentTargetAttr) and ret_reg.attr.startswith(TEMP_ATTR_NAME):
   661	            ret_val = builder.read_nullable_attr(ret_reg.obj, ret_reg.attr, -1)
   662	        else:
   663	            ret_val = builder.read(ret_reg)
   664	        builder.add(Branch(ret_val, rest, return_block, Branch.IS_ERROR))
   665
   666	        builder.activate_block(return_block)
   667	        builder.nonlocal_control[-1].gen_return(builder, ret_val, -1)
   668
   669	    # TODO: handle break/continue
   670	    builder.activate_block(rest)
   671	    out_block = BasicBlock()
   672	    builder.goto(out_block)
   673
   674	    # If there was an exception, restore again
   675	    builder.activate_block(cleanup_block)
   676	    finally_control.gen_cleanup(builder, -1)
   677	    builder.call_c(keep_propagating_op, [], NO_TRACEBACK_LINE_NO)
   678	    builder.add(Unreachable())
   679
   680	    return out_block

   683	def transform_try_finally_stmt(
   684	    builder: IRBuilder, try_body: GenFunc, finally_body: GenFunc, line: int = -1
   685	) -> None:
   686	    """Generalized try/finally handling that takes functions to gen the bodies.
   687
   688	    The point of this is to also be able to support with."""
   689	    # Finally is a big pain, because there are so many ways that
   690	    # exits can occur. We emit 10+ basic blocks for every finally!
   691
   692	    err_handler, main_entry, return_entry, finally_block = (
   693	        BasicBlock(),
   694	        BasicBlock(),
   695	        BasicBlock(),
   696	        BasicBlock(),
   697	    )
   698
   699	    # Compile the body of the try
   700	    ret_reg = try_finally_try(builder, err_handler, return_entry, main_entry, try_body)
   701
   702	    # Set up the entry blocks for the finally statement
   703	    old_exc = try_finally_entry_blocks(
   704	        builder, err_handler, return_entry, main_entry, finally_block, ret_reg
   705	    )
   706
   707	    # Compile the body of the finally
   708	    cleanup_block, finally_control = try_finally_body(
   709	        builder, finally_block, finally_body, old_exc
   710	    )
   711
   712	    # Resolve the control flow out of the finally block
   713	    out_block = try_finally_resolve_control(
   714	        builder, cleanup_block, finally_control, old_exc, ret_reg
   715	    )
   716
   717	    builder.activate_block(out_block)

   720	def transform_try_finally_stmt_async(
   721	    builder: IRBuilder, try_body: GenFunc, finally_body: GenFunc, line: int = -1
   722	) -> None:
   723	    """Async-aware try/finally handling for when finally contains await.
   724
   725	    This version uses a modified approach that preserves exceptions across await."""
   726
   727	    # We need to handle returns properly, so we'll use TryFinallyNonlocalControl
   728	    # to track return values, similar to the regular try/finally implementation
   729
   730	    err_handler, main_entry, return_entry, finally_entry = (
   731	        BasicBlock(),
   732	        BasicBlock(),
   733	        BasicBlock(),
   734	        BasicBlock(),
   735	    )
   736
   737	    # Track if we're returning from the try block
   738	    control = TryFinallyNonlocalControl(return_entry)
   739	    builder.builder.push_error_handler(err_handler)
   740	    builder.nonlocal_control.append(control)
   741	    builder.goto_and_activate(BasicBlock())
   742	    try_body()
   743	    builder.goto(main_entry)
   744	    builder.nonlocal_control.pop()
   745	    builder.builder.pop_error_handler()
   746	    ret_reg = control.ret_reg
   747
   748	    # Normal case - no exception or return
   749	    builder.activate_block(main_entry)
   750	    builder.goto(finally_entry)
   751
   752	    # Return case
   753	    builder.activate_block(return_entry)
   754	    builder.goto(finally_entry)
   755
   756	    # Exception case - need to catch to clear the error indicator
   757	    builder.activate_block(err_handler)
   758	    # Catch the error to clear Python's error indicator
   759	    builder.call_c(error_catch_op, [], line)
   760	    # We're not going to use old_exc since it won't survive await
   761	    # The exception is now in sys.exc_info()
   762	    builder.goto(finally_entry)
   763
   764	    # Finally block
   765	    builder.activate_block(finally_entry)
   766
   767	    # Execute finally body
   768	    finally_body()
   769
   770	    # After finally, we need to handle exceptions carefully:
   771	    # 1. If finally raised a new exception, it's in the error indicator - let it propagate
   772	    # 2. If finally didn't raise, check if we need to reraise the original from sys.exc_info()
   773	    # 3. If there was a return, return that value
   774	    # 4. Otherwise, normal exit
   775
   776	    # First, check if there's a current exception in the error indicator
   777	    # (this would be from the finally block)
   778	    no_current_exc = builder.call_c(no_err_occurred_op, [], line)
   779	    finally_raised = BasicBlock()
   780	    check_original = BasicBlock()
   781	    builder.add(Branch(no_current_exc, check_original, finally_raised, Branch.BOOL))
   782
   783	    # Finally raised an exception - let it propagate naturally
   784	    builder.activate_block(finally_raised)
   785	    builder.call_c(keep_propagating_op, [], NO_TRACEBACK_LINE_NO)
   786	    builder.add(Unreachable())
   787
   788	    # No exception from finally, check if we need to handle return or original exception
   789	    builder.activate_block(check_original)
   790
   791	    # Check if we have a return value
   792	    if ret_reg:
   793	        return_block, check_old_exc = BasicBlock(), BasicBlock()
   794	        builder.add(Branch(builder.read(ret_reg), check_old_exc, return_block, Branch.IS_ERROR))
   795
   796	        builder.activate_block(return_block)
   797	        builder.nonlocal_control[-1].gen_return(builder, builder.read(ret_reg), -1)
   798
   799	        builder.activate_block(check_old_exc)
   800
   801	    # Check if we need to reraise the original exception from sys.exc_info
   802	    exc_info = builder.call_c(get_exc_info_op, [], line)
   803	    exc_type = builder.add(TupleGet(exc_info, 0, line))
   804
   805	    # Check if exc_type is None
   806	    none_obj = builder.none_object()
   807	    has_exc = builder.binary_op(exc_type, none_obj, "is not", line)
   808
   809	    reraise_block, exit_block = BasicBlock(), BasicBlock()
   810	    builder.add(Branch(has_exc, reraise_block, exit_block, Branch.BOOL))
   811
   812	    # Reraise the original exception
   813	    builder.activate_block(reraise_block)
   814	    builder.call_c(reraise_exception_op, [], NO_TRACEBACK_LINE_NO)
   815	    builder.add(Unreachable())
   816
   817	    # Normal exit
   818	    builder.activate_block(exit_block)

   832	def transform_try_stmt(builder: IRBuilder, t: TryStmt) -> None:
   833	    # Our compilation strategy for try/except/else/finally is to
   834	    # treat try/except/else and try/finally as separate language
   835	    # constructs that we compile separately. When we have a
   836	    # try/except/else/finally, we treat the try/except/else as the
   837	    # body of a try/finally block.
   838	    if t.is_star:
   839	        builder.error("Exception groups and except* cannot be compiled yet", t.line)
   840
   841	    # Check if we're in an async function with a finally block that contains await
   842	    use_async_version = False
   843	    if t.finally_body and builder.fn_info.is_coroutine:
   844	        detector = AwaitDetector()
   845	        t.finally_body.accept(detector)
   846
   847	        if detector.has_await:
   848	            # Use the async version that handles exceptions correctly
   849	            use_async_version = True
   850
   851	    if t.finally_body:
   852
   853	        def transform_try_body() -> None:
   854	            if t.handlers:
   855	                transform_try_except_stmt(builder, t)
   856	            else:
   857	                builder.accept(t.body)
   858
   859	        body = t.finally_body
   860
   861	        if use_async_version:
   862	            transform_try_finally_stmt_async(
   863	                builder, transform_try_body, lambda: builder.accept(body), t.line
   864	            )
   865	        else:
   866	            transform_try_finally_stmt(
   867	                builder, transform_try_body, lambda: builder.accept(body), t.line
   868	            )
   869	    else:
   870	        transform_try_except_stmt(builder, t)
