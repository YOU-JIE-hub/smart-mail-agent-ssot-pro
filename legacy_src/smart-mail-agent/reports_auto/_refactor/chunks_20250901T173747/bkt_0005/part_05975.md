   773	def generate_arg_check(

   810	            emitter.emit_unbox(
   811	                f"obj_{name}",
   812	                f"arg_{name}",
   813	                typ,
   814	                declare_dest=True,
   815	                raise_exception=raise_exception,
   816	                error=error,
   817	                borrow=True,
   818	                optional=optional,
   819	            )
   820	    elif is_object_rprimitive(typ):

   217	def format_int(n: int) -> bytes:
   218	    """Format an integer using a variable-length binary encoding."""
   219	    if n < 128:
   220	        a = [n]
   221	    else:
   222	        a = []
   223	        while n > 0:
   224	            a.insert(0, n & 0x7F)
   225	            n >>= 7
   226	        for i in range(len(a) - 1):
   227	            # If the highest bit is set, more 7-bit digits follow

    13	# Some notes on the vtable layout: Each concrete class has a vtable
    14	# that contains function pointers for its methods. So that subclasses
    15	# may be efficiently used when their parent class is expected, the
    16	# layout of child vtables must be an extension of their base class's
    17	# vtable.
    18	#
    19	# This makes multiple inheritance tricky, since obviously we cannot be
    20	# an extension of multiple parent classes. We solve this by requiring
    21	# all but one parent to be "traits", which we can operate on in a
    22	# somewhat less efficient way. For each trait implemented by a class,
    23	# we generate a separate vtable for the methods in that trait.
    24	# We then store an array of (trait type, trait vtable) pointers alongside
    25	# a class's main vtable. When we want to call a trait method, we
    26	# (at runtime!) search the array of trait vtables to find the correct one,
    27	# then call through it.
    28	# Trait vtables additionally need entries for attribute getters and setters,
    29	# since they can't always be in the same location.
    30	#
    31	# To keep down the number of indirections necessary, we store the
    32	# array of trait vtables in the memory *before* the class vtable, and
    33	# search it backwards.  (This is a trick we can only do once---there
    34	# are only two directions to store data in---but I don't think we'll
    35	# need it again.)
    36	# There are some tricks we could try in the future to store the trait
    37	# vtables inline in the trait table (which would cut down one indirection),
    38	# but this seems good enough for now.
    39	#
    40	# As an example:
    41	# Imagine that we have a class B that inherits from a concrete class A
    42	# and traits T1 and T2, and that A has methods foo() and
    43	# bar() and B overrides bar() with a more specific type.
    44	# Then B's vtable will look something like:
    45	#
    46	#      T1 type object
    47	#      ptr to B's T1 trait vtable
    48	#      T2 type object
    49	#      ptr to B's T2 trait vtable
    50	# -> | A.foo
    51	#    | Glue function that converts between A.bar's type and B.bar
    52	#      B.bar
    53	#      B.baz
    54	#
    55	# The arrow points to the "start" of the vtable (what vtable pointers
    56	# point to) and the bars indicate which parts correspond to the parent
    57	# class A's vtable layout.
    58	#
    59	# Classes that allow interpreted code to subclass them also have a
    60	# "shadow vtable" that contains implementations that delegate to
    61	# making a pycall, so that overridden methods in interpreted children
    62	# will be called. (A better strategy could dynamically generate these
    63	# vtables based on which methods are overridden in the children.)

    50	class BasicBlock:
    51	    """IR basic block.
    52
    53	    Contains a sequence of Ops and ends with a ControlOp (Goto,
    54	    Branch, Return or Unreachable). Only the last op can be a
    55	    ControlOp.
    56
    57	    All generated Ops live in basic blocks. Basic blocks determine the
    58	    order of evaluation and control flow within a function. A basic
    59	    block is always associated with a single function/method (FuncIR).
    60
    61	    When building the IR, ops that raise exceptions can be included in
    62	    the middle of a basic block, but the exceptions aren't checked.
    63	    Afterwards we perform a transform that inserts explicit checks for
    64	    all error conditions and splits basic blocks accordingly to preserve
    65	    the invariant that a jump, branch or return can only ever appear
    66	    as the final op in a block. Manually inserting error checking ops
    67	    would be boring and error-prone.
    68
    69	    BasicBlocks have an error_handler attribute that determines where
    70	    to jump if an error occurs. If none is specified, an error will
    71	    propagate up out of the function. This is compiled away by the
    72	    `exceptions` module.
    73
    74	    Block labels are used for pretty printing and emitting C code, and
    75	    get filled in by those passes.

    77	    Ops that may terminate the program aren't treated as exits.

    91	        this is false during construction.

   458	class Unreachable(ControlOp):
   459	    """Mark the end of basic block as unreachable.
   460
   461	    This is sometimes necessary when the end of a basic block is never
   462	    reached. This can also be explicitly added to the end of non-None
   463	    returning functions (in None-returning function we can just return
   464	    None).
   465
   466	    Mypy statically guarantees that the end of the function is not
   467	    unreachable if there is not a return statement.
   468
   469	    This prevents the block formatter from being confused due to lack
   470	    of a leave and also leaves a nifty note in the IR. It is not
   471	    generally processed by visitors.
   472	    """
   473
   474	    error_kind = ERR_NEVER
   475
   476	    def __init__(self, line: int = -1) -> None:
   477	        super().__init__(line)
   478
   479	    def sources(self) -> list[Value]:
   480	        return []
   481
   482	    def set_sources(self, new: list[Value]) -> None:
   483	        assert not new
   484
   485	    def accept(self, visitor: OpVisitor[T]) -> T:
   486	        return visitor.visit_unreachable(self)

   489	class RegisterOp(Op):
   490	    """Abstract base class for operations that can be written as r1 = f(r2, ..., rn).
   491
   492	    Takes some values, performs an operation, and generates an output
   493	    (unless the 'type' attribute is void_rtype, which is the default).
   494	    Other ops can refer to the result of the Op by referring to the Op
   495	    instance. This doesn't do any explicit control flow, but can raise an
   496	    error.
   497
   498	    Note that the operands can be arbitrary Values, not just Register
   499	    instances, even though the naming may suggest otherwise.
   500	    """
   501
   502	    error_kind = -1  # Can this raise exception and how is it signalled; one of ERR_*
   503
   504	    _type: RType | None = None
   505
   506	    def __init__(self, line: int) -> None:
   507	        super().__init__(line)
   508	        assert self.error_kind != -1, "error_kind not defined"
   509
   510	    def can_raise(self) -> bool:
   511	        return self.error_kind != ERR_NEVER

   562	class Call(RegisterOp):
   563	    """Native call f(arg, ...).
   564
   565	    The call target can be a module-level function or a class.
   566	    """
   567
   568	    def __init__(self, fn: FuncDecl, args: Sequence[Value], line: int) -> None:
   569	        self.fn = fn
   570	        self.args = list(args)
   571	        assert len(self.args) == len(fn.sig.args)
   572	        self.type = fn.sig.ret_type
   573	        ret_type = fn.sig.ret_type
   574	        if not ret_type.error_overlap:
   575	            self.error_kind = ERR_MAGIC
   576	        else:
   577	            self.error_kind = ERR_MAGIC_OVERLAPPING
   578	        super().__init__(line)
   579
   580	    def sources(self) -> list[Value]:
   581	        return list(self.args.copy())
   582
   583	    def set_sources(self, new: list[Value]) -> None:
   584	        self.args = new[:]
   585
   586	    def accept(self, visitor: OpVisitor[T]) -> T:
   587	        return visitor.visit_call(self)
