    34	def assert_blobs_same(x: Any, y: Any, trail: tuple[Any, ...]) -> None:
    35	    """Compare two blobs of IR as best we can.
    36
    37	    FuncDecls, FuncIRs, and ClassIRs are compared by fullname to avoid
    38	    infinite recursion.
    39	    (More detailed comparisons should be done manually.)
    40
    41	    Types and signatures are compared using mypyc.sametype.
    42
    43	    Containers are compared recursively.
    44
    45	    Anything else is compared with ==.
    46
    47	    The `trail` argument is used in error messages.
    48	    """
    49
    50	    assert type(x) is type(y), (f"Type mismatch at {trail}", type(x), type(y))
    51	    if isinstance(x, (FuncDecl, FuncIR, ClassIR)):
    52	        assert x.fullname == y.fullname, f"Name mismatch at {trail}"
    53	    elif isinstance(x, dict):
    54	        assert len(x.keys()) == len(y.keys()), f"Keys mismatch at {trail}"
    55	        for (xk, xv), (yk, yv) in zip(x.items(), y.items()):
    56	            assert_blobs_same(xk, yk, trail + ("keys",))
    57	            assert_blobs_same(xv, yv, trail + (xk,))
    58	    elif isinstance(x, dict):
    59	        assert x.keys() == y.keys(), f"Keys mismatch at {trail}"
    60	        for k in x.keys():
    61	            assert_blobs_same(x[k], y[k], trail + (k,))
    62	    elif isinstance(x, Iterable) and not isinstance(x, (str, set)):
    63	        # Special case iterables to generate better assert error messages.
    64	        # We can't use this for sets since the ordering is unpredictable,
    65	        # and strings should be treated as atomic values.
    66	        for i, (xv, yv) in enumerate(zip(x, y)):
    67	            assert_blobs_same(xv, yv, trail + (i,))
    68	    elif isinstance(x, RType):
    69	        assert is_same_type(x, y), f"RType mismatch at {trail}"
    70	    elif isinstance(x, FuncSignature):
    71	        assert is_same_signature(x, y), f"Signature mismatch at {trail}"
    72	    else:
    73	        assert x == y, f"Value mismatch at {trail}"

     4
     5	    x = f()
     6	    y = x
     7

    13	"""

    16

    25	def do_copy_propagation(fn: FuncIR, options: CompilerOptions) -> None:
    26	    """Perform copy propagation optimization for fn."""
    27
    28	    # Anything with an assignment count >1 will not be optimized
    29	    # here, as it would be require data flow analysis and we want to
    30	    # keep this simple and fast, at least until we've made data flow
    31	    # analysis much faster.
    32	    counts: dict[Value, int] = {}
    33	    replacements: dict[Value, Value] = {}
    34	    for arg in fn.arg_regs:
    35	        # Arguments are always assigned to initially
    36	        counts[arg] = 1
    37
    38	    for block in fn.blocks:
    39	        for op in block.ops:
    40	            if isinstance(op, Assign):
    41	                c = counts.get(op.dest, 0)
    42	                counts[op.dest] = c + 1
    43	                # Does this look like a supported assignment?
    44	                # TODO: Something needs LoadErrorValue assignments to be preserved?
    45	                if (
    46	                    c == 0
    47	                    and is_same_type(op.dest.type, op.src.type)
    48	                    and not isinstance(op.src, LoadErrorValue)
    49	                ):
    50	                    replacements[op.dest] = op.src
    51	                elif c == 1:
    52	                    # Too many assignments -- don't replace this one
    53	                    replacements.pop(op.dest, 0)
    54	            elif isinstance(op, AssignMulti):
    55	                # Copy propagation not supported for AssignMulti destinations
    56	                counts[op.dest] = 2
    57	                replacements.pop(op.dest, 0)
    58	            elif isinstance(op, LoadAddress):
    59	                # We don't support taking the address of an arbitrary Value,
    60	                # so we'll need to preserve the operands of LoadAddress.
    61	                if isinstance(op.src, Register):
    62	                    counts[op.src] = 2
    63	                    replacements.pop(op.src, 0)
    64
    65	    # Follow chains of propagation with more than one assignment.
    66	    for src, dst in list(replacements.items()):
    67	        if counts.get(dst, 0) > 1:
    68	            # Not supported
    69	            del replacements[src]
    70	        else:
    71	            while dst in replacements:
    72	                dst = replacements[dst]
    73	                if counts.get(dst, 0) > 1:
    74	                    # Not supported
    75	                    del replacements[src]
    76	        if src in replacements:
    77	            replacements[src] = dst
    78
    79	    builder = LowLevelIRBuilder(None, options)
    80	    transform = CopyPropagationTransform(builder, replacements)
    81	    transform.transform_blocks(fn.blocks)
    82	    fn.blocks = builder.blocks

   111	    post_must_defined: AnalysisDict[Value],
   112	) -> None:
   113	    old_ops = block.ops
   114	    ops: list[Op] = []
   115	    for i, op in enumerate(old_ops):
   116	        key = (block, i)
   117
   118	        assert op not in pre_live[key]
   119	        dest = op.dest if isinstance(op, Assign) else op
   120	        stolen = op.stolen()
   121
   122	        # Incref any references that are being stolen that stay live, were borrowed,
   123	        # or are stolen more than once by this operation.
   124	        for j, src in enumerate(stolen):
   125	            if src in post_live[key] or src in pre_borrow[key] or src in stolen[:j]:
   126	                maybe_append_inc_ref(ops, src)
   127	                # For assignments to registers that were already live,
   128	                # decref the old value.
   129	                if dest not in pre_borrow[key] and dest in pre_live[key]:
   130	                    assert isinstance(op, Assign)
   131	                    maybe_append_dec_ref(ops, dest, post_must_defined, key)
   132
   133	        # Strip KeepAlive. Its only purpose is to help with this transform.
   134	        if not isinstance(op, KeepAlive):
   135	            ops.append(op)
   136
   137	        # Control ops don't have any space to insert ops after them, so
   138	        # their inc/decrefs get inserted by insert_branch_inc_and_decrefs.
   139	        if isinstance(op, ControlOp):
   140	            continue
   141
   142	        for src in op.unique_sources():
   143	            # Decrement source that won't be live afterwards.
   144	            if src not in post_live[key] and src not in pre_borrow[key] and src not in stolen:
   145	                maybe_append_dec_ref(ops, src, post_must_defined, key)
   146	        # Decrement the destination if it is dead after the op and
   147	        # wasn't a borrowed RegisterOp
   148	        if (
   149	            not dest.is_void
   150	            and dest not in post_live[key]
   151	            and not (isinstance(op, RegisterOp) and dest.is_borrowed)
   152	        ):
   153	            maybe_append_dec_ref(ops, dest, post_must_defined, key)
   154	    block.ops = ops

   157	def insert_branch_inc_and_decrefs(
   158	    block: BasicBlock,
   159	    cache: BlockCache,
   160	    blocks: list[BasicBlock],
   161	    pre_live: AnalysisDict[Value],
   162	    pre_borrow: AnalysisDict[Value],
   163	    post_borrow: AnalysisDict[Value],
   164	    post_must_defined: AnalysisDict[Value],
   165	    ordering: dict[Value, int],
   166	) -> None:
   167	    """Insert inc_refs and/or dec_refs after a branch/goto.
   168
   169	    Add dec_refs for registers that become dead after a branch.
   170	    Add inc_refs for registers that become unborrowed after a branch or goto.
   171
   172	    Branches are special as the true and false targets may have a different
   173	    live and borrowed register sets. Add new blocks before the true/false target
   174	    blocks that tweak reference counts.
   175
   176	    Example where we need to add an inc_ref:
   177
   178	      def f(a: int) -> None
   179	          if a:
   180	              a = 1
   181	          return a  # a is borrowed if condition is false and unborrowed if true
   182	    """
   183	    prev_key = (block, len(block.ops) - 1)
   184	    source_live_regs = pre_live[prev_key]
   185	    source_borrowed = post_borrow[prev_key]
   186	    source_defined = post_must_defined[prev_key]
   187
   188	    term = block.terminator
   189	    for i, target in enumerate(term.targets()):
   190	        # HAX: After we've checked against an error value the value we must not touch the
   191	        #      refcount since it will be a null pointer. The correct way to do this would be
   192	        #      to perform data flow analysis on whether a value can be null (or is always
   193	        #      null).
   194	        omitted: Iterable[Value]
   195	        if isinstance(term, Branch) and term.op == Branch.IS_ERROR and i == 0:
   196	            omitted = (term.value,)
   197	        else:
   198	            omitted = ()
   199
   200	        decs = after_branch_decrefs(
   201	            target, pre_live, source_defined, source_borrowed, source_live_regs, ordering, omitted
   202	        )
   203	        incs = after_branch_increfs(target, pre_live, pre_borrow, source_borrowed, ordering)
   204	        term.set_target(i, add_block(decs, incs, cache, blocks, target))
   205
   206
   207	def after_branch_decrefs(
   208	    label: BasicBlock,
   209	    pre_live: AnalysisDict[Value],
   210	    source_defined: set[Value],
   211	    source_borrowed: set[Value],
   212	    source_live_regs: set[Value],
   213	    ordering: dict[Value, int],
   214	    omitted: Iterable[Value],
   215	) -> tuple[tuple[Value, bool], ...]:
   216	    target_pre_live = pre_live[label, 0]
   217	    decref = source_live_regs - target_pre_live - source_borrowed
   218	    if decref:
   219	        return tuple(
   220	            (reg, is_maybe_undefined(source_defined, reg))
   221	            for reg in sorted(decref, key=lambda r: ordering[r])
   222	            if reg.type.is_refcounted and reg not in omitted
   223	        )
   224	    return ()
   225
   226
   227	def after_branch_increfs(
   228	    label: BasicBlock,
   229	    pre_live: AnalysisDict[Value],
   230	    pre_borrow: AnalysisDict[Value],
   231	    source_borrowed: set[Value],
   232	    ordering: dict[Value, int],
   233	) -> tuple[Value, ...]:
   234	    target_pre_live = pre_live[label, 0]
   235	    target_borrowed = pre_borrow[label, 0]
   236	    incref = (source_borrowed - target_borrowed) & target_pre_live
   237	    if incref:
   238	        return tuple(
   239	            reg for reg in sorted(incref, key=lambda r: ordering[r]) if reg.type.is_refcounted
   240	        )
   241	    return ()
   242
   243
   244	def add_block(
   245	    decs: Decs, incs: Incs, cache: BlockCache, blocks: list[BasicBlock], label: BasicBlock
   246	) -> BasicBlock:
   247	    if not decs and not incs:
   248	        return label
   249
   250	    # TODO: be able to share *partial* results
   251	    if (label, decs, incs) in cache:
   252	        return cache[label, decs, incs]
   253
   254	    block = BasicBlock()
   255	    blocks.append(block)
   256	    block.ops.extend(DecRef(reg, is_xdec=xdec) for reg, xdec in decs)
   257	    block.ops.extend(IncRef(reg) for reg in incs)
   258	    block.ops.append(Goto(label))
   259	    cache[label, decs, incs] = block
   260	    return block

   106	    @classmethod
   107	    def _load(cls, configfiles, verbose=False):
   108	        """
   109	        Load configuration from the given files in reverse order,
   110	        if they exist and have a [nodeenv] section.
   111	        Additionally, load version from .node-version if file exists.
   112	        """
   113	        for configfile in reversed(configfiles):
   114	            configfile = os.path.expanduser(configfile)
   115	            if not os.path.exists(configfile):
   116	                continue
   117
   118	            ini_file = ConfigParser()
   119	            ini_file.read(configfile)
   120	            section = "nodeenv"
   121	            if not ini_file.has_section(section):
   122	                continue
   123
   124	            for attr, val in iteritems(vars(cls)):
   125	                if attr.startswith('_') or not \
   126	                   ini_file.has_option(section, attr):
   127	                    continue
   128
   129	                if isinstance(val, bool):
   130	                    val = ini_file.getboolean(section, attr)
   131	                else:
   132	                    val = ini_file.get(section, attr)
   133
   134	                if verbose:
   135	                    print('CONFIG {0}: {1} = {2}'.format(
   136	                        os.path.basename(configfile), attr, val))
   137	                setattr(cls, attr, val)
   138
   139	        if os.path.exists(".node-version"):
   140	            with open(".node-version", "r") as v_file:
   141	                setattr(cls, "node", v_file.readlines(1)[0].strip())

   174	def parse_version(version_str):
   175	    """
   176	    Parse version string to a tuple of integer parts
   177	    """
   178	    v = version_str.replace('v', '').split('.')[:3]
   179	    # remove all after '+' in the PATCH part of the version
   180	    if len(v) >= 3:
   181	        v[2] = v[2].split('+')[0]
   182	    return tuple(map(int, v))

   197	def create_logger():
   198	    """
   199	    Create logger for diagnostic
   200	    """
   201	    # create logger
   202	    loggr = logging.getLogger("nodeenv")
   203	    loggr.setLevel(logging.INFO)
   204
   205	    # monkey patch
   206	    def emit(self, record):
   207	        msg = self.format(record)
   208	        fs = "%s" if getattr(record, "continued", False) else "%s\n"
   209	        self.stream.write(fs % to_utf8(msg))
   210	        self.flush()
   211	    logging.StreamHandler.emit = emit
   212
   213	    # create console handler and set level to debug
   214	    ch = logging.StreamHandler()
   215	    ch.setLevel(logging.DEBUG)
   216
   217	    # create formatter
   218	    formatter = logging.Formatter(fmt="%(message)s")
   219
   220	    # add formatter to ch
   221	    ch.setFormatter(formatter)
   222
   223	    # add ch to logger
   224	    loggr.addHandler(ch)
   225	    return loggr
