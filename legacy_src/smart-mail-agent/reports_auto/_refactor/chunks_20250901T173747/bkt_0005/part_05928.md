  3025	    def union_overload_result(
  3026	        self,
  3027	        plausible_targets: list[CallableType],
  3028	        args: list[Expression],
  3029	        arg_types: list[Type],
  3030	        arg_kinds: list[ArgKind],
  3031	        arg_names: Sequence[str | None] | None,
  3032	        callable_name: str | None,
  3033	        object_type: Type | None,
  3034	        none_type_var_overlap: bool,
  3035	        context: Context,
  3036	        level: int = 0,
  3037	    ) -> list[tuple[Type, Type]] | None:
  3038	        """Accepts a list of overload signatures and attempts to match calls by destructuring
  3039	        the first union.
  3040
  3041	        Return a list of (<return type>, <inferred variant type>) if call succeeds for every
  3042	        item of the desctructured union. Returns None if there is no match.

  3044	        # Step 1: If we are already too deep, then stop immediately. Otherwise mypy might
  3045	        # hang for long time because of a weird overload call. The caller will get
  3046	        # the exception and generate an appropriate note message, if needed.
  3047	        if level >= MAX_UNIONS:
  3048	            raise TooManyUnions
  3049
  3050	        # Step 2: Find position of the first union in arguments. Return the normal inferred
  3051	        # type if no more unions left.
  3052	        for idx, typ in enumerate(arg_types):
  3053	            if self.real_union(typ):
  3054	                break
  3055	        else:
  3056	            # No unions in args, just fall back to normal inference
  3057	            with self.type_overrides_set(args, arg_types):
  3058	                res = self.infer_overload_return_type(
  3059	                    plausible_targets,
  3060	                    args,
  3061	                    arg_types,
  3062	                    arg_kinds,
  3063	                    arg_names,
  3064	                    callable_name,
  3065	                    object_type,
  3066	                    context,
  3067	                )
  3068	            if res is not None:
  3069	                return [res]
  3070	            return None
  3071
  3072	        # Step 3: Try a direct match before splitting to avoid unnecessary union splits
  3073	        # and save performance.
  3074	        if not none_type_var_overlap:
  3075	            with self.type_overrides_set(args, arg_types):
  3076	                direct = self.infer_overload_return_type(
  3077	                    plausible_targets,
  3078	                    args,
  3079	                    arg_types,
  3080	                    arg_kinds,
  3081	                    arg_names,
  3082	                    callable_name,
  3083	                    object_type,
  3084	                    context,
  3085	                )
  3086	            if direct is not None and not isinstance(
  3087	                get_proper_type(direct[0]), (UnionType, AnyType)
  3088	            ):
  3089	                # We only return non-unions soon, to avoid greedy match.
  3090	                return [direct]
  3091
  3092	        # Step 4: Split the first remaining union type in arguments into items and
  3093	        # try to match each item individually (recursive).
  3094	        first_union = get_proper_type(arg_types[idx])
  3095	        assert isinstance(first_union, UnionType)
  3096	        res_items = []
  3097	        for item in first_union.relevant_items():
  3098	            new_arg_types = arg_types.copy()
  3099	            new_arg_types[idx] = item
  3100	            sub_result = self.union_overload_result(
  3101	                plausible_targets,
  3102	                args,
  3103	                new_arg_types,
  3104	                arg_kinds,
  3105	                arg_names,
  3106	                callable_name,
  3107	                object_type,
  3108	                none_type_var_overlap,
  3109	                context,
  3110	                level + 1,
  3111	            )
  3112	            if sub_result is not None:
  3113	                res_items.extend(sub_result)
  3114	            else:
  3115	                # Some item doesn't match, return soon.
  3116	                return None
  3117
  3118	        # Step 5: If splitting succeeded, then filter out duplicate items before returning.
  3119	        seen: set[tuple[Type, Type]] = set()
  3120	        result = []
  3121	        for pair in res_items:
  3122	            if pair not in seen:
  3123	                seen.add(pair)
  3124	                result.append(pair)
  3125	        return result

  3552	                    context=e,
  3553	                    allow_reverse=False,

  3564
  3565	        Comparison expressions are type checked consecutive-pair-wise
  3566	        That is, 'a < b > c == d' is check as 'a < b and b > c and c == d'
  3567	        """

  3569	        sub_result: Type
  3570
  3571	        # Check each consecutive operand pair and their operator
  3572	        for left, right, operator in zip(e.operands, e.operands[1:], e.operators):
  3573	            left_type = self.accept(left)
  3574
  3575	            if operator == "in" or operator == "not in":
  3576	                # This case covers both iterables and containers, which have different meanings.
  3577	                # For a container, the in operator calls the __contains__ method.
  3578	                # For an iterable, the in operator iterates over the iterable, and compares each item one-by-one.
  3579	                # We allow `in` for a union of containers and iterables as long as at least one of them matches the
  3580	                # type of the left operand, as the operation will simply return False if the union's container/iterator
  3581	                # type doesn't match the left operand.

  3584	                # a "Need type annotation ..." message, as it would be noise.

  3586	                if right_type is None:
  3587	                    right_type = self.accept(right)  # Validate the right operand
  3588
  3589	                right_type = get_proper_type(right_type)

  3591	                if isinstance(right_type, UnionType):
  3592	                    item_types = list(right_type.relevant_items())

  3594	                sub_result = self.bool_type()

  3633	                            )

  3636	                            failed_out = True
  3637	                        else:
  3638	                            method_type = CallableType(
  3639	                                [left_type],
  3640	                                [nodes.ARG_POS],
  3641	                                [None],
  3642	                                self.bool_type(),
  3643	                                self.named_type("builtins.function"),

  3646	                            iterable_types.append(itertype)
  3647	                    elif not container_errors.has_new_errors() and cont_type:
  3648	                        container_types.append(cont_type)
  3649	                        e.method_types.append(method_type)
  3650	                    else:

  3652	                        failed_out = True
  3653

  3655	                    iterable_type = UnionType.make_union(iterable_types)

  3658	                            self.msg.unsupported_operand_types("in", left_type, right_type, e)
  3659	                        else:
  3660	                            container_type = UnionType.make_union(container_types)
  3661	                            if self.dangerous_comparison(

  3663	                                container_type,
  3664	                                original_container=right_type,

  3666	                            ):
  3667	                                self.msg.dangerous_comparison(
  3668	                                    left_type, container_type, "container", e
  3669	                                )

  3671	            elif operator in operators.op_methods:

  3674	                with ErrorWatcher(self.msg.errors) as w:
  3675	                    sub_result, method_type = self.check_op(
  3676	                        method, left_type, right, e, allow_reverse=True

  3678	                    e.method_types.append(method_type)

  3687	                        right_type = try_getting_literal(right_type)

  3689
  3690	            elif operator == "is" or operator == "is not":
  3691	                right_type = self.accept(right)  # validate the right operand
  3692	                sub_result = self.bool_type()
  3693	                if self.dangerous_comparison(left_type, right_type):
  3694	                    # Show the most specific literal types possible
  3695	                    left_type = try_getting_literal(left_type)
  3696	                    right_type = try_getting_literal(right_type)
  3697	                    self.msg.dangerous_comparison(left_type, right_type, "identity", e)
  3698	                e.method_types.append(None)
  3699	            else:
  3700	                raise RuntimeError(f"Unknown comparison operator {operator}")
  3701
  3702	            #  Determine type of boolean-and of result and sub_result
  3703	            if result is None:
  3704	                result = sub_result
  3705	            else:
  3706	                result = join.join_types(result, sub_result)

  3708	        assert result is not None
  3709	        return result

  3727	    def dangerous_comparison(
  3728	        self,
  3729	        left: Type,
  3730	        right: Type,
  3731	        *,
  3732	        original_container: Type | None = None,
  3733	        seen_types: set[tuple[Type, Type]] | None = None,
  3734	        prefer_literal: bool = True,
  3735	    ) -> bool:
  3736	        """Check for dangerous non-overlapping comparisons like 42 == 'no'.
  3737
  3738	        The original_container is the original container type for 'in' checks
  3739	        (and None for equality checks).
  3740
  3741	        Rules:
  3742	            * X and None are overlapping even in strict-optional mode. This is to allow
  3743	            'assert x is not None' for x defined as 'x = None  # type: str' in class body
  3744	            (otherwise mypy itself would have couple dozen errors because of this).
  3745	            * Optional[X] and Optional[Y] are non-overlapping if X and Y are
  3746	            non-overlapping, although technically None is overlap, it is most
  3747	            likely an error.
  3748	            * Any overlaps with everything, i.e. always safe.
  3749	            * Special case: b'abc' in b'cde' is safe.
  3750	        """
  3751	        if not self.chk.options.strict_equality:
  3752	            return False
  3753
  3754	        if seen_types is None:
  3755	            seen_types = set()
  3756	        if (left, right) in seen_types:
  3757	            return False
  3758	        seen_types.add((left, right))
  3759
  3760	        left, right = get_proper_types((left, right))
  3761
  3762	        # We suppress the error if there is a custom __eq__() method on either
  3763	        # side. User defined (or even standard library) classes can define this
  3764	        # to return True for comparisons between non-overlapping types.
  3765	        if custom_special_method(left, "__eq__") or custom_special_method(right, "__eq__"):
  3766	            return False
  3767
  3768	        if prefer_literal:
  3769	            # Also flag non-overlapping literals in situations like:
  3770	            #    x: Literal['a', 'b']
  3771	            #    if x == 'c':
  3772	            #        ...
  3773	            left = try_getting_literal(left)
  3774	            right = try_getting_literal(right)
  3775
  3776	        if self.chk.binder.is_unreachable_warning_suppressed():
  3777	            # We are inside a function that contains type variables with value restrictions in
  3778	            # its signature. In this case we just suppress all strict-equality checks to avoid
  3779	            # false positives for code like:
  3780	            #
  3781	            #     T = TypeVar('T', str, int)
  3782	            #     def f(x: T) -> T:
  3783	            #         if x == 0:
  3784	            #             ...
  3785	            #         return x
  3786	            #
  3787	            # TODO: find a way of disabling the check only for types resulted from the expansion.
  3788	            return False
  3789	        if isinstance(left, NoneType) or isinstance(right, NoneType):
  3790	            return False
  3791	        if isinstance(left, UnionType) and isinstance(right, UnionType):
  3792	            left = remove_optional(left)
  3793	            right = remove_optional(right)
  3794	            left, right = get_proper_types((left, right))
  3795	        if (
  3796	            original_container
  3797	            and has_bytes_component(original_container)
  3798	            and has_bytes_component(left)
  3799	        ):
  3800	            # We need to special case bytes and bytearray, because 97 in b'abc', b'a' in b'abc',
  3801	            # b'a' in bytearray(b'abc') etc. all return True (and we want to show the error only
  3802	            # if the check can _never_ be True).
  3803	            return False
  3804	        if isinstance(left, Instance) and isinstance(right, Instance):
  3805	            # Special case some builtin implementations of AbstractSet.
  3806	            left_name = left.type.fullname
  3807	            right_name = right.type.fullname
  3808	            if (
  3809	                left_name in OVERLAPPING_TYPES_ALLOWLIST
  3810	                and right_name in OVERLAPPING_TYPES_ALLOWLIST
  3811	            ):
  3812	                abstract_set = self.chk.lookup_typeinfo("typing.AbstractSet")
  3813	                left = map_instance_to_supertype(left, abstract_set)
  3814	                right = map_instance_to_supertype(right, abstract_set)
  3815	                return self.dangerous_comparison(
  3816	                    left.args[0], right.args[0], seen_types=seen_types
  3817	                )
  3818	            elif left.type.has_base("typing.Mapping") and right.type.has_base("typing.Mapping"):
  3819	                # Similar to above: Mapping ignores the classes, it just compares items.
  3820	                abstract_map = self.chk.lookup_typeinfo("typing.Mapping")
  3821	                left = map_instance_to_supertype(left, abstract_map)
  3822	                right = map_instance_to_supertype(right, abstract_map)
  3823	                return self.dangerous_comparison(
  3824	                    left.args[0], right.args[0], seen_types=seen_types
  3825	                ) or self.dangerous_comparison(left.args[1], right.args[1], seen_types=seen_types)
  3826	            elif left_name in ("builtins.list", "builtins.tuple") and right_name == left_name:
  3827	                return self.dangerous_comparison(
  3828	                    left.args[0], right.args[0], seen_types=seen_types
  3829	                )
  3830	            elif left_name in OVERLAPPING_BYTES_ALLOWLIST and right_name in (
  3831	                OVERLAPPING_BYTES_ALLOWLIST
  3832	            ):
  3833	                return False
  3834	        if isinstance(left, LiteralType) and isinstance(right, LiteralType):
  3835	            if isinstance(left.value, bool) and isinstance(right.value, bool):
  3836	                # Comparing different booleans is not dangerous.
  3837	                return False
  3838	        if isinstance(left, LiteralType) and isinstance(right, Instance):
  3839	            # bytes/bytearray comparisons are supported
  3840	            if left.fallback.type.fullname == "builtins.bytes" and right.type.has_base(
  3841	                "builtins.bytearray"
  3842	            ):
  3843	                return False
  3844	        if isinstance(right, LiteralType) and isinstance(left, Instance):
  3845	            # bytes/bytearray comparisons are supported
  3846	            if right.fallback.type.fullname == "builtins.bytes" and left.type.has_base(
  3847	                "builtins.bytearray"
  3848	            ):
  3849	                return False
  3850	        return not is_overlapping_types(left, right, ignore_promotions=False)

  3990	            """
  3991	            for cls in typ.type.mro:
  3992	                if cls.names.get(attr_name):
  3993	                    return cls.fullname
  3994	            return None
  3995
  3996	        left_type = get_proper_type(left_type)
  3997	        right_type = get_proper_type(right_type)
  3998
  3999	        # If either the LHS or the RHS are Any, we can't really concluding anything
  4000	        # about the operation since the Any type may or may not define an
  4001	        # __op__ or __rop__ method. So, we punt and return Any instead.
  4002
  4003	        if isinstance(left_type, AnyType):
  4004	            any_type = AnyType(TypeOfAny.from_another_any, source_any=left_type)
  4005	            return any_type, any_type
  4006	        if isinstance(right_type, AnyType):
  4007	            any_type = AnyType(TypeOfAny.from_another_any, source_any=right_type)
  4008	            return any_type, any_type
  4009
  4010	        # STEP 1:
  4011	        # We start by getting the __op__ and __rop__ methods, if they exist.
  4012
  4013	        rev_op_name = operators.reverse_op_methods[op_name]
  4014
  4015	        left_op = lookup_operator(op_name, left_type)
  4016	        right_op = lookup_operator(rev_op_name, right_type)
  4017
  4018	        # STEP 2a:
  4019	        # We figure out in which order Python will call the operator methods. As it
  4020	        # turns out, it's not as simple as just trying to call __op__ first and
  4021	        # __rop__ second.
  4022	        #
  4023	        # We store the determined order inside the 'variants_raw' variable,
  4024	        # which records tuples containing the method, base type, and the argument.
  4025
  4026	        if op_name in operators.op_methods_that_shortcut and is_same_type(left_type, right_type):
  4027	            # When we do "A() + A()", for example, Python will only call the __add__ method,
  4028	            # never the __radd__ method.
  4029	            #
  4030	            # This is the case even if the __add__ method is completely missing and the __radd__
  4031	            # method is defined.
  4032
  4033	            variants_raw = [(op_name, left_op, left_type, right_expr)]
  4034	        elif (
  4035	            is_subtype(right_type, left_type)
  4036	            and isinstance(left_type, Instance)
  4037	            and isinstance(right_type, Instance)
  4038	            and not (
  4039	                left_type.type.alt_promote is not None
  4040	                and left_type.type.alt_promote.type is right_type.type
  4041	            )
  4042	            and lookup_definer(left_type, op_name) != lookup_definer(right_type, rev_op_name)
  4043	        ):
  4044	            # When we do "A() + B()" where B is a subclass of A, we'll actually try calling
  4045	            # B's __radd__ method first, but ONLY if B explicitly defines or overrides the
  4046	            # __radd__ method.
  4047	            #
  4048	            # This mechanism lets subclasses "refine" the expected outcome of the operation, even
  4049	            # if they're located on the RHS.
  4050	            #
  4051	            # As a special case, the alt_promote check makes sure that we don't use the
  4052	            # __radd__ method of int if the LHS is a native int type.
  4053
  4054	            variants_raw = [
  4055	                (rev_op_name, right_op, right_type, left_expr),
  4056	                (op_name, left_op, left_type, right_expr),
  4057	            ]
  4058	        else:
  4059	            # In all other cases, we do the usual thing and call __add__ first and
  4060	            # __radd__ second when doing "A() + B()".
  4061
  4062	            variants_raw = [
  4063	                (op_name, left_op, left_type, right_expr),
  4064	                (rev_op_name, right_op, right_type, left_expr),
  4065	            ]
  4066
  4067	        # STEP 3:
  4068	        # We now filter out all non-existent operators. The 'variants' list contains
  4069	        # all operator methods that are actually present, in the order that Python
  4070	        # attempts to invoke them.
  4071
  4072	        variants = [(na, op, obj, arg) for (na, op, obj, arg) in variants_raw if op is not None]
  4073
  4074	        # STEP 4:
  4075	        # We now try invoking each one. If an operation succeeds, end early and return
  4076	        # the corresponding result. Otherwise, return the result and errors associated
  4077	        # with the first entry.
  4078
  4079	        errors = []
  4080	        results = []
  4081	        for name, method, obj, arg in variants:
  4082	            with self.msg.filter_errors(save_filtered_errors=True) as local_errors:
  4083	                result = self.check_method_call(name, obj, method, [arg], [ARG_POS], context)
  4084	            if local_errors.has_new_errors():
  4085	                errors.append(local_errors.filtered_errors())
  4086	                results.append(result)
  4087	            else:
  4088	                if isinstance(obj, Instance) and isinstance(
  4089	                    defn := obj.type.get_method(name), OverloadedFuncDef
  4090	                ):
  4091	                    for item in defn.items:
  4092	                        if (
  4093	                            isinstance(item, Decorator)
  4094	                            and isinstance(typ := item.func.type, CallableType)
  4095	                            and bind_self(typ) == result[1]
  4096	                        ):
  4097	                            self.chk.check_deprecated(item.func, context)
  4098	                return result
  4099
  4100	        # We finish invoking above operators and no early return happens. Therefore,
  4101	        # we check if either the LHS or the RHS is Instance and fallbacks to Any,
  4102	        # if so, we also return Any
  4103	        if (isinstance(left_type, Instance) and left_type.type.fallback_to_any) or (
  4104	            isinstance(right_type, Instance) and right_type.type.fallback_to_any
  4105	        ):
  4106	            any_type = AnyType(TypeOfAny.special_form)
  4107	            return any_type, any_type
  4108
  4109	        # STEP 4b:
  4110	        # Sometimes, the variants list is empty. In that case, we fall-back to attempting to
  4111	        # call the __op__ method (even though it's missing).
  4112
  4113	        if not variants:
  4114	            with self.msg.filter_errors(save_filtered_errors=True) as local_errors:
  4115	                result = self.check_method_call_by_name(
  4116	                    op_name, left_type, [right_expr], [ARG_POS], context
  4117	                )
  4118
  4119	            if local_errors.has_new_errors():
  4120	                errors.append(local_errors.filtered_errors())
  4121	                results.append(result)
  4122	            else:
  4123	                # Although we should not need this case anymore, we keep it just in case, as
  4124	                # otherwise we will get a crash if we introduce inconsistency in checkmember.py
  4125	                return result
  4126
  4127	        self.msg.add_errors(errors[0])
  4128	        if len(results) == 1:
  4129	            return results[0]
  4130	        else:
  4131	            error_any = AnyType(TypeOfAny.from_error)
  4132	            result = error_any, error_any
  4133	            return result

  4135	    def check_op(
  4136	        self,
  4137	        method: str,
  4138	        base_type: Type,
  4139	        arg: Expression,
  4140	        context: Context,
  4141	        allow_reverse: bool = False,
  4142	    ) -> tuple[Type, Type]:
  4143	        """Type check a binary operation which maps to a method call.
  4144
  4145	        Return tuple (result type, inferred operator method type).
  4146	        """
  4147
  4148	        if allow_reverse:
  4149	            left_variants = [base_type]
  4150	            base_type = get_proper_type(base_type)
  4151	            if isinstance(base_type, UnionType):
  4152	                left_variants = list(flatten_nested_unions(base_type.relevant_items()))
  4153	            right_type = self.accept(arg)
  4154
  4155	            # Step 1: We first try leaving the right arguments alone and destructure
  4156	            # just the left ones. (Mypy can sometimes perform some more precise inference
  4157	            # if we leave the right operands a union -- see testOperatorWithEmptyListAndSum.)
  4158	            all_results = []
  4159	            all_inferred = []
  4160
  4161	            with self.msg.filter_errors() as local_errors:
  4162	                for left_possible_type in left_variants:
  4163	                    result, inferred = self.check_op_reversible(
  4164	                        op_name=method,
  4165	                        left_type=left_possible_type,
  4166	                        left_expr=TempNode(left_possible_type, context=context),
  4167	                        right_type=right_type,
  4168	                        right_expr=arg,
  4169	                        context=context,
  4170	                    )
  4171	                    all_results.append(result)
  4172	                    all_inferred.append(inferred)
  4173
  4174	            if not local_errors.has_new_errors():
  4175	                results_final = make_simplified_union(all_results)
  4176	                inferred_final = make_simplified_union(all_inferred)
  4177	                return results_final, inferred_final
  4178
  4179	            # Step 2: If that fails, we try again but also destructure the right argument.
  4180	            # This is also necessary to make certain edge cases work -- see
  4181	            # testOperatorDoubleUnionInterwovenUnionAdd, for example.
  4182
  4183	            # Note: We want to pass in the original 'arg' for 'left_expr' and 'right_expr'
  4184	            # whenever possible so that plugins and similar things can introspect on the original
  4185	            # node if possible.
  4186	            #
  4187	            # We don't do the same for the base expression because it could lead to weird
  4188	            # type inference errors -- e.g. see 'testOperatorDoubleUnionSum'.
  4189	            # TODO: Can we use `type_overrides_set()` here?
  4190	            right_variants = [(right_type, arg)]
  4191	            right_type = get_proper_type(right_type)
  4192	            if isinstance(right_type, UnionType):
  4193	                right_variants = [
  4194	                    (item, TempNode(item, context=context))
  4195	                    for item in flatten_nested_unions(right_type.relevant_items())
  4196	                ]
  4197
  4198	            all_results = []
  4199	            all_inferred = []
  4200
  4201	            with self.msg.filter_errors(save_filtered_errors=True) as local_errors:
  4202	                for left_possible_type in left_variants:
  4203	                    for right_possible_type, right_expr in right_variants:
  4204	                        result, inferred = self.check_op_reversible(
  4205	                            op_name=method,
  4206	                            left_type=left_possible_type,
  4207	                            left_expr=TempNode(left_possible_type, context=context),
  4208	                            right_type=right_possible_type,
  4209	                            right_expr=right_expr,
  4210	                            context=context,
  4211	                        )
  4212	                        all_results.append(result)
  4213	                        all_inferred.append(inferred)
  4214
  4215	            if local_errors.has_new_errors():
  4216	                self.msg.add_errors(local_errors.filtered_errors())
  4217	                # Point any notes to the same location as an existing message.
  4218	                err = local_errors.filtered_errors()[-1]
  4219	                recent_context = TempNode(NoneType())
  4220	                recent_context.line = err.line
  4221	                recent_context.column = err.column
  4222	                if len(left_variants) >= 2 and len(right_variants) >= 2:
  4223	                    self.msg.warn_both_operands_are_from_unions(recent_context)
  4224	                elif len(left_variants) >= 2:
  4225	                    self.msg.warn_operand_was_from_union("Left", base_type, context=recent_context)
  4226	                elif len(right_variants) >= 2:
  4227	                    self.msg.warn_operand_was_from_union(
  4228	                        "Right", right_type, context=recent_context
  4229	                    )
  4230
  4231	            # See the comment in 'check_overload_call' for more details on why
  4232	            # we call 'combine_function_signature' instead of just unioning the inferred
  4233	            # callable types.
  4234	            results_final = make_simplified_union(all_results)
  4235	            inferred_final = self.combine_function_signatures(get_proper_types(all_inferred))
  4236	            return results_final, inferred_final
  4237	        else:
  4238	            return self.check_method_call_by_name(
  4239	                method=method,
  4240	                base_type=base_type,
  4241	                args=[arg],
  4242	                arg_kinds=[ARG_POS],
  4243	                context=context,
  4244	            )

  4246	    def check_boolean_op(self, e: OpExpr, context: Context) -> Type:
  4247	        """Type check a boolean operation ('and' or 'or')."""
  4248
  4249	        # A boolean operation can evaluate to either of the operands.
  4250
  4251	        # We use the current type context to guide the type inference of of
  4252	        # the left operand. We also use the left operand type to guide the type
  4253	        # inference of the right operand so that expressions such as
  4254	        # '[1] or []' are inferred correctly.
  4255	        ctx = self.type_context[-1]
  4256	        left_type = self.accept(e.left, ctx)
  4257	        expanded_left_type = try_expanding_sum_type_to_union(
  4258	            self.accept(e.left, ctx), "builtins.bool"
  4259	        )
  4260
  4261	        assert e.op in ("and", "or")  # Checked by visit_op_expr
  4262
  4263	        if e.right_always:
  4264	            left_map: mypy.checker.TypeMap = None
  4265	            right_map: mypy.checker.TypeMap = {}
  4266	        elif e.right_unreachable:
  4267	            left_map, right_map = {}, None
  4268	        elif e.op == "and":
  4269	            right_map, left_map = self.chk.find_isinstance_check(e.left)
  4270	        elif e.op == "or":
  4271	            left_map, right_map = self.chk.find_isinstance_check(e.left)
  4272
  4273	        # If left_map is None then we know mypy considers the left expression
  4274	        # to be redundant.
  4275	        if (
  4276	            codes.REDUNDANT_EXPR in self.chk.options.enabled_error_codes
  4277	            and left_map is None
  4278	            # don't report an error if it's intentional
  4279	            and not e.right_always
  4280	        ):
  4281	            self.msg.redundant_left_operand(e.op, e.left)
  4282
  4283	        if (
  4284	            self.chk.should_report_unreachable_issues()
  4285	            and right_map is None
  4286	            # don't report an error if it's intentional
  4287	            and not e.right_unreachable
  4288	        ):
  4289	            self.msg.unreachable_right_operand(e.op, e.right)
  4290
  4291	        right_type = self.analyze_cond_branch(right_map, e.right, expanded_left_type)
  4292
  4293	        if left_map is None and right_map is None:
  4294	            return UninhabitedType()
  4295
  4296	        if right_map is None:
  4297	            # The boolean expression is statically known to be the left value
  4298	            assert left_map is not None
  4299	            return left_type
  4300	        if left_map is None:
  4301	            # The boolean expression is statically known to be the right value
  4302	            assert right_map is not None
  4303	            return right_type
  4304
  4305	        if e.op == "and":
  4306	            restricted_left_type = false_only(expanded_left_type)
  4307	            result_is_left = not expanded_left_type.can_be_true
  4308	        elif e.op == "or":
  4309	            restricted_left_type = true_only(expanded_left_type)
  4310	            result_is_left = not expanded_left_type.can_be_false
  4311
  4312	        if isinstance(restricted_left_type, UninhabitedType):
  4313	            # The left operand can never be the result
  4314	            return right_type
  4315	        elif result_is_left:
  4316	            # The left operand is always the result
  4317	            return left_type
  4318	        else:
  4319	            return make_simplified_union([restricted_left_type, right_type])

  4562	    def try_getting_int_literals(self, index: Expression) -> list[int] | None:
  4563	        """If the given expression or type corresponds to an int literal
  4564	        or a union of int literals, returns a list of the underlying ints.
  4565	        Otherwise, returns None.
  4566
  4567	        Specifically, this function is guaranteed to return a list with
  4568	        one or more ints if one the following is true:
  4569
  4570	        1. 'expr' is a IntExpr or a UnaryExpr backed by an IntExpr
  4571	        2. 'typ' is a LiteralType containing an int
  4572	        3. 'typ' is a UnionType containing only LiteralType of ints
  4573	        """
  4574	        if isinstance(index, IntExpr):
  4575	            return [index.value]
  4576	        elif isinstance(index, UnaryExpr):
  4577	            if index.op == "-":
  4578	                operand = index.expr
  4579	                if isinstance(operand, IntExpr):
  4580	                    return [-1 * operand.value]
  4581	            if index.op == "+":
  4582	                operand = index.expr
  4583	                if isinstance(operand, IntExpr):
  4584	                    return [operand.value]
  4585	        typ = get_proper_type(self.accept(index))
  4586	        if isinstance(typ, Instance) and typ.last_known_value is not None:
  4587	            typ = typ.last_known_value
  4588	        if isinstance(typ, LiteralType) and isinstance(typ.value, int):
  4589	            return [typ.value]
  4590	        if isinstance(typ, UnionType):
  4591	            out = []
  4592	            for item in get_proper_types(typ.items):
  4593	                if isinstance(item, LiteralType) and isinstance(item.value, int):
  4594	                    out.append(item.value)
  4595	                else:
  4596	                    return None
  4597	            return out
  4598	        return None
