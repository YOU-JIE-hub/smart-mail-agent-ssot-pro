   104	class MypycPlugin(Plugin):
   105	    """Plugin for making mypyc interoperate properly with mypy incremental mode.
   106
   107	    Basically the point of this plugin is to force mypy to recheck things
   108	    based on the demands of mypyc in a couple situations:
   109	      * Any modules in the same group must be compiled together, so we
   110	        tell mypy that modules depend on all their groupmates.
   111	      * If the IR metadata is missing or stale or any of the generated
   112	        C source files associated missing or stale, then we need to
   113	        recompile the module so we mark it as stale.
   114	    """
   115
   116	    def __init__(
   117	        self, options: Options, compiler_options: CompilerOptions, groups: Groups
   118	    ) -> None:
   119	        super().__init__(options)
   120	        self.group_map: dict[str, tuple[str | None, list[str]]] = {}
   121	        for sources, name in groups:
   122	            modules = sorted(source.module for source in sources)
   123	            for id in modules:
   124	                self.group_map[id] = (name, modules)
   125
   126	        self.compiler_options = compiler_options
   127	        self.metastore = create_metastore(options)

   129	    def report_config_data(self, ctx: ReportConfigContext) -> tuple[str | None, list[str]] | None:
   130	        # The config data we report is the group map entry for the module.
   131	        # If the data is being used to check validity, we do additional checks
   132	        # that the IR cache exists and matches the metadata cache and all
   133	        # output source files exist and are up to date.
   134
   135	        id, path, is_check = ctx.id, ctx.path, ctx.is_check
   136
   137	        if id not in self.group_map:
   138	            return None
   139
   140	        # If we aren't doing validity checks, just return the cache data
   141	        if not is_check:
   142	            return self.group_map[id]
   143
   144	        # Load the metadata and IR cache
   145	        meta_path, _, _ = get_cache_names(id, path, self.options)
   146	        ir_path = get_ir_cache_name(id, path, self.options)
   147	        try:
   148	            meta_json = self.metastore.read(meta_path)
   149	            ir_json = self.metastore.read(ir_path)
   150	        except FileNotFoundError:
   151	            # This could happen if mypyc failed after mypy succeeded
   152	            # in the previous run or if some cache files got
   153	            # deleted. No big deal, just fail to load the cache.
   154	            return None
   155
   156	        ir_data = json.loads(ir_json)
   157
   158	        # Check that the IR cache matches the metadata cache
   159	        if hash_digest(meta_json) != ir_data["meta_hash"]:
   160	            return None
   161
   162	        # Check that all of the source files are present and as
   163	        # expected. The main situation where this would come up is the
   164	        # user deleting the build directory without deleting
   165	        # .mypy_cache, which we should handle gracefully.
   166	        for path, hash in ir_data["src_hashes"].items():
   167	            try:
   168	                with open(os.path.join(self.compiler_options.target_dir, path), "rb") as f:
   169	                    contents = f.read()
   170	            except FileNotFoundError:
   171	                return None
   172	            real_hash = hash_digest(contents)
   173	            if hash != real_hash:
   174	                return None
   175
   176	        return self.group_map[id]

   223	    Returns the IR of the modules.
   224	    """
   225
   226	    if compiler_options.verbose:
   227	        print("Compiling {}".format(", ".join(x.name for x in scc)))
   228
   229	    # Generate basic IR, with missing exception and refcount handling.
   230	    modules = build_ir(scc, result.graph, result.types, mapper, compiler_options, errors)
   231	    if errors.num_errors > 0:
   232	        return modules
   233
   234	    env_user_functions = {}
   235	    for module in modules.values():
   236	        for cls in module.classes:
   237	            if cls.env_user_function:
   238	                env_user_functions[cls.env_user_function] = cls
   239
   240	    for module in modules.values():
   241	        for fn in module.functions:
   242	            # Insert uninit checks.
   243	            insert_uninit_checks(fn)
   244	            # Insert exception handling.
   245	            insert_exception_handling(fn)
   246	            # Insert refcount handling.
   247	            insert_ref_count_opcodes(fn)
   248
   249	            if fn in env_user_functions:
   250	                insert_spills(fn, env_user_functions[fn])
   251
   252	            # Switch to lower abstraction level IR.
   253	            lower_ir(fn, compiler_options)
   254	            # Perform optimizations.
   255	            do_copy_propagation(fn, compiler_options)
   256	            do_flag_elimination(fn, compiler_options)
   257
   258	    return modules

   261	def compile_modules_to_ir(
   262	    result: BuildResult, mapper: Mapper, compiler_options: CompilerOptions, errors: Errors
   263	) -> ModuleIRs:
   264	    """Compile a collection of modules into ModuleIRs.
   265
   266	    The modules to compile are specified as part of mapper's group_map.
   267
   268	    Returns the IR of the modules.
   269	    """
   270	    deser_ctx = DeserMaps({}, {})
   271	    modules = {}
   272
   273	    # Process the graph by SCC in topological order, like we do in mypy.build
   274	    for scc in sorted_components(result.graph):
   275	        scc_states = [result.graph[id] for id in scc]
   276	        trees = [st.tree for st in scc_states if st.id in mapper.group_map and st.tree]
   277
   278	        if not trees:
   279	            continue
   280
   281	        fresh = all(id not in result.manager.rechecked_modules for id in scc)
   282	        if fresh:
   283	            load_scc_from_cache(trees, result, mapper, deser_ctx)
   284	        else:
   285	            scc_ir = compile_scc_to_ir(trees, result, mapper, compiler_options, errors)
   286	            modules.update(scc_ir)
   287
   288	    return modules

   291	def compile_ir_to_c(
   292	    groups: Groups,
   293	    modules: ModuleIRs,
   294	    result: BuildResult,
   295	    mapper: Mapper,
   296	    compiler_options: CompilerOptions,
   297	) -> dict[str | None, list[tuple[str, str]]]:
   298	    """Compile a collection of ModuleIRs to C source text.
   299
   300	    Returns a dictionary mapping group names to a list of (file name,
   301	    file text) pairs.
   302	    """
   303	    source_paths = {
   304	        source.module: result.graph[source.module].xpath
   305	        for sources, _ in groups
   306	        for source in sources
   307	    }
   308
   309	    names = NameGenerator(
   310	        [[source.module for source in sources] for sources, _ in groups],
   311	        separate=compiler_options.separate,
   312	    )
   313
   314	    # Generate C code for each compilation group. Each group will be
   315	    # compiled into a separate extension module.
   316	    ctext: dict[str | None, list[tuple[str, str]]] = {}
   317	    for group_sources, group_name in groups:
   318	        group_modules = {
   319	            source.module: modules[source.module]
   320	            for source in group_sources
   321	            if source.module in modules
   322	        }
   323	        if not group_modules:
   324	            ctext[group_name] = []
   325	            continue
   326	        generator = GroupGenerator(
   327	            group_modules, source_paths, group_name, mapper.group_map, names, compiler_options
   328	        )
   329	        ctext[group_name] = generator.generate_c_for_modules()
   330
   331	    return ctext

   343	def write_cache(
   344	    modules: ModuleIRs,
   345	    result: BuildResult,
   346	    group_map: dict[str, str | None],
   347	    ctext: dict[str | None, list[tuple[str, str]]],
   348	) -> None:
   349	    """Write out the cache information for modules.
   350
   351	    Each module has the following cache information written (which is
   352	    in addition to the cache information written by mypy itself):
   353	      * A serialized version of its mypyc IR, minus the bodies of
   354	        functions. This allows code that depends on it to use
   355	        these serialized data structures when compiling against it
   356	        instead of needing to recompile it. (Compiling against a
   357	        module requires access to both its mypy and mypyc data
   358	        structures.)
   359	      * The hash of the mypy metadata cache file for the module.
   360	        This is used to ensure that the mypyc cache and the mypy
   361	        cache are in sync and refer to the same version of the code.
   362	        This is particularly important if mypyc crashes/errors/is
   363	        stopped after mypy has written its cache but before mypyc has.
   364	      * The hashes of all of the source file outputs for the group
   365	        the module is in. This is so that the module will be
   366	        recompiled if the source outputs are missing.
   367	    """
   368
   369	    hashes = {}
   370	    for name, files in ctext.items():
   371	        hashes[name] = {file: compute_hash(data) for file, data in files}
   372
   373	    # Write out cache data
   374	    for id, module in modules.items():
   375	        st = result.graph[id]
   376
   377	        meta_path, _, _ = get_cache_names(id, st.xpath, result.manager.options)
   378	        # If the metadata isn't there, skip writing the cache.
   379	        try:
   380	            meta_data = result.manager.metastore.read(meta_path)
   381	        except OSError:
   382	            continue
   383
   384	        newpath = get_state_ir_cache_name(st)
   385	        ir_data = {
   386	            "ir": module.serialize(),
   387	            "meta_hash": hash_digest(meta_data),
   388	            "src_hashes": hashes[group_map[id]],
   389	        }
   390
   391	        result.manager.metastore.write(newpath, json_dumps(ir_data))
   392
   393	    result.manager.metastore.commit()

   397	    scc: list[MypyFile], result: BuildResult, mapper: Mapper, ctx: DeserMaps

   414	def compile_modules_to_c(
   415	    result: BuildResult, compiler_options: CompilerOptions, errors: Errors, groups: Groups
   416	) -> tuple[ModuleIRs, list[FileContents], Mapper]:
   417	    """Compile Python module(s) to the source of Python C extension modules.
   418
   419	    This generates the source code for the "shared library" module
   420	    for each group. The shim modules are generated in mypyc.build.
   421	    Each shared library module provides, for each module in its group,
   422	    a PyCapsule containing an initialization function.
   423	    Additionally, it provides a capsule containing an export table of
   424	    pointers to all of the group's functions and static variables.
   425
   426	    Arguments:
   427	        result: The BuildResult from the mypy front-end
   428	        compiler_options: The compilation options
   429	        errors: Where to report any errors encountered
   430	        groups: The groups that we are compiling. See documentation of Groups type above.
   431
   432	    Returns the IR of the modules and a list containing the generated files for each group.
   433	    """
   434	    # Construct a map from modules to what group they belong to
   435	    group_map = {source.module: lib_name for group, lib_name in groups for source in group}
   436	    mapper = Mapper(group_map)
   437
   438	    # Sometimes when we call back into mypy, there might be errors.
   439	    # We don't want to crash when that happens.
   440	    result.manager.errors.set_file(
   441	        "<mypyc>", module=None, scope=None, options=result.manager.options
   442	    )
   443
   444	    modules = compile_modules_to_ir(result, mapper, compiler_options, errors)
   445	    if errors.num_errors > 0:
   446	        return {}, [], Mapper({})
   447
   448	    ctext = compile_ir_to_c(groups, modules, result, mapper, compiler_options)
   449	    write_cache(modules, result, group_map, ctext)
   450
   451	    return modules, [ctext[name] for _, name in groups], mapper

   469	def pointerize(decl: str, name: str) -> str:
   470	    """Given a C decl and its name, modify it to be a declaration to a pointer."""
   471	    # This doesn't work in general but does work for all our types...
   472	    if "(" in decl:
   473	        # Function pointer. Stick an * in front of the name and wrap it in parens.
   474	        return decl.replace(name, f"(*{name})")
   475	    else:
   476	        # Non-function pointer. Just stick an * in front of the name.
   477	        return decl.replace(name, f"*{name}")

   485	class GroupGenerator:
   486	    def __init__(
   487	        self,
   488	        modules: dict[str, ModuleIR],
   489	        source_paths: dict[str, str],
   490	        group_name: str | None,
   491	        group_map: dict[str, str | None],
   492	        names: NameGenerator,
   493	        compiler_options: CompilerOptions,
   494	    ) -> None:
   495	        """Generator for C source for a compilation group.
   496
   497	        The code for a compilation group contains an internal and an
   498	        external .h file, and then one .c if not in multi_file mode or
   499	        one .c file per module if in multi_file mode.)
   500
   501	        Arguments:
   502	            modules: (name, ir) pairs for each module in the group
   503	            source_paths: Map from module names to source file paths
   504	            group_name: The name of the group (or None if this is single-module compilation)
   505	            group_map: A map of modules to their group names
   506	            names: The name generator for the compilation
   507	            multi_file: Whether to put each module in its own source file regardless
   508	                        of group structure.
   509	        """
   510	        self.modules = modules
   511	        self.source_paths = source_paths
   512	        self.context = EmitterContext(names, group_name, group_map)
   513	        self.names = names
   514	        # Initializations of globals to simple values that we can't
   515	        # do statically because the windows loader is bad.
   516	        self.simple_inits: list[tuple[str, str]] = []
   517	        self.group_name = group_name
   518	        self.use_shared_lib = group_name is not None
   519	        self.compiler_options = compiler_options
   520	        self.multi_file = compiler_options.multi_file
   521	        # Multi-phase init is needed to enable free-threading. In the future we'll
   522	        # probably want to enable it always, but we'll wait until it's stable.
   523	        self.multi_phase_init = IS_FREE_THREADED
   524
   525	    @property
   526	    def group_suffix(self) -> str:
   527	        return "_" + exported_name(self.group_name) if self.group_name else ""
   528
   529	    @property
   530	    def short_group_suffix(self) -> str:
   531	        return "_" + exported_name(self.group_name.split(".")[-1]) if self.group_name else ""
   532
   533	    def generate_c_for_modules(self) -> list[tuple[str, str]]:
   534	        file_contents = []
   535	        multi_file = self.use_shared_lib and self.multi_file
   536
   537	        # Collect all literal refs in IR.
   538	        for module in self.modules.values():
   539	            for fn in module.functions:
   540	                collect_literals(fn, self.context.literals)
   541
   542	        base_emitter = Emitter(self.context)
   543	        # Optionally just include the runtime library c files to
   544	        # reduce the number of compiler invocations needed
   545	        if self.compiler_options.include_runtime_files:
   546	            for name in RUNTIME_C_FILES:
   547	                base_emitter.emit_line(f'#include "{name}"')
   548	        base_emitter.emit_line(f'#include "__native{self.short_group_suffix}.h"')
   549	        base_emitter.emit_line(f'#include "__native_internal{self.short_group_suffix}.h"')
   550	        emitter = base_emitter
   551
   552	        self.generate_literal_tables()
   553
   554	        for module_name, module in self.modules.items():
   555	            if multi_file:
   556	                emitter = Emitter(self.context)
   557	                emitter.emit_line(f'#include "__native{self.short_group_suffix}.h"')
   558	                emitter.emit_line(f'#include "__native_internal{self.short_group_suffix}.h"')
   559
   560	            self.declare_module(module_name, emitter)
   561	            self.declare_internal_globals(module_name, emitter)
   562	            self.declare_imports(module.imports, emitter)
   563
   564	            for cl in module.classes:
   565	                if cl.is_ext_class:
   566	                    generate_class(cl, module_name, emitter)
   567
   568	            # Generate Python extension module definitions and module initialization functions.
   569	            self.generate_module_def(emitter, module_name, module)
   570
   571	            for fn in module.functions:
   572	                emitter.emit_line()
   573	                generate_native_function(fn, emitter, self.source_paths[module_name], module_name)
   574	                if fn.name != TOP_LEVEL_NAME and not fn.internal:
   575	                    emitter.emit_line()
   576	                    if is_fastcall_supported(fn, emitter.capi_version):
   577	                        generate_wrapper_function(
   578	                            fn, emitter, self.source_paths[module_name], module_name
   579	                        )
   580	                    else:
   581	                        generate_legacy_wrapper_function(
   582	                            fn, emitter, self.source_paths[module_name], module_name
   583	                        )
   584	            if multi_file:
   585	                name = f"__native_{exported_name(module_name)}.c"
   586	                file_contents.append((name, "".join(emitter.fragments)))
   587
   588	        # The external header file contains type declarations while
   589	        # the internal contains declarations of functions and objects
   590	        # (which are shared between shared libraries via dynamic
   591	        # exports tables and not accessed directly.)
   592	        ext_declarations = Emitter(self.context)
   593	        ext_declarations.emit_line(f"#ifndef MYPYC_NATIVE{self.group_suffix}_H")
   594	        ext_declarations.emit_line(f"#define MYPYC_NATIVE{self.group_suffix}_H")
   595	        ext_declarations.emit_line("#include <Python.h>")
   596	        ext_declarations.emit_line("#include <CPy.h>")
   597
   598	        declarations = Emitter(self.context)
   599	        declarations.emit_line(f"#ifndef MYPYC_NATIVE_INTERNAL{self.group_suffix}_H")
   600	        declarations.emit_line(f"#define MYPYC_NATIVE_INTERNAL{self.group_suffix}_H")
   601	        declarations.emit_line("#include <Python.h>")
   602	        declarations.emit_line("#include <CPy.h>")
   603	        declarations.emit_line(f'#include "__native{self.short_group_suffix}.h"')
   604	        declarations.emit_line()
   605	        declarations.emit_line("int CPyGlobalsInit(void);")
   606	        declarations.emit_line()
   607
   608	        for module_name, module in self.modules.items():
   609	            self.declare_finals(module_name, module.final_names, declarations)
   610	            for cl in module.classes:
   611	                generate_class_type_decl(cl, emitter, ext_declarations, declarations)
   612	            self.declare_type_vars(module_name, module.type_var_names, declarations)
   613	            for fn in module.functions:
   614	                generate_function_declaration(fn, declarations)
   615
   616	        for lib in sorted(self.context.group_deps):
   617	            elib = exported_name(lib)
   618	            short_lib = exported_name(lib.split(".")[-1])
   619	            declarations.emit_lines(
   620	                "#include <{}>".format(os.path.join(group_dir(lib), f"__native_{short_lib}.h")),
   621	                f"struct export_table_{elib} exports_{elib};",
   622	            )
   623
   624	        sorted_decls = self.toposort_declarations()
   625
   626	        emitter = base_emitter
   627	        self.generate_globals_init(emitter)
   628
   629	        emitter.emit_line()
   630
   631	        for declaration in sorted_decls:
   632	            decls = ext_declarations if declaration.is_type else declarations
   633	            if not declaration.is_type:
   634	                decls.emit_lines(f"extern {declaration.decl[0]}", *declaration.decl[1:])
   635	                # If there is a definition, emit it. Otherwise repeat the declaration
   636	                # (without an extern).
   637	                if declaration.defn:
   638	                    emitter.emit_lines(*declaration.defn)
   639	                else:
   640	                    emitter.emit_lines(*declaration.decl)
   641	            else:
   642	                decls.emit_lines(*declaration.decl)
   643
   644	        if self.group_name:
   645	            self.generate_export_table(ext_declarations, emitter)
   646
   647	            self.generate_shared_lib_init(emitter)
   648
   649	        ext_declarations.emit_line("#endif")
   650	        declarations.emit_line("#endif")
   651
   652	        output_dir = group_dir(self.group_name) if self.group_name else ""
   653	        return file_contents + [
   654	            (
   655	                os.path.join(output_dir, f"__native{self.short_group_suffix}.c"),
   656	                "".join(emitter.fragments),
   657	            ),
   658	            (
   659	                os.path.join(output_dir, f"__native_internal{self.short_group_suffix}.h"),
   660	                "".join(declarations.fragments),
   661	            ),
   662	            (
   663	                os.path.join(output_dir, f"__native{self.short_group_suffix}.h"),
   664	                "".join(ext_declarations.fragments),
   665	            ),
   666	        ]

   668	    def generate_literal_tables(self) -> None:
   669	        """Generate tables containing descriptions of Python literals to construct.
   670
   671	        We will store the constructed literals in a single array that contains
   672	        literals of all types. This way we can refer to an arbitrary literal by
   673	        its index.
   674	        """

   676	        # During module initialization we store all the constructed objects here
   677	        self.declare_global("PyObject *[%d]" % literals.num_literals(), "CPyStatics")
   678	        # Descriptions of str literals
   679	        init_str = c_string_array_initializer(literals.encoded_str_values())
   680	        self.declare_global("const char * const []", "CPyLit_Str", initializer=init_str)
   681	        # Descriptions of bytes literals
   682	        init_bytes = c_string_array_initializer(literals.encoded_bytes_values())
   683	        self.declare_global("const char * const []", "CPyLit_Bytes", initializer=init_bytes)
   684	        # Descriptions of int literals
   685	        init_int = c_string_array_initializer(literals.encoded_int_values())
   686	        self.declare_global("const char * const []", "CPyLit_Int", initializer=init_int)
   687	        # Descriptions of float literals
   688	        init_floats = c_array_initializer(literals.encoded_float_values())
   689	        self.declare_global("const double []", "CPyLit_Float", initializer=init_floats)
   690	        # Descriptions of complex literals
   691	        init_complex = c_array_initializer(literals.encoded_complex_values())
   692	        self.declare_global("const double []", "CPyLit_Complex", initializer=init_complex)
   693	        # Descriptions of tuple literals
   694	        init_tuple = c_array_initializer(literals.encoded_tuple_values())
   695	        self.declare_global("const int []", "CPyLit_Tuple", initializer=init_tuple)
   696	        # Descriptions of frozenset literals

   700	    def generate_export_table(self, decl_emitter: Emitter, code_emitter: Emitter) -> None:
   701	        """Generate the declaration and definition of the group's export struct.
   702
   703	        To avoid needing to deal with deeply platform specific issues
   704	        involving dynamic library linking (and some possibly
   705	        insurmountable issues involving cyclic dependencies), compiled
   706	        code accesses functions and data in other compilation groups
   707	        via an explicit "export struct".
   708
   709	        Each group declares a struct type that contains a pointer to
   710	        every function and static variable it exports. It then
   711	        populates this struct and stores a pointer to it in a capsule
   712	        stored as an attribute named 'exports' on the group's shared
   713	        library's python module.
   714
   715	        On load, a group's init function will import all of its
   716	        dependencies' exports tables using the capsule mechanism and
   717	        copy the contents into a local copy of the table (to eliminate
   718	        the need for a pointer indirection when accessing it).
   719
   720	        Then, all calls to functions in another group and accesses to statics
   721	        from another group are done indirectly via the export table.
   722
   723	        For example, a group containing a module b, where b contains a class B
   724	        and a function bar, would declare an export table like:
   725	            struct export_table_b {
   726	                PyTypeObject **CPyType_B;
   727	                PyObject *(*CPyDef_B)(CPyTagged cpy_r_x);
   728	                CPyTagged (*CPyDef_B___foo)(PyObject *cpy_r_self, CPyTagged cpy_r_y);
   729	                tuple_T2OI (*CPyDef_bar)(PyObject *cpy_r_x);
   730	                char (*CPyDef___top_level__)(void);
   731	            };
   732	        that would be initialized with:
   733	            static struct export_table_b exports = {
   734	                &CPyType_B,
   735	                &CPyDef_B,
   736	                &CPyDef_B___foo,
   737	                &CPyDef_bar,
   738	                &CPyDef___top_level__,
   739	            };
   740	        To call `b.foo`, then, a function in another group would do
   741	        `exports_b.CPyDef_bar(...)`.
   742	        """
   743
   744	        decls = decl_emitter.context.declarations
   745
   746	        decl_emitter.emit_lines("", f"struct export_table{self.group_suffix} {{")
   747	        for name, decl in decls.items():
   748	            if decl.needs_export:

    40	# Generic vectorcall wrapper functions (Python 3.7+)
    41	#
    42	# A wrapper function has a signature like this:
    43	#
    44	# PyObject *fn(PyObject *self, PyObject *const *args, Py_ssize_t nargs, PyObject *kwnames)
    45	#
    46	# The function takes a self object, pointer to an array of arguments,
    47	# the number of positional arguments, and a tuple of keyword argument
    48	# names (that are stored starting in args[nargs]).
    49	#
    50	# It returns the returned object, or NULL on an exception.
    51	#
    52	# These are more efficient than legacy wrapper functions, since
    53	# usually no tuple or dict objects need to be created for the
    54	# arguments. Vectorcalls also use pre-constructed str objects for
    55	# keyword argument names and other pre-computed information, instead
    56	# of processing the argument format string on each call.
    57
    58
    59	def wrapper_function_header(fn: FuncIR, names: NameGenerator) -> str:
    60	    """Return header of a vectorcall wrapper function.
    61
    62	    See comment above for a summary of the arguments.
    63	    """
    64	    assert not fn.internal
    65	    return (
    66	        "PyObject *{prefix}{name}("
    67	        "PyObject *self, PyObject *const *args, size_t nargs, PyObject *kwnames)"
    68	    ).format(prefix=PREFIX, name=fn.cname(names))

    71	def generate_traceback_code(
    72	    fn: FuncIR, emitter: Emitter, source_path: str, module_name: str
    73	) -> str:
    74	    # If we hit an error while processing arguments, then we emit a
    75	    # traceback frame to make it possible to debug where it happened.
    76	    # Unlike traceback frames added for exceptions seen in IR, we do this
    77	    # even if there is no `traceback_name`. This is because the error will
    78	    # have originated here and so we need it in the traceback.
    79	    globals_static = emitter.static_name("globals", module_name)
    80	    traceback_code = 'CPy_AddTraceback("%s", "%s", %d, %s);' % (
    81	        source_path.replace("\\", "\\\\"),
    82	        fn.traceback_name or fn.name,
    83	        fn.line,
    84	        globals_static,
    85	    )
    86	    return traceback_code

   104	def make_format_string(func_name: str | None, groups: dict[ArgKind, list[RuntimeArg]]) -> str:
   105	    """Return a format string that specifies the accepted arguments.
   106
   107	    The format string is an extended subset of what is supported by
   108	    PyArg_ParseTupleAndKeywords(). Only the type 'O' is used, and we
   109	    also support some extensions:
   110
   111	    - Required keyword-only arguments are introduced after '@'
   112	    - If the function receives *args or **kwargs, we add a '%' prefix
   113
   114	    Each group requires the previous groups' delimiters to be present
   115	    first.
   116
   117	    These are used by both vectorcall and legacy wrapper functions.
   118	    """
   119	    format = ""
   120	    if groups[ARG_STAR] or groups[ARG_STAR2]:
   121	        format += "%"
   122	    format += "O" * len(groups[ARG_POS])
   123	    if groups[ARG_OPT] or groups[ARG_NAMED_OPT] or groups[ARG_NAMED]:
   124	        format += "|" + "O" * len(groups[ARG_OPT])
   125	    if groups[ARG_NAMED_OPT] or groups[ARG_NAMED]:
   126	        format += "$" + "O" * len(groups[ARG_NAMED_OPT])
   127	    if groups[ARG_NAMED]:
   128	        format += "@" + "O" * len(groups[ARG_NAMED])
   129	    if func_name is not None:
   130	        format += f":{func_name}"
   131	    return format

   134	def generate_wrapper_function(
   135	    fn: FuncIR, emitter: Emitter, source_path: str, module_name: str
   136	) -> None:
   137	    """Generate a CPython-compatible vectorcall wrapper for a native function.
   138
   139	    In particular, this handles unboxing the arguments, calling the native function, and
   140	    then boxing the return value.
   141	    """
   142	    emitter.emit_line(f"{wrapper_function_header(fn, emitter.names)} {{")
   143
   144	    # If fn is a method, then the first argument is a self param
   145	    real_args = list(fn.args)
   146	    if fn.sig.num_bitmap_args:
   147	        real_args = real_args[: -fn.sig.num_bitmap_args]
   148	    if fn.class_name and fn.decl.kind != FUNC_STATICMETHOD:
   149	        arg = real_args.pop(0)
   150	        emitter.emit_line(f"PyObject *obj_{arg.name} = self;")
   151
   152	    # Need to order args as: required, optional, kwonly optional, kwonly required
   153	    # This is because CPyArg_ParseStackAndKeywords format string requires
   154	    # them grouped in that way.
   155	    groups = make_arg_groups(real_args)
   156	    reordered_args = reorder_arg_groups(groups)
   157
   158	    emitter.emit_line(make_static_kwlist(reordered_args))
   159	    fmt = make_format_string(fn.name, groups)
   160	    # Define the arguments the function accepts (but no types yet)
   161	    emitter.emit_line(f'static CPyArg_Parser parser = {{"{fmt}", kwlist, 0}};')
   162
   163	    for arg in real_args:
   164	        emitter.emit_line(
   165	            "PyObject *obj_{}{};".format(arg.name, " = NULL" if arg.optional else "")
   166	        )
   167
   168	    cleanups = [f"CPy_DECREF(obj_{arg.name});" for arg in groups[ARG_STAR] + groups[ARG_STAR2]]
   169
   170	    arg_ptrs: list[str] = []
   171	    if groups[ARG_STAR] or groups[ARG_STAR2]:
   172	        arg_ptrs += [f"&obj_{groups[ARG_STAR][0].name}" if groups[ARG_STAR] else "NULL"]
   173	        arg_ptrs += [f"&obj_{groups[ARG_STAR2][0].name}" if groups[ARG_STAR2] else "NULL"]
   174	    arg_ptrs += [f"&obj_{arg.name}" for arg in reordered_args]
   175
   176	    if fn.name == "__call__":
   177	        nargs = "PyVectorcall_NARGS(nargs)"
   178	    else:
   179	        nargs = "nargs"
   180	    parse_fn = "CPyArg_ParseStackAndKeywords"
   181	    # Special case some common signatures
   182	    if not real_args:
   183	        # No args
   184	        parse_fn = "CPyArg_ParseStackAndKeywordsNoArgs"
   185	    elif len(real_args) == 1 and len(groups[ARG_POS]) == 1:
   186	        # Single positional arg
   187	        parse_fn = "CPyArg_ParseStackAndKeywordsOneArg"
   188	    elif len(real_args) == len(groups[ARG_POS]) + len(groups[ARG_OPT]):
   189	        # No keyword-only args, *args or **kwargs
   190	        parse_fn = "CPyArg_ParseStackAndKeywordsSimple"
   191	    emitter.emit_lines(
   192	        "if (!{}(args, {}, kwnames, &parser{})) {{".format(
   193	            parse_fn, nargs, "".join(", " + n for n in arg_ptrs)
   194	        ),
   195	        "return NULL;",
   196	        "}",
   197	    )
   198	    for i in range(fn.sig.num_bitmap_args):
   199	        name = bitmap_name(i)
   200	        emitter.emit_line(f"{BITMAP_TYPE} {name} = 0;")
   201	    traceback_code = generate_traceback_code(fn, emitter, source_path, module_name)
   202	    generate_wrapper_core(
   203	        fn,
   204	        emitter,
   205	        groups[ARG_OPT] + groups[ARG_NAMED_OPT],
   206	        cleanups=cleanups,
   207	        traceback_code=traceback_code,
   208	    )
   209
   210	    emitter.emit_line("}")

   330	def generate_bin_op_wrapper(cl: ClassIR, fn: FuncIR, emitter: Emitter) -> str:
   331	    """Generates a wrapper for a native binary dunder method.
   332
   333	    The same wrapper that handles the forward method (e.g. __add__) also handles
   334	    the corresponding reverse method (e.g. __radd__), if defined.
   335
   336	    Both arguments and the return value are PyObject *.
   337	    """
   338	    gen = WrapperGenerator(cl, emitter)
   339	    gen.set_target(fn)
   340	    if fn.name in ("__pow__", "__rpow__"):
   341	        gen.arg_names = ["left", "right", "mod"]
   342	    else:
   343	        gen.arg_names = ["left", "right"]
   344	    wrapper_name = gen.wrapper_name()
   345
   346	    gen.emit_header()
   347	    if fn.name not in reverse_op_methods and fn.name in reverse_op_method_names:
   348	        # There's only a reverse operator method.
   349	        generate_bin_op_reverse_only_wrapper(fn, emitter, gen)
   350	    else:
   351	        rmethod = reverse_op_methods[fn.name]
   352	        fn_rev = cl.get_method(rmethod)
   353	        if fn_rev is None:
   354	            # There's only a forward operator method.
   355	            generate_bin_op_forward_only_wrapper(fn, emitter, gen)
   356	        else:
   357	            # There's both a forward and a reverse operator method.
   358	            generate_bin_op_both_wrappers(cl, fn, fn_rev, emitter, gen)
   359	    return wrapper_name

   362	def generate_bin_op_forward_only_wrapper(
   363	    fn: FuncIR, emitter: Emitter, gen: WrapperGenerator
   364	) -> None:
   365	    gen.emit_arg_processing(error=GotoHandler("typefail"), raise_exception=False)
   366	    handle_third_pow_argument(fn, emitter, gen, if_unsupported=["goto typefail;"])
   367	    gen.emit_call(not_implemented_handler="goto typefail;")
   368	    gen.emit_error_handling()
   369	    emitter.emit_label("typefail")
   370	    # If some argument has an incompatible type, treat this the same as
   371	    # returning NotImplemented, and try to call the reverse operator method.
   372	    #
   373	    # Note that in normal Python you'd instead of an explicit
   374	    # return of NotImplemented, but it doesn't generally work here
   375	    # the body won't be executed at all if there is an argument
   376	    # type check failure.
   377	    #
   378	    # The recommended way is to still use a type check in the
   379	    # body. This will only be used in interpreted mode:
   380	    #
   381	    #    def __add__(self, other: int) -> Foo:
   382	    #        if not isinstance(other, int):
   383	    #            return NotImplemented
   384	    #        ...
   385	    generate_bin_op_reverse_dunder_call(fn, emitter, reverse_op_methods[fn.name])
   386	    gen.finish()

   389	def generate_bin_op_reverse_only_wrapper(
   390	    fn: FuncIR, emitter: Emitter, gen: WrapperGenerator
   391	) -> None:
   392	    gen.arg_names = ["right", "left"]
   393	    gen.emit_arg_processing(error=GotoHandler("typefail"), raise_exception=False)
   394	    handle_third_pow_argument(fn, emitter, gen, if_unsupported=["goto typefail;"])
   395	    gen.emit_call()
   396	    gen.emit_error_handling()
   397	    emitter.emit_label("typefail")
   398	    emitter.emit_line("Py_INCREF(Py_NotImplemented);")

   403	def generate_bin_op_both_wrappers(
   404	    cl: ClassIR, fn: FuncIR, fn_rev: FuncIR, emitter: Emitter, gen: WrapperGenerator
   405	) -> None:
   406	    # There's both a forward and a reverse operator method. First
   407	    # check if we should try calling the forward one. If the
   408	    # argument type check fails, fall back to the reverse method.
   409	    #
   410	    # Similar to above, we can't perfectly match Python semantics.
   411	    # In regular Python code you'd return NotImplemented if the
   412	    # operand has the wrong type, but in compiled code we'll never
   413	    # get to execute the type check.
   414	    emitter.emit_line(
   415	        "if (PyObject_IsInstance(obj_left, (PyObject *){})) {{".format(
   416	            emitter.type_struct_name(cl)
   417	        )
   418	    )
   419	    gen.emit_arg_processing(error=GotoHandler("typefail"), raise_exception=False)
   420	    handle_third_pow_argument(fn, emitter, gen, if_unsupported=["goto typefail2;"])
   421	    # Ternary __rpow__ calls aren't a thing so immediately bail
   422	    # if ternary __pow__ returns NotImplemented.
   423	    if fn.name == "__pow__" and len(fn.args) == 3:
   424	        fwd_not_implemented_handler = "goto typefail2;"
   425	    else:
   426	        fwd_not_implemented_handler = "goto typefail;"
   427	    gen.emit_call(not_implemented_handler=fwd_not_implemented_handler)
   428	    gen.emit_error_handling()
   429	    emitter.emit_line("}")
   430	    emitter.emit_label("typefail")
   431	    emitter.emit_line(
   432	        "if (PyObject_IsInstance(obj_right, (PyObject *){})) {{".format(
   433	            emitter.type_struct_name(cl)
   434	        )
   435	    )
   436	    gen.set_target(fn_rev)
   437	    gen.arg_names = ["right", "left"]
   438	    gen.emit_arg_processing(error=GotoHandler("typefail2"), raise_exception=False)
   439	    handle_third_pow_argument(fn_rev, emitter, gen, if_unsupported=["goto typefail2;"])
   440	    gen.emit_call()
   441	    gen.emit_error_handling()
   442	    emitter.emit_line("} else {")
   443	    generate_bin_op_reverse_dunder_call(fn, emitter, fn_rev.name)
   444	    emitter.emit_line("}")
   445	    emitter.emit_label("typefail2")
   446	    emitter.emit_line("Py_INCREF(Py_NotImplemented);")
   447	    emitter.emit_line("return Py_NotImplemented;")
   448	    gen.finish()

   451	def generate_bin_op_reverse_dunder_call(fn: FuncIR, emitter: Emitter, rmethod: str) -> None:
   452	    if fn.name in ("__pow__", "__rpow__"):
   453	        # Ternary pow() will never call the reverse dunder.
   454	        emitter.emit_line("if (obj_mod == Py_None) {")
   455	    emitter.emit_line(f"_Py_IDENTIFIER({rmethod});")
   456	    emitter.emit_line(
   457	        'return CPy_CallReverseOpMethod(obj_left, obj_right, "{}", &PyId_{});'.format(
   458	            op_methods_to_symbols[fn.name], rmethod
   459	        )
   460	    )
   461	    if fn.name in ("__pow__", "__rpow__"):
   462	        emitter.emit_line("} else {")
   463	        emitter.emit_line("Py_INCREF(Py_NotImplemented);")
   464	        emitter.emit_line("return Py_NotImplemented;")
   465	        emitter.emit_line("}")

   468	def handle_third_pow_argument(
   469	    fn: FuncIR, emitter: Emitter, gen: WrapperGenerator, *, if_unsupported: list[str]
   470	) -> None:
   471	    if fn.name not in ("__pow__", "__rpow__", "__ipow__"):
   472	        return
   473
   474	    if (fn.name in ("__pow__", "__ipow__") and len(fn.args) == 2) or fn.name == "__rpow__":
   475	        # If the power dunder only supports two arguments and the third
   476	        # argument (AKA mod) is set to a non-default value, simply bail.
   477	        #
   478	        # Importantly, this prevents any ternary __rpow__ calls from
   479	        # happening (as per the language specification).
   480	        emitter.emit_line("if (obj_mod != Py_None) {")
   481	        for line in if_unsupported:
   482	            emitter.emit_line(line)
   483	        emitter.emit_line("}")
   484	        # The slot wrapper will receive three arguments, but the call only
   485	        # supports two so make sure that the third argument isn't passed
   486	        # along. This is needed as two-argument __(i)pow__ is allowed and
   487	        # rather common.
   488	        if len(gen.arg_names) == 3:
   489	            gen.arg_names.pop()

   502	def generate_richcompare_wrapper(cl: ClassIR, emitter: Emitter) -> str | None:
   503	    """Generates a wrapper for richcompare dunder methods."""
   504	    # Sort for determinism on Python 3.5
   505	    matches = sorted(name for name in RICHCOMPARE_OPS if cl.has_method(name))
   506	    if not matches:
   507	        return None
   508
   509	    name = f"{DUNDER_PREFIX}_RichCompare_{cl.name_prefix(emitter.names)}"
   510	    emitter.emit_line(
   511	        "static PyObject *{name}(PyObject *obj_lhs, PyObject *obj_rhs, int op) {{".format(
   512	            name=name
   513	        )
   514	    )
   515	    emitter.emit_line("switch (op) {")
   516	    for func in matches:
   517	        emitter.emit_line(f"case {RICHCOMPARE_OPS[func]}: {{")
   518	        method = cl.get_method(func)
   519	        assert method is not None
   520	        generate_wrapper_core(method, emitter, arg_names=["lhs", "rhs"])
   521	        emitter.emit_line("}")
   522	    emitter.emit_line("}")
   523
   524	    emitter.emit_line("Py_INCREF(Py_NotImplemented);")
   525	    emitter.emit_line("return Py_NotImplemented;")
   526
   527	    emitter.emit_line("}")
   528
   529	    return name

   547	def generate_hash_wrapper(cl: ClassIR, fn: FuncIR, emitter: Emitter) -> str:
   548	    """Generates a wrapper for native __hash__ methods."""
   549	    name = f"{DUNDER_PREFIX}{fn.name}{cl.name_prefix(emitter.names)}"
   550	    emitter.emit_line(f"static Py_ssize_t {name}(PyObject *self) {{")
   551	    emitter.emit_line(
   552	        "{}retval = {}{}{}(self);".format(
   553	            emitter.ctype_spaced(fn.ret_type),
   554	            emitter.get_group_prefix(fn.decl),
   555	            NATIVE_PREFIX,
   556	            fn.cname(emitter.names),
   557	        )
   558	    )
   559	    emitter.emit_error_check("retval", fn.ret_type, "return -1;")
   560	    if is_int_rprimitive(fn.ret_type):
   561	        emitter.emit_line("Py_ssize_t val = CPyTagged_AsSsize_t(retval);")
   562	    else:
   563	        emitter.emit_line("Py_ssize_t val = PyLong_AsSsize_t(retval);")
   564	    emitter.emit_dec_ref("retval", fn.ret_type)
   565	    emitter.emit_line("if (PyErr_Occurred()) return -1;")
   566	    # We can't return -1 from a hash function..
   567	    emitter.emit_line("if (val == -1) return -2;")
   568	    emitter.emit_line("return val;")
   569	    emitter.emit_line("}")
   570
   571	    return name
