   377	        self,
   378	        path: os.PathLike[str] | str | None = None,
   379	        lineno: int | None = None,
   380	        firstlineno: int | None = None,
   381	        excludepath: os.PathLike[str] | None = None,
   382	    ) -> Traceback:
   383	        """Return a Traceback instance wrapping part of this Traceback.
   384
   385	        By providing any combination of path, lineno and firstlineno, the
   386	        first frame to start the to-be-returned traceback is determined.
   387
   388	        This allows cutting the first part of a Traceback instance e.g.
   389	        for formatting reasons (removing some uninteresting bits that deal
   390	        with handling of the exception/traceback).
   391	        """
   392	        path_ = None if path is None else os.fspath(path)
   393	        excludepath_ = None if excludepath is None else os.fspath(excludepath)
   394	        for x in self:
   395	            code = x.frame.code

   400	                excludepath is not None

   403	            ):

   406	                continue

   408	                continue

   427	        /,
   428	    ) -> Traceback:
   429	        """Return a Traceback instance with certain items removed.
   430
   431	        If the filter is an `ExceptionInfo`, removes all the ``TracebackEntry``s
   432	        which are hidden (see ishidden() above).
   433
   434	        Otherwise, the filter is a function that gets a single argument, a
   435	        ``TracebackEntry`` instance, and should return True when the item should
   436	        be added to the ``Traceback``, False when not.
   437	        """
   438	        if isinstance(excinfo_or_fn, ExceptionInfo):

   445	        """Return the index of the frame/TracebackEntry where recursion originates if
   446	        appropriate, None if no recursion occurred."""
   447	        cache: dict[tuple[Any, int, int], list[dict[str, Any]]] = {}
   448	        for i, entry in enumerate(self):
   449	            # id for the code.raw is needed to work around
   450	            # the strange metaprogramming in the decorator lib from pypi
   451	            # which generates code objects that have hash/value equality
   452	            # XXX needs a test
   453	            key = entry.frame.code.path, id(entry.frame.code.raw), entry.lineno

   456	            loc = dict(entry.frame.f_locals)

   458	                for otherloc in values:
   459	                    if otherloc == loc:
   460	                        return i
   461	            values.append(loc)
   462	        return None

   465	def stringify_exception(
   466	    exc: BaseException, include_subexception_msg: bool = True
   467	) -> str:
   468	    try:
   469	        notes = getattr(exc, "__notes__", [])
   470	    except KeyError:
   471	        # Workaround for https://github.com/python/cpython/issues/98778 on
   472	        # Python <= 3.9, and some 3.10 and 3.11 patch versions.
   473	        HTTPError = getattr(sys.modules.get("urllib.error", None), "HTTPError", ())
   474	        if sys.version_info < (3, 12) and isinstance(exc, HTTPError):
   475	            notes = []
   476	        else:  # pragma: no cover
   477	            # exception not related to above bug, reraise
   478	            raise
   479	    if not include_subexception_msg and isinstance(exc, BaseExceptionGroup):
   480	        message = exc.message
   481	    else:
   482	        message = str(exc)
   483
   484	    return "\n".join(
   485	        [
   486	            message,
   487	            *notes,
   488	        ]
   489	    )

   783	    def _group_contains(
   784	        self,
   785	        exc_group: BaseExceptionGroup[BaseException],
   786	        expected_exception: EXCEPTION_OR_MORE,
   787	        match: str | re.Pattern[str] | None,
   788	        target_depth: int | None = None,
   789	        current_depth: int = 1,
   790	    ) -> bool:
   791	        """Return `True` if a `BaseExceptionGroup` contains a matching exception."""
   792	        if (target_depth is not None) and (current_depth > target_depth):
   793	            # already descended past the target depth
   794	            return False
   795	        for exc in exc_group.exceptions:
   796	            if isinstance(exc, BaseExceptionGroup):
   797	                if self._group_contains(
   798	                    exc, expected_exception, match, target_depth, current_depth + 1
   799	                ):
   800	                    return True
   801	            if (target_depth is not None) and (current_depth != target_depth):
   802	                # not at the target depth, no match
   803	                continue
   804	            if not isinstance(exc, expected_exception):
   805	                continue
   806	            if match is not None:
   807	                value = stringify_exception(exc)
   808	                if not re.search(match, value):
   809	                    continue
   810	            return True
   811	        return False

   813	    def group_contains(
   814	        self,
   815	        expected_exception: EXCEPTION_OR_MORE,
   816	        *,
   817	        match: str | re.Pattern[str] | None = None,
   818	        depth: int | None = None,
   819	    ) -> bool:
   820	        """Check whether a captured exception group contains a matching exception.
   821
   822	        :param Type[BaseException] | Tuple[Type[BaseException]] expected_exception:
   823	            The expected exception type, or a tuple if one of multiple possible
   824	            exception types are expected.
   825
   826	        :param str | re.Pattern[str] | None match:
   827	            If specified, a string containing a regular expression,
   828	            or a regular expression object, that is tested against the string
   829	            representation of the exception and its `PEP-678 <https://peps.python.org/pep-0678/>` `__notes__`
   830	            using :func:`re.search`.
   831
   832	            To match a literal string that may contain :ref:`special characters
   833	            <re-syntax>`, the pattern can first be escaped with :func:`re.escape`.
   834
   835	        :param Optional[int] depth:
   836	            If `None`, will search for a matching exception at any nesting depth.
   837	            If >= 1, will only match an exception if it's at the specified depth (depth = 1 being
   838	            the exceptions contained within the topmost exception group).
   839
   840	        .. versionadded:: 8.0
   841
   842	        .. warning::
   843	           This helper makes it easy to check for the presence of specific exceptions,
   844	           but it is very bad for checking that the group does *not* contain
   845	           *any other exceptions*.
   846	           You should instead consider using :class:`pytest.RaisesGroup`
   847
   848	        """
   849	        msg = "Captured exception is not an instance of `BaseExceptionGroup`"
   850	        assert isinstance(self.value, BaseExceptionGroup), msg
   851	        msg = "`depth` must be >= 1 if specified"
   852	        assert (depth is None) or (depth >= 1), msg
   853	        return self._group_contains(self.value, expected_exception, match, depth)

   973	    def get_highlight_arrows_for_line(
   974	        self,
   975	        line: str,
   976	        raw_line: str,
   977	        lineno: int | None,
   978	        end_lineno: int | None,
   979	        colno: int | None,
   980	        end_colno: int | None,
   981	    ) -> list[str]:
   982	        """Return characters highlighting a source line.
   983
   984	        Example with colno and end_colno pointing to the bar expression:
   985	                   "foo() + bar()"
   986	        returns    "        ^^^^^"
   987	        """
   988	        if lineno != end_lineno:
   989	            # Don't handle expressions that span multiple lines.
   990	            return []
   991	        if colno is None or end_colno is None:
   992	            # Can't do anything without column information.
   993	            return []
   994
   995	        num_stripped_chars = len(raw_line) - len(line)
   996
   997	        start_char_offset = _byte_offset_to_character_offset(raw_line, colno)
   998	        end_char_offset = _byte_offset_to_character_offset(raw_line, end_colno)
   999	        num_carets = end_char_offset - start_char_offset
  1000	        # If the highlight would span the whole line, it is redundant, don't
  1001	        # show it.
  1002	        if num_carets >= len(line.strip()):
  1003	            return []
  1004
  1005	        highlights = "    "
  1006	        highlights += " " * (start_char_offset - num_stripped_chars + 1)
  1007	        highlights += "^" * num_carets
  1008	        return [highlights]

  1140	    def _truncate_recursive_traceback(
  1141	        self, traceback: Traceback
  1142	    ) -> tuple[Traceback, str | None]:
  1143	        """Truncate the given recursive traceback trying to find the starting
  1144	        point of the recursion.
  1145
  1146	        The detection is done by going through each traceback entry and
  1147	        finding the point in which the locals of the frame are equal to the
  1148	        locals of a previous frame (see ``recursionindex()``).
  1149
  1150	        Handle the situation where the recursion process might raise an
  1151	        exception (for example comparing numpy arrays using equality raises a
  1152	        TypeError), in which case we do our best to warn the user of the
  1153	        error and show a limited traceback.
  1154	        """
  1155	        try:
  1156	            recursionindex = traceback.recursionindex()
  1157	        except Exception as e:
  1158	            max_frames = 10
  1159	            extraline: str | None = (
  1160	                "!!! Recursion error detected, but an error occurred locating the origin of recursion.\n"
  1161	                "  The following exception happened when comparing locals in the stack frame:\n"
  1162	                f"    {type(e).__name__}: {e!s}\n"
  1163	                f"  Displaying first and last {max_frames} stack frames out of {len(traceback)}."
  1164	            )
  1165	            # Type ignored because adding two instances of a List subtype
  1166	            # currently incorrectly has type List instead of the subtype.
  1167	            traceback = traceback[:max_frames] + traceback[-max_frames:]  # type: ignore
  1168	        else:
  1169	            if recursionindex is not None:
  1170	                extraline = "!!! Recursion detected (same locals & position)"
  1171	                traceback = traceback[: recursionindex + 1]
  1172	            else:
  1173	                extraline = None
  1174
  1175	        return traceback, extraline

  1177	    def repr_excinfo(self, excinfo: ExceptionInfo[BaseException]) -> ExceptionChainRepr:
  1178	        repr_chain: list[tuple[ReprTraceback, ReprFileLocation | None, str | None]] = []
  1179	        e: BaseException | None = excinfo.value
  1180	        excinfo_: ExceptionInfo[BaseException] | None = excinfo
  1181	        descr = None
  1182	        seen: set[int] = set()
  1183	        while e is not None and id(e) not in seen:
  1184	            seen.add(id(e))
  1185
  1186	            if excinfo_:
  1187	                # Fall back to native traceback as a temporary workaround until
  1188	                # full support for exception groups added to ExceptionInfo.
  1189	                # See https://github.com/pytest-dev/pytest/issues/9159
  1190	                reprtraceback: ReprTraceback | ReprTracebackNative
  1191	                if isinstance(e, BaseExceptionGroup):
  1192	                    # don't filter any sub-exceptions since they shouldn't have any internal frames
  1193	                    traceback = filter_excinfo_traceback(self.tbfilter, excinfo)
  1194	                    reprtraceback = ReprTracebackNative(
  1195	                        format_exception(
  1196	                            type(excinfo.value),
  1197	                            excinfo.value,
  1198	                            traceback[0]._rawentry,
  1199	                        )
  1200	                    )
  1201	                else:
  1202	                    reprtraceback = self.repr_traceback(excinfo_)
  1203	                reprcrash = excinfo_._getreprcrash()
  1204	            else:
  1205	                # Fallback to native repr if the exception doesn't have a traceback:
  1206	                # ExceptionInfo objects require a full traceback to work.
  1207	                reprtraceback = ReprTracebackNative(format_exception(type(e), e, None))
  1208	                reprcrash = None
  1209	            repr_chain += [(reprtraceback, reprcrash, descr)]
  1210
  1211	            if e.__cause__ is not None and self.chain:
  1212	                e = e.__cause__
  1213	                excinfo_ = ExceptionInfo.from_exception(e) if e.__traceback__ else None
  1214	                descr = "The above exception was the direct cause of the following exception:"
  1215	            elif (
  1216	                e.__context__ is not None and not e.__suppress_context__ and self.chain
  1217	            ):
  1218	                e = e.__context__
  1219	                excinfo_ = ExceptionInfo.from_exception(e) if e.__traceback__ else None
  1220	                descr = "During handling of the above exception, another exception occurred:"
  1221	            else:
  1222	                e = None
  1223	        repr_chain.reverse()
  1224	        return ExceptionChainRepr(repr_chain)

  1348	    def _write_entry_lines(self, tw: TerminalWriter) -> None:
  1349	        """Write the source code portions of a list of traceback entries with syntax highlighting.
  1350
  1351	        Usually entries are lines like these:
  1352
  1353	            "     x = 1"
  1354	            ">    assert x == 2"
  1355	            "E    assert 1 == 2"
  1356
  1357	        This function takes care of rendering the "source" portions of it (the lines without
  1358	        the "E" prefix) using syntax highlighting, taking care to not highlighting the ">"
  1359	        character, as doing so might break line continuations.
  1360	        """
  1361	        if not self.lines:
  1362	            return
  1363
  1364	        if self.style == "value":
  1365	            # Using tw.write instead of tw.line for testing purposes due to TWMock implementation;
  1366	            # lines written with TWMock.line and TWMock._write_source cannot be distinguished
  1367	            # from each other, whereas lines written with TWMock.write are marked with TWMock.WRITE
  1368	            for line in self.lines:
  1369	                tw.write(line)
  1370	                tw.write("\n")
  1371	            return
  1372
  1373	        # separate indents and source lines that are not failures: we want to
  1374	        # highlight the code but not the indentation, which may contain markers
  1375	        # such as ">   assert 0"
  1376	        fail_marker = f"{FormattedExcinfo.fail_marker}   "
  1377	        indent_size = len(fail_marker)
  1378	        indents: list[str] = []
  1379	        source_lines: list[str] = []
  1380	        failure_lines: list[str] = []
  1381	        for index, line in enumerate(self.lines):
  1382	            is_failure_line = line.startswith(fail_marker)
  1383	            if is_failure_line:
  1384	                # from this point on all lines are considered part of the failure
  1385	                failure_lines.extend(self.lines[index:])
  1386	                break
  1387	            else:
  1388	                indents.append(line[:indent_size])
  1389	                source_lines.append(line[indent_size:])
  1390
  1391	        tw._write_source(source_lines, indents)
  1392
  1393	        # failure lines are always completely red and bold
  1394	        for line in failure_lines:
  1395	            tw.line(line, bold=True, red=True)

   341	def _extract_decimal_digits_info(decimal: Decimal) -> tuple[int, int]:
   342	    """Compute the total number of digits and decimal places for a given [`Decimal`][decimal.Decimal] instance.
   343
   344	    This function handles both normalized and non-normalized Decimal instances.
   345	    Example: Decimal('1.230') -> 4 digits, 3 decimal places
   346
   347	    Args:
   348	        decimal (Decimal): The decimal number to analyze.
   349
   350	    Returns:
   351	        tuple[int, int]: A tuple containing the number of decimal places and total digits.
   352
   353	    Though this could be divided into two separate functions, the logic is easier to follow if we couple the computation
   354	    of the number of decimals and digits together.
   355	    """
   356	    try:
   357	        decimal_tuple = decimal.as_tuple()
   358
   359	        assert isinstance(decimal_tuple.exponent, int)
   360
   361	        exponent = decimal_tuple.exponent
   362	        num_digits = len(decimal_tuple.digits)
   363
   364	        if exponent >= 0:
   365	            # A positive exponent adds that many trailing zeros
   366	            # Ex: digit_tuple=(1, 2, 3), exponent=2 -> 12300 -> 0 decimal places, 5 digits
   367	            num_digits += exponent
   368	            decimal_places = 0
   369	        else:
   370	            # If the absolute value of the negative exponent is larger than the
   371	            # number of digits, then it's the same as the number of digits,
   372	            # because it'll consume all the digits in digit_tuple and then
   373	            # add abs(exponent) - len(digit_tuple) leading zeros after the decimal point.
   374	            # Ex: digit_tuple=(1, 2, 3), exponent=-2 -> 1.23 -> 2 decimal places, 3 digits
   375	            # Ex: digit_tuple=(1, 2, 3), exponent=-4 -> 0.0123 -> 4 decimal places, 4 digits
   376	            decimal_places = abs(exponent)
   377	            num_digits = max(num_digits, decimal_places)
   378
   379	        return decimal_places, num_digits
   380	    except (AssertionError, AttributeError):
   381	        raise TypeError(f'Unable to extract decimal digits info from supplied value {decimal}')
