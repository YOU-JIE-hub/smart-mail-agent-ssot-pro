   135	    """Get the closest parent node (including self) which matches the given

   173	@dataclasses.dataclass(frozen=True)
   174	class ParamArgKey:
   175	    """A key for a high-scoped parameter used by an item.
   176
   177	    For use as a hashable key in `reorder_items`. The combination of fields
   178	    is meant to uniquely identify a particular "instance" of a param,
   179	    potentially shared by multiple items in a scope.
   180	    """
   181
   182	    #: The param name.
   183	    argname: str
   184	    param_index: int
   185	    #: For scopes Package, Module, Class, the path to the file (directory in
   186	    #: Package's case) of the package/module/class where the item is defined.
   187	    scoped_item_path: Path | None
   188	    #: For Class scope, the class where the item is defined.
   189	    item_cls: type | None
   190
   191
   192	_V = TypeVar("_V")
   193	OrderedSet = dict[_V, None]
   194
   195
   196	def get_param_argkeys(item: nodes.Item, scope: Scope) -> Iterator[ParamArgKey]:
   197	    """Return all ParamArgKeys for item matching the specified high scope."""
   198	    assert scope is not Scope.Function
   199
   200	    try:
   201	        callspec: CallSpec2 = item.callspec  # type: ignore[attr-defined]
   202	    except AttributeError:
   203	        return
   204
   205	    item_cls = None
   206	    if scope is Scope.Session:
   207	        scoped_item_path = None
   208	    elif scope is Scope.Package:
   209	        # Package key = module's directory.
   210	        scoped_item_path = item.path.parent
   211	    elif scope is Scope.Module:
   212	        scoped_item_path = item.path
   213	    elif scope is Scope.Class:
   214	        scoped_item_path = item.path
   215	        item_cls = item.cls  # type: ignore[attr-defined]
   216	    else:
   217	        assert_never(scope)
   218
   219	    for argname in callspec.indices:
   220	        if callspec._arg2scope[argname] != scope:
   221	            continue
   222	        param_index = callspec.indices[argname]
   223	        yield ParamArgKey(argname, param_index, scoped_item_path, item_cls)

   226	def reorder_items(items: Sequence[nodes.Item]) -> list[nodes.Item]:
   227	    argkeys_by_item: dict[Scope, dict[nodes.Item, OrderedSet[ParamArgKey]]] = {}
   228	    items_by_argkey: dict[Scope, dict[ParamArgKey, OrderedDict[nodes.Item, None]]] = {}
   229	    for scope in HIGH_SCOPES:
   230	        scoped_argkeys_by_item = argkeys_by_item[scope] = {}
   231	        scoped_items_by_argkey = items_by_argkey[scope] = defaultdict(OrderedDict)
   232	        for item in items:
   233	            argkeys = dict.fromkeys(get_param_argkeys(item, scope))
   234	            if argkeys:
   235	                scoped_argkeys_by_item[item] = argkeys
   236	                for argkey in argkeys:
   237	                    scoped_items_by_argkey[argkey][item] = None
   238
   239	    items_set = dict.fromkeys(items)
   240	    return list(
   241	        reorder_items_atscope(
   242	            items_set, argkeys_by_item, items_by_argkey, Scope.Session
   243	        )
   244	    )

   247	def reorder_items_atscope(
   248	    items: OrderedSet[nodes.Item],
   249	    argkeys_by_item: Mapping[Scope, Mapping[nodes.Item, OrderedSet[ParamArgKey]]],
   250	    items_by_argkey: Mapping[
   251	        Scope, Mapping[ParamArgKey, OrderedDict[nodes.Item, None]]
   252	    ],
   253	    scope: Scope,
   254	) -> OrderedSet[nodes.Item]:
   255	    if scope is Scope.Function or len(items) < 3:
   256	        return items
   257
   258	    scoped_items_by_argkey = items_by_argkey[scope]
   259	    scoped_argkeys_by_item = argkeys_by_item[scope]
   260
   261	    ignore: set[ParamArgKey] = set()
   262	    items_deque = deque(items)
   263	    items_done: OrderedSet[nodes.Item] = {}
   264	    while items_deque:
   265	        no_argkey_items: OrderedSet[nodes.Item] = {}
   266	        slicing_argkey = None
   267	        while items_deque:
   268	            item = items_deque.popleft()
   269	            if item in items_done or item in no_argkey_items:
   270	                continue
   271	            argkeys = dict.fromkeys(
   272	                k for k in scoped_argkeys_by_item.get(item, ()) if k not in ignore
   273	            )
   274	            if not argkeys:
   275	                no_argkey_items[item] = None
   276	            else:
   277	                slicing_argkey, _ = argkeys.popitem()
   278	                # We don't have to remove relevant items from later in the
   279	                # deque because they'll just be ignored.
   280	                matching_items = [
   281	                    i for i in scoped_items_by_argkey[slicing_argkey] if i in items
   282	                ]
   283	                for i in reversed(matching_items):
   284	                    items_deque.appendleft(i)
   285	                    # Fix items_by_argkey order.
   286	                    for other_scope in HIGH_SCOPES:
   287	                        other_scoped_items_by_argkey = items_by_argkey[other_scope]
   288	                        for argkey in argkeys_by_item[other_scope].get(i, ()):
   289	                            argkey_dict = other_scoped_items_by_argkey[argkey]
   290	                            if not hasattr(sys, "pypy_version_info"):
   291	                                argkey_dict[i] = None
   292	                                argkey_dict.move_to_end(i, last=False)
   293	                            else:
   294	                                # Work around a bug in PyPy:
   295	                                # https://github.com/pypy/pypy/issues/5257
   296	                                # https://github.com/pytest-dev/pytest/issues/13312
   297	                                bkp = argkey_dict.copy()
   298	                                argkey_dict.clear()
   299	                                argkey_dict[i] = None
   300	                                argkey_dict.update(bkp)
   301	                break
   302	        if no_argkey_items:
   303	            reordered_no_argkey_items = reorder_items_atscope(
   304	                no_argkey_items, argkeys_by_item, items_by_argkey, scope.next_lower()
   305	            )
   306	            items_done.update(reordered_no_argkey_items)
   307	        if slicing_argkey is not None:
   308	            ignore.add(slicing_argkey)
   309	    return items_done

   312	@dataclasses.dataclass(frozen=True)
   313	class FuncFixtureInfo:
   314	    """Fixture-related information for a fixture-requesting item (e.g. test
   315	    function).
   316
   317	    This is used to examine the fixtures which an item requests statically
   318	    (known during collection). This includes autouse fixtures, fixtures
   319	    requested by the `usefixtures` marker, fixtures requested in the function
   320	    parameters, and the transitive closure of these.
   321
   322	    An item may also request fixtures dynamically (using `request.getfixturevalue`);
   323	    these are not reflected here.
   324	    """
   325
   326	    __slots__ = ("argnames", "initialnames", "name2fixturedefs", "names_closure")
   327
   328	    # Fixture names that the item requests directly by function parameters.
   329	    argnames: tuple[str, ...]
   330	    # Fixture names that the item immediately requires. These include
   331	    # argnames + fixture names specified via usefixtures and via autouse=True in
   332	    # fixture definitions.
   333	    initialnames: tuple[str, ...]
   334	    # The transitive closure of the fixture names that the item requires.
   335	    # Note: can't include dynamic dependencies (`request.getfixturevalue` calls).
   336	    names_closure: list[str]
   337	    # A map from a fixture name in the transitive closure to the FixtureDefs
   338	    # matching the name which are applicable to this function.
   339	    # There may be multiple overriding fixtures with the same name. The
   340	    # sequence is ordered from furthest to closes to the function.
   341	    name2fixturedefs: dict[str, Sequence[FixtureDef[Any]]]
   342
   343	    def prune_dependency_tree(self) -> None:
   344	        """Recompute names_closure from initialnames and name2fixturedefs.
   345
   346	        Can only reduce names_closure, which means that the new closure will
   347	        always be a subset of the old one. The order is preserved.
   348
   349	        This method is needed because direct parametrization may shadow some
   350	        of the fixtures that were included in the originally built dependency
   351	        tree. In this way the dependency tree can get pruned, and the closure
   352	        of argnames may get reduced.
   353	        """
   354	        closure: set[str] = set()
   355	        working_set = set(self.initialnames)
   356	        while working_set:
   357	            argname = working_set.pop()
   358	            # Argname may be something not included in the original names_closure,
   359	            # in which case we ignore it. This currently happens with pseudo
   360	            # FixtureDefs which wrap 'get_direct_param_fixture_func(request)'.
   361	            # So they introduce the new dependency 'request' which might have
   362	            # been missing in the original tree (closure).
   363	            if argname not in closure and argname in self.names_closure:
   364	                closure.add(argname)
   365	                if argname in self.name2fixturedefs:
   366	                    working_set.update(self.name2fixturedefs[argname][-1].argnames)
   367
   368	        self.names_closure[:] = sorted(closure, key=self.names_closure.index)

   371	class FixtureRequest(abc.ABC):
   372	    """The type of the ``request`` fixture.
   373
   374	    A request object gives access to the requesting test context and has a
   375	    ``param`` attribute in case the fixture is parametrized.
   376	    """
   377
   378	    def __init__(
   379	        self,
   380	        pyfuncitem: Function,
   381	        fixturename: str | None,
   382	        arg2fixturedefs: dict[str, Sequence[FixtureDef[Any]]],
   383	        fixture_defs: dict[str, FixtureDef[Any]],
   384	        *,
   385	        _ispytest: bool = False,
   386	    ) -> None:
   387	        check_ispytest(_ispytest)
   388	        #: Fixture for which this request is being performed.
   389	        self.fixturename: Final = fixturename
   390	        self._pyfuncitem: Final = pyfuncitem
   391	        # The FixtureDefs for each fixture name requested by this item.
   392	        # Starts from the statically-known fixturedefs resolved during
   393	        # collection. Dynamically requested fixtures (using
   394	        # `request.getfixturevalue("foo")`) are added dynamically.
   395	        self._arg2fixturedefs: Final = arg2fixturedefs
   396	        # The evaluated argnames so far, mapping to the FixtureDef they resolved
   397	        # to.
   398	        self._fixture_defs: Final = fixture_defs
   399	        # Notes on the type of `param`:
   400	        # -`request.param` is only defined in parametrized fixtures, and will raise
   401	        #   AttributeError otherwise. Python typing has no notion of "undefined", so
   402	        #   this cannot be reflected in the type.
   403	        # - Technically `param` is only (possibly) defined on SubRequest, not
   404	        #   FixtureRequest, but the typing of that is still in flux so this cheats.
   405	        # - In the future we might consider using a generic for the param type, but
   406	        #   for now just using Any.
   407	        self.param: Any

   526	    def getfixturevalue(self, argname: str) -> Any:
   527	        """Dynamically run a named fixture function.
   528
   529	        Declaring fixtures via function argument is recommended where possible.
   530	        But if you can only decide whether to use another fixture at test
   531	        setup time, you may use this function to retrieve it inside a fixture
   532	        or test function body.
   533
   534	        This method can be used during the test setup phase or the test run
   535	        phase, but during the test teardown phase a fixture's value may not
   536	        be available.
   537
   538	        :param argname:
   539	            The fixture name.
   540	        :raises pytest.FixtureLookupError:
   541	            If the given fixture could not be found.
   542	        """
   543	        # Note that in addition to the use case described in the docstring,
   544	        # getfixturevalue() is also called by pytest itself during item and fixture
   545	        # setup to evaluate the fixtures that are requested statically
   546	        # (using function parameters, autouse, etc).
   547
   548	        fixturedef = self._get_active_fixturedef(argname)
   549	        assert fixturedef.cached_result is not None, (
   550	            f'The fixture value for "{argname}" is not available.  '
   551	            "This can happen when the fixture has already been torn down."
   552	        )
   553	        return fixturedef.cached_result[0]

   556	        """Yield all SubRequests in the chain, from self up.

   558	        Note: does *not* yield the TopRequest.
   559	        """
   560	        current = self
   561	        while isinstance(current, SubRequest):
   562	            yield current
   563	            current = current._parent_request

   565	    def _get_active_fixturedef(
   566	        self, argname: str
   567	    ) -> FixtureDef[object] | PseudoFixtureDef[object]:
   568	        if argname == "request":
   569	            cached_result = (self, [0], None)
   570	            return PseudoFixtureDef(cached_result, Scope.Function)
   571
   572	        # If we already finished computing a fixture by this name in this item,
   573	        # return it.
   574	        fixturedef = self._fixture_defs.get(argname)
   575	        if fixturedef is not None:
   576	            self._check_scope(fixturedef, fixturedef._scope)
   577	            return fixturedef
   578
   579	        # Find the appropriate fixturedef.
   580	        fixturedefs = self._arg2fixturedefs.get(argname, None)
   581	        if fixturedefs is None:
   582	            # We arrive here because of a dynamic call to
   583	            # getfixturevalue(argname) which was naturally
   584	            # not known at parsing/collection time.
   585	            fixturedefs = self._fixturemanager.getfixturedefs(argname, self._pyfuncitem)
   586	            if fixturedefs is not None:
   587	                self._arg2fixturedefs[argname] = fixturedefs
   588	        # No fixtures defined with this name.
   589	        if fixturedefs is None:
   590	            raise FixtureLookupError(argname, self)
   591	        # The are no fixtures with this name applicable for the function.
   592	        if not fixturedefs:
   593	            raise FixtureLookupError(argname, self)
   594
   595	        # A fixture may override another fixture with the same name, e.g. a
   596	        # fixture in a module can override a fixture in a conftest, a fixture in
   597	        # a class can override a fixture in the module, and so on.
   598	        # An overriding fixture can request its own name (possibly indirectly);
   599	        # in this case it gets the value of the fixture it overrides, one level
   600	        # up.
   601	        # Check how many `argname`s deep we are, and take the next one.
   602	        # `fixturedefs` is sorted from furthest to closest, so use negative
   603	        # indexing to go in reverse.
   604	        index = -1
   605	        for request in self._iter_chain():
   606	            if request.fixturename == argname:
   607	                index -= 1
   608	        # If already consumed all of the available levels, fail.
   609	        if -index > len(fixturedefs):
   610	            raise FixtureLookupError(argname, self)
   611	        fixturedef = fixturedefs[index]
   612
   613	        # Prepare a SubRequest object for calling the fixture.
   614	        try:
   615	            callspec = self._pyfuncitem.callspec
   616	        except AttributeError:
   617	            callspec = None
   618	        if callspec is not None and argname in callspec.params:
   619	            param = callspec.params[argname]
   620	            param_index = callspec.indices[argname]
   621	            # The parametrize invocation scope overrides the fixture's scope.
   622	            scope = callspec._arg2scope[argname]
   623	        else:
   624	            param = NOTSET
   625	            param_index = 0
   626	            scope = fixturedef._scope
   627	            self._check_fixturedef_without_param(fixturedef)
   628	        # The parametrize invocation scope only controls caching behavior while
   629	        # allowing wider-scoped fixtures to keep depending on the parametrized
   630	        # fixture. Scope control is enforced for parametrized fixtures
   631	        # by recreating the whole fixture tree on parameter change.
   632	        # Hence `fixturedef._scope`, not `scope`.
   633	        self._check_scope(fixturedef, fixturedef._scope)
   634	        subrequest = SubRequest(
   635	            self, scope, param, param_index, fixturedef, _ispytest=True
   636	        )
   637
   638	        # Make sure the fixture value is cached, running it if it isn't
   639	        fixturedef.execute(request=subrequest)
   640
   641	        self._fixture_defs[argname] = fixturedef
   642	        return fixturedef
   643
   644	    def _check_fixturedef_without_param(self, fixturedef: FixtureDef[object]) -> None:
   645	        """Check that this request is allowed to execute this fixturedef without
   646	        a param."""
   647	        funcitem = self._pyfuncitem
   648	        has_params = fixturedef.params is not None
   649	        fixtures_not_supported = getattr(funcitem, "nofuncargs", False)
   650	        if has_params and fixtures_not_supported:
   651	            msg = (
   652	                f"{funcitem.name} does not support fixtures, maybe unittest.TestCase subclass?\n"
   653	                f"Node id: {funcitem.nodeid}\n"
   654	                f"Function type: {type(funcitem).__name__}"
   655	            )
   656	            fail(msg, pytrace=False)
   657	        if has_params:
   658	            frame = inspect.stack()[3]
   659	            frameinfo = inspect.getframeinfo(frame[0])
   660	            source_path = absolutepath(frameinfo.filename)
   661	            source_lineno = frameinfo.lineno
   662	            try:
   663	                source_path_str = str(source_path.relative_to(funcitem.config.rootpath))
   664	            except ValueError:
   665	                source_path_str = str(source_path)
   666	            location = getlocation(fixturedef.func, funcitem.config.rootpath)
   667	            msg = (
   668	                "The requested fixture has no parameter defined for test:\n"
   669	                f"    {funcitem.nodeid}\n\n"
   670	                f"Requested fixture '{fixturedef.argname}' defined in:\n"
   671	                f"{location}\n\n"
   672	                f"Requested here:\n"
   673	                f"{source_path_str}:{source_lineno}"
   674	            )
   675	            fail(msg, pytrace=False)
   676
   677	    def _get_fixturestack(self) -> list[FixtureDef[Any]]:
   678	        values = [request._fixturedef for request in self._iter_chain()]
   679	        values.reverse()
   680	        return values

   780	        self,

   790	                self._format_fixturedef_line(fixturedef)
   791	                for fixturedef in self._get_fixturestack()

   793	            requested_fixture = self._format_fixturedef_line(requested_fixturedef)

   796	                f"fixture {argname} with a {self._scope.value} scoped request object. "
   797	                f"Requesting fixture stack:\n{fixture_stack}\n"
   798	                f"Requested fixture:\n{requested_fixture}",
   799	                pytrace=False,
   800	            )

   807	        signature = inspect.signature(factory)
   808	        return f"{path}:{lineno + 1}:  def {factory.__name__}{signature}"

   814	@final
   815	class FixtureLookupError(LookupError):
   816	    """Could not return a requested fixture (missing or invalid)."""
   817
   818	    def __init__(
   819	        self, argname: str | None, request: FixtureRequest, msg: str | None = None
   820	    ) -> None:
   821	        self.argname = argname
   822	        self.request = request
   823	        self.fixturestack = request._get_fixturestack()
   824	        self.msg = msg
   825
   826	    def formatrepr(self) -> FixtureLookupErrorRepr:
   827	        tblines: list[str] = []
   828	        addline = tblines.append
   829	        stack = [self.request._pyfuncitem.obj]
   830	        stack.extend(map(lambda x: x.func, self.fixturestack))
   831	        msg = self.msg
   832	        # This function currently makes an assumption that a non-None msg means we
   833	        # have a non-empty `self.fixturestack`. This is currently true, but if
   834	        # somebody at some point want to extend the use of FixtureLookupError to
   835	        # new cases it might break.
   836	        # Add the assert to make it clearer to developer that this will fail, otherwise
   837	        # it crashes because `fspath` does not get set due to `stack` being empty.
   838	        assert self.msg is None or self.fixturestack, (
   839	            "formatrepr assumptions broken, rewrite it to handle it"
   840	        )
   841	        if msg is not None:
   842	            # The last fixture raise an error, let's present
   843	            # it at the requesting side.
   844	            stack = stack[:-1]
   845	        for function in stack:
   846	            fspath, lineno = getfslineno(function)
   847	            try:
   848	                lines, _ = inspect.getsourcelines(get_real_func(function))
   849	            except (OSError, IndexError, TypeError):
   850	                error_msg = "file %s, line %s: source code not available"
   851	                addline(error_msg % (fspath, lineno + 1))
   852	            else:
   853	                addline(f"file {fspath}, line {lineno + 1}")
   854	                for i, line in enumerate(lines):
   855	                    line = line.rstrip()
   856	                    addline("  " + line)
   857	                    if line.lstrip().startswith("def"):
   858	                        break
   859
   860	        if msg is None:
   861	            fm = self.request._fixturemanager
   862	            available = set()
   863	            parent = self.request._pyfuncitem.parent
   864	            assert parent is not None
   865	            for name, fixturedefs in fm._arg2fixturedefs.items():
   866	                faclist = list(fm._matchfactories(fixturedefs, parent))
   867	                if faclist:
   868	                    available.add(name)
   869	            if self.argname in available:
   870	                msg = (
   871	                    f" recursive dependency involving fixture '{self.argname}' detected"
   872	                )
   873	            else:
   874	                msg = f"fixture '{self.argname}' not found"
   875	            msg += "\n available fixtures: {}".format(", ".join(sorted(available)))
   876	            msg += "\n use 'pytest --fixtures [testpath]' for help on them."
   877
   878	        return FixtureLookupErrorRepr(fspath, lineno, tblines, msg, self.argname)

   915	def call_fixture_func(
   916	    fixturefunc: _FixtureFunc[FixtureValue], request: FixtureRequest, kwargs
   917	) -> FixtureValue:
   918	    if inspect.isgeneratorfunction(fixturefunc):
   919	        fixturefunc = cast(Callable[..., Generator[FixtureValue]], fixturefunc)
   920	        generator = fixturefunc(**kwargs)
   921	        try:
   922	            fixture_result = next(generator)
   923	        except StopIteration:
   924	            raise ValueError(f"{request.fixturename} did not yield a value") from None
   925	        finalizer = functools.partial(_teardown_yield_fixture, fixturefunc, generator)
   926	        request.addfinalizer(finalizer)
   927	    else:
   928	        fixturefunc = cast(Callable[..., FixtureValue], fixturefunc)
   929	        fixture_result = fixturefunc(**kwargs)
   930	    return fixture_result

   933	def _teardown_yield_fixture(fixturefunc, it) -> None:
   934	    """Execute the teardown of a fixture function by advancing the iterator
   935	    after the yield and ensure the iteration ends (if not it means there is
   936	    more than one yield in the function)."""
   937	    try:
   938	        next(it)
   939	    except StopIteration:
   940	        pass
   941	    else:
   942	        fs, lineno = getfslineno(fixturefunc)
   943	        fail(
   944	            f"fixture function has more than one 'yield':\n\n"
   945	            f"{Source(fixturefunc).indent()}\n"
   946	            f"{fs}:{lineno + 1}",
   947	            pytrace=False,
   948	        )

   951	def _eval_scope_callable(
   952	    scope_callable: Callable[[str, Config], _ScopeName],
   953	    fixture_name: str,
   954	    config: Config,
   955	) -> _ScopeName:
   956	    try:
   957	        # Type ignored because there is no typing mechanism to specify
   958	        # keyword arguments, currently.
   959	        result = scope_callable(fixture_name=fixture_name, config=config)  # type: ignore[call-arg]
   960	    except Exception as e:
   961	        raise TypeError(
   962	            f"Error evaluating {scope_callable} while defining fixture '{fixture_name}'.\n"
   963	            "Expected a function with the signature (*, fixture_name, config)"
   964	        ) from e
   965	    if not isinstance(result, str):
   966	        fail(
   967	            f"Expected {scope_callable} to return a 'str' while defining fixture '{fixture_name}', but it returned:\n"
   968	            f"{result!r}",
   969	            pytrace=False,
   970	        )
   971	    return result

   974	@final
   975	class FixtureDef(Generic[FixtureValue]):
   976	    """A container for a fixture definition.
   977
   978	    Note: At this time, only explicitly documented fields and methods are
   979	    considered public stable API.
   980	    """
   981
   982	    def __init__(
   983	        self,
   984	        config: Config,
   985	        baseid: str | None,
   986	        argname: str,
   987	        func: _FixtureFunc[FixtureValue],
   988	        scope: Scope | _ScopeName | Callable[[str, Config], _ScopeName] | None,
   989	        params: Sequence[object] | None,
   990	        ids: tuple[object | None, ...] | Callable[[Any], object | None] | None = None,
   991	        *,
   992	        _ispytest: bool = False,
   993	        # only used in a deprecationwarning msg, can be removed in pytest9
   994	        _autouse: bool = False,
   995	    ) -> None:
   996	        check_ispytest(_ispytest)
   997	        # The "base" node ID for the fixture.
   998	        #
   999	        # This is a node ID prefix. A fixture is only available to a node (e.g.
  1000	        # a `Function` item) if the fixture's baseid is a nodeid of a parent of
  1001	        # node.
  1002	        #
  1003	        # For a fixture found in a Collector's object (e.g. a `Module`s module,
  1004	        # a `Class`'s class), the baseid is the Collector's nodeid.
  1005	        #
  1006	        # For a fixture found in a conftest plugin, the baseid is the conftest's
  1007	        # directory path relative to the rootdir.
  1008	        #
  1009	        # For other plugins, the baseid is the empty string (always matches).
  1010	        self.baseid: Final = baseid or ""
  1011	        # Whether the fixture was found from a node or a conftest in the
  1012	        # collection tree. Will be false for fixtures defined in non-conftest
  1013	        # plugins.
  1014	        self.has_location: Final = baseid is not None
  1015	        # The fixture factory function.
  1016	        self.func: Final = func
  1017	        # The name by which the fixture may be requested.
  1018	        self.argname: Final = argname
  1019	        if scope is None:
  1020	            scope = Scope.Function
  1021	        elif callable(scope):
  1022	            scope = _eval_scope_callable(scope, argname, config)
  1023	        if isinstance(scope, str):
  1024	            scope = Scope.from_user(
  1025	                scope, descr=f"Fixture '{func.__name__}'", where=baseid
  1026	            )
  1027	        self._scope: Final = scope
  1028	        # If the fixture is directly parametrized, the parameter values.
  1029	        self.params: Final = params
  1030	        # If the fixture is directly parametrized, a tuple of explicit IDs to
  1031	        # assign to the parameter values, or a callable to generate an ID given
  1032	        # a parameter value.
  1033	        self.ids: Final = ids
  1034	        # The names requested by the fixtures.
  1035	        self.argnames: Final = getfuncargnames(func, name=argname)
  1036	        # If the fixture was executed, the current value of the fixture.
  1037	        # Can change if the fixture is executed with different parameters.
  1038	        self.cached_result: _FixtureCachedResult[FixtureValue] | None = None
  1039	        self._finalizers: Final[list[Callable[[], object]]] = []
  1040
  1041	        # only used to emit a deprecationwarning, can be removed in pytest9
  1042	        self._autouse = _autouse
  1043
  1044	    @property
  1045	    def scope(self) -> _ScopeName:
  1046	        """Scope string, one of "function", "class", "module", "package", "session"."""
  1047	        return self._scope.value
  1048
  1049	    def addfinalizer(self, finalizer: Callable[[], object]) -> None:
  1050	        self._finalizers.append(finalizer)
  1051
  1052	    def finish(self, request: SubRequest) -> None:
  1053	        exceptions: list[BaseException] = []
  1054	        while self._finalizers:
  1055	            fin = self._finalizers.pop()
  1056	            try:
  1057	                fin()
  1058	            except BaseException as e:
  1059	                exceptions.append(e)
  1060	        node = request.node
  1061	        node.ihook.pytest_fixture_post_finalizer(fixturedef=self, request=request)
  1062	        # Even if finalization fails, we invalidate the cached fixture
  1063	        # value and remove all finalizers because they may be bound methods
  1064	        # which will keep instances alive.
  1065	        self.cached_result = None
  1066	        self._finalizers.clear()
  1067	        if len(exceptions) == 1:
  1068	            raise exceptions[0]
  1069	        elif len(exceptions) > 1:
  1070	            msg = f'errors while tearing down fixture "{self.argname}" of {node}'
  1071	            raise BaseExceptionGroup(msg, exceptions[::-1])
  1072
  1073	    def execute(self, request: SubRequest) -> FixtureValue:
  1074	        """Return the value of this fixture, executing it if not cached."""
  1075	        # Ensure that the dependent fixtures requested by this fixture are loaded.
  1076	        # This needs to be done before checking if we have a cached value, since
  1077	        # if a dependent fixture has their cache invalidated, e.g. due to
  1078	        # parametrization, they finalize themselves and fixtures depending on it
  1079	        # (which will likely include this fixture) setting `self.cached_result = None`.
  1080	        # See #4871
  1081	        requested_fixtures_that_should_finalize_us = []
  1082	        for argname in self.argnames:
  1083	            fixturedef = request._get_active_fixturedef(argname)
  1084	            # Saves requested fixtures in a list so we later can add our finalizer
  1085	            # to them, ensuring that if a requested fixture gets torn down we get torn
  1086	            # down first. This is generally handled by SetupState, but still currently
  1087	            # needed when this fixture is not parametrized but depends on a parametrized
  1088	            # fixture.
  1089	            if not isinstance(fixturedef, PseudoFixtureDef):
  1090	                requested_fixtures_that_should_finalize_us.append(fixturedef)
  1091
  1092	        # Check for (and return) cached value/exception.
  1093	        if self.cached_result is not None:
  1094	            request_cache_key = self.cache_key(request)
  1095	            cache_key = self.cached_result[1]
  1096	            try:
  1097	                # Attempt to make a normal == check: this might fail for objects
  1098	                # which do not implement the standard comparison (like numpy arrays -- #6497).
  1099	                cache_hit = bool(request_cache_key == cache_key)
  1100	            except (ValueError, RuntimeError):
  1101	                # If the comparison raises, use 'is' as fallback.
  1102	                cache_hit = request_cache_key is cache_key
  1103
  1104	            if cache_hit:
  1105	                if self.cached_result[2] is not None:
  1106	                    exc, exc_tb = self.cached_result[2]
  1107	                    raise exc.with_traceback(exc_tb)
  1108	                else:
  1109	                    result = self.cached_result[0]
  1110	                    return result
  1111	            # We have a previous but differently parametrized fixture instance
  1112	            # so we need to tear it down before creating a new one.
  1113	            self.finish(request)
  1114	            assert self.cached_result is None
  1115
  1116	        # Add finalizer to requested fixtures we saved previously.
  1117	        # We make sure to do this after checking for cached value to avoid
  1118	        # adding our finalizer multiple times. (#12135)
  1119	        finalizer = functools.partial(self.finish, request=request)
  1120	        for parent_fixture in requested_fixtures_that_should_finalize_us:
  1121	            parent_fixture.addfinalizer(finalizer)
  1122
  1123	        ihook = request.node.ihook
  1124	        try:
  1125	            # Setup the fixture, run the code in it, and cache the value
  1126	            # in self.cached_result
  1127	            result = ihook.pytest_fixture_setup(fixturedef=self, request=request)
  1128	        finally:
  1129	            # schedule our finalizer, even if the setup failed
  1130	            request.node.addfinalizer(finalizer)
  1131
  1132	        return result
  1133
  1134	    def cache_key(self, request: SubRequest) -> object:
  1135	        return getattr(request, "param", None)
  1136
  1137	    def __repr__(self) -> str:
  1138	        return f"<FixtureDef argname={self.argname!r} scope={self.scope!r} baseid={self.baseid!r}>"

  1142	    fixturedef: FixtureDef[FixtureValue], request: FixtureRequest

  1147	    # The fixture function needs to be bound to the actual
  1148	    # request.instance so that code working with "fixturedef" behaves

  1150	    instance = request.instance
  1151	    if instance is not None:

  1166	    fixturedef: FixtureDef[FixtureValue], request: SubRequest
  1167	) -> FixtureValue:
  1168	    """Execution of fixture setup."""
  1169	    kwargs = {}
  1170	    for argname in fixturedef.argnames:
  1171	        kwargs[argname] = request.getfixturevalue(argname)
  1172
  1173	    fixturefunc = resolve_fixture_function(fixturedef, request)
  1174	    my_cache_key = fixturedef.cache_key(request)
  1175
  1176	    if inspect.isasyncgenfunction(fixturefunc) or inspect.iscoroutinefunction(
  1177	        fixturefunc
  1178	    ):
  1179	        auto_str = " with autouse=True" if fixturedef._autouse else ""
  1180
  1181	        warnings.warn(
  1182	            PytestRemovedIn9Warning(
  1183	                f"{request.node.name!r} requested an async fixture "
  1184	                f"{request.fixturename!r}{auto_str}, with no plugin or hook that "
  1185	                "handled it. This is usually an error, as pytest does not natively "
  1186	                "support it. "
  1187	                "This will turn into an error in pytest 9.\n"
  1188	                "See: https://docs.pytest.org/en/stable/deprecations.html#sync-test-depending-on-async-fixture"
  1189	            ),
  1190	            # no stacklevel will point at users code, so we just point here
  1191	            stacklevel=1,
  1192	        )
  1193
  1194	    try:
  1195	        result = call_fixture_func(fixturefunc, request, kwargs)
  1196	    except TEST_OUTCOME as e:

  1198	            # The test requested a fixture which caused a skip.
  1199	            # Don't show the fixture as the skip location, as then the user
  1200	            # wouldn't know which test skipped.

  1202	        fixturedef.cached_result = (None, my_cache_key, (e, e.__traceback__))
  1203	        raise

  1322	def fixture(
  1323	    fixture_function: FixtureFunction | None = None,
  1324	    *,
  1325	    scope: _ScopeName | Callable[[str, Config], _ScopeName] = "function",
  1326	    params: Iterable[object] | None = None,
  1327	    autouse: bool = False,
  1328	    ids: Sequence[object | None] | Callable[[Any], object | None] | None = None,
  1329	    name: str | None = None,
  1330	) -> FixtureFunctionMarker | FixtureFunctionDefinition:
  1331	    """Decorator to mark a fixture factory function.
  1332
  1333	    This decorator can be used, with or without parameters, to define a
  1334	    fixture function.
  1335
  1336	    The name of the fixture function can later be referenced to cause its
  1337	    invocation ahead of running tests: test modules or classes can use the
  1338	    ``pytest.mark.usefixtures(fixturename)`` marker.
  1339
  1340	    Test functions can directly use fixture names as input arguments in which
  1341	    case the fixture instance returned from the fixture function will be
  1342	    injected.
  1343
  1344	    Fixtures can provide their values to test functions using ``return`` or
  1345	    ``yield`` statements. When using ``yield`` the code block after the
  1346	    ``yield`` statement is executed as teardown code regardless of the test
  1347	    outcome, and must yield exactly once.
  1348
  1349	    :param scope:
  1350	        The scope for which this fixture is shared; one of ``"function"``
  1351	        (default), ``"class"``, ``"module"``, ``"package"`` or ``"session"``.
  1352
  1353	        This parameter may also be a callable which receives ``(fixture_name, config)``
  1354	        as parameters, and must return a ``str`` with one of the values mentioned above.
  1355
  1356	        See :ref:`dynamic scope` in the docs for more information.
  1357
  1358	    :param params:
  1359	        An optional list of parameters which will cause multiple invocations
  1360	        of the fixture function and all of the tests using it. The current
  1361	        parameter is available in ``request.param``.
  1362
  1363	    :param autouse:
  1364	        If True, the fixture func is activated for all tests that can see it.
  1365	        If False (the default), an explicit reference is needed to activate
  1366	        the fixture.
  1367
  1368	    :param ids:
  1369	        Sequence of ids each corresponding to the params so that they are
  1370	        part of the test id. If no ids are provided they will be generated
  1371	        automatically from the params.
  1372
  1373	    :param name:
  1374	        The name of the fixture. This defaults to the name of the decorated
  1375	        function. If a fixture is used in the same module in which it is
  1376	        defined, the function name of the fixture will be shadowed by the
  1377	        function arg that requests the fixture; one way to resolve this is to
  1378	        name the decorated function ``fixture_<fixturename>`` and then use
  1379	        ``@pytest.fixture(name='<fixturename>')``.
  1380	    """
  1381	    fixture_marker = FixtureFunctionMarker(
  1382	        scope=scope,
  1383	        params=tuple(params) if params is not None else None,
  1384	        autouse=autouse,
  1385	        ids=None if ids is None else ids if callable(ids) else tuple(ids),
  1386	        name=name,
  1387	        _ispytest=True,
  1388	    )
  1389
  1390	    # Direct decoration.
  1391	    if fixture_function:
  1392	        return fixture_marker(fixture_function)
  1393
  1394	    return fixture_marker

  1499	class FixtureManager:
  1500	    """pytest fixture definitions and information is stored and managed
  1501	    from this class.
  1502
  1503	    During collection fm.parsefactories() is called multiple times to parse
  1504	    fixture function definitions into FixtureDef objects and internal
  1505	    data structures.
  1506
  1507	    During collection of test functions, metafunc-mechanics instantiate
  1508	    a FuncFixtureInfo object which is cached per node/func-name.
  1509	    This FuncFixtureInfo object is later retrieved by Function nodes
  1510	    which themselves offer a fixturenames attribute.
  1511
  1512	    The FuncFixtureInfo object holds information about fixtures and FixtureDefs
  1513	    relevant for a particular function. An initial list of fixtures is
  1514	    assembled like this:
  1515
  1516	    - ini-defined usefixtures
  1517	    - autouse-marked fixtures along the collection chain up from the function
  1518	    - usefixtures markers at module/class/function level
  1519	    - test function funcargs
  1520
  1521	    Subsequently the funcfixtureinfo.fixturenames attribute is computed
  1522	    as the closure of the fixtures needed to setup the initial fixtures,
  1523	    i.e. fixtures needed by fixture functions themselves are appended
  1524	    to the fixturenames list.
  1525
  1526	    Upon the test-setup phases all fixturenames are instantiated, retrieved
  1527	    by a lookup of their FuncFixtureInfo.
  1528	    """
  1529
  1530	    def __init__(self, session: Session) -> None:
  1531	        self.session = session
  1532	        self.config: Config = session.config
  1533	        # Maps a fixture name (argname) to all of the FixtureDefs in the test
  1534	        # suite/plugins defined with this name. Populated by parsefactories().
  1535	        # TODO: The order of the FixtureDefs list of each arg is significant,
  1536	        #       explain.
  1537	        self._arg2fixturedefs: Final[dict[str, list[FixtureDef[Any]]]] = {}
  1538	        self._holderobjseen: Final[set[object]] = set()
  1539	        # A mapping from a nodeid to a list of autouse fixtures it defines.
  1540	        self._nodeid_autousenames: Final[dict[str, list[str]]] = {
  1541	            "": self.config.getini("usefixtures"),
  1542	        }
  1543	        session.config.pluginmanager.register(self, "funcmanage")

  1545	    def getfixtureinfo(
  1546	        self,

  1548	        func: Callable[..., object] | None,
  1549	        cls: type | None,
  1550	    ) -> FuncFixtureInfo:
  1551	        """Calculate the :class:`FuncFixtureInfo` for an item.
  1552
  1553	        If ``func`` is None, or if the item sets an attribute
  1554	        ``nofuncargs = True``, then ``func`` is not examined at all.
  1555
  1556	        :param node:
  1557	            The item requesting the fixtures.
  1558	        :param func:
  1559	            The item's function.
  1560	        :param cls:

  1574	            parentnode=node,

  1793

  1795	        object), the node's object is traversed and the node's nodeid is used to
  1796	        determine the fixtures' visibility. `nodeid` must not be specified in
  1797	        this case.

  1801	        visibility. `nodeid` must be specified in this case; None and "" mean
  1802	        total visibility.

  1805	            holderobj = node_or_obj

  1808	            holderobj = cast(object, node_or_obj.obj)  # type: ignore[attr-defined]

  1815	        if not safe_isclass(holderobj) and not isinstance(holderobj, types.ModuleType):

  1823	            # access below can raise. safe_getattr() ignores such exceptions.

   130	        sys.stderr.write(f"writing pytest debug information to {path}\n")
   131
   132	        def unset_tracing() -> None:
   133	            debugfile.close()
   134	            sys.stderr.write(f"wrote pytest debug information to {debugfile.name}\n")
   135	            config.trace.root.setwriter(None)
   136	            undo_tracing()
   137
   138	        config.add_cleanup(unset_tracing)
   139
   140	    return config

   156	def pytest_cmdline_main(config: Config) -> int | ExitCode | None:
   157	    if config.option.version > 0:
   158	        showversion(config)
   159	        return 0
   160	    elif config.option.help:
   161	        config._do_configure()
   162	        showhelp(config)
   163	        config._ensure_unconfigure()
   164	        return 0
   165	    return None

   168	def showhelp(config: Config) -> None:
   169	    import textwrap
   170
   171	    reporter: TerminalReporter | None = config.pluginmanager.get_plugin(
   172	        "terminalreporter"
   173	    )
   174	    assert reporter is not None
   175	    tw = reporter._tw
   176	    tw.write(config._parser.optparser.format_help())
   177	    tw.line()
   178	    tw.line(
   179	        "[pytest] ini-options in the first "
   180	        "pytest.ini|tox.ini|setup.cfg|pyproject.toml file found:"
   181	    )
   182	    tw.line()
   183
   184	    columns = tw.fullwidth  # costly call
   185	    indent_len = 24  # based on argparse's max_help_position=24
   186	    indent = " " * indent_len
   187	    for name in config._parser._ininames:
   188	        help, type, default = config._parser._inidict[name]
   189	        if type is None:
   190	            type = "string"
   191	        if help is None:
   192	            raise TypeError(f"help argument cannot be None for {name}")
   193	        spec = f"{name} ({type}):"
   194	        tw.write(f"  {spec}")
   195	        spec_len = len(spec)
   196	        if spec_len > (indent_len - 3):
   197	            # Display help starting at a new line.
   198	            tw.line()
   199	            helplines = textwrap.wrap(
   200	                help,
   201	                columns,
   202	                initial_indent=indent,
   203	                subsequent_indent=indent,
   204	                break_on_hyphens=False,
   205	            )
   206
   207	            for line in helplines:
   208	                tw.line(line)
   209	        else:
   210	            # Display help starting after the spec, following lines indented.
   211	            tw.write(" " * (indent_len - spec_len - 2))
   212	            wrapped = textwrap.wrap(help, columns - indent_len, break_on_hyphens=False)
   213
   214	            if wrapped:
   215	                tw.line(wrapped[0])
   216	                for line in wrapped[1:]:
   217	                    tw.line(indent + line)
   218
   219	    tw.line()
   220	    tw.line("Environment variables:")
   221	    vars = [
   222	        (
   223	            "CI",
   224	            "When set (regardless of value), pytest knows it is running in a "
   225	            "CI process and does not truncate summary info",
   226	        ),
   227	        ("BUILD_NUMBER", "Equivalent to CI")
