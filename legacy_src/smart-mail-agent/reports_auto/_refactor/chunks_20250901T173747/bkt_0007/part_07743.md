   262	def is_debugging(trace_func=None):
   263	    """Detect if a debugging session is in progress.
   264
   265	    This looks at both pytest's builtin pdb support as well as
   266	    externally installed debuggers using some heuristics.
   267
   268	     This is done by checking if either of the following conditions is
   269	     true:
   270
   271	     1. Examines the trace function to see if the module it originates
   272	        from is in KNOWN_DEBUGGING_MODULES.
   273	     2. Check is SUPPRESS_TIMEOUT is set to True.
   274
   275	    :param trace_func: the current trace function, if not given will use
   276	        sys.gettrace(). Used to unit-test this function.
   277	    """
   278	    global SUPPRESS_TIMEOUT, KNOWN_DEBUGGING_MODULES
   279	    if SUPPRESS_TIMEOUT:
   280	        return True
   281	    if trace_func is None:
   282	        trace_func = sys.gettrace()
   283	    trace_module = None
   284	    if trace_func:
   285	        trace_module = inspect.getmodule(trace_func) or inspect.getmodule(
   286	            trace_func.__class__
   287	        )
   288	    if trace_module:
   289	        parts = trace_module.__name__.split(".")
   290	        for name in KNOWN_DEBUGGING_MODULES:
   291	            if any(part.startswith(name) for part in parts):
   292	                return True
   293
   294	    # For 3.12, sys.monitoring is used for tracing. Check if any debugger has been registered.
   295	    if hasattr(sys, "monitoring"):
   296	        return sys.monitoring.get_tool(sys.monitoring.DEBUGGER_ID) != None
   297	    return False

   303	@pytest.hookimpl(trylast=True)
   304	def pytest_timeout_set_timer(item, settings):
   305	    """Setup up a timeout trigger and handler."""
   306	    timeout_method = settings.method
   307	    if (
   308	        timeout_method == "signal"
   309	        and threading.current_thread() is not threading.main_thread()
   310	    ):
   311	        timeout_method = "thread"
   312
   313	    if timeout_method == "signal":
   314
   315	        def handler(signum, frame):
   316	            __tracebackhide__ = True
   317	            timeout_sigalrm(item, settings)
   318
   319	        def cancel():
   320	            signal.setitimer(signal.ITIMER_REAL, 0)
   321	            signal.signal(signal.SIGALRM, signal.SIG_DFL)
   322
   323	        item.cancel_timeout = cancel
   324	        signal.signal(signal.SIGALRM, handler)
   325	        signal.setitimer(signal.ITIMER_REAL, settings.timeout)
   326	    elif timeout_method == "thread":
   327	        timer = threading.Timer(settings.timeout, timeout_timer, (item, settings))
   328	        timer.name = "%s %s" % (__name__, item.nodeid)
   329
   330	        def cancel():
   331	            timer.cancel()
   332	            timer.join()
   333
   334	        item.cancel_timeout = cancel
   335	        timer.start()
   336	    return True

   412	def _parse_marker(marker):
   413	    """Return (timeout, method) tuple from marker.
   414
   415	    Either could be None.  The values are not interpreted, so
   416	    could still be bogus and even the wrong type.
   417	    """
   418	    if not marker.args and not marker.kwargs:
   419	        raise TypeError("Timeout marker must have at least one argument")
   420	    timeout = method = func_only = NOTSET = object()
   421	    for kw, val in marker.kwargs.items():
   422	        if kw == "timeout":
   423	            timeout = val
   424	        elif kw == "method":
   425	            method = val
   426	        elif kw == "func_only":
   427	            func_only = val
   428	        else:
   429	            raise TypeError("Invalid keyword argument for timeout marker: %s" % kw)
   430	    if len(marker.args) >= 1 and timeout is not NOTSET:
   431	        raise TypeError("Multiple values for timeout argument of timeout marker")
   432	    elif len(marker.args) >= 1:
   433	        timeout = marker.args[0]
   434	    if len(marker.args) >= 2 and method is not NOTSET:
   435	        raise TypeError("Multiple values for method argument of timeout marker")
   436	    elif len(marker.args) >= 2:
   437	        method = marker.args[1]
   438	    if len(marker.args) > 2:
   439	        raise TypeError("Too many arguments for timeout marker")
   440	    if timeout is NOTSET:
   441	        timeout = None
   442	    if method is NOTSET:
   443	        method = None
   444	    if func_only is NOTSET:
   445	        func_only = None
   446	    return Settings(timeout, method, func_only, None)

   505	def timeout_timer(item, settings):
   506	    """Dump stack of threads and call os._exit().
   507
   508	    This disables the capturemanager and dumps stdout and stderr.
   509	    Then the stacks are dumped and os._exit(1) is called.
   510	    """
   511	    if not settings.disable_debugger_detection and is_debugging():
   512	        return
   513	    terminal = item.config.get_terminal_writer()
   514	    try:
   515	        capman = item.config.pluginmanager.getplugin("capturemanager")
   516	        if capman:
   517	            capman.suspend_global_capture(item)
   518	            stdout, stderr = capman.read_global_capture()
   519	        else:
   520	            stdout, stderr = None, None
   521	        terminal.sep("+", title="Timeout")
   522	        caplog = item.config.pluginmanager.getplugin("_capturelog")
   523	        if caplog and hasattr(item, "capturelog_handler"):
   524	            log = item.capturelog_handler.stream.getvalue()
   525	            if log:
   526	                terminal.sep("~", title="Captured log")
   527	                terminal.write(log)
   528	        if stdout:
   529	            terminal.sep("~", title="Captured stdout")
   530	            terminal.write(stdout)
   531	        if stderr:
   532	            terminal.sep("~", title="Captured stderr")
   533	            terminal.write(stderr)
   534	        dump_stacks(terminal)
   535	        terminal.sep("+", title="Timeout")
   536	    except Exception:
   537	        traceback.print_exc()
   538	    finally:
   539	        terminal.flush()
   540	        sys.stdout.flush()
   541	        sys.stderr.flush()
   542	        os._exit(1)

   545	def dump_stacks(terminal):
   546	    """Dump the stacks of all threads except the current thread."""
   547	    current_ident = threading.current_thread().ident
   548	    for thread_ident, frame in sys._current_frames().items():
   549	        if thread_ident == current_ident:
   550	            continue
   551	        for t in threading.enumerate():
   552	            if t.ident == thread_ident:
   553	                thread_name = t.name
   554	                break
   555	        else:
   556	            thread_name = "<unknown>"
   557	        terminal.sep("~", title="Stack of %s (%s)" % (thread_name, thread_ident))
   558	        terminal.write("".join(traceback.format_stack(frame)))

   173	class Code128(MultiWidthBarcode):
   174	    """
   175	    Code 128 is a very compact symbology that can encode the entire
   176	    128 character ASCII set, plus 4 special control codes,
   177	    (FNC1-FNC4, expressed in the input string as \xf1 to \xf4).
   178	    Code 128 can also encode digits at double density (2 per byte)
   179	    and has a mandatory checksum.  Code 128 is well supported and
   180	    commonly used -- for example, by UPS for tracking labels.
   181
   182	    Because of these qualities, Code 128 is probably the best choice
   183	    for a linear symbology today (assuming you have a choice).
   184
   185	    Options that may be passed to constructor:
   186
   187	        value (int, or numeric string. required.):
   188	            The value to encode.
   189
   190	        barWidth (float, default .0075):
   191	            X-Dimension, or width of the smallest element
   192	            Minumum is .0075 inch (7.5 mils).
   193
   194	        barHeight (float, see default below):
   195	            Height of the symbol.  Default is the height of the two
   196	            bearer bars (if they exist) plus the greater of .25 inch
   197	            or .15 times the symbol's length.
   198
   199	        quiet (bool, default 1):
   200	            Wether to include quiet zones in the symbol.
   201
   202	        lquiet (float, see default below):
   203	            Quiet zone size to left of code, if quiet is true.
   204	            Default is the greater of .25 inch, or 10 barWidth
   205
   206	        rquiet (float, defaults as above):
   207	            Quiet zone size to right left of code, if quiet is true.
   208
   209	    Sources of Information on Code 128:
   210
   211	    http://www.semiconductor.agilent.com/barcode/sg/Misc/code_128.html
   212	    http://www.adams1.com/pub/russadam/128code.html
   213	    http://www.barcodeman.com/c128.html
   214
   215	    Official Spec, "ANSI/AIM BC4-1999, ISS" is available for US$45 from
   216	    http://www.aimglobal.org/aimstore/
