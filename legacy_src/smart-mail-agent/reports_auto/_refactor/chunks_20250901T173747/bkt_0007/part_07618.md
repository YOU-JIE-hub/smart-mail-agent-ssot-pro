   161	class ErrorWatcher:
   162	    """Context manager that can be used to keep track of new errors recorded
   163	    around a given operation.
   164
   165	    Errors maintain a stack of such watchers. The handler is called starting
   166	    at the top of the stack, and is propagated down the stack unless filtered
   167	    out by one of the ErrorWatcher instances.
   168	    """
   169
   170	    # public attribute for the special treatment of `reveal_type` by
   171	    # `MessageBuilder.reveal_type`:
   172	    filter_revealed_type: bool
   173
   174	    def __init__(
   175	        self,
   176	        errors: Errors,
   177	        *,
   178	        filter_errors: bool | Callable[[str, ErrorInfo], bool] = False,
   179	        save_filtered_errors: bool = False,
   180	        filter_deprecated: bool = False,
   181	        filter_revealed_type: bool = False,
   182	    ) -> None:
   183	        self.errors = errors
   184	        self._has_new_errors = False
   185	        self._filter = filter_errors
   186	        self._filter_deprecated = filter_deprecated
   187	        self.filter_revealed_type = filter_revealed_type
   188	        self._filtered: list[ErrorInfo] | None = [] if save_filtered_errors else None
   189
   190	    def __enter__(self) -> Self:
   191	        self.errors._watchers.append(self)
   192	        return self
   193
   194	    def __exit__(self, exc_type: object, exc_val: object, exc_tb: object) -> Literal[False]:
   195	        last = self.errors._watchers.pop()
   196	        assert last == self
   197	        return False
   198
   199	    def on_error(self, file: str, info: ErrorInfo) -> bool:
   200	        """Handler called when a new error is recorded.
   201
   202	        The default implementation just sets the has_new_errors flag
   203
   204	        Return True to filter out the error, preventing it from being seen by other
   205	        ErrorWatcher further down the stack and from being recorded by Errors
   206	        """
   207	        if info.code == codes.DEPRECATED:
   208	            # Deprecated is not a type error, so it is handled on opt-in basis here.
   209	            return self._filter_deprecated
   210
   211	        self._has_new_errors = True
   212	        if isinstance(self._filter, bool):
   213	            should_filter = self._filter
   214	        elif callable(self._filter):
   215	            should_filter = self._filter(file, info)
   216	        else:
   217	            raise AssertionError(f"invalid error filter: {type(self._filter)}")
   218	        if should_filter and self._filtered is not None:
   219	            self._filtered.append(info)
   220
   221	        return should_filter
   222
   223	    def has_new_errors(self) -> bool:
   224	        return self._has_new_errors
   225
   226	    def filtered_errors(self) -> list[ErrorInfo]:
   227	        assert self._filtered is not None
   228	        return self._filtered

   231	class IterationDependentErrors:
   232	    """An `IterationDependentErrors` instance serves to collect the `unreachable`,
   233	    `redundant-expr`, and `redundant-casts` errors, as well as the revealed types,
   234	    handled by the individual `IterationErrorWatcher` instances sequentially applied to
   235	    the same code section."""
   236
   237	    # One set of `unreachable`, `redundant-expr`, and `redundant-casts` errors per
   238	    # iteration step.  Meaning of the tuple items: ErrorCode, message, line, column,
   239	    # end_line, end_column.
   240	    uselessness_errors: list[set[tuple[ErrorCode, str, int, int, int, int]]]
   241
   242	    # One set of unreachable line numbers per iteration step.  Not only the lines where
   243	    # the error report occurs but really all unreachable lines.
   244	    unreachable_lines: list[set[int]]
   245
   246	    # One list of revealed types for each `reveal_type` statement.  Each created list
   247	    # can grow during the iteration.  Meaning of the tuple items: line, column,
   248	    # end_line, end_column:
   249	    revealed_types: dict[tuple[int, int, int | None, int | None], list[Type]]
   250
   251	    def __init__(self) -> None:
   252	        self.uselessness_errors = []
   253	        self.unreachable_lines = []
   254	        self.revealed_types = defaultdict(list)
   255
   256	    def yield_uselessness_error_infos(self) -> Iterator[tuple[str, Context, ErrorCode]]:
   257	        """Report only those `unreachable`, `redundant-expr`, and `redundant-casts`
   258	        errors that could not be ruled out in any iteration step."""
   259
   260	        persistent_uselessness_errors = set()
   261	        for candidate in set(chain(*self.uselessness_errors)):
   262	            if all(
   263	                (candidate in errors) or (candidate[2] in lines)
   264	                for errors, lines in zip(self.uselessness_errors, self.unreachable_lines)
   265	            ):
   266	                persistent_uselessness_errors.add(candidate)
   267	        for error_info in persistent_uselessness_errors:
   268	            context = Context(line=error_info[2], column=error_info[3])
   269	            context.end_line = error_info[4]
   270	            context.end_column = error_info[5]
   271	            yield error_info[1], context, error_info[0]
   272
   273	    def yield_revealed_type_infos(self) -> Iterator[tuple[list[Type], Context]]:
   274	        """Yield all types revealed in at least one iteration step."""
   275
   276	        for note_info, types in self.revealed_types.items():
   277	            context = Context(line=note_info[0], column=note_info[1])
   278	            context.end_line = note_info[2]
   279	            context.end_column = note_info[3]
   280	            yield types, context

   283	class IterationErrorWatcher(ErrorWatcher):
   284	    """Error watcher that filters and separately collects `unreachable` errors,
   285	    `redundant-expr` and `redundant-casts` errors, and revealed types when analysing
   286	    code sections iteratively to help avoid making too-hasty reports."""
   287
   288	    iteration_dependent_errors: IterationDependentErrors
   289
   290	    def __init__(
   291	        self,
   292	        errors: Errors,
   293	        iteration_dependent_errors: IterationDependentErrors,
   294	        *,
   295	        filter_errors: bool | Callable[[str, ErrorInfo], bool] = False,
   296	        save_filtered_errors: bool = False,
   297	        filter_deprecated: bool = False,
   298	    ) -> None:
   299	        super().__init__(
   300	            errors,
   301	            filter_errors=filter_errors,
   302	            save_filtered_errors=save_filtered_errors,
   303	            filter_deprecated=filter_deprecated,
   304	        )
   305	        self.iteration_dependent_errors = iteration_dependent_errors
   306	        iteration_dependent_errors.uselessness_errors.append(set())
   307	        iteration_dependent_errors.unreachable_lines.append(set())
   308
   309	    def on_error(self, file: str, info: ErrorInfo) -> bool:
   310	        """Filter out the "iteration-dependent" errors and notes and store their
   311	        information to handle them after iteration is completed."""
   312
   313	        iter_errors = self.iteration_dependent_errors
   314
   315	        if info.code in (codes.UNREACHABLE, codes.REDUNDANT_EXPR, codes.REDUNDANT_CAST):
   316	            iter_errors.uselessness_errors[-1].add(
   317	                (info.code, info.message, info.line, info.column, info.end_line, info.end_column)
   318	            )
   319	            if info.code == codes.UNREACHABLE:
   320	                iter_errors.unreachable_lines[-1].update(range(info.line, info.end_line + 1))
   321	            return True
   322
   323	        return super().on_error(file, info)

   487	    def report(
   488	        self,
   489	        line: int,
   490	        column: int | None,
   491	        message: str,
   492	        code: ErrorCode | None = None,
   493	        *,
   494	        blocker: bool = False,
   495	        severity: str = "error",
   496	        file: str | None = None,
   497	        only_once: bool = False,
   498	        origin_span: Iterable[int] | None = None,
   499	        offset: int = 0,
   500	        end_line: int | None = None,
   501	        end_column: int | None = None,
   502	        parent_error: ErrorInfo | None = None,
   503	    ) -> ErrorInfo:
   504	        """Report message at the given line using the current error context.
   505
   506	        Args:
   507	            line: line number of error
   508	            column: column number of error
   509	            message: message to report
   510	            code: error code (defaults to 'misc'; not shown for notes)
   511	            blocker: if True, don't continue analysis after this error
   512	            severity: 'error' or 'note'
   513	            file: if non-None, override current file as context
   514	            only_once: if True, only report this exact message once per build
   515	            origin_span: if non-None, override current context as origin
   516	                         (type: ignores have effect here)
   517	            end_line: if non-None, override current context as end
   518	            parent_error: an error this note is attached to (for notes only).
   519	        """
   520	        if self.scope:
   521	            type = self.scope.current_type_name()
   522	            if self.scope.ignored > 0:
   523	                type = None  # Omit type context if nested function
   524	            function = self.scope.current_function_name()
   525	        else:
   526	            type = None
   527	            function = None
   528
   529	        if column is None:
   530	            column = -1
   531	        if end_column is None:
   532	            if column == -1:
   533	                end_column = -1
   534	            else:
   535	                end_column = column + 1
   536
   537	        if file is None:
   538	            file = self.file
   539	        if offset:
   540	            message = " " * offset + message
   541
   542	        if origin_span is None:
   543	            origin_span = [line]
   544
   545	        if end_line is None:
   546	            end_line = line
   547
   548	        code = code or (parent_error.code if parent_error else None)
   549	        code = code or (codes.MISC if not blocker else None)
   550
   551	        info = ErrorInfo(
   552	            import_ctx=self.import_context(),
   553	            file=file,
   554	            module=self.current_module(),
   555	            typ=type,
   556	            function_or_member=function,
   557	            line=line,
   558	            column=column,
   559	            end_line=end_line,
   560	            end_column=end_column,
   561	            severity=severity,
   562	            message=message,
   563	            code=code,
   564	            blocker=blocker,
   565	            only_once=only_once,
   566	            origin=(self.file, origin_span),
   567	            target=self.current_target(),
   568	            parent_error=parent_error,
   569	        )
   570	        self.add_error_info(info)
   571	        return info

   573	    def _add_error_info(self, file: str, info: ErrorInfo) -> None:
   574	        assert file not in self.flushed_files
   575	        # process the stack of ErrorWatchers before modifying any internal state
   576	        # in case we need to filter out the error entirely
   577	        if self._filter_error(file, info):
   578	            return
   579	        if file not in self.error_info_map:
   580	            self.error_info_map[file] = []
   581	        self.error_info_map[file].append(info)
   582	        if info.blocker:
   583	            self.has_blockers.add(file)
   584	        if info.code in (IMPORT, IMPORT_UNTYPED, IMPORT_NOT_FOUND):
   585	            self.seen_import_error = True
   586
   587	    def get_watchers(self) -> Iterator[ErrorWatcher]:
   588	        """Yield the `ErrorWatcher` stack from top to bottom."""
   589	        i = len(self._watchers)
   590	        while i > 0:
   591	            i -= 1
   592	            yield self._watchers[i]
   593
   594	    def _filter_error(self, file: str, info: ErrorInfo) -> bool:
   595	        """
   596	        process ErrorWatcher stack from top to bottom,
   597	        stopping early if error needs to be filtered out
   598	        """
   599	        return any(w.on_error(file, info) for w in self.get_watchers())
   600
   601	    def add_error_info(self, info: ErrorInfo) -> None:
   602	        file, lines = info.origin
   603	        # process the stack of ErrorWatchers before modifying any internal state
   604	        # in case we need to filter out the error entirely
   605	        # NB: we need to do this both here and in _add_error_info, otherwise we
   606	        # might incorrectly update the sets of ignored or only_once messages
   607	        if self._filter_error(file, info):
   608	            return
   609	        if not info.blocker:  # Blockers cannot be ignored
   610	            if file in self.ignored_lines:
   611	                # Check each line in this context for "type: ignore" comments.
   612	                # line == end_line for most nodes, so we only loop once.
   613	                for scope_line in lines:
   614	                    if self.is_ignored_error(scope_line, info, self.ignored_lines[file]):
   615	                        err_code = info.code or codes.MISC
   616	                        if not self.is_error_code_enabled(err_code):
   617	                            # Error code is disabled - don't mark the current
   618	                            # "type: ignore" comment as used.
   619	                            return
   620	                        # Annotation requests us to ignore all errors on this line.
   621	                        self.used_ignored_lines[file][scope_line].append(err_code.code)
   622	                        return
   623	            if file in self.ignored_files:
   624	                return
   625	        if info.only_once:
   626	            if info.message in self.only_once_messages:
   627	                return
   628	            self.only_once_messages.add(info.message)
   629	        if (
   630	            self.seen_import_error
   631	            and info.code not in (IMPORT, IMPORT_UNTYPED, IMPORT_NOT_FOUND)
   632	            and self.has_many_errors()
   633	        ):
   634	            # Missing stubs can easily cause thousands of errors about
   635	            # Any types, especially when upgrading to mypy 0.900,
   636	            # which no longer bundles third-party library stubs. Avoid
   637	            # showing too many errors to make it easier to see
   638	            # import-related errors.
   639	            info.hidden = True
   640	            self.report_hidden_errors(info)
   641	        self._add_error_info(file, info)
   642	        ignored_codes = self.ignored_lines.get(file, {}).get(info.line, [])
   643	        if ignored_codes and info.code:
   644	            # Something is ignored on the line, but not this error, so maybe the error
   645	            # code is incorrect.
   646	            msg = f'Error code "{info.code.code}" not covered by "type: ignore" comment'
   647	            if info.code in original_error_codes:
   648	                # If there seems to be a "type: ignore" with a stale error
   649	                # code, report a more specific note.
   650	                old_code = original_error_codes[info.code].code
   651	                if old_code in ignored_codes:
   652	                    msg = (
   653	                        f'Error code changed to {info.code.code}; "type: ignore" comment '
   654	                        + "may be out of date"
   655	                    )
   656	            note = ErrorInfo(
   657	                import_ctx=info.import_ctx,
   658	                file=info.file,
   659	                module=info.module,
   660	                typ=info.type,
   661	                function_or_member=info.function_or_member,
   662	                line=info.line,
   663	                column=info.column,
   664	                end_line=info.end_line,
   665	                end_column=info.end_column,
   666	                severity="note",
   667	                message=msg,
   668	                code=None,
   669	                blocker=False,
   670	                only_once=False,
   671	            )
   672	            self._add_error_info(file, note)
   673	        if (
   674	            self.options.show_error_code_links

   951	    def format_messages(
   952	        self, error_tuples: list[ErrorTuple], source_lines: list[str] | None
   953	    ) -> list[str]:
   954	        """Return a string list that represents the error messages.
   955
   956	        Use a form suitable for displaying to the user. If self.pretty
   957	        is True also append a relevant trimmed source code line (only for
   958	        severity 'error').
   959	        """
   960	        a: list[str] = []
   961	        for file, line, column, end_line, end_column, severity, message, code in error_tuples:
   962	            s = ""
   963	            if file is not None:
   964	                if self.options.show_column_numbers and line >= 0 and column >= 0:
   965	                    srcloc = f"{file}:{line}:{1 + column}"
   966	                    if self.options.show_error_end and end_line >= 0 and end_column >= 0:
   967	                        srcloc += f":{end_line}:{end_column}"
   968	                elif line >= 0:
   969	                    srcloc = f"{file}:{line}"
   970	                else:
   971	                    srcloc = file
   972	                s = f"{srcloc}: {severity}: {message}"
   973	            else:
   974	                s = message
   975	            if (
   976	                not self.hide_error_codes
   977	                and code
   978	                and (severity != "note" or code in SHOW_NOTE_CODES)
   979	            ):
   980	                # If note has an error code, it is related to a previous error. Avoid
   981	                # displaying duplicate error codes.
   982	                s = f"{s}  [{code.code}]"
   983	            a.append(s)
   984	            if self.options.pretty:
   985	                # Add source code fragment and a location marker.
   986	                if severity == "error" and source_lines and line > 0:
   987	                    source_line = source_lines[line - 1]
   988	                    source_line_expanded = source_line.expandtabs()
   989	                    if column < 0:
   990	                        # Something went wrong, take first non-empty column.
   991	                        column = len(source_line) - len(source_line.lstrip())
   992
   993	                    # Shifts column after tab expansion
   994	                    column = len(source_line[:column].expandtabs())
   995	                    end_column = len(source_line[:end_column].expandtabs())
   996
   997	                    # Note, currently coloring uses the offset to detect source snippets,
   998	                    # so these offsets should not be arbitrary.
   999	                    a.append(" " * DEFAULT_SOURCE_OFFSET + source_line_expanded)
  1000	                    marker = "^"
  1001	                    if end_line == line and end_column > column:
  1002	                        marker = f'^{"~" * (end_column - column - 1)}'
  1003	                    a.append(" " * (DEFAULT_SOURCE_OFFSET + column) + marker)
  1004	        return a

  1006	    def file_messages(self, path: str, formatter: ErrorFormatter | None = None) -> list[str]:
  1007	        """Return a string list of new error messages from a given file.
  1008
  1009	        Use a form suitable for displaying to the user.
  1010	        """
  1011	        if path not in self.error_info_map:
  1012	            return []
  1013
  1014	        error_info = self.error_info_map[path]
  1015	        error_info = [info for info in error_info if not info.hidden]
  1016	        error_info = self.remove_duplicates(self.sort_messages(error_info))
  1017	        error_tuples = self.render_messages(error_info)
  1018
  1019	        if formatter is not None:
  1020	            errors = create_errors(error_tuples)
  1021	            return [formatter.report_error(err) for err in errors]
  1022
  1023	        self.flushed_files.add(path)
  1024	        source_lines = None
  1025	        if self.options.pretty and self.read_source:
  1026	            # Find shadow file mapping and read source lines if a shadow file exists for the given path.
  1027	            # If shadow file mapping is not found, read source lines
  1028	            mapped_path = self.find_shadow_file_mapping(path)
  1029	            if mapped_path:
  1030	                source_lines = self.read_source(mapped_path)
  1031	            else:
  1032	                source_lines = self.read_source(path)
  1033	        return self.format_messages(error_tuples, source_lines)

  1039

  1043	        return None

  1066	    def render_messages(self, errors: list[ErrorInfo]) -> list[ErrorTuple]:
  1067	        """Translate the messages into a sequence of tuples.
  1068
  1069	        Each tuple is of form (path, line, col, severity, message, code).
  1070	        The rendered sequence includes information about error contexts.
  1071	        The path item may be None. If the line item is negative, the
  1072	        line number is not defined for the tuple.
  1073	        """
  1074	        result: list[ErrorTuple] = []
  1075	        prev_import_context: list[tuple[str, int]] = []
  1076	        prev_function_or_member: str | None = None
  1077	        prev_type: str | None = None
  1078
  1079	        for e in errors:
  1080	            # Report module import context, if different from previous message.
  1081	            if not self.options.show_error_context:
  1082	                pass
  1083	            elif e.import_ctx != prev_import_context:
  1084	                last = len(e.import_ctx) - 1
  1085	                i = last
  1086	                while i >= 0:
  1087	                    path, line = e.import_ctx[i]
  1088	                    fmt = "{}:{}: note: In module imported here"
  1089	                    if i < last:
  1090	                        fmt = "{}:{}: note: ... from here"
  1091	                    if i > 0:
  1092	                        fmt += ","
  1093	                    else:
  1094	                        fmt += ":"
  1095	                    # Remove prefix to ignore from path (if present) to
  1096	                    # simplify path.
  1097	                    path = remove_path_prefix(path, self.ignore_prefix)
  1098	                    result.append((None, -1, -1, -1, -1, "note", fmt.format(path, line), None))
  1099	                    i -= 1
  1100
  1101	            file = self.simplify_path(e.file)
  1102
  1103	            # Report context within a source file.
  1104	            if not self.options.show_error_context:
  1105	                pass
  1106	            elif e.function_or_member != prev_function_or_member or e.type != prev_type:
  1107	                if e.function_or_member is None:
  1108	                    if e.type is None:
  1109	                        result.append((file, -1, -1, -1, -1, "note", "At top level:", None))
  1110	                    else:
  1111	                        result.append(
  1112	                            (file, -1, -1, -1, -1, "note", f'In class "{e.type}":', None)
  1113	                        )
  1114	                else:
  1115	                    if e.type is None:
  1116	                        result.append(
  1117	                            (
  1118	                                file,
  1119	                                -1,
  1120	                                -1,
  1121	                                -1,
  1122	                                -1,
  1123	                                "note",
  1124	                                f'In function "{e.function_or_member}":',
  1125	                                None,
  1126	                            )
  1127	                        )
  1128	                    else:
  1129	                        result.append(
  1130	                            (
  1131	                                file,
  1132	                                -1,
  1133	                                -1,
  1134	                                -1,
  1135	                                -1,
  1136	                                "note",
  1137	                                'In member "{}" of class "{}":'.format(
  1138	                                    e.function_or_member, e.type
  1139	                                ),
  1140	                                None,
  1141	                            )
  1142	                        )
  1143	            elif e.type != prev_type:
  1144	                if e.type is None:
  1145	                    result.append((file, -1, -1, -1, -1, "note", "At top level:", None))
  1146	                else:
  1147	                    result.append((file, -1, -1, -1, -1, "note", f'In class "{e.type}":', None))
  1148
  1149	            result.append(
  1150	                (file, e.line, e.column, e.end_line, e.end_column, e.severity, e.message, e.code)
  1151	            )
  1152
  1153	            prev_import_context = e.import_ctx
  1154	            prev_function_or_member = e.function_or_member
  1155	            prev_type = e.type
  1156
  1157	        return result

  1159	    def sort_messages(self, errors: list[ErrorInfo]) -> list[ErrorInfo]:
  1160	        """Sort an array of error messages locally by line number.
  1161
  1162	        I.e., sort a run of consecutive messages with the same
  1163	        context by line number, but otherwise retain the general
  1164	        ordering of the messages.
  1165	        """
  1166	        result: list[ErrorInfo] = []
  1167	        i = 0
  1168	        while i < len(errors):
  1169	            i0 = i
  1170	            # Find neighbouring errors with the same context and file.
  1171	            while (
  1172	                i + 1 < len(errors)
  1173	                and errors[i + 1].import_ctx == errors[i].import_ctx
  1174	                and errors[i + 1].file == errors[i].file
  1175	            ):
  1176	                i += 1
  1177	            i += 1
  1178
  1179	            # Sort the errors specific to a file according to line number and column.
  1180	            a = sorted(errors[i0:i], key=lambda x: (x.line, x.column))
  1181	            a = self.sort_within_context(a)
  1182	            result.extend(a)
  1183	        return result

  1185	    def sort_within_context(self, errors: list[ErrorInfo]) -> list[ErrorInfo]:
  1186	        """For the same location decide which messages to show first/last.
  1187
  1188	        Currently, we only compare within the same error code, to decide the
  1189	        order of various additional notes.
  1190	        """
  1191	        result = []
  1192	        i = 0
  1193	        while i < len(errors):
  1194	            i0 = i
  1195	            # Find neighbouring errors with the same position and error code.
  1196	            while (
  1197	                i + 1 < len(errors)
  1198	                and errors[i + 1].line == errors[i].line
  1199	                and errors[i + 1].column == errors[i].column
  1200	                and errors[i + 1].end_line == errors[i].end_line
  1201	                and errors[i + 1].end_column == errors[i].end_column
  1202	                and errors[i + 1].code == errors[i].code
  1203	            ):
  1204	                i += 1
  1205	            i += 1
  1206
  1207	            # Sort the messages specific to a given error by priority.
  1208	            a = sorted(errors[i0:i], key=lambda x: x.priority)
  1209	            result.extend(a)
  1210	        return result

  1212	    def remove_duplicates(self, errors: list[ErrorInfo]) -> list[ErrorInfo]:
  1213	        filtered_errors = []
  1214	        seen_by_line: defaultdict[int, set[tuple[str, str]]] = defaultdict(set)
  1215	        removed = set()
  1216	        for err in errors:
  1217	            if err.parent_error is not None:
  1218	                # Notes with specified parent are removed together with error below.

   211	    def visit_instance(self, t: Instance) -> Type:
   212	        args = self.expand_types_with_unpack(list(t.args))
   213
   214	        if isinstance(t.type, FakeInfo):
   215	            # The type checker expands function definitions and bodies
   216	            # if they depend on constrained type variables but the body
   217	            # might contain a tuple type comment (e.g., # type: (int, float)),
   218	            # in which case 't.type' is not yet available.
   219	            #
   220	            # See: https://github.com/python/mypy/issues/16649
   221	            return t.copy_modified(args=args)
   222
   223	        if t.type.fullname == "builtins.tuple":
   224	            # Normalize Tuple[*Tuple[X, ...], ...] -> Tuple[X, ...]
   225	            arg = args[0]
   226	            if isinstance(arg, UnpackType):
   227	                unpacked = get_proper_type(arg.type)
   228	                if isinstance(unpacked, Instance):
   229	                    # TODO: this and similar asserts below may be unsafe because get_proper_type()
   230	                    # may be called during semantic analysis before all invalid types are removed.
   231	                    assert unpacked.type.fullname == "builtins.tuple"
   232	                    args = list(unpacked.args)
   233	        return t.copy_modified(args=args)

   235	    def visit_type_var(self, t: TypeVarType) -> Type:
   236	        # Normally upper bounds can't contain other type variables, the only exception is
   237	        # special type variable Self`0 <: C[T, S], where C is the class where Self is used.
   238	        if t.id.is_self():

   240	        repl = self.variables.get(t.id, t)
   241	        if isinstance(repl, ProperType) and isinstance(repl, Instance):

   245	        if isinstance(repl, TypeVarType) and repl.has_default():

   252	            self.recursive_tvar_guard[tvar_id] = repl
