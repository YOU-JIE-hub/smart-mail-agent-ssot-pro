   626	    def showPage(self):
   627	        """Close the current page and possibly start on a new page."""
   628	        # ensure a space at the end of the stream - Acrobat does
   629	        # not mind, but Ghostscript dislikes 'Qendstream' even if
   630	        # the length marker finishes after 'Q'
   631
   632	        pageWidth = self._pagesize[0]
   633	        pageHeight = self._pagesize[1]
   634	        cM = self._cropMarks
   635	        code = self._code
   636	        if cM:
   637	            bw = max(0,getattr(cM,'borderWidth',36))
   638	            if bw:
   639	                markLast = getattr(cM,'markLast',1)
   640	                ml = min(bw,max(0,getattr(cM,'markLength',18)))
   641	                mw = getattr(cM,'markWidth',0.5)
   642	                mc = getattr(cM,'markColor',black)
   643	                mg = 2*bw-ml
   644	                cx0 = len(code)
   645	                if ml and mc:
   646	                    self.saveState()
   647	                    self.setStrokeColor(mc)
   648	                    self.setLineWidth(mw)
   649	                    self.lines([
   650	                        (bw,0,bw,ml),
   651	                        (pageWidth+bw,0,pageWidth+bw,ml),
   652	                        (bw,pageHeight+mg,bw,pageHeight+2*bw),
   653	                        (pageWidth+bw,pageHeight+mg,pageWidth+bw,pageHeight+2*bw),
   654	                        (0,bw,ml,bw),
   655	                        (pageWidth+mg,bw,pageWidth+2*bw,bw),
   656	                        (0,pageHeight+bw,ml,pageHeight+bw),
   657	                        (pageWidth+mg,pageHeight+bw,pageWidth+2*bw,pageHeight+bw),
   658	                        ])
   659	                    self.restoreState()
   660	                    if markLast:
   661	                        #if the marks are to be drawn after the content
   662	                        #save the code we just drew for later use
   663	                        L = code[cx0:]
   664	                        del code[cx0:]
   665	                        cx0 = len(code)
   666
   667	                bleedW = max(0,getattr(cM,'bleedWidth',0))
   668	                self.saveState()
   669	                self.translate(bw-bleedW,bw-bleedW)
   670	                if bleedW:
   671	                    #scale everything
   672	                    self.scale(1+(2.0*bleedW)/pageWidth,1+(2.0*bleedW)/pageHeight)
   673
   674	                #move our translation/expansion code to the beginning
   675	                C = code[cx0:]
   676	                del code[cx0:]
   677	                code[0:0] = C
   678	                self.restoreState()
   679	                if markLast:
   680	                    code.extend(L)
   681	                pageWidth = 2*bw + pageWidth
   682	                pageHeight = 2*bw + pageHeight
   683
   684	        code.append(' ')
   685	        page = pdfdoc.PDFPage()
   686	        page.pagewidth = pageWidth
   687	        page.pageheight = pageHeight
   688	        page.Rotate = self._pageRotation
   689	        page.hasImages = self._currentPageHasImages
   690	        page.setPageTransition(self._pageTransition)
   691	        page.setCompression(self._pageCompression)
   692	        for box in ('crop','art','bleed','trim'):
   693	            size = getattr(self,'_%sBox'%box,None)
   694	            if size:
   695	                setattr(page,box.capitalize()+'Box',pdfdoc.PDFArray(size))
   696	        if self._pageDuration is not None:
   697	            page.Dur = self._pageDuration
   698
   699	        strm =  self._psCommandsBeforePage + [self._preamble] + code + self._psCommandsAfterPage
   700	        page.setStream(strm)
   701	        self._setColorSpace(page)
   702	        self._setExtGState(page)
   703	        self._setXObjects(page)
   704	        self._setShadingUsed(page)
   705	        self._setAnnotations(page)
   706	        self._doc.addPage(page)
   707
   708	        if self._onPage: self._onPage(self._pageNumber)
   709	        self._startPage()

   755	    def bookmarkPage(self, key,
   756	                      fit="Fit",
   757	                      left=None,
   758	                      top=None,
   759	                      bottom=None,
   760	                      right=None,
   761	                      zoom=None
   762	                      ):
   763	        """
   764	        This creates a bookmark to the current page which can
   765	        be referred to with the given key elsewhere.
   766
   767	        PDF offers very fine grained control over how Acrobat
   768	        reader is zoomed when people link to this. The default
   769	        is to keep the user's current zoom settings. the last
   770	        arguments may or may not be needed depending on the
   771	        choice of 'fitType'.
   772
   773	        Fit types and the other arguments they use are:
   774
   775	        - XYZ left top zoom - fine grained control.  null
   776	          or zero for any of the parameters means 'leave
   777	          as is', so "0,0,0" will keep the reader's settings.
   778	          NB. Adobe Reader appears to prefer "null" to 0's.
   779
   780	        - Fit - entire page fits in window
   781
   782	        - FitH top - top coord at top of window, width scaled
   783	          to fit.
   784
   785	        - FitV left - left coord at left of window, height
   786	          scaled to fit
   787
   788	        - FitR left bottom right top - scale window to fit
   789	          the specified rectangle
   790
   791	        (question: do we support /FitB, FitBH and /FitBV
   792	        which are hangovers from version 1.1 / Acrobat 3.0?)"""
   793	        dest = self._bookmarkReference(key)
   794	        self._doc.inPage() # try to enable page-only features
   795	        pageref = self._doc.thisPageRef()
   796
   797	        #None = "null" for PDF
   798	        if left is None:
   799	            left = "null"
   800	        if top is None:
   801	            top = "null"
   802	        if bottom is None:
   803	            bottom = "null"
   804	        if right is None:
   805	            right = "null"
   806	        if zoom is None:
   807	            zoom = "null"
   808
   809	        if fit == "XYZ":
   810	            dest.xyz(left,top,zoom)
   811	        elif fit == "Fit":
   812	            dest.fit()
   813	        elif fit == "FitH":
   814	            dest.fith(top)
   815	        elif fit == "FitV":
   816	            dest.fitv(left)
   817	        elif fit == "FitR":
   818	            dest.fitr(left,bottom,right,top)
   819	        #Do we need these (version 1.1 / Acrobat 3 versions)?
   820	        elif fit == "FitB":
   821	            dest.fitb()
   822	        elif fit == "FitBH":

   899	    def drawImage(self, image, x, y, width=None, height=None, mask=None,
   900	            preserveAspectRatio=False, anchor='c', anchorAtXY=False, showBoundary=False):
   901	        """Draws the image (ImageReader object or filename) as specified.
   902
   903	        "image" may be an image filename or an ImageReader object.
   904
   905	        x and y define the lower left corner of the image you wish to
   906	        draw (or of its bounding box, if using preserveAspectRation below).
   907
   908	        If width and height are not given, the width and height of the
   909	        image in pixels is used at a scale of 1 point to 1 pixel.
   910
   911	        If width and height are given, the image will be stretched to fill
   912	        the given rectangle bounded by (x, y, x+width, y-height).
   913
   914	        If you supply negative widths and/or heights, it inverts them and adjusts
   915	        x and y accordingly.
   916
   917	        The method returns the width and height of the underlying image, since
   918	        this is often useful for layout algorithms and saves you work if you have
   919	        not specified them yourself.
   920
   921	        The mask parameter supports transparent backgrounds. It takes 6 numbers
   922	        and defines the range of RGB values which will be masked out or treated
   923	        as transparent.  For example with [0,2,40,42,136,139], it will mask out
   924	        any pixels with a Red value from 0-2, Green from 40-42 and
   925	        Blue from 136-139  (on a scale of 0-255).
   926
   927	        New post version 2.0:  drawImage can center an image in a box you
   928	        provide, while preserving its aspect ratio.  For example, you might
   929	        have a fixed square box in your design, and a collection of photos
   930	        which might be landscape or portrait that you want to appear within
   931	        the box.  If preserveAspectRatio is true, your image will appear within
   932	        the box specified.
   933
   934
   935	        If preserveAspectRatio is True, the anchor property can be used to
   936	        specify how images should fit into the given box.  It should
   937	        be set to one of the following values, taken from the points of
   938	        the compass (plus 'c' for 'centre'):
   939
   940	                nw   n   ne
   941	                w    c    e
   942	                sw   s   se
   943
   944	        The default value is 'c' for 'centre'.  Thus, if you want your
   945	        bitmaps to always be centred and appear at the top of the given box,
   946	        set anchor='n'.      There are good examples of this in the output
   947	        of test_pdfgen_general.py
   948
   949	        Unlike drawInlineImage, this creates 'external images' which
   950	        are only stored once in the PDF file but can be drawn many times.
   951	        If you give it the same filename twice, even at different locations
   952	        and sizes, it will reuse the first occurrence, resulting in a saving
   953	        in file size and generation time.  If you use ImageReader objects,
   954	        it tests whether the image content has changed before deciding
   955	        whether to reuse it.
   956
   957	        In general you should use drawImage in preference to drawInlineImage
   958	        unless you have read the PDF Spec and understand the tradeoffs."""
   959
   960	        self._currentPageHasImages = 1
   961
   962	        # first, generate a unique name/signature for the image.  If ANYTHING
   963	        # is different, even the mask, this should be different.
   964	        if isinstance(image,ImageReader):
   965	            rawdata = image.getRGBData()
   966	            smask = image._dataA
   967	            if mask=='auto' and smask:
   968	                mdata = smask.getRGBData()
   969	            else:
   970	                mdata = str(mask)
   971	            if isUnicode(mdata):
   972	                mdata = mdata.encode('utf8')
   973	            name = _digester(rawdata+mdata)
   974	        else:
   975	            #filename, use it
   976	            s = '%s%s' % (image, mask)
   977	            if isUnicode(s):
   978	                s = s.encode('utf-8')
   979	            name = _digester(s)
   980
   981	        # in the pdf document, this will be prefixed with something to
   982	        # say it is an XObject.  Does it exist yet?
   983	        regName = self._doc.getXObjectName(name)
   984	        imgObj = self._doc.idToObject.get(regName, None)
   985	        if not imgObj:
   986	            #first time seen, create and register the PDFImageXobject
   987	            imgObj = pdfdoc.PDFImageXObject(name, image, mask=mask)
   988	            imgObj.name = name
   989	            self._setXObjects(imgObj)
   990	            self._doc.Reference(imgObj, regName)
   991	            self._doc.addForm(name, imgObj)
   992	            smask = getattr(imgObj,'_smask',None)
   993	            if smask:   #set up the softmask obtained above
   994	                mRegName = self._doc.getXObjectName(smask.name)
   995	                mImgObj = self._doc.idToObject.get(mRegName, None)
   996	                if not mImgObj:
   997	                    self._setXObjects(smask)
   998	                    imgObj.smask = self._doc.Reference(smask,mRegName)
   999	                else:
  1000	                    imgObj.smask = pdfdoc.PDFObjectReference(mRegName)
  1001	                del imgObj._smask
  1002
  1003	        # ensure we have a size, as PDF will make it 1x1 pixel otherwise!
  1004	        x,y,width,height,scaled = aspectRatioFix(preserveAspectRatio,anchor,x,y,width,height,imgObj.width,imgObj.height,anchorAtXY)
  1005
  1006	        # scale and draw
  1007	        self.saveState()
  1008	        self.translate(x, y)
  1009	        self.scale(width, height)
  1010	        self._code.append("/%s Do" % regName)
  1011	        self.restoreState()
  1012	        if showBoundary:
  1013	            #self.rect(x,y,width,height,stroke=1,fill=0)
  1014	            self.drawBoundary(showBoundary,x,y,width,height)
  1015
  1016	        # track what's been used on this page
  1017	        self._formsinuse.append(name)
  1018
  1019	        return (imgObj.width, imgObj.height)

  1333	    def resetTransforms(self):
  1334	        """I want to draw something (eg, string underlines) w.r.t. the default user space.
  1335	           Reset the matrix! This should be used usually as follows::
  1336
  1337	              canv.saveState()
  1338	              canv.resetTransforms()
  1339	              #...draw some stuff in default space coords...
  1340	              canv.restoreState() # go back!
  1341	        """
  1342	        # we have to adjoin the inverse, since reset is not a basic operation (without save/restore)
  1343	        (selfa, selfb, selfc, selfd, selfe, selff) = self._currentMatrix
  1344	        det = selfa*selfd - selfc*selfb
  1345	        resulta = selfd/det
  1346	        resultc = -selfc/det
  1347	        resulte = (selfc*selff - selfd*selfe)/det
  1348	        resultd = selfa/det
  1349	        resultb = -selfb/det
  1350	        resultf = (selfe*selfb - selff*selfa)/det
  1351	        self.transform(resulta, resultb, resultc, resultd, resulte, resultf)

  1353	    def transform(self, a,b,c,d,e,f):
  1354	        """adjoin a mathematical transform to the current graphics state matrix.
  1355	           Not recommended for beginners."""
  1356	        #How can Python track this?
  1357	        if ENABLE_TRACKING:
  1358	            a0,b0,c0,d0,e0,f0 = self._currentMatrix
  1359	            self._currentMatrix = (a0*a+c0*b,    b0*a+d0*b,
  1360	                                   a0*c+c0*d,    b0*c+d0*d,
  1361	                                   a0*e+c0*f+e0, b0*e+d0*f+f0)
  1362	        if self._code and self._code[-1][-3:]==' cm':
  1363	            L = self._code[-1].split()
  1364	            a0, b0, c0, d0, e0, f0 = list(map(float,L[-7:-1]))
  1365	            s = len(L)>7 and join(L)+ ' %s cm' or '%s cm'
  1366	            self._code[-1] = s % fp_str(a0*a+c0*b,b0*a+d0*b,a0*c+c0*d,b0*c+d0*d,a0*e+c0*f+e0,b0*e+d0*f+f0)
  1367	        else:

  1641	    def drawAlignedString(self, x, y, text, pivotChar=rl_config.decimalSymbol, mode=None, charSpace=0, direction=None, wordSpace=None):
  1642	        """Draws a string aligned on the first '.' (or other pivot character).
  1643
  1644	        The centre position of the pivot character will be used as x.
  1645	        So, you could draw a straight line down through all the decimals in a
  1646	        column of numbers, and anything without a decimal should be
  1647	        optically aligned with those that have.
  1648
  1649	        There is one special rule to help with accounting formatting.  Here's
  1650	        how normal numbers should be aligned on the 'dot'. Look at the
  1651	        LAST two::
  1652
  1653	           12,345,67
  1654	              987.15
  1655	               42
  1656	           -1,234.56
  1657	             (456.78)
  1658	             (456)
  1659	               27 inches
  1660	               13cm
  1661
  1662	        Since the last three do not contain a dot, a crude dot-finding
  1663	        rule would place them wrong. So we test for the special case
  1664	        where no pivot is found, digits are present, but the last character
  1665	        is not a digit.  We then work back from the end of the string
  1666	        This case is a tad slower but hopefully rare.
  1667
  1668	        """
  1669	        parts = text.split(pivotChar,1)
  1670	        pivW = self.stringWidth(pivotChar, self._fontname, self._fontsize)
  1671
  1672	        if len(parts) == 1 and digitPat.search(text) is not None and text[-1] not in digits:
  1673	            #we have no decimal but it ends in a bracket, or 'in' or something.
  1674	            #the cut should be after the last digit.
  1675	            leftText = parts[0][0:-1]
  1676	            rightText = parts[0][-1]
  1677	            #any more?
  1678	            while leftText[-1] not in digits:
  1679	                rightText = leftText[-1] + rightText
  1680	                leftText = leftText[0:-1]
  1681
  1682	            self.drawRightString(x-0.5*pivW, y, leftText, mode=mode, charSpace=charSpace,
  1683	                    direction=direction, wordSpace=wordSpace)
  1684	            self.drawString(x-0.5*pivW, y, rightText, mode=mode, charSpace=charSpace,
  1685	                    direction=direction, wordSpace=wordSpace)
  1686
  1687	        else:
  1688	            #normal case
  1689	            leftText = parts[0]
  1690	            self.drawRightString(x-0.5*pivW, y, leftText, mode=mode, charSpace=charSpace,
  1691	                    direction=direction, wordSpace=wordSpace)
  1692	            if len(parts) > 1:
  1693	                rightText = pivotChar + parts[1]
  1694	                self.drawString(x-0.5*pivW, y, rightText, mode=mode, charSpace=charSpace,
  1695	                        direction=direction, wordSpace=wordSpace)

    15	def bezierArc(x1,y1, x2,y2, startAng=0, extent=90):
    16	    """bezierArc(x1,y1, x2,y2, startAng=0, extent=90) --> List of Bezier
    17	curve control points.
    18
    19	(x1, y1) and (x2, y2) are the corners of the enclosing rectangle.  The
    20	coordinate system has coordinates that increase to the right and down.
    21	Angles, measured in degress, start with 0 to the right (the positive X
    22	axis) and increase counter-clockwise.  The arc extends from startAng
    23	to startAng+extent.  I.e. startAng=0 and extent=180 yields an openside-down
    24	semi-circle.
    25
    26	The resulting coordinates are of the form (x1,y1, x2,y2, x3,y3, x4,y4)
    27	such that the curve goes from (x1, y1) to (x4, y4) with (x2, y2) and
    28	(x3, y3) as their respective Bezier control points."""
    29
    30	    x1,y1, x2,y2 = min(x1,x2), max(y1,y2), max(x1,x2), min(y1,y2)
    31
    32	    if abs(extent) <= 90:
    33	        arcList = [startAng]
    34	        fragAngle = float(extent)
    35	        Nfrag = 1
    36	    else:
    37	        arcList = []
    38	        Nfrag = int(ceil(abs(extent)/90.))
    39	        fragAngle = float(extent) / Nfrag
    40
    41	    x_cen = (x1+x2)/2.
    42	    y_cen = (y1+y2)/2.
    43	    rx = (x2-x1)/2.
    44	    ry = (y2-y1)/2.
    45	    halfAng = fragAngle * pi / 360.
    46	    kappa = abs(4. / 3. * (1. - cos(halfAng)) / sin(halfAng))
    47
    48	    if fragAngle < 0:
    49	        sign = -1
    50	    else:
    51	        sign = 1
    52
    53	    pointList = []
    54
    55	    for i in range(Nfrag):
    56	        theta0 = (startAng + i*fragAngle) * pi / 180.
    57	        theta1 = (startAng + (i+1)*fragAngle) *pi / 180.
    58	        if fragAngle > 0:
    59	            pointList.append((x_cen + rx * cos(theta0),
    60	                              y_cen - ry * sin(theta0),
    61	                              x_cen + rx * (cos(theta0) - kappa * sin(theta0)),
    62	                              y_cen - ry * (sin(theta0) + kappa * cos(theta0)),
    63	                              x_cen + rx * (cos(theta1) + kappa * sin(theta1)),
    64	                              y_cen - ry * (sin(theta1) - kappa * cos(theta1)),
    65	                              x_cen + rx * cos(theta1),
    66	                              y_cen - ry * sin(theta1)))
    67	        else:
