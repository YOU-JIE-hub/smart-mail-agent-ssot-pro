   187	        """Works out where they go.
   188
   189	        Sets an attribute _positions which is a list of
   190	        lists of (x, y) matching the data.

   192
   193	        self._seriesCount = len(self.data)

   199	        yscale = self.yValueAxis.scale

   202	        for rowNo, row in enumerate(data):
   203	            if isinstance(row, FillPairedData):
   204	                other = row.other
   205	                if 0<=other<n:
   206	                    if other==rowNo:
   207	                        raise ValueError('data row %r may not be paired with itself' % rowNo)
   208	                    pairs.add((rowNo,other))
   209	                else:
   210	                    raise ValueError('data row %r is paired with invalid data row %r' % (rowNo, other))
   211	            line = [].append
   212	            for colNo, datum in enumerate(row):
   213	                xv = datum[0]

   215	                    (

   217	                    yscale(datum[1])

   220	            P(line.__self__)

   229	    def _innerDrawLabel(self, rowNo, colNo, x, y):
   230	        "Draw a label for a given item in the list."
   231
   232	        labelFmt = self.lineLabelFormat
   233	        labelValue = self.data[rowNo][colNo][1] ###
   234
   235	        if labelFmt is None:
   236	            labelText = None
   237	        elif isinstance(labelFmt,str):
   238	            if labelFmt == 'values':
   239	                labelText = self.lineLabelArray[rowNo][colNo]
   240	            else:
   241	                labelText = labelFmt % labelValue
   242	        elif hasattr(labelFmt,'__call__'):
   243	            if not hasattr(labelFmt,'__labelFmtEX__'):
   244	                labelText = labelFmt(labelValue)
   245	            else:
   246	                labelText = labelFmt(self,rowNo,colNo,x,y)
   247	        else:
   248	            raise ValueError("Unknown formatter type %s, expected string or function"% labelFmt)
   249
   250	        if labelText:
   251	            label = self.lineLabels[(rowNo, colNo)]
   252	            if not label.visible: return
   253	            #hack to make sure labels are outside the bar
   254	            if y > 0:
   255	                label.setOrigin(x, y + self.lineLabelNudge)
   256	            else:
   257	                label.setOrigin(x, y - self.lineLabelNudge)
   258	            label.setText(labelText)
   259	        else:
   260	            label = None
   261	        return label

   268	    def makeLines(self):
   269	        g = Group()
   270	        yA = self.yValueAxis
   271	        xA = self.xValueAxis
   272	        bubblePlot = getattr(self,'_bubblePlot',None)
   273	        if bubblePlot:
   274	            bubbleR = min(yA._bubbleRadius,xA._bubbleRadius)
   275	            bubbleMax = xA._bubbleMax
   276
   277	        labelFmt = self.lineLabelFormat
   278
   279	        P = self._positions
   280	        _inFill = getattr(self,'_inFill',None)
   281	        lines = self.lines
   282	        styleCount = len(lines)
   283	        if (_inFill or self._pairInFills or
   284	                [rowNo for rowNo in range(len(P))
   285	                        if getattr(lines[rowNo%styleCount],'inFill',False)]
   286	                ):
   287	            inFillY = getattr(_inFill,'yValue',None)
   288	            if inFillY is None:
   289	                inFillY = xA._y
   290	            else:
   291	                inFillY = yA.scale(inFillY)
   292	            inFillX0 = yA._x
   293	            inFillX1 = inFillX0 + xA._length
   294	            inFillG = getattr(self,'_inFillG',g)
   295	        lG = getattr(self,'_lineG',g)
   296	        # Iterate over data rows.
   297	        R = range(len(P))
   298	        if self.reversePlotOrder: R = reversed(R)
   299	        for rowNo in R:
   300	            row = P[rowNo]
   301	            styleRowNo = rowNo % styleCount
   302	            rowStyle = lines[styleRowNo]
   303	            strokeColor = getattr(rowStyle,'strokeColor',None)
   304	            fillColor = getattr(rowStyle, 'fillColor', strokeColor)
   305	            inFill = getattr(rowStyle,'inFill',_inFill)
   306	            dash = getattr(rowStyle, 'strokeDashArray', None)
   307
   308	            if hasattr(rowStyle, 'strokeWidth'):
   309	                width = rowStyle.strokeWidth
   310	            elif hasattr(lines, 'strokeWidth'):
   311	                width = lines.strokeWidth
   312	            else:
   313	                width = None
   314
   315	            # Iterate over data columns.
   316	            if self.joinedLines:
   317	                points = flatten(row)
   318	                if inFill or isinstance(row,FillPairedData):
   319	                    filler = getattr(rowStyle, 'filler', None)
   320	                    if isinstance(row,FillPairedData):
   321	                        fpoints = points + flatten(reversed(P[row.other]))
   322	                    else:
   323	                        fpoints = [inFillX0,inFillY] + points + [inFillX1,inFillY]
   324	                    if filler:
   325	                        filler.fill(self,inFillG,rowNo,fillColor,fpoints)
   326	                    else:
   327	                        inFillG.add(Polygon(fpoints,fillColor=fillColor,strokeColor=strokeColor if strokeColor==fillColor else None,strokeWidth=width or 0.1))
   328	                if not inFill or inFill==2 or strokeColor!=fillColor:
   329	                    line = PolyLine(points,strokeColor=strokeColor,strokeLineCap=0,strokeLineJoin=1)
   330	                    if width:
   331	                        line.strokeWidth = width
   332	                    if dash:
   333	                        line.strokeDashArray = dash
   334	                    lG.add(line)
   335
   336	            if hasattr(rowStyle, 'symbol'):
   337	                uSymbol = rowStyle.symbol
   338	            elif hasattr(lines, 'symbol'):
   339	                uSymbol = lines.symbol
   340	            else:
   341	                uSymbol = None
   342
   343	            if uSymbol:
   344	                if bubblePlot: drow = self.data[rowNo]
   345	                for j,xy in enumerate(row):
   346	                    if (styleRowNo,j) in lines:
   347	                        juSymbol = getattr(lines[styleRowNo,j],'symbol',uSymbol)
   348	                    else:
   349	                        juSymbol = uSymbol
   350	                    if juSymbol is uSymbol:
   351	                        symbol = uSymbol
   352	                        symColor = strokeColor
   353	                    else:
   354	                        symbol = juSymbol
   355	                        symColor = getattr(symbol,'fillColor',strokeColor)
   356	                    symbol = uSymbol2Symbol(tpcGetItem(symbol,j),xy[0],xy[1],symColor)
   357	                    if symbol:
   358	                        if bubblePlot:
   359	                            symbol.size = bubbleR*(drow[j][2]/bubbleMax)**0.5
   360	                        g.add(symbol)
   361	            else:
   362	                if bubblePlot: drow = self.data[rowNo]
   363	                for j,xy in enumerate(row):
   364	                    juSymbol = getattr(lines[styleRowNo,j],'symbol',None)
   365	                    if not juSymbol: continue
   366	                    symColor = getattr(juSymbol,'fillColor',getattr(juSymbol,'strokeColor',strokeColor))
   367	                    symbol = uSymbol2Symbol(juSymbol,xy[0],xy[1],symColor)
   368	                    if symbol:
   369	                        if bubblePlot:
   370	                            symbol.size = bubbleR*(drow[j][2]/bubbleMax)**0.5
   371	                        g.add(symbol)
   372
   373	            # Draw data labels.
   374	            for colNo,datum in enumerate(row):
   375	                x1, y1 = datum
   376	                self.drawLabel(g, rowNo, colNo, x1, y1)
   377
   378	            shader = getattr(rowStyle, 'shader', None)
   379	            if shader: shader.shade(self,g,rowNo,strokeColor,row)
   380
   381	        return g

   383	    def draw(self):
   384	        yA = self.yValueAxis
   385	        xA = self.xValueAxis
   386	        if getattr(self,'_bubblePlot',None):
   387	            yA._bubblePlot = xA._bubblePlot = 1
   388	        yA.setPosition(self.x, self.y, self.height)
   389	        if yA: yA.joinAxis = xA
   390	        if xA: xA.joinAxis = yA
   391	        yA.configure(self.data)
   392
   393	        # if zero is in chart, put x axis there, otherwise use bottom.
   394	        xAxisCrossesAt = yA.scale(0)
   395	        if ((xAxisCrossesAt > self.y + self.height) or (xAxisCrossesAt < self.y)):
   396	            y = self.y
   397	        else:
   398	            y = xAxisCrossesAt
   399
   400	        xA.setPosition(self.x, y, self.width)
   401	        xA.configure(self.data)
   402	        self.calcPositions()
   403	        g = Group()
   404	        g.add(self.makeBackground())
   405	        if self._inFill or self.behindAxes:
   406	            xA._joinToAxis()
   407	            if self._inFill:
   408	                self._inFillG = Group()
   409	                g.add(self._inFillG)
   410	            if self.behindAxes:
   411	                self._lineG = Group()
   412	                g.add(self._lineG)
   413	        xA._joinToAxis()
   414	        yA._joinToAxis()
   415	        xAex = xA.visibleAxis and [xA._y] or []
   416	        yAex = yA.visibleAxis and [yA._x] or []
   417	        skipGrid = getattr(xA,'skipGrid','none')
   418	        if skipGrid!=None:
   419	            if skipGrid in ('both','top'):
   420	                yAex.append(xA._x+xA._length)
   421	            if skipGrid in ('both','bottom'):
   422	                yAex.append(xA._x)
   423	        skipGrid = getattr(yA,'skipGrid','none')
   424	        if skipGrid!=None:
   425	            if skipGrid in ('both','top'):
   426	                xAex.append(yA._y+yA._length)
   427	            if skipGrid in ('both','bottom'):
   428	                xAex.append(yA._y)
   429	        if self.gridFirst:
   430	            xA.makeGrid(g,parent=self,dim=yA.getGridDims,exclude=yAex)
   431	            yA.makeGrid(g,parent=self,dim=xA.getGridDims,exclude=xAex)
   432	        g.add(xA.draw())
   433	        g.add(yA.draw())
   434	        if not self.gridFirst:
   435	            xAdgl = getattr(xA,'drawGridLast',False)
   436	            yAdgl = getattr(yA,'drawGridLast',False)
   437	            if not xAdgl: xA.makeGrid(g,parent=self,dim=yA.getGridDims,exclude=yAex)
   438	            if not yAdgl: yA.makeGrid(g,parent=self,dim=xA.getGridDims,exclude=xAex)
   439	        annotations = getattr(self,'annotations',[])
   440	        for a in annotations:
   441	            if getattr(a,'beforeLines',None):
   442	                g.add(a(self,xA.scale,yA.scale))
   443	        g.add(self.makeLines())
   444	        if not self.gridFirst:
   445	            if xAdgl: xA.makeGrid(g,parent=self,dim=yA.getGridDims,exclude=yAex)
   446	            if yAdgl: yA.makeGrid(g,parent=self,dim=xA.getGridDims,exclude=xAex)
   447	        for a in annotations:
   448	            if not getattr(a,'beforeLines',None):
   449	                g.add(a(self,xA.scale,yA.scale))
   450	        return g
