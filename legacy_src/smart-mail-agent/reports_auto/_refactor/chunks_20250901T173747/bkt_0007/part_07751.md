   417	    def _bezierArcSegmentCCW(self, cx,cy, rx,ry, theta0, theta1):
   418	        """compute the control points for a bezier arc with theta1-theta0 <= 90.
   419	        Points are computed for an arc with angle theta increasing in the
   420	        counter-clockwise (CCW) direction.  returns a tuple with starting point
   421	        and 3 control points of a cubic bezier curve for the curvto opertator"""
   422
   423	        # Requires theta1 - theta0 <= 90 for a good approximation
   424	        assert abs(theta1 - theta0) <= 90
   425	        cos0 = cos(pi*theta0/180.0)
   426	        sin0 = sin(pi*theta0/180.0)
   427	        x0 = cx + rx*cos0
   428	        y0 = cy + ry*sin0
   429
   430	        cos1 = cos(pi*theta1/180.0)
   431	        sin1 = sin(pi*theta1/180.0)
   432
   433	        x3 = cx + rx*cos1
   434	        y3 = cy + ry*sin1
   435
   436	        dx1 = -rx * sin0
   437	        dy1 = ry * cos0
   438
   439	        #from pdfgeom
   440	        halfAng = pi*(theta1-theta0)/(2.0 * 180.0)
   441	        k = abs(4.0 / 3.0 * (1.0 - cos(halfAng) ) /(sin(halfAng)) )
   442	        x1 = x0 + dx1 * k
   443	        y1 = y0 + dy1 * k
   444
   445	        dx2 = -rx * sin1
   446	        dy2 = ry * cos1
   447
   448	        x2 = x3 - dx2 * k
   449	        y2 = y3 - dy2 * k
   450	        return ((x0,y0), ((x1,y1), (x2,y2), (x3,y3)) )

   452	    def bezierArcCCW(self, cx,cy, rx,ry, theta0, theta1):
   453	        """return a set of control points for Bezier approximation to an arc
   454	        with angle increasing counter clockwise. No requirement on (theta1-theta0) <= 90
   455	        However, it must be true that theta1-theta0 > 0."""
   456
   457	        # I believe this is also clockwise
   458	        # pretty much just like Robert Kern's pdfgeom.BezierArc
   459	        angularExtent = theta1 - theta0
   460	        # break down the arc into fragments of <=90 degrees
   461	        if abs(angularExtent) <= 90.0:  # we just need one fragment
   462	            angleList = [(theta0,theta1)]
   463	        else:
   464	            Nfrag = int( ceil( abs(angularExtent)/90.) )
   465	            fragAngle = float(angularExtent)/ Nfrag  # this could be negative
   466	            angleList = []
   467	            for ii in range(Nfrag):
   468	                a = theta0 + ii * fragAngle
   469	                b = a + fragAngle # hmm.. is I wonder if this is precise enought
   470	                angleList.append((a,b))
   471
   472	        ctrlpts = []
   473	        for (a,b) in angleList:
   474	            if not ctrlpts: # first time
   475	                [(x0,y0), pts] = self._bezierArcSegmentCCW(cx,cy, rx,ry, a,b)
   476	                ctrlpts.append(pts)
   477	            else:
   478	                [(tmpx,tmpy), pts] = self._bezierArcSegmentCCW(cx,cy, rx,ry, a,b)
   479	                ctrlpts.append(pts)
   480	        return ((x0,y0), ctrlpts)

   587	    def ellipse(self, cx,cy,rx,ry):
   588	        """add closed path ellipse with center cx,cy and axes rx,ry: counter-clockwise orientation
   589	        (remember y-axis increases downward) """
   590	        self.pathBegin()
   591	        # first segment
   592	        x0 = cx + rx   # (x0,y0) start pt
   593	        y0 = cy
   594
   595	        x3 = cx        # (x3,y3) end pt of arc
   596	        y3 = cy-ry
   597
   598	        x1 = cx+rx
   599	        y1 = cy-ry*BEZIER_ARC_MAGIC
   600
   601	        x2 = x3 + rx*BEZIER_ARC_MAGIC
   602	        y2 = y3
   603	        self.moveTo(x0, y0)
   604	        self.curveTo(x1,y1,x2,y2,x3,y3)
   605	        # next segment
   606	        x0 = x3
   607	        y0 = y3
   608
   609	        x3 = cx-rx
   610	        y3 = cy
   611
   612	        x1 = cx-rx*BEZIER_ARC_MAGIC
   613	        y1 = cy-ry
   614
   615	        x2 = x3
   616	        y2 = cy- ry*BEZIER_ARC_MAGIC
   617	        self.curveTo(x1,y1,x2,y2,x3,y3)
   618	        # next segment
   619	        x0 = x3
   620	        y0 = y3
   621
   622	        x3 = cx
   623	        y3 = cy+ry
   624
   625	        x1 = cx-rx
   626	        y1 = cy+ry*BEZIER_ARC_MAGIC

   394	    def drawArc(self, x1,y1, x2,y2, startAng=0, extent=360, fromcenter=0):
   395	        """Draw a partial ellipse inscribed within the rectangle x1,y1,x2,y2,
   396	        starting at startAng degrees and covering extent degrees.   Angles
   397	        start with 0 to the right (+x) and increase counter-clockwise.
   398	        These should have x1<x2 and y1<y2."""
   399	        #calculate centre of ellipse
   400	        #print "x1,y1,x2,y2,startAng,extent,fromcenter", x1,y1,x2,y2,startAng,extent,fromcenter
   401	        cx, cy = (x1+x2)/2.0, (y1+y2)/2.0
   402	        rx, ry = (x2-x1)/2.0, (y2-y1)/2.0
   403
   404	        codeline = self._genArcCode(x1, y1, x2, y2, startAng, extent)
   405
   406	        startAngleRadians = math.pi*startAng/180.0
   407	        extentRadians = math.pi*extent/180.0
   408	        endAngleRadians = startAngleRadians + extentRadians
   409
   410	        codelineAppended = 0
   411
   412	        # fill portion
   413
   414	        if self._fillColor != None:
   415	            self.setColor(self._fillColor)
   416	            self.code_append(codeline)
   417	            codelineAppended = 1
   418	            if self._strokeColor!=None: self.code_append('gsave')
   419	            self.lineTo(cx,cy)
   420	            self.code_append('eofill')
   421	            if self._strokeColor!=None: self.code_append('grestore')
   422
   423	        # stroke portion
   424	        if self._strokeColor != None:
   425	            # this is a bit hacked up.  There is certainly a better way...
   426	            self.setColor(self._strokeColor)
   427	            (startx, starty) = (cx+rx*math.cos(startAngleRadians), cy+ry*math.sin(startAngleRadians))
   428	            if not codelineAppended:
   429	                self.code_append(codeline)
   430	            if fromcenter:
   431	                # move to center
   432	                self.lineTo(cx,cy)
   433	                self.lineTo(startx, starty)
   434	                self.code_append('closepath')
   435	            self.code_append('stroke')

   437	    def _genArcCode(self, x1, y1, x2, y2, startAng, extent):
   438	        "Calculate the path for an arc inscribed in rectangle defined by (x1,y1),(x2,y2)"
   439	        #calculate semi-minor and semi-major axes of ellipse
   440	        xScale = abs((x2-x1)/2.0)
   441	        yScale = abs((y2-y1)/2.0)
   442	        #calculate centre of ellipse
   443	        x, y = (x1+x2)/2.0, (y1+y2)/2.0
   444
   445	        codeline = 'matrix currentmatrix %s %s translate %s %s scale 0 0 1 %s %s %s setmatrix'
   446
   447	        if extent >= 0:
   448	            arc='arc'
   449	        else:
   450	            arc='arcn'
   451	        data = (x,y, xScale, yScale, startAng, startAng+extent, arc)
   452
   453	        return codeline % data

   569	    def _drawImageLevel1(self, image, x1, y1, width=None, height=None):
   570	        # Postscript Level1 version available for fallback mode when Level2 doesn't work
   571	        # For now let's start with 24 bit RGB images (following piddlePDF again)
   572	        component_depth = 8
   573	        myimage = image.convert('RGB')
   574	        imgwidth, imgheight = myimage.size
   575	        if not width:
   576	            width = imgwidth
   577	        if not height:
   578	            height = imgheight
   579	        #print 'Image size (%d, %d); Draw size (%d, %d)' % (imgwidth, imgheight, width, height)

   581
   582	        # "image operators assume that they receive sample data from
   583	        # their data source in x-axis major index order.  The coordinate
   584	        # of the lower-left corner of the first sample is (0,0), of the
   585	        # second (1,0) and so on" -PS2 ref manual p. 215
   586	        #
   587	        # The ImageMatrix maps unit squre of user space to boundary of the source image
   588	        #
   589
   590	        # The CurrentTransformationMatrix (CTM) maps the unit square of
   591	        # user space to the rect...on the page that is to receive the
   592	        # image. A common ImageMatrix is [width 0 0 -height 0 height]
   593	        # (for a left to right, top to bottom image )
   594
   595	        # first let's map the user coordinates start at offset x1,y1 on page
   596
   597	        self.code.extend([
   598	            'gsave',
   599	            '%s %s translate' % (x1,y1), # need to start are lower left of image
   600	            '%s %s scale' % (width,height),
   601	            '/scanline %d 3 mul string def' % imgwidth  # scanline by multiples of image width
   602	            ])
   603
   604	        # now push the dimensions and depth info onto the stack
   605	        # and push the ImageMatrix to map the source to the target rectangle (see above)
   606	        # finally specify source (PS2 pp. 225 ) and by exmample
   607	        self.code.extend([
   608	            '%s %s %s' % (imgwidth, imgheight, component_depth),
   609	            '[%s %s %s %s %s %s]' % (imgwidth, 0, 0, -imgheight, 0, imgheight),
   610	            '{ currentfile scanline readhexstring pop } false 3',
   611	            'colorimage '
   612	            ])
   613
   614	        # data source output--now we just need to deliver a hex encode
   615	        # series of lines of the right overall size can follow
   616	        # piddlePDF again
   617	        rawimage = (myimage.tobytes if hasattr(myimage,'tobytes') else myimage.tostring)()
   618	        hex_encoded = self._AsciiHexEncode(rawimage)
   619
   620	        # write in blocks of 78 chars per line
   621	        outstream = StringIO(hex_encoded)
   622
   623	        dataline = outstream.read(78)
   624	        while dataline != "":
   625	            self.code_append(dataline)
   626	            dataline= outstream.read(78)
   627	        self.code_append('% end of image data') # for clarity
   628	        self.code_append('grestore') # return coordinates to normal

   583	    def drawArc(self, x1,y1, x2,y2, startAng=0, extent=360, fromcenter=0):
   584	        """Draw a partial ellipse inscribed within the rectangle x1,y1,x2,y2.
   585
   586	        Starting at startAng degrees and covering extent degrees. Angles
   587	        start with 0 to the right (+x) and increase counter-clockwise.
   588	        These should have x1<x2 and y1<y2.
   589	        """
   590
   591	        cx, cy = (x1+x2)/2.0, (y1+y2)/2.0
   592	        rx, ry = (x2-x1)/2.0, (y2-y1)/2.0
   593	        mx = rx * cos(startAng*pi/180) + cx
   594	        my = ry * sin(startAng*pi/180) + cy
   595	        ax = rx * cos((startAng+extent)*pi/180) + cx
   596	        ay = ry * sin((startAng+extent)*pi/180) + cy
   597
   598	        cfp_str = self.cfp_str
   599	        s = [].append
   600	        if fromcenter:
   601	            s("M %s L %s" % (cfp_str(cx, cy), cfp_str(ax, ay)))
   602
   603	        if fromcenter:
   604	            s("A %s %d %d %d %s" % \
   605	              (cfp_str(rx, ry), 0, extent>=180, 0, cfp_str(mx, my)))
   606	        else:
   607	            s("M %s A %s %d %d %d %s Z" % \
   608	              (cfp_str(mx, my), cfp_str(rx, ry), 0, extent>=180, 0, cfp_str(mx, my)))
   609
   610	        if fromcenter:
   611	            s("L %s Z" % cfp_str(cx, cy))
   612
   613	        path = transformNode(self.doc, "path",
   614	            d=' '.join(s.__self__), style=self._formatStyle())
   615	        self.currGroup.appendChild(path)
