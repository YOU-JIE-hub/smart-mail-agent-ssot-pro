   125	    file will not be deleted.

   140	def info_formatter(info: Iterable[tuple[str, Any]]) -> Iterator[str]:
   141	    """Produce a sequence of formatted lines from info.
   142
   143	    `info` is a sequence of pairs (label, data).  The produced lines are
   144	    nicely formatted, ready to print.
   145
   146	    """
   147	    info = list(info)
   148	    if not info:
   149	        return
   150	    label_len = 30
   151	    assert all(len(l) < label_len for l, _ in info)
   152	    for label, data in info:
   153	        if data == []:
   154	            data = "-none-"
   155	        if isinstance(data, tuple) and len(repr(tuple(data))) < 30:
   156	            # Convert to tuple to scrub namedtuples.
   157	            yield "%*s: %r" % (label_len, label, tuple(data))
   158	        elif isinstance(data, (list, set, tuple)):
   159	            prefix = "%*s:" % (label_len, label)
   160	            for e in data:
   161	                yield "%*s %s" % (label_len+1, prefix, e)
   162	                prefix = ""
   163	        else:
   164	            yield "%*s: %s" % (label_len, label, data)

   234	def short_stack(
   235	    skip: int = 0,
   236	    full: bool = False,
   237	    frame_ids: bool = False,
   238	    short_filenames: bool = False,
   239	) -> str:
   240	    """Return a string summarizing the call stack.
   241
   242	    The string is multi-line, with one line per stack frame. Each line shows
   243	    the function name, the file name, and the line number:
   244
   245	        ...
   246	        start_import_stop : /Users/ned/coverage/trunk/tests/coveragetest.py:95
   247	        import_local_file : /Users/ned/coverage/trunk/tests/coveragetest.py:81
   248	        import_local_file : /Users/ned/coverage/trunk/coverage/backward.py:159
   249	        ...
   250
   251	    `skip` is the number of closest immediate frames to skip, so that debugging
   252	    functions can call this and not be included in the result.
   253
   254	    If `full` is true, then include all frames.  Otherwise, initial "boring"
   255	    frames (ones in site-packages and earlier) are omitted.
   256
   257	    `short_filenames` will shorten filenames using `short_filename`, to reduce
   258	    the amount of repetitive noise in stack traces.
   259
   260	    """
   261	    # Regexes in initial frames that we don't care about.
   262	    BORING_PRELUDE = [
   263	        "<string>",             # pytest-xdist has string execution.
   264	        r"\bigor.py$",          # Our test runner.
   265	        r"\bsite-packages\b",   # pytest etc getting to our tests.
   266	    ]
   267
   268	    stack: Iterable[inspect.FrameInfo] = inspect.stack()[:skip:-1]
   269	    if not full:
   270	        for pat in BORING_PRELUDE:
   271	            stack = itertools.dropwhile(
   272	                (lambda fi, pat=pat: re.search(pat, fi.filename)),  # type: ignore[misc]
   273	                stack,
   274	            )
   275	    lines = []
   276	    for frame_info in stack:
   277	        line = f"{frame_info.function:>30s} : "
   278	        if frame_ids:
   279	            line += f"{id(frame_info.frame):#x} "
   280	        filename = frame_info.filename
   281	        if short_filenames:
   282	            filename = short_filename(filename)
   283	        line += f"{filename}:{frame_info.lineno}"
   284	        lines.append(line)
   285	    return "\n".join(lines)

   351	def filter_text(text: str, filters: Iterable[Callable[[str], str]]) -> str:
   352	    """Run `text` through a series of filters.
   353
   354	    `filters` is a list of functions. Each takes a string and returns a
   355	    string.  Each is run in turn. After each filter, the text is split into
   356	    lines, and each line is passed through the next filter.
   357
   358	    Returns: the final string that results after all of the filters have
   359	    run.
   360
   361	    """
   362	    clean_text = text.rstrip()
   363	    ending = text[len(clean_text):]
   364	    text = clean_text
   365	    for filter_fn in filters:
   366	        lines = []
   367	        for line in text.splitlines():
   368	            lines.extend(filter_fn(line).splitlines())
   369	        text = "\n".join(lines)
   370	    return text + ending

   429	class DebugOutputFile:
   430	    """A file-like object that includes pid and cwd information."""
   431	    def __init__(
   432	        self,
   433	        outfile: IO[str] | None,
   434	        filters: Iterable[Callable[[str], str]],
   435	    ):
   436	        self.outfile = outfile
   437	        self.filters = list(filters)
   438	        self.pid = os.getpid()
   439
   440	    @classmethod
   441	    def get_one(
   442	        cls,
   443	        fileobj: IO[str] | None = None,
   444	        file_name: str | None = None,
   445	        filters: Iterable[Callable[[str], str]] = (),
   446	        interim: bool = False,
   447	    ) -> DebugOutputFile:
   448	        """Get a DebugOutputFile.
   449
   450	        If `fileobj` is provided, then a new DebugOutputFile is made with it.
   451
   452	        If `fileobj` isn't provided, then a file is chosen (`file_name` if
   453	        provided, or COVERAGE_DEBUG_FILE, or stderr), and a process-wide
   454	        singleton DebugOutputFile is made.
   455
   456	        `filters` are the text filters to apply to the stream to annotate with
   457	        pids, etc.
   458
   459	        If `interim` is true, then a future `get_one` can replace this one.
   460
   461	        """
   462	        if fileobj is not None:
   463	            # Make DebugOutputFile around the fileobj passed.
   464	            return cls(fileobj, filters)
   465
   466	        the_one, is_interim = cls._get_singleton_data()
   467	        if the_one is None or is_interim:
   468	            if file_name is not None:
   469	                fileobj = open(file_name, "a", encoding="utf-8")
   470	            else:
   471	                # $set_env.py: COVERAGE_DEBUG_FILE - Where to write debug output
   472	                file_name = os.getenv("COVERAGE_DEBUG_FILE", FORCED_DEBUG_FILE)
   473	                if file_name in ["stdout", "stderr"]:
   474	                    fileobj = getattr(sys, file_name)
   475	                elif file_name:
   476	                    fileobj = open(file_name, "a", encoding="utf-8")
   477	                    atexit.register(fileobj.close)
   478	                else:
   479	                    fileobj = sys.stderr
   480	            the_one = cls(fileobj, filters)
   481	            cls._set_singleton_data(the_one, interim)
   482
   483	        if not(the_one.filters):
   484	            the_one.filters = list(filters)
   485	        return the_one
   486
   487	    # Because of the way igor.py deletes and re-imports modules,
   488	    # this class can be defined more than once. But we really want
   489	    # a process-wide singleton. So stash it in sys.modules instead of
   490	    # on a class attribute. Yes, this is aggressively gross.
   491
   492	    SYS_MOD_NAME: Final[str] = "$coverage.debug.DebugOutputFile.the_one"
   493	    SINGLETON_ATTR: Final[str] = "the_one_and_is_interim"
   494
   495	    @classmethod
   496	    def _set_singleton_data(cls, the_one: DebugOutputFile, interim: bool) -> None:
   497	        """Set the one DebugOutputFile to rule them all."""
   498	        singleton_module = types.ModuleType(cls.SYS_MOD_NAME)
   499	        setattr(singleton_module, cls.SINGLETON_ATTR, (the_one, interim))
   500	        sys.modules[cls.SYS_MOD_NAME] = singleton_module
   501
   502	    @classmethod
   503	    def _get_singleton_data(cls) -> tuple[DebugOutputFile | None, bool]:
   504	        """Get the one DebugOutputFile."""
   505	        singleton_module = sys.modules.get(cls.SYS_MOD_NAME)
   506	        return getattr(singleton_module, cls.SINGLETON_ATTR, (None, True))
   507
   508	    @classmethod
   509	    def _del_singleton_data(cls) -> None:
   510	        """Delete the one DebugOutputFile, just for tests to use."""
   511	        if cls.SYS_MOD_NAME in sys.modules:
   512	            del sys.modules[cls.SYS_MOD_NAME]
   513
   514	    def write(self, text: str) -> None:
   515	        """Just like file.write, but filter through all our filters."""
   516	        assert self.outfile is not None
   517	        if not self.outfile.closed:
   518	            self.outfile.write(filter_text(text, self.filters))
   519	            self.outfile.flush()
   520
   521	    def flush(self) -> None:
   522	        """Flush our file."""
   523	        assert self.outfile is not None
   524	        if not self.outfile.closed:
   525	            self.outfile.flush()

   308	# Tokenizer for _glob_to_regex.
   309	# None as a sub means disallowed.
   310	G2RX_TOKENS = [(re.compile(rx), sub) for rx, sub in [
   311	    (r"\*\*\*+", None),             # Can't have ***
   312	    (r"[^/]+\*\*+", None),          # Can't have x**
   313	    (r"\*\*+[^/]+", None),          # Can't have **x
   314	    (r"\*\*/\*\*", None),           # Can't have **/**
   315	    (r"^\*+/", r"(.*[/\\\\])?"),    # ^*/ matches any prefix-slash, or nothing.
   316	    (r"/\*+$", r"[/\\\\].*"),       # /*$ matches any slash-suffix.
   317	    (r"\*\*/", r"(.*[/\\\\])?"),    # **/ matches any subdirs, including none
   318	    (r"/", r"[/\\\\]"),             # / matches either slash or backslash
   319	    (r"\*", r"[^/\\\\]*"),          # * matches any number of non slash-likes
   320	    (r"\?", r"[^/\\\\]"),           # ? matches one non slash-like
   321	    (r"\[.*?\]", r"\g<0>"),         # [a-f] matches [a-f]
   322	    (r"[a-zA-Z0-9_-]+", r"\g<0>"),  # word chars match themselves
   323	    (r"[\[\]]", None),              # Can't have single square brackets
   324	    (r".", r"\\\g<0>"),             # Anything else is escaped to be safe
   325	]]

   327	def _glob_to_regex(pattern: str) -> str:
   328	    """Convert a file-path glob pattern into a regex."""
   329	    # Turn all backslashes into slashes to simplify the tokenizer.
   330	    pattern = pattern.replace("\\", "/")
   331	    if "/" not in pattern:
   332	        pattern = f"**/{pattern}"
   333	    path_rx = []
   334	    pos = 0
   335	    while pos < len(pattern):
   336	        for rx, sub in G2RX_TOKENS:                     # pragma: always breaks
   337	            if m := rx.match(pattern, pos=pos):
   338	                if sub is None:
   339	                    raise ConfigError(f"File pattern can't include {m[0]!r}")
   340	                path_rx.append(m.expand(sub))
   341	                pos = m.end()
   342	                break
   343	    return "".join(path_rx)

   346	def globs_to_regex(
   347	    patterns: Iterable[str],
   348	    case_insensitive: bool = False,
   349	    partial: bool = False,
   350	) -> re.Pattern[str]:
   351	    """Convert glob patterns to a compiled regex that matches any of them.
   352
   353	    Slashes are always converted to match either slash or backslash, for
   354	    Windows support, even when running elsewhere.
   355
   356	    If the pattern has no slash or backslash, then it is interpreted as
   357	    matching a file name anywhere it appears in the tree.  Otherwise, the glob
   358	    pattern must match the whole file path.
   359
   360	    If `partial` is true, then the pattern will match if the target string
   361	    starts with the pattern. Otherwise, it must match the entire string.
   362
   363	    Returns: a compiled regex object.  Use the .match method to compare target
   364	    strings.
   365
   366	    """
   367	    flags = 0
   368	    if case_insensitive:
   369	        flags |= re.IGNORECASE
   370	    rx = join_regex(map(_glob_to_regex, patterns))
   371	    if not partial:
   372	        rx = fr"(?:{rx})\Z"
   373	    compiled = re.compile(rx, flags=flags)
   374	    return compiled

   377	class PathAliases:
   378	    """A collection of aliases for paths.
   379
   380	    When combining data files from remote machines, often the paths to source
   381	    code are different, for example, due to OS differences, or because of
   382	    serialized checkouts on continuous integration machines.
   383
   384	    A `PathAliases` object tracks a list of pattern/result pairs, and can
   385	    map a path through those aliases to produce a unified path.
   386
   387	    """
   388	    def __init__(
   389	        self,
   390	        debugfn: Callable[[str], None] | None = None,
   391	        relative: bool = False,
   392	    ) -> None:
   393	        # A list of (original_pattern, regex, result)
   394	        self.aliases: list[tuple[str, re.Pattern[str], str]] = []
   395	        self.debugfn = debugfn or (lambda msg: 0)
   396	        self.relative = relative
   397	        self.pprinted = False
   398
   399	    def pprint(self) -> None:
   400	        """Dump the important parts of the PathAliases, for debugging."""
   401	        self.debugfn(f"Aliases (relative={self.relative}):")
   402	        for original_pattern, regex, result in self.aliases:
   403	            self.debugfn(f" Rule: {original_pattern!r} -> {result!r} using regex {regex.pattern!r}")
   404
   405	    def add(self, pattern: str, result: str) -> None:
   406	        """Add the `pattern`/`result` pair to the list of aliases.
   407
   408	        `pattern` is an `glob`-style pattern.  `result` is a simple
   409	        string.  When mapping paths, if a path starts with a match against
   410	        `pattern`, then that match is replaced with `result`.  This models
   411	        isomorphic source trees being rooted at different places on two
   412	        different machines.
   413
   414	        `pattern` can't end with a wildcard component, since that would
   415	        match an entire tree, and not just its root.
   416
   417	        """
   418	        original_pattern = pattern
   419	        pattern_sep = sep(pattern)
   420
   421	        if len(pattern) > 1:
   422	            pattern = pattern.rstrip(r"\/")
   423
   424	        # The pattern can't end with a wildcard component.
   425	        if pattern.endswith("*"):
   426	            raise ConfigError("Pattern must not end with wildcards.")
   427
   428	        # The pattern is meant to match a file path.  Let's make it absolute
   429	        # unless it already is, or is meant to match any prefix.
   430	        if not self.relative:
   431	            if not pattern.startswith("*") and not isabs_anywhere(pattern + pattern_sep):
   432	                pattern = abs_file(pattern)
   433	        if not pattern.endswith(pattern_sep):
   434	            pattern += pattern_sep
   435
   436	        # Make a regex from the pattern.
   437	        regex = globs_to_regex([pattern], case_insensitive=True, partial=True)
   438
   439	        # Normalize the result: it must end with a path separator.
   440	        result_sep = sep(result)
   441	        result = result.rstrip(r"\/") + result_sep
   442	        self.aliases.append((original_pattern, regex, result))
   443
   444	    def map(self, path: str, exists:Callable[[str], bool] = source_exists) -> str:
   445	        """Map `path` through the aliases.
   446
   447	        `path` is checked against all of the patterns.  The first pattern to
   448	        match is used to replace the root of the path with the result root.
   449	        Only one pattern is ever used.  If no patterns match, `path` is
   450	        returned unchanged.
   451
   452	        The separator style in the result is made to match that of the result
   453	        in the alias.
   454
   455	        `exists` is a function to determine if the resulting path actually
   456	        exists.
   457
   458	        Returns the mapped path.  If a mapping has happened, this is a
   459	        canonical path.  If no mapping has happened, it is the original value
   460	        of `path` unchanged.
   461
   462	        """
   463	        if not self.pprinted:
   464	            self.pprint()
   465	            self.pprinted = True
   466
   467	        for original_pattern, regex, result in self.aliases:
   468	            if m := regex.match(path):
   469	                new = path.replace(m[0], result)
   470	                new = new.replace(sep(path), sep(result))
   471	                if not self.relative:
   472	                    new = canonical_filename(new)
   473	                dot_start = result.startswith(("./", ".\\")) and len(result) > 2
   474	                if new.startswith(("./", ".\\")) and not dot_start:
   475	                    new = new[2:]
   476	                if not exists(new):
   477	                    self.debugfn(
   478	                        f"Rule {original_pattern!r} changed {path!r} to {new!r} " +
   479	                        "which doesn't exist, continuing",
   480	                    )
   481	                    continue
   482	                self.debugfn(
   483	                    f"Matched path {path!r} to rule {original_pattern!r} -> {result!r}, " +
   484	                    f"producing {new!r}",
   485	                )
   486	                return new
   487
   488	        # If we get here, no pattern matched.
   489
   490	        if self.relative:
   491	            path = relative_filename(path)
   492
   493	        if self.relative and not isabs_anywhere(path):
   494	            # Auto-generate a pattern to implicitly match relative files
   495	            parts = re.split(r"[/\\]", path)
   496	            if len(parts) > 1:
   497	                dir1 = parts[0]
   498	                pattern = f"*/{dir1}"
   499	                regex_pat = fr"^(.*[\\/])?{re.escape(dir1)}[\\/]"
   500	                result = f"{dir1}{os.sep}"
   501	                # Only add a new pattern if we don't already have this pattern.
   502	                if not any(p == pattern for p, _, _ in self.aliases):
   503	                    self.debugfn(
   504	                        f"Generating rule: {pattern!r} -> {result!r} using regex {regex_pat!r}",
   505	                    )
   506	                    self.aliases.append((pattern, re.compile(regex_pat), result))
   507	                    return self.map(path, exists=exists)
   508
   509	        self.debugfn(f"No rules match, path {path!r} is unchanged")
   510	        return path

   513	def find_python_files(dirname: str, include_namespace_packages: bool) -> Iterable[str]:
   514	    """Yield all of the importable Python files in `dirname`, recursively.
   515
   516	    To be importable, the files have to be in a directory with a __init__.py,
   517	    except for `dirname` itself, which isn't required to have one.  The
   518	    assumption is that `dirname` was specified directly, so the user knows
   519	    best, but sub-directories are checked for a __init__.py to be sure we only
   520	    find the importable files.
   521
   522	    If `include_namespace_packages` is True, then the check for __init__.py
   523	    files is skipped.
   524

   528	    """
   529	    for i, (dirpath, dirnames, filenames) in enumerate(os.walk(dirname)):
   530	        if not include_namespace_packages:
   531	            if i > 0 and "__init__.py" not in filenames:
   532	                # If a directory doesn't have __init__.py, then it isn't
   533	                # importable and neither are its files
   534	                del dirnames[:]
   535	                continue
   536	        for filename in filenames:
   537	            # We're only interested in files that look like reasonable Python
   538	            # files: Must end with .py or .pyw, and must not have certain funny
   539	            # characters that probably mean they are editor junk.
   540	            if re.match(r"^[^.#~!$@%^&*()+=,]+\.pyw?$", filename):
   541	                yield os.path.join(dirpath, filename)
