   130	class Frame:
   131	    """Wrapper around a Python frame holding f_locals and f_globals
   132	    in which expressions can be evaluated."""
   133
   134	    __slots__ = ("raw",)
   135
   136	    def __init__(self, frame: FrameType) -> None:
   137	        self.raw = frame
   138
   139	    @property
   140	    def lineno(self) -> int:
   141	        return self.raw.f_lineno - 1
   142
   143	    @property
   144	    def f_globals(self) -> dict[str, Any]:
   145	        return self.raw.f_globals
   146
   147	    @property
   148	    def f_locals(self) -> dict[str, Any]:
   149	        return self.raw.f_locals

   151	    @property
   152	    def code(self) -> Code:
   153	        return Code(self.raw.f_code)

   155	    @property
   156	    def statement(self) -> Source:
   157	        """Statement this frame is at."""
   158	        if self.code.fullsource is None:
   159	            return Source("")

   162	    def eval(self, code, **vars):
   163	        """Evaluate 'code' in the frame.
   164
   165	        'vars' are optional additional local variables.
   166
   167	        Returns the result of the evaluation.
   168	        """
   169	        f_locals = self.f_locals.copy()
   170	        f_locals.update(vars)
   171	        return eval(code, self.f_globals, f_locals)

   173	    def repr(self, object: object) -> str:
   174	        """Return a 'safe' (non-recursive, one-line) string repr for 'object'."""
   175	        return saferepr(object)

   177	    def getargs(self, var: bool = False):
   178	        """Return a list of tuples (name, value) for all arguments.
   179
   180	        If 'var' is set True, also include the variable and keyword arguments
   181	        when present.
   182	        """
   183	        retval = []
   184	        for arg in self.code.getargs(var):
   185	            try:
   186	                retval.append((arg, self.f_locals[arg]))
   187	            except KeyError:
   188	                pass  # this can occur when using Psyco
   189	        return retval

   284	    def getsource(
   285	        self, astcache: dict[str | Path, ast.AST] | None = None
   286	    ) -> Source | None:
   287	        """Return failing source code."""
   288	        # we use the passed in astcache to not reparse asttrees
   289	        # within exception info printing
   290	        source = self.frame.code.fullsource
   291	        if source is None:
   292	            return None
   293	        key = astnode = None
   294	        if astcache is not None:
   295	            key = self.frame.code.path
   296	            if key is not None:
   297	                astnode = astcache.get(key, None)
   298	        start = self.getfirstlinesource()
   299	        try:
   300	            astnode, _, end = getstatementrange_ast(
   301	                self.lineno, source, astnode=astnode
   302	            )
   303	        except SyntaxError:
   304	            end = self.lineno + 1
   305	        else:
   306	            if key is not None and astcache is not None:
   307	                astcache[key] = astnode
   308	        return source[start:end]

   312	    def ishidden(self, excinfo: ExceptionInfo[BaseException] | None) -> bool:
   313	        """Return True if the current frame has a var __tracebackhide__
   314	        resolving to True.
   315
   316	        If __tracebackhide__ is a callable, it gets called with the
   317	        ExceptionInfo instance and can decide whether to hide the traceback.
   318
   319	        Mostly for internal use.
   320	        """
   321	        tbh: bool | Callable[[ExceptionInfo[BaseException] | None], bool] = False
   322	        for maybe_ns_dct in (self.frame.f_locals, self.frame.f_globals):
   323	            # in normal cases, f_locals and f_globals are dictionaries
   324	            # however via `exec(...)` / `eval(...)` they can be other types
   325	            # (even incorrect types!).
   326	            # as such, we suppress all exceptions while accessing __tracebackhide__
   327	            try:
   328	                tbh = maybe_ns_dct["__tracebackhide__"]
   329	            except Exception:
   330	                pass
   331	            else:
   332	                break
   333	        if tbh and callable(tbh):
   334	            return tbh(excinfo)
   335	        return tbh

   356	class Traceback(list[TracebackEntry]):
   357	    """Traceback objects encapsulate and offer higher level access to Traceback entries."""

   359	    def __init__(
   360	        self,
   361	        tb: TracebackType | Iterable[TracebackEntry],
   362	    ) -> None:
   363	        """Initialize from given python traceback object and ExceptionInfo."""
   364	        if isinstance(tb, TracebackType):
   365
   366	            def f(cur: TracebackType) -> Iterable[TracebackEntry]:
   367	                cur_: TracebackType | None = cur
   368	                while cur_ is not None:
   369	                    yield TracebackEntry(cur_)
   370	                    cur_ = cur_.tb_next
   371
   372	            super().__init__(f(tb))
   373	        else:
   374	            super().__init__(tb)

   376	    def cut(
   377	        self,
   378	        path: os.PathLike[str] | str | None = None,
   379	        lineno: int | None = None,
   380	        firstlineno: int | None = None,
   381	        excludepath: os.PathLike[str] | None = None,
   382	    ) -> Traceback:
   383	        """Return a Traceback instance wrapping part of this Traceback.
   384
   385	        By providing any combination of path, lineno and firstlineno, the
   386	        first frame to start the to-be-returned traceback is determined.
   387
   388	        This allows cutting the first part of a Traceback instance e.g.
   389	        for formatting reasons (removing some uninteresting bits that deal
   390	        with handling of the exception/traceback).
   391	        """
   392	        path_ = None if path is None else os.fspath(path)
   393	        excludepath_ = None if excludepath is None else os.fspath(excludepath)
   394	        for x in self:
   395	            code = x.frame.code
   396	            codepath = code.path
   397	            if path is not None and str(codepath) != path_:
   398	                continue
   399	            if (
   400	                excludepath is not None
   401	                and isinstance(codepath, Path)
   402	                and excludepath_ in (str(p) for p in codepath.parents)  # type: ignore[operator]
   403	            ):
   404	                continue
   405	            if lineno is not None and x.lineno != lineno:
   406	                continue
   407	            if firstlineno is not None and x.frame.code.firstlineno != firstlineno:
   408	                continue
   409	            return Traceback(x._rawentry)
   410	        return self

   424	    def filter(
   425	        self,
   426	        excinfo_or_fn: ExceptionInfo[BaseException] | Callable[[TracebackEntry], bool],
   427	        /,
   428	    ) -> Traceback:
   429	        """Return a Traceback instance with certain items removed.
   430
   431	        If the filter is an `ExceptionInfo`, removes all the ``TracebackEntry``s
   432	        which are hidden (see ishidden() above).
   433
   434	        Otherwise, the filter is a function that gets a single argument, a
   435	        ``TracebackEntry`` instance, and should return True when the item should
   436	        be added to the ``Traceback``, False when not.
   437	        """
   438	        if isinstance(excinfo_or_fn, ExceptionInfo):
   439	            fn = lambda x: not x.ishidden(excinfo_or_fn)  # noqa: E731
   440	        else:
   441	            fn = excinfo_or_fn
   442	        return Traceback(filter(fn, self))

   444	    def recursionindex(self) -> int | None:
   445	        """Return the index of the frame/TracebackEntry where recursion originates if
   446	        appropriate, None if no recursion occurred."""
   447	        cache: dict[tuple[Any, int, int], list[dict[str, Any]]] = {}
   448	        for i, entry in enumerate(self):
   449	            # id for the code.raw is needed to work around
   450	            # the strange metaprogramming in the decorator lib from pypi
   451	            # which generates code objects that have hash/value equality
   452	            # XXX needs a test
   453	            key = entry.frame.code.path, id(entry.frame.code.raw), entry.lineno
   454	            values = cache.setdefault(key, [])
   455	            # Since Python 3.13 f_locals is a proxy, freeze it.
   456	            loc = dict(entry.frame.f_locals)
   457	            if values:
   458	                for otherloc in values:
   459	                    if otherloc == loc:
   460	                        return i
   461	            values.append(loc)
   462	        return None

   465	def stringify_exception(
   466	    exc: BaseException, include_subexception_msg: bool = True
   467	) -> str:
   468	    try:
   469	        notes = getattr(exc, "__notes__", [])
   470	    except KeyError:
   471	        # Workaround for https://github.com/python/cpython/issues/98778 on
   472	        # Python <= 3.9, and some 3.10 and 3.11 patch versions.
   473	        HTTPError = getattr(sys.modules.get("urllib.error", None), "HTTPError", ())
   474	        if sys.version_info < (3, 12) and isinstance(exc, HTTPError):
   475	            notes = []
   476	        else:  # pragma: no cover
   477	            # exception not related to above bug, reraise
   478	            raise
   479	    if not include_subexception_msg and isinstance(exc, BaseExceptionGroup):
   480	        message = exc.message
   481	    else:
   482	        message = str(exc)
   483
   484	    return "\n".join(
   485	        [
   486	            message,
   487	            *notes,
   488	        ]
   489	    )

   519	    @classmethod
   520	    def from_exception(
   521	        cls,
   522	        # Ignoring error: "Cannot use a covariant type variable as a parameter".
   523	        # This is OK to ignore because this class is (conceptually) readonly.
   524	        # See https://github.com/python/mypy/issues/7049.
   525	        exception: E,  # type: ignore[misc]
   526	        exprinfo: str | None = None,
   527	    ) -> ExceptionInfo[E]:
   528	        """Return an ExceptionInfo for an existing exception.
   529
   530	        The exception must have a non-``None`` ``__traceback__`` attribute,
   531	        otherwise this function fails with an assertion error. This means that
   532	        the exception must have been raised, or added a traceback with the
   533	        :py:meth:`~BaseException.with_traceback()` method.
   534
   535	        :param exprinfo:
   536	            A text string helping to determine if we should strip
   537	            ``AssertionError`` from the output. Defaults to the exception
   538	            message/``__str__()``.
   539
   540	        .. versionadded:: 7.4
   541	        """
   542	        assert exception.__traceback__, (
   543	            "Exceptions passed to ExcInfo.from_exception(...)"
   544	            " must have a non-None __traceback__."
   545	        )
   546	        exc_info = (type(exception), exception, exception.__traceback__)
   547	        return cls.from_exc_info(exc_info, exprinfo)

   549	    @classmethod
   550	    def from_exc_info(
   551	        cls,
   552	        exc_info: tuple[type[E], E, TracebackType],
   553	        exprinfo: str | None = None,
   554	    ) -> ExceptionInfo[E]:
   555	        """Like :func:`from_exception`, but using old-style exc_info tuple."""
   556	        _striptext = ""
   557	        if exprinfo is None and isinstance(exc_info[1], AssertionError):
   558	            exprinfo = getattr(exc_info[1], "msg", None)
   559	            if exprinfo is None:
   560	                exprinfo = saferepr(exc_info[1])
   561	            if exprinfo and exprinfo.startswith(cls._assert_start_repr):
   562	                _striptext = "AssertionError: "

   564	        return cls(exc_info, _striptext, _ispytest=True)

   570	        .. warning::
   571
   572	            Experimental API

   574	        :param exprinfo:
   575	            A text string helping to determine if we should strip

   644	    def exconly(self, tryshort: bool = False) -> str:
   645	        """Return the exception as a string.
   646
   647	        When 'tryshort' resolves to True, and the exception is an
   648	        AssertionError, only the actual exception part of the exception
   649	        representation is returned (so 'AssertionError: ' is removed from
   650	        the beginning).
   651	        """
   652
   653	        def _get_single_subexc(
   654	            eg: BaseExceptionGroup[BaseException],
   655	        ) -> BaseException | None:
   656	            if len(eg.exceptions) != 1:
   657	                return None
   658	            if isinstance(e := eg.exceptions[0], BaseExceptionGroup):
   659	                return _get_single_subexc(e)
   660	            return e
   661
   662	        if (
   663	            tryshort
   664	            and isinstance(self.value, BaseExceptionGroup)
   665	            and (subexc := _get_single_subexc(self.value)) is not None
   666	        ):
   667	            return f"{subexc!r} [single exception in {type(self.value).__name__}]"
   668
   669	        lines = format_exception_only(self.type, self.value)
   670	        text = "".join(lines)
   671	        text = text.rstrip()
   672	        if tryshort:
   673	            if text.startswith(self._striptext):
   674	                text = text[len(self._striptext) :]
   675	        return text

   684	    def _getreprcrash(self) -> ReprFileLocation | None:
   685	        # Find last non-hidden traceback entry that led to the exception of the
   686	        # traceback, or None if all hidden.
   687	        for i in range(-1, -len(self.traceback) - 1, -1):
   688	            entry = self.traceback[i]
   689	            if not entry.ishidden(self):
   690	                path, lineno = entry.frame.code.raw.co_filename, entry.lineno
   691	                exconly = self.exconly(tryshort=True)
   692	                return ReprFileLocation(path, lineno + 1, exconly)
   693	        return None

   768	    def match(self, regexp: str | re.Pattern[str]) -> Literal[True]:
   769	        """Check whether the regular expression `regexp` matches the string
   770	        representation of the exception using :func:`python:re.search`.
   771
   772	        If it matches `True` is returned, otherwise an `AssertionError` is raised.
   773	        """
   774	        __tracebackhide__ = True
   775	        value = stringify_exception(self.value)
   776	        msg = f"Regex pattern did not match.\n Regex: {regexp!r}\n Input: {value!r}"
   777	        if regexp == value:
   778	            msg += "\n Did you mean to `re.escape()` the regex?"
   779	        assert re.search(regexp, value), msg
   780	        # Return True to allow for "assert excinfo.match()".
   781	        return True

   783	    def _group_contains(
   784	        self,
   785	        exc_group: BaseExceptionGroup[BaseException],
   786	        expected_exception: EXCEPTION_OR_MORE,
   787	        match: str | re.Pattern[str] | None,
   788	        target_depth: int | None = None,
   789	        current_depth: int = 1,
   790	    ) -> bool:
   791	        """Return `True` if a `BaseExceptionGroup` contains a matching exception."""
   792	        if (target_depth is not None) and (current_depth > target_depth):
   793	            # already descended past the target depth
   794	            return False
   795	        for exc in exc_group.exceptions:
   796	            if isinstance(exc, BaseExceptionGroup):
   797	                if self._group_contains(
   798	                    exc, expected_exception, match, target_depth, current_depth + 1
   799	                ):
   800	                    return True
   801	            if (target_depth is not None) and (current_depth != target_depth):
   802	                # not at the target depth, no match
   803	                continue
   804	            if not isinstance(exc, expected_exception):
   805	                continue
   806	            if match is not None:
   807	                value = stringify_exception(exc)
   808	                if not re.search(match, value):
   809	                    continue
   810	            return True
   811	        return False

   813	    def group_contains(
   814	        self,
   815	        expected_exception: EXCEPTION_OR_MORE,
   816	        *,
   817	        match: str | re.Pattern[str] | None = None,
   818	        depth: int | None = None,
   819	    ) -> bool:
   820	        """Check whether a captured exception group contains a matching exception.
   821
   822	        :param Type[BaseException] | Tuple[Type[BaseException]] expected_exception:
   823	            The expected exception type, or a tuple if one of multiple possible
   824	            exception types are expected.

   827	            If specified, a string containing a regular expression,
   828	            or a regular expression object, that is tested against the string
   829	            representation of the exception and its `PEP-678 <https://peps.python.org/pep-0678/>` `__notes__`
   830	            using :func:`re.search`.

   833	            <re-syntax>`, the pattern can first be escaped with :func:`re.escape`.

   836	            If `None`, will search for a matching exception at any nesting depth.
   837	            If >= 1, will only match an exception if it's at the specified depth (depth = 1 being
   838	            the exceptions contained within the topmost exception group).
   839
   840	        .. versionadded:: 8.0
