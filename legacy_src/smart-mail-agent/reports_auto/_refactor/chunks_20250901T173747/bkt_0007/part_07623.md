 in left.items:
   649	            return False
   650	        if not is_overlapping(left.items[key], right.items[key]):
   651	            return False
   652
   653	    # The presence of any additional optional keys does not affect whether the two
   654	    # TypedDicts are partially overlapping: the dicts would be overlapping if the
   655	    # keys happened to be missing.
   656	    return True

   659	def are_tuples_overlapping(
   660	    left: Type, right: Type, is_overlapping: Callable[[Type, Type], bool]
   661	) -> bool:
   662	    """Returns true if left and right are overlapping tuples."""
   663	    left, right = get_proper_types((left, right))
   664	    left = adjust_tuple(left, right) or left
   665	    right = adjust_tuple(right, left) or right
   666	    assert isinstance(left, TupleType), f"Type {left} is not a tuple"
   667	    assert isinstance(right, TupleType), f"Type {right} is not a tuple"
   668
   669	    # This algorithm works well if only one tuple is variadic, if both are
   670	    # variadic we may get rare false negatives for overlapping prefix/suffix.
   671	    # Also, this ignores empty unpack case, but it is probably consistent with
   672	    # how we handle e.g. empty lists in overload overlaps.
   673	    # TODO: write a more robust algorithm for cases where both types are variadic.
   674	    left_unpack = find_unpack_in_list(left.items)
   675	    right_unpack = find_unpack_in_list(right.items)
   676	    if left_unpack is not None:
   677	        left = expand_tuple_if_possible(left, len(right.items))
   678	    if right_unpack is not None:
   679	        right = expand_tuple_if_possible(right, len(left.items))
   680
   681	    if len(left.items) != len(right.items):
   682	        return False
   683	    if not all(is_overlapping(l, r) for l, r in zip(left.items, right.items)):
   684	        return False
   685
   686	    # Check that the tuples aren't from e.g. different NamedTuples.
   687	    if is_named_instance(right.partial_fallback, "builtins.tuple") or is_named_instance(
   688	        left.partial_fallback, "builtins.tuple"
   689	    ):
   690	        return True
   691	    else:
   692	        return is_overlapping(left.partial_fallback, right.partial_fallback)
   693

   695	def expand_tuple_if_possible(tup: TupleType, target: int) -> TupleType:
   696	    if len(tup.items) > target + 1:
   697	        return tup
   698	    extra = target + 1 - len(tup.items)
   699	    new_items = []
   700	    for it in tup.items:
   701	        if not isinstance(it, UnpackType):
   702	            new_items.append(it)
   703	            continue
   704	        unpacked = get_proper_type(it.type)
   705	        if isinstance(unpacked, TypeVarTupleType):
   706	            instance = unpacked.tuple_fallback
   707	        else:
   708	            # Nested non-variadic tuples should be normalized at this point.
   709	            assert isinstance(unpacked, Instance)
   710	            instance = unpacked
   711	        assert instance.type.fullname == "builtins.tuple"
   712	        new_items.extend([instance.args[0]] * extra)
   713	    return tup.copy_modified(items=new_items)

   716	def adjust_tuple(left: ProperType, r: ProperType) -> TupleType | None:
   717	    """Find out if `left` is a Tuple[A, ...], and adjust its length to `right`"""
   718	    if isinstance(left, Instance) and left.type.fullname == "builtins.tuple":
   719	        n = r.length() if isinstance(r, TupleType) else 1
   720	        return TupleType([left.args[0]] * n, left)
   721	    return None

   727	        isinstance(typ, Instance) and typ.type.fullname == "builtins.tuple"

   822	    def visit_instance(self, t: Instance) -> ProperType:
   823	        if isinstance(self.s, Instance):
   824	            if t.type == self.s.type:
   825	                if is_subtype(t, self.s) or is_subtype(self.s, t):
   826	                    # Combine type arguments. We could have used join below
   827	                    # equivalently.
   828	                    args: list[Type] = []
   829	                    # N.B: We use zip instead of indexing because the lengths might have
   830	                    # mismatches during daemon reprocessing.
   831	                    if t.type.has_type_var_tuple_type:
   832	                        # We handle meet of variadic instances by simply creating correct mapping
   833	                        # for type arguments and compute the individual meets same as for regular
   834	                        # instances. All the heavy lifting is done in the meet of tuple types.
   835	                        s = self.s
   836	                        assert s.type.type_var_tuple_prefix is not None
   837	                        assert s.type.type_var_tuple_suffix is not None
   838	                        prefix = s.type.type_var_tuple_prefix
   839	                        suffix = s.type.type_var_tuple_suffix
   840	                        tvt = s.type.defn.type_vars[prefix]
   841	                        assert isinstance(tvt, TypeVarTupleType)
   842	                        fallback = tvt.tuple_fallback
   843	                        s_prefix, s_middle, s_suffix = split_with_prefix_and_suffix(
   844	                            s.args, prefix, suffix
   845	                        )
   846	                        t_prefix, t_middle, t_suffix = split_with_prefix_and_suffix(
   847	                            t.args, prefix, suffix
   848	                        )
   849	                        s_args = s_prefix + (TupleType(list(s_middle), fallback),) + s_suffix
   850	                        t_args = t_prefix + (TupleType(list(t_middle), fallback),) + t_suffix
   851	                    else:
   852	                        t_args = t.args
   853	                        s_args = self.s.args
   854	                    for ta, sa, tv in zip(t_args, s_args, t.type.defn.type_vars):
   855	                        meet = self.meet(ta, sa)
   856	                        if isinstance(tv, TypeVarTupleType):
   857	                            # Correctly unpack possible outcomes of meets of tuples: it can be
   858	                            # either another tuple type or Never (normalized as *tuple[Never, ...])
   859	                            if isinstance(meet, TupleType):
   860	                                args.extend(meet.items)
   861	                                continue
   862	                            else:
   863	                                assert isinstance(meet, UninhabitedType)
   864	                                meet = UnpackType(tv.tuple_fallback.copy_modified(args=[meet]))
   865	                        args.append(meet)
   866	                    return Instance(t.type, args)
   867	                else:
   868	                    if state.strict_optional:
   869	                        return UninhabitedType()
   870	                    else:
   871	                        return NoneType()
   872	            else:
   873	                alt_promote = t.type.alt_promote
   874	                if alt_promote and alt_promote.type is self.s.type:
   875	                    return t
   876	                alt_promote = self.s.type.alt_promote
   877	                if alt_promote and alt_promote.type is t.type:
   878	                    return self.s
   879	                if is_subtype(t, self.s):
   880	                    return t
   881	                elif is_subtype(self.s, t):
   882	                    # See also above comment.
   883	                    return self.s
   884	                else:
   885	                    if state.strict_optional:
   886	                        return UninhabitedType()
   887	                    else:
   888	                        return NoneType()
   889	        elif isinstance(self.s, FunctionLike) and t.type.is_protocol:
   890	            call = join.unpack_callback_protocol(t)
   891	            if call:
   892	                return meet_types(call, self.s)
   893	        elif isinstance(self.s, FunctionLike) and self.s.is_type_obj() and t.type.is_metaclass():
   894	            if is_subtype(self.s.fallback, t):
   895	                return self.s
   896	            return self.default(self.s)
   897	        elif isinstance(self.s, TypeType):
   898	            return meet_types(t, self.s)
   899	        elif isinstance(self.s, TupleType):
   900	            return meet_types(t, self.s)
   901	        elif isinstance(self.s, LiteralType):
   902	            return meet_types(t, self.s)
   903	        elif isinstance(self.s, TypedDictType):
   904	            return meet_types(t, self.s)
   905	        return self.default(self.s)

   908	        if isinstance(self.s, CallableType) and join.is_similar_callables(t, self.s):
   909	            if is_equivalent(t, self.s):
   910	                return join.combine_similar_callables(t, self.s)
   911	            result = meet_similar_callables(t, self.s)

   913	            # concrete class objects gets inferred as their common abstract superclass.
   914	            if not (
   915	                (t.is_type_obj() and t.type_object().is_abstract)

   929	        elif isinstance(self.s, Instance) and self.s.type.is_protocol:
   930	            call = join.unpack_callback_protocol(self.s)
   931	            if call:
   932	                return meet_types(t, call)
   933	        return self.default(self.s)

   954	    def meet_tuples(self, s: TupleType, t: TupleType) -> list[Type] | None:
   955	        """Meet two tuple types while handling variadic entries.
   956
   957	        This is surprisingly tricky, and we don't handle some tricky corner cases.
   958	        Most of the trickiness comes from the variadic tuple items like *tuple[X, ...]
   959	        since they can have arbitrary partial overlaps (while *Ts can't be split). This
   960	        function is roughly a mirror of join_tuples() w.r.t. to the fact that fixed
   961	        tuples are subtypes of variadic ones but not vice versa.
   962	        """
   963	        s_unpack_index = find_unpack_in_list(s.items)
   964	        t_unpack_index = find_unpack_in_list(t.items)
   965	        if s_unpack_index is None and t_unpack_index is None:
   966	            if s.length() == t.length():
   967	                items: list[Type] = []
   968	                for i in range(t.length()):
   969	                    items.append(self.meet(t.items[i], s.items[i]))
   970	                return items
   971	            return None
   972	        if s_unpack_index is not None and t_unpack_index is not None:
   973	            # The only simple case we can handle if both tuples are variadic
   974	            # is when their structure fully matches. Other cases are tricky because
   975	            # a variadic item is effectively a union of tuples of all length, thus
   976	            # potentially causing overlap between a suffix in `s` and a prefix
   977	            # in `t` (see how this is handled in is_subtype() for details).
   978	            # TODO: handle more cases (like when both prefix/suffix are shorter in s or t).
   979	            if s.length() == t.length() and s_unpack_index == t_unpack_index:
   980	                unpack_index = s_unpack_index
   981	                s_unpack = s.items[unpack_index]
   982	                assert isinstance(s_unpack, UnpackType)
   983	                s_unpacked = get_proper_type(s_unpack.type)
   984	                t_unpack = t.items[unpack_index]
   985	                assert isinstance(t_unpack, UnpackType)
   986	                t_unpacked = get_proper_type(t_unpack.type)
   987	                if not (isinstance(s_unpacked, Instance) and isinstance(t_unpacked, Instance)):
   988	                    return None
   989	                meet = self.meet(s_unpacked, t_unpacked)
   990	                if not isinstance(meet, Instance):
   991	                    return None
   992	                m_prefix: list[Type] = []
   993	                for si, ti in zip(s.items[:unpack_index], t.items[:unpack_index]):
   994	                    m_prefix.append(meet_types(si, ti))
   995	                m_suffix: list[Type] = []
   996	                for si, ti in zip(s.items[unpack_index + 1 :], t.items[unpack_index + 1 :]):
   997	                    m_suffix.append(meet_types(si, ti))
   998	                return m_prefix + [UnpackType(meet)] + m_suffix
   999	            return None
  1000	        if s_unpack_index is not None:
  1001	            variadic = s
  1002	            unpack_index = s_unpack_index
  1003	            fixed = t
  1004	        else:
  1005	            assert t_unpack_index is not None
  1006	            variadic = t
  1007	            unpack_index = t_unpack_index
  1008	            fixed = s
  1009	        # If one tuple is variadic one, and the other one is fixed, the meet will be fixed.
  1010	        unpack = variadic.items[unpack_index]
  1011	        assert isinstance(unpack, UnpackType)
  1012	        unpacked = get_proper_type(unpack.type)
  1013	        if not isinstance(unpacked, Instance):
  1014	            return None
  1015	        if fixed.length() < variadic.length() - 1:
  1016	            return None
  1017	        prefix_len = unpack_index
  1018	        suffix_len = variadic.length() - prefix_len - 1
  1019	        prefix, middle, suffix = split_with_prefix_and_suffix(
  1020	            tuple(fixed.items), prefix_len, suffix_len
  1021	        )
  1022	        items = []
  1023	        for fi, vi in zip(prefix, variadic.items[:prefix_len]):
  1024	            items.append(self.meet(fi, vi))
  1025	        for mi in middle:
  1026	            items.append(self.meet(mi, unpacked.args[0]))
  1027	        if suffix_len:
  1028	            for fi, vi in zip(suffix, variadic.items[-suffix_len:]):
  1029	                items.append(self.meet(fi, vi))
  1030	        return items
  1031
  1032	    def visit_tuple_type(self, t: TupleType) -> ProperType:
  1033	        if isinstance(self.s, TupleType):
  1034	            items = self.meet_tuples(self.s, t)
  1035	            if items is None:
  1036	                return self.default(self.s)
  1037	            # TODO: What if the fallbacks are different?
  1038	            return TupleType(items, tuple_fallback(t))
  1039	        elif isinstance(self.s, Instance):
  1040	            # meet(Tuple[t1, t2, <...>], Tuple[s, ...]) == Tuple[meet(t1, s), meet(t2, s), <...>].
  1041	            if self.s.type.fullname in TUPLE_LIKE_INSTANCE_NAMES and self.s.args:
  1042	                return t.copy_modified(items=[meet_types(it, self.s.args[0]) for it in t.items])
  1043	            elif is_proper_subtype(t, self.s):
  1044	                # A named tuple that inherits from a normal class
  1045	                return t
  1046	            elif self.s.type.has_type_var_tuple_type and is_subtype(t, self.s):
  1047	                # This is a bit ad-hoc but more principled handling is tricky, and this
  1048	                # special case is important for type narrowing in binder to work.
  1049	                return t

  1118	def meet_similar_callables(t: CallableType, s: CallableType) -> CallableType:
  1119	    from mypy.join import match_generic_callables, safe_join
  1120
  1121	    t, s = match_generic_callables(t, s)
  1122	    arg_types: list[Type] = []
  1123	    for i in range(len(t.arg_types)):

  1171	def typed_dict_mapping_overlap(
  1172	    left: Type, right: Type, overlapping: Callable[[Type, Type], bool]
  1173	) -> bool:
  1174	    """Check if a TypedDict type is overlapping with a Mapping.
  1175

  1177

  1179	      if and only if every key type is overlapping with <some type>. For example:
  1180
  1181	      - TypedDict(x=int, y=str) overlaps with Dict[str, Union[str, int]]
  1182	      - TypedDict(x=int, y=str) doesn't overlap with Dict[str, int]
  1183

  1185
  1186	    * A TypedDict with no required keys overlaps with Mapping[str, <some type>] if and
  1187	      only if at least one of key types overlaps with <some type>. For example:
  1188
  1189	      - TypedDict(x=str, y=str, total=False) overlaps with Dict[str, str]
  1190	      - TypedDict(x=str, y=str, total=False) doesn't overlap with Dict[str, int]
  1191	      - TypedDict(x=int, y=str, total=False) overlaps with Dict[str, str]
  1192
  1193	    * A TypedDict with at least one ReadOnly[] key does not overlap
  1194	      with Dict or MutableMapping, because they assume mutable data.
  1195
  1196	    As usual empty, dictionaries lie in a gray area. In general, List[str] and List[str]
  1197	    are considered non-overlapping despite empty list belongs to both. However, List[int]
  1198	    and List[Never] are considered overlapping.
  1199
  1200	    So here we follow the same logic: a TypedDict with no required keys is considered
  1201	    non-overlapping with Mapping[str, <some type>], but is considered overlapping with
  1202	    Mapping[Never, Never]. This way we avoid false positives for overloads, and also
  1203	    avoid false positives for comparisons like SomeTypedDict == {} under --strict-equality.
  1204	    """

  1206	    assert not isinstance(left, TypedDictType) or not isinstance(right, TypedDictType)
  1207
  1208	    if isinstance(left, TypedDictType):
  1209	        assert isinstance(right, Instance)
  1210	        typed, other = left, right
  1211	    else:
  1212	        assert isinstance(left, Instance)
  1213	        assert isinstance(right, TypedDictType)
  1214	        typed, other = right, left

  1216	    mutable_mapping = next(
  1217	        (base for base in other.type.mro if base.fullname == "typing.MutableMapping"), None
  1218	    )
  1219	    if mutable_mapping is not None and typed.readonly_keys:
  1220	        return False
  1221
  1222	    mapping = next(base for base in other.type.mro if base.fullname == "typing.Mapping")
  1223	    other = map_instance_to_supertype(other, mapping)
  1224	    key_type, value_type = get_proper_types(other.args)
  1225
  1226	    # TODO: is there a cleaner way to get str_type here?
  1227	    fallback = typed.as_anonymous().fallback
  1228	    str_type = fallback.type.bases[0].args[0]  # typing._TypedDict inherits Mapping[str, object]
  1229
  1230	    # Special case: a TypedDict with no required keys overlaps with an empty dict.
  1231	    if isinstance(key_type, UninhabitedType) and isinstance(value_type, UninhabitedType):
  1232	        return not typed.required_keys
  1233
  1234	    if typed.required_keys:
  1235	        if not overlapping(key_type, str_type):
  1236	            return False
  1237	        return all(overlapping(typed.items[k], value_type) for k in typed.required_keys)
  1238	    else:
  1239	        if not overlapping(key_type, str_type):
  1240	            return False
  1241	        non_required = set(typed.items.keys()) - typed.required_keys
  1242	        return any(overlapping(typed.items[k], value_type) for k in non_required)

    21	    """Return stats about memory use.

    94	def find_recursive_objects(objs: list[object]) -> None:
    95	    """Find additional objects referenced by objs and append them to objs.
    96
    97	    We use this since gc.get_objects() does not return objects without pointers
    98	    in them such as strings.
    99	    """
   100	    seen = {id(o) for o in objs}
   101
   102	    def visit(o: object) -> None:
   103	        if id(o) not in seen:
   104	            objs.append(o)
   105	            seen.add(id(o))
   106
   107	    for obj in objs.copy():
   108	        if type(obj) is FakeInfo:
   109	            # Processing these would cause a crash.
   110	            continue
   111	        if type(obj) in (dict, defaultdict):
   112	            for key, val in cast(dict[object, object], obj).items():
   113	                visit(key)
   114	                visit(val)
   115	        if type(obj) in (list, tuple, set):
   116	            for x in cast(Iterable[object], obj):
   117	                visit(x)
   118	        if hasattr(obj, "__slots__"):
   119	            for base in type.mro(type(obj)):
   120	                for slot in getattr(base, "__slots__", ()):
   121	                    if hasattr(obj, slot):
   122	                        visit(getattr(obj, slot))

   460	                self.fail("Cannot call function of unknown type", context, code=codes.OPERATOR)
   461	                return codes.OPERATOR
   462	            else:
   463	                self.fail(
   464	                    message_registry.NOT_CALLABLE.format(format_type(original_type, self.options)),
   465	                    context,
   466	                    code=codes.OPERATOR,
   467	                )
   468	                return codes.OPERATOR
   469	        else:
   470	            # The non-special case: a missing ordinary attribute.
   471	            extra = ""
   472	            if member == "__iter__":
   473	                extra = " (not iterable)"
   474	            elif member == "__aiter__":
   475	                extra = " (not async iterable)"
   476	            if not self.are_type_names_disabled():
   477	                failed = False
   478	                if isinstance(original_type, Instance) and original_type.type.names:
   479	                    if (
   480	                        module_symbol_table is not None
   481	                        and member in module_symbol_table
   482	                        and not module_symbol_table[member].module_public
   483	                    ):
   484	                        self.fail(
   485	                            f"{format_type(original_type, self.options, module_names=True)} does not "
   486	                            f'explicitly export attribute "{member}"',
   487	                            context,
   488	                            code=codes.ATTR_DEFINED,
   489	                        )
   490	                        failed = True
   491	                    else:
   492	                        alternatives = set(original_type.type.names.keys())
   493	                        if module_symbol_table is not None:
   494	                            alternatives |= {
   495	                                k for k, v in module_symbol_table.items() if v.module_public
   496	                            }
   497	                        # Rare but possible, see e.g. testNewAnalyzerCyclicDefinitionCrossModule
   498	                        alternatives.discard(member)
   499
   500	                        matches = [m for m in COMMON_MISTAKES.get(member, []) if m in alternatives]
   501	                        matches.extend(best_matches(member, alternatives, n=3))
   502	                        if member == "__aiter__" and matches == ["__iter__"]:
   503	                            matches = []  # Avoid misleading suggestion
   504	                        if matches:
   505	                            self.fail(
   506	                                '{} has no attribute "{}"; maybe {}?{}'.format(
   507	                                    format_type(original_type, self.options),
   508	                                    member,
   509	                                    pretty_seq(matches, "or"),
   510	                                    extra,
   511	                                ),
   512	                                context,
   513	                                code=codes.ATTR_DEFINED,
   514	                            )
   515	                            failed = True
   516	                if not failed:
   517	                    self.fail(
   518	                        '{} has no attribute "{}"{}'.format(
   519	                            format_type(original_type, self.options), member, extra
   520	                        ),
   521	                        context,
   522	                        code=codes.ATTR_DEFINED,
   523	                    )
   524	                return codes.ATTR_DEFINED
   525	            elif isinstance(original_type, UnionType):
   526	                # The checker passes "object" in lieu of "None" for attribute
   527	                # checks, so we manually convert it back.
   528	                typ_format, orig_type_format = format_type_distinctly(
   529	                    typ, original_type, options=self.options
   530	                )
   531	                if typ_format == '"object"' and any(
   532	                    type(item) == NoneType for item in original_type.items
   533	                ):
   534	                    typ_format = '"None"'
   535	                self.fail(
   536	                    'Item {} of {} has no attribute "{}"{}'.format(
   537	                        typ_format, orig_type_format, member, extra
   538	                    ),
   539	                    context,
   540	                    code=codes.UNION_ATTR,
   541	                )
   542	                return codes.UNION_ATTR
   543	            elif isinstance(original_type, TypeVarType):
   544	                bound = get_proper_type(original_type.upper_bound)
   545	                if isinstance(bound, UnionType):
   546	                    typ_fmt, bound_fmt = format_type_distinctly(typ, bound, options=self.options)
   547	                    original_type_fmt = format_type(original_type, self.options)
   548	                    self.fail(
   549	                        "Item {} of the upper bound {} of type variable {} has no "
   550	                        'attribute "{}"{}'.format(
   551	                            typ_fmt, bound_fmt, original_type_fmt, member, extra
   552	                        ),
   553	                        context,
   554	                        code=codes.UNION_ATTR,
   555	                    )
   556	                    return codes.UNION_ATTR
   557	            else:
   558	                self.fail(
   559	                    '{} has no attribute "{}"{}'.format(
   560	                        format_type(original_type, self.options), member, extra
   561	                    ),
   562	                    context,
   563	                    code=codes.ATTR_DEFINED,
   564	                )
   565	                return codes.ATTR_DEFINED
   566	        return None

   619	    def incompatible_argument(
   620	        self,
   621	        n: int,
   622	        m: int,
   623	        callee: CallableType,
   624	        arg_type: Type,
   625	        arg_kind: ArgKind,
   626	        object_type: Type | None,
   627	        context: Context,
   628	        outer_context: Context,
   629	    ) -> ErrorInfo:
   630	        """Report an error about an incompatible argument type.
   631
   632	        The argument type is arg_type, argument number is n and the
   633	        callee type is 'callee'. If the callee represents a method
   634	        that corresponds to an operator, use the corresponding
   635	        operator name in the messages.
   636
   637	        Return the error code that used for the argument (multiple error
   638	        codes are possible).
   639	        """
   640	        arg_type = get_proper_type(arg_type)
   641
   642	        target = ""
   643	        callee_name = callable_name(callee)
   644	        if callee_name is not None:
   645	            name = callee_name
   646	            if object_type is not None:
   647	                base = format_type(object_type, self.options)
   648	            else:
   649	                base = extract_type(name)
   650
   651	            for method, op in op_methods_to_symbols.items():
   652	                for variant in method, "__r" + method[2:]:
   653	                    # FIX: do not rely on textual formatting
   654	                    if name.startswith(f'"{variant}" of'):
   655	                        if op == "in" or variant != method:
   656	                            # Reversed order of base/argument.
   657	                            return self.unsupported_operand_types(
   658	                                op, arg_type, base, context, code=codes.OPERATOR
   659	                            )
   660	                        else:
   661	                            return self.unsupported_operand_types(
   662	                                op, base, arg_type, context, code=codes.OPERATOR
   663	                            )
   664
   665	            if name.startswith('"__getitem__" of'):
   666	                return self.invalid_index_type(
   667	                    arg_type, callee.arg_types[n - 1], base, context, code=codes.INDEX
   668	                )
   669
   670	            if name.startswith('"__setitem__" of'):
   671	                if n == 1:
   672	                    return self.invalid_index_type(
   673	                        arg_type, callee.arg_types[n - 1], base, context, code=codes.INDEX
   674	                    )
   675	                else:
   676	                    arg_type_str, callee_type_str = format_type_distinctly(
   677	                        arg_type, callee.arg_types[n - 1], options=self.options
   678	                    )
   679	                    info = (
   680	                        f" (expression has type {arg_type_str}, target has type {callee_type_str})"
   681	                    )
   682	                    error_msg = (
   683	                        message_registry.INCOMPATIBLE_TYPES_IN_ASSIGNMENT.with_additional_msg(info)
   684	                    )
   685	                    return self.fail(error_msg.value, context, code=error_msg.code)
   686
   687	            target = f"to {name} "
   688
   689	        msg = ""
   690	        code = codes.MISC
   691	        notes: list[str] = []
   692	        if callee_name == "<list>":
   693	            name = callee_name[1:-1]
   694	            n -= 1
   695	            actual_type_str, expected_type_str = format_type_distinctly(
   696	                arg_type, callee.arg_types[0], options=self.options
   697	            )
   698	            msg = "{} item {} has incompatible type {}; expected {}".format(
   699	                name.title(), n, actual_type_str, expected_type_str
   700	            )
   701	            code = codes.LIST_ITEM
   702	        elif callee_name == "<dict>" and isinstance(
   703	            get_proper_type(callee.arg_types[n - 1]), TupleType
   704	        ):
   705	            name = callee_name[1:-1]
   706	            n -= 1
   707	            key_type, value_type = cast(TupleType, arg_type).items
   708	            expected_key_type, expected_value_type = cast(TupleType, callee.arg_types[n]).items
   709
   710	            # don't increase verbosity unless there is need to do so
   711	            if is_subtype(key_type, expected_key_type):
   712	                key_type_str = format_type(key_type, self.options)
   713	                expected_key_type_str = format_type(expected_key_type, self.options)
   714	            else:
   715	                key_type_str, expected_key_type_str = format_type_distinctly(
   716	                    key_type, expected_key_type, options=self.options
   717	                )
   718	            if is_subtype(value_type, expected_value_type):
   719	                value_type_str = format_type(value_type, self.options)
   720	                expected_value_type_str = format_type(expected_value_type, self.options)
   721	            else:
   722	                value_type_str, expected_value_type_str = format_type_distinctly(
   723	                    value_type, expected_value_type, options=self.options
   724	                )
   725
   726	            msg = "{} entry {} has incompatible type {}: {}; expected {}: {}".format(
   727	                name.title(),
   728	                n,
   729	                key_type_str,
   730	                value_type_str,
   731	                expected_key_type_str,
   732	                expected_value_type_str,
   733	            )
   734	            code = codes.DICT_ITEM
   735	        elif callee_name == "<dict>":
   736	            value_type_str, expected_value_type_str = format_type_distinctly(
   737	                arg_type, callee.arg_types[n - 1], options=self.options
   738	            )
   739	            msg = "Unpacked dict entry {} has incompatible type {}; expected {}".format(
   740	                n - 1, value_type_str, expected_value_type_str
   741	            )
   742	            code = codes.DICT_ITEM
   743	        elif callee_name == "<list-comprehension>":
   744	            actual_type_str, expected_type_str = map(
   745	                strip_quotes,
   746	                format_type_distinctly(arg_type, callee.arg_types[0], options=self.options),
   747	            )
   748	            msg = "List comprehension has incompatible type List[{}]; expected List[{}]".format(
   749	                actual_type_str, expected_type_str
   750	            )
   751	        elif callee_name == "<set-comprehension>":
   752	            actual_type_str, expected_type_str = map(
   753	                strip_quotes,
   754	                format_type_distinctly(arg_type, callee.arg_types[0], options=self.options),
   755	            )
   756	            msg = "Set comprehension has incompatible type Set[{}]; expected Set[{}]".format(
   757	                actual_type_str, expected_type_str
   758	            )
   759	        elif callee_name == "<dictionary-comprehension>":
   760	            actual_type_str, expected_type_str = format_type_distinctly(
   761	                arg_type, callee.arg_types[n - 1], options=self.options
   762	            )
   763	            msg = (
   764	                "{} expression in dictionary comprehension has incompatible type {}; "
   765	                "expected type {}"
   766	            ).format("Key" if n == 1 else "Value", actual_type_str, expected_type_str)
   767	        elif callee_name == "<generator>":
   768	            actual_type_str, expected_type_str = format_type_distinctly(
   769	                arg_type, callee.arg_types[0], options=self.options
   770	            )
   771	            msg = "Generator has incompatible item type {}; expected {}".format(
   772	                actual_type_str, expected_type_str
   773	            )
   774	        else:
   775	            if self.prefer_simple_messages():
   776	                msg = "Argument has incompatible type"
   777	            else:
   778	                try:
   779	                    expected_type = callee.arg_types[m - 1]
   780	                except IndexError:  # Varargs callees
   781	                    expected_type = callee.arg_types[-1]
   782	                arg_type_str, expected_type_str = format_type_distinctly(
   783	                    arg_type, expected_type, bare=True, options=self.options
   784	                )
   785	                if arg_kind == ARG_STAR:
   786	                    arg_type_str = "*" + arg_type_str
   787	                elif arg_kind == ARG_STAR2:
   788	                    arg_type_str = "**" + arg_type_str
   789
   790	                # For function calls with keyword arguments, display the argument name rather
   791	                # than the number.
   792	                arg_label = str(n)
   793	                if isinstance(outer_context, CallExpr) and len(outer_context.arg_names) >= n:
   794	                    arg_name = outer_context.arg_names[n - 1]
   795	                    if arg_name is not None:
   796	                        arg_label = f'"{arg_name}"'
   797	                if (
   798	                    arg_kind == ARG_STAR2
   799	                    and isinstance(arg_type, TypedDictType)
   800	                    and m <= len(callee.arg_names)
   801	                    and callee.arg_names[m - 1] is not None
   802	                    and callee.arg_kinds[m - 1] != ARG_STAR2
   803	                ):
   804	                    arg_name = callee.arg_names[m - 1]
   805	                    assert arg_name is not None
   806	                    arg_type_str, expected_type_str = format_type_distinctly(
   807	                        arg_type.items[arg_name], expected_type, bare=True, options=self.options
   808	                    )
   809	                    arg_label = f'"{arg_name}"'
   810	                if isinstance(outer_context, IndexExpr) and isinstance(
   811	                    outer_context.index, StrExpr
   812	                ):
   813	                    msg = 'Value of "{}" has incompatible type {}; expected {}'.format(
   814	                        outer_context.index.value,
   815	                        quote_type_string(arg_type_str),
   816	                        quote_type_string(expected_type_str),
   817	                    )
   818	                else:
   819	                    msg = "Argument {} {}has incompatible type {}; expected {}".format(
   820	                        arg_label,
   821	                        target,
   822	                        quote_type_string(arg_type_str),
   823	                        quote_type_string(expected_type_str),
   824	                    )
   825	                expected_type = get_proper_type(expected_type)
   826	                if isinstance(expected_type, UnionType):
   827	                    expected_types = list(expected_type.items)
   828	                else:
   829	                    expected_types = [expected_type]
   830	                for type in get_proper_types(expected_types):
   831	                    if isinstance(arg_type, Instance) and isinstance(type, Instance):
   832	                        notes = append_invariance_notes(notes, arg_type, type)
   833	                        notes = append_numbers_notes(notes, arg_type, type)
   834	            object_type = get_proper_type(object_type)
   835	            if isinstance(object_type, TypedDictType):
   836	                code = codes.TYPEDDICT_ITEM
   837	            else:
   838	                code = codes.ARG_TYPE
   839	        error = self.fail(msg, context, code=code)
   840	        if notes:
   841	            for note_msg in notes:
   842	                self.note(note_msg, context, code=code)
   843	        return error

   845	    def incompatible_argument_note(
   846	        self,
   847	        original_caller_type: ProperType,
   848	        callee_type: ProperType,
   849	        context: Context,
   850	        parent_error: ErrorInfo,
   851	    ) -> None:
   852	        if self.prefer_simple_messages():
   853	            return
   854	        if isinstance(
   855	            original_caller_type, (Instance, TupleType, TypedDictType, TypeType, CallableType)
   856	        ):
   857	            if isinstance(callee_type, Instance) and callee_type.type.is_protocol:
   858	                self.report_protocol_problems(
   859	                    original_caller_type, callee_type, context, parent_error=parent_error
   860	                )
   861	            if isinstance(callee_type, UnionType):
   862	                for item in callee_type.items:
   863	                    item = get_proper_type(item)
   864	                    if isinstance(item, Instance) and item.type.is_protocol:
   865	                        self.report_protocol_problems(
   866	                            original_caller_type, item, context, parent_error=parent_error
   867	                        )
   868	        if isinstance(callee_type, CallableType) and isinstance(original_caller_type, Instance):
   869	            call = find_member(
   870	                "__call__", original_caller_type, original_caller_type, is_operator=True
   871	            )
   872	            if call:
   873	                self.note_call(original_caller_type, call, context, code=parent_error.code)
   874	        if isinstance(callee_type, Instance) and callee_type.type.is_protocol:
   875	            call = find_member("__call__", callee_type, callee_type, is_operator=True)
   876	            if call:
   877	                self.note_call(callee_type, call, context, code=parent_error.code)
   878	        self.maybe_note_concatenate_pos_args(
   879	            original_caller_type, callee_type, context, parent_error.code
   880	        )

   882	    def maybe_note_concatenate_pos_args(
   883	        self,
   884	        original_caller_type: ProperType,
   885	        callee_type: ProperType,
   886	        context: Context,
   887	        code: ErrorCode | None = None,
   888	    ) -> None:
   889	        # pos-only vs positional can be confusing, with Concatenate
   890	        if (
   891	            isinstance(callee_type, CallableType)
   892	            and isinstance(original_caller_type, CallableType)
   893	            and (original_caller_type.from_concatenate or callee_type.from_concatenate)
   894	        ):
   895	            names: list[str] = []
   896	            for c, o in zip(
   897	                callee_type.formal_arguments(), original_caller_type.formal_arguments()
   898	            ):
   899	                if None in (c.pos, o.pos):
   900	                    # non-positional
   901	                    continue
   902	                if c.name != o.name and c.name is None and o.name is not None:
   903	                    names.append(o.name)
   904
   905	            if names:
   906	                missing_arguments = '"' + '", "'.join(names) + '"'
   907	                self.note(
   908	                    f'This is likely because "{original_caller_type.name}" has named arguments: '
   909	                    f"{missing_arguments}. Consider marking them positional-only",
   910	                    context,
   911	                    code=code,
   912	                )

   947	    ) -> None:
   948	        if self.prefer_simple_messages():
   949	            msg = "Too few arguments"

   952	            arguments_left = callee.arg_names[num_positional_args : callee.min_args]
   953	            diff = [k for k in arguments_left if k not in argument_names]

   959	            if callee_name is not None and diff and all(d is not None for d in diff):
   960	                args = '", "'.join(cast(list[str], diff))
   961	                msg += f' "{args}" in call to {callee_name}'
   962	            else:
   963	                msg = "Too few arguments" + for_function(callee)
   964
   965	        else:
   966	            msg = "Too few arguments" + for_function(callee)
   967	        self.fail(msg, context, code=codes.CALL_ARG)

   981	    def too_many_arguments_from_typed_dict(
   982	        self, callee: CallableType, arg_type: TypedDictType, context: Context
   983	    ) -> None:
   984	        # Try to determine the name of the extra argument.
   985	        for key in arg_type.items:
   986	            if key not in callee.arg_names:
   987	                msg = f'Extra argument "{key}" from **args' + for_function(callee)
