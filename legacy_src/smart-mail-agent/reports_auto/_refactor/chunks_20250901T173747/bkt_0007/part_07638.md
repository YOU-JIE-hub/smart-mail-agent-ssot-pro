           self.add_operator_method_dependency(lvalue.base, "__setitem__")
   521	        elif isinstance(lvalue, NameExpr):
   522	            if lvalue.kind in (MDEF, GDEF):
   523	                # Assignment to an attribute in the class body, or direct assignment to a
   524	                # global variable.
   525	                lvalue_type = self.get_non_partial_lvalue_type(lvalue)
   526	                type_triggers = self.get_type_triggers(lvalue_type)
   527	                attr_trigger = make_trigger(f"{self.scope.current_full_target()}.{lvalue.name}")
   528	                for type_trigger in type_triggers:
   529	                    self.add_dependency(type_trigger, attr_trigger)
   530	        elif isinstance(lvalue, MemberExpr):
   531	            if self.is_self_member_ref(lvalue) and lvalue.is_new_def:
   532	                node = lvalue.node
   533	                if isinstance(node, Var):
   534	                    info = node.info
   535	                    if info and has_user_bases(info):
   536	                        # Recheck Liskov for self definitions
   537	                        self.add_dependency(make_trigger(info.fullname + "." + lvalue.name))
   538	            if lvalue.kind is None:
   539	                # Reference to a non-module attribute
   540	                if lvalue.expr not in self.type_map:
   541	                    # Unreachable assignment -> not checked so no dependencies to generate.
   542	                    return
   543	                object_type = self.type_map[lvalue.expr]
   544	                lvalue_type = self.get_non_partial_lvalue_type(lvalue)
   545	                type_triggers = self.get_type_triggers(lvalue_type)
   546	                for attr_trigger in self.attribute_triggers(object_type, lvalue.name):
   547	                    for type_trigger in type_triggers:
   548	                        self.add_dependency(type_trigger, attr_trigger)
   549	        elif isinstance(lvalue, TupleExpr):
   550	            for item in lvalue.items:
   551	                self.process_lvalue(item)
   552	        elif isinstance(lvalue, StarExpr):
   553	            self.process_lvalue(lvalue.expr)
   554
   555	    def is_self_member_ref(self, memberexpr: MemberExpr) -> bool:
   556	        """Does memberexpr to refer to an attribute of self?"""
   557	        if not isinstance(memberexpr.expr, NameExpr):
   558	            return False
   559	        node = memberexpr.expr.node
   560	        return isinstance(node, Var) and node.is_self

   562	    def get_non_partial_lvalue_type(self, lvalue: RefExpr) -> Type:
   563	        if lvalue not in self.type_map:
   564	            # Likely a block considered unreachable during type checking.
   565	            return UninhabitedType()
   566	        lvalue_type = get_proper_type(self.type_map[lvalue])
   567	        if isinstance(lvalue_type, PartialType):
   568	            if isinstance(lvalue.node, Var):
   569	                if lvalue.node.type:
   570	                    lvalue_type = get_proper_type(lvalue.node.type)
   571	                else:
   572	                    lvalue_type = UninhabitedType()
   573	            else:
   574	                # Probably a secondary, non-definition assignment that doesn't
   575	                # result in a non-partial type. We won't be able to infer any
   576	                # dependencies from this so just return something. (The first,
   577	                # definition assignment with a partial type is handled
   578	                # differently, in the semantic analyzer.)
   579	                assert not lvalue.is_new_def
   580	                return UninhabitedType()
   581	        return lvalue_type

   583	    def visit_operator_assignment_stmt(self, o: OperatorAssignmentStmt) -> None:
   584	        super().visit_operator_assignment_stmt(o)
   585	        self.process_lvalue(o.lvalue)
   586	        method = op_methods[o.op]
   587	        self.add_attribute_dependency_for_expr(o.lvalue, method)
   588	        if o.op in ops_with_inplace_method:
   589	            inplace_method = "__i" + method[2:]
   590	            self.add_attribute_dependency_for_expr(o.lvalue, inplace_method)

   592	    def visit_for_stmt(self, o: ForStmt) -> None:
   593	        super().visit_for_stmt(o)
   594	        if not o.is_async:
   595	            # __getitem__ is only used if __iter__ is missing but for simplicity we
   596	            # just always depend on both.
   597	            self.add_attribute_dependency_for_expr(o.expr, "__iter__")
   598	            self.add_attribute_dependency_for_expr(o.expr, "__getitem__")
   599	            if o.inferred_iterator_type:
   600	                self.add_attribute_dependency(o.inferred_iterator_type, "__next__")
   601	        else:
   602	            self.add_attribute_dependency_for_expr(o.expr, "__aiter__")
   603	            if o.inferred_iterator_type:
   604	                self.add_attribute_dependency(o.inferred_iterator_type, "__anext__")
   605
   606	        self.process_lvalue(o.index)
   607	        if isinstance(o.index, TupleExpr):
   608	            # Process multiple assignment to index variables.
   609	            item_type = o.inferred_item_type
   610	            if item_type:
   611	                # This is similar to above.
   612	                self.add_attribute_dependency(item_type, "__iter__")
   613	                self.add_attribute_dependency(item_type, "__getitem__")
   614	        if o.index_type:
   615	            self.add_type_dependencies(o.index_type)

   617	    def visit_with_stmt(self, o: WithStmt) -> None:
   618	        super().visit_with_stmt(o)
   619	        for e in o.expr:
   620	            if not o.is_async:
   621	                self.add_attribute_dependency_for_expr(e, "__enter__")
   622	                self.add_attribute_dependency_for_expr(e, "__exit__")
   623	            else:
   624	                self.add_attribute_dependency_for_expr(e, "__aenter__")
   625	                self.add_attribute_dependency_for_expr(e, "__aexit__")
   626	        for typ in o.analyzed_types:
   627	            self.add_type_dependencies(typ)

   630	        super().visit_del_stmt(o)
   631	        if isinstance(o.expr, IndexExpr):
   632	            self.add_attribute_dependency_for_expr(o.expr.base, "__delitem__")

   636	    def process_global_ref_expr(self, o: RefExpr) -> None:
   637	        if o.fullname:
   638	            self.add_dependency(make_trigger(o.fullname))
   639
   640	        # If this is a reference to a type, generate a dependency to its
   641	        # constructor.
   642	        # IDEA: Avoid generating spurious dependencies for except statements,
   643	        #       class attribute references, etc., if performance is a problem.
   644	        typ = get_proper_type(self.type_map.get(o))
   645	        if isinstance(typ, FunctionLike) and typ.is_type_obj():
   646	            class_name = typ.type_object().fullname
   647	            self.add_dependency(make_trigger(class_name + ".__init__"))
   648	            self.add_dependency(make_trigger(class_name + ".__new__"))

   650	    def visit_name_expr(self, o: NameExpr) -> None:
   651	        if o.kind == LDEF:
   652	            # We don't track dependencies to local variables, since they
   653	            # aren't externally visible.
   654	            return
   655	        if o.kind == MDEF:
   656	            # Direct reference to member is only possible in the scope that
   657	            # defined the name, so no dependency is required.
   658	            return
   659	        self.process_global_ref_expr(o)

   661	    def visit_member_expr(self, e: MemberExpr) -> None:
   662	        if isinstance(e.expr, RefExpr) and isinstance(e.expr.node, TypeInfo):
   663	            # Special case class attribute so that we don't depend on "__init__".
   664	            self.add_dependency(make_trigger(e.expr.node.fullname))
   665	        else:
   666	            super().visit_member_expr(e)
   667	        if e.kind is not None:
   668	            # Reference to a module attribute
   669	            self.process_global_ref_expr(e)
   670	        else:
   671	            # Reference to a non-module (or missing) attribute
   672	            if e.expr not in self.type_map:
   673	                # No type available -- this happens for unreachable code. Since it's unreachable,
   674	                # it wasn't type checked and we don't need to generate dependencies.
   675	                return
   676	            if isinstance(e.expr, RefExpr) and isinstance(e.expr.node, MypyFile):
   677	                # Special case: reference to a missing module attribute.
   678	                self.add_dependency(make_trigger(e.expr.node.fullname + "." + e.name))
   679	                return
   680	            typ = get_proper_type(self.type_map[e.expr])
   681	            self.add_attribute_dependency(typ, e.name)
   682	            if self.use_logical_deps() and isinstance(typ, AnyType):
   683	                name = self.get_unimported_fullname(e, typ)
   684	                if name is not None:
   685	                    # Generate a logical dependency from an unimported
   686	                    # definition (which comes from a missing module).
   687	                    # Example:
   688	                    #     import missing  # "missing" not in build
   689	                    #
   690	                    #     def g() -> None:
   691	                    #         missing.f()  # Generate dependency from "missing.f"
   692	                    self.add_dependency(make_trigger(name))

   694	    def get_unimported_fullname(self, e: MemberExpr, typ: AnyType) -> str | None:
   695	        """If e refers to an unimported definition, infer the fullname of this.
   696
   697	        Return None if e doesn't refer to an unimported definition or if we can't
   698	        determine the name.
   699	        """
   700	        suffix = ""
   701	        # Unwrap nested member expression to handle cases like "a.b.c.d" where
   702	        # "a.b" is a known reference to an unimported module. Find the base
   703	        # reference to an unimported module (such as "a.b") and the name suffix
   704	        # (such as "c.d") needed to build a full name.
   705	        while typ.type_of_any == TypeOfAny.from_another_any and isinstance(e.expr, MemberExpr):
   706	            suffix = "." + e.name + suffix
   707	            e = e.expr
   708	            if e.expr not in self.type_map:
   709	                return None
   710	            obj_type = get_proper_type(self.type_map[e.expr])
   711	            if not isinstance(obj_type, AnyType):
   712	                # Can't find the base reference to the unimported module.
   713	                return None
   714	            typ = obj_type
   715	        if typ.type_of_any == TypeOfAny.from_unimported_type and typ.missing_import_name:
   716	            # Infer the full name of the unimported definition.
   717	            return typ.missing_import_name + "." + e.name + suffix
   718	        return None

   720	    def visit_super_expr(self, e: SuperExpr) -> None:
   721	        # Arguments in "super(C, self)" won't generate useful logical deps.
   722	        if not self.use_logical_deps():
   723	            super().visit_super_expr(e)
   724	        if e.info is not None:
   725	            name = e.name
   726	            for base in non_trivial_bases(e.info):
   727	                self.add_dependency(make_trigger(base.fullname + "." + name))
   728	                if name in base.names:
   729	                    # No need to depend on further base classes, since we found
   730	                    # the target.  This is safe since if the target gets
   731	                    # deleted or modified, we'll trigger it.
   732	                    break

   734	    def visit_call_expr(self, e: CallExpr) -> None:
   735	        if isinstance(e.callee, RefExpr) and e.callee.fullname == "builtins.isinstance":
   736	            self.process_isinstance_call(e)
   737	        else:
   738	            super().visit_call_expr(e)
   739	            typ = self.type_map.get(e.callee)
   740	            if typ is not None:
   741	                typ = get_proper_type(typ)
   742	                if not isinstance(typ, FunctionLike):
   743	                    self.add_attribute_dependency(typ, "__call__")

   745	    def process_isinstance_call(self, e: CallExpr) -> None:
   746	        """Process "isinstance(...)" in a way to avoid some extra dependencies."""
   747	        if len(e.args) == 2:
   748	            arg = e.args[1]
   749	            if (
   750	                isinstance(arg, RefExpr)
   751	                and arg.kind == GDEF
   752	                and isinstance(arg.node, TypeInfo)
   753	                and arg.fullname
   754	            ):
   755	                # Special case to avoid redundant dependencies from "__init__".
   756	                self.add_dependency(make_trigger(arg.fullname))
   757	                return
   758	        # In uncommon cases generate normal dependencies. These will include
   759	        # spurious dependencies, but the performance impact is small.
   760	        super().visit_call_expr(e)

   797	    def process_binary_op(self, op: str, left: Expression, right: Expression) -> None:
   798	        method = op_methods.get(op)
   799	        if method:
   800	            if op == "in":
   801	                self.add_operator_method_dependency(right, method)
   802	            else:
   803	                self.add_operator_method_dependency(left, method)
   804	                rev_method = reverse_op_methods.get(method)
   805	                if rev_method:
   806	                    self.add_operator_method_dependency(right, rev_method)

   810	        if typ is not None:
   811	            self.add_operator_method_dependency_for_type(typ, method)

   813	    def add_operator_method_dependency_for_type(self, typ: ProperType, method: str) -> None:
   814	        # Note that operator methods can't be (non-metaclass) methods of type objects
   815	        # (that is, TypeType objects or Callables representing a type).
   816	        if isinstance(typ, TypeVarType):
   817	            typ = get_proper_type(typ.upper_bound)
   818	        if isinstance(typ, TupleType):
   819	            typ = typ.partial_fallback
   820	        if isinstance(typ, Instance):
   821	            trigger = make_trigger(typ.type.fullname + "." + method)
   822	            self.add_dependency(trigger)
   823	        elif isinstance(typ, UnionType):
   824	            for item in typ.items:
   825	                self.add_operator_method_dependency_for_type(get_proper_type(item), method)
   826	        elif isinstance(typ, FunctionLike) and typ.is_type_obj():
   827	            self.add_operator_method_dependency_for_type(typ.fallback, method)
   828	        elif isinstance(typ, TypeType):
   829	            if isinstance(typ.item, Instance) and typ.item.type.metaclass_type is not None:
   830	                self.add_operator_method_dependency_for_type(typ.item.type.metaclass_type, method)
   831

   863	    def add_dependency(self, trigger: str, target: str | None = None) -> None:
   864	        """Add dependency from trigger to a target.
   865
   866	        If the target is not given explicitly, use the current target.
   867	        """
   868	        if trigger.startswith(
   869	            ("<builtins.", "<typing.", "<mypy_extensions.", "<typing_extensions.")
   870	        ):
   871	            # Don't track dependencies to certain library modules to keep the size of
   872	            # the dependencies manageable. These dependencies should only
   873	            # change on mypy version updates, which will require a full rebuild
   874	            # anyway.
   875	            return
   876	        if target is None:
   877	            target = self.scope.current_target()
   878	        self.map.setdefault(trigger, set()).add(target)

   880	    def add_type_dependencies(self, typ: Type, target: str | None = None) -> None:
   881	        """Add dependencies to all components of a type.
   882
   883	        Args:
   884	            target: If not None, override the default (current) target of the
   885	                generated dependency.
   886	        """
   887	        for trigger in self.get_type_triggers(typ):
   888	            self.add_dependency(trigger, target)

   890	    def add_attribute_dependency(self, typ: Type, name: str) -> None:
   891	        """Add dependencies for accessing a named attribute of a type."""
   892	        targets = self.attribute_triggers(typ, name)
   893	        for target in targets:
   894	            self.add_dependency(target)

   896	    def attribute_triggers(self, typ: Type, name: str) -> list[str]:
   897	        """Return all triggers associated with the attribute of a type."""
   898	        typ = get_proper_type(typ)
   899	        if isinstance(typ, TypeVarType):
   900	            typ = get_proper_type(typ.upper_bound)
   901	        if isinstance(typ, TupleType):
   902	            typ = typ.partial_fallback
   903	        if isinstance(typ, Instance):
   904	            member = f"{typ.type.fullname}.{name}"
   905	            return [make_trigger(member)]
   906	        elif isinstance(typ, FunctionLike) and typ.is_type_obj():
   907	            member = f"{typ.type_object().fullname}.{name}"
   908	            triggers = [make_trigger(member)]
   909	            triggers.extend(self.attribute_triggers(typ.fallback, name))
   910	            return triggers
   911	        elif isinstance(typ, UnionType):
   912	            targets = []
   913	            for item in typ.items:
   914	                targets.extend(self.attribute_triggers(item, name))
   915	            return targets
   916	        elif isinstance(typ, TypeType):
   917	            triggers = self.attribute_triggers(typ.item, name)
   918	            if isinstance(typ.item, Instance) and typ.item.type.metaclass_type is not None:
   919	                triggers.append(
   920	                    make_trigger(f"{typ.item.type.metaclass_type.type.fullname}.{name}")
   921	                )
   922	            return triggers
   923	        else:
   924	            return []

   927	        typ = self.type_map.get(e)
   928	        if typ is not None:
   929	            self.add_attribute_dependency(typ, name)

   950	class TypeTriggersVisitor(TypeVisitor[list[str]]):
   951	    def __init__(
   952	        self, use_logical_deps: bool, seen_aliases: set[TypeAliasType] | None = None
   953	    ) -> None:
   954	        self.deps: list[str] = []
   955	        self.seen_aliases: set[TypeAliasType] = seen_aliases or set()
   956	        self.use_logical_deps = use_logical_deps
   957
   958	    def get_type_triggers(self, typ: Type) -> list[str]:
   959	        return get_type_triggers(typ, self.use_logical_deps, self.seen_aliases)
   960
   961	    def visit_instance(self, typ: Instance) -> list[str]:
   962	        trigger = make_trigger(typ.type.fullname)
   963	        triggers = [trigger]
   964	        for arg in typ.args:
   965	            triggers.extend(self.get_type_triggers(arg))
   966	        if typ.last_known_value:
   967	            triggers.extend(self.get_type_triggers(typ.last_known_value))
   968	        if typ.extra_attrs and typ.extra_attrs.mod_name:
   969	            # Module as type effectively depends on all module attributes, use wildcard.
   970	            triggers.append(make_wildcard_trigger(typ.extra_attrs.mod_name))
   971	        return triggers
   972
   973	    def visit_type_alias_type(self, typ: TypeAliasType) -> list[str]:
   974	        if typ in self.seen_aliases:
   975	            return []
   976	        self.seen_aliases.add(typ)
   977	        assert typ.alias is not None
   978	        trigger = make_trigger(typ.alias.fullname)
   979	        triggers = [trigger]
   980	        for arg in typ.args:
   981	            triggers.extend(self.get_type_triggers(arg))
   982	        # TODO: Now that type aliases are its own kind of types we can simplify
   983	        # the logic to rely on intermediate dependencies (like for instance types).
   984	        triggers.extend(self.get_type_triggers(typ.alias.target))
   985	        return triggers
   986
   987	    def visit_any(self, typ: AnyType) -> list[str]:
   988	        if typ.missing_import_name is not None:
   989	            return [make_trigger(typ.missing_import_name)]
   990	        return []
   991
   992	    def visit_none_type(self, typ: NoneType) -> list[str]:
   993	        return []
   994
   995	    def visit_callable_type(self, typ: CallableType) -> list[str]:
   996	        triggers = []
   997	        for arg in typ.arg_types:
   998	            triggers.extend(self.get_type_triggers(arg))

    26	ATOMIC_TYPE_BLACKLIST: Final = {bool, int, float, str, type(None), object}

    35	def isproperty(o: object, attr: str) -> bool:
    36	    return isinstance(getattr(type(o), attr, None), property)

    39	def get_edge_candidates(o: object) -> Iterator[tuple[object, object]]:
    40	    # use getattr because mypyc expects dict, not mappingproxy
    41	    if "__getattribute__" in getattr(type(o), "__dict__"):  # noqa: B009
    42	        return
    43	    if type(o) not in COLLECTION_TYPE_BLACKLIST:
    44	        for attr in dir(o):
    45	            try:
    46	                if attr not in ATTR_BLACKLIST and hasattr(o, attr) and not isproperty(o, attr):
    47	                    e = getattr(o, attr)
    48	                    if type(e) not in ATOMIC_TYPE_BLACKLIST:
    49	                        yield attr, e
    50	            except AssertionError:
    51	                pass
    52	    if isinstance(o, Mapping):
    53	        yield from o.items()
    54	    elif isinstance(o, Iterable) and not isinstance(o, str):
    55	        for i, e in enumerate(o):
    56	            yield i, e
    57

    59	def get_edges(o: object) -> Iterator[tuple[object, object]]:
    60	    for s, e in get_edge_candidates(o):
    61	        if isinstance(e, FUNCTION_TYPES):
    62	            # We don't want to collect methods, but do want to collect values
    63	            # in closures and self pointers to other objects
    64
    65	            if hasattr(e, "__closure__"):
    66	                yield (s, "__closure__"), e.__closure__
    67	            if hasattr(e, "__self__"):
    68	                se = e.__self__
    69	                if se is not o and se is not type(o) and hasattr(s, "__self__"):
    70	                    yield s.__self__, se
    71	        else:
    72	            if type(e) not in TYPE_BLACKLIST:
    73	                yield s, e

    76	def get_reachable_graph(root: object) -> tuple[dict[int, object], dict[int, tuple[int, object]]]:
    77	    parents = {}
    78	    seen = {id(root): root}
    79	    worklist = [root]
    80	    while worklist:
    81	        o = worklist.pop()
    82	        for s, e in get_edges(o):
    83	            if id(e) in seen:
    84	                continue
    85	            parents[id(e)] = (id(o), s)
    86	            seen[id(e)] = e
    87	            worklist.append(e)
    88
    89	    return seen, parents

    96	    while id(o) in parents:
    97	        pid, attr = parents[id(o)]
    98	        o = seen[pid]
    99	        path.append((attr, o))

   101	    return path

     1	"""Update build by processing changes using fine-grained dependencies.
     2
     3	Use fine-grained dependencies to update targets in other modules that
     4	may be affected by externally-visible changes in the changed modules.
     5
     6	This forms the core of the fine-grained incremental daemon mode. This
     7	module is not used at all by the 'classic' (non-daemon) incremental
     8	mode.
     9
    10	Here is some motivation for this mode:
    11
    12	* By keeping program state in memory between incremental runs, we
    13	  only have to process changed modules, not their dependencies. The
    14	  classic incremental mode has to deserialize the symbol tables of
    15	  all dependencies of changed modules, which can be slow for large
    16	  programs.
    17
    18	* Fine-grained dependencies allow processing only the relevant parts
    19	  of modules indirectly affected by a change. Say, if only one function
    20	  in a large module is affected by a change in another module, only this
    21	  function is processed. The classic incremental mode always processes
    22	  an entire file as a unit, which is typically much slower.
    23
    24	* It's possible to independently process individual modules within an
    25	  import cycle (SCC). Small incremental changes can be fast independent
    26	  of the size of the related SCC. In classic incremental mode, any change
    27	  within a SCC requires the entire SCC to be processed, which can slow
    28	  things down considerably.
    29
    30	Some terms:
    31
    32	* A *target* is a function/method definition or the top level of a module.
    33	  We refer to targets using their fully qualified name (e.g.
    34	  'mod.Cls.method'). Targets are the smallest units of processing during
    35	  fine-grained incremental checking.
    36
    37	* A *trigger* represents the properties of a part of a program, and it
    38	  gets triggered/fired when these properties change. For example,
    39	  '<mod.func>' refers to a module-level function. It gets triggered if
    40	  the signature of the function changes, or if the function is removed,
    41	  for example.
    42
    43	Some program state is maintained across multiple build increments in
    44	memory:
    45
    46	* The full ASTs of all modules are stored in memory all the time (this
    47	  includes the type map).
    48
    49	* A fine-grained dependency map is maintained, which maps triggers to
    50	  affected program locations (these can be targets, triggers, or
    51	  classes). The latter determine what other parts of a program need to
    52	  be processed again due to a fired trigger.
    53
    54	Here's a summary of how a fine-grained incremental program update happens:
    55
    56	* Determine which modules have changes in their source code since the
    57	  previous update.
    58
    59	* Process changed modules one at a time. Perform a separate full update
    60	  for each changed module, but only report the errors after all modules
    61	  have been processed, since the intermediate states can generate bogus
    62	  errors due to only seeing a partial set of changes.
    63
    64	* Each changed module is processed in full. We parse the module, and
    65	  run semantic analysis to create a new AST and symbol table for the
    66	  module. Reuse the existing ASTs and symbol tables of modules that
    67	  have no changes in their source code. At the end of this stage, we have
    68	  two ASTs and symbol tables for the changed module (the old and the new
    69	  versions). The latter AST has not yet been type checked.
    70
    71	* Take a snapshot of the old symbol table. This is used later to determine
    72	  which properties of the module have changed and which triggers to fire.
    73
    74	* Merge the old AST with the new AST, preserving the identities of
    75	  externally visible AST nodes for which we can find a corresponding node
    76	  in the new AST. (Look at mypy.server.astmerge for the details.) This
    77	  way all external references to AST nodes in the changed module will
    78	  continue to point to the right nodes (assuming they still have a valid
    79	  target).
    80
    81	* Type check the new module.
    82
    83	* Take another snapshot of the symbol table of the changed module.
    84	  Look at the differences between the old and new snapshots to determine
    85	  which parts of the changed modules have changed. The result is a set of
    86	  fired triggers.
    87
    88	* Using the dependency map and the fired triggers, decide which other
    89	  targets have become stale and need to be reprocessed.
    90
    91	* Create new fine-grained dependencies for the changed module. We don't
    92	  garbage collect old dependencies, since extra dependencies are relatively
    93	  harmless (they take some memory and can theoretically slow things down
    94	  a bit by causing redundant work). This is implemented in
    95	  mypy.server.deps.
    96
    97	* Strip the stale AST nodes that we found above. This returns them to a
    98	  state resembling the end of semantic analysis pass 1. We'll run semantic
    99	  analysis again on the existing AST nodes, and since semantic analysis
   100	  is not idempotent, we need to revert some changes made during semantic
   101	  analysis. This is implemented in mypy.server.aststrip.
   102
   103	* Run semantic analyzer passes 2 and 3 on the stale AST nodes, and type
   104	  check them. We also need to do the symbol table snapshot comparison
   105	  dance to find any changes, and we need to merge ASTs to preserve AST node
   106	  identities.
   107
   108	* If some triggers haven been fired, continue processing and repeat the
   109	  previous steps until no triggers are fired.
   110
   111	This is module is tested using end-to-end fine-grained incremental mode
   112	test cases (test-data/unit/fine-grained*.test).

   344	        Returns:
   345	            Tuple with these items:

   380	        self, module: str, path: str, force_removed: bool, followed: bool
   381	    ) -> tuple[list[tuple[str, str]], tuple[str, str], list[str] | None]:

   383
   384	        If the module contains imports of previously unseen modules, only process one of
   385	        the new modules and return the remaining work to be done.
   386
   387	        Args:
   388	            module: Id of the module
   389	            path: File system path of the module

   391	                exists (used for removing an existing file from the build)

   395	            Tuple with these items:

   397	            - Remaining modules to process as (module id, path) tuples

   402	        self.updated_modules.append(module)

   404	        # builtins and friends could potentially get triggered because
   405	        # of protocol stuff, but nothing good could possibly come from
   406	        # actually updating them.

   408	            is_stdlib_file(self.manager.options.abs_custom_typeshed_dir, path)

   434	        )

   436	            # Blocking error -- just give up
   437	            module, path, remaining, errors = result
   438	            self.previous_modules = get_module_to_path_map(graph)
   439	            return remaining, (module, path), errors
   440	        assert isinstance(result, NormalUpdate)  # Work around #4124
   441	        module, path, remaining, tree = result

   443	        # TODO: What to do with stale dependencies?
   444	        t1 = time.time()

   446	        if is_verbose(self.manager):

   450	        if module in graph:
   451	            graph[module].update_fine_grained_deps(self.deps)
   452	            graph[module].free_state()

   454	            manager,
   455	            graph,
   456	            self.deps,
   457	            triggered,
   458	            {module},
   459	            targets_with_errors=set(),
   460	            processed_targets=self.processed_targets,
   461	        )
   462	        t2 = time.time()
   463	        manager.add_stats(update_isolated_time=t1 - t0, propagate_time=t2 - t1)
   464
   465	        # Preserve state needed for the next update.
   466	        self.previous_targets_with_errors.update(manager.errors.targets())
   467	        self.previous_modules = get_module_to_path_map(graph)
   468
   469	        return remaining, (module, path), None

   472	def find_unloaded_deps(
   473	    manager: BuildManager, graph: dict[str, State], initial: Sequence[str]
   474	) -> list[str]:
   475	    """Find all the deps of the nodes in initial that haven't had their tree loaded.
   476
   477	    The key invariant here is that if a module is loaded, so are all
   478	    of their dependencies. This means that when we encounter a loaded
   479	    module, we don't need to explore its dependencies.  (This
   480	    invariant is slightly violated when dependencies are added, which
   481	    can be handled by calling find_unloaded_deps directly on the new
   482	    dependencies.)
   483	    """
   484	    worklist = list(initial)
   485	    seen: set[str] = set()
   486	    unloaded = []
   487	    while worklist:
   488	        node = worklist.pop()
   489	        if node in seen or node not in graph:
   490	            continue
   491	        seen.add(node)
   492	        if node not in manager.modules:
   493	            ancestors = graph[node].ancestors or []
   494	            worklist.extend(graph[node].dependencies + ancestors)
   495	            unloaded.append(node)
   496
   497	    return unloaded

   500	def ensure_deps_loaded(module: str, deps: dict[str, set[str]], graph: dict[str, State]) -> None:
   501	    """Ensure that the dependencies on a module are loaded.
   502
   503	    Dependencies are loaded into the 'deps' dictionary.
   504
   505	    This also requires loading dependencies from any parent modules,
   506	    since dependencies will get stored with parent modules when a module
   507	    doesn't exist.
   508	    """
   509	    if module in graph and graph[module].fine_grained_deps_loaded:
   510	        return
   511	    parts = module.split(".")
   512	    for i in range(len(parts)):
   513	        base = ".".join(parts[: i + 1])
   514	        if base in graph and not graph[base].fine_grained_deps_loaded:
   515	            merge_dependencies(graph[base].load_fine_grained_deps(), deps)
   516	            graph[base].fine_grained_deps_loaded = True

   571
   572	    Don't propagate changes to elsewhere in the program. Raise CompileError on
   573	    encountering a blocking error.
   574
   575	    Args:
   576	        module: Changed module (modified, created or deleted)
   577	        path: Path of the changed module
   578	        manager: Build manager
   579	        graph: Build graph
   580	        force_removed: If True, consider the module removed from the build even it the
   581	            file exists
   582
   583	    Returns a named tuple describing the result (see above for details).
   584	    """
   585	    if module not in graph:
   586	        manager.log_fine_grained(f"new module {module!r}")
   587
   588	    if not manager.fscache.isfile(path) or force_removed:
   589	        delete_module(module, path, graph, manager)
   590	        return NormalUpdate(module, path, [], None)
   591
   592	    sources = get_sources(manager.fscache, previous_modules, [(module, path)], followed)
   593
   594	    if module in manager.missing_modules:
   595	        manager.missing_modules.remove(module)
   596
   597	    orig_module = module
   598	    orig_state = graph.get(module)
   599	    orig_tree = manager.modules.get(module)
   600
   601	    def restore(ids: list[str]) -> None:
   602	        # For each of the modules in ids, restore that id's old
   603	        # manager.modules and graphs entries. (Except for the original
   604	        # module, this means deleting them.)
   605	        for id in ids:
   606	            if id == orig_module and orig_tree:
   607	                manager.modules[id] = orig_tree
   608	            elif id in manager.modules:
   609	                del manager.modules[id]
   610	            if id == orig_module and orig_state:
   611	                graph[id] = orig_state
   612	            elif id in graph:
   613	                del graph[id]
   614
   615	    new_modules: list[State] = []
   616	    try:
   617	        if module in graph:
   618	            del graph[module]
   619	        load_graph(sources, manager, graph, new_modules)
   620	    except CompileError as err:
   621	        # Parse error somewhere in the program -- a blocker
   622	        assert err.module_with_blocker
   623	        restore([module] + [st.id for st in new_modules])
   624	        return BlockedUpdate(err.module_with_blocker, path, [], err.messages)
   625
   626	    # Reparsing the file may have brought in dependencies that we
   627	    # didn't have before. Make sure that they are loaded to restore
   628	    # the invariant that a module having a loaded tree implies that
   629	    # its dependencies do as well.
   630	    ensure_trees_loaded(manager, graph, graph[module].dependencies)
   631
   632	    # Find any other modules brought in by imports.
   633	    changed_modules = [(st.id, st.xpath) for st in new_modules]
   634
   635	    # If there are multiple modules to process, only process one of them and return
   636	    # the remaining ones to the caller.
   637	    if len(changed_modules) > 1:
   638	        # As an optimization, look for a module that imports no other changed modules.
   639	        module, path = find_relative_leaf_module(changed_modules, graph)
   640	        changed_modules.remove((module, path))
   641	        remaining_modules = changed_modules
   642	        # The remaining modules haven't been processed yet so drop them.
   643	        restore([id for id, _ in remaining_modules])
   644	        manager.log_fine_grained(f"--> {module!r} (newly imported)")
   645	    else:
   646	        remaining_modules = []
   647
   648	    state = graph[module]
   649
   650	    # Process the changed file.
   651	    state.parse_file()
   652	    assert state.tree is not None, "file must be at least parsed"
   653	    t0 = time.time()
   654	    try:
   655	        semantic_analysis_for_scc(graph, [state.id], manager.errors)
   656	    except CompileError as err:
   657	        # There was a blocking error, so module AST is incomplete. Restore old modules.
   658	        restore([module])
   659	        return BlockedUpdate(module, path, remaining_modules, err.messages)
   660
   661	    # Merge old and new ASTs.
   662	    new_modules_dict: dict[str, MypyFile | None] = {module: state.tree}
   663	    replace_modules_with_new_variants(manager, graph, {orig_module: orig_tree}, new_modules_dict)
   664
   665	    t1 = time.time()
   666	    # Perform type checking.
   667	    state.type_checker().reset()
   668	    state.type_check_first_pass()
   669	    state.type_check_second_pass()
   670	    state.detect_possibly_undefined_vars()
   671	    state.generate_unused_ignore_notes()
   672	    state.generate_ignore_without_code_notes()
   673	    t2 = time.time()
   674	    state.finish_passes()
   675	    t3 = time.time()
   676	    manager.add_stats(semanal_time=t1 - t0, typecheck_time=t2 - t1, finish_passes_time=t3 - t2)
   677
   678	    graph[module] = state
   679
   680	    return NormalUpdate(module, path, remaining_modules, state.tree)

   683	def find_relative_leaf_module(modules: list[tuple[str, str]], graph: Graph) -> tuple[str, str]:
   684	    """Find a module in a list that directly imports no other module in the list.
   685
   686	    If no such module exists, return the lexicographically first module from the list.
   687	    Always return one of the items in the modules list.
   688
   689	    NOTE: If both 'abc' and 'typing' have changed, an effect of the above rule is that
   690	        we prefer 'abc', even if both are in the same SCC. This works around a false
   691	        positive in 'typing', at least in tests.
   692
   693	    Args:
   694	        modules: List of (module, path) tuples (non-empty)
   695	        graph: Program import graph that contains all modules in the module list
   696	    """
   697	    assert modules
   698	    # Sort for repeatable results.
   699	    modules = sorted(modules)
   700	    module_set = {module for module, _ in modules}
   701	    for module, path in modules:
   702	        state = graph[module]
   703	        if len(set(state.dependencies) & module_set) == 0:
   704	            # Found it!
   705	            return module, path
   706	    # Could not find any. Just return the first module (by lexicographic order).
   707	    return modules[0]

   710	def delete_module(module_id: str, path: str, graph: Graph, manager: BuildManager) -> None:
   711	    manager.log_fine_grained(f"delete module {module_id!r}")
   712	    # TODO: Remove deps for the module (this only affects memory use, not correctness)
   713	    if module_id in graph:
   714	        del graph[module_id]
   715	    if module_id in manager.modules:
   716	        del manager.modules[module_id]
   717	    components = module_id.split(".")
   718	    if len(components) > 1:
   719	        # Delete reference to module in parent module.
   720	        parent_id = ".".join(components[:-1])
   721	        # If parent module is ignored, it won't be included in the modules dictionary.
   722	        if parent_id in manager.modules:
   723	            parent = manager.modules[parent_id]
   724	            if components[-1] in parent.names:
   725	                del parent.names[components[-1]]
   726	    # If the module is removed from the build but still exists, then
   727	    # we mark it as missing so that it will get picked up by import from still.
   728	    if manager.fscache.isfile(path):
   729	        manager.missing_modules.add(module_id)

   759	def calculate_active_triggers(
   760	    manager: BuildManager,
   761	    old_snapshots: dict[str, dict[str, SymbolSnapshot]],
   762	    new_modules: dict[str, MypyFile | None],
   763	) -> set[str]:
   764	    """Determine activated triggers by comparing old and new symbol tables.
   765
   766	    For example, if only the signature of function m.f is different in the new
   767	    symbol table, return {'<m.f>'}.
   768	    """
   769	    names: set[str] = set()
   770	    for id in new_modules:
   771	        snapshot1 = old_snapshots.get(id)
   772	        if snapshot1 is None:
   773	            names.add(id)
   774	            snapshot1 = {}
   775	        new = new_modules[id]
   776	        if new is None:
   777	            snapshot2 = snapshot_symbol_table(id, SymbolTable())
   778	            names.add(id)
   779	        else:
   780	            snapshot2 = snapshot_symbol_table(id, new.names)
   781	        diff = compare_symbol_table_snapshots(id, snapshot1, snapshot2)
   782	        package_nesting_level = id.count(".")
   783	        for item in diff.copy():
   784	            if item.count(".") <= package_nesting_level + 1 and item.split(".")[-1] not in (
   785	                "__builtins__",
   786	                "__file__",
   787	                "__name__",
   788	                "__package__",
   789	                "__doc__",
   790	            ):
   791	                # Activate catch-all wildcard trigger for top-level module changes (used for
   792	                # "from m import *"). This also gets triggered by changes to module-private
   793	                # entries, but as these unneeded dependencies only result in extra processing,
   794	                # it's a minor problem.
   795	                #
   796	                # TODO: Some __* names cause mistriggers. Fix the underlying issue instead of
   797	                #     special casing them here.
   798	                diff.add(id + WILDCARD_TAG)
   799	            if item.count(".") > package_nesting_level + 1:
   800	                # These are for changes within classes, used by protocols.
   801	                diff.add(item.rsplit(".", 1)[0] + WILDCARD_TAG)
   802
   803	        names |= diff
   804	    return {make_trigger(name) for name in names}

   807	def replace_modules_with_new_variants(
   808	    manager: BuildManager,
   809	    graph: dict[str, State],
   810	    old_modules: dict[str, MypyFile | None],
   811	    new_modules: dict[str, MypyFile | None],
   812	) -> None:
   813	    """Replace modules with newly builds versions.
   814
   815	    Retain the identities of externally visible AST nodes in the
   816	    old ASTs so that references to the affected modules from other
   817	    modules will still be valid (unless something was deleted or
   818	    replaced with an incompatible definition, in which case there
   819	    will be dangling references that will be handled by
   820	    propagate_changes_using_dependencies).
   821	    """
   822	    for id in new_modules:
   823	        preserved_module = old_modules.get(id)
   824	        new_module = new_modules[id]
   825	        if preserved_module and new_module is not None:
   826	            merge_asts(preserved_module, preserved_module.names, new_module, new_module.names)
   827	            manager.modules[id] = preserved_module
   828	            graph[id].tree = preserved_module

   831	def propagate_changes_using_dependencies(
   832	    manager: BuildManager,
   833	    graph: dict[str, State],
   834	    deps: dict[str, set[str]],
   835	    triggered: set[str],
   836	    up_to_date_modules: set[str],
   837	    targets_with_errors: set[str],
   838	    processed_targets: list[str],
   839	) -> list[tuple[str, str]]:
   840	    """Transitively rechecks targets based on triggers and the dependency map.
   841
   842	    Returns a list (module id, path) tuples representing modules that contain
   843	    a target that needs to be reprocessed but that has not been parsed yet.
   844
   845	    Processed targets should be appended to processed_targets (used in tests only,
   846	    to test the order of processing targets).
   847	    """
   848
   849	    num_iter = 0
   850	    remaining_modules: list[tuple[str, str]] = []
   851
   852	    # Propagate changes until nothing visible has changed during the last
   853	    # iteration.
   854	    while triggered or targets_with_errors:
   855	        num_iter += 1
   856	        if num_iter > MAX_ITER:
   857	            raise RuntimeError("Max number of iterations (%d) reached (endless loop?)" % MAX_ITER)
   858
   859	        todo, unloaded, stale_protos = find_targets_recursive(
   860	            manager, graph, triggered, deps, up_to_date_modules
   861	        )
   862	        # TODO: we sort to make it deterministic, but this is *incredibly* ad hoc
   863	        remaining_modules.extend((id, graph[id].xpath) for id in sorted(unloaded))
   864	        # Also process targets that used to have errors, as otherwise some
   865	        # errors might be lost.
   866	        for target in targets_with_errors:
   867	            id = module_prefix(graph, target)
   868	            if id is not None and id not in up_to_date_modules:
   869	                if id not in todo:
   870	                    todo[id] = set()
   871	                manager.log_fine_grained(f"process target with error: {target}")
   872	                more_nodes, _ = lookup_target(manager, target)
   873	                todo[id].update(more_nodes)
   874	        triggered = set()
   875	        # First invalidate subtype caches in all stale protocols.
   876	        # We need to do this to avoid false negatives if the protocol itself is
   877	        # unchanged, but was marked stale because its sub- (or super-) type changed.
   878	        for info in stale_protos:
   879	            type_state.reset_subtype_caches_for(info)
   880	        # Then fully reprocess all targets.
   881	        # TODO: Preserve order (set is not optimal)
   882	        for id, nodes in sorted(todo.items(), key=lambda x: x[0]):
   883	            assert id not in up_to_date_modules
   884	            triggered |= reprocess_nodes(manager, graph, id, nodes, deps, processed_targets)
   885	        # Changes elsewhere may require us to reprocess modules that were
   886	        # previously considered up to date. For example, there may be a
   887	        # dependency loop that loops back to an originally processed module.
   888	        up_to_date_modules = set()
   889	        targets_with_errors = set()
   890	        if is_verbose(manager):
   891	            manager.log_fine_grained(f"triggered: {list(triggered)!r}")
   892
   893	    return remaining_modules
   894

   896	def find_targets_recursive(
   897	    manager: BuildManager,
   898	    graph: Graph,
   899	    triggers: set[str],
   900	    deps: dict[str, set[str]],
   901	    up_to_date_modules: set[str],
   902	) -> tuple[dict[str, set[FineGrainedDeferredNode]], set[str], set[TypeInfo]]:
   903	    """Find names of all targets that need to reprocessed, given some triggers.
   904
   905	    Returns: A tuple containing a:
   906	     * Dictionary from module id to a set of stale targets.
   907	     * A set of module ids for unparsed modules with stale targets.
   908	    """
   909	    result: dict[str, set[FineGrainedDeferredNode]] = {}
   910	    worklist = triggers
   911	    processed: set[str] = set()
   912	    stale_protos: set[TypeInfo] = set()
   913	    unloaded_files: set[str] = set()
   914
   915	    # Find AST nodes corresponding to each target.
   916	    #
   917	    # TODO: Don't rely on a set, since the items are in an unpredictable order.
   918	    while worklist:
   919	        processed |= worklist
   920	        current = worklist
   921	        worklist = set()
   922	        for target in current:
   923	            if target.startswith("<"):
   924	                module_id = module_prefix(graph, trigger_to_target(target))
   925	                if module_id:
   926	                    ensure_deps_loaded(module_id, deps, graph)
   927
   928	                worklist |= deps.get(target, set()) - processed
   929	            else:
   930	                module_id = module_prefix(graph, target)
   931	                if module_id is None:
   932	                    # Deleted module.
   933	                    continue
   934	                if module_id in up_to_date_modules:
   935	                    # Already processed.
   936	                    continue
   937	                if (
   938	                    module_id not in manager.modules
   939	                    or manager.modules[module_id].is_cache_skeleton
   940	                ):
   941	                    # We haven't actually parsed and checked the module, so we don't have
   942	                    # access to the actual nodes.
   943	                    # Add it to the queue of files that need to be processed fully.
   944	                    unloaded_files.add(module_id)
   945	                    continue
   946
   947	                if module_id not in result:
   948	                    result[module_id] = set()
   949	                manager.log_fine_grained(f"process: {target}")
   950	                deferred, stale_proto = lookup_target(manager, target)
   951	                if stale_proto:
   952	                    stale_protos.add(stale_proto)
   953	                result[module_id].update(deferred)
   954
   955	    return result, unloaded_files, stale_protos

   958	def reprocess_nodes(
   959	    manager: BuildManager,
   960	    graph: dict[str, State],
   961	    module_id: str,
   962	    nodeset: set[FineGrainedDeferredNode],
   963	    deps: dict[str, set[str]],
   964	    processed_targets: list[str],
   965	) -> set[str]:
   966	    """Reprocess a set of nodes within a single module.
   967
   968	    Return fired triggers.
   969	    """
   970	    if module_id not in graph:
   971	        manager.log_fine_grained("%s not in graph (blocking errors or deleted?)" % module_id)
   972	        return set()
   973
   974	    file_node = manager.modules[module_id]
   975	    old_symbols = find_symbol_tables_recursive(file_node.fullname, file_node.names)
   976	    old_symbols = {name: names.copy() for name, names in old_symbols.items()}
   977	    old_symbols_snapshot = snapshot_symbol_table(file_node.fullname, file_node.names)
   978
   979	    def key(node: FineGrainedDeferredNode) -> int:
   980	        # Unlike modules which are sorted by name within SCC,
   981	        # nodes within the same module are sorted by line number, because
   982	        # this is how they are processed in normal mode.
   983	        return node.node.line
   984
   985	    nodes = sorted(nodeset, key=key)
   986
   987	    state = graph[module_id]
   988	    options = state.options
   989	    manager.errors.set_file_ignored_lines(
   990	        file_node.path, file_node.ignored_lines, options.ignore_errors or state.ignore_all
   991	    )
   992	    manager.errors.set_skipped_lines(file_node.path, file_node.skipped_lines)
   993
   994	    targets = set()
   995	    for node in nodes:
   996	        target = target_from_node(module_id, node.node)
   997	        if target is not None:
   998	            targets.add(target)
   999	    manager.errors.clear_errors_in_targets(file_node.path, targets)
  1000
  1001	    # If one of the nodes is the module itself, emit any errors that
  1002	    # happened before semantic analysis.
  1003	    for target in targets:
  1004	        if target == module_id:
  1005	            for info in graph[module_id].early_errors:
  1006	                manager.errors.add_error_info(info)
  1007
  1008	    # Strip semantic analysis information.
  1009	    saved_attrs: SavedAttributes = {}
  1010	    for deferred in nodes:
  1011	        processed_targets.append(deferred.node.fullname)
  1012	        strip_target(deferred.node, saved_attrs)
  1013	    semantic_analysis_for_targets(graph[module_id], nodes, graph, saved_attrs)
  1014	    # Merge symbol tables to preserve identities of AST nodes. The file node will remain
  1015	    # the same, but other nodes may have been recreated with different identities, such as
  1016	    # NamedTuples defined using assignment statements.
  1017	    new_symbols = find_symbol_tables_recursive(file_node.fullname, file_node.names)
  1018	    for name in old_symbols:
  1019	        if name in new_symbols:
  1020	            merge_asts(file_node, old_symbols[name], file_node, new_symbols[name])
  1021
  1022	    # Type check.
  1023	    checker = graph[module_id].type_checker()
  1024	    checker.reset()
  1025	    # We seem to need additional passes in fine-grained incremental mode.
  1026	    checker.pass_num = 0
  1027	    checker.last_pass = 3
  1028	    more = checker.check_second_pass(nodes)
  1029	    while more:
  1030	        more = False
  1031	        if graph[module_id].type_checker().check_second_pass():
  1032	            more = True
  1033
  1034	    if manager.options.export_types:
  1035	        manager.all_types.update(graph[module_id].type_map())
  1036
  1037	    new_symbols_snapshot = snapshot_symbol_table(file_node.fullname, file_node.names)
  1038	    # Check if any attribute types were changed and need to be propagated further.
  1039	    changed = compare_symbol_table_snapshots(
  1040	        file_node.fullname, old_symbols_snapshot, new_symbols_snapshot
  1041	    )
  1042	    new_triggered = {make_trigger(name) for name in changed}
  1043
  1044	    # Dependencies may have changed.
  1045	    update_deps(module_id, nodes, graph, deps, options)
  1046
  1047	    # Report missing imports.
  1048	    graph[module_id].verify_dependencies()
  1049
  1050	    graph[module_id].free_state()
  1051
  1052	    return new_triggered
  1053

  1055	def find_symbol_tables_recursive(prefix: str, symbols: SymbolTable) -> dict[str, SymbolTable]:
  1056	    """Find all nested symbol tables.
  1057
  1058	    Args:
  1059	        prefix: Full name prefix (used for return value keys and to filter result so that
  1060	            cross references to other modules aren't included)
  1061	        symbols: Root symbol table
  1062
  1063	    Returns a dictionary from full name to corresponding symbol table.
  1064	    """
  1065	    result = {prefix: symbols}
  1066	    for name, node in symbols.items():
  1067	        if isinstance(node.node, TypeInfo) and node.node.fullname.startswith(prefix + "."):
  1068	            more = find_symbol_tables_recursive(prefix + "." + name, node.node.names)
  1069	            result.update(more)
  1070	    return result

  1073	def update_deps(
  1074	    module_id: str,
  1075	    nodes: list[FineGrainedDeferredNode],
  1076	    graph: dict[str, State],
  1077	    deps: dict[str, set[str]],
  1078	    options: Options,
  1079	) -> None:
  1080	    for deferred in nodes:
  1081	        node = deferred.node
  1082	        type_map = graph[module_id].type_map()
  1083	        tree = graph[module_id].tree
  1084	        assert tree is not None, "Tree must be processed at this stage"
  1085	        new_deps = get_dependencies_of_target(
  1086	            module_id, tree, node, type_map, options.python_version
  1087	        )
  1088	        for trigger, targets in new_deps.items():
  1089	            deps.setdefault(trigger, set()).update(targets)
  1090	    # Merge also the newly added protocol deps (if any).
  1091	    type_state.update_protocol_deps(deps)

  1094	def lookup_target(
  1095	    manager: BuildManager, target: str
  1096	) -> tuple[list[FineGrainedDeferredNode], TypeInfo | None]:
  1097	    """Look up a target by fully-qualified name.
  1098
  1099	    The first item in the return tuple is a list of deferred nodes that
  1100	    needs to be reprocessed. If the target represents a TypeInfo corresponding
  1101	    to a protocol, return it as a second item in the return tuple, otherwise None.
  1102	    """
  1103
  1104	    def not_found() -> None:
  1105	        manager.log_fine_grained(f"Can't find matching target for {target} (stale dependency?)")
  1106
  1107	    modules = manager.modules
  1108	    items = split_target(modules, target)
