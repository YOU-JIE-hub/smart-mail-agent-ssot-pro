    35	def nextRoundNumber(x):
    36	    """Return the first 'nice round number' greater than or equal to x
    37
    38	    Used in selecting apropriate tick mark intervals; we say we want
    39	    an interval which places ticks at least 10 points apart, work out
    40	    what that is in chart space, and ask for the nextRoundNumber().
    41	    Tries the series 1,2,5,10,20,50,100.., going up or down as needed.
    42	    """
    43
    44	    #guess to nearest order of magnitude
    45	    if x in (0, 1):
    46	        return x
    47
    48	    if x < 0:
    49	        return -1.0 * nextRoundNumber(-x)
    50	    else:
    51	        lg = int(log10(x))
    52
    53	        if lg == 0:
    54	            if x < 1:
    55	                base = 0.1
    56	            else:
    57	                base = 1.0
    58	        elif lg < 0:
    59	            base = 10.0 ** (lg - 1)
    60	        else:
    61	            base = 10.0 ** lg    # e.g. base(153) = 100
    62	        # base will always be lower than x
    63
    64	        if base >= x:
    65	            return base * 1.0
    66	        elif (base * 2) >= x:
    67	            return base * 2.0
    68	        elif (base * 5) >= x:
    69	            return base * 5.0
    70	        else:
    71	            return base * 10.0

   121	class _Segment:
   122	    def __init__(self,s,i,data):
   123	        S = data[s]
   124	        x0 = S[i-1][0]
   125	        y0 = S[i-1][1]
   126	        x1 = S[i][0]
   127	        y1 = S[i][1]

   129	            x0,y0,x1,y1 = x1,y1,x0,y0
   130	        # (y-y0)*(x1-x0) = (y1-y0)*(x-x0)
   131	        # (x1-x0)*y + (y0-y1)*x = y0*(x1-x0)+x0*(y0-y1)
   132	        # a*y+b*x = c

   134	        self.b = float(y1-y0)

   136	        self.x1 = x1
   137	        self.y0 = y0

   141	        self.s = s

   148	    def intersect(self,o,I):
   149	        '''try to find an intersection with _Segment o
   150	        '''
   151	        x0 = self.x0
   152	        ox0 = o.x0
   153	        assert x0<=ox0
   154	        if ox0>self.x1: return 1
   155	        if o.s==self.s and o.i in (self.i-1,self.i+1): return
   156	        a = self.a
   157	        b = self.b
   158	        oa = o.a
   159	        ob = o.b
   160	        det = ob*a - oa*b
   161	        if -1e-8<det<1e-8: return
   162	        dx = x0 - ox0
   163	        dy = self.y0 - o.y0
   164	        u = (oa*dy - ob*dx)/det
   165	        ou = (a*dy - b*dx)/det
   166	        if u<0 or u>1 or ou<0 or ou>1: return
   167	        x = x0 + u*a
   168	        y = self.y0 + u*b
   169	        if _ZERO<u<_ONE:
   170	            t = self.s,self.i,x,y
   171	            if t not in I: I.append(t)
   172	        if _ZERO<ou<_ONE:
   173	            t = o.s,o.i,x,y
   174	            if t not in I:  I.append(t)

   179	def find_intersections(data,small=0):
   180	    '''
   181	    data is a sequence of series
   182	    each series is a list of (x,y) coordinates
   183	    where x & y are ints or floats
   184
   185	    find_intersections returns a sequence of 4-tuples
   186	        i, j, x, y
   187
   188	    where i is a data index j is an insertion position for data[i]
   189	    and x, y are coordinates of an intersection of series data[i]
   190	    with some other series. If correctly implemented we get all such
   191	    intersections. We don't count endpoint intersections and consider
   192	    parallel lines as non intersecting (even when coincident).
   193	    We ignore segments that have an estimated size less than small.
   194	    '''
   195
   196	    #find all line segments
   197	    S = []
   198	    a = S.append
   199	    for s in range(len(data)):
   200	        ds = data[s]
   201	        if not ds: continue
   202	        n = len(ds)
   203	        if n==1: continue
   204	        for i in range(1,n):
   205	            seg = _Segment(s,i,data)
   206	            if seg.a+abs(seg.b)>=small: a(seg)
   207	    S.sort(key=_segKey)
   208	    I = []
   209	    n = len(S)
   210	    for i in range(0,n-1):
   211	        s = S[i]
   212	        for j in range(i+1,n):
   213	            if s.intersect(S[j],I)==1: break
   214	    I.sort()
