   181	    """Identity type transformation.
   182
   183	    Subclass this and override some methods to implement a non-trivial
   184	    transformation.

   186	    We cache the results of certain translations to avoid
   187	    massively expanding the sizes of types.

   189
   190	    def __init__(self, cache: dict[Type, Type] | None = None) -> None:
   191	        # For deduplication of results

   194	    def get_cached(self, t: Type) -> Type | None:

   257	        return t.copy_modified(

   259	            ret_type=t.ret_type.accept(self),
   260	            variables=self.translate_variables(t.variables),

   264	        return TupleType(
   265	            self.translate_types(t.items),

   267	            cast(Any, t.partial_fallback.accept(self)),
   268	            t.line,
   269	            t.column,
   270	        )

   272	    def visit_typeddict_type(self, t: TypedDictType, /) -> Type:
   273	        # Use cache to avoid O(n**2) or worse expansion of types during translation
   274	        if cached := self.get_cached(t):
   275	            return cached
   276	        items = {item_name: item_type.accept(self) for (item_name, item_type) in t.items.items()}
   277	        result = TypedDictType(
   278	            items,
   279	            t.required_keys,
   280	            t.readonly_keys,
   281	            # TODO: This appears to be unsafe.
   282	            cast(Any, t.fallback.accept(self)),
   283	            t.line,
   284	            t.column,

   286	        self.set_cached(t, result)

   290	        fallback = t.fallback.accept(self)
   291	        assert isinstance(fallback, Instance)  # type: ignore[misc]
   292	        return LiteralType(value=t.value, fallback=fallback, line=t.line, column=t.column)

   295	        # Use cache to avoid O(n**2) or worse expansion of types during translation
   296	        # (only for large unions, since caching adds overhead)
   297	        use_cache = len(t.items) > 3
   298	        if use_cache and (cached := self.get_cached(t)):
   299	            return cached

   301	        result = UnionType(
   302	            self.translate_types(t.items),
   303	            t.line,
   304	            t.column,
   305	            uses_pep604_syntax=t.uses_pep604_syntax,
   306	        )

   308	            self.set_cached(t, result)

   317	        return variables

   319	    def visit_overloaded(self, t: Overloaded, /) -> Type:
   320	        items: list[CallableType] = []
   321	        for item in t.items:
   322	            new = item.accept(self)
   323	            assert isinstance(new, CallableType)  # type: ignore[misc]
   324	            items.append(new)
   325	        return Overloaded(items=items)

   327	    def visit_type_type(self, t: TypeType, /) -> Type:
   328	        return TypeType.make_normalized(t.item.accept(self), line=t.line, column=t.column)

   331	    def visit_type_alias_type(self, t: TypeAliasType, /) -> Type:
   332	        # This method doesn't have a default implementation for type translators,
   333	        # because type aliases are special: some information is contained in the
   334	        # TypeAlias node, and we normally don't generate new nodes. Every subclass
   335	        # must implement this depending on its semantics.
   336	        pass

   185	class TypeAnalyser(SyntheticTypeVisitor[Type], TypeAnalyzerPluginInterface):
   186	    """Semantic analyzer for types.
   187
   188	    Converts unbound types into bound types. This is a no-op for already
   189	    bound types.
   190
   191	    If an incomplete reference is encountered, this does a defer. The
   192	    caller never needs to defer.
   193	    """
   194
   195	    # Is this called from an untyped function definition?
   196	    in_dynamic_func: bool = False
   197	    # Is this called from global scope?
   198	    global_scope: bool = True
   199
   200	    def __init__(
   201	        self,
   202	        api: SemanticAnalyzerCoreInterface,
   203	        tvar_scope: TypeVarLikeScope,
   204	        plugin: Plugin,
   205	        options: Options,
   206	        cur_mod_node: MypyFile,
   207	        is_typeshed_stub: bool,
   208	        *,
   209	        defining_alias: bool = False,
   210	        python_3_12_type_alias: bool = False,
   211	        allow_tuple_literal: bool = False,
   212	        allow_unbound_tvars: bool = False,
   213	        allow_placeholder: bool = False,
   214	        allow_typed_dict_special_forms: bool = False,
   215	        allow_final: bool = True,
   216	        allow_param_spec_literals: bool = False,
   217	        allow_unpack: bool = False,
   218	        report_invalid_types: bool = True,
   219	        prohibit_self_type: str | None = None,
   220	        prohibit_special_class_field_types: str | None = None,
   221	        allowed_alias_tvars: list[TypeVarLikeType] | None = None,
   222	        allow_type_any: bool = False,
   223	        alias_type_params_names: list[str] | None = None,
   224	    ) -> None:
   225	        self.api = api
   226	        self.fail_func = api.fail
   227	        self.note_func = api.note
   228	        self.tvar_scope = tvar_scope
   229	        # Are we analysing a type alias definition rvalue?
   230	        self.defining_alias = defining_alias
   231	        self.python_3_12_type_alias = python_3_12_type_alias
   232	        self.allow_tuple_literal = allow_tuple_literal
   233	        # Positive if we are analyzing arguments of another (outer) type
   234	        self.nesting_level = 0
   235	        # Should we allow new type syntax when targeting older Python versions
   236	        # like 'list[int]' or 'X | Y' (allowed in stubs and with `__future__` import)?
   237	        self.always_allow_new_syntax = self.api.is_stub_file or self.api.is_future_flag_set(
   238	            "annotations"
   239	        )
   240	        # Should we accept unbound type variables? This is currently used for class bases,
   241	        # and alias right hand sides (before they are analyzed as type aliases).
   242	        self.allow_unbound_tvars = allow_unbound_tvars
   243	        if allowed_alias_tvars is None:
   244	            allowed_alias_tvars = []
   245	        self.allowed_alias_tvars = allowed_alias_tvars
   246	        self.alias_type_params_names = alias_type_params_names
   247	        # If false, record incomplete ref if we generate PlaceholderType.
   248	        self.allow_placeholder = allow_placeholder
   249	        # Are we in a context where Required[] is allowed?
   250	        self.allow_typed_dict_special_forms = allow_typed_dict_special_forms
   251	        # Set True when we analyze ClassVar else False
   252	        self.allow_final = allow_final
   253	        # Are we in a context where ParamSpec literals are allowed?
   254	        self.allow_param_spec_literals = allow_param_spec_literals
   255	        # Are we in context where literal "..." specifically is allowed?
   256	        self.allow_ellipsis = False
   257	        # Should we report an error whenever we encounter a RawExpressionType outside
   258	        # of a Literal context: e.g. whenever we encounter an invalid type? Normally,
   259	        # we want to report an error, but the caller may want to do more specialized
   260	        # error handling.
   261	        self.report_invalid_types = report_invalid_types
   262	        self.plugin = plugin
   263	        self.options = options
   264	        self.cur_mod_node = cur_mod_node
   265	        self.is_typeshed_stub = is_typeshed_stub
   266	        # Names of type aliases encountered while analysing a type will be collected here.
   267	        self.aliases_used: set[str] = set()
   268	        self.prohibit_self_type = prohibit_self_type
   269	        # Set when we analyze TypedDicts or NamedTuples, since they are special:
   270	        self.prohibit_special_class_field_types = prohibit_special_class_field_types
   271	        # Allow variables typed as Type[Any] and type (useful for base classes).
   272	        self.allow_type_any = allow_type_any
   273	        self.allow_type_var_tuple = False
   274	        self.allow_unpack = allow_unpack

   292	    def not_declared_in_type_params(self, tvar_name: str) -> bool:
   293	        return (
   294	            self.alias_type_params_names is not None
   295	            and tvar_name not in self.alias_type_params_names

   298	    def visit_unbound_type_nonoptional(self, t: UnboundType, defining_literal: bool) -> Type:
   299	        sym = self.lookup_qualified(t.name, t)
   300	        param_spec_name = None
   301	        if t.name.endswith((".args", ".kwargs")):
   302	            param_spec_name = t.name.rsplit(".", 1)[0]
   303	            maybe_param_spec = self.lookup_qualified(param_spec_name, t)
   304	            if maybe_param_spec and isinstance(maybe_param_spec.node, ParamSpecExpr):
   305	                sym = maybe_param_spec
   306	            else:
   307	                param_spec_name = None
   308
   309	        if sym is not None:
   310	            node = sym.node
   311	            if isinstance(node, PlaceholderNode):
   312	                if node.becomes_typeinfo:
   313	                    # Reference to placeholder type.
   314	                    if self.api.final_iteration:
   315	                        self.cannot_resolve_type(t)
   316	                        return AnyType(TypeOfAny.from_error)
   317	                    elif self.allow_placeholder:
   318	                        self.api.defer()
   319	                    else:
   320	                        self.api.record_incomplete_ref()
   321	                    # Always allow ParamSpec for placeholders, if they are actually not valid,
   322	                    # they will be reported later, after we resolve placeholders.
   323	                    return PlaceholderType(
   324	                        node.fullname,
   325	                        self.anal_array(
   326	                            t.args,
   327	                            allow_param_spec=True,
   328	                            allow_param_spec_literals=True,
   329	                            allow_unpack=True,
   330	                        ),
   331	                        t.line,
   332	                    )
   333	                else:
   334	                    if self.api.final_iteration:
   335	                        self.cannot_resolve_type(t)
   336	                        return AnyType(TypeOfAny.from_error)
   337	                    else:
   338	                        # Reference to an unknown placeholder node.
   339	                        self.api.record_incomplete_ref()
   340	                        return AnyType(TypeOfAny.special_form)
   341	            if node is None:
   342	                self.fail(f"Internal error (node is None, kind={sym.kind})", t)
   343	                return AnyType(TypeOfAny.special_form)
   344	            fullname = node.fullname
   345	            hook = self.plugin.get_type_analyze_hook(fullname)
   346	            if hook is not None:
   347	                return hook(AnalyzeTypeContext(t, t, self))
   348	            tvar_def = self.tvar_scope.get_binding(sym)
   349	            if isinstance(sym.node, ParamSpecExpr):
   350	                if tvar_def is None:
   351	                    if self.allow_unbound_tvars:
   352	                        return t
   353	                    name = param_spec_name or t.name
   354	                    if self.defining_alias and self.not_declared_in_type_params(t.name):
   355	                        msg = f'ParamSpec "{name}" is not included in type_params'
   356	                    else:
   357	                        msg = f'ParamSpec "{name}" is unbound'
   358	                    self.fail(msg, t, code=codes.VALID_TYPE)
   359	                    return AnyType(TypeOfAny.from_error)
   360	                assert isinstance(tvar_def, ParamSpecType)
   361	                if len(t.args) > 0:
   362	                    self.fail(
   363	                        f'ParamSpec "{t.name}" used with arguments', t, code=codes.VALID_TYPE
   364	                    )
   365	                if param_spec_name is not None and not self.allow_param_spec_literals:
   366	                    self.fail(
   367	                        "ParamSpec components are not allowed here", t, code=codes.VALID_TYPE
   368	                    )
   369	                    return AnyType(TypeOfAny.from_error)
   370	                # Change the line number
   371	                return ParamSpecType(
   372	                    tvar_def.name,
   373	                    tvar_def.fullname,
   374	                    tvar_def.id,
   375	                    tvar_def.flavor,
   376	                    tvar_def.upper_bound,
   377	                    tvar_def.default,
   378	                    line=t.line,
   379	                    column=t.column,
   380	                )
   381	            if (
   382	                isinstance(sym.node, TypeVarExpr)
   383	                and self.defining_alias
   384	                and not defining_literal
   385	                and (tvar_def is None or tvar_def not in self.allowed_alias_tvars)
   386	            ):
   387	                if self.not_declared_in_type_params(t.name):
   388	                    if self.python_3_12_type_alias:
   389	                        msg = message_registry.TYPE_PARAMETERS_SHOULD_BE_DECLARED.format(
   390	                            f'"{t.name}"'
   391	                        )
   392	                    else:
   393	                        msg = f'Type variable "{t.name}" is not included in type_params'
   394	                else:
   395	                    msg = f'Can\'t use bound type variable "{t.name}" to define generic alias'
   396	                self.fail(msg, t, code=codes.VALID_TYPE)
   397	                return AnyType(TypeOfAny.from_error)
   398	            if isinstance(sym.node, TypeVarExpr) and tvar_def is not None:
   399	                assert isinstance(tvar_def, TypeVarType)
   400	                if len(t.args) > 0:
   401	                    self.fail(
   402	                        f'Type variable "{t.name}" used with arguments', t, code=codes.VALID_TYPE
   403	                    )
   404	                # Change the line number
   405	                return tvar_def.copy_modified(line=t.line, column=t.column)
   406	            if isinstance(sym.node, TypeVarTupleExpr) and (
   407	                tvar_def is not None
   408	                and self.defining_alias
   409	                and tvar_def not in self.allowed_alias_tvars
   410	            ):
   411	                if self.not_declared_in_type_params(t.name):
   412	                    msg = f'Type variable "{t.name}" is not included in type_params'
   413	                else:
   414	                    msg = f'Can\'t use bound type variable "{t.name}" to define generic alias'
   415	                self.fail(msg, t, code=codes.VALID_TYPE)
   416	                return AnyType(TypeOfAny.from_error)
   417	            if isinstance(sym.node, TypeVarTupleExpr):
   418	                if tvar_def is None:
   419	                    if self.allow_unbound_tvars:
   420	                        return t
   421	                    if self.defining_alias and self.not_declared_in_type_params(t.name):
   422	                        if self.python_3_12_type_alias:
   423	                            msg = message_registry.TYPE_PARAMETERS_SHOULD_BE_DECLARED.format(
   424	                                f'"{t.name}"'
   425	                            )
   426	                        else:
   427	                            msg = f'TypeVarTuple "{t.name}" is not included in type_params'
   428	                    else:
   429	                        msg = f'TypeVarTuple "{t.name}" is unbound'
   430	                    self.fail(msg, t, code=codes.VALID_TYPE)
   431	                    return AnyType(TypeOfAny.from_error)
   432	                assert isinstance(tvar_def, TypeVarTupleType)
   433	                if not self.allow_type_var_tuple:
   434	                    self.fail(
   435	                        f'TypeVarTuple "{t.name}" is only valid with an unpack',
   436	                        t,
   437	                        code=codes.VALID_TYPE,
   438	                    )
   439	                    return AnyType(TypeOfAny.from_error)
   440	                if len(t.args) > 0:
   441	                    self.fail(
   442	                        f'Type variable "{t.name}" used with arguments', t, code=codes.VALID_TYPE
   443	                    )
   444
   445	                # Change the line number
   446	                return TypeVarTupleType(
   447	                    tvar_def.name,
   448	                    tvar_def.fullname,
   449	                    tvar_def.id,
   450	                    tvar_def.upper_bound,
   451	                    sym.node.tuple_fallback,
   452	                    tvar_def.default,
   453	                    line=t.line,
   454	                    column=t.column,
   455	                )
   456	            special = self.try_analyze_special_unbound_type(t, fullname)
   457	            if special is not None:
   458	                return special
   459	            if isinstance(node, TypeAlias):
   460	                self.aliases_used.add(fullname)
   461	                an_args = self.anal_array(
   462	                    t.args,
   463	                    allow_param_spec=True,
   464	                    allow_param_spec_literals=node.has_param_spec_type,
   465	                    allow_unpack=True,  # Fixed length unpacks can be used for non-variadic aliases.
   466	                )
   467	                if node.has_param_spec_type and len(node.alias_tvars) == 1:
   468	                    an_args = self.pack_paramspec_args(an_args)
   469
   470	                disallow_any = self.options.disallow_any_generics and not self.is_typeshed_stub
   471	                res = instantiate_type_alias(
   472	                    node,
   473	                    an_args,
   474	                    self.fail,
   475	                    node.no_args,
   476	                    t,
   477	                    self.options,
   478	                    unexpanded_type=t,
   479	                    disallow_any=disallow_any,
   480	                    empty_tuple_index=t.empty_tuple_index,
   481	                )
   482	                # The only case where instantiate_type_alias() can return an incorrect instance is
   483	                # when it is top-level instance, so no need to recurse.
   484	                if (
   485	                    isinstance(res, ProperType)
   486	                    and isinstance(res, Instance)
   487	                    and not (self.defining_alias and self.nesting_level == 0)
   488	                    and not validate_instance(res, self.fail, t.empty_tuple_index)
   489	                ):
   490	                    fix_instance(
   491	                        res,
   492	                        self.fail,
   493	                        self.note,
   494	                        disallow_any=disallow_any,
   495	                        options=self.options,
   496	                        use_generic_error=True,
   497	                        unexpanded_type=t,
   498	                    )
   499	                if node.eager:
   500	                    res = get_proper_type(res)
   501	                return res
   502	            elif isinstance(node, TypeInfo):
   503	                return self.analyze_type_with_type_info(node, t.args, t, t.empty_tuple_index)
   504	            elif node.fullname in TYPE_ALIAS_NAMES:
   505	                return AnyType(TypeOfAny.special_form)
   506	            # Concatenate is an operator, no need for a proper type
   507	            elif node.fullname in CONCATENATE_TYPE_NAMES:
   508	                # We check the return type further up the stack for valid use locations
   509	                return self.apply_concatenate_operator(t)
   510	            else:
   511	                return self.analyze_unbound_type_without_type_info(t, sym, defining_literal)
   512	        else:  # sym is None
   513	            return AnyType(TypeOfAny.special_form)

   515	    def pack_paramspec_args(self, an_args: Sequence[Type]) -> list[Type]:
   516	        # "Aesthetic" ParamSpec literals for single ParamSpec: C[int, str] -> C[[int, str]].
   517	        # These do not support mypy_extensions VarArgs, etc. as they were already analyzed
   518	        # TODO: should these be re-analyzed to get rid of this inconsistency?
   519	        count = len(an_args)
   520	        if count == 0:
   521	            return []
   522	        if count == 1 and isinstance(get_proper_type(an_args[0]), AnyType):
   523	            # Single Any is interpreted as ..., rather that a single argument with Any type.
   524	            # I didn't find this in the PEP, but it sounds reasonable.
   525	            return list(an_args)
   526	        if any(isinstance(a, (Parameters, ParamSpecType)) for a in an_args):
   527	            if len(an_args) > 1:
   528	                first_wrong = next(
   529	                    arg for arg in an_args if isinstance(arg, (Parameters, ParamSpecType))
   530	                )
   531	                self.fail(
   532	                    "Nested parameter specifications are not allowed",
   533	                    first_wrong,
   534	                    code=codes.VALID_TYPE,
   535	                )
   536	                return [AnyType(TypeOfAny.from_error)]
   537	            return list(an_args)
   538	        first = an_args[0]
   539	        return [
   540	            Parameters(
   541	                an_args, [ARG_POS] * count, [None] * count, line=first.line, column=first.column
   542	            )
   543	        ]

   545	    def cannot_resolve_type(self, t: UnboundType) -> None:
   546	        # TODO: Move error message generation to messages.py. We'd first
   547	        #       need access to MessageBuilder here. Also move the similar
   548	        #       message generation logic in semanal.py.
   549	        self.api.fail(f'Cannot resolve name "{t.name}" (possible cyclic definition)', t)
   550	        if self.api.is_func_scope():
   551	            self.note("Recursive types are not allowed at function scope", t)

   553	    def apply_concatenate_operator(self, t: UnboundType) -> Type:
   554	        if len(t.args) == 0:
   555	            self.api.fail("Concatenate needs type arguments", t, code=codes.VALID_TYPE)
   556	            return AnyType(TypeOfAny.from_error)
   557
   558	        # Last argument has to be ParamSpec or Ellipsis.
   559	        ps = self.anal_type(t.args[-1], allow_param_spec=True, allow_ellipsis=True)
   560	        if not isinstance(ps, (ParamSpecType, Parameters)):
   561	            if isinstance(ps, UnboundType) and self.allow_unbound_tvars:
   562	                sym = self.lookup_qualified(ps.name, t)
   563	                if sym is not None and isinstance(sym.node, ParamSpecExpr):
   564	                    return ps
   565	            self.api.fail(
   566	                "The last parameter to Concatenate needs to be a ParamSpec",
   567	                t,
   568	                code=codes.VALID_TYPE,
   569	            )
   570	            return AnyType(TypeOfAny.from_error)
   571	        elif isinstance(ps, ParamSpecType) and ps.prefix.arg_types:
   572	            self.api.fail("Nested Concatenates are invalid", t, code=codes.VALID_TYPE)
   573
   574	        args = self.anal_array(t.args[:-1])
   575	        pre = ps.prefix if isinstance(ps, ParamSpecType) else ps
   576
   577	        # mypy can't infer this :(
   578	        names: list[str | None] = [None] * len(args)
   579
   580	        pre = Parameters(
   581	            args + pre.arg_types,
   582	            [ARG_POS] * len(args) + pre.arg_kinds,
   583	            names + pre.arg_names,
   584	            line=t.line,
   585	            column=t.column,
   586	        )
   587	        return ps.copy_modified(prefix=pre) if isinstance(ps, ParamSpecType) else pre

   589	    def try_analyze_special_unbound_type(self, t: UnboundType, fullname: str) -> Type | None:
   590	        """Bind special type that is recognized through magic name such as 'typing.Any'.
   591
   592	        Return the bound type if successful, and return None if the type is a normal type.
   593	        """
   594	        if fullname == "builtins.None":
   595	            return NoneType()
   596	        elif fullname == "typing.Any":
   597	            return AnyType(TypeOfAny.explicit, line=t.line, column=t.column)
   598	        elif fullname in FINAL_TYPE_NAMES:
   599	            if self.prohibit_special_class_field_types:
   600	                self.fail(
   601	                    f"Final[...] can't be used inside a {self.prohibit_special_class_field_types}",
   602	                    t,
   603	                    code=codes.VALID_TYPE,
   604	                )
   605	            else:
   606	                if not self.allow_final:
   607	                    self.fail(
   608	                        "Final can be only used as an outermost qualifier in a variable annotation",
   609	                        t,
   610	                        code=codes.VALID_TYPE,
   611	                    )
   612	            return AnyType(TypeOfAny.from_error)
   613	        elif fullname in TUPLE_NAMES:
   614	            # Tuple is special because it is involved in builtin import cycle
   615	            # and may be not ready when used.
   616	            sym = self.api.lookup_fully_qualified_or_none("builtins.tuple")
   617	            if not sym or isinstance(sym.node, PlaceholderNode):
   618	                if self.api.is_incomplete_namespace("builtins"):
   619	                    self.api.record_incomplete_ref()
   620	                else:
   621	                    self.fail('Name "tuple" is not defined', t)
   622	                return AnyType(TypeOfAny.special_form)
   623	            if len(t.args) == 0 and not t.empty_tuple_index:
   624	                # Bare 'Tuple' is same as 'tuple'
   625	                any_type = self.get_omitted_any(t)
   626	                return self.named_type("builtins.tuple", [any_type], line=t.line, column=t.column)
   627	            if len(t.args) == 2 and isinstance(t.args[1], EllipsisType):
   628	                # Tuple[T, ...] (uniform, variable-length tuple)
   629	                instance = self.named_type("builtins.tuple", [self.anal_type(t.args[0])])
   630	                instance.line = t.line
   631	                return instance
   632	            return self.tuple_type(
   633	                self.anal_array(t.args, allow_unpack=True), line=t.line, column=t.column
   634	            )
   635	        elif fullname == "typing.Union":
   636	            items = self.anal_array(t.args)
   637	            return UnionType.make_union(items)
   638	        elif fullname == "typing.Optional":
   639	            if len(t.args) != 1:
   640	                self.fail(
   641	                    "Optional[...] must have exactly one type argument", t, code=codes.VALID_TYPE
   642	                )
   643	                return AnyType(TypeOfAny.from_error)
   644	            item = self.anal_type(t.args[0])
   645	            return make_optional_type(item)
   646	        elif fullname == "typing.Callable":
   647	            return self.analyze_callable_type(t)
   648	        elif fullname in TYPE_NAMES:
   649	            if len(t.args) == 0:
   650	                if fullname == "typing.Type":
   651	                    any_type = self.get_omitted_any(t)
   652	                    return TypeType(any_type, line=t.line, column=t.column)
   653	                else:
   654	                    # To prevent assignment of 'builtins.type' inferred as 'builtins.object'
   655	                    # See https://github.com/python/mypy/issues/9476 for more information
   656	                    return None
   657	            type_str = "Type[...]" if fullname == "typing.Type" else "type[...]"
   658	            if len(t.args) != 1:
   659	                self.fail(
   660	                    f"{type_str} must have exactly one type argument", t, code=codes.VALID_TYPE
   661	                )
   662	            item = self.anal_type(t.args[0])
   663	            bad_item_name = get_bad_type_type_item(item)
   664	            if bad_item_name:
   665	                self.fail(f'{type_str} can\'t contain "{bad_item_name}"', t, code=codes.VALID_TYPE)
   666	                item = AnyType(TypeOfAny.from_error)
   667	            return TypeType.make_normalized(item, line=t.line, column=t.column)
   668	        elif fullname == "typing.ClassVar":
   669	            if self.nesting_level > 0:
   670	                self.fail(
   671	                    "Invalid type: ClassVar nested inside other type", t, code=codes.VALID_TYPE
   672	                )
   673	            if self.prohibit_special_class_field_types:
   674	                self.fail(
   675	                    f"ClassVar[...] can't be used inside a {self.prohibit_special_class_field_types}",
   676	                    t,
   677	                    code=codes.VALID_TYPE,
   678	                )
   679	            if self.defining_alias:
   680	                self.fail(
   681	                    "ClassVar[...] can't be used inside a type alias", t, code=codes.VALID_TYPE
   682	                )
   683	            if len(t.args) == 0:
   684	                return AnyType(TypeOfAny.from_omitted_generics, line=t.line, column=t.column)
   685	            if len(t.args) != 1:
   686	                self.fail(
   687	                    "ClassVar[...] must have at most one type argument", t, code=codes.VALID_TYPE
   688	                )
   689	                return AnyType(TypeOfAny.from_error)
   690	            return self.anal_type(t.args[0], allow_final=self.options.python_version >= (3, 13))
   691	        elif fullname in NEVER_NAMES:
   692	            return UninhabitedType()
   693	        elif fullname in LITERAL_TYPE_NAMES:
   694	            return self.analyze_literal_type(t)
   695	        elif fullname in ANNOTATED_TYPE_NAMES:
   696	            if len(t.args) < 2:
   697	                self.fail(
   698	                    "Annotated[...] must have exactly one type argument"
   699	                    " and at least one annotation",
   700	                    t,
   701	                    code=codes.VALID_TYPE,
   702	                )
   703	                return AnyType(TypeOfAny.from_error)
   704	            return self.anal_type(
   705	                t.args[0], allow_typed_dict_special_forms=self.allow_typed_dict_special_forms
   706	            )
   707	        elif fullname in ("typing_extensions.Required", "typing.Required"):
   708	            if not self.allow_typed_dict_special_forms:
   709	                self.fail(
   710	                    "Required[] can be only used in a TypedDict definition",
   711	                    t,
   712	                    code=codes.VALID_TYPE,
   713	                )
   714	                return AnyType(TypeOfAny.from_error)
   715	            if len(t.args) != 1:
   716	                self.fail(
   717	                    "Required[] must have exactly one type argument", t, code=codes.VALID_TYPE
   718	                )
   719	                return AnyType(TypeOfAny.from_error)
   720	            return RequiredType(
   721	                self.anal_type(t.args[0], allow_typed_dict_special_forms=True), required=True
   722	            )
   723	        elif fullname in ("typing_extensions.NotRequired", "typing.NotRequired"):
   724	            if not self.allow_typed_dict_special_forms:
   725	                self.fail(
   726	                    "NotRequired[] can be only used in a TypedDict definition",
   727	                    t,
   728	                    code=codes.VALID_TYPE,
   729	                )
   730	                return AnyType(TypeOfAny.from_error)
   731	            if len(t.args) != 1:
   732	                self.fail(
   733	                    "NotRequired[] must have exactly one type argument", t, code=codes.VALID_TYPE
   734	                )
   735	                return AnyType(TypeOfAny.from_error)
   736	            return RequiredType(
   737	                self.anal_type(t.args[0], allow_typed_dict_special_forms=True), required=False
   738	            )
   739	        elif fullname in ("typing_extensions.ReadOnly", "typing.ReadOnly"):
   740	            if not self.allow_typed_dict_special_forms:
   741	                self.fail(
   742	                    "ReadOnly[] can be only used in a TypedDict definition",
   743	                    t,
   744	                    code=codes.VALID_TYPE,
   745	                )
   746	                return AnyType(TypeOfAny.from_error)
   747	            if len(t.args) != 1:
   748	                self.fail(
   749	                    '"ReadOnly[]" must have exactly one type argument', t, code=codes.VALID_TYPE
   750	                )
   751	                return AnyType(TypeOfAny.from_error)
   752	            return ReadOnlyType(self.anal_type(t.args[0], allow_typed_dict_special_forms=True))
   753	        elif (
   754	            self.anal_type_guard_arg(t, fullname) is not None
   755	            or self.anal_type_is_arg(t, fullname) is not None
   756	        ):
   757	            # In most contexts, TypeGuard[...] acts as an alias for bool (ignoring its args)
   758	            return self.named_type("builtins.bool")
   759	        elif fullname in UNPACK_TYPE_NAMES:
   760	            if len(t.args) != 1:
   761	                self.fail("Unpack[...] requires exactly one type argument", t)
   762	                return AnyType(TypeOfAny.from_error)
   763	            if not self.allow_unpack:
   764	                self.fail(message_registry.INVALID_UNPACK_POSITION, t, code=codes.VALID_TYPE)
   765	                return AnyType(TypeOfAny.from_error)
   766	            self.allow_type_var_tuple = True
   767	            result = UnpackType(self.anal_type(t.args[0]), line=t.line, column=t.column)
   768	            self.allow_type_var_tuple = False
   769	            return result
   770	        elif fullname in SELF_TYPE_NAMES:
   771	            if t.args:
   772	                self.fail("Self type cannot have type arguments", t)
   773	            if self.prohibit_self_type is not None:
   774	                self.fail(f"Self type cannot be used in {self.prohibit_self_type}", t)
   775	                return AnyType(TypeOfAny.from_error)
   776	            if self.api.type is None:
   777	                self.fail("Self type is only allowed in annotations within class definition", t)
   778	                return AnyType(TypeOfAny.from_error)
   779	            if self.api.type.has_base("builtins.type"):
   780	                self.fail("Self type cannot be used in a metaclass", t)
   781	            if self.api.type.self_type is not None:
   782	                if self.api.type.is_final:
   783	                    return fill_typevars(self.api.type)
   784	                return self.api.type.self_type.copy_modified(line=t.line, column=t.column)
   785	            # TODO: verify this is unreachable and replace with an assert?
   786	            self.fail("Unexpected Self type", t)
   787	            return AnyType(TypeOfAny.from_error)
   788	        return None

   794	    def check_and_warn_deprecated(self, info: TypeInfo, ctx: Context) -> None:
   795	        """Similar logic to `TypeChecker.check_deprecated` and `TypeChecker.warn_deprecated."""
   796
   797	        if (
   798	            (deprecated := info.deprecated)
   799	            and not self.is_typeshed_stub
   800	            and not (self.api.type and (self.api.type.fullname == info.fullname))
   801	            and not any(
   802	                info.fullname == p or info.fullname.startswith(f"{p}.")
   803	                for p in self.options.deprecated_calls_exclude
   804	            )

   806	            for imp in self.cur_mod_node.imports:
   807	                if isinstance(imp, ImportFrom) and any(info.name == n[0] for n in imp.names):
   808	                    break
   809	            else:
   810	                warn = self.note if self.options.report_deprecated_as_note else self.fail
   811	                warn(deprecated, ctx, code=codes.DEPRECATED)

   813	    def analyze_type_with_type_info(
   814	        self, info: TypeInfo, args: Sequence[Type], ctx: Context, empty_tuple_index: bool
   815	    ) -> Type:
   816	        """Bind unbound type when were able to find target TypeInfo.
   817
   818	        This handles simple cases like 'int', 'modname.UserClass[str]', etc.
   819	        """
   820
   821	        self.check_and_warn_deprecated(info, ctx)
   822
   823	        if len(args) > 0 and info.fullname == "builtins.tuple":
   824	            fallback = Instance(info, [AnyType(TypeOfAny.special_form)], ctx.line)
   825	            return TupleType(self.anal_array(args, allow_unpack=True), fallback, ctx.line)
   826
   827	        # Analyze arguments and (usually) construct Instance type. The
   828	        # number of type arguments and their values are
   829	        # checked only later, since we do not always know the
   830	        # valid count at this point. Thus we may construct an
   831	        # Instance with an invalid number of type arguments.
   832	        #
   833	        # We allow ParamSpec literals based on a heuristic: it will be
   834	        # checked later anyways but the error message may be worse.
   835	        instance = Instance(
   836	            info,
   837	            self.anal_array(
   838	                args,
   839	                allow_param_spec=True,
   840	                allow_param_spec_literals=info.has_param_spec_type,
   841	                allow_unpack=True,  # Fixed length tuples can be used for non-variadic types.
   842	            ),
   843	            ctx.line,
   844	            ctx.column,
   845	        )
   846	        instance.end_line = ctx.end_line
   847	        instance.end_column = ctx.end_column
   848	        if len(info.type_vars) == 1 and info.has_param_spec_type:
   849	            instance.args = tuple(self.pack_paramspec_args(instance.args))
   850
   851	        # Check type argument count.
   852	        instance.args = tuple(flatten_nested_tuples(instance.args))
   853	        if not (self.defining_alias and self.nesting_level == 0) and not validate_instance(
   854	            instance, self.fail, empty_tuple_index
   855	        ):
   856	            fix_instance(
   857	                instance,
   858	                self.fail,
   859	                self.note,
   860	                disallow_any=self.options.disallow_any_generics and not self.is_typeshed_stub,
   861	                options=self.options,
   862	            )
   863
   864	        tup = info.tuple_type
   865	        if tup is not None:
   866	            # The class has a Tuple[...] base class so it will be
   867	            # represented as a tuple type.
   868	            if info.special_alias:
   869	                return instantiate_type_alias(
   870	                    info.special_alias,
   871	                    # TODO: should we allow NamedTuples generic in ParamSpec?
   872	                    self.anal_array(args, allow_unpack=True),
   873	                    self.fail,
   874	                    False,
   875	                    ctx,
   876	                    self.options,
   877	                    use_standard_error=True,
   878	                )
   879	            return tup.copy_modified(
   880	                items=self.anal_array(tup.items, allow_unpack=True), fallback=instance
   881	            )
   882	        td = info.typeddict_type
   883	        if td is not None:
   884	            # The class has a TypedDict[...] base class so it will be
   885	            # represented as a typeddict type.
   886	            if info.special_alias:
   887	                return instantiate_type_alias(
   888	                    info.special_alias,
   889	                    # TODO: should we allow TypedDicts generic in ParamSpec?
   890	                    self.anal_array(args, allow_unpack=True),
   891	                    self.fail,
   892	                    False,
   893	                    ctx,
   894	                    self.options,
   895	                    use_standard_error=True,
   896	                )
   897	            # Create a named TypedDictType
   898	            return td.copy_modified(
   899	                item_types=self.anal_array(list(td.items.values())), fallback=instance
   900	            )
   901
   902	        if info.fullname == "types.NoneType":
   903	            self.fail(
   904	                "NoneType should not be used as a type, please use None instead",
   905	                ctx,
   906	                code=codes.VALID_TYPE,
   907	            )
   908	            return NoneType(ctx.line, ctx.column)
   909
   910	        return instance

   912	    def analyze_unbound_type_without_type_info(
   913	        self, t: UnboundType, sym: SymbolTableNode, defining_literal: bool
   914	    ) -> Type:
   915	        """Figure out what an unbound type that doesn't refer to a TypeInfo node means.
   916
   917	        This is something unusual. We try our best to find out what it is.
   918	        """
   919	        name = sym.fullname
   920	        if name is None:
   921	            assert sym.node is not None
   922	            name = sym.node.name
   923	        # Option 1:
   924	        # Something with an Any type -- make it an alias for Any in a type
   925	        # context. This is slightly problematic as it allows using the type 'Any'
   926	        # as a base class -- however, this will fail soon at runtime so the problem
   927	        # is pretty minor.
   928	        if isinstance(sym.node, Var):
   929	            typ = get_proper_type(sym.node.type)
   930	            if isinstance(typ, AnyType):
   931	                return AnyType(
   932	                    TypeOfAny.from_unimported_type, missing_import_name=typ.missing_import_name
   933	                )
   934	            elif self.allow_type_any:
   935	                if isinstance(typ, Instance) and typ.type.fullname == "builtins.type":
   936	                    return AnyType(TypeOfAny.special_form)
   937	                if isinstance(typ, TypeType) and isinstance(typ.item, AnyType):
   938	                    return AnyType(TypeOfAny.from_another_any, source_any=typ.item)
   939	        # Option 2:
   940	        # Unbound type variable. Currently these may be still valid,
   941	        # for example when defining a generic type alias.
   942	        unbound_tvar = (
   943	            isinstance(sym.node, (TypeVarExpr, TypeVarTupleExpr))
   944	            and self.tvar_scope.get_binding(sym) is None
   945	        )
   946	        if self.allow_unbound_tvars and unbound_tvar:
   947	            return t
   948
   949	        # Option 3:
   950	        # Enum value. Note: we only want to return a LiteralType when
   951	        # we're using this enum value specifically within context of
   952	        # a "Literal[...]" type. So, if `defining_literal` is not set,
   953	        # we bail out early with an error.
   954	        #
   955	        # If, in the distant future, we decide to permit things like
   956	        # `def foo(x: Color.RED) -> None: ...`, we can remove that
   957	        # check entirely.
   958	        if (
   959	            isinstance(sym.node, Var)
   960	            and sym.node.info
   961	            and sym.node.info.is_enum
   962	            and not sym.node.name.startswith("__")
   963	        ):
   964	            value = sym.node.name
   965	            base_enum_short_name = sym.node.info.name
   966	            if not defining_literal:
   967	                msg = message_registry.INVALID_TYPE_RAW_ENUM_VALUE.format(
   968	                    base_enum_short_name, value
   969	                )
   970	                self.fail(msg.value, t, code=msg.code)
   971	                return AnyType(TypeOfAny.from_error)
   972	            return LiteralType(
   973	                value=value,
   974	                fallback=Instance(sym.node.info, [], line=t.line, column=t.column),
   975	                line=t.line,
   976	                column=t.column,
   977	            )
   978
   979	        # None of the above options worked. We parse the args (if there are any)
   980	        # to make sure there are no remaining semanal-only types, then give up.
   981	        t = t.copy_modified(args=self.anal_array(t.args))
   982	        # TODO: Move this message building logic to messages.py.
   983	        notes: list[str] = []
   984	        error_code = codes.VALID_TYPE
   985	        if isinstance(sym.node, Var):
   986	            notes.append(
   987	                "See https://mypy.readthedocs.io/en/"
   988	                "stable/common_issues.html#variables-vs-type-aliases"
   989	            )
   990	            message = 'Variable "{}" is not valid as a type'
   991	        elif isinstance(sym.node, (SYMBOL_FUNCBASE_TYPES, Decorator)):
   992	            message = 'Function "{}" is not valid as a type'
   993	            if name == "builtins.any":
   994	                notes.append('Perhaps you meant "typing.Any" instead of "any"?')
   995	            elif name == "builtins.callable":
   996	                notes.append('Perhaps you meant "typing.Callable" instead of "callable"?')
   997	            else:
   998	                notes.append('Perhaps you need "Callable[...]" or a callback protocol?')
   999	        elif isinstance(sym.node, MypyFile):
  1000	            message = 'Module "{}" is not valid as a type'
  1001	            notes.append("Perhaps you meant to use a protocol matching the module structure?")
  1002	        elif unbound_tvar:
  1003	            assert isinstance(sym.node, TypeVarLikeExpr)
  1004	            if sym.node.is_new_style:
  1005	                # PEP 695 type parameters are never considered unbound -- they are undefined
  1006	                # in contexts where they aren't valid, such as in argument default values.
  1007	                message = 'Name "{}" is not defined'
  1008	                name = name.split(".")[-1]
  1009	                error_code = codes.NAME_DEFINED
  1010	            else:
  1011	                message = 'Type variable "{}" is unbound'
  1012	                short = name.split(".")[-1]
  1013	                notes.append(
  1014	                    f'(Hint: Use "Generic[{short}]" or "Protocol[{short}]" base class'
  1015	                    f' to bind "{short}" inside a class)'
  1016	                )
  1017	                notes.append(
  1018	                    f'(Hint: Use "{short}" in function signature '
  1019	                    f'to bind "{short}" inside a function)'
  1020	                )
  1021	        else:
  1022	            message = 'Cannot interpret reference "{}" as a type'
  1023	        if not defining_literal:
  1024	            # Literal check already gives a custom error. Avoid duplicating errors.
  1025	            self.fail(message.format(name), t, code=error_code)
  1026	            for note in notes:
  1027	                self.note(note, t, code=error_code)
  1028
  1029	        # TODO: Would it be better to always return Any instead of UnboundType
  1030	        # in case of an error? On one hand, UnboundType has a name so error messages
  1031	        # are more detailed, on the other hand, some of them may be bogus,
  1032	        # see https://github.com/python/mypy/issues/4987.
  1033	        return t

  1101	    def visit_callable_type(
  1102	        self, t: CallableType, nested: bool = True, namespace: str = ""
  1103	    ) -> Type:
  1104	        # Every Callable can bind its own type variables, if they're not in the outer scope
  1105	        # TODO: attach namespace for nested free type variables (these appear in return type only).
  1106	        with self.tvar_scope_frame(namespace=namespace):
  1107	            unpacked_kwargs = t.unpack_kwargs
  1108	            if self.defining_alias:
  1109	                variables = t.variables
  1110	            else:
  1111	                variables, _ = self.bind_function_type_variables(t, t)
  1112	            type_guard = self.anal_type_guard(t.ret_type)
  1113	            type_is = self.anal_type_is(t.ret_type)
  1114
  1115	            arg_kinds = t.arg_kinds
  1116	            arg_types = []
  1117	            param_spec_with_args = param_spec_with_kwargs = None
  1118	            param_spec_invalid = False
  1119	            for kind, ut in zip(arg_kinds, t.arg_types):
  1120	                if kind == ARG_STAR:
  1121	                    param_spec_with_args, at = self.anal_star_arg_type(ut, kind, nested=nested)
  1122	                elif kind == ARG_STAR2:
  1123	                    param_spec_with_kwargs, at = self.anal_star_arg_type(ut, kind, nested=nested)
  1124	                else:
  1125	                    if param_spec_with_args:
  1126	                        param_spec_invalid = True
  1127	                        self.fail(
  1128	                            "Arguments not allowed after ParamSpec.args", t, code=codes.VALID_TYPE
  1129	                        )
  1130	                    at = self.anal_type(ut, nested=nested, allow_unpack=False)
  1131	                arg_types.append(at)
  1132
  1133	            if nested and arg_types:
  1134	                # If we've got a Callable[[Unpack[SomeTypedDict]], None], make sure
  1135	                # Unpack is interpreted as `**` and not as `*`.
  1136	                last = arg_types[-1]
  1137	                if isinstance(last, UnpackType):
  1138	                    # TODO: it would be better to avoid this get_proper_type() call.
  1139	                    p_at = get_proper_type(last.type)
  1140	                    if isinstance(p_at, TypedDictType) and not last.from_star_syntax:
  1141	                        # Automatically detect Unpack[Foo] in Callable as backwards
  1142	                        # compatible syntax for **Foo, if Foo is a TypedDict.
  1143	                        arg_kinds[-1] = ARG_STAR2
  1144	                        arg_types[-1] = p_at
  1145	                        unpacked_kwargs = True
  1146	                arg_types = self.check_unpacks_in_list(arg_types)
  1147
  1148	            if not param_spec_invalid and param_spec_with_args != param_spec_with_kwargs:
  1149	                # If already invalid, do not report more errors - definition has
  1150	                # to be fixed anyway
  1151	                name = param_spec_with_args or param_spec_with_kwargs
  1152	                self.fail(
  1153	                    f'ParamSpec must have "*args" typed as "{name}.args" and "**kwargs" typed as "{name}.kwargs"',
  1154	                    t,
  1155	                    code=codes.VALID_TYPE,
  1156	                )
  1157	                param_spec_invalid = True
  1158
  1159	            if param_spec_invalid:
  1160	                if ARG_STAR in arg_kinds:
  1161	                    arg_types[arg_kinds.index(ARG_STAR)] = AnyType(TypeOfAny.from_error)
  1162	                if ARG_STAR2 in arg_kinds:
  1163	                    arg_types[arg_kinds.index(ARG_STAR2)] = AnyType(TypeOfAny.from_error)
  1164
  1165	            # If there were multiple (invalid) unpacks, the arg types list will become shorter,
  1166	            # we need to trim the kinds/names as well to avoid crashes.
  1167	            arg_kinds = t.arg_kinds[: len(arg_types)]
  1168	            arg_names = t.arg_names[: len(arg_types)]
  1169
  1170	            ret = t.copy_modified(
  1171	                arg_types=arg_types,
  1172	                arg_kinds=arg_kinds,
  1173	                arg_names=arg_names,
  1174	                ret_type=self.anal_type(t.ret_type, nested=nested),
  1175	                # If the fallback isn't filled in yet,
  1176	                # its type will be the falsey FakeInfo
  1177	                fallback=(t.fallback if t.fallback.type else self.named_type("builtins.function")),
  1178	                variables=self.anal_var_defs(variables),
  1179	                type_guard=type_guard,
  1180	                type_is=type_is,
  1181	                unpack_kwargs=unpacked_kwargs,
  1182	            )
  1183	        return ret

  1343	    def visit_raw_expression_type(self, t: RawExpressionType) -> Type:
  1344	        # We should never see a bare Literal. We synthesize these raw literals

  1346	        # "fake literals" should always be wrapped in an UnboundType
  1347	        # corresponding to 'Literal'.
  1348	        #
  1349	        # Note: if at some point in the distant future, we decide to

  1534	    def analyze_callable_type(self, t: UnboundType) -> Type:
  1535	        fallback = self.named_type("builtins.function")
  1536	        if len(t.args) == 0:
  1537	            # Callable (bare). Treat as Callable[..., Any].
  1538	            any_type = self.get_omitted_any(t)
  1539	            ret = callable_with_ellipsis(any_type, any_type, fallback)
  1540	        elif len(t.args) == 2:
  1541	            callable_args = t.args[0]
  1542	            ret_type = t.args[1]
  1543	            if isinstance(callable_args, TypeList):
  1544	                # Callable[[ARG, ...], RET] (ordinary callable type)
  1545	                analyzed_args = self.analyze_callable_args(callable_args)
  1546	                if analyzed_args is None:
  1547	                    return AnyType(TypeOfAny.from_error)
  1548	                args, kinds, names = analyzed_args
  1549	                ret = CallableType(args, kinds, names, ret_type=ret_type, fallback=fallback)
  1550	            elif isinstance(callable_args, EllipsisType):
  1551	                # Callable[..., RET] (with literal ellipsis; accept arbitrary arguments)
  1552	                ret = callable_with_ellipsis(
  1553	                    AnyType(TypeOfAny.explicit), ret_type=ret_type, fallback=fallback
  1554	                )
  1555	            else:
  1556	                # Callable[P, RET] (where P is ParamSpec)
  1557	                with self.tvar_scope_frame(namespace=""):
  1558	                    # Temporarily bind ParamSpecs to allow code like this:
  1559	                    #     my_fun: Callable[Q, Foo[Q]]
  1560	                    # We usually do this later in visit_callable_type(), but the analysis
  1561	                    # below happens at very early stage.
