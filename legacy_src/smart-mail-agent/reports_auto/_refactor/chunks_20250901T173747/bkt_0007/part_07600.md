   131	    def data_for_file(self, fr: FileReporter, analysis: Analysis) -> FileData:
   132	        """Produce the data needed for one file's report."""
   133	        if self.has_arcs:
   134	            missing_branch_arcs = analysis.missing_branch_arcs()
   135	            arcs_executed = analysis.arcs_executed
   136	        else:
   137	            missing_branch_arcs = {}
   138	            arcs_executed = []
   139
   140	        if self.config.show_contexts:
   141	            contexts_by_lineno = self.data.contexts_by_lineno(analysis.filename)
   142
   143	        lines = []
   144	        branch_stats = analysis.branch_stats()
   145	        multiline_map = {}
   146	        if hasattr(fr, "multiline_map"):
   147	            multiline_map = fr.multiline_map()
   148
   149	        for lineno, tokens in enumerate(fr.source_token_lines(), start=1):
   150	            # Figure out how to mark this line.
   151	            category = category2 = ""
   152	            short_annotations = []
   153	            long_annotations = []
   154
   155	            if lineno in analysis.excluded:
   156	                category = "exc"
   157	            elif lineno in analysis.missing:
   158	                category = "mis"
   159	            elif self.has_arcs and lineno in missing_branch_arcs:
   160	                category = "par"
   161	                mba = missing_branch_arcs[lineno]
   162	                if len(mba) == branch_stats[lineno][0]:
   163	                    # None of the branches were taken from this line.
   164	                    short_annotations.append("anywhere")
   165	                    long_annotations.append(
   166	                        f"line {lineno} didn't jump anywhere: it always raised an exception."
   167	                    )
   168	                else:
   169	                    for b in missing_branch_arcs[lineno]:
   170	                        if b < 0:
   171	                            short_annotations.append("exit")
   172	                        else:
   173	                            short_annotations.append(str(b))
   174	                        long_annotations.append(
   175	                            fr.missing_arc_description(lineno, b, arcs_executed)
   176	                        )
   177	            elif lineno in analysis.statements:
   178	                category = "run"
   179	            elif first_line := multiline_map.get(lineno):
   180	                if first_line in analysis.excluded:
   181	                    category2 = "exc2"
   182	                elif first_line in analysis.missing:
   183	                    category2 = "mis2"
   184	                elif self.has_arcs and first_line in missing_branch_arcs:
   185	                    category2 = "par2"
   186	                # I don't understand why this last condition is marked as
   187	                # partial.  If I add an else with an exception, the exception
   188	                # is raised.
   189	                elif first_line in analysis.statements:     # pragma: part covered
   190	                    category2 = "run2"
   191
   192	            contexts = []
   193	            contexts_label = ""
   194	            context_list = []
   195	            if category and self.config.show_contexts:
   196	                contexts = human_sorted(c or self.EMPTY for c in contexts_by_lineno.get(lineno, ()))
   197	                if contexts == [self.EMPTY]:
   198	                    contexts_label = self.EMPTY
   199	                else:
   200	                    contexts_label = f"{len(contexts)} ctx"
   201	                    context_list = contexts
   202
   203	            lines.append(LineData(
   204	                tokens=tokens,
   205	                number=lineno,
   206	                category=category or category2,
   207	                contexts=contexts,
   208	                contexts_label=contexts_label,
   209	                context_list=context_list,
   210	                short_annotations=short_annotations,
   211	                long_annotations=long_annotations,
   212	            ))
   213
   214	        file_data = FileData(
   215	            relative_filename=fr.relative_filename(),
   216	            nums=analysis.numbers,
   217	            lines=lines,
   218	        )
   219
   220	        return file_data

   233	HTML_SAFE = string.ascii_letters + string.digits + "!#$%'()*+,-./:;=?@[]^_`{|}~"

   235	@functools.cache
   236	def encode_int(n: int) -> str:
   237	    """Create a short HTML-safe string from an integer, using HTML_SAFE."""
   238	    if n == 0:
   239	        return HTML_SAFE[0]
   240
   241	    r = []
   242	    while n:
   243	        n, t = divmod(n, len(HTML_SAFE))
   244	        r.append(HTML_SAFE[t])
   245	    return "".join(r)

   248	def copy_with_cache_bust(src: str, dest_dir: str) -> str:
   249	    """Copy `src` to `dest_dir`, adding a hash to the name.
   250
   251	    Returns the updated destination file name with hash.
   252	    """
   253	    with open(src, "rb") as f:
   254	        text = f.read()
   255	    h = Hasher()
   256	    h.update(text)
   257	    cache_bust = h.hexdigest()[:8]
   258	    src_base = os.path.basename(src)
   259	    dest = src_base.replace(".", f"_cb_{cache_bust}.")
   260	    with open(os.path.join(dest_dir, dest), "wb") as f:
   261	        f.write(text)
   262	    return dest

   351	        """Generate an HTML report for `morfs`.
   352
   353	        `morfs` is a list of modules or file names.
   354
   355	        """
   356	        # Read the status data and check that this run used the same
   357	        # global data as the last run.
   358	        self.incr.read()
   359	        self.incr.check_global_data(self.config, self.pyfile_html_source)
   360
   361	        # Process all the files. For each page we need to supply a link
   362	        # to the next and previous page.
   363	        files_to_report = []
   364
   365	        have_data = False
   366	        for fr, analysis in get_analysis_to_report(self.coverage, morfs):
   367	            have_data = True

   369	            if self.should_report(analysis, self.index_pages["file"]):
   370	                files_to_report.append(ftr)

   374	        if not have_data:
   375	            raise NoDataError("No data to report.")
   376
   377	        self.make_directory()
   378	        self.make_local_static_report_files()
   379
   380	        if files_to_report:
   381	            for ftr1, ftr2 in zip(files_to_report[:-1], files_to_report[1:]):
   382	                ftr1.next_html = ftr2.html_filename
   383	                ftr2.prev_html = ftr1.html_filename
   384	            files_to_report[0].prev_html = "index.html"

   390	                if noun not in self.index_pages:

   397	        else:
   398	            first_html = final_html = "index.html"

   401	        # Write function and class index pages.
   402	        self.write_region_index_pages(files_to_report)

   441	    def should_report(self, analysis: Analysis, index_page: IndexPage) -> bool:
   442	        """Determine if we'll report this file or region."""
   443	        # Get the numbers for this file.
   444	        nums = analysis.numbers
   445	        index_page.totals += nums
   446
   447	        if self.skip_covered:
   448	            # Don't report on 100% files.
   449	            no_missing_lines = (nums.n_missing == 0)
   450	            no_missing_branches = (nums.n_partial_branches == 0)
   451	            if no_missing_lines and no_missing_branches:
   452	                index_page.skipped_covered_count += 1
   453	                return False
   454
   455	        if self.skip_empty:
   456	            # Don't report on empty files.
   457	            if nums.n_statements == 0:
   458	                index_page.skipped_empty_count += 1
   459	                return False
   460
   461	        return True

   463	    def write_html_page(self, ftr: FileToReport) -> None:
   464	        """Generate an HTML page for one source file.
   465
   466	        If the page on disk is already correct based on our incremental status
   467	        checking, then the page doesn't have to be generated, and this function
   468	        only does page summary bookkeeping.
   469
   470	        """
   471	        # Find out if the page on disk is already correct.
   472	        if self.incr.can_skip_file(self.data, ftr.fr, ftr.rootname):
   473	            self.index_pages["file"].summaries.append(self.incr.index_info(ftr.rootname))
   474	            return
   475
   476	        # Write the HTML page for this source file.
   477	        file_data = self.datagen.data_for_file(ftr.fr, ftr.analysis)
   478
   479	        contexts = collections.Counter(c for cline in file_data.lines for c in cline.contexts)
   480	        context_codes = {y: i for (i, y) in enumerate(x[0] for x in contexts.most_common())}
   481	        if context_codes:
   482	            contexts_json = json.dumps(
   483	                {encode_int(v): k for (k, v) in context_codes.items()},
   484	                indent=2,
   485	            )
   486	        else:
   487	            contexts_json = None
   488
   489	        for ldata in file_data.lines:
   490	            # Build the HTML for the line.
   491	            html_parts = []
   492	            for tok_type, tok_text in ldata.tokens:
   493	                if tok_type == "ws":
   494	                    html_parts.append(escape(tok_text))
   495	                else:
   496	                    tok_html = escape(tok_text) or "&nbsp;"
   497	                    html_parts.append(f'<span class="{tok_type}">{tok_html}</span>')
   498	            ldata.html = "".join(html_parts)
   499	            if ldata.context_list:
   500	                encoded_contexts = [
   501	                    encode_int(context_codes[c_context]) for c_context in ldata.context_list
   502	                ]
   503	                code_width = max(len(ec) for ec in encoded_contexts)
   504	                ldata.context_str = (
   505	                    str(code_width)
   506	                    + "".join(ec.ljust(code_width) for ec in encoded_contexts)
   507	                )
   508	            else:
   509	                ldata.context_str = ""
   510
   511	            if ldata.short_annotations:
   512	                # 202F is NARROW NO-BREAK SPACE.
   513	                # 219B is RIGHTWARDS ARROW WITH STROKE.
   514	                ldata.annotate = ",&nbsp;&nbsp; ".join(
   515	                    f"{ldata.number}&#x202F;&#x219B;&#x202F;{d}"
   516	                    for d in ldata.short_annotations
   517	                )
   518	            else:
   519	                ldata.annotate = None
   520
   521	            if ldata.long_annotations:
   522	                longs = ldata.long_annotations
   523	                # A line can only have two branch destinations. If there were
   524	                # two missing, we would have written one as "always raised."
   525	                assert len(longs) == 1, (
   526	                    f"Had long annotations in {ftr.fr.relative_filename()}: {longs}"
   527	                )
   528	                ldata.annotate_long = longs[0]
   529	            else:
   530	                ldata.annotate_long = None
   531
   532	            css_classes = []
   533	            if ldata.category:
   534	                css_classes.append(
   535	                    self.template_globals["category"][ldata.category],   # type: ignore[index]
   536	                )
   537	            ldata.css_class = " ".join(css_classes) or "pln"
   538
   539	        html_path = os.path.join(self.directory, ftr.html_filename)
   540	        html = self.source_tmpl.render({
   541	            **file_data.__dict__,
   542	            "contexts_json": contexts_json,
   543	            "prev_html": ftr.prev_html,
   544	            "next_html": ftr.next_html,
   545	        })
   546	        write_html(html_path, html)
   547
   548	        # Save this file's information for the index page.
   549	        index_info = IndexItem(
   550	            url = ftr.html_filename,
   551	            file = escape(ftr.fr.relative_filename()),
   552	            nums = ftr.analysis.numbers,
   553	        )
   554	        self.index_pages["file"].summaries.append(index_info)
   555	        self.incr.set_index_info(ftr.rootname, index_info)

   571	    def write_region_index_pages(self, files_to_report: Iterable[FileToReport]) -> None:

   597	                        ),
   598	                        nums=analysis.numbers,
   599	                    ))

   601	                analysis = ftr.analysis.narrow(outside_lines)
   602	                if self.should_report(analysis, page_data):
   603	                    page_data.summaries.append(IndexItem(
   604	                        url=ftr.html_filename,

   607	                            "<data value=''>"

   610	                        ),
   611	                        nums=analysis.numbers,

   618	    def write_index_page(self, index_page: IndexPage, **kwargs: str) -> str:
   619	        """Write an index page specified by `index_page`.
   620
   621	        Returns the filename created.
   622	        """
   623	        skipped_covered_msg = skipped_empty_msg = ""
   624	        if n := index_page.skipped_covered_count:
   625	            word = plural(n, index_page.noun, index_page.plural)
   626	            skipped_covered_msg = f"{n} {word} skipped due to complete coverage."
   627	        if n := index_page.skipped_empty_count:
   628	            word = plural(n, index_page.noun, index_page.plural)
   629	            skipped_empty_msg = f"{n} empty {word} skipped."
   630
   631	        index_buttons = [
   632	            {
   633	                "label": ip.plural.title(),
   634	                "url": ip.filename if ip.noun != index_page.noun else "",
   635	                "current": ip.noun == index_page.noun,

   667	    """Logic and data to support incremental reporting.
   668
   669	    When generating an HTML report, often only a few of the source files have
   670	    changed since the last time we made the HTML report.  This means previously
   671	    created HTML pages can be reused without generating them again, speeding
   672	    the command.
   673
   674	    This class manages a JSON data file that captures enough information to
   675	    know whether an HTML page for a .py file needs to be regenerated or not.
   676	    The data file also needs to store all the information needed to create the
   677	    entry for the file on the index page so that if the HTML page is reused,
   678	    the index page can still be created to refer to it.
   679
   680	    The data looks like::
   681
   682	        {
   683	            "note": "This file is an internal implementation detail ...",
   684	            // A fixed number indicating the data format.  STATUS_FORMAT
   685	            "format": 5,
   686	            // The version of coverage.py
   687	            "version": "7.4.4",
   688	            // A hash of a number of global things, including the configuration
   689	            // settings and the pyfile.html template itself.
   690	            "globals": "540ee119c15d52a68a53fe6f0897346d",
   691	            "files": {
   692	                // An entry for each source file keyed by the flat_rootname().
   693	                "z_7b071bdc2a35fa80___init___py": {
   694	                    // Hash of the source, the text of the .py file.
   695	                    "hash": "e45581a5b48f879f301c0f30bf77a50c",
   696	                    // Information for the index.html file.
   697	                    "index": {
   698	                        "url": "z_7b071bdc2a35fa80___init___py.html",
   699	                        "file": "cogapp/__init__.py",
   700	                        "description": "",
   701	                        // The Numbers for this file.
   702	                        "nums": { "precision": 2, "n_files": 1, "n_statements": 43, ... }
   703	                    }
   704	                },
   705	                ...
   706	            }
   707	        }

   709	    """

   716	        + " for the JSON report: https://coverage.rtfd.io/cmd.html#cmd-json"

   736	            usable = False

   763	        status_data = {
   764	            "note": self.NOTE,
   765	            "format": self.STATUS_FORMAT,

   772	        }

   783	        """
   784	        m = Hasher()

   342	                        # exec'ing code.  If it seems like something is wrong with
   343	                        # the frame's file name, then just use the original.

    23	def line_hash(line: str) -> str:
    24	    """Produce a hash of a source line for use in the LCOV file."""
    25	    # The LCOV file format optionally allows each line to be MD5ed as a
    26	    # fingerprint of the file.  This is not a security use.  Some security
    27	    # scanners raise alarms about the use of MD5 here, but it is a false
    28	    # positive.  This is not a security concern.
    29	    # The unusual encoding of the MD5 hash, as a base64 sequence with the
    30	    # trailing = signs stripped, is specified by the LCOV file format.
    31	    hashed = hashlib.md5(line.encode("utf-8"), usedforsecurity=False).digest()
    32	    return base64.b64encode(hashed).decode("ascii").rstrip("=")

   165	        """Add `v` to the hash, recursively if needed."""
   166	        self.hash.update(str(type(v)).encode("utf-8"))
   167	        if isinstance(v, str):

   177	                self.update(e)

   179	            keys = v.keys()
   180	            for k in sorted(keys):

   188	                if inspect.isroutine(a):

   228	def substitute_variables(text: str, variables: Mapping[str, str]) -> str:
   229	    """Substitute ``${VAR}`` variables in `text` with their values.
   230
   231	    Variables in the text can take a number of shell-inspired forms::
   232
   233	        $VAR
   234	        ${VAR}
   235	        ${VAR?}             strict: an error if VAR isn't defined.
   236	        ${VAR-missing}      defaulted: "missing" if VAR isn't defined.
   237	        $$                  just a dollar sign.
   238
   239	    `variables` is a dictionary of variable values.
   240
   241	    Returns the resulting text with values substituted.
   242
   243	    """
   244	    dollar_pattern = r"""(?x)   # Use extended regex syntax
   245	        \$                      # A dollar sign,
   246	        (?:                     # then
   247	            (?P<dollar> \$ ) |      # a dollar sign, or
   248	            (?P<word1> \w+ ) |      # a plain word, or
   249	            \{                      # a {-wrapped
   250	                (?P<word2> \w+ )        # word,
   251	                (?:                         # either
   252	                    (?P<strict> \? ) |      # with a strict marker
   253	                    -(?P<defval> [^}]* )    # or a default value
   254	                )?                      # maybe.
   255	            }
   256	        )
   257	        """
   258
   259	    dollar_groups = ("dollar", "word1", "word2")
   260
   261	    def dollar_replace(match: re.Match[str]) -> str:
   262	        """Called for each $replacement."""
   263	        # Only one of the dollar_groups will have matched, just get its text.
   264	        word = next(g for g in match.group(*dollar_groups) if g)    # pragma: always breaks
   265	        if word == "$":
   266	            return "$"
   267	        elif word in variables:
   268	            return variables[word]
   269	        elif match["strict"]:
   270	            msg = f"Variable {word} is undefined: {text!r}"
   271	            raise CoverageException(msg)
   272	        else:
   273	            return match["defval"]
   274
   275	    text = re.sub(dollar_pattern, dollar_replace, text)
   276	    return text

    32	        A binary blob.
    33	    """
    34	    try:
    35	        nbytes = max(nums) // 8 + 1
    36	    except ValueError:
    37	        # nums was empty.
    38	        return b""
    39	    b = bytearray(nbytes)
    40	    for num in nums:
    41	        b[num//8] |= 1 << num % 8

    45	def numbits_to_nums(numbits: bytes) -> list[int]:
    46	    """Convert a numbits into a list of numbers.
    47
    48	    Arguments:
    49	        numbits: a binary blob, the packed number set.
    50
    51	    Returns:
    52	        A list of ints.
    53
    54	    When registered as a SQLite function by :func:`register_sqlite_functions`,
    55	    this returns a string, a JSON-encoded list of ints.
    56
    57	    """
    58	    nums = []
    59	    for byte_i, byte in enumerate(numbits):
    60	        for bit_i in range(8):
    61	            if (byte & (1 << bit_i)):
    62	                nums.append(byte_i * 8 + bit_i)
    63	    return nums

    66	def numbits_union(numbits1: bytes, numbits2: bytes) -> bytes:
    67	    """Compute the union of two numbits.
    68
    69	    Returns:
    70	        A new numbits, the union of `numbits1` and `numbits2`.
    71	    """
    72	    byte_pairs = zip_longest(numbits1, numbits2, fillvalue=0)
    73	    return bytes(b1 | b2 for b1, b2 in byte_pairs)

    77	    """Compute the intersection of two numbits.
    78
    79	    Returns:
    80	        A new numbits, the intersection `numbits1` and `numbits2`.
    81	    """
    82	    byte_pairs = zip_longest(numbits1, numbits2, fillvalue=0)
