   200	def parse(
   201	    source: str | bytes,
   202	    fnam: str,
   203	    module: str | None,
   204	    errors: Errors,
   205	    options: Options | None = None,
   206	) -> MypyFile:
   207	    """Parse a source file, without doing any semantic analysis.
   208
   209	    Return the parse tree. If errors is not provided, raise ParseError
   210	    on failure. Otherwise, use the errors object to report parse errors.
   211	    """
   212	    ignore_errors = (options is not None and options.ignore_errors) or (
   213	        fnam in errors.ignored_files
   214	    )
   215	    # If errors are ignored, we can drop many function bodies to speed up type checking.
   216	    strip_function_bodies = ignore_errors and (options is None or not options.preserve_asts)
   217
   218	    if options is None:
   219	        options = Options()
   220	    errors.set_file(fnam, module, options=options)
   221	    is_stub_file = fnam.endswith(".pyi")
   222	    if is_stub_file:
   223	        feature_version = defaults.PYTHON3_VERSION[1]
   224	        if options.python_version[0] == 3 and options.python_version[1] > feature_version:
   225	            feature_version = options.python_version[1]
   226	    else:
   227	        assert options.python_version[0] >= 3
   228	        feature_version = options.python_version[1]
   229	    try:
   230	        # Disable deprecation warnings about \u
   231	        with warnings.catch_warnings():
   232	            warnings.filterwarnings("ignore", category=DeprecationWarning)
   233	            ast = ast3_parse(source, fnam, "exec", feature_version=feature_version)
   234
   235	        tree = ASTConverter(
   236	            options=options,
   237	            is_stub=is_stub_file,
   238	            errors=errors,
   239	            strip_function_bodies=strip_function_bodies,
   240	            path=fnam,
   241	        ).visit(ast)
   242
   243	    except RecursionError as e:
   244	        # For very complex expressions it is possible to hit recursion limit
   245	        # before reaching a leaf node.
   246	        # Should reject at top level instead at bottom, since bottom would already
   247	        # be at the threshold of the recursion limit, and may fail again later.
   248	        # E.G. x1+x2+x3+...+xn -> BinOp(left=BinOp(left=BinOp(left=...
   249	        try:
   250	            # But to prove that is the cause of this particular recursion error,
   251	            # try to walk the tree using builtin visitor
   252	            ast3.NodeVisitor().visit(ast)
   253	        except RecursionError:
   254	            errors.report(
   255	                -1, -1, "Source expression too complex to parse", blocker=False, code=codes.MISC
   256	            )
   257
   258	            tree = MypyFile([], [], False, {})
   259
   260	        else:
   261	            # re-raise original recursion error if it *can* be unparsed,
   262	            # maybe this is some other issue that shouldn't be silenced/misdirected
   263	            raise e
   264
   265	    except SyntaxError as e:
   266	        message = e.msg
   267	        if feature_version > sys.version_info.minor and message.startswith("invalid syntax"):
   268	            python_version_str = f"{options.python_version[0]}.{options.python_version[1]}"
   269	            message += f"; you likely need to run mypy using Python {python_version_str} or newer"
   270	        errors.report(
   271	            e.lineno if e.lineno is not None else -1,
   272	            e.offset,
   273	            re.sub(
   274	                r"^(\s*\w)", lambda m: m.group(1).upper(), message
   275	            ),  # Standardizing error message
   276	            blocker=True,
   277	            code=codes.SYNTAX,
   278	        )
   279	        tree = MypyFile([], [], False, {})
   280
   281	    assert isinstance(tree, MypyFile)
   282	    return tree

   285	def parse_type_ignore_tag(tag: str | None) -> list[str] | None:
   286	    """Parse optional "[code, ...]" tag after "# type: ignore".
   287
   288	    Return:
   289	     * [] if no tag was found (ignore all errors)
   290	     * list of ignored error codes if a tag was found
   291	     * None if the tag was invalid.
   292	    """
   293	    if not tag or tag.strip() == "" or tag.strip().startswith("#"):
   294	        # No tag -- ignore all errors.
   295	        return []
   296	    m = re.match(r"\s*\[([^]#]*)\]\s*(#.*)?$", tag)
   297	    if m is None:
   298	        # Invalid "# type: ignore" comment.
   299	        return None
   300	    return [code.strip() for code in m.group(1).split(",")]

   303	def parse_type_comment(
   304	    type_comment: str, line: int, column: int, errors: Errors | None
   305	) -> tuple[list[str] | None, ProperType | None]:
   306	    """Parse type portion of a type comment (+ optional type ignore).
   307
   308	    Return (ignore info, parsed type).
   309	    """
   310	    try:
   311	        typ = ast3_parse(type_comment, "<type_comment>", "eval")
   312	    except SyntaxError:
   313	        if errors is not None:
   314	            stripped_type = type_comment.split("#", 2)[0].strip()
   315	            err_msg = message_registry.TYPE_COMMENT_SYNTAX_ERROR_VALUE.format(stripped_type)
   316	            errors.report(line, column, err_msg.value, blocker=True, code=err_msg.code)
   317	            return None, None
   318	        else:
   319	            raise
   320	    else:
   321	        extra_ignore = TYPE_IGNORE_PATTERN.match(type_comment)
   322	        if extra_ignore:
   323	            tag: str | None = extra_ignore.group(1)
   324	            ignored: list[str] | None = parse_type_ignore_tag(tag)
   325	            if ignored is None:
   326	                if errors is not None:
   327	                    errors.report(
   328	                        line, column, message_registry.INVALID_TYPE_IGNORE.value, code=codes.SYNTAX
   329	                    )
   330	                else:
   331	                    raise SyntaxError
   332	        else:
   333	            ignored = None
   334	        assert isinstance(typ, ast3.Expression)
   335	        converted = TypeConverter(
   336	            errors, line=line, override_column=column, is_evaluated=False
   337	        ).visit(typ.body)
   338	        return ignored, converted

   341	def parse_type_string(
   342	    expr_string: str, expr_fallback_name: str, line: int, column: int
   343	) -> ProperType:
   344	    """Parses a type that was originally present inside of an explicit string.
   345
   346	    For example, suppose we have the type `Foo["blah"]`. We should parse the
   347	    string expression "blah" using this function.
   348	    """
   349	    try:
   350	        _, node = parse_type_comment(f"({expr_string})", line=line, column=column, errors=None)
   351	        if isinstance(node, (UnboundType, UnionType)) and node.original_str_expr is None:
   352	            node.original_str_expr = expr_string
   353	            node.original_str_fallback = expr_fallback_name
   354	            return node
   355	        else:
   356	            return RawExpressionType(expr_string, expr_fallback_name, line, column)
   357	    except (SyntaxError, ValueError):
   358	        # Note: the parser will raise a `ValueError` instead of a SyntaxError if
   359	        # the string happens to contain things like \x00.
   360	        return RawExpressionType(expr_string, expr_fallback_name, line, column)

   382	    def __init__(
   383	        self,
   384	        options: Options,
   385	        is_stub: bool,
   386	        errors: Errors,
   387	        *,
   388	        strip_function_bodies: bool,

   391	        # 'C' for class, 'D' for function signature, 'F' for function, 'L' for lambda

   426	        if node is None:
   427	            return None

   456	        if (
   457	            isinstance(node, (ast3.AsyncFunctionDef, ast3.ClassDef, ast3.FunctionDef))
   458	            and node.decorator_list

   463	    def translate_stmt_list(
   464	        self,
   465	        stmts: Sequence[ast3.stmt],
   466	        *,
   467	        ismodule: bool = False,
   468	        can_strip: bool = False,
   469	        is_coroutine: bool = False,
   470	    ) -> list[Statement]:
   471	        # A "# type: ignore" comment before the first statement of a module
   472	        # ignores the whole module:
   473	        if (
   474	            ismodule
   475	            and stmts
   476	            and self.type_ignores
   477	            and min(self.type_ignores) < self.get_lineno(stmts[0])
   478	        ):
   479	            ignores = self.type_ignores[min(self.type_ignores)]
   480	            if ignores:
   481	                joined_ignores = ", ".join(ignores)
   482	                self.fail(
   483	                    message_registry.TYPE_IGNORE_WITH_ERRCODE_ON_MODULE.format(joined_ignores),
   484	                    line=min(self.type_ignores),
   485	                    column=0,
   486	                    blocker=False,
   487	                )
   488	            self.errors.used_ignored_lines[self.errors.file][min(self.type_ignores)].append(
   489	                codes.FILE.code
   490	            )
   491	            block = Block(self.fix_function_overloads(self.translate_stmt_list(stmts)))
   492	            self.set_block_lines(block, stmts)
   493	            mark_block_unreachable(block)
   494	            return [block]
   495
   496	        stack = self.class_and_function_stack
   497	        # Fast case for stripping function bodies
   498	        if (
   499	            can_strip
   500	            and self.strip_function_bodies
   501	            and len(stack) == 1
   502	            and stack[0] == "F"
   503	            and not is_coroutine
   504	        ):
   505	            return []
   506
   507	        res: list[Statement] = []
   508	        for stmt in stmts:
   509	            node = self.visit(stmt)
   510	            res.append(node)
   511
   512	        # Slow case for stripping function bodies
   513	        if can_strip and self.strip_function_bodies:
   514	            if stack[-2:] == ["C", "F"]:
   515	                if is_possible_trivial_body(res):
   516	                    can_strip = False
   517	                else:
   518	                    # We only strip method bodies if they don't assign to an attribute, as
   519	                    # this may define an attribute which has an externally visible effect.
   520	                    visitor = FindAttributeAssign()
   521	                    for s in res:
   522	                        s.accept(visitor)
   523	                        if visitor.found:
   524	                            can_strip = False
   525	                            break
   526
   527	            if can_strip and stack[-1] == "F" and is_coroutine:
   528	                # Yields inside an async function affect the return type and should not
   529	                # be stripped.
   530	                yield_visitor = FindYield()
   531	                for s in res:
   532	                    s.accept(yield_visitor)
   533	                    if yield_visitor.found:
   534	                        can_strip = False
   535	                        break
   536
   537	            if can_strip:
   538	                return []
   539	        return res

   630	    def fix_function_overloads(self, stmts: list[Statement]) -> list[Statement]:
   631	        ret: list[Statement] = []
   632	        current_overload: list[OverloadPart] = []
   633	        current_overload_name: str | None = None
   634	        seen_unconditional_func_def = False
   635	        last_if_stmt: IfStmt | None = None
   636	        last_if_overload: Decorator | FuncDef | OverloadedFuncDef | None = None
   637	        last_if_stmt_overload_name: str | None = None
   638	        last_if_unknown_truth_value: IfStmt | None = None
   639	        skipped_if_stmts: list[IfStmt] = []
   640	        for stmt in stmts:
   641	            if_overload_name: str | None = None
   642	            if_block_with_overload: Block | None = None
   643	            if_unknown_truth_value: IfStmt | None = None
   644	            if isinstance(stmt, IfStmt) and seen_unconditional_func_def is False:
   645	                # Check IfStmt block to determine if function overloads can be merged
   646	                if_overload_name = self._check_ifstmt_for_overloads(stmt, current_overload_name)
   647	                if if_overload_name is not None:
   648	                    (if_block_with_overload, if_unknown_truth_value) = (
   649	                        self._get_executable_if_block_with_overloads(stmt)
   650	                    )
   651
   652	            if (
   653	                current_overload_name is not None
   654	                and isinstance(stmt, (Decorator, FuncDef))
   655	                and stmt.name == current_overload_name
   656	            ):
   657	                if last_if_stmt is not None:
   658	                    skipped_if_stmts.append(last_if_stmt)
   659	                if last_if_overload is not None:
   660	                    # Last stmt was an IfStmt with same overload name
   661	                    # Add overloads to current_overload
   662	                    if isinstance(last_if_overload, OverloadedFuncDef):
   663	                        current_overload.extend(last_if_overload.items)
   664	                    else:
   665	                        current_overload.append(last_if_overload)
   666	                    last_if_stmt, last_if_overload = None, None
   667	                if last_if_unknown_truth_value:
   668	                    self.fail_merge_overload(last_if_unknown_truth_value)
   669	                    last_if_unknown_truth_value = None
   670	                current_overload.append(stmt)
   671	                if isinstance(stmt, FuncDef):
   672	                    seen_unconditional_func_def = True
   673	            elif (
   674	                current_overload_name is not None
   675	                and isinstance(stmt, IfStmt)
   676	                and if_overload_name == current_overload_name
   677	            ):
   678	                # IfStmt only contains stmts relevant to current_overload.
   679	                # Check if stmts are reachable and add them to current_overload,
   680	                # otherwise skip IfStmt to allow subsequent overload
   681	                # or function definitions.
   682	                skipped_if_stmts.append(stmt)
   683	                if if_block_with_overload is None:
   684	                    if if_unknown_truth_value is not None:
   685	                        self.fail_merge_overload(if_unknown_truth_value)
   686	                    continue
   687	                if last_if_overload is not None:
   688	                    # Last stmt was an IfStmt with same overload name
   689	                    # Add overloads to current_overload
   690	                    if isinstance(last_if_overload, OverloadedFuncDef):
   691	                        current_overload.extend(last_if_overload.items)
   692	                    else:
   693	                        current_overload.append(last_if_overload)
   694	                    last_if_stmt, last_if_overload = None, None
   695	                if isinstance(if_block_with_overload.body[-1], OverloadedFuncDef):
   696	                    skipped_if_stmts.extend(cast(list[IfStmt], if_block_with_overload.body[:-1]))
   697	                    current_overload.extend(if_block_with_overload.body[-1].items)
   698	                else:
   699	                    current_overload.append(
   700	                        cast(Union[Decorator, FuncDef], if_block_with_overload.body[0])
   701	                    )
   702	            else:
   703	                if last_if_stmt is not None:
   704	                    ret.append(last_if_stmt)
   705	                    last_if_stmt_overload_name = current_overload_name
   706	                    last_if_stmt, last_if_overload = None, None
   707	                    last_if_unknown_truth_value = None
   708
   709	                if current_overload and current_overload_name == last_if_stmt_overload_name:
   710	                    # Remove last stmt (IfStmt) from ret if the overload names matched
   711	                    # Only happens if no executable block had been found in IfStmt
   712	                    popped = ret.pop()
   713	                    assert isinstance(popped, IfStmt)
   714	                    skipped_if_stmts.append(popped)
   715	                if current_overload and skipped_if_stmts:
   716	                    # Add bare IfStmt (without overloads) to ret
   717	                    # Required for mypy to be able to still check conditions
   718	                    for if_stmt in skipped_if_stmts:
   719	                        self._strip_contents_from_if_stmt(if_stmt)
   720	                        ret.append(if_stmt)
   721	                    skipped_if_stmts = []
   722	                if len(current_overload) == 1:
   723	                    ret.append(current_overload[0])
   724	                elif len(current_overload) > 1:
   725	                    ret.append(OverloadedFuncDef(current_overload))
   726
   727	                # If we have multiple decorated functions named "_" next to each, we want to treat
   728	                # them as a series of regular FuncDefs instead of one OverloadedFuncDef because
   729	                # most of mypy/mypyc assumes that all the functions in an OverloadedFuncDef are
   730	                # related, but multiple underscore functions next to each other aren't necessarily
   731	                # related
   732	                seen_unconditional_func_def = False
   733	                if isinstance(stmt, Decorator) and not unnamed_function(stmt.name):
   734	                    current_overload = [stmt]
   735	                    current_overload_name = stmt.name
   736	                elif isinstance(stmt, IfStmt) and if_overload_name is not None:
   737	                    current_overload = []
   738	                    current_overload_name = if_overload_name
   739	                    last_if_stmt = stmt
   740	                    last_if_stmt_overload_name = None
   741	                    if if_block_with_overload is not None:
   742	                        skipped_if_stmts.extend(
   743	                            cast(list[IfStmt], if_block_with_overload.body[:-1])
   744	                        )
   745	                        last_if_overload = cast(
   746	                            Union[Decorator, FuncDef, OverloadedFuncDef],
   747	                            if_block_with_overload.body[-1],
   748	                        )
   749	                    last_if_unknown_truth_value = if_unknown_truth_value
   750	                else:
   751	                    current_overload = []
   752	                    current_overload_name = None
   753	                    ret.append(stmt)
   754
   755	        if current_overload and skipped_if_stmts:
   756	            # Add bare IfStmt (without overloads) to ret
   757	            # Required for mypy to be able to still check conditions
   758	            for if_stmt in skipped_if_stmts:
   759	                self._strip_contents_from_if_stmt(if_stmt)
   760	                ret.append(if_stmt)
   761	        if len(current_overload) == 1:
   762	            ret.append(current_overload[0])
   763	        elif len(current_overload) > 1:
   764	            ret.append(OverloadedFuncDef(current_overload))
   765	        elif last_if_overload is not None:
   766	            ret.append(last_if_overload)
   767	        elif last_if_stmt is not None:
   768	            ret.append(last_if_stmt)
   769	        return ret

   771	    def _check_ifstmt_for_overloads(
   772	        self, stmt: IfStmt, current_overload_name: str | None = None
   773	    ) -> str | None:
   774	        """Check if IfStmt contains only overloads with the same name.
   775	        Return overload_name if found, None otherwise.
   776	        """
   777	        # Check that block only contains a single Decorator, FuncDef, or OverloadedFuncDef.
   778	        # Multiple overloads have already been merged as OverloadedFuncDef.
   779	        if not (
   780	            len(stmt.body[0].body) == 1
   781	            and (
   782	                isinstance(stmt.body[0].body[0], (Decorator, OverloadedFuncDef))
   783	                or current_overload_name is not None
   784	                and isinstance(stmt.body[0].body[0], FuncDef)
   785	            )
   786	            or len(stmt.body[0].body) > 1
   787	            and isinstance(stmt.body[0].body[-1], OverloadedFuncDef)
   788	            and all(self._is_stripped_if_stmt(if_stmt) for if_stmt in stmt.body[0].body[:-1])
   789	        ):
   790	            return None
   791
   792	        overload_name = cast(
   793	            Union[Decorator, FuncDef, OverloadedFuncDef], stmt.body[0].body[-1]
   794	        ).name
   795	        if stmt.else_body is None:
   796	            return overload_name
   797
   798	        if len(stmt.else_body.body) == 1:
   799	            # For elif: else_body contains an IfStmt itself -> do a recursive check.
   800	            if (
   801	                isinstance(stmt.else_body.body[0], (Decorator, FuncDef, OverloadedFuncDef))
   802	                and stmt.else_body.body[0].name == overload_name
   803	            ):
   804	                return overload_name
   805	            if (
   806	                isinstance(stmt.else_body.body[0], IfStmt)
   807	                and self._check_ifstmt_for_overloads(stmt.else_body.body[0], current_overload_name)
   808	                == overload_name
   809	            ):
   810	                return overload_name
   811
   812	        return None

   814	    def _get_executable_if_block_with_overloads(
   815	        self, stmt: IfStmt
   816	    ) -> tuple[Block | None, IfStmt | None]:
   817	        """Return block from IfStmt that will get executed.
   818
   819	        Return
   820	            0 -> A block if sure that alternative blocks are unreachable.
   821	            1 -> An IfStmt if the reachability of it can't be inferred,
   822	                 i.e. the truth value is unknown.
   823	        """
   824	        infer_reachability_of_if_statement(stmt, self.options)
   825	        if stmt.else_body is None and stmt.body[0].is_unreachable is True:
   826	            # always False condition with no else
   827	            return None, None
   828	        if (
   829	            stmt.else_body is None
   830	            or stmt.body[0].is_unreachable is False
   831	            and stmt.else_body.is_unreachable is False
   832	        ):
   833	            # The truth value is unknown, thus not conclusive
   834	            return None, stmt
   835	        if stmt.else_body.is_unreachable is True:
   836	            # else_body will be set unreachable if condition is always True
   837	            return stmt.body[0], None
   838	        if stmt.body[0].is_unreachable is True:
   839	            # body will be set unreachable if condition is always False
   840	            # else_body can contain an IfStmt itself (for elif) -> do a recursive check
   841	            if isinstance(stmt.else_body.body[0], IfStmt):
   842	                return self._get_executable_if_block_with_overloads(stmt.else_body.body[0])
   843	            return stmt.else_body, None
   844	        return None, stmt

   846	    def _strip_contents_from_if_stmt(self, stmt: IfStmt) -> None:
   847	        """Remove contents from IfStmt.
   848
   849	        Needed to still be able to check the conditions after the contents
   850	        have been merged with the surrounding function overloads.
   851	        """
   852	        if len(stmt.body) == 1:
   853	            stmt.body[0].body = []
   854	        if stmt.else_body and len(stmt.else_body.body) == 1:
   855	            if isinstance(stmt.else_body.body[0], IfStmt):
   856	                self._strip_contents_from_if_stmt(stmt.else_body.body[0])
   857	            else:
   858	                stmt.else_body.body = []

   860	    def _is_stripped_if_stmt(self, stmt: Statement) -> bool:
   861	        """Check stmt to make sure it is a stripped IfStmt.
   862
   863	        See also: _strip_contents_from_if_stmt
   864	        """
   865	        if not isinstance(stmt, IfStmt):
   866	            return False
   867
   868	        if not (len(stmt.body) == 1 and len(stmt.body[0].body) == 0):
   869	            # Body not empty
   870	            return False
   871
   872	        if not stmt.else_body or len(stmt.else_body.body) == 0:
   873	            # No or empty else_body
   874	            return True
   875
   876	        # For elif, IfStmt are stored recursively in else_body
   877	        return self._is_stripped_if_stmt(stmt.else_body.body[0])

  1021	                )

  1023	                self.fail(

  1025	                    lineno,

  1028	                )

  1083	    def transform_args(
  1084	        self, args: ast3.arguments, line: int, no_type_check: bool = False
  1085	    ) -> list[Argument]:
  1086	        new_args = []
  1087	        names: list[ast3.arg] = []
  1088	        posonlyargs = getattr(args, "posonlyargs", cast(list[ast3.arg], []))
  1089	        args_args = posonlyargs + args.args
  1090	        args_defaults = args.defaults
  1091	        num_no_defaults = len(args_args) - len(args_defaults)
  1092	        # positional arguments without defaults
  1093	        for i, a in enumerate(args_args[:num_no_defaults]):
  1094	            pos_only = i < len(posonlyargs)
  1095	            new_args.append(self.make_argument(a, None, ARG_POS, no_type_check, pos_only))
  1096	            names.append(a)
  1097
  1098	        # positional arguments with defaults
  1099	        for i, (a, d) in enumerate(zip(args_args[num_no_defaults:], args_defaults)):
  1100	            pos_only = num_no_defaults + i < len(posonlyargs)
  1101	            new_args.append(self.make_argument(a, d, ARG_OPT, no_type_check, pos_only))
  1102	            names.append(a)
  1103
  1104	        # *arg
  1105	        if args.vararg is not None:
  1106	            new_args.append(self.make_argument(args.vararg, None, ARG_STAR, no_type_check))
  1107	            names.append(args.vararg)
  1108
  1109	        # keyword-only arguments with defaults
  1110	        for a, kd in zip(args.kwonlyargs, args.kw_defaults):
  1111	            new_args.append(
  1112	                self.make_argument(
  1113	                    a, kd, ARG_NAMED if kd is None else ARG_NAMED_OPT, no_type_check
  1114	                )
  1115	            )
  1116	            names.append(a)
  1117
  1118	        # **kwarg
  1119	        if args.kwarg is not None:
  1120	            new_args.append(self.make_argument(args.kwarg, None, ARG_STAR2, no_type_check))
  1121	            names.append(args.kwarg)
  1122
  1123	        check_arg_names([arg.variable.name for arg in new_args], names, self.fail_arg)
  1124
  1125	        return new_args
