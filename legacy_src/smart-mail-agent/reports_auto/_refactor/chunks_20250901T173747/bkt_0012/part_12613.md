   471	class DropShorterLongHelpFormatter(argparse.HelpFormatter):
   472	    """Shorten help for long options that differ only in extra hyphens.
   473
   474	    - Collapse **long** options that are the same except for extra hyphens.
   475	    - Shortcut if there are only two options and one of them is a short one.
   476	    - Cache result on the action object as this is called at least 2 times.
   477	    """
   478
   479	    def __init__(self, *args: Any, **kwargs: Any) -> None:
   480	        # Use more accurate terminal width.
   481	        if "width" not in kwargs:
   482	            kwargs["width"] = _pytest._io.get_terminal_width()
   483	        super().__init__(*args, **kwargs)

   485	    def _format_action_invocation(self, action: argparse.Action) -> str:
   486	        orgstr = super()._format_action_invocation(action)
   487	        if orgstr and orgstr[0] != "-":  # only optional arguments
   488	            return orgstr
   489	        res: str | None = getattr(action, "_formatted_action_invocation", None)
   490	        if res:
   491	            return res
   492	        options = orgstr.split(", ")
   493	        if len(options) == 2 and (len(options[0]) == 2 or len(options[1]) == 2):
   494	            # a shortcut for '-h, --help' or '--abc', '-a'
   495	            action._formatted_action_invocation = orgstr  # type: ignore
   496	            return orgstr
   497	        return_list = []
   498	        short_long: dict[str, str] = {}
   499	        for option in options:
   500	            if len(option) == 2 or option[2] == " ":
   501	                continue
   502	            if not option.startswith("--"):
   503	                raise ArgumentError(
   504	                    f'long optional argument without "--": [{option}]', option
   505	                )
   506	            xxoption = option[2:]
   507	            shortened = xxoption.replace("-", "")
   508	            if shortened not in short_long or len(short_long[shortened]) < len(
   509	                xxoption
   510	            ):
   511	                short_long[shortened] = xxoption
   512	        # now short_long has been filled out to the longest with dashes
   513	        # **and** we keep the right option ordering from add_argument
   514	        for option in options:
   515	            if len(option) == 2 or option[2] == " ":
   516	                return_list.append(option)
   517	            if option[2:] == short_long.get(option.replace("-", "")):
   518	                return_list.append(option.replace(" ", "=", 1))
   519	        formatted_action_invocation = ", ".join(return_list)
   520	        action._formatted_action_invocation = formatted_action_invocation  # type: ignore
   521	        return formatted_action_invocation

   472	    """Context manager which replaces ``inspect.unwrap`` with a version
   473	    that's aware of mock objects and doesn't recurse into them."""
   474	    real_unwrap = inspect.unwrap

   477	        func: Callable[..., Any], *, stop: Callable[[Any], Any] | None = None
   478	    ) -> Any:

   480	            if stop is None or stop is _is_mocked:

   489	                PytestWarning,

   491	            raise

   494	    try:
   495	        yield

   594	def _init_checker_class() -> type[doctest.OutputChecker]:
   595	    import doctest
   596
   597	    class LiteralsOutputChecker(doctest.OutputChecker):
   598	        # Based on doctest_nose_plugin.py from the nltk project
   599	        # (https://github.com/nltk/nltk) and on the "numtest" doctest extension
   600	        # by Sebastien Boisgerault (https://github.com/boisgera/numtest).
   601
   602	        _unicode_literal_re = re.compile(r"(\W|^)[uU]([rR]?[\'\"])", re.UNICODE)
   603	        _bytes_literal_re = re.compile(r"(\W|^)[bB]([rR]?[\'\"])", re.UNICODE)
   604	        _number_re = re.compile(
   605	            r"""
   606	            (?P<number>
   607	              (?P<mantissa>
   608	                (?P<integer1> [+-]?\d*)\.(?P<fraction>\d+)
   609	                |
   610	                (?P<integer2> [+-]?\d+)\.
   611	              )
   612	              (?:
   613	                [Ee]
   614	                (?P<exponent1> [+-]?\d+)
   615	              )?
   616	              |
   617	              (?P<integer3> [+-]?\d+)
   618	              (?:
   619	                [Ee]
   620	                (?P<exponent2> [+-]?\d+)
   621	              )
   622	            )
   623	            """,
   624	            re.VERBOSE,
   625	        )
   626
   627	        def check_output(self, want: str, got: str, optionflags: int) -> bool:
   628	            if super().check_output(want, got, optionflags):
   629	                return True
   630
   631	            allow_unicode = optionflags & _get_allow_unicode_flag()
   632	            allow_bytes = optionflags & _get_allow_bytes_flag()
   633	            allow_number = optionflags & _get_number_flag()
   634
   635	            if not allow_unicode and not allow_bytes and not allow_number:
   636	                return False
   637
   638	            def remove_prefixes(regex: re.Pattern[str], txt: str) -> str:
   639	                return re.sub(regex, r"\1\2", txt)
   640
   641	            if allow_unicode:
   642	                want = remove_prefixes(self._unicode_literal_re, want)
   643	                got = remove_prefixes(self._unicode_literal_re, got)
   644
   645	            if allow_bytes:
   646	                want = remove_prefixes(self._bytes_literal_re, want)
   647	                got = remove_prefixes(self._bytes_literal_re, got)
   648
   649	            if allow_number:
   650	                got = self._remove_unwanted_precision(want, got)
   651
   652	            return super().check_output(want, got, optionflags)
   653
   654	        def _remove_unwanted_precision(self, want: str, got: str) -> str:
   655	            wants = list(self._number_re.finditer(want))
   656	            gots = list(self._number_re.finditer(got))
   657	            if len(wants) != len(gots):
   658	                return got
   659	            offset = 0
   660	            for w, g in zip(wants, gots):
   661	                fraction: str | None = w.group("fraction")
   662	                exponent: str | None = w.group("exponent1")
   663	                if exponent is None:
   664	                    exponent = w.group("exponent2")
   665	                precision = 0 if fraction is None else len(fraction)
   666	                if exponent is not None:
   667	                    precision -= int(exponent)
   668	                if float(w.group()) == approx(float(g.group()), abs=10**-precision):
   669	                    # They're close enough. Replace the text we actually
   670	                    # got with the text we want, so that it will match when we
   671	                    # check the string literally.
   672	                    got = (
   673	                        got[: g.start() + offset] + w.group() + got[g.end() + offset :]
   674	                    )
   675	                    offset += w.end() - w.start() - (g.end() - g.start())
   676	            return got
   677
   678	    return LiteralsOutputChecker

   135	    """Get the closest parent node (including self) which matches the given
   136	    scope.
   137

   151	    elif scope is Scope.Package:

   155	    else:
