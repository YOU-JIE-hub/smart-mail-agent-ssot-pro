   356	    @staticmethod
   357	    def _extract_syntax_information(exception: Exception) -> tuple[int, int]:
   358	        if (
   359	            len(exception.args) > 1
   360	            and exception.args[1]
   361	            and len(exception.args[1]) > 2
   362	        ):
   363	            token = exception.args[1]
   364	            row, column = token[1:3]
   365	        elif (
   366	            isinstance(exception, tokenize.TokenError)
   367	            and len(exception.args) == 2
   368	            and len(exception.args[1]) == 2
   369	        ):
   370	            token = ()
   371	            row, column = exception.args[1]
   372	        else:
   373	            token = ()
   374	            row, column = (1, 0)
   375
   376	        if (
   377	            column > 0
   378	            and token
   379	            and isinstance(exception, SyntaxError)
   380	            and len(token) == 4  # Python 3.9 or earlier
   381	        ):
   382	            # NOTE(sigmavirus24): SyntaxErrors report 1-indexed column
   383	            # numbers. We need to decrement the column number by 1 at
   384	            # least.
   385	            column_offset = 1
   386	            row_offset = 0
   387	            # See also: https://github.com/pycqa/flake8/issues/169,
   388	            # https://github.com/PyCQA/flake8/issues/1372
   389	            # On Python 3.9 and earlier, token will be a 4-item tuple with the
   390	            # last item being the string. Starting with 3.10, they added to
   391	            # the tuple so now instead of it ending with the code that failed
   392	            # to parse, it ends with the end of the section of code that
   393	            # failed to parse. Luckily the absolute position in the tuple is
   394	            # stable across versions so we can use that here
   395	            physical_line = token[3]
   396
   397	            # NOTE(sigmavirus24): Not all "tokens" have a string as the last
   398	            # argument. In this event, let's skip trying to find the correct
   399	            # column and row values.
   400	            if physical_line is not None:
   401	                # NOTE(sigmavirus24): SyntaxErrors also don't exactly have a
   402	                # "physical" line so much as what was accumulated by the point
   403	                # tokenizing failed.
   404	                # See also: https://github.com/pycqa/flake8/issues/169
   405	                lines = physical_line.rstrip("\n").split("\n")
   406	                row_offset = len(lines) - 1
   407	                logical_line = lines[0]
   408	                logical_line_length = len(logical_line)
   409	                if column > logical_line_length:
   410	                    column = logical_line_length
   411	            row -= row_offset
   412	            column -= column_offset
   413	        return row, column

   415	    def run_ast_checks(self) -> None:
   416	        """Run all checks expecting an abstract syntax tree."""
   417	        assert self.processor is not None, self.filename
   418	        ast = self.processor.build_ast()

   420	        for plugin in self.plugins.tree:
   421	            checker = self.run_check(plugin, tree=ast)
   422	            # If the plugin uses a class, call the run method of it, otherwise
   423	            # the call should return something iterable itself

   425	                runner = checker.run()

   427	                runner = checker

   429	                self.report(
   430	                    error_code=None,

   436	    def run_logical_checks(self) -> None:
   437	        """Run all checks expecting a logical line."""
   438	        assert self.processor is not None
   439	        comments, logical_line, mapping = self.processor.build_logical_line()
   440	        if not mapping:
   441	            return

   444	        LOG.debug('Logical line: "%s"', logical_line.rstrip())

   446	        for plugin in self.plugins.logical_line:

   451	                if line_number == column_offset == 0:

   453	                self.report(
   454	                    error_code=None,
   455	                    line_number=line_number,
   456	                    column=column_offset,
   457	                    text=text,
   458	                )

   460	        self.processor.next_logical_line()

   462	    def run_physical_checks(self, physical_line: str) -> None:
   463	        """Run all checks for a given physical line.

   465	        A single physical check may return multiple errors.

   493	    def process_tokens(self) -> None:
   494	        """Process tokens and trigger checks.
   495
   496	        Instead of using this directly, you should use
   497	        :meth:`flake8.checker.FileChecker.run_checks`.
   498	        """
   499	        assert self.processor is not None
   500	        parens = 0
   501	        statistics = self.statistics
   502	        file_processor = self.processor
   503	        prev_physical = ""
   504	        for token in file_processor.generate_tokens():
   505	            statistics["tokens"] += 1
   506	            self.check_physical_eol(token, prev_physical)
   507	            token_type, text = token[0:2]
   508	            if token_type == tokenize.OP:
   509	                parens = processor.count_parentheses(parens, text)
   510	            elif parens == 0:
   511	                if processor.token_is_newline(token):
   512	                    self.handle_newline(token_type)
   513	            prev_physical = token[4]
   514
   515	        if file_processor.tokens:
   516	            # If any tokens are left over, process them
   517	            self.run_physical_checks(file_processor.lines[-1])
   518	            self.run_logical_checks()

   520	    def run_checks(self) -> tuple[str, Results, dict[str, int]]:
   521	        """Run checks against the file."""
   522	        if self.processor is None or not self.should_process:
   523	            return self.display_name, self.results, self.statistics
   524
   525	        try:
   526	            self.run_ast_checks()
   527	            self.process_tokens()
   528	        except (SyntaxError, tokenize.TokenError) as e:
   529	            code = "E902" if isinstance(e, tokenize.TokenError) else "E999"
   530	            row, column = self._extract_syntax_information(e)
   531	            self.report(code, row, column, f"{type(e).__name__}: {e.args[0]}")
   532	            return self.display_name, self.results, self.statistics
   533
   534	        logical_lines = self.processor.statistics["logical lines"]
   535	        self.statistics["logical lines"] = logical_lines
   536	        return self.display_name, self.results, self.statistics
   537
   538	    def handle_newline(self, token_type: int) -> None:
   539	        """Handle the logic when encountering a newline token."""
   540	        assert self.processor is not None
   541	        if token_type == tokenize.NEWLINE:
   542	            self.run_logical_checks()
   543	            self.processor.reset_blank_before()
   544	        elif len(self.processor.tokens) == 1:
   545	            # The physical line contains only this token.
   546	            self.processor.visited_new_blank_line()
   547	            self.processor.delete_first_token()
   548	        else:
   549	            self.run_logical_checks()
   550
   551	    def check_physical_eol(
   552	        self, token: tokenize.TokenInfo, prev_physical: str
   553	    ) -> None:
   554	        """Run physical checks if and only if it is at the end of the line."""
   555	        assert self.processor is not None
   556	        if token.type == FSTRING_START:  # pragma: >=3.12 cover
   557	            self.processor.fstring_start(token.start[0])
   558	        elif token.type == TSTRING_START:  # pragma: >=3.14 cover
   559	            self.processor.tstring_start(token.start[0])
   560	        # a newline token ends a single physical line.
   561	        elif processor.is_eol_token(token):
   562	            # if the file does not end with a newline, the NEWLINE
   563	            # token is inserted by the parser, but it does not contain
   564	            # the previous physical line in `token[4]`
   565	            if token.line == "":
   566	                self.run_physical_checks(prev_physical)
   567	            else:
   568	                self.run_physical_checks(token.line)
   569	        elif processor.is_multiline_string(token):
   570	            # Less obviously, a string that contains newlines is a
   571	            # multiline string, either triple-quoted or with internal
   572	            # newlines backslash-escaped. Check every physical line in the
   573	            # string *except* for the last one: its newline is outside of
   574	            # the multiline string, so we consider it a regular physical
   575	            # line, and will check it like any other physical line.
   576	            #
   577	            # Subtleties:
   578	            # - have to wind self.line_number back because initially it
   579	            #   points to the last line of the string, and we want
   580	            #   check_physical() to give accurate feedback
   581	            for line in self.processor.multiline_string(token):
   582	                self.run_physical_checks(line)

   601	def find_offset(
   602	    offset: int, mapping: processor._LogicalMapping
   603	) -> tuple[int, int]:
   604	    """Find the offset tuple for a single offset."""
   605	    if isinstance(offset, tuple):
   606	        return offset
   607
   608	    for token in mapping:
   609	        token_offset = token[0]
   610	        if offset <= token_offset:
   611	            position = token[1]
   612	            break
   613	    else:
   614	        position = (0, 0)
   615	        offset = token_offset = 0
   616	    return (position[0], position[1] + offset - token_offset)

   252	def extract_refexpr_names(expr: RefExpr) -> set[str]:
   253	    """Recursively extracts all module references from a reference expression.
   254
   255	    Note that currently, the only two subclasses of RefExpr are NameExpr and
   256	    MemberExpr."""
   257	    output: set[str] = set()
   258	    while isinstance(expr.node, MypyFile) or expr.fullname:
   259	        if isinstance(expr.node, MypyFile) and expr.fullname:
   260	            # If it's None, something's wrong (perhaps due to an
   261	            # import cycle or a suppressed error).  For now we just
   262	            # skip it.
   263	            output.add(expr.fullname)
   264
   265	        if isinstance(expr, NameExpr):
   266	            is_suppressed_import = isinstance(expr.node, Var) and expr.node.is_suppressed_import
   267	            if isinstance(expr.node, TypeInfo):
   268	                # Reference to a class or a nested class
   269	                output.update(split_module_names(expr.node.module_name))
   270	            elif "." in expr.fullname and not is_suppressed_import:
   271	                # Everything else (that is not a silenced import within a class)
   272	                output.add(expr.fullname.rsplit(".", 1)[0])
   273	            break
   274	        elif isinstance(expr, MemberExpr):
   275	            if isinstance(expr.expr, RefExpr):
   276	                expr = expr.expr
   277	            else:
   278	                break
   279	        else:
   280	            raise AssertionError(f"Unknown RefExpr subclass: {type(expr)}")
   281	    return output

   303	    """Expression type checker.
   304
   305	    This class works closely together with checker.TypeChecker.
   306	    """

   325	        plugin: Plugin,
   326	        per_line_checking_time_ns: dict[int, int],
   327	    ) -> None:

   330	        self.msg = msg

   332	        self.per_line_checking_time_ns = per_line_checking_time_ns
   333	        self.collect_line_checking_stats = chk.options.line_checking_stats is not None
   334	        # Are we already visiting some expression? This is used to avoid double counting
   335	        # time for nested expressions.

   348	        # Callee in a call expression is in some sense both runtime context and

   358	    def visit_name_expr(self, e: NameExpr) -> Type:
   359	        """Type check a name expression.
   360
   361	        It can be of any kind: local, member or global.
   362	        """
   363	        self.chk.module_refs.update(extract_refexpr_names(e))
   364	        result = self.analyze_ref_expr(e)
   365	        narrowed = self.narrow_type_from_binder(e, result)
   366	        self.chk.check_deprecated(e.node, e)
   367	        return narrowed

   369	    def analyze_ref_expr(self, e: RefExpr, lvalue: bool = False) -> Type:
   370	        result: Type | None = None
   371	        node = e.node
   372
   373	        if isinstance(e, NameExpr) and e.is_special_form:
   374	            # A special form definition, nothing to check here.
   375	            return AnyType(TypeOfAny.special_form)
   376
   377	        if isinstance(node, Var):
   378	            # Variable reference.
   379	            result = self.analyze_var_ref(node, e)
   380	            if isinstance(result, PartialType):
   381	                result = self.chk.handle_partial_var_type(result, lvalue, node, e)
   382	        elif isinstance(node, Decorator):
   383	            result = self.analyze_var_ref(node.var, e)
   384	        elif isinstance(node, OverloadedFuncDef):
   385	            if node.type is None:
   386	                if self.chk.in_checked_function() and node.items:
   387	                    self.chk.handle_cannot_determine_type(node.name, e)
   388	                result = AnyType(TypeOfAny.from_error)
   389	            else:
   390	                result = node.type
   391	        elif isinstance(node, (FuncDef, TypeInfo, TypeAlias, MypyFile, TypeVarLikeExpr)):
   392	            result = self.analyze_static_reference(node, e, e.is_alias_rvalue or lvalue)
   393	        else:
   394	            if isinstance(node, PlaceholderNode):
   395	                assert False, f"PlaceholderNode {node.fullname!r} leaked to checker"
   396	            # Unknown reference; use any type implicitly to avoid
   397	            # generating extra type errors.
   398	            result = AnyType(TypeOfAny.from_error)
   399	        if isinstance(node, TypeInfo):
   400	            if isinstance(result, CallableType) and isinstance(  # type: ignore[misc]
   401	                result.ret_type, Instance
   402	            ):
   403	                # We need to set correct line and column
   404	                # TODO: always do this in type_object_type by passing the original context
   405	                result.ret_type.line = e.line
   406	                result.ret_type.column = e.column
   407	            if is_type_type_context(self.type_context[-1]):
   408	                # This is the type in a type[] expression, so substitute type
   409	                # variables with Any.
   410	                result = erasetype.erase_typevars(result)
   411	        assert result is not None
   412	        return result

   414	    def analyze_static_reference(
   415	        self,
   416	        node: SymbolNode,
   417	        ctx: Context,
   418	        is_lvalue: bool,
   419	        *,
   420	        include_modules: bool = True,
   421	        suppress_errors: bool = False,
   422	    ) -> Type:
   423	        """
   424	        This is the version of analyze_ref_expr() that doesn't do any deferrals.
   425
   426	        This function can be used by member access to "static" attributes. For example,
   427	        when accessing module attributes in protocol checks, or accessing attributes of
   428	        special kinds (like TypeAlias, TypeInfo, etc.) on an instance or class object.
   429	        # TODO: merge with analyze_ref_expr() when we are confident about performance.
   430	        """
   431	        if isinstance(node, (Var, Decorator, OverloadedFuncDef)):
   432	            return node.type or AnyType(TypeOfAny.special_form)
   433	        elif isinstance(node, FuncDef):
   434	            return function_type(node, self.named_type("builtins.function"))
   435	        elif isinstance(node, TypeInfo):
   436	            # Reference to a type object.
   437	            if node.typeddict_type:
   438	                # We special-case TypedDict, because they don't define any constructor.
   439	                return self.typeddict_callable(node)
   440	            elif node.fullname == "types.NoneType":
   441	                # We special case NoneType, because its stub definition is not related to None.
   442	                return TypeType(NoneType())
   443	            else:
   444	                return type_object_type(node, self.named_type)
   445	        elif isinstance(node, TypeAlias):
   446	            # Something that refers to a type alias appears in runtime context.
   447	            # Note that we suppress bogus errors for alias redefinitions,
   448	            # they are already reported in semanal.py.
   449	            with self.msg.filter_errors() if suppress_errors else nullcontext():
   450	                return self.alias_type_in_runtime_context(
   451	                    node, ctx=ctx, alias_definition=is_lvalue
   452	                )
   453	        elif isinstance(node, TypeVarExpr):
   454	            return self.named_type("typing.TypeVar")
   455	        elif isinstance(node, (ParamSpecExpr, TypeVarTupleExpr)):
   456	            return self.object_type()
   457	        elif isinstance(node, MypyFile):
   458	            # Reference to a module object.
   459	            return self.module_type(node) if include_modules else AnyType(TypeOfAny.special_form)
   460	        return AnyType(TypeOfAny.from_error)
   461
   462	    def analyze_var_ref(self, var: Var, context: Context) -> Type:
   463	        if var.type:
   464	            var_type = get_proper_type(var.type)
   465	            if isinstance(var_type, Instance):
   466	                if var.fullname == "typing.Any":
   467	                    # The typeshed type is 'object'; give a more useful type in runtime context
   468	                    return self.named_type("typing._SpecialForm")
   469	                if self.is_literal_context() and var_type.last_known_value is not None:
   470	                    return var_type.last_known_value
   471	                if var.name in {"True", "False"}:
   472	                    return self.infer_literal_expr_type(var.name == "True", "builtins.bool")
   473	            return var.type
   474	        else:
   475	            if not var.is_ready and self.chk.in_checked_function():
   476	                self.chk.handle_cannot_determine_type(var.name, context)
   477	            # Implicit 'Any' type.
   478	            return AnyType(TypeOfAny.special_form)

   480	    def module_type(self, node: MypyFile) -> Instance:
   481	        try:
   482	            result = self.named_type("types.ModuleType")
   483	        except KeyError:
   484	            # In test cases might 'types' may not be available.
   485	            # Fall back to a dummy 'object' type instead to
   486	            # avoid a crash.
   487	            result = self.named_type("builtins.object")
   488	        module_attrs: dict[str, Type] = {}
   489	        immutable = set()
   490	        for name, n in node.names.items():
   491	            if not n.module_public:
   492	                continue
   493	            if isinstance(n.node, Var) and n.node.is_final:
   494	                immutable.add(name)
   495	            if n.node is None:
   496	                module_attrs[name] = AnyType(TypeOfAny.from_error)
   497	            else:
   498	                # TODO: what to do about nested module references?
   499	                # They are non-trivial because there may be import cycles.
   500	                module_attrs[name] = self.analyze_static_reference(
   501	                    n.node, n.node, False, include_modules=False, suppress_errors=True
   502	                )
   503	        result.extra_attrs = ExtraAttrs(module_attrs, immutable, node.fullname)
   504	        return result

   506	    def visit_call_expr(self, e: CallExpr, allow_none_return: bool = False) -> Type:
   507	        """Type check a call expression."""
   508	        if e.analyzed:
   509	            if isinstance(e.analyzed, NamedTupleExpr) and not e.analyzed.is_typed:
   510	                # Type check the arguments, but ignore the results. This relies
   511	                # on the typeshed stubs to type check the arguments.
   512	                self.visit_call_expr_inner(e)
   513	            # It's really a special form that only looks like a call.
   514	            return self.accept(e.analyzed, self.type_context[-1])
   515	        return self.visit_call_expr_inner(e, allow_none_return=allow_none_return)

   517	    def refers_to_typeddict(self, base: Expression) -> bool:
   518	        if not isinstance(base, RefExpr):
   519	            return False
   520	        if isinstance(base.node, TypeInfo) and base.node.typeddict_type is not None:
   521	            # Direct reference.
   522	            return True
   523	        return isinstance(base.node, TypeAlias) and isinstance(
   524	            get_proper_type(base.node.target), TypedDictType
   525	        )

   527	    def visit_call_expr_inner(self, e: CallExpr, allow_none_return: bool = False) -> Type:
   528	        if (
   529	            self.refers_to_typeddict(e.callee)
   530	            or isinstance(e.callee, IndexExpr)
   531	            and self.refers_to_typeddict(e.callee.base)
   532	        ):
   533	            typeddict_callable = get_proper_type(self.accept(e.callee, is_callee=True))
   534	            if isinstance(typeddict_callable, CallableType):
   535	                typeddict_type = get_proper_type(typeddict_callable.ret_type)
   536	                assert isinstance(typeddict_type, TypedDictType)
   537	                return self.check_typeddict_call(
   538	                    typeddict_type, e.arg_kinds, e.arg_names, e.args, e, typeddict_callable
   539	                )
   540	        if (
   541	            isinstance(e.callee, NameExpr)
   542	            and e.callee.name in ("isinstance", "issubclass")
   543	            and len(e.args) == 2
   544	        ):
   545	            for typ in mypy.checker.flatten(e.args[1]):
   546	                node = None
   547	                if isinstance(typ, NameExpr):
   548	                    try:
   549	                        node = self.chk.lookup_qualified(typ.name)
   550	                    except KeyError:
   551	                        # Undefined names should already be reported in semantic analysis.
   552	                        pass
   553	                if is_expr_literal_type(typ):
   554	                    self.msg.cannot_use_function_with_type(e.callee.name, "Literal", e)
   555	                    continue
   556	                if node and isinstance(node.node, TypeAlias):
   557	                    target = get_proper_type(node.node.target)
   558	                    if isinstance(target, AnyType):
   559	                        self.msg.cannot_use_function_with_type(e.callee.name, "Any", e)
   560	                        continue
   561	                    if isinstance(target, NoneType):
   562	                        continue
   563	                if (
   564	                    isinstance(typ, IndexExpr)
   565	                    and isinstance(typ.analyzed, (TypeApplication, TypeAliasExpr))
   566	                ) or (
   567	                    isinstance(typ, NameExpr)
   568	                    and node
   569	                    and isinstance(node.node, TypeAlias)
   570	                    and not node.node.no_args
   571	                    and not (
   572	                        isinstance(union_target := get_proper_type(node.node.target), UnionType)
   573	                        and union_target.uses_pep604_syntax
   574	                    )
   575	                ):
   576	                    self.msg.type_arguments_not_allowed(e)
   577	                if isinstance(typ, RefExpr) and isinstance(typ.node, TypeInfo):
   578	                    if typ.node.typeddict_type:
   579	                        self.msg.cannot_use_function_with_type(e.callee.name, "TypedDict", e)
   580	                    elif typ.node.is_newtype:
   581	                        self.msg.cannot_use_function_with_type(e.callee.name, "NewType", e)
   582	        self.try_infer_partial_type(e)
   583	        type_context = None
   584	        if isinstance(e.callee, LambdaExpr):
   585	            formal_to_actual = map_actuals_to_formals(
   586	                e.arg_kinds,
   587	                e.arg_names,
   588	                e.callee.arg_kinds,
   589	                e.callee.arg_names,
   590	                lambda i: self.accept(e.args[i]),
   591	            )
   592
   593	            arg_types = [
   594	                join.join_type_list([self.accept(e.args[j]) for j in formal_to_actual[i]])
   595	                for i in range(len(e.callee.arg_kinds))
   596	            ]
   597	            type_context = CallableType(
   598	                arg_types,
   599	                e.callee.arg_kinds,
   600	                e.callee.arg_names,
   601	                ret_type=self.object_type(),
   602	                fallback=self.named_type("builtins.function"),
   603	            )
   604	        callee_type = get_proper_type(
   605	            self.accept(e.callee, type_context, always_allow_any=True, is_callee=True)
   606	        )
   607
   608	        # Figure out the full name of the callee for plugin lookup.
   609	        object_type = None
   610	        member = None
   611	        fullname = None
   612	        if isinstance(e.callee, RefExpr):
   613	            # There are two special cases where plugins might act:
   614	            # * A "static" reference/alias to a class or function;
   615	            #   get_function_hook() will be invoked for these.
   616	            fullname = e.callee.fullname or None
   617	            if isinstance(e.callee.node, TypeAlias):
   618	                target = get_proper_type(e.callee.node.target)
   619	                if isinstance(target, Instance):
   620	                    fullname = target.type.fullname
   621	            # * Call to a method on object that has a full name (see
   622	            #   method_fullname() for details on supported objects);
   623	            #   get_method_hook() and get_method_signature_hook() will
   624	            #   be invoked for these.
   625	            if (
   626	                not fullname
   627	                and isinstance(e.callee, MemberExpr)
   628	                and self.chk.has_type(e.callee.expr)
   629	            ):
   630	                member = e.callee.name
   631	                object_type = self.chk.lookup_type(e.callee.expr)
   632
   633	        if (
   634	            self.chk.options.disallow_untyped_calls
   635	            and self.chk.in_checked_function()
   636	            and isinstance(callee_type, CallableType)
   637	            and callee_type.implicit
   638	            and callee_type.name != LAMBDA_NAME
   639	        ):
   640	            if fullname is None and member is not None:
   641	                assert object_type is not None
   642	                fullname = self.method_fullname(object_type, member)
   643	            if not fullname or not any(
   644	                fullname == p or fullname.startswith(f"{p}.")
   645	                for p in self.chk.options.untyped_calls_exclude
   646	            ):
   647	                self.msg.untyped_function_call(callee_type, e)
   648
   649	        ret_type = self.check_call_expr_with_callee_type(
   650	            callee_type, e, fullname, object_type, member
   651	        )
   652	        if isinstance(e.callee, RefExpr) and len(e.args) == 2:
   653	            if e.callee.fullname in ("builtins.isinstance", "builtins.issubclass"):
   654	                self.check_runtime_protocol_test(e)
   655	            if e.callee.fullname == "builtins.issubclass":
   656	                self.check_protocol_issubclass(e)
   657	        if isinstance(e.callee, MemberExpr) and e.callee.name == "format":
   658	            self.check_str_format_call(e)
   659	        ret_type = get_proper_type(ret_type)
   660	        if isinstance(ret_type, UnionType):
   661	            ret_type = make_simplified_union(ret_type.items)
   662	        if isinstance(ret_type, UninhabitedType) and not ret_type.ambiguous:
   663	            self.chk.binder.unreachable()
   664	        # Warn on calls to functions that always return None. The check
