   185	def consider_sys_version_info(expr: Expression, pyversion: tuple[int, ...]) -> int:
   186	    """Consider whether expr is a comparison involving sys.version_info.
   187
   188	    Return ALWAYS_TRUE, ALWAYS_FALSE, or TRUTH_VALUE_UNKNOWN.
   189	    """
   190	    # Cases supported:
   191	    # - sys.version_info[<int>] <compare_op> <int>
   192	    # - sys.version_info[:<int>] <compare_op> <tuple_of_n_ints>
   193	    # - sys.version_info <compare_op> <tuple_of_1_or_2_ints>
   194	    #   (in this case <compare_op> must be >, >=, <, <=, but cannot be ==, !=)
   195	    if not isinstance(expr, ComparisonExpr):
   196	        return TRUTH_VALUE_UNKNOWN
   197	    # Let's not yet support chained comparisons.
   198	    if len(expr.operators) > 1:
   199	        return TRUTH_VALUE_UNKNOWN
   200	    op = expr.operators[0]
   201	    if op not in ("==", "!=", "<=", ">=", "<", ">"):
   202	        return TRUTH_VALUE_UNKNOWN
   203
   204	    index = contains_sys_version_info(expr.operands[0])
   205	    thing = contains_int_or_tuple_of_ints(expr.operands[1])
   206	    if index is None or thing is None:
   207	        index = contains_sys_version_info(expr.operands[1])
   208	        thing = contains_int_or_tuple_of_ints(expr.operands[0])
   209	        op = reverse_op[op]
   210	    if isinstance(index, int) and isinstance(thing, int):
   211	        # sys.version_info[i] <compare_op> k
   212	        if 0 <= index <= 1:
   213	            return fixed_comparison(pyversion[index], op, thing)
   214	        else:
   215	            return TRUTH_VALUE_UNKNOWN
   216	    elif isinstance(index, tuple) and isinstance(thing, tuple):
   217	        lo, hi = index
   218	        if lo is None:
   219	            lo = 0
   220	        if hi is None:
   221	            hi = 2
   222	        if 0 <= lo < hi <= 2:
   223	            val = pyversion[lo:hi]
   224	            if len(val) == len(thing) or len(val) > len(thing) and op not in ("==", "!="):
   225	                return fixed_comparison(val, op, thing)
   226	    return TRUTH_VALUE_UNKNOWN

   230	    """Consider whether expr is a comparison involving sys.platform.

   232	    Return ALWAYS_TRUE, ALWAYS_FALSE, or TRUTH_VALUE_UNKNOWN.
   233	    """
   234	    # Cases supported:
   235	    # - sys.platform == 'linux'
   236	    # - sys.platform != 'win32'

   259	            return TRUTH_VALUE_UNKNOWN

   262	    def handle_refine(self, expr: NameExpr) -> None:
   263	        """Store assignment to an existing name (that replaces previous value, if any)."""
   264	        name = expr.name
   265	        if name in self.refs[-1]:
   266	            names = self.refs[-1][name]
   267	            if not names:
   268	                names.append([])
   269	            names[-1].append(expr)
   270
   271	    def handle_ref(self, expr: NameExpr) -> None:
   272	        """Store reference to defined name."""
   273	        name = expr.name
   274	        if name in self.refs[-1]:
   275	            names = self.refs[-1][name]
   276	            if not names:
   277	                names.append([])
   278	            names[-1].append(expr)
   279	        num_reads = self.num_reads[-1]
   280	        num_reads[name] = num_reads.get(name, 0) + 1
   281
   282	    def flush_refs(self) -> None:
   283	        """Rename all references within the current scope.
   284
   285	        This will be called at the end of a scope.
   286	        """
   287	        is_func = self.scope_kinds[-1] == FUNCTION
   288	        for refs in self.refs[-1].values():
   289	            if len(refs) == 1:
   290	                # Only one definition -- no renaming needed.
   291	                continue
   292	            if is_func:
   293	                # In a function, don't rename the first definition, as it
   294	                # may be an argument that must preserve the name.
   295	                to_rename = refs[1:]
   296	            else:
   297	                # At module top level, don't rename the final definition,
   298	                # as it will be publicly visible outside the module.
   299	                to_rename = refs[:-1]
   300	            for i, item in enumerate(to_rename):
   301	                rename_refs(item, i)
   302	        self.refs.pop()

   332	            yield

   334	            self.loop_depth -= 1

   336	    def current_block(self) -> int:

   339	    @contextmanager
   340	    def enter_scope(self, kind: int) -> Iterator[None]:
   341	        self.var_blocks.append({})
   342	        self.refs.append({})
   343	        self.num_reads.append({})
   344	        self.scope_kinds.append(kind)
   345	        try:

   347	        finally:

   351	            self.scope_kinds.pop()

   353	    def is_nested(self) -> int:
   354	        return len(self.var_blocks) > 1

   356	    def reject_redefinition_of_vars_in_scope(self) -> None:
   357	        """Make it impossible to redefine defined variables in the current scope.
   358
   359	        This is used if we encounter a function definition that
   360	        can make it ambiguous which definition is live. Example:
   361
   362	          x = 0
   363
   364	          def f() -> int:
   365	              return x
   366
   367	          x = ''  # Error -- cannot redefine x across function definition
   368	        """
   369	        var_blocks = self.var_blocks[-1]
   370	        for key in var_blocks:
   371	            var_blocks[key] = -1
   372
   373	    def reject_redefinition_of_vars_in_loop(self) -> None:
   374	        """Reject redefinition of variables in the innermost loop.
   375
   376	        If there is an early exit from a loop, there may be ambiguity about which
   377	        value may escape the loop. Example where this matters:
   378
   379	          while f():
   380	              x = 0
   381	              if g():
   382	                  break
   383	              x = ''  # Error -- not a redefinition
   384	          reveal_type(x)  # int
   385
   386	        This method ensures that the second assignment to 'x' doesn't introduce a new
   387	        variable.
   388	        """
   389	        var_blocks = self.var_blocks[-1]
   390	        for key, block in var_blocks.items():
   391	            if self.block_loop_depth.get(block) == self.loop_depth:
   392	                var_blocks[key] = -1
