   555	        """@property with setters and getters, @overload chain and some others."""

   580	    def _get_func_args(self, o: FuncDef, ctx: FunctionContext) -> list[ArgSig]:
   581	        args: list[ArgSig] = []
   582
   583	        # Ignore pos-only status of magic methods whose args names are elided by mypy at parse
   584	        actually_pos_only_args = o.name not in MAGIC_METHODS_POS_ARGS_ONLY
   585	        pos_only_marker_position = 0  # Where to insert "/", if any
   586	        for i, arg_ in enumerate(o.arguments):
   587	            var = arg_.variable
   588	            kind = arg_.kind
   589	            name = var.name
   590	            annotated_type = (
   591	                o.unanalyzed_type.arg_types[i]
   592	                if isinstance(o.unanalyzed_type, CallableType)
   593	                else None
   594	            )
   595	            # I think the name check is incorrect: there are libraries which
   596	            # name their 0th argument other than self/cls
   597	            is_self_arg = i == 0 and name == "self"
   598	            is_cls_arg = i == 0 and name == "cls"
   599	            typename: str | None = None
   600	            if annotated_type and not is_self_arg and not is_cls_arg:
   601	                # Luckily, an argument explicitly annotated with "Any" has
   602	                # type "UnboundType" and will not match.
   603	                if not isinstance(get_proper_type(annotated_type), AnyType):
   604	                    typename = self.print_annotation(annotated_type)
   605
   606	            if actually_pos_only_args and arg_.pos_only:
   607	                pos_only_marker_position += 1
   608
   609	            if kind.is_named() and not any(arg.name.startswith("*") for arg in args):
   610	                args.append(ArgSig("*"))
   611
   612	            default = "..."
   613	            if arg_.initializer:
   614	                if not typename:
   615	                    typename = self.get_str_type_of_node(arg_.initializer, can_be_incomplete=False)
   616	                potential_default, valid = self.get_str_default_of_node(arg_.initializer)
   617	                if valid and len(potential_default) <= 200:
   618	                    default = potential_default
   619	            elif kind == ARG_STAR:
   620	                name = f"*{name}"
   621	            elif kind == ARG_STAR2:
   622	                name = f"**{name}"
   623
   624	            args.append(
   625	                ArgSig(name, typename, default=bool(arg_.initializer), default_value=default)
   626	            )
   627	        if pos_only_marker_position:
   628	            args.insert(pos_only_marker_position, ArgSig("/"))
   629
   630	        if ctx.class_info is not None and all(
   631	            arg.type is None and arg.default is False for arg in args
   632	        ):
   633	            new_args = infer_method_arg_types(
   634	                ctx.name, ctx.class_info.self_var, [arg.name for arg in args]
   635	            )
   636
   637	            if ctx.name == "__exit__":
   638	                self.import_tracker.add_import("types")
   639	                self.import_tracker.require_name("types")
   640
   641	            if new_args is not None:
   642	                args = new_args
   643
   644	        return args

   646	    def _get_func_return(self, o: FuncDef, ctx: FunctionContext) -> str | None:
   647	        if o.name != "__init__" and isinstance(o.unanalyzed_type, CallableType):
   648	            if isinstance(get_proper_type(o.unanalyzed_type.ret_type), AnyType):
   649	                # Luckily, a return type explicitly annotated with "Any" has
   650	                # type "UnboundType" and will enter the else branch.
   651	                return None  # implicit Any
   652	            else:
   653	                return self.print_annotation(o.unanalyzed_type.ret_type)
   654	        if o.abstract_status == IS_ABSTRACT or o.name in METHODS_WITH_RETURN_VALUE:
   655	            # Always assume abstract methods return Any unless explicitly annotated. Also
   656	            # some dunder methods should not have a None return type.
   657	            return None  # implicit Any
   658	        retname = infer_method_ret_type(o.name)
   659	        if retname is not None:
   660	            return retname
   661	        if has_yield_expression(o) or has_yield_from_expression(o):
   662	            generator_name = self.add_name("collections.abc.Generator")
   663	            yield_name = "None"
   664	            send_name: str | None = None
   665	            return_name: str | None = None
   666	            if has_yield_from_expression(o):
   667	                yield_name = send_name = self.add_name("_typeshed.Incomplete")
   668	            else:
   669	                for expr, in_assignment in all_yield_expressions(o):
   670	                    if expr.expr is not None and not is_none_expr(expr.expr):
   671	                        yield_name = self.add_name("_typeshed.Incomplete")
   672	                    if in_assignment:
   673	                        send_name = self.add_name("_typeshed.Incomplete")
   674	            if has_return_statement(o):
   675	                return_name = self.add_name("_typeshed.Incomplete")
   676	            if return_name is not None:
   677	                if send_name is None:
   678	                    send_name = "None"
   679	                return f"{generator_name}[{yield_name}, {send_name}, {return_name}]"
   680	            elif send_name is not None:
   681	                return f"{generator_name}[{yield_name}, {send_name}]"
   682	            else:
   683	                return f"{generator_name}[{yield_name}]"
   684	        if not has_return_statement(o) and o.abstract_status == NOT_ABSTRACT:
   685	            return "None"
   686	        return None

   696	    def visit_func_def(self, o: FuncDef) -> None:
   697	        is_dataclass_generated = (
   698	            self.analyzed and self.processing_dataclass and o.info.names[o.name].plugin_generated
   699	        )
   700	        if is_dataclass_generated:
   701	            # Skip methods generated by the @dataclass decorator
   702	            return
   703	        if (
   704	            self.is_private_name(o.name, o.fullname)
   705	            or self.is_not_in_all(o.name)
   706	            or (self.is_recorded_name(o.name) and not o.is_overload)
   707	        ):
   708	            self.clear_decorators()
   709	            return
   710	        if self.is_top_level() and self._state not in (EMPTY, FUNC):
   711	            self.add("\n")
   712	        if not self.is_top_level():
   713	            self_inits = find_self_initializers(o)
   714	            for init, value, annotation in self_inits:
   715	                if init in self.method_names:
   716	                    # Can't have both an attribute and a method/property with the same name.
   717	                    continue
   718	                init_code = self.get_init(init, value, annotation)
   719	                if init_code:
   720	                    self.add(init_code)
   721
   722	        if self._class_stack:
   723	            if len(o.arguments):
   724	                self_var = o.arguments[0].variable.name
   725	            else:
   726	                self_var = "self"
   727	            class_info: ClassInfo | None = None
   728	            for class_def in self._class_stack:
   729	                class_info = ClassInfo(class_def.name, self_var, parent=class_info)
   730	        else:
   731	            class_info = None
   732
   733	        ctx = FunctionContext(
   734	            module_name=self.module_name,
   735	            name=o.name,
   736	            docstring=self._get_func_docstring(o),
   737	            is_abstract=o.abstract_status != NOT_ABSTRACT,
   738	            class_info=class_info,
   739	        )
   740
   741	        self.record_name(o.name)
   742
   743	        default_sig = self.get_default_function_sig(o, ctx)
   744	        sigs = self.get_signatures(default_sig, self.sig_generators, ctx)
   745
   746	        for output in self.format_func_def(
   747	            sigs, is_coroutine=o.is_coroutine, decorators=self._decorators, docstring=ctx.docstring
   748	        ):
   749	            self.add(output + "\n")
   750
   751	        self.clear_decorators()
   752	        self._state = FUNC

   760	    def process_decorator(self, o: Decorator) -> None:
   761	        """Process a series of decorators.
   762
   763	        Only preserve certain special decorators such as @abstractmethod.
   764	        """
   765	        o.func.is_overload = False
   766	        for decorator in o.original_decorators:
   767	            d = decorator
   768	            if isinstance(d, CallExpr):
   769	                d = d.callee
   770	            if not isinstance(d, (NameExpr, MemberExpr)):
   771	                continue
   772	            qualname = get_qualified_name(d)
   773	            fullname = self.get_fullname(d)
   774	            if fullname in (
   775	                "builtins.property",
   776	                "builtins.staticmethod",
   777	                "builtins.classmethod",
   778	                "functools.cached_property",
   779	            ):
   780	                self.add_decorator(qualname, require_name=True)
   781	            elif fullname in (
   782	                "asyncio.coroutine",
   783	                "asyncio.coroutines.coroutine",
   784	                "types.coroutine",
   785	            ):
   786	                o.func.is_awaitable_coroutine = True
   787	                self.add_decorator(qualname, require_name=True)
   788	            elif fullname == "abc.abstractmethod":
   789	                self.add_decorator(qualname, require_name=True)
   790	                o.func.abstract_status = IS_ABSTRACT
   791	            elif fullname in (
   792	                "abc.abstractproperty",
   793	                "abc.abstractstaticmethod",
   794	                "abc.abstractclassmethod",
   795	            ):
   796	                abc_module = qualname.rpartition(".")[0]
   797	                if not abc_module:
   798	                    self.import_tracker.add_import("abc")
   799	                builtin_decorator_replacement = fullname[len("abc.abstract") :]
   800	                self.add_decorator(builtin_decorator_replacement, require_name=False)
   801	                self.add_decorator(f"{abc_module or 'abc'}.abstractmethod", require_name=True)
   802	                o.func.abstract_status = IS_ABSTRACT
   803	            elif fullname in OVERLOAD_NAMES:
   804	                self.add_decorator(qualname, require_name=True)
   805	                o.func.is_overload = True
   806	            elif qualname.endswith((".setter", ".deleter")):
   807	                self.add_decorator(qualname, require_name=False)
   808	            elif fullname in DATACLASS_TRANSFORM_NAMES:
   809	                p = AliasPrinter(self)
   810	                self._decorators.append(f"@{decorator.accept(p)}")
   811	            elif isinstance(decorator, (NameExpr, MemberExpr)):
   812	                p = AliasPrinter(self)
   813	                self._decorators.append(f"@{decorator.accept(p)}")

   827	    def visit_class_def(self, o: ClassDef) -> None:
   828	        self._class_stack.append(o)
   829	        self.method_names = find_method_names(o.defs.body)
   830	        sep: int | None = None
   831	        if self.is_top_level() and self._state != EMPTY:
   832	            sep = len(self._output)
   833	            self.add("\n")
   834	        decorators = self.get_class_decorators(o)
   835	        for d in decorators:
   836	            self.add(f"{self._indent}@{d}\n")
   837	        self.record_name(o.name)
   838	        base_types = self.get_base_types(o)
   839	        if base_types:
   840	            for base in base_types:
   841	                self.import_tracker.require_name(base)
   842	        if self.analyzed and o.info.is_enum:
   843	            self.processing_enum = True
   844	        if isinstance(o.metaclass, (NameExpr, MemberExpr)):
   845	            meta = o.metaclass.accept(AliasPrinter(self))
   846	            base_types.append("metaclass=" + meta)
   847	        elif self.analyzed and o.info.is_abstract and not o.info.is_protocol:
   848	            base_types.append("metaclass=abc.ABCMeta")
   849	            self.import_tracker.add_import("abc")
   850	            self.import_tracker.require_name("abc")
   851	        bases = f"({', '.join(base_types)})" if base_types else ""
   852	        type_args = self.format_type_args(o)
   853	        self.add(f"{self._indent}class {o.name}{type_args}{bases}:\n")
   854	        self.indent()
   855	        if self._include_docstrings and o.docstring:
   856	            docstring = mypy.util.quote_docstring(o.docstring)
   857	            self.add(f"{self._indent}{docstring}\n")
   858	        n = len(self._output)
   859	        self._vars.append([])
   860	        if self.analyzed and (spec := find_dataclass_transform_spec(o)):
   861	            self.processing_dataclass = True
   862	            self.dataclass_field_specifier = spec.field_specifiers
   863	        super().visit_class_def(o)
   864	        self.dedent()
   865	        self._vars.pop()
   866	        self._vars[-1].append(o.name)
   867	        if len(self._output) == n:
   868	            if self._state == EMPTY_CLASS and sep is not None:
   869	                self._output[sep] = ""
   870	            if not (self._include_docstrings and o.docstring):
   871	                self._output[-1] = self._output[-1][:-1] + " ...\n"
   872	            self._state = EMPTY_CLASS
   873	        else:
   874	            self._state = CLASS
   875	        self.method_names = set()
   876	        self.processing_dataclass = False
   877	        self.dataclass_field_specifier = ()
   878	        self._class_stack.pop(-1)
   879	        self.processing_enum = False

   881	    def get_base_types(self, cdef: ClassDef) -> list[str]:
   882	        """Get list of base classes for a class."""
   883	        base_types: list[str] = []
   884	        p = AliasPrinter(self)
   885	        for base in cdef.base_type_exprs + cdef.removed_base_type_exprs:
   886	            if isinstance(base, (NameExpr, MemberExpr)):
   887	                if self.get_fullname(base) != "builtins.object":
   888	                    base_types.append(get_qualified_name(base))
   889	            elif isinstance(base, IndexExpr):
   890	                base_types.append(base.accept(p))
   891	            elif isinstance(base, CallExpr):
   892	                # namedtuple(typename, fields), NamedTuple(typename, fields) calls can
   893	                # be used as a base class. The first argument is a string literal that
   894	                # is usually the same as the class name.
   895	                #
   896	                # Note:
   897	                # A call-based named tuple as a base class cannot be safely converted to
   898	                # a class-based NamedTuple definition because class attributes defined
   899	                # in the body of the class inheriting from the named tuple call are not
   900	                # namedtuple fields at runtime.
   901	                if self.is_namedtuple(base):
   902	                    nt_fields = self._get_namedtuple_fields(base)
   903	                    assert isinstance(base.args[0], StrExpr)
   904	                    typename = base.args[0].value
   905	                    if nt_fields is None:
   906	                        # Invalid namedtuple() call, cannot determine fields
   907	                        base_types.append(self.add_name("_typeshed.Incomplete"))
   908	                        continue
   909	                    fields_str = ", ".join(f"({f!r}, {t})" for f, t in nt_fields)
   910	                    namedtuple_name = self.add_name("typing.NamedTuple")
   911	                    base_types.append(f"{namedtuple_name}({typename!r}, [{fields_str}])")
   912	                elif self.is_typed_namedtuple(base):
   913	                    base_types.append(base.accept(p))
   914	                else:
   915	                    # At this point, we don't know what the base class is, so we
   916	                    # just use Incomplete as the base class.
   917	                    base_types.append(self.add_name("_typeshed.Incomplete"))
   918	        for name, value in cdef.keywords.items():
   919	            if name == "metaclass":
   920	                continue  # handled separately
   921	            processed_value = value.accept(p) or "..."  # at least, don't crash
   922	            base_types.append(f"{name}={processed_value}")
   923	        return base_types

   960	    def visit_assignment_stmt(self, o: AssignmentStmt) -> None:
   961	        foundl = []
   962
   963	        for lvalue in o.lvalues:
   964	            if isinstance(lvalue, NameExpr) and isinstance(o.rvalue, CallExpr):
   965	                if self.is_namedtuple(o.rvalue) or self.is_typed_namedtuple(o.rvalue):
   966	                    self.process_namedtuple(lvalue, o.rvalue)
   967	                    foundl.append(False)  # state is updated in process_namedtuple
   968	                    continue
   969	                if self.is_typeddict(o.rvalue):
   970	                    self.process_typeddict(lvalue, o.rvalue)
   971	                    foundl.append(False)  # state is updated in process_typeddict
   972	                    continue
   973	            if (
   974	                isinstance(lvalue, NameExpr)
   975	                and self.is_alias_expression(o.rvalue)
   976	                and not self.is_private_name(lvalue.name)
   977	            ):
   978	                is_explicit_type_alias = (
   979	                    o.unanalyzed_type and getattr(o.type, "name", None) == "TypeAlias"
   980	                )
   981	                if is_explicit_type_alias:
   982	                    self.process_typealias(lvalue, o.rvalue, is_explicit_type_alias=True)
   983	                    continue
   984
   985	                if not o.unanalyzed_type:
   986	                    self.process_typealias(lvalue, o.rvalue)
   987	                    continue
   988
   989	            if isinstance(lvalue, (TupleExpr, ListExpr)):
   990	                items = lvalue.items
   991	                if isinstance(o.unanalyzed_type, TupleType):  # type: ignore[misc]
   992	                    annotations: Iterable[Type | None] = o.unanalyzed_type.items
   993	                else:
   994	                    annotations = [None] * len(items)
   995	            else:
   996	                items = [lvalue]
   997	                annotations = [o.unanalyzed_type]
   998	            sep = False
   999	            found = False
  1000	            for item, annotation in zip(items, annotations):
  1001	                if isinstance(item, NameExpr):
  1002	                    init = self.get_init(item.name, o.rvalue, annotation)

  1146	    def is_alias_expression(self, expr: Expression, top_level: bool = True) -> bool:
  1147	        """Return True for things that look like target for an alias.
  1148
  1149	        Used to know if assignments look like type aliases, function alias,
  1150	        or module alias.
  1151	        """
  1152	        # Assignment of TypeVar(...)  and other typevar-likes are passed through
  1153	        if isinstance(expr, CallExpr) and self.get_fullname(expr.callee) in TYPE_VAR_LIKE_NAMES:
  1154	            return True
  1155	        elif isinstance(expr, EllipsisExpr):
  1156	            return not top_level
  1157	        elif isinstance(expr, NameExpr):
  1158	            if expr.name in ("True", "False"):
  1159	                return False
  1160	            elif expr.name == "None":
  1161	                return not top_level
  1162	            else:
  1163	                return not self.is_private_name(expr.name)
  1164	        elif isinstance(expr, MemberExpr) and self.analyzed:
  1165	            # Also add function and module aliases.
  1166	            return (
  1167	                top_level
  1168	                and isinstance(expr.node, (FuncDef, Decorator, MypyFile))
  1169	                or isinstance(expr.node, TypeInfo)
  1170	            ) and not self.is_private_member(expr.node.fullname)
  1171	        elif isinstance(expr, IndexExpr) and (
  1172	            (isinstance(expr.base, NameExpr) and not self.is_private_name(expr.base.name))
  1173	            or (  # Also some known aliases that could be member expression
  1174	                isinstance(expr.base, MemberExpr)
  1175	                and not self.is_private_member(get_qualified_name(expr.base))
  1176	                and self.get_fullname(expr.base).startswith(
  1177	                    ("builtins.", "typing.", "typing_extensions.", "collections.abc.")
  1178	                )
  1179	            )
  1180	        ):
  1181	            if isinstance(expr.index, TupleExpr):
  1182	                indices = expr.index.items
  1183	            else:
  1184	                indices = [expr.index]
  1185	            if expr.base.name == "Callable" and len(indices) == 2:
  1186	                args, ret = indices
  1187	                if isinstance(args, EllipsisExpr):
  1188	                    indices = [ret]
  1189	                elif isinstance(args, ListExpr):
  1190	                    indices = args.items + [ret]
  1191	                else:
  1192	                    return False
  1193	            return all(self.is_alias_expression(i, top_level=False) for i in indices)
  1194	        elif isinstance(expr, OpExpr) and expr.op == "|":
  1195	            return self.is_alias_expression(
  1196	                expr.left, top_level=False
  1197	            ) and self.is_alias_expression(expr.right, top_level=False)
  1198	        else:
  1199	            return False

  1385	    def maybe_unwrap_unary_expr(self, expr: Expression) -> Expression:
  1386	        """Unwrap (possibly nested) unary expressions.
  1387
  1388	        But, some unary expressions can change the type of expression.
  1389	        While we want to preserve it. For example, `~True` is `int`.
  1390	        So, we only allow a subset of unary expressions to be unwrapped.
  1391	        """
  1392	        if not isinstance(expr, UnaryExpr):
  1393	            return expr
  1394
  1395	        # First, try to unwrap `[+-]+ (int|float|complex)` expr:
  1396	        math_ops = ("+", "-")
  1397	        if expr.op in math_ops:
  1398	            while isinstance(expr, UnaryExpr):
  1399	                if expr.op not in math_ops or not isinstance(
  1400	                    expr.expr, (IntExpr, FloatExpr, ComplexExpr, UnaryExpr)
  1401	                ):
  1402	                    break
  1403	                expr = expr.expr
  1404	            return expr
  1405
  1406	        # Next, try `not bool` expr:
  1407	        if expr.op == "not":
  1408	            while isinstance(expr, UnaryExpr):
  1409	                if expr.op != "not" or not isinstance(expr.expr, (NameExpr, UnaryExpr)):
  1410	                    break
  1411	                if isinstance(expr.expr, NameExpr) and expr.expr.name not in ("True", "False"):
  1412	                    break
  1413	                expr = expr.expr
  1414	            return expr
  1415
  1416	        # This is some other unary expr, we cannot do anything with it (yet?).
  1417	        return expr

  1419	    def get_str_default_of_node(self, rvalue: Expression) -> tuple[str, bool]:

  1421
  1422	        Returns a 2-tuple of the default and whether or not it is valid.
  1423	        """
  1424	        if isinstance(rvalue, NameExpr):
  1425	            if rvalue.name in ("None", "True", "False"):
  1426	                return rvalue.name, True
  1427	        elif isinstance(rvalue, (IntExpr, FloatExpr)):
  1428	            return f"{rvalue.value}", True
  1429	        elif isinstance(rvalue, UnaryExpr):
  1430	            if isinstance(rvalue.expr, (IntExpr, FloatExpr)):
  1431	                return f"{rvalue.op}{rvalue.expr.value}", True
  1432	        elif isinstance(rvalue, StrExpr):
  1433	            return repr(rvalue.value), True
  1434	        elif isinstance(rvalue, BytesExpr):
  1435	            return "b" + repr(rvalue.value).replace("\\\\", "\\"), True
  1436	        elif isinstance(rvalue, TupleExpr):
  1437	            items_defaults = []
  1438	            for e in rvalue.items:
  1439	                e_default, valid = self.get_str_default_of_node(e)
  1440	                if not valid:
  1441	                    break
  1442	                items_defaults.append(e_default)
  1443	            else:
  1444	                closing = ",)" if len(items_defaults) == 1 else ")"
  1445	                default = "(" + ", ".join(items_defaults) + closing
  1446	                return default, True
  1447	        elif isinstance(rvalue, ListExpr):
  1448	            items_defaults = []
  1449	            for e in rvalue.items:
  1450	                e_default, valid = self.get_str_default_of_node(e)
  1451	                if not valid:
  1452	                    break
  1453	                items_defaults.append(e_default)
  1454	            else:
  1455	                default = "[" + ", ".join(items_defaults) + "]"
  1456	                return default, True
  1457	        elif isinstance(rvalue, SetExpr):
  1458	            items_defaults = []
  1459	            for e in rvalue.items:
  1460	                e_default, valid = self.get_str_default_of_node(e)
  1461	                if not valid:
  1462	                    break
  1463	                items_defaults.append(e_default)
  1464	            else:
  1465	                if items_defaults:
  1466	                    default = "{" + ", ".join(items_defaults) + "}"
  1467	                    return default, True
  1468	        elif isinstance(rvalue, DictExpr):
  1469	            items_defaults = []
  1470	            for k, v in rvalue.items:
  1471	                if k is None:
  1472	                    break
  1473	                k_default, k_valid = self.get_str_default_of_node(k)
  1474	                v_default, v_valid = self.get_str_default_of_node(v)
  1475	                if not (k_valid and v_valid):
  1476	                    break
  1477	                items_defaults.append(f"{k_default}: {v_default}")
  1478	            else:
  1479	                default = "{" + ", ".join(items_defaults) + "}"
  1480	                return default, True
  1481	        return "...", False

  1604	    modules: list[str], packages: list[str], verbose: bool, quiet: bool

  1611	        py_modules: list[StubSource] = []
  1612	        c_modules: list[StubSource] = []

  1614	        modules = modules + found
  1615	        modules = [
  1616	            mod for mod in modules if not is_non_library_module(mod)

  1677	def find_module_paths_using_search(
  1678	    modules: list[str], packages: list[str], search_path: list[str], pyversion: tuple[int, int]
  1679	) -> list[StubSource]:
  1680	    """Find sources for modules and packages requested.
  1681
  1682	    This function just looks for source files at the file system level.
  1683	    This is used if user passes --no-import, and will not find C modules.
  1684	    Exit if some of the modules or packages can't be found.
  1685	    """
  1686	    result: list[StubSource] = []
  1687	    typeshed_path = default_lib_path(mypy.build.default_data_dir(), pyversion, None)
  1688	    search_paths = SearchPaths((".",) + tuple(search_path), (), (), tuple(typeshed_path))
  1689	    cache = FindModuleCache(search_paths, fscache=None, options=None)
  1690	    for module in modules:
  1691	        m_result = cache.find_module(module)
  1692	        if isinstance(m_result, ModuleNotFoundReason):
  1693	            fail_missing(module, m_result)
  1694	            module_path = None
  1695	        else:
  1696	            module_path = m_result
  1697	        result.append(StubSource(module, module_path))
  1698	    for package in packages:
  1699	        p_result = cache.find_modules_recursive(package)
  1700	        if p_result:
  1701	            fail_missing(package, ModuleNotFoundReason.NOT_FOUND)
  1702	        sources = [StubSource(m.module, m.path) for m in p_result]
  1703	        result.extend(sources)
  1704
  1705	    result = [m for m in result if not is_non_library_module(m.module)]
  1706
  1707	    return result

  1732	def parse_source_file(mod: StubSource, mypy_options: MypyOptions) -> None:
  1733	    """Parse a source file.
  1734
  1735	    On success, store AST in the corresponding attribute of the stub source.
  1736	    If there are syntax errors, print them and exit.
  1737	    """
  1738	    assert mod.path is not None, "Not found module was not skipped"
  1739	    with open(mod.path, "rb") as f:
  1740	        data = f.read()
  1741	    source = mypy.util.decode_python_encoding(data)
  1742	    errors = Errors(mypy_options)
  1743	    mod.ast = mypy.parse.parse(
  1744	        source, fnam=mod.path, module=mod.module, errors=errors, options=mypy_options
  1745	    )
  1746	    mod.ast._fullname = mod.module
  1747	    if errors.is_blockers():
  1748	        # Syntax error!
  1749	        for m in errors.new_messages():
  1750	            sys.stderr.write(f"{m}\n")
  1751	        sys.exit(1)

  1754	def generate_asts_for_modules(
  1755	    py_modules: list[StubSource], parse_only: bool, mypy_options: MypyOptions, verbose: bool
  1756	) -> None:
  1757	    """Use mypy to parse (and optionally analyze) source files."""
  1758	    if not py_modules:
  1759	        return  # Nothing to do here, but there may be C modules
  1760	    if verbose:
  1761	        print(f"Processing {len(py_modules)} files...")
  1762	    if parse_only:
  1763	        for mod in py_modules:
  1764	            parse_source_file(mod, mypy_options)
  1765	        return
  1766	    # Perform full semantic analysis of the source set.
  1767	    try:
  1768	        res = build([module.source for module in py_modules], mypy_options)
  1769	    except CompileError as e:
  1770	        raise SystemExit(f"Critical error during semantic analysis: {e}") from e
  1771
  1772	    for mod in py_modules:
  1773	        mod.ast = res.graph[mod.module].tree
  1774	        # Use statically inferred __all__ if there is no runtime one.
  1775	        if mod.runtime_all is None:
  1776	            mod.runtime_all = res.manager.semantic_analyzer.export_map[mod.module]

  1833	    all_modules = py_modules + pyc_modules + c_modules

  1839	        assert mod.path is not None, "Not found module was not skipped"

  1860	    # Separately analyse C modules using different logic.
