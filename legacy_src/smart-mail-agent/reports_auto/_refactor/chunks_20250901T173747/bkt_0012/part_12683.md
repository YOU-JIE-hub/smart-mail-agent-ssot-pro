 235	    ignore_pos_arg_names: bool = False,
   236	    options: Options | None = None,
   237	    subtype_context: SubtypeContext | None = None,
   238	) -> bool:
   239	    return is_subtype(
   240	        a,
   241	        b,
   242	        ignore_type_params=ignore_type_params,
   243	        ignore_pos_arg_names=ignore_pos_arg_names,
   244	        options=options,
   245	        subtype_context=subtype_context,
   246	    ) and is_subtype(
   247	        b,
   248	        a,
   249	        ignore_type_params=ignore_type_params,
   250	        ignore_pos_arg_names=ignore_pos_arg_names,
   251	        options=options,
   252	        subtype_context=subtype_context,
   253	    )

   256	def is_same_type(
   257	    a: Type, b: Type, ignore_promotions: bool = True, subtype_context: SubtypeContext | None = None
   258	) -> bool:
   259	    """Are these types proper subtypes of each other?
   260
   261	    This means types may have different representation (e.g. an alias, or
   262	    a non-simplified union) but are semantically exchangeable in all contexts.
   263	    """
   264	    # First, use fast path for some common types. This is performance-critical.
   265	    if (
   266	        type(a) is Instance
   267	        and type(b) is Instance
   268	        and a.type == b.type
   269	        and len(a.args) == len(b.args)
   270	        and a.last_known_value is b.last_known_value
   271	    ):
   272	        return all(is_same_type(x, y) for x, y in zip(a.args, b.args))
   273	    elif isinstance(a, TypeVarType) and isinstance(b, TypeVarType) and a.id == b.id:
   274	        return True
   275
   276	    # Note that using ignore_promotions=True (default) makes types like int and int64
   277	    # considered not the same type (which is the case at runtime).
   278	    # Also Union[bool, int] (if it wasn't simplified before) will be different
   279	    # from plain int, etc.
   280	    return is_proper_subtype(
   281	        a, b, ignore_promotions=ignore_promotions, subtype_context=subtype_context
   282	    ) and is_proper_subtype(
   283	        b, a, ignore_promotions=ignore_promotions, subtype_context=subtype_context
   284	    )
   285
   286
   287	# This is a common entry point for subtyping checks (both proper and non-proper).
   288	# Never call this private function directly, use the public versions.
   289	def _is_subtype(
   290	    left: Type, right: Type, subtype_context: SubtypeContext, proper_subtype: bool
   291	) -> bool:
   292	    subtype_context.check_context(proper_subtype)
   293	    orig_right = right
   294	    orig_left = left
   295	    left = get_proper_type(left)
   296	    right = get_proper_type(right)
   297
   298	    # Note: Unpack type should not be a subtype of Any, since it may represent
   299	    # multiple types. This should always go through the visitor, to check arity.
   300	    if (
   301	        not proper_subtype
   302	        and isinstance(right, (AnyType, UnboundType, ErasedType))
   303	        and not isinstance(left, UnpackType)
   304	    ):
   305	        # TODO: should we consider all types proper subtypes of UnboundType and/or
   306	        # ErasedType as we do for non-proper subtyping.
   307	        return True
   308
   309	    if isinstance(right, UnionType) and not isinstance(left, UnionType):
   310	        # Normally, when 'left' is not itself a union, the only way
   311	        # 'left' can be a subtype of the union 'right' is if it is a
   312	        # subtype of one of the items making up the union.
   313	        if proper_subtype:
   314	            is_subtype_of_item = any(
   315	                is_proper_subtype(orig_left, item, subtype_context=subtype_context)
   316	                for item in right.items
   317	            )
   318	        else:
   319	            is_subtype_of_item = any(
   320	                is_subtype(orig_left, item, subtype_context=subtype_context)
   321	                for item in right.items
   322	            )
   323	        # Recombine rhs literal types, to make an enum type a subtype
   324	        # of a union of all enum items as literal types. Only do it if
   325	        # the previous check didn't succeed, since recombining can be
   326	        # expensive.
   327	        # `bool` is a special case, because `bool` is `Literal[True, False]`.
   328	        if (
   329	            not is_subtype_of_item
   330	            and isinstance(left, Instance)
   331	            and (left.type.is_enum or left.type.fullname == "builtins.bool")
   332	        ):
   333	            right = UnionType(
   334	                mypy.typeops.try_contracting_literals_in_union(flatten_nested_unions(right.items))
   335	            )
   336	            if proper_subtype:
   337	                is_subtype_of_item = any(
   338	                    is_proper_subtype(orig_left, item, subtype_context=subtype_context)
   339	                    for item in right.items
   340	                )
   341	            else:
   342	                is_subtype_of_item = any(
   343	                    is_subtype(orig_left, item, subtype_context=subtype_context)
   344	                    for item in right.items
   345	                )
   346	        # However, if 'left' is a type variable T, T might also have
   347	        # an upper bound which is itself a union. This case will be
   348	        # handled below by the SubtypeVisitor. We have to check both
   349	        # possibilities, to handle both cases like T <: Union[T, U]
   350	        # and cases like T <: B where B is the upper bound of T and is
   351	        # a union. (See #2314.)
   352	        if not isinstance(left, TypeVarType):
   353	            return is_subtype_of_item
   354	        elif is_subtype_of_item:
   355	            return True
   356	        # otherwise, fall through
   357	    return left.accept(SubtypeVisitor(orig_right, subtype_context, proper_subtype))

   360	def check_type_parameter(
   361	    left: Type, right: Type, variance: int, proper_subtype: bool, subtype_context: SubtypeContext
   362	) -> bool:
   363	    # It is safe to consider empty collection literals and similar as covariant, since
   364	    # such type can't be stored in a variable, see checker.is_valid_inferred_type().
   365	    if variance == INVARIANT:
   366	        p_left = get_proper_type(left)
   367	        if isinstance(p_left, UninhabitedType) and p_left.ambiguous:
   368	            variance = COVARIANT
   369	    # If variance hasn't been inferred yet, we are lenient and default to
   370	    # covariance. This shouldn't happen often, but it's very difficult to
   371	    # avoid these cases altogether.
   372	    if variance == COVARIANT or variance == VARIANCE_NOT_READY:
   373	        if proper_subtype:
   374	            return is_proper_subtype(left, right, subtype_context=subtype_context)
   375	        else:
   376	            return is_subtype(left, right, subtype_context=subtype_context)
   377	    elif variance == CONTRAVARIANT:
   378	        if proper_subtype:
   379	            return is_proper_subtype(right, left, subtype_context=subtype_context)
   380	        else:
   381	            return is_subtype(right, left, subtype_context=subtype_context)
   382	    else:
   383	        if proper_subtype:
   384	            # We pass ignore_promotions=False because it is a default for subtype checks.
   385	            # The actual value will be taken from the subtype_context, and it is whatever
   386	            # the original caller passed.
   387	            return is_same_type(
   388	                left, right, ignore_promotions=False, subtype_context=subtype_context
   389	            )
   390	        else:
   391	            return is_equivalent(left, right, subtype_context=subtype_context)

   394	class SubtypeVisitor(TypeVisitor[bool]):
   395	    def __init__(self, right: Type, subtype_context: SubtypeContext, proper_subtype: bool) -> None:
   396	        self.right = get_proper_type(right)
   397	        self.orig_right = right
   398	        self.proper_subtype = proper_subtype
   399	        self.subtype_context = subtype_context
   400	        self.options = subtype_context.options
   401	        self._subtype_kind = SubtypeVisitor.build_subtype_kind(subtype_context, proper_subtype)
   402
   403	    @staticmethod
   404	    def build_subtype_kind(subtype_context: SubtypeContext, proper_subtype: bool) -> SubtypeKind:
   405	        return (
   406	            state.strict_optional,
   407	            proper_subtype,
   408	            subtype_context.ignore_type_params,
   409	            subtype_context.ignore_pos_arg_names,
   410	            subtype_context.ignore_declared_variance,
   411	            subtype_context.always_covariant,
   412	            subtype_context.ignore_promotions,
   413	            subtype_context.erase_instances,
   414	            subtype_context.keep_erased_types,
   415	        )
   416
   417	    def _is_subtype(self, left: Type, right: Type) -> bool:
   418	        if self.proper_subtype:
   419	            return is_proper_subtype(left, right, subtype_context=self.subtype_context)
   420	        return is_subtype(left, right, subtype_context=self.subtype_context)
   421
   422	    def _all_subtypes(self, lefts: Iterable[Type], rights: Iterable[Type]) -> bool:
   423	        return all(self._is_subtype(li, ri) for (li, ri) in zip(lefts, rights))
   424
   425	    # visit_x(left) means: is left (which is an instance of X) a subtype of right?
   426
   427	    def visit_unbound_type(self, left: UnboundType) -> bool:
   428	        # This can be called if there is a bad type annotation. The result probably
   429	        # doesn't matter much but by returning True we simplify these bad types away
   430	        # from unions, which could filter out some bogus messages.
   431	        return True
   432
   433	    def visit_any(self, left: AnyType) -> bool:
   434	        return isinstance(self.right, AnyType) if self.proper_subtype else True
   435
   436	    def visit_none_type(self, left: NoneType) -> bool:
   437	        if state.strict_optional:
   438	            if isinstance(self.right, NoneType) or is_named_instance(
   439	                self.right, "builtins.object"
   440	            ):
   441	                return True
   442	            if isinstance(self.right, Instance) and self.right.type.is_protocol:
   443	                members = self.right.type.protocol_members
   444	                # None is compatible with Hashable (and other similar protocols). This is
   445	                # slightly sloppy since we don't check the signature of "__hash__".
   446	                # None is also compatible with `SupportsStr` protocol.
   447	                return not members or all(member in ("__hash__", "__str__") for member in members)
   448	            return False
   449	        else:
   450	            return True
   451
   452	    def visit_uninhabited_type(self, left: UninhabitedType) -> bool:
   453	        return True
   454
   455	    def visit_erased_type(self, left: ErasedType) -> bool:
   456	        # This may be encountered during type inference. The result probably doesn't
   457	        # matter much.
   458	        # TODO: it actually does matter, figure out more principled logic about this.
   459	        return not self.subtype_context.keep_erased_types
   460
   461	    def visit_deleted_type(self, left: DeletedType) -> bool:
   462	        return True
   463
   464	    def visit_instance(self, left: Instance) -> bool:
   465	        if left.type.fallback_to_any and not self.proper_subtype:
   466	            # NOTE: `None` is a *non-subclassable* singleton, therefore no class
   467	            # can by a subtype of it, even with an `Any` fallback.
   468	            # This special case is needed to treat descriptors in classes with
   469	            # dynamic base classes correctly, see #5456.
   470	            return not isinstance(self.right, NoneType)
   471	        right = self.right
   472	        if isinstance(right, TupleType) and right.partial_fallback.type.is_enum:
   473	            return self._is_subtype(left, mypy.typeops.tuple_fallback(right))
   474	        if isinstance(right, TupleType):
   475	            if len(right.items) == 1:
   476	                # Non-normalized Tuple type (may be left after semantic analysis
   477	                # because semanal_typearg visitor is not a type translator).
   478	                item = right.items[0]
   479	                if isinstance(item, UnpackType):
   480	                    unpacked = get_proper_type(item.type)
   481	                    if isinstance(unpacked, Instance):
   482	                        return self._is_subtype(left, unpacked)
   483	            if left.type.has_base(right.partial_fallback.type.fullname):
   484	                if not self.proper_subtype:
   485	                    # Special cases to consider:
   486	                    #   * Plain tuple[Any, ...] instance is a subtype of all tuple types.
   487	                    #   * Foo[*tuple[Any, ...]] (normalized) instance is a subtype of all
   488	                    #     tuples with fallback to Foo (e.g. for variadic NamedTuples).
   489	                    mapped = map_instance_to_supertype(left, right.partial_fallback.type)
   490	                    if is_erased_instance(mapped):
   491	                        if (
   492	                            mapped.type.fullname == "builtins.tuple"
   493	                            or mapped.type.has_type_var_tuple_type
   494	                        ):
   495	                            return True
   496	            return False
   497	        if isinstance(right, TypeVarTupleType):
   498	            # tuple[Any, ...] is like Any in the world of tuples (see special case above).
   499	            if left.type.has_base("builtins.tuple"):
   500	                mapped = map_instance_to_supertype(left, right.tuple_fallback.type)
   501	                if isinstance(get_proper_type(mapped.args[0]), AnyType):
   502	                    return not self.proper_subtype
   503	        if isinstance(right, Instance):
   504	            if type_state.is_cached_subtype_check(self._subtype_kind, left, right):
   505	                return True
   506	            if type_state.is_cached_negative_subtype_check(self._subtype_kind, left, right):
   507	                return False
   508	            if not self.subtype_context.ignore_promotions and not right.type.is_protocol:
   509	                for base in left.type.mro:
   510	                    if base._promote and any(
   511	                        self._is_subtype(p, self.right) for p in base._promote
   512	                    ):
   513	                        type_state.record_subtype_cache_entry(self._subtype_kind, left, right)
   514	                        return True
   515	                # Special case: Low-level integer types are compatible with 'int'. We can't
   516	                # use promotions, since 'int' is already promoted to low-level integer types,
   517	                # and we can't have circular promotions.
   518	                if left.type.alt_promote and left.type.alt_promote.type is right.type:
   519	                    return True
   520	            rname = right.type.fullname
   521	            # Always try a nominal check if possible,
   522	            # there might be errors that a user wants to silence *once*.
   523	            # NamedTuples are a special case, because `NamedTuple` is not listed
   524	            # in `TypeInfo.mro`, so when `(a: NamedTuple) -> None` is used,
   525	            # we need to check for `is_named_tuple` property
   526	            if (
   527	                left.type.has_base(rname)
   528	                or rname == "builtins.object"
   529	                or (
   530	                    rname in TYPED_NAMEDTUPLE_NAMES
   531	                    and any(l.is_named_tuple for l in left.type.mro)
   532	                )
   533	            ) and not self.subtype_context.ignore_declared_variance:
   534	                # Map left type to corresponding right instances.
   535	                t = map_instance_to_supertype(left, right.type)
   536	                if self.subtype_context.erase_instances:
   537	                    erased = erase_type(t)
   538	                    assert isinstance(erased, Instance)
   539	                    t = erased
   540	                nominal = True
   541	                if right.type.has_type_var_tuple_type:
   542	                    # For variadic instances we simply find the correct type argument mappings,
   543	                    # all the heavy lifting is done by the tuple subtyping.
   544	                    assert right.type.type_var_tuple_prefix is not None
   545	                    assert right.type.type_var_tuple_suffix is not None
   546	                    prefix = right.type.type_var_tuple_prefix
   547	                    suffix = right.type.type_var_tuple_suffix
   548	                    tvt = right.type.defn.type_vars[prefix]
   549	                    assert isinstance(tvt, TypeVarTupleType)
   550	                    fallback = tvt.tuple_fallback
   551	                    left_prefix, left_middle, left_suffix = split_with_prefix_and_suffix(
   552	                        t.args, prefix, suffix
   553	                    )
   554	                    right_prefix, right_middle, right_suffix = split_with_prefix_and_suffix(
   555	                        right.args, prefix, suffix
   556	                    )
   557	                    left_args = (
   558	                        left_prefix + (TupleType(list(left_middle), fallback),) + left_suffix
   559	                    )
   560	                    right_args = (
   561	                        right_prefix + (TupleType(list(right_middle), fallback),) + right_suffix
   562	                    )
   563	                    if not self.proper_subtype and is_erased_instance(t):
   564	                        return True
   565	                    if len(left_args) != len(right_args):
   566	                        return False
   567	                    type_params = zip(left_args, right_args, right.type.defn.type_vars)
   568	                else:
   569	                    type_params = zip(t.args, right.args, right.type.defn.type_vars)
   570	                if not self.subtype_context.ignore_type_params:
   571	                    tried_infer = False
   572	                    for lefta, righta, tvar in type_params:
   573	                        if isinstance(tvar, TypeVarType):
   574	                            if tvar.variance == VARIANCE_NOT_READY and not tried_infer:
   575	                                infer_class_variances(right.type)
   576	                                tried_infer = True
   577	                            if (
   578	                                self.subtype_context.always_covariant
   579	                                and tvar.variance == INVARIANT
   580	                            ):
   581	                                variance = COVARIANT
   582	                            else:
   583	                                variance = tvar.variance
   584	                            if not check_type_parameter(
   585	                                lefta, righta, variance, self.proper_subtype, self.subtype_context
   586	                            ):
   587	                                nominal = False
   588	                        else:
   589	                            # TODO: everywhere else ParamSpecs are handled as invariant.
   590	                            if not check_type_parameter(
   591	                                lefta, righta, COVARIANT, self.proper_subtype, self.subtype_context
   592	                            ):
   593	                                nominal = False
   594	                if nominal:
   595	                    type_state.record_subtype_cache_entry(self._subtype_kind, left, right)
   596	                else:
   597	                    type_state.record_negative_subtype_cache_entry(self._subtype_kind, left, right)
   598	                return nominal
   599	            if right.type.is_protocol and is_protocol_implementation(
   600	                left, right, proper_subtype=self.proper_subtype, options=self.options
   601	            ):
   602	                return True
   603	            # We record negative cache entry here, and not in the protocol check like we do for
   604	            # positive cache, to avoid accidentally adding a type that is not a structural
   605	            # subtype, but is a nominal subtype (involving type: ignore override).
   606	            type_state.record_negative_subtype_cache_entry(self._subtype_kind, left, right)
   607	            return False
   608	        if isinstance(right, TypeType):
   609	            item = right.item
   610	            if isinstance(item, TupleType):
   611	                item = mypy.typeops.tuple_fallback(item)
   612	            # TODO: this is a bit arbitrary, we should only skip Any-related cases.
   613	            if not self.proper_subtype:
   614	                if is_named_instance(left, "builtins.type"):
   615	                    return self._is_subtype(TypeType(AnyType(TypeOfAny.special_form)), right)
   616	                if left.type.is_metaclass():
   617	                    if isinstance(item, AnyType):
   618	                        return True
   619	                    if isinstance(item, Instance):
   620	                        return is_named_instance(item, "builtins.object")
   621	        if isinstance(right, LiteralType) and left.last_known_value is not None:
   622	            return self._is_subtype(left.last_known_value, right)
   623	        if isinstance(right, FunctionLike):
   624	            # Special case: Instance can be a subtype of Callable / Overloaded.
   625	            call = find_member("__call__", left, left, is_operator=True)
   626	            if call:
   627	                return self._is_subtype(call, right)
   628	            return False
   629	        else:
   630	            return False
   631
   632	    def visit_type_var(self, left: TypeVarType) -> bool:
   633	        right = self.right
   634	        if isinstance(right, TypeVarType) and left.id == right.id:
   635	            # Fast path for most common case.
   636	            if left.upper_bound == right.upper_bound:
   637	                return True
   638	            # Corner case for self-types in classes generic in type vars
   639	            # with value restrictions.
   640	            if left.id.is_self():
   641	                return True
   642	            return self._is_subtype(left.upper_bound, right.upper_bound)
   643	        if left.values and self._is_subtype(UnionType.make_union(left.values), right):
   644	            return True
   645	        return self._is_subtype(left.upper_bound, self.right)
   646
   647	    def visit_param_spec(self, left: ParamSpecType) -> bool:
   648	        right = self.right
   649	        if (
   650	            isinstance(right, ParamSpecType)
   651	            and right.id == left.id
   652	            and right.flavor == left.flavor
   653	        ):
   654	            return self._is_subtype(left.prefix, right.prefix)
   655	        if isinstance(right, Parameters) and are_trivial_parameters(right):
   656	            return True
   657	        return self._is_subtype(left.upper_bound, self.right)
   658
   659	    def visit_type_var_tuple(self, left: TypeVarTupleType) -> bool:
   660	        right = self.right
   661	        if isinstance(right, TypeVarTupleType) and right.id == left.id:
   662	            return left.min_len >= right.min_len
   663	        return self._is_subtype(left.upper_bound, self.right)
   664
   665	    def visit_unpack_type(self, left: UnpackType) -> bool:
   666	        # TODO: Ideally we should not need this (since it is not a real type).
   667	        # Instead callers (upper level types) should handle it when it appears in type list.
   668	        if isinstance(self.right, UnpackType):
   669	            return self._is_subtype(left.type, self.right.type)
   670	        if isinstance(self.right, Instance) and self.right.type.fullname == "builtins.object":
   671	            return True
   672	        return False
   673
   674	    def visit_parameters(self, left: Parameters) -> bool:
   675	        if isinstance(self.right, Parameters):
   676	            return are_parameters_compatible(
   677	                left,
   678	                self.right,
   679	                is_compat=self._is_subtype,
   680	                # TODO: this should pass the current value, but then couple tests fail.
   681	                is_proper_subtype=False,
   682	                ignore_pos_arg_names=self.subtype_context.ignore_pos_arg_names,
   683	            )
   684	        elif isinstance(self.right, Instance):
   685	            return self.right.type.fullname == "builtins.object"
   686	        else:
   687	            return False
   688
   689	    def visit_callable_type(self, left: CallableType) -> bool:
   690	        right = self.right
   691	        if isinstance(right, CallableType):
   692	            if left.type_guard is not None and right.type_guard is not None:
   693	                if not self._is_subtype(left.type_guard, right.type_guard):
   694	                    return False
   695	            elif left.type_is is not None and right.type_is is not None:
   696	                # For TypeIs we have to check both ways; it is unsafe to pass
   697	                # a TypeIs[Child] when a TypeIs[Parent] is expected, because
   698	                # if the narrower returns False, we assume that the narrowed value is
   699	                # *not* a Parent.
   700	                if not self._is_subtype(left.type_is, right.type_is) or not self._is_subtype(
   701	                    right.type_is, left.type_is
   702	                ):
   703	                    return False
   704	            elif right.type_guard is not None and left.type_guard is None:
   705	                # This means that one function has `TypeGuard` and other does not.
   706	                # They are not compatible. See https://github.com/python/mypy/issues/11307
   707	                return False
   708	            elif right.type_is is not None and left.type_is is None:
   709	                # Similarly, if one function has `TypeIs` and the other does not,
   710	                # they are not compatible.
   711	                return False
   712	            return is_callable_compatible(
   713	                left,
   714	                right,
   715	                is_compat=self._is_subtype,
   716	                is_proper_subtype=self.proper_subtype,
   717	                ignore_pos_arg_names=self.subtype_context.ignore_pos_arg_names,
   718	                strict_concatenate=(
   719	                    (self.options.extra_checks or self.options.strict_concatenate)
   720	                    if self.options
   721	                    else False
   722	                ),
   723	            )
   724	        elif isinstance(right, Overloaded):
   725	            return all(self._is_subtype(left, item) for item in right.items)
   726	        elif isinstance(right, Instance):
   727	            if right.type.is_protocol and "__call__" in right.type.protocol_members:
   728	                # OK, a callable can implement a protocol with a `__call__` member.
   729	                call = find_member("__call__", right, right, is_operator=True)
   730	                assert call is not None
   731	                if self._is_subtype(left, call):
   732	                    if len(right.type.protocol_members) == 1:
   733	                        return True
   734	                    if is_protocol_implementation(left.fallback, right, skip=["__call__"]):
   735	                        return True
   736	            if right.type.is_protocol and left.is_type_obj():
   737	                ret_type = get_proper_type(left.ret_type)
   738	                if isinstance(ret_type, TupleType):
   739	                    ret_type = mypy.typeops.tuple_fallback(ret_type)
   740	                if isinstance(ret_type, Instance) and is_protocol_implementation(
   741	                    ret_type, right, proper_subtype=self.proper_subtype, class_obj=True
   742	                ):
   743	                    return True
   744	            return self._is_subtype(left.fallback, right)
   745	        elif isinstance(right, TypeType):
   746	            # This is unsound, we don't check the __init__ signature.
   747	            return left.is_type_obj() and self._is_subtype(left.ret_type, right.item)
   748	        else:
   749	            return False

   751	    def visit_tuple_type(self, left: TupleType) -> bool:
   752	        right = self.right
   753	        if isinstance(right, Instance):
   754	            if is_named_instance(right, "typing.Sized"):
   755	                return True
   756	            elif is_named_instance(right, TUPLE_LIKE_INSTANCE_NAMES):
   757	                if right.args:
   758	                    iter_type = right.args[0]
   759	                else:
   760	                    if self.proper_subtype:
   761	                        return False
   762	                    iter_type = AnyType(TypeOfAny.special_form)
   763	                if is_named_instance(right, "builtins.tuple") and isinstance(
   764	                    get_proper_type(iter_type), AnyType
   765	                ):
   766	                    # TODO: We shouldn't need this special case. This is currently needed
   767	                    #       for isinstance(x, tuple), though it's unclear why.
   768	                    return True
   769	                for li in left.items:
   770	                    if isinstance(li, UnpackType):
   771	                        unpack = get_proper_type(li.type)
   772	                        if isinstance(unpack, TypeVarTupleType):
   773	                            unpack = get_proper_type(unpack.upper_bound)
   774	                        assert (
   775	                            isinstance(unpack, Instance)
   776	                            and unpack.type.fullname == "builtins.tuple"
   777	                        )
   778	                        li = unpack.args[0]
   779	                    if not self._is_subtype(li, iter_type):
   780	                        return False
   781	                return True
   782	            elif self._is_subtype(left.partial_fallback, right) and self._is_subtype(
   783	                mypy.typeops.tuple_fallback(left), right
   784	            ):
   785	                return True
   786	            return False
   787	        elif isinstance(right, TupleType):
   788	            # If right has a variadic unpack this needs special handling. If there is a TypeVarTuple
   789	            # unpack, item count must coincide. If the left has variadic unpack but right
   790	            # doesn't have one, we will fall through to False down the line.
   791	            if self.variadic_tuple_subtype(left, right):
   792	                return True
   793	            if len(left.items) != len(right.items):
   794	                return False
   795	            if any(not self._is_subtype(l, r) for l, r in zip(left.items, right.items)):
   796	                return False
   797	            if is_named_instance(right.partial_fallback, "builtins.tuple"):
   798	                # No need to verify fallback. This is useful since the calculated fallback
   799	                # may be inconsistent due to how we calculate joins between unions vs.
   800	                # non-unions. For example, join(int, str) == object, whereas
   801	                # join(Union[int, C], Union[str, C]) == Union[int, str, C].
   802	                return True
   803	            if is_named_instance(left.partial_fallback, "builtins.tuple"):
   804	                # Again, no need to verify. At this point we know the right fallback
   805	                # is a subclass of tuple, so if left is plain tuple, it cannot be a subtype.
   806	                return False
   807	            # At this point we know both fallbacks are non-tuple.
   808	            return self._is_subtype(left.partial_fallback, right.partial_fallback)
   809	        else:
   810	            return False

   812	    def variadic_tuple_subtype(self, left: TupleType, right: TupleType) -> bool:
   813	        """Check subtyping between two potentially variadic tuples.
   814
   815	        Most non-trivial cases here are due to variadic unpacks like *tuple[X, ...],
   816	        we handle such unpacks as infinite unions Tuple[()] | Tuple[X] | Tuple[X, X] | ...
   817
   818	        Note: the cases where right is fixed or has *Ts unpack should be handled
   819	        by the caller.
   820	        """
   821	        right_unpack_index = find_unpack_in_list(right.items)
   822	        if right_unpack_index is None:
   823	            # This case should be handled by the caller.
   824	            return False
   825	        right_unpack = right.items[right_unpack_index]
   826	        assert isinstance(right_unpack, UnpackType)
   827	        right_unpacked = get_proper_type(right_unpack.type)
   828	        if not isinstance(right_unpacked, Instance):
   829	            # This case should be handled by the caller.
   830	            return False
   831	        assert right_unpacked.type.fullname == "builtins.tuple"
   832	        right_item = right_unpacked.args[0]
   833	        right_prefix = right_unpack_index
   834	        right_suffix = len(right.items) - right_prefix - 1
   835	        left_unpack_index = find_unpack_in_list(left.items)
   836	        if left_unpack_index is None:
   837	            # Simple case: left is fixed, simply find correct mapping to the right
   838	            # (effectively selecting item with matching length from an infinite union).
   839	            if len(left.items) < right_prefix + right_suffix:
   840	                return False
   841	            prefix, middle, suffix = split_with_prefix_and_suffix(
   842	                tuple(left.items), right_prefix, right_suffix
   843	            )
   844	            if not all(
   845	                self._is_subtype(li, ri) for li, ri in zip(prefix, right.items[:right_prefix])
   846	            ):
   847	                return False
   848	            if right_suffix and not all(
   849	                self._is_subtype(li, ri) for li, ri in zip(suffix, right.items[-right_suffix:])
   850	            ):
   851	                return False
   852	            return all(self._is_subtype(li, right_item) for li in middle)
   853	        else:
   854	            if len(left.items) < len(right.items):
   855	                # There are some items on the left that will never have a matching length
   856	                # on the right.
   857	                return False
   858	            left_prefix = left_unpack_index
   859	            left_suffix = len(left.items) - left_prefix - 1
   860	            left_unpack = left.items[left_unpack_index]
   861	            assert isinstance(left_unpack, UnpackType)
   862	            left_unpacked = get_proper_type(left_unpack.type)
   863	            if not isinstance(left_unpacked, Instance):
   864	                # *Ts unpack can't be split, except if it is all mapped to Anys or objects.
   865	                if self.is_top_type(right_item):
   866	                    right_prefix_types, middle, right_suffix_types = split_with_prefix_and_suffix(
   867	                        tuple(right.items), left_prefix, left_suffix
   868	                    )
   869	                    if not all(
   870	                        self.is_top_type(ri) or isinstance(ri, UnpackType) for ri in middle
   871	                    ):
   872	                        return False
   873	                    # Also check the tails match as well.
   874	                    return self._all_subtypes(
   875	                        left.items[:left_prefix], right_prefix_types
   876	                    ) and self._all_subtypes(left.items[-left_suffix:], right_suffix_types)
   877	                return False
   878	            assert left_unpacked.type.fullname == "builtins.tuple"
   879	            left_item = left_unpacked.args[0]
   880
   881	            # The most tricky case with two variadic unpacks we handle similar to union
   882	            # subtyping: *each* item on the left, must be a subtype of *some* item on the right.
   883	            # For this we first check the "asymptotic case", i.e. that both unpacks a subtypes,
   884	            # and then check subtyping for all finite overlaps.
   885	            if not self._is_subtype(left_item, right_item):
   886	                return False
   887	            max_overlap = max(0, right_prefix - left_prefix, right_suffix - left_suffix)
   888	            for overlap in range(max_overlap + 1):
   889	                repr_items = left.items[:left_prefix] + [left_item] * overlap
   890	                if left_suffix:
   891	                    repr_items += left.items[-left_suffix:]
   892	                left_repr = left.copy_modified(items=repr_items)
   893	                if not self._is_subtype(left_repr, right):
   894	                    return False
   895	            return True
   896
   897	    def is_top_type(self, typ: Type) -> bool:
   898	        if not self.proper_subtype and isinstance(get_proper_type(typ), AnyType):
   899	            return True
   900	        return is_named_instance(typ, "builtins.object")

   902	    def visit_typeddict_type(self, left: TypedDictType) -> bool:
   903	        right = self.right
   904	        if isinstance(right, Instance):
   905	            return self._is_subtype(left.fallback, right)
   906	        elif isinstance(right, TypedDictType):
   907	            if left == right:
   908	                return True  # Fast path
   909	            if not left.names_are_wider_than(right):
   910	                return False
   911	            for name, l, r in left.zip(right):
   912	                # TODO: should we pass on the full subtype_context here and below?
   913	                right_readonly = name in right.readonly_keys
   914	                if not right_readonly:
   915	                    if self.proper_subtype:
   916	                        check = is_same_type(l, r)
   917	                    else:
   918	                        check = is_equivalent(
   919	                            l,
   920	                            r,
   921	                            ignore_type_params=self.subtype_context.ignore_type_params,
   922	                            options=self.options,
   923	                        )
   924	                else:
   925	                    # Read-only items behave covariantly
   926	                    check = self._is_subtype(l, r)
   927	                if not check:
   928	                    return False
   929	                # Non-required key is not compatible with a required key since
   930	                # indexing may fail unexpectedly if a required key is missing.
   931	                # Required key is not compatible with a non-read-only non-required
   932	                # key since the prior doesn't support 'del' but the latter should
   933	                # support it.
   934	                # Required key is compatible with a read-only non-required key.
   935	                required_differ = (name in left.required_keys) != (name in right.required_keys)
   936	                if not right_readonly and required_differ:
   937	                    return False
   938	                # Readonly fields check:
   939	                #
   940	                # A = TypedDict('A', {'x': ReadOnly[int]})
   941	                # B = TypedDict('B', {'x': int})
   942	                # def reset_x(b: B) -> None:
   943	                #     b['x'] = 0
   944	                #
   945	                # So, `A` cannot be a subtype of `B`, while `B` can be a subtype of `A`,
   946	                # because you can use `B` everywhere you use `A`, but not the other way around.
   947	                if name in left.readonly_keys and name not in right.readonly_keys:
   948	                    return False
   949	            # (NOTE: Fallbacks don't matter.)
   950	            return True
   951	        else:
   952	            return False

   960	    def visit_overloaded(self, left: Overloaded) -> bool:
   961	        right = self.right
   962	        if isinstance(right, Instance):
   963	            if right.type.is_protocol and "__call__" in right.type.protocol_members:
   964	                # same as for CallableType
   965	                call = find_member("__call__", right, right, is_operator=True)
   966	                assert call is not None
   967	                if self._is_subtype(left, call):
   968	                    if len(right.type.protocol_members) == 1:
   969	                        return True
   970	                    if is_protocol_implementation(left.fallback, right, skip=["__call__"]):
   971	                        return True
   972	            return self._is_subtype(left.fallback, right)
   973	        elif isinstance(right, CallableType):
   974	            for item in left.items:
   975	                if self._is_subtype(item, right):
   976	                    return True
   977	            return False
   978	        elif isinstance(right, Overloaded):
   979	            if left == self.right:
   980	                # When it is the same overload, then the types are equal.
   981	                return True
   982
   983	            # Ensure each overload on the right side (the supertype) is accounted for.
   984	            previous_match_left_index = -1
   985	            matched_overloads = set()
   986
   987	            for right_item in right.items:
   988	                found_match = False
   989
   990	                for left_index, left_item in enumerate(left.items):
   991	                    subtype_match = self._is_subtype(left_item, right_item)
   992
   993	                    # Order matters: we need to make sure that the index of
   994	                    # this item is at least the index of the previous one.
   995	                    if subtype_match and previous_match_left_index <= left_index:
   996	                        previous_match_left_index = left_index
   997	                        found_match = True
   998	                        matched_overloads.add(left_index)
   999	                        break
  1000	                    else:
  1001	                        # If this one overlaps with the supertype in any way, but it wasn't
  1002	                        # an exact match, then it's a potential error.
  1003	                        strict_concat = (
  1004	                            (self.options.extra_checks or self.options.strict_concatenate)
  1005	                            if self.options
  1006	                            else False
  1007	                        )
  1008	                        if left_index not in matched_overloads and (
  1009	                            is_callable_compatible(
  1010	                                left_item,
  1011	                                right_item,
  1012	                                is_compat=self._is_subtype,
  1013	                                is_proper_subtype=self.proper_subtype,
  1014	                                ignore_return=True,
  1015	                                ignore_pos_arg_names=self.subtype_context.ignore_pos_arg_names,
  1016	                                strict_concatenate=strict_concat,
  1017	                            )
  1018	                            or is_callable_compatible(
  1019	                                right_item,
  1020	                                left_item,
  1021	                                is_compat=self._is_subtype,
  1022	                                is_proper_subtype=self.proper_subtype,
  1023	                                ignore_return=True,
  1024	                                ignore_pos_arg_names=self.subtype_context.ignore_pos_arg_names,
  1025	                                strict_concatenate=strict_concat,
  1026	                            )
  1027	                        ):
  1028	                            return False
  1029
  1030	                if not found_match:
  1031	                    return False
  1032	            return True
  1033	        elif isinstance(right, UnboundType):
  1034	            return True
  1035	        elif isinstance(right, TypeType):
  1036	            # All the items must have the same type object status, so
  1037	            # it's sufficient to query only (any) one of them.
  1038	            # This is unsound, we don't check all the __init__ signatures.
  1039	            return left.is_type_obj() and self._is_subtype(left.items[0], right)
  1040	        else:
  1041	            return False

  1043	    def visit_union_type(self, left: UnionType) -> bool:
  1044	        if isinstance(self.right, Instance):
  1045	            literal_types: set[Instance] = set()
  1046	            # avoid redundant check for union of literals
  1047	            for item in left.relevant_items():
  1048	                p_item = get_proper_type(item)
  1049	                lit_type = mypy.typeops.simple_literal_type(p_item)
  1050	                if lit_type is not None:
  1051	                    if lit_type in literal_types:
  1052	                        continue
  1053	                    literal_types.add(lit_type)
  1054	                    item = lit_type
  1055	                if not self._is_subtype(item, self.orig_right):
  1056	                    return False
  1057	            return True
  1058
  1059	        elif isinstance(self.right, UnionType):
  1060	            # prune literals early to avoid nasty quadratic behavior which would otherwise arise when checking
  1061	            # subtype relationships between slightly different narrowings of an Enum
  1062	            # we achieve O(N+M) instead of O(N*M)
  1063
  1064	            fast_check: set[ProperType] = set()
  1065
  1066	            for item in flatten_types(self.right.relevant_items()):
  1067	                p_item = get_proper_type(item)
  1068	                fast_check.add(p_item)
  1069	                if isinstance(p_item, Instance) and p_item.last_known_value is not None:
  1070	                    fast_check.add(p_item.last_known_value)
  1071
  1072	            for item in left.relevant_items():
  1073	                p_item = get_proper_type(item)
  1074	                if p_item in fast_check:
  1075	                    continue
  1076	                lit_type = mypy.typeops.simple_literal_type(p_item)
  1077	                if lit_type in fast_check:
  1078	                    continue
  1079	                if not self._is_subtype(item, self.orig_right):
  1080	                    return False
  1081	            return True
  1082
  1083	        return all(self._is_subtype(item, self.orig_right) for item in left.items)

  1147	def is_protocol_implementation(
  1148	    left: Instance,
  1149	    right: Instance,
  1150	    proper_subtype: bool = False,
  1151	    class_obj: bool = False,
  1152	    skip: list[str] | None = None,
  1153	    options: Options | None = None,
  1154	) -> bool:
  1155	    """Check whether 'left' implements the protocol 'right'.
  1156
  1157	    If 'proper_subtype' is True, then check for a proper subtype.
  1158	    Treat recursive protocols by using the 'assuming' structural subtype matrix
  1159	    (in sparse representation, i.e. as a list of pairs (subtype, supertype)),
  1160	    see also comment in nodes.TypeInfo. When we enter a check for classes
  1161	    (A, P), defined as following::
  1162
  1163	      class P(Protocol):
  1164	          def f(self) -> P: ...
  1165	      class A:
  1166	          def f(self) -> A: ...
  1167
  1168	    this results in A being a subtype of P without infinite recursion.
  1169	    On every false result, we pop the assumption, thus avoiding an infinite recursion
  1170	    as well.
  1171	    """
  1172	    assert right.type.is_protocol
  1173	    if skip is None:
  1174	        skip = []
  1175	    # We need to record this check to generate protocol fine-grained dependencies.
  1176	    type_state.record_protocol_subtype_check(left.type, right.type)
  1177	    # nominal subtyping currently ignores '__init__' and '__new__' signatures
  1178	    members_not_to_check = {"__init__", "__new__"}
  1179	    members_not_to_check.update(skip)
  1180	    # Trivial check that circumvents the bug described in issue 9771:
  1181	    if left.type.is_protocol:
  1182	        members_right = set(right.type.protocol_members) - members_not_to_check
  1183	        members_left = set(left.type.protocol_members) - members_not_to_check
  1184	        if not members_right.issubset(members_left):
  1185	            return False
  1186	    assuming = right.type.assuming_proper if proper_subtype else right.type.assuming
  1187	    for l, r in reversed(assuming):
  1188	        if l == left and r == right:
  1189	            return True
  1190	    with pop_on_exit(assuming, left, right):
  1191	        for member in right.type.protocol_members:
  1192	            if member in members_not_to_check:
  1193	                continue
  1194	            ignore_names = member != "__call__"  # __call__ can be passed kwargs
  1195	            # The third argument below indicates to what self type is bound.
  1196	            # We always bind self to the subtype. (Similarly to nominal types).
  1197	            supertype = find_member(member, right, left)
  1198	            assert supertype is not None
  1199
  1200	            subtype = mypy.typeops.get_protocol_member(left, member, class_obj)
  1201	            # Useful for debugging:
  1202	            # print(member, 'of', left, 'has type', subtype)
  1203	            # print(member, 'of', right, 'has type', supertype)
  1204	            if not subtype:
  1205	                return False
  1206	            if not proper_subtype:
  1207	                # Nominal check currently ignores arg names
  1208	                # NOTE: If we ever change this, be sure to also change the call to
  1209	                # SubtypeVisitor.build_subtype_kind(...) down below.
  1210	                is_compat = is_subtype(
  1211	                    subtype, supertype, ignore_pos_arg_names=ignore_names, options=options
  1212	                )
  1213	            else:
  1214	                is_compat = is_proper_subtype(subtype, supertype)
  1215	            if not is_compat:
  1216	                return False
  1217	            if isinstance(get_proper_type(subtype), NoneType) and isinstance(
  1218	                get_proper_type(supertype), CallableType
  1219	            ):
  1220	                # We want __hash__ = None idiom to work even without --strict-optional
  1221	                return False
  1222	            subflags = get_member_flags(member, left, class_obj=class_obj)
  1223	            superflags = get_member_flags(member, right)
  1224	            if IS_SETTABLE in superflags:
  1225	                # Check opposite direction for settable attributes.
  1226	                if IS_EXPLICIT_SETTER in superflags:
  1227	                    supertype = find_member(member, right, left, is_lvalue=True)
  1228	                if IS_EXPLICIT_SETTER in subflags:
  1229	                    subtype = mypy.typeops.get_protocol_member(
  1230	                        left, member, class_obj, is_lvalue=True
  1231	                    )
  1232	                # At this point we know attribute is present on subtype, otherwise we
  1233	                # would return False above.
  1234	                assert supertype is not None and subtype is not None
  1235	                if not is_subtype(supertype, subtype, options=options):
  1236	                    return False
  1237	            if IS_SETTABLE in superflags and IS_SETTABLE not in subflags:
  1238	                return False
  1239	            if not class_obj:
  1240	                if IS_SETTABLE not in superflags:
  1241	                    if IS_CLASSVAR in superflags and IS_CLASSVAR not in subflags:
  1242	                        return False
  1243	                elif (IS_CLASSVAR in subflags) != (IS_CLASSVAR in superflags):
  1244	                    return False
  1245	            else:
  1246	                if IS_VAR in superflags and IS_CLASSVAR not in subflags:
  1247	                    # Only class variables are allowed for class object access.
  1248	                    return False
  1249	                if IS_CLASSVAR in superflags:
  1250	                    # This can be never matched by a class object.
  1251	                    return False
  1252	            # This rule is copied from nominal check in checker.py
  1253	            if IS_CLASS_OR_STATIC in superflags and IS_CLASS_OR_STATIC not in subflags:
  1254	                return False
  1255
  1256	    if not proper_subtype:
  1257	        # Nominal check currently ignores arg names, but __call__ is special for protocols
  1258	        ignore_names = right.type.protocol_members != ["__call__"]
  1259	    else:
  1260	        ignore_names = False
  1261	    subtype_kind = SubtypeVisitor.build_subtype_kind(
  1262	        subtype_context=SubtypeContext(ignore_pos_arg_names=ignore_names),
  1263	        proper_subtype=proper_subtype,
  1264	    )
  1265	    type_state.record_subtype_cache_entry(subtype_kind, left, right)
  1266	    return True

  1269	def find_member(
  1270	    name: str,
  1271	    itype: Instance,
  1272	    subtype: Type,
  1273	    *,
  1274	    is_operator: bool = False,
  1275	    class_obj: bool = False,
  1276	    is_lvalue: bool = False,
  1277	) -> Type | None:
  1278	    type_checker = checker_state.type_checker
  1279	    if type_checker is None:
  1280	        # Unfortunately, there are many scenarios where someone calls is_subtype() before
  1281	        # type checking phase. In this case we fallback to old (incomplete) logic.
  1282	        # TODO: reduce number of such cases (e.g. semanal_typeargs, post-semanal plugins).
  1283	        return find_member_simple(
  1284	            name, itype, subtype, is_operator=is_operator, class_obj=class_obj, is_lvalue=is_lvalue
  1285	        )
  1286
  1287	    # We don't use ATTR_DEFINED error code below (since missing attributes can cause various
  1288	    # other error codes), instead we perform quick node lookup with all the fallbacks.
  1289	    info = itype.type
  1290	    sym = info.get(name)
  1291	    node = sym.node if sym else None
  1292	    if not node:
  1293	        name_not_found = True
  1294	        if (
  1295	            name not in ["__getattr__", "__setattr__", "__getattribute__"]
  1296	            and not is_operator
  1297	            and not class_obj
  1298	            and itype.extra_attrs is None  # skip ModuleType.__getattr__
  1299	        ):
  1300	            for method_name in ("__getattribute__", "__getattr__"):
  1301	                method = info.get_method(method_name)
  1302	                if method and method.info.fullname != "builtins.object":
  1303	                    name_not_found = False
  1304	                    break
  1305	        if name_not_found:
  1306	            if info.fallback_to_any or class_obj and info.meta_fallback_to_any:
  1307	                return AnyType(TypeOfAny.special_form)
  1308	            if itype.extra_attrs and name in itype.extra_attrs.attrs:
  1309	                return itype.extra_attrs.attrs[name]
  1310	            return None
  1311
  1312	    from mypy.checkmember import (
  1313	        MemberContext,
  1314	        analyze_class_attribute_access,
  1315	        analyze_instance_member_access,
  1316	    )
  1317
  1318	    mx = MemberContext(
  1319	        is_lvalue=is_lvalue,
  1320	        is_super=False,
  1321	        is_operator=is_operator,
  1322	        original_type=TypeType.make_normalized(itype) if class_obj else itype,
  1323	        self_type=TypeType.make_normalized(subtype) if class_obj else subtype,
  1324	        context=Context(),  # all errors are filtered, but this is a required argument
  1325	        chk=type_checker,
  1326	        suppress_errors=True,
  1327	        # This is needed to avoid infinite recursion in situations involving protocols like
  1328	        #     class P(Protocol[T]):
  1329	        #         def combine(self, other: P[S]) -> P[Tuple[T, S]]: ...
  1330	        # Normally we call freshen_all_functions_type_vars() during attribute access,
  1331	        # to avoid type variable id collisions, but for protocols this means we can't
  1332	        # use the assumption stack, that will grow indefinitely.
  1333	        # TODO: find a cleaner solution that doesn't involve massive perf impact.
  1334	        preserve_type_var_ids=True,
  1335	    )
  1336	    with type_checker.msg.filter_errors(filter_deprecated=True):
  1337	        if class_obj:
  1338	            fallback = itype.type.metaclass_type or mx.named_type("builtins.type")
  1339	            return analyze_class_attribute_access(itype, name, mx, mcs_fallback=fallback)
  1340	        else:
  1341	            return analyze_instance_member_access(name, itype, mx, info)

  1344	def find_member_simple(
  1345	    name: str,
  1346	    itype: Instance,
  1347	    subtype: Type,
  1348	    *,
  1349	    is_operator: bool = False,
  1350	    class_obj: bool = False,
  1351	    is_lvalue: bool = False,
  1352	) -> Type | None:
  1353	    """Find the type of member by 'name' in 'itype's TypeInfo.
  1354
  1355	    Find the member type after applying type arguments from 'itype', and binding
  1356	    'self' to 'subtype'. Return None if member was not found.
  1357	    """
  1358	    info = itype.type
  1359	    method = info.get_method(name)
  1360	    if method:
  1361	        if isinstance(method, Decorator):
  1362	            return find_node_type(method.var, itype, subtype, class_obj=class_obj)
  1363	        if method.is_property:
  1364	            assert isinstance(method, OverloadedFuncDef)
  1365	            dec = method.items[0]
  1366	            assert isinstance(dec, Decorator)
  1367	            # Pass on is_lvalue flag as this may be a property with different setter type.
  1368	            return find_node_type(
  1369	                dec.var, itype, subtype, class_obj=class_obj, is_lvalue=is_lvalue
  1370	            )
  1371	        return find_node_type(method, itype, subtype, class_obj=class_obj)
  1372	    else:
  1373	        # don't have such method, maybe variable or decorator?
  1374	        node = info.get(name)
  1375	        v = node.node if node else None
  1376	        if isinstance(v, Var):
  1377	            return find_node_type(v, itype, subtype, class_obj=class_obj)
  1378	        if (
  1379	            not v
  1380	            and name not in ["__getattr__", "__setattr__", "__getattribute__"]
  1381	            and not is_operator
  1382	            and not class_obj
  1383	            and itype.extra_attrs is None  # skip ModuleType.__getattr__
  1384	        ):
  1385	            for method_name in ("__getattribute__", "__getattr__"):
  1386	                # Normally, mypy assumes that instances that define __getattr__ have all
  1387	                # attributes with the corresponding return type. If this will produce
  1388	                # many false negatives, then this could be prohibited for
  1389	                # structural subtyping.
  1390	                method = info.get_method(method_name)
  1391	                if method and method.info.fullname != "builtins.object":
  1392	                    if isinstance(method, Decorator):
  1393	                        getattr_type = get_proper_type(find_node_type(method.var, itype, subtype))
  1394	                    else:
  1395	                        getattr_type = get_proper_type(find_node_type(method, itype, subtype))
  1396	                    if isinstance(getattr_type, CallableType):
  1397	                        return getattr_type.ret_type
  1398	                    return getattr_type
  1399	        if itype.type.fallback_to_any or class_obj and itype.type.meta_fallback_to_any:
  1400	            return AnyType(TypeOfAny.special_form)
  1401	        if isinstance(v, TypeInfo):
  1402	            # PEP 544 doesn't specify anything about such use cases. So we just try
  1403	            # to do something meaningful (at least we should not crash).
  1404	            return TypeType(fill_typevars_with_any(v))
  1405	    if itype.extra_attrs and name in itype.extra_attrs.attrs:
  1406	        return itype.extra_attrs.attrs[name]
  1407	    return None

  1410	def get_member_flags(name: str, itype: Instance, class_obj: bool = False) -> set[int]:
  1411	    """Detect whether a member 'name' is settable, whether it is an
  1412	    instance or class variable, and whether it is class or static method.
  1413
  1414	    The flags are defined as following:
  1415	    * IS_SETTABLE: whether this attribute can be set, not set for methods and
  1416	      non-settable properties;
  1417	    * IS_CLASSVAR: set if the variable is annotated as 'x: ClassVar[t]';
  1418	    * IS_CLASS_OR_STATIC: set for methods decorated with @classmethod or
  1419	      with @staticmethod.
  1420	    """
  1421	    info = itype.type
  1422	    method = info.get_method(name)
  1423	    setattr_meth = info.get_method("__setattr__")
  1424	    if method:
  1425	        if isinstance(method, Decorator):
  1426	            if method.var.is_staticmethod or method.var.is_classmethod:
  1427	                return {IS_CLASS_OR_STATIC}
  1428	            elif method.var.is_property:
  1429	                return {IS_VAR}
  1430	        elif method.is_property:  # this could be settable property
  1431	            assert isinstance(method, OverloadedFuncDef)
  1432	            dec = method.items[0]
  1433	            assert isinstance(dec, Decorator)
  1434	            if dec.var.is_settable_property or setattr_meth:
  1435	                flags = {IS_VAR, IS_SETTABLE}
  1436	                if dec.var.setter_type is not None:
  1437	                    flags.add(IS_EXPLICIT_SETTER)
  1438	                return flags
  1439	            else:
  1440	                return {IS_VAR}
  1441	        return set()  # Just a regular method
  1442	    node = info.get(name)
  1443	    if not node:
  1444	        if setattr_meth:
  1445	            return {IS_SETTABLE}
  1446	        if itype.extra_attrs and name in itype.extra_attrs.attrs:
  1447	            flags = set()
  1448	            if name not in itype.extra_attrs.immutable:
  1449	                flags.add(IS_SETTABLE)
  1450	            return flags
  1451	        return set()
  1452	    v = node.node
  1453	    # just a variable
  1454	    if isinstance(v, Var):
  1455	        if v.is_property:
  1456	            return {IS_VAR}
  1457	        flags = {IS_VAR}
  1458	        if not v.is_final:
  1459	            flags.add(IS_SETTABLE)
  1460	        # TODO: define cleaner rules for class vs instance variables.
  1461	        if v.is_classvar and not is_descriptor(v.type):
  1462	            flags.add(IS_CLASSVAR)
  1463	        if class_obj and v.is_inferred:
  1464	            flags.add(IS_CLASSVAR)
  1465	        return flags
  1466	    return set()

  1478	def find_node_type(
  1479	    node: Var | FuncBase,
  1480	    itype: Instance,
  1481	    subtype: Type,
  1482	    class_obj: bool = False,
  1483	    is_lvalue: bool = False,
  1484	) -> Type:
  1485	    """Find type of a variable or method 'node' (maybe also a decorated method).
  1486	    Apply type arguments from 'itype', and bind 'self' to 'subtype'.
  1487	    """
  1488	    from mypy.typeops import bind_self
  1489
  1490	    if isinstance(node, FuncBase):
  1491	        typ: Type | None = mypy.typeops.function_type(
  1492	            node, fallback=Instance(itype.type.mro[-1], [])
  1493	        )
  1494	    else:
  1495	        # This part and the one below are simply copies of the logic from checkmember.py.
  1496	        if node.is_settable_property and is_lvalue:
  1497	            typ = node.setter_type
  1498	            if typ is None and node.is_ready:
  1499	                typ = node.type
  1500	        else:
  1501	            typ = node.type
  1502	        if typ is not None:
  1503	            typ = expand_self_type(node, typ, subtype)
  1504	    p_typ = get_proper_type(typ)
  1505	    if typ is None:
  1506	        return AnyType(TypeOfAny.from_error)
  1507	    # We don't need to bind 'self' for static methods, since there is no 'self'.
  1508	    if isinstance(node, FuncBase) or (
  1509	        isinstance(p_typ, FunctionLike)
  1510	        and node.is_initialized_in_class
  1511	        and not node.is_staticmethod
  1512	    ):
  1513	        assert isinstance(p_typ, FunctionLike)
  1514	        if class_obj and not (
  1515	            node.is_class if isinstance(node, FuncBase) else node.is_classmethod
  1516	        ):
  1517	            # Don't bind instance methods on class objects.
  1518	            signature = p_typ
  1519	        else:
  1520	            signature = bind_self(
  1521	                p_typ, subtype, is_classmethod=isinstance(node, Var) and node.is_classmethod
  1522	            )
  1523	        if node.is_property and not class_obj:
  1524	            assert isinstance(signature, CallableType)
  1525	            if (
  1526	                isinstance(node, Var)
  1527	                and node.is_settable_property
  1528	                and is_lvalue
  1529	                and node.setter_type is not None
  1530	            ):
  1531	                typ = signature.arg_types[0]
  1532	            else:
  1533	                typ = signature.ret_type
  1534	        else:
  1535	            typ = signature
  1536	    itype = map_instance_to_supertype(itype, node.info)
  1537	    typ = expand_type_by_instance(typ, itype)
  1538	    return typ

  1556	def is_callable_compatible(
  1557	    left: CallableType,
  1558	    right: CallableType,
  1559	    *,
  1560	    is_compat: Callable[[Type, Type], bool],
  1561	    is_proper_subtype: bool,
  1562	    is_compat_return: Callable[[Type, Type], bool] | None = None,
  1563	    ignore_return: bool = False,
  1564	    ignore_pos_arg_names: bool = False,
  1565	    check_args_covariantly: bool = False,
  1566	    allow_partial_overlap: bool = False,
  1567	    strict_concatenate: bool = False,
  1568	) -> bool:
  1569	    """Is the left compatible with the right, using the provided compatibility check?
  1570
  1571	    is_compat:
  1572	        The check we want to run against the parameters.
  1573
  1574	    is_compat_return:
  1575	        The check we want to run against the return type.
  1576	        If None, use the 'is_compat' check.
  1577
  1578	    check_args_covariantly:
  1579	        If true, check if the left's args is compatible with the right's
  1580	        instead of the other way around (contravariantly).
  1581
  1582	        This function is mostly used to check if the left is a subtype of the right which
  1583	        is why the default is to check the args contravariantly. However, it's occasionally
  1584	        useful to check the args using some other check, so we leave the variance
  1585	        configurable.
  1586
  1587	        For example, when checking the validity of overloads, it's useful to see if
  1588	        the first overload alternative has more precise arguments than the second.
  1589	        We would want to check the arguments covariantly in that case.
  1590
  1591	        Note! The following two function calls are NOT equivalent:
  1592
  1593	            is_callable_compatible(f, g, is_compat=is_subtype, check_args_covariantly=False)
  1594	            is_callable_compatible(g, f, is_compat=is_subtype, check_args_covariantly=True)
  1595
  1596	        The two calls are similar in that they both check the function arguments in
  1597	        the same direction: they both run `is_subtype(argument_from_g, argument_from_f)`.
  1598
  1599	        However, the two calls differ in which direction they check things like
  1600	        keyword arguments. For example, suppose f and g are defined like so:
  1601
  1602	            def f(x: int, *y: int) -> int: ...
  1603	            def g(x: int) -> int: ...
  1604
  1605	        In this case, the first call will succeed and the second will fail: f is a
  1606	        valid stand-in for g but not vice-versa.
  1607
  1608	    allow_partial_overlap:
  1609	        By default this function returns True if and only if *all* calls to left are
  1610	        also calls to right (with respect to the provided 'is_compat' function).
  1611
  1612	        If this parameter is set to 'True', we return True if *there exists at least one*
  1613	        call to left that's also a call to right.
  1614
  1615	        In other words, we perform an existential check instead of a universal one;
  1616	        we require left to only overlap with right instead of being a subset.
  1617
  1618	        For example, suppose we set 'is_compat' to some subtype check and compare following:
  1619
  1620	            f(x: float, y: str = "...", *args: bool) -> str
  1621	            g(*args: int) -> str
  1622
  1623	        This function would normally return 'False': f is not a subtype of g.
  1624	        However, we would return True if this parameter is set to 'True': the two
  1625	        calls are compatible if the user runs "f_or_g(3)". In the context of that
  1626	        specific call, the two functions effectively have signatures of:
  1627
  1628	            f2(float) -> str
  1629	            g2(int) -> str
  1630
  1631	        Here, f2 is a valid subtype of g2 so we return True.
  1632
  1633	        Specifically, if this parameter is set this function will:
  1634
  1635	        -   Ignore optional arguments on either the left or right that have no
  1636	            corresponding match.
  1637	        -   No longer mandate optional arguments on either side are also optional
  1638	            on the other.
  1639	        -   No longer mandate that if right has a *arg or **kwarg that left must also
  1640	            have the same.
  1641
  1642	        Note: when this argument is set to True, this function becomes "symmetric" --
  1643	        the following calls are equivalent:
  1644
  1645	            is_callable_compatible(f, g,
  1646	                                   is_compat=some_check,
  1647	                                   check_args_covariantly=False,
  1648	                                   allow_partial_overlap=True)
  1649	            is_callable_compatible(g, f,
  1650	                                   is_compat=some_check,
  1651	                                   check_args_covariantly=True,
  1652	                                   allow_partial_overlap=True)
  1653
  1654	        If the 'some_check' function is also symmetric, the two calls would be equivalent
  1655	        whether or not we check the args covariantly.
  1656	    """
  1657	    # Normalize both types before comparing them.
  1658	    left = left.with_unpacked_kwargs().with_normalized_var_args()
  1659	    right = right.with_unpacked_kwargs().with_normalized_var_args()
  1660
  1661	    if is_compat_return is None:
  1662	        is_compat_return = is_compat
  1663
  1664	    # If either function is implicitly typed, ignore positional arg names too
  1665	    if left.implicit or right.implicit:
  1666	        ignore_pos_arg_names = True
  1667
  1668	    # Non-type cannot be a subtype of type.
  1669	    if right.is_type_obj() and not left.is_type_obj() and not allow_partial_overlap:
  1670	        return False
  1671
  1672	    # A callable L is a subtype of a generic callable R if L is a
  1673	    # subtype of every type obtained from R by substituting types for
  1674	    # the variables of R. We can check this by simply leaving the
  1675	    # generic variables of R as type variables, effectively varying
  1676	    # over all possible values.
  1677
  1678	    # It's okay even if these variables share ids with generic
  1679	    # type variables of L, because generating and solving
  1680	    # constraints for the variables of L to make L a subtype of R
  1681	    # (below) treats type variables on the two sides as independent.
  1682	    if left.variables:
  1683	        # Apply generic type variables away in left via type inference.
  1684	        unified = unify_generic_callable(left, right, ignore_return=ignore_return)
  1685	        if unified is None:
  1686	            return False
  1687	        left = unified
  1688
  1689	    # Check return types.
  1690	    if not ignore_return and not is_compat_return(left.ret_type, right.ret_type):
  1691	        return False
  1692
  1693	    if check_args_covariantly:
  1694	        is_compat = flip_compat_check(is_compat)
  1695
  1696	    if not strict_concatenate and (left.from_concatenate or right.from_concatenate):
  1697	        strict_concatenate_check = False
  1698	    else:
  1699	        strict_concatenate_check = True
  1700
  1701	    return are_parameters_compatible(
  1702	        left,
  1703	        right,
  1704	        is_compat=is_compat,
  1705	        is_proper_subtype=is_proper_subtype,
  1706	        ignore_pos_arg_names=ignore_pos_arg_names,
  1707	        allow_partial_overlap=allow_partial_overlap,
  1708	        strict_concatenate_check=strict_concatenate_check,

  1736	def are_parameters_compatible(
  1737	    left: Parameters | NormalizedCallableType,
  1738	    right: Parameters | NormalizedCallableType,
  1739	    *,
  1740	    is_compat: Callable[[Type, Type], bool],
  1741	    is_proper_subtype: bool,
  1742	    ignore_pos_arg_names: bool = False,
  1743	    allow_partial_overlap: bool = False,
  1744	    strict_concatenate_check: bool = False,
  1745	) -> bool:
  1746	    """Helper function for is_callable_compatible, used for Parameter compatibility"""
  1747	    if right.is_ellipsis_args and not is_proper_subtype:
  1748	        return True
  1749
  1750	    left_star = left.var_arg()
  1751	    left_star2 = left.kw_arg()
  1752	    right_star = right.var_arg()
  1753	    right_star2 = right.kw_arg()
  1754
  1755	    # Treat "def _(*a: Any, **kw: Any) -> X" similarly to "Callable[..., X]"
  1756	    if are_trivial_parameters(right) and not is_proper_subtype:
  1757	        return True
  1758	    trivial_suffix = is_trivial_suffix(right) and not is_proper_subtype
  1759
  1760	    trivial_vararg_suffix = False
  1761	    if (
  1762	        right.arg_kinds[-1:] == [ARG_STAR]
  1763	        and isinstance(get_proper_type(right.arg_types[-1]), AnyType)
  1764	        and not is_proper_subtype
  1765	        and all(k.is_positional(star=True) for k in left.arg_kinds)
  1766	    ):
  1767	        # Similar to how (*Any, **Any) is considered a supertype of all callables, we consider
  1768	        # (*Any) a supertype of all callables with positional arguments. This is needed in
  1769	        # particular because we often refuse to try type inference if actual type is not
  1770	        # a subtype of erased template type.
  1771	        trivial_vararg_suffix = True
  1772
  1773	    # Match up corresponding arguments and check them for compatibility. In
  1774	    # every pair (argL, argR) of corresponding arguments from L and R, argL must
  1775	    # be "more general" than argR if L is to be a subtype of R.
  1776
  1777	    # Arguments are corresponding if they either share a name, share a position,
  1778	    # or both. If L's corresponding argument is ambiguous, L is not a subtype of R.
  1779
  1780	    # If left has one corresponding argument by name and another by position,
  1781	    # consider them to be one "merged" argument (and not ambiguous) if they're
  1782	    # both optional, they're name-only and position-only respectively, and they
  1783	    # have the same type.  This rule allows functions with (*args, **kwargs) to
  1784	    # properly stand in for the full domain of formal arguments that they're
  1785	    # used for in practice.
  1786
  1787	    # Every argument in R must have a corresponding argument in L, and every
  1788	    # required argument in L must have a corresponding argument in R.
  1789
  1790	    # Phase 1: Confirm every argument in R has a corresponding argument in L.
  1791
  1792	    # Phase 1a: If left and right can both accept an infinite number of args,
  1793	    #           their types must be compatible.
  1794	    #
  1795	    #           Furthermore, if we're checking for compatibility in all cases,
  1796	    #           we confirm that if R accepts an infinite number of arguments,
  1797	    #           L must accept the same.
  1798	    def _incompatible(left_arg: FormalArgument | None, right_arg: FormalArgument | None) -> bool:
  1799	        if right_arg is None:
  1800	            return False
  1801	        if left_arg is None:
  1802	            return not allow_partial_overlap and not trivial_suffix
  1803	        return not is_compat(right_arg.typ, left_arg.typ)
  1804
  1805	    if (
  1806	        _incompatible(left_star, right_star)
  1807	        and not trivial_vararg_suffix
  1808	        or _incompatible(left_star2, right_star2)
  1809	    ):
  1810	        return False
  1811
  1812	    # Phase 1b: Check non-star args: for every arg right can accept, left must
  1813	    #           also accept. The only exception is if we are allowing partial
  1814	    #           overlaps: in that case, we ignore optional args on the right.
  1815	    for right_arg in right.formal_arguments():
  1816	        left_arg = mypy.typeops.callable_corresponding_argument(left, right_arg)
  1817	        if left_arg is None:
  1818	            if allow_partial_overlap and not right_arg.required:
  1819	                continue
  1820	            return False
  1821	        if not are_args_compatible(
  1822	            left_arg,
  1823	            right_arg,
  1824	            is_compat,
  1825	            ignore_pos_arg_names=ignore_pos_arg_names,
  1826	            allow_partial_overlap=allow_partial_overlap,
  1827	            allow_imprecise_kinds=right.imprecise_arg_kinds,
  1828	        ):
  1829	            return False
  1830
  1831	    if trivial_suffix:
  1832	        # For trivial right suffix we *only* check that every non-star right argument
  1833	        # has a valid match on the left.
  1834	        return True
  1835
  1836	    # Phase 1c: Check var args. Right has an infinite series of optional positional
  1837	    #           arguments. Get all further positional args of left, and make sure
  1838	    #           they're more general than the corresponding member in right.
  1839	    # TODO: handle suffix in UnpackType (i.e. *args: *Tuple[Ts, X, Y]).
  1840	    if right_star is not None and not trivial_vararg_suffix:
  1841	        # Synthesize an anonymous formal argument for the right
  1842	        right_by_position = right.try_synthesizing_arg_from_vararg(None)
  1843	        assert right_by_position is not None
  1844
  1845	        i = right_star.pos
  1846	        assert i is not None
  1847	        while i < len(left.arg_kinds) and left.arg_kinds[i].is_positional():
  1848	            if allow_partial_overlap and left.arg_kinds[i].is_optional():
  1849	                break
  1850
  1851	            left_by_position = left.argument_by_position(i)
  1852	            assert left_by_position is not None
  1853
  1854	            if not are_args_compatible(
  1855	                left_by_position,
  1856	                right_by_position,
  1857	                is_compat,
  1858	                ignore_pos_arg_names=ignore_pos_arg_names,
  1859	                allow_partial_overlap=allow_partial_overlap,
  1860	            ):
  1861	                return False
  1862	            i += 1
  1863
  1864	    # Phase 1d: Check kw args. Right has an infinite series of optional named
  1865	    #           arguments. Get all further named args of left, and make sure
  1866	    #           they're more general than the corresponding member in right.
  1867	    if right_star2 is not None:
  1868	        right_names = {name for name in right.arg_names if name is not None}
  1869	        left_only_names = set()
  1870	        for name, kind in zip(left.arg_names, left.arg_kinds):
  1871	            if (
  1872	                name is None
  1873	                or kind.is_star()
  1874	                or name in right_names
  1875	                or not strict_concatenate_check
  1876	            ):
  1877	                continue
  1878	            left_only_names.add(name)
  1879
  1880	        # Synthesize an anonymous formal argument for the right
  1881	        right_by_name = right.try_synthesizing_arg_from_kwarg(None)
  1882	        assert right_by_name is not None
  1883
  1884	        for name in left_only_names:
  1885	            left_by_name = left.argument_by_name(name)
  1886	            assert left_by_name is not None
  1887
  1888	            if allow_partial_overlap and not left_by_name.required:
  1889	                continue
  1890
  1891	            if not are_args_compatible(
  1892	                left_by_name,
  1893	                right_by_name,
  1894	                is_compat,
  1895	                ignore_pos_arg_names=ignore_pos_arg_names,
  1896	                allow_partial_overlap=allow_partial_overlap,
  1897	            ):
  1898	                return False
  1899
  1900	    # Phase 2: Left must not impose additional restrictions.
  1901	    #          (Every required argument in L must have a corresponding argument in R)
  1902	    #          Note: we already checked the *arg and **kwarg arguments in phase 1a.
  1903	    for left_arg in left.formal_arguments():
  1904	        right_by_name = (
  1905	            right.argument_by_name(left_arg.name) if left_arg.name is not None else None
  1906	        )
  1907
  1908	        right_by_pos = (
  1909	            right.argument_by_position(left_arg.pos) if left_arg.pos is not None else None
  1910	        )
  1911
  1912	        # If the left hand argument corresponds to two right-hand arguments,
  1913	        # neither of them can be required.
  1914	        if (
  1915	            right_by_name is not None
  1916	            and right_by_pos is not None
  1917	            and right_by_name != right_by_pos
  1918	            and (right_by_pos.required or right_by_name.required)
  1919	            and strict_concatenate_check
  1920	            and not right.imprecise_arg_kinds
  1921	        ):
  1922	            return False
  1923
  1924	        # All *required* left-hand arguments must have a corresponding
  1925	        # right-hand argument.  Optional args do not matter.
  1926	        if left_arg.required and right_by_pos is None and right_by_name is None:
  1927	            return False
  1928
  1929	    return True

  1932	def are_args_compatible(
  1933	    left: FormalArgument,
  1934	    right: FormalArgument,
  1935	    is_compat: Callable[[Type, Type], bool],
  1936	    *,
  1937	    ignore_pos_arg_names: bool,
  1938	    allow_partial_overlap: bool,
  1939	    allow_imprecise_kinds: bool = False,
  1940	) -> bool:
  1941	    if left.required and right.required:
  1942	        # If both arguments are required allow_partial_overlap has no effect.
  1943	        allow_partial_overlap = False
  1944
  1945	    def is_different(
  1946	        left_item: object | None, right_item: object | None, allow_overlap: bool
  1947	    ) -> bool:
  1948	        """Checks if the left and right items are different.
  1949
  1950	        If the right item is unspecified (e.g. if the right callable doesn't care
  1951	        about what name or position its arg has), we default to returning False.
  1952
  1953	        If we're allowing partial overlap, we also default to returning False
  1954	        if the left callable also doesn't care."""
  1955	        if right_item is None:
  1956	            return False
  1957	        if allow_overlap and left_item is None:
  1958	            return False
  1959	        return left_item != right_item
  1960
  1961	    # If right has a specific name it wants this argument to be, left must
  1962	    # have the same.
  1963	    if is_different(left.name, right.name, allow_partial_overlap):
  1964	        # But pay attention to whether we're ignoring positional arg names
  1965	        if not ignore_pos_arg_names or right.pos is None:
  1966	            return False
  1967
  1968	    # If right is at a specific position, left must have the same.
  1969	    # TODO: partial overlap logic is flawed for positions.
  1970	    # We disable it to avoid false positives at a cost of few false negatives.
  1971	    if is_different(left.pos, right.pos, allow_overlap=False) and not allow_imprecise_kinds:
  1972	        return False
  1973
  1974	    # If right's argument is optional, left's must also be
  1975	    # (unless we're relaxing the checks to allow potential
  1976	    # rather than definite compatibility).
  1977	    if not allow_partial_overlap and not right.required and left.required:
  1978	        return False
  1979
  1980	    # If we're allowing partial overlaps and neither arg is required,
  1981	    # the types don't actually need to be the same
  1982	    if allow_partial_overlap and not left.required and not right.required:
  1983	        return True
  1984
  1985	    # Left must have a more general type
  1986	    return is_compat(right.typ, left.typ)
