   231	        """Starts a new line at an offset dx,dy from the start of the
   232	        current line. This does not move the cursor relative to the
   233	        current position, and it changes the current offset of every
   234	        future line drawn (i.e. if you next do a textLine() call, it
   235	        will move the cursor to a position one line lower than the
   236	        position specificied in this call.  """
   237
   238	        # Check if we have a previous move cursor call, and combine
   239	        # them if possible.
   240	        if self._code and self._code[-1][-3:]==' Td':
   241	            L = self._code[-1].split()

   243	                del self._code[-1]
   244	            else:
   245	                self._code[-1] = ''.join(L[:-4])

   249	            lastDy = float(L[-2])

   256	            # we've already done this for lastDx, lastDy, remove it

   258	            self._x0 -= lastDx
   259	            self._y0 -= lastDy

   261	        # Output the move text cursor call.
   262	        self._code.append('%s Td' % fp_str(dx, -dy))

   264	        # Keep track of the new line offsets and the cursor position
   265	        self._x0 += dx
   266	        self._y0 += dy

   268	        self._y = self._y0

   423	class BaseDocTemplate:
   424	    """
   425	    First attempt at defining a document template class.
   426
   427	    The basic idea is simple.
   428
   429	    1)  The document has a list of data associated with it
   430	        this data should derive from flowables. We'll have
   431	        special classes like PageBreak, FrameBreak to do things
   432	        like forcing a page end etc.
   433
   434	    2)  The document has one or more page templates.
   435
   436	    3)  Each page template has one or more frames.
   437
   438	    4)  The document class provides base methods for handling the
   439	        story events and some reasonable methods for getting the
   440	        story flowables into the frames.
   441
   442	    5)  The document instances can override the base handler routines.
   443
   444	    Most of the methods for this class are not called directly by the user,
   445	    but in some advanced usages they may need to be overridden via subclassing.
   446
   447	    EXCEPTION: doctemplate.build(...) must be called for most reasonable uses
   448	    since it builds a document using the page template.
   449
   450	    Each document template builds exactly one document into a file specified
   451	    by the filename argument on initialization.
   452
   453	    Possible keyword arguments for the initialization:
   454
   455	    - pageTemplates: A list of templates.  Must be nonempty.  Names
   456	      assigned to the templates are used for referring to them so no two used
   457	      templates should have the same name.  For example you might want one template
   458	      for a title page, one for a section first page, one for a first page of
   459	      a chapter and two more for the interior of a chapter on odd and even pages.
   460	      If this argument is omitted then at least one pageTemplate should be provided
   461	      using the addPageTemplates method before the document is built.
   462	    - pageSize: a 2-tuple or a size constant from reportlab/lib/pagesizes.pu.
   463	      Used by the SimpleDocTemplate subclass which does NOT accept a list of
   464	      pageTemplates but makes one for you; ignored when using pageTemplates.

   466	    - showBoundary: if set draw a box around the frame boundaries.
   467	    - leftMargin:
   468	    - rightMargin:
   469	    - topMargin:
   470	    - bottomMargin:  Margin sizes in points (default 1 inch).  These margins may be
   471	      overridden by the pageTemplates.  They are primarily of interest for the
   472	      SimpleDocumentTemplate subclass.

   474	    - allowSplitting:  If set flowables (eg, paragraphs) may be split across frames or pages
   475	      (default: 1)

   900	        #allow document a chance to look at, modify or ignore
   901	        #the object(s) about to be processed
   902	        self.filterFlowables(flowables)

   904	        f = flowables[0]

   933	                    self._curPageFlowableCount += 1
   934	                    self.afterFlowable(f)

   960	                        #leave to keep apart from the raise
   961	                        raise LayoutError(ident)

   963	                    f._postponed = 1
   964	                    mbe = getattr(self,'_multiBuildEdits',None)
   965	                    if mbe:
   966	                        mbe((delattr,f,'_postponed'))

  1049	           rather than the one provided upon initialization.

  1057	        flowableCount = len(flowables)

  1061	        self._startBuild(filename,canvasmaker)
  1062

  1064	        #to take priority, so cache and reapply our own info dictionary after the build.

  1067	        handled = 0

  1069	        try:
  1070	            canv._doctemplate = self
  1071	            while len(flowables):

  1101	        finally:
