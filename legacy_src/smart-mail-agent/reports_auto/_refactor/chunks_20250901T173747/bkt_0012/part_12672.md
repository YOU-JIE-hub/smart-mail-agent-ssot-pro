  1227	    def analyze_overloaded_func_def(self, defn: OverloadedFuncDef) -> None:
  1228	        # OverloadedFuncDef refers to any legitimate situation where you have
  1229	        # more than one declaration for the same function in a row.  This occurs
  1230	        # with a @property with a setter or a deleter, and for a classic
  1231	        # @overload.
  1232
  1233	        defn._fullname = self.qualified_name(defn.name)
  1234	        # TODO: avoid modifying items.
  1235	        defn.items = defn.unanalyzed_items.copy()
  1236
  1237	        first_item = defn.items[0]
  1238	        first_item.is_overload = True
  1239	        with self.overload_item_set(0):
  1240	            first_item.accept(self)
  1241
  1242	        bare_setter_type = None
  1243	        is_property = False
  1244	        if isinstance(first_item, Decorator) and first_item.func.is_property:
  1245	            is_property = True
  1246	            # This is a property.
  1247	            first_item.func.is_overload = True
  1248	            bare_setter_type = self.analyze_property_with_multi_part_definition(defn)
  1249	            typ = function_type(first_item.func, self.named_type("builtins.function"))
  1250	            assert isinstance(typ, CallableType)
  1251	            types = [typ]
  1252	        else:
  1253	            # This is a normal overload. Find the item signatures, the
  1254	            # implementation (if outside a stub), and any missing @overload
  1255	            # decorators.
  1256	            types, impl, non_overload_indexes = self.analyze_overload_sigs_and_impl(defn)
  1257	            defn.impl = impl
  1258	            if non_overload_indexes:
  1259	                self.handle_missing_overload_decorators(
  1260	                    defn, non_overload_indexes, some_overload_decorators=len(types) > 0
  1261	                )
  1262	            # If we found an implementation, remove it from the overload item list,
  1263	            # as it's special.
  1264	            if impl is not None:
  1265	                assert impl is defn.items[-1]
  1266	                defn.items = defn.items[:-1]
  1267	            elif not non_overload_indexes:
  1268	                self.handle_missing_overload_implementation(defn)
  1269
  1270	        if types and not any(
  1271	            # If some overload items are decorated with other decorators, then
  1272	            # the overload type will be determined during type checking.
  1273	            # Note: bare @property is removed in visit_decorator().
  1274	            isinstance(it, Decorator)
  1275	            and len(it.decorators) > (1 if i > 0 or not is_property else 0)
  1276	            for i, it in enumerate(defn.items)
  1277	        ):
  1278	            # TODO: should we enforce decorated overloads consistency somehow?
  1279	            # Some existing code uses both styles:
  1280	            #   * Put decorator only on implementation, use "effective" types in overloads
  1281	            #   * Put decorator everywhere, use "bare" types in overloads.
  1282	            defn.type = Overloaded(types)
  1283	            defn.type.line = defn.line
  1284	            # In addition, we can set the getter/setter type for valid properties as some
  1285	            # code paths may either use the above type, or var.type etc. of the first item.
  1286	            if isinstance(first_item, Decorator) and bare_setter_type:
  1287	                first_item.var.type = types[0]
  1288	                first_item.var.setter_type = bare_setter_type
  1289
  1290	        if not defn.items:
  1291	            # It was not a real overload after all, but function redefinition. We've
  1292	            # visited the redefinition(s) already.
  1293	            if not defn.impl:
  1294	                # For really broken overloads with no items and no implementation we need to keep
  1295	                # at least one item to hold basic information like function name.
  1296	                defn.impl = defn.unanalyzed_items[-1]
  1297	            return
  1298
  1299	        # We know this is an overload def. Infer properties and perform some checks.
  1300	        self.process_deprecated_overload(defn)
  1301	        self.process_final_in_overload(defn)
  1302	        self.process_static_or_class_method_in_overload(defn)
  1303	        self.process_overload_impl(defn)

  1361	    def analyze_overload_sigs_and_impl(
  1362	        self, defn: OverloadedFuncDef
  1363	    ) -> tuple[list[CallableType], OverloadPart | None, list[int]]:
  1364	        """Find overload signatures, the implementation, and items with missing @overload.
  1365
  1366	        Assume that the first was already analyzed. As a side effect:
  1367	        analyzes remaining items and updates 'is_overload' flags.
  1368	        """
  1369	        types = []
  1370	        non_overload_indexes = []
  1371	        impl: OverloadPart | None = None
  1372	        for i, item in enumerate(defn.items):
  1373	            if i != 0:
  1374	                # Assume that the first item was already visited
  1375	                item.is_overload = True
  1376	                with self.overload_item_set(i if i < len(defn.items) - 1 else None):
  1377	                    item.accept(self)
  1378	            # TODO: support decorated overloaded functions properly
  1379	            if isinstance(item, Decorator):
  1380	                callable = function_type(item.func, self.named_type("builtins.function"))
  1381	                assert isinstance(callable, CallableType)
  1382	                if not any(refers_to_fullname(dec, OVERLOAD_NAMES) for dec in item.decorators):
  1383	                    if i == len(defn.items) - 1 and not self.is_stub_file:
  1384	                        # Last item outside a stub is impl
  1385	                        impl = item
  1386	                    else:
  1387	                        # Oops it wasn't an overload after all. A clear error
  1388	                        # will vary based on where in the list it is, record
  1389	                        # that.
  1390	                        non_overload_indexes.append(i)
  1391	                else:
  1392	                    item.func.is_overload = True
  1393	                    types.append(callable)
  1394	                    if item.var.is_property:
  1395	                        self.fail("An overload can not be a property", item)
  1396	                # If any item was decorated with `@override`, the whole overload
  1397	                # becomes an explicit override.
  1398	                defn.is_explicit_override |= item.func.is_explicit_override
  1399	            elif isinstance(item, FuncDef):
  1400	                if i == len(defn.items) - 1 and not self.is_stub_file:
  1401	                    impl = item
  1402	                else:
  1403	                    non_overload_indexes.append(i)
  1404	        return types, impl, non_overload_indexes

  1532	                item.func.accept(self)

  1542	                                item.func, self.named_type("builtins.function")

  1549	                        for other_node in item.decorators[1:]:

  1553	                            f"Only supported top decorator is @{first_item.func.name}.setter", item
  1554	                        )
  1555	            else:

  1561	        for item in items[1:]:

  1643	        dec.decorators = dec.original_decorators.copy()

  1645	        if not dec.is_overload:
  1646	            self.add_symbol(dec.name, dec, dec)

  1648	        dec.var._fullname = self.qualified_name(dec.name)

  1650	            d.accept(self)
  1651	        removed: list[int] = []
  1652	        no_type_check = False
  1653	        could_be_decorated_property = False

  1655	            # A bunch of decorators are special cased here.
  1656	            if refers_to_fullname(d, "abc.abstractmethod"):
  1657	                removed.append(i)
  1658	                dec.func.abstract_status = IS_ABSTRACT
  1659	                self.check_decorated_function_is_method("abstractmethod", dec)
  1660	            elif refers_to_fullname(d, ("asyncio.coroutines.coroutine", "types.coroutine")):
  1661	                removed.append(i)
  1662	                dec.func.is_awaitable_coroutine = True
  1663	            elif refers_to_fullname(d, "builtins.staticmethod"):
  1664	                removed.append(i)
  1665	                dec.func.is_static = True
  1666	                dec.var.is_staticmethod = True
  1667	                self.check_decorated_function_is_method("staticmethod", dec)
  1668	            elif refers_to_fullname(d, "builtins.classmethod"):
  1669	                removed.append(i)
  1670	                dec.func.is_class = True
  1671	                dec.var.is_classmethod = True
  1672	                self.check_decorated_function_is_method("classmethod", dec)
  1673	            elif refers_to_fullname(d, OVERRIDE_DECORATOR_NAMES):
  1674	                removed.append(i)
  1675	                dec.func.is_explicit_override = True
  1676	                self.check_decorated_function_is_method("override", dec)
  1677	            elif refers_to_fullname(
  1678	                d,
  1679	                (
  1680	                    "builtins.property",
  1681	                    "abc.abstractproperty",
  1682	                    "functools.cached_property",
  1683	                    "enum.property",
  1684	                    "types.DynamicClassAttribute",
  1685	                ),
  1686	            ):
  1687	                removed.append(i)
  1688	                dec.func.is_property = True
  1689	                dec.var.is_property = True
  1690	                if refers_to_fullname(d, "abc.abstractproperty"):
  1691	                    dec.func.abstract_status = IS_ABSTRACT
  1692	                elif refers_to_fullname(d, "functools.cached_property"):
  1693	                    dec.var.is_settable_property = True
  1694	                self.check_decorated_function_is_method("property", dec)
  1695	            elif refers_to_fullname(d, "typing.no_type_check"):
  1696	                dec.var.type = AnyType(TypeOfAny.special_form)
  1697	                no_type_check = True
  1698	            elif refers_to_fullname(d, FINAL_DECORATOR_NAMES):
  1699	                if self.is_class_scope():
  1700	                    assert self.type is not None, "No type set at class scope"
  1701	                    if self.type.is_protocol:
  1702	                        self.msg.protocol_members_cant_be_final(d)
  1703	                    else:
  1704	                        dec.func.is_final = True
  1705	                        dec.var.is_final = True
  1706	                    removed.append(i)
  1707	                else:
  1708	                    self.fail("@final cannot be used with non-method functions", d)
  1709	            elif refers_to_fullname(d, TYPE_CHECK_ONLY_NAMES):
  1710	                # TODO: support `@overload` funcs.
  1711	                dec.func.is_type_check_only = True
  1712	            elif isinstance(d, CallExpr) and refers_to_fullname(
  1713	                d.callee, DATACLASS_TRANSFORM_NAMES
  1714	            ):
  1715	                dec.func.dataclass_transform_spec = self.parse_dataclass_transform_spec(d)
  1716	            elif (deprecated := self.get_deprecated(d)) is not None:
  1717	                dec.func.deprecated = f"function {dec.fullname} is deprecated: {deprecated}"
  1718	            elif not dec.var.is_property:
  1719	                # We have seen a "non-trivial" decorator before seeing @property, if
  1720	                # we will see a @property later, give an error, as we don't support this.
  1721	                could_be_decorated_property = True
  1722	        for i in reversed(removed):
  1723	            del dec.decorators[i]
  1724	        if (not dec.is_overload or dec.var.is_property) and self.type:
  1725	            dec.var.info = self.type
  1726	            dec.var.is_initialized_in_class = True
  1727	        if not no_type_check and self.recurse_into_functions:
  1728	            dec.func.accept(self)
  1729	        if could_be_decorated_property and dec.decorators and dec.var.is_property:
  1730	            self.fail(
  1731	                "Decorators on top of @property are not supported", dec, code=PROPERTY_DECORATOR
  1732	            )
  1733	        if (dec.func.is_static or dec.func.is_class) and dec.var.is_property:
  1734	            self.fail("Only instance methods can be decorated with @property", dec)
  1735	        if dec.func.abstract_status == IS_ABSTRACT and dec.func.is_final:
  1736	            self.fail(f"Method {dec.func.name} is both abstract and final", dec)
  1737	        if dec.func.is_static and dec.func.is_class:
  1738	            self.fail(message_registry.CLASS_PATTERN_CLASS_OR_STATIC_METHOD, dec)

  1761	    def push_type_args(
  1762	        self, type_args: list[TypeParam] | None, context: Context
  1763	    ) -> list[tuple[str, TypeVarLikeExpr]] | None:
  1764	        if not type_args:
  1765	            return []
  1766	        self.locals.append(SymbolTable())
  1767	        self.scope_stack.append(SCOPE_ANNOTATION)
  1768	        tvs: list[tuple[str, TypeVarLikeExpr]] = []
  1769	        for p in type_args:

  1771	            if tv is None:
  1772	                return None

  1776	            if self.is_defined_type_param(name):
  1777	                self.fail(f'"{name}" already defined as a type parameter', context)
  1778	            else:
  1779	                self.add_symbol(name, tv, context, no_progress=True, type_param=True)

  1793	    def analyze_type_param(
  1794	        self, type_param: TypeParam, context: Context
  1795	    ) -> TypeVarLikeExpr | None:
  1796	        fullname = self.qualified_name(type_param.name)
  1797	        if type_param.upper_bound:
  1798	            upper_bound = self.anal_type(type_param.upper_bound, allow_placeholder=True)
  1799	            # TODO: we should validate the upper bound is valid for a given kind.
  1800	            if upper_bound is None:
  1801	                # This and below copies special-casing for old-style type variables, that
  1802	                # is equally necessary for new-style classes to break a vicious circle.
  1803	                upper_bound = PlaceholderType(None, [], context.line)
  1804	        else:
  1805	            if type_param.kind == TYPE_VAR_TUPLE_KIND:
  1806	                upper_bound = self.named_type("builtins.tuple", [self.object_type()])
  1807	            else:
  1808	                upper_bound = self.object_type()
  1809	        if type_param.default:
  1810	            default = self.anal_type(
  1811	                type_param.default,
  1812	                allow_placeholder=True,
  1813	                allow_unbound_tvars=True,
  1814	                report_invalid_types=False,
  1815	                allow_param_spec_literals=type_param.kind == PARAM_SPEC_KIND,
  1816	                allow_tuple_literal=type_param.kind == PARAM_SPEC_KIND,
  1817	                allow_unpack=type_param.kind == TYPE_VAR_TUPLE_KIND,
  1818	            )
  1819	            if default is None:
  1820	                default = PlaceholderType(None, [], context.line)
  1821	            elif type_param.kind == TYPE_VAR_KIND:
  1822	                default = self.check_typevar_default(default, type_param.default)
  1823	            elif type_param.kind == PARAM_SPEC_KIND:
  1824	                default = self.check_paramspec_default(default, type_param.default)
  1825	            elif type_param.kind == TYPE_VAR_TUPLE_KIND:
  1826	                default = self.check_typevartuple_default(default, type_param.default)
  1827	        else:
  1828	            default = AnyType(TypeOfAny.from_omitted_generics)
  1829	        if type_param.kind == TYPE_VAR_KIND:
  1830	            values: list[Type] = []
  1831	            if type_param.values:
  1832	                for value in type_param.values:
  1833	                    analyzed = self.anal_type(value, allow_placeholder=True)
  1834	                    if analyzed is None:
  1835	                        analyzed = PlaceholderType(None, [], context.line)
  1836	                    if has_type_vars(analyzed):
  1837	                        self.fail(message_registry.TYPE_VAR_GENERIC_CONSTRAINT_TYPE, context)
  1838	                        values.append(AnyType(TypeOfAny.from_error))
  1839	                    else:
  1840	                        values.append(analyzed)
  1841	            return TypeVarExpr(
  1842	                name=type_param.name,
  1843	                fullname=fullname,
  1844	                values=values,
  1845	                upper_bound=upper_bound,
  1846	                default=default,
  1847	                variance=VARIANCE_NOT_READY,
  1848	                is_new_style=True,
  1849	                line=context.line,
  1850	            )
  1851	        elif type_param.kind == PARAM_SPEC_KIND:
  1852	            return ParamSpecExpr(
  1853	                name=type_param.name,
  1854	                fullname=fullname,
  1855	                upper_bound=upper_bound,
  1856	                default=default,
  1857	                is_new_style=True,
  1858	                line=context.line,
  1859	            )
  1860	        else:
  1861	            assert type_param.kind == TYPE_VAR_TUPLE_KIND
  1862	            tuple_fallback = self.named_type("builtins.tuple", [self.object_type()])
  1863	            return TypeVarTupleExpr(
  1864	                name=type_param.name,
  1865	                fullname=fullname,
  1866	                upper_bound=upper_bound,
  1867	                tuple_fallback=tuple_fallback,

  1879	    def analyze_class(self, defn: ClassDef) -> None:
  1880	        fullname = self.qualified_name(defn.name)
  1881	        if not defn.info and not self.is_core_builtin_class(defn):
  1882	            # Add placeholder so that self-references in base classes can be
  1883	            # resolved.  We don't want this to cause a deferral, since if there
  1884	            # are no incomplete references, we'll replace this with a TypeInfo
  1885	            # before returning.
  1886	            placeholder = PlaceholderNode(fullname, defn, defn.line, becomes_typeinfo=True)
  1887	            self.add_symbol(defn.name, placeholder, defn, can_defer=False)
  1888
  1889	        tag = self.track_incomplete_refs()
  1890
  1891	        # Restore base classes after previous iteration (things like Generic[T] might be removed).
  1892	        defn.base_type_exprs.extend(defn.removed_base_type_exprs)
  1893	        defn.removed_base_type_exprs.clear()
  1894
  1895	        self.infer_metaclass_and_bases_from_compat_helpers(defn)
  1896
  1897	        bases = defn.base_type_exprs
  1898	        bases, tvar_defs, is_protocol = self.clean_up_bases_and_infer_type_variables(
  1899	            defn, bases, context=defn
  1900	        )
  1901
  1902	        self.check_type_alias_bases(bases)
  1903
  1904	        for tvd in tvar_defs:
  1905	            if isinstance(tvd, TypeVarType) and any(
  1906	                has_placeholder(t) for t in [tvd.upper_bound] + tvd.values
  1907	            ):
  1908	                # Some type variable bounds or values are not ready, we need
  1909	                # to re-analyze this class.
  1910	                self.defer()
  1911	            if has_placeholder(tvd.default):
  1912	                # Placeholder values in TypeVarLikeTypes may get substituted in.
  1913	                # Defer current target until they are ready.
  1914	                self.mark_incomplete(defn.name, defn)
  1915	                return
  1916
  1917	        self.analyze_class_keywords(defn)
  1918	        bases_result = self.analyze_base_classes(bases)
  1919	        if bases_result is None or self.found_incomplete_ref(tag):
  1920	            # Something was incomplete. Defer current target.
  1921	            self.mark_incomplete(defn.name, defn)
  1922	            return
  1923
  1924	        base_types, base_error = bases_result
  1925	        if any(isinstance(base, PlaceholderType) for base, _ in base_types):
  1926	            # We need to know the TypeInfo of each base to construct the MRO. Placeholder types
  1927	            # are okay in nested positions, since they can't affect the MRO.
  1928	            self.mark_incomplete(defn.name, defn)
  1929	            return
  1930
  1931	        declared_metaclass, should_defer, any_meta = self.get_declared_metaclass(
  1932	            defn.name, defn.metaclass
  1933	        )
  1934	        if should_defer or self.found_incomplete_ref(tag):
  1935	            # Metaclass was not ready. Defer current target.
  1936	            self.mark_incomplete(defn.name, defn)
  1937	            return
  1938
  1939	        if self.analyze_typeddict_classdef(defn):
  1940	            if defn.info:
  1941	                self.setup_type_vars(defn, tvar_defs)
  1942	                self.setup_alias_type_vars(defn)
  1943	            return
  1944
  1945	        if self.analyze_namedtuple_classdef(defn, tvar_defs):
  1946	            return
  1947
  1948	        # Create TypeInfo for class now that base classes and the MRO can be calculated.
  1949	        self.prepare_class_def(defn)
  1950	        self.setup_type_vars(defn, tvar_defs)
  1951	        if base_error:
  1952	            defn.info.fallback_to_any = True
  1953	        if any_meta:
  1954	            defn.info.meta_fallback_to_any = True
  1955
  1956	        with self.scope.class_scope(defn.info):
  1957	            self.configure_base_classes(defn, base_types)
  1958	            defn.info.is_protocol = is_protocol
  1959	            self.recalculate_metaclass(defn, declared_metaclass)
  1960	            defn.info.runtime_protocol = False
  1961
  1962	            if defn.type_args:
  1963	                # PEP 695 type parameters are not in scope in class decorators, so
  1964	                # temporarily disable type parameter namespace.
  1965	                type_params_names = self.locals.pop()
  1966	                self.scope_stack.pop()
  1967	            for decorator in defn.decorators:
  1968	                self.analyze_class_decorator(defn, decorator)
  1969	            if defn.type_args:
  1970	                self.locals.append(type_params_names)
  1971	                self.scope_stack.append(SCOPE_ANNOTATION)
  1972
  1973	            self.analyze_class_body_common(defn)

  2180	    def clean_up_bases_and_infer_type_variables(
  2181	        self, defn: ClassDef, base_type_exprs: list[Expression], context: Context
  2182	    ) -> tuple[list[Expression], list[TypeVarLikeType], bool]:
  2183	        """Remove extra base classes such as Generic and infer type vars.
  2184
  2185	        For example, consider this class:
  2186
  2187	          class Foo(Bar, Generic[T]): ...
  2188
  2189	        Now we will remove Generic[T] from bases of Foo and infer that the
  2190	        type variable 'T' is a type argument of Foo.
  2191
  2192	        Note that this is performed *before* semantic analysis.
  2193
  2194	        Returns (remaining base expressions, inferred type variables, is protocol).
  2195	        """
  2196	        removed: list[int] = []
  2197	        declared_tvars: TypeVarLikeList = []
  2198	        is_protocol = False
  2199	        if defn.type_args is not None:
  2200	            for p in defn.type_args:
  2201	                node = self.lookup(p.name, context)
  2202	                assert node is not None
  2203	                assert isinstance(node.node, TypeVarLikeExpr)
  2204	                declared_tvars.append((p.name, node.node))
  2205
  2206	        for i, base_expr in enumerate(base_type_exprs):
  2207	            if isinstance(base_expr, StarExpr):
  2208	                base_expr.valid = True
  2209	            self.analyze_type_expr(base_expr)
  2210
  2211	            try:
  2212	                base = self.expr_to_unanalyzed_type(base_expr)
  2213	            except TypeTranslationError:
  2214	                # This error will be caught later.
  2215	                continue
  2216	            result = self.analyze_class_typevar_declaration(base)
  2217	            if result is not None:
  2218	                tvars = result[0]
  2219	                is_protocol |= result[1]
  2220	                if declared_tvars:
  2221	                    if defn.type_args:
  2222	                        if is_protocol:
  2223	                            self.fail('No arguments expected for "Protocol" base class', context)
  2224	                        else:
  2225	                            self.fail("Generic[...] base class is redundant", context)
  2226	                    else:
  2227	                        self.fail(
  2228	                            "Only single Generic[...] or Protocol[...] can be in bases", context
  2229	                        )
  2230	                removed.append(i)
  2231	                declared_tvars.extend(tvars)
  2232	            if isinstance(base, UnboundType):
  2233	                sym = self.lookup_qualified(base.name, base)
  2234	                if sym is not None and sym.node is not None:
  2235	                    if sym.node.fullname in PROTOCOL_NAMES and i not in removed:
  2236	                        # also remove bare 'Protocol' bases
  2237	                        removed.append(i)
  2238	                        is_protocol = True
  2239
  2240	        all_tvars = self.get_all_bases_tvars(base_type_exprs, removed)
  2241	        if declared_tvars:
  2242	            if len(remove_dups(declared_tvars)) < len(declared_tvars) and not defn.type_args:
  2243	                self.fail("Duplicate type variables in Generic[...] or Protocol[...]", context)
  2244	            declared_tvars = remove_dups(declared_tvars)
  2245	            if not set(all_tvars).issubset(set(declared_tvars)):
  2246	                if defn.type_args:
  2247	                    undeclared = sorted(set(all_tvars) - set(declared_tvars))
  2248	                    self.msg.type_parameters_should_be_declared(
  2249	                        [tv[0] for tv in undeclared], context
  2250	                    )
  2251	                else:
  2252	                    self.fail(
  2253	                        "If Generic[...] or Protocol[...] is present"
  2254	                        " it should list all type variables",
  2255	                        context,
  2256	                    )
  2257	                # In case of error, Generic tvars will go first
  2258	                declared_tvars = remove_dups(declared_tvars + all_tvars)
  2259	        else:
  2260	            declared_tvars = all_tvars
  2261	        for i in reversed(removed):
  2262	            # We need to actually remove the base class expressions like Generic[T],
  2263	            # mostly because otherwise they will create spurious dependencies in fine
  2264	            # grained incremental mode.
  2265	            defn.removed_base_type_exprs.append(defn.base_type_exprs[i])
  2266	            del base_type_exprs[i]
  2267	        tvar_defs = self.tvar_defs_from_tvars(declared_tvars, context)
  2268	        return base_type_exprs, tvar_defs, is_protocol

  2270	    def analyze_class_typevar_declaration(self, base: Type) -> tuple[TypeVarLikeList, bool] | None:
  2271	        """Analyze type variables declared using Generic[...] or Protocol[...].
  2272
  2273	        Args:
  2274	            base: Non-analyzed base class
  2275
  2276	        Return None if the base class does not declare type variables. Otherwise,
  2277	        return the type variables.
  2278	        """
  2279	        if not isinstance(base, UnboundType):
  2280	            return None
  2281	        unbound = base
  2282	        sym = self.lookup_qualified(unbound.name, unbound)
  2283	        if sym is None or sym.node is None:
  2284	            return None
  2285	        if (
  2286	            sym.node.fullname == "typing.Generic"
  2287	            or sym.node.fullname in PROTOCOL_NAMES
  2288	            and base.args
  2289	        ):
  2290	            is_proto = sym.node.fullname != "typing.Generic"
  2291	            tvars: TypeVarLikeList = []
  2292	            have_type_var_tuple = False

  2294	                tag = self.track_incomplete_refs()

  2299	                            self.fail("Can only use one type var tuple in a class def", base)
  2300	                            continue
  2301	                        have_type_var_tuple = True
  2302	                    tvars.append(tvar)
  2303	                elif not self.found_incomplete_ref(tag):
  2304	                    self.fail("Free type variable expected in %s[...]" % sym.node.name, base)

  2485	    def analyze_base_classes(
  2486	        self, base_type_exprs: list[Expression]
  2487	    ) -> tuple[list[tuple[ProperType, Expression]], bool] | None:
  2488	        """Analyze base class types.
  2489
  2490	        Return None if some definition was incomplete. Otherwise, return a tuple
  2491	        with these items:
  2492
  2493	         * List of (analyzed type, original expression) tuples
  2494	         * Boolean indicating whether one of the bases had a semantic analysis error
  2495	        """
  2496	        is_error = False
  2497	        bases = []
  2498	        for base_expr in base_type_exprs:
  2499	            if (
  2500	                isinstance(base_expr, RefExpr)
  2501	                and base_expr.fullname in TYPED_NAMEDTUPLE_NAMES + TPDICT_NAMES
  2502	            ) or (
  2503	                isinstance(base_expr, CallExpr)
  2504	                and isinstance(base_expr.callee, RefExpr)
  2505	                and base_expr.callee.fullname in TPDICT_NAMES
  2506	            ):
  2507	                # Ignore magic bases for now.
  2508	                # For example:
  2509	                #  class Foo(TypedDict): ...  # RefExpr
  2510	                #  class Foo(NamedTuple): ...  # RefExpr
  2511	                #  class Foo(TypedDict("Foo", {"a": int})): ...  # CallExpr
  2512	                continue
  2513
  2514	            try:
  2515	                base = self.expr_to_analyzed_type(
  2516	                    base_expr, allow_placeholder=True, allow_type_any=True
  2517	                )
  2518	            except TypeTranslationError:
  2519	                name = self.get_name_repr_of_expr(base_expr)
  2520	                if isinstance(base_expr, CallExpr):
  2521	                    msg = "Unsupported dynamic base class"
  2522	                else:
  2523	                    msg = "Invalid base class"
  2524	                if name:
  2525	                    msg += f' "{name}"'
  2526	                self.fail(msg, base_expr)
  2527	                is_error = True
  2528	                continue
  2529	            if base is None:
  2530	                return None
  2531	            base = get_proper_type(base)
  2532	            bases.append((base, base_expr))
  2533	        return bases, is_error

  2535	    def configure_base_classes(
  2536	        self, defn: ClassDef, bases: list[tuple[ProperType, Expression]]
  2537	    ) -> None:
  2538	        """Set up base classes.
  2539
  2540	        This computes several attributes on the corresponding TypeInfo defn.info
  2541	        related to the base classes: defn.info.bases, defn.info.mro, and
  2542	        miscellaneous others (at least tuple_type, fallback_to_any, and is_enum.)
  2543	        """
  2544	        base_types: list[Instance] = []
  2545	        info = defn.info
  2546
  2547	        for base, base_expr in bases:
  2548	            if isinstance(base, TupleType):
  2549	                actual_base = self.configure_tuple_base_class(defn, base)
  2550	                base_types.append(actual_base)
  2551	            elif isinstance(base, Instance):
  2552	                if base.type.is_newtype:
  2553	                    self.fail('Cannot subclass "NewType"', defn)
  2554	                base_types.append(base)
  2555	            elif isinstance(base, AnyType):
  2556	                if self.options.disallow_subclassing_any:
  2557	                    if isinstance(base_expr, (NameExpr, MemberExpr)):
  2558	                        msg = f'Class cannot subclass "{base_expr.name}" (has type "Any")'
  2559	                    else:
  2560	                        msg = 'Class cannot subclass value of type "Any"'
  2561	                    self.fail(msg, base_expr)
  2562	                info.fallback_to_any = True
  2563	            elif isinstance(base, TypedDictType):
  2564	                base_types.append(base.fallback)
  2565	            else:
  2566	                msg = "Invalid base class"
  2567	                name = self.get_name_repr_of_expr(base_expr)
  2568	                if name:
  2569	                    msg += f' "{name}"'
  2570	                self.fail(msg, base_expr)
  2571	                info.fallback_to_any = True
  2572	            if self.options.disallow_any_unimported and has_any_from_unimported_type(base):
  2573	                if isinstance(base_expr, (NameExpr, MemberExpr)):
  2574	                    prefix = f"Base type {base_expr.name}"
  2575	                else:
  2576	                    prefix = "Base type"
  2577	                self.msg.unimported_type_becomes_any(prefix, base, base_expr)
  2578	            check_for_explicit_any(
  2579	                base, self.options, self.is_typeshed_stub_file, self.msg, context=base_expr
  2580	            )
  2581
  2582	        # Add 'object' as implicit base if there is no other base class.
  2583	        if not base_types and defn.fullname != "builtins.object":
  2584	            base_types.append(self.object_type())
  2585
  2586	        info.bases = base_types
  2587
  2588	        # Calculate the MRO.
  2589	        if not self.verify_base_classes(defn):
  2590	            self.set_dummy_mro(defn.info)
  2591	            return
  2592	        if not self.verify_duplicate_base_classes(defn):
  2593	            # We don't want to block the typechecking process,
  2594	            # so, we just insert `Any` as the base class and show an error.
  2595	            self.set_any_mro(defn.info)
  2596	        self.calculate_class_mro(defn, self.object_type)

  2652	    def infer_metaclass_and_bases_from_compat_helpers(self, defn: ClassDef) -> None:
  2653	        """Lookup for special metaclass declarations, and update defn fields accordingly.
  2654
  2655	        * six.with_metaclass(M, B1, B2, ...)

  2657	        * future.utils.with_metaclass(M, B1, B2, ...)
  2658	        * past.utils.with_metaclass(M, B1, B2, ...)

  2660
  2661	        # Look for six.with_metaclass(M, B1, B2, ...)
  2662	        with_meta_expr: Expression | None = None
  2663	        if len(defn.base_type_exprs) == 1:
  2664	            base_expr = defn.base_type_exprs[0]
  2665	            if isinstance(base_expr, CallExpr) and isinstance(base_expr.callee, RefExpr):
  2666	                self.analyze_type_expr(base_expr)

  2668	                    base_expr.callee.fullname
  2669	                    in {
  2670	                        "six.with_metaclass",
  2671	                        "future.utils.with_metaclass",
  2672	                        "past.utils.with_metaclass",
  2673	                    }
  2674	                    and len(base_expr.args) >= 1
  2675	                    and all(kind == ARG_POS for kind in base_expr.arg_kinds)
  2676	                ):
  2677	                    with_meta_expr = base_expr.args[0]
  2678	                    defn.base_type_exprs = base_expr.args[1:]

  2680	        # Look for @six.add_metaclass(M)

  2682	        for dec_expr in defn.decorators:

  2684	                dec_expr.callee.accept(self)

  2686	                    dec_expr.callee.fullname == "six.add_metaclass"
  2687	                    and len(dec_expr.args) == 1

  2690	                    add_meta_expr = dec_expr.args[0]

  2717	    def is_base_class(self, t: TypeInfo, s: TypeInfo) -> bool:
  2718	        """Determine if t is a base class of s (but do not use mro)."""
  2719	        # Search the base class graph for t, starting from s.
  2720	        worklist = [s]
  2721	        visited = {s}
  2722	        while worklist:
  2723	            nxt = worklist.pop()
  2724	            if nxt == t:
  2725	                return True
  2726	            for base in nxt.bases:
  2727	                if base.type not in visited:
  2728	                    worklist.append(base.type)
  2729	                    visited.add(base.type)
  2730	        return False

  2732	    def get_declared_metaclass(
  2733	        self, name: str, metaclass_expr: Expression | None
  2734	    ) -> tuple[Instance | None, bool, bool]:
  2735	        """Get declared metaclass from metaclass expression.
  2736
  2737	        Returns a tuple of three values:
  2738	          * A metaclass instance or None
  2739	          * A boolean indicating whether we should defer
  2740	          * A boolean indicating whether we should set metaclass Any fallback
  2741	            (either for Any metaclass or invalid/dynamic metaclass).
  2742
  2743	        The two boolean flags can only be True if instance is None.
  2744	        """
  2745	        declared_metaclass = None
  2746	        if metaclass_expr:
  2747	            metaclass_name = None
  2748	            if isinstance(metaclass_expr, NameExpr):
  2749	                metaclass_name = metaclass_expr.name
  2750	            elif isinstance(metaclass_expr, MemberExpr):
  2751	                metaclass_name = get_member_expr_fullname(metaclass_expr)
  2752	            if metaclass_name is None:
  2753	                self.fail(f'Dynamic metaclass not supported for "{name}"', metaclass_expr)
  2754	                return None, False, True
  2755	            sym = self.lookup_qualified(metaclass_name, metaclass_expr)
  2756	            if sym is None:
  2757	                # Probably a name error - it is already handled elsewhere
  2758	                return None, False, True
  2759	            if isinstance(sym.node, Var) and isinstance(get_proper_type(sym.node.type), AnyType):
  2760	                if self.options.disallow_subclassing_any:
  2761	                    self.fail(
  2762	                        f'Class cannot use "{sym.node.name}" as a metaclass (has type "Any")',
  2763	                        metaclass_expr,
  2764	                    )
  2765	                return None, False, True
  2766	            if isinstance(sym.node, PlaceholderNode):
  2767	                return None, True, False  # defer later in the caller
  2768
  2769	            # Support type aliases, like `_Meta: TypeAlias = type`
  2770	            metaclass_info: Node | None = sym.node
  2771	            if (
  2772	                isinstance(sym.node, TypeAlias)
  2773	                and not sym.node.python_3_12_type_alias
  2774	                and not sym.node.alias_tvars
  2775	            ):
  2776	                target = get_proper_type(sym.node.target)
  2777	                if isinstance(target, Instance):
  2778	                    metaclass_info = target.type
  2779
  2780	            if not isinstance(metaclass_info, TypeInfo) or metaclass_info.tuple_type is not None:
  2781	                self.fail(f'Invalid metaclass "{metaclass_name}"', metaclass_expr)
  2782	                return None, False, False
  2783	            if not metaclass_info.is_metaclass():
  2784	                self.fail(
  2785	                    'Metaclasses not inheriting from "type" are not supported', metaclass_expr
  2786	                )
  2787	                return None, False, False
  2788	            inst = fill_typevars(metaclass_info)
  2789	            assert isinstance(inst, Instance)
  2790	            declared_metaclass = inst
  2791	        return declared_metaclass, False, False

  2869	                # could more precisely model runtime semantics by taking into account
  2870	                # the line number beyond which the local definition should take

  2877
