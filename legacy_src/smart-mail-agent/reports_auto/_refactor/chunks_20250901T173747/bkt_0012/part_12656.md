    73	class Constraint:
    74	    """A representation of a type constraint.
    75
    76	    It can be either T <: type or T :> type (T is a type variable).
    77	    """
    78
    79	    type_var: TypeVarId
    80	    op = 0  # SUBTYPE_OF or SUPERTYPE_OF
    81	    target: Type
    82
    83	    def __init__(self, type_var: TypeVarLikeType, op: int, target: Type) -> None:
    84	        self.type_var = type_var.id
    85	        self.op = op
    86	        # TODO: should we add "assert not isinstance(target, UnpackType)"?
    87	        # UnpackType is a synthetic type, and is never valid as a constraint target.
    88	        self.target = target
    89	        self.origin_type_var = type_var
    90	        # These are additional type variables that should be solved for together with type_var.
    91	        # TODO: A cleaner solution may be to modify the return type of infer_constraints()
    92	        # to include these instead, but this is a rather big refactoring.
    93	        self.extra_tvars: list[TypeVarLikeType] = []
    94
    95	    def __repr__(self) -> str:
    96	        op_str = "<:"
    97	        if self.op == SUPERTYPE_OF:
    98	            op_str = ":>"
    99	        return f"{self.type_var} {op_str} {self.target}"
   100
   101	    def __hash__(self) -> int:
   102	        return hash((self.type_var, self.op, self.target))
   103
   104	    def __eq__(self, other: object) -> bool:
   105	        if not isinstance(other, Constraint):
   106	            return False
   107	        return (self.type_var, self.op, self.target) == (other.type_var, other.op, other.target)

   110	def infer_constraints_for_callable(
   111	    callee: CallableType,
   112	    arg_types: Sequence[Type | None],
   113	    arg_kinds: list[ArgKind],
   114	    arg_names: Sequence[str | None] | None,
   115	    formal_to_actual: list[list[int]],
   116	    context: ArgumentInferContext,
   117	) -> list[Constraint]:
   118	    """Infer type variable constraints for a callable and actual arguments.
   119
   120	    Return a list of constraints.
   121	    """
   122	    constraints: list[Constraint] = []
   123	    mapper = ArgTypeExpander(context)
   124
   125	    param_spec = callee.param_spec()
   126	    param_spec_arg_types = []
   127	    param_spec_arg_names = []
   128	    param_spec_arg_kinds = []
   129
   130	    incomplete_star_mapping = False
   131	    for i, actuals in enumerate(formal_to_actual):  # TODO: isn't this `enumerate(arg_types)`?
   132	        for actual in actuals:
   133	            if actual is None and callee.arg_kinds[i] in (ARG_STAR, ARG_STAR2):  # type: ignore[unreachable]
   134	                # We can't use arguments to infer ParamSpec constraint, if only some
   135	                # are present in the current inference pass.
   136	                incomplete_star_mapping = True  # type: ignore[unreachable]
   137	                break
   138
   139	    for i, actuals in enumerate(formal_to_actual):
   140	        if isinstance(callee.arg_types[i], UnpackType):
   141	            unpack_type = callee.arg_types[i]
   142	            assert isinstance(unpack_type, UnpackType)
   143
   144	            # In this case we are binding all the actuals to *args,
   145	            # and we want a constraint that the typevar tuple being unpacked
   146	            # is equal to a type list of all the actuals.
   147	            actual_types = []
   148
   149	            unpacked_type = get_proper_type(unpack_type.type)
   150	            if isinstance(unpacked_type, TypeVarTupleType):
   151	                tuple_instance = unpacked_type.tuple_fallback
   152	            elif isinstance(unpacked_type, TupleType):
   153	                tuple_instance = unpacked_type.partial_fallback
   154	            else:
   155	                assert False, "mypy bug: unhandled constraint inference case"
   156
   157	            for actual in actuals:
   158	                actual_arg_type = arg_types[actual]
   159	                if actual_arg_type is None:
   160	                    continue
   161
   162	                expanded_actual = mapper.expand_actual_type(
   163	                    actual_arg_type,
   164	                    arg_kinds[actual],
   165	                    callee.arg_names[i],
   166	                    callee.arg_kinds[i],
   167	                    allow_unpack=True,
   168	                )
   169
   170	                if arg_kinds[actual] != ARG_STAR or isinstance(
   171	                    get_proper_type(actual_arg_type), TupleType
   172	                ):
   173	                    actual_types.append(expanded_actual)
   174	                else:
   175	                    # If we are expanding an iterable inside * actual, append a homogeneous item instead
   176	                    actual_types.append(
   177	                        UnpackType(tuple_instance.copy_modified(args=[expanded_actual]))
   178	                    )
   179
   180	            if isinstance(unpacked_type, TypeVarTupleType):
   181	                constraints.append(
   182	                    Constraint(
   183	                        unpacked_type,
   184	                        SUPERTYPE_OF,
   185	                        TupleType(actual_types, unpacked_type.tuple_fallback),
   186	                    )
   187	                )
   188	            elif isinstance(unpacked_type, TupleType):
   189	                # Prefixes get converted to positional args, so technically the only case we
   190	                # should have here is like Tuple[Unpack[Ts], Y1, Y2, Y3]. If this turns out
   191	                # not to hold we can always handle the prefixes too.
   192	                inner_unpack = unpacked_type.items[0]
   193	                assert isinstance(inner_unpack, UnpackType)
   194	                inner_unpacked_type = get_proper_type(inner_unpack.type)
   195	                suffix_len = len(unpacked_type.items) - 1
   196	                if isinstance(inner_unpacked_type, TypeVarTupleType):
   197	                    # Variadic item can be either *Ts...
   198	                    constraints.append(
   199	                        Constraint(
   200	                            inner_unpacked_type,
   201	                            SUPERTYPE_OF,
   202	                            TupleType(
   203	                                actual_types[:-suffix_len], inner_unpacked_type.tuple_fallback
   204	                            ),
   205	                        )
   206	                    )
   207	                else:
   208	                    # ...or it can be a homogeneous tuple.
   209	                    assert (
   210	                        isinstance(inner_unpacked_type, Instance)
   211	                        and inner_unpacked_type.type.fullname == "builtins.tuple"
   212	                    )
   213	                    for at in actual_types[:-suffix_len]:
   214	                        constraints.extend(
   215	                            infer_constraints(inner_unpacked_type.args[0], at, SUPERTYPE_OF)
   216	                        )
   217	                # Now handle the suffix (if any).
   218	                if suffix_len:
   219	                    for tt, at in zip(unpacked_type.items[1:], actual_types[-suffix_len:]):
   220	                        constraints.extend(infer_constraints(tt, at, SUPERTYPE_OF))
   221	            else:
   222	                assert False, "mypy bug: unhandled constraint inference case"
   223	        else:
   224	            for actual in actuals:
   225	                actual_arg_type = arg_types[actual]
   226	                if actual_arg_type is None:
   227	                    continue
   228
   229	                if param_spec and callee.arg_kinds[i] in (ARG_STAR, ARG_STAR2):
   230	                    # If actual arguments are mapped to ParamSpec type, we can't infer individual
   231	                    # constraints, instead store them and infer single constraint at the end.
   232	                    # It is impossible to map actual kind to formal kind, so use some heuristic.
   233	                    # This inference is used as a fallback, so relying on heuristic should be OK.
   234	                    if not incomplete_star_mapping:
   235	                        param_spec_arg_types.append(
   236	                            mapper.expand_actual_type(
   237	                                actual_arg_type, arg_kinds[actual], None, arg_kinds[actual]
   238	                            )
   239	                        )
   240	                        actual_kind = arg_kinds[actual]
   241	                        param_spec_arg_kinds.append(
   242	                            ARG_POS if actual_kind not in (ARG_STAR, ARG_STAR2) else actual_kind
   243	                        )
   244	                        param_spec_arg_names.append(arg_names[actual] if arg_names else None)
   245	                else:
   246	                    actual_type = mapper.expand_actual_type(
   247	                        actual_arg_type,
   248	                        arg_kinds[actual],
   249	                        callee.arg_names[i],
   250	                        callee.arg_kinds[i],
   251	                    )
   252	                    c = infer_constraints(callee.arg_types[i], actual_type, SUPERTYPE_OF)
   253	                    constraints.extend(c)
   254	    if (
   255	        param_spec
   256	        and not any(c.type_var == param_spec.id for c in constraints)
   257	        and not incomplete_star_mapping
   258	    ):
   259	        # Use ParamSpec constraint from arguments only if there are no other constraints,
   260	        # since as explained above it is quite ad-hoc.
   261	        constraints.append(
   262	            Constraint(
   263	                param_spec,
   264	                SUPERTYPE_OF,
   265	                Parameters(
   266	                    arg_types=param_spec_arg_types,
   267	                    arg_kinds=param_spec_arg_kinds,
   268	                    arg_names=param_spec_arg_names,
   269	                    imprecise_arg_kinds=True,
   270	                ),
   271	            )
   272	        )
   273	    if any(isinstance(v, ParamSpecType) for v in callee.variables):
   274	        # As a perf optimization filter imprecise constraints only when we can have them.
   275	        constraints = filter_imprecise_kinds(constraints)
   276	    return constraints

   279	def infer_constraints(
   280	    template: Type, actual: Type, direction: int, skip_neg_op: bool = False
   281	) -> list[Constraint]:
   282	    """Infer type constraints.
   283
   284	    Match a template type, which may contain type variable references,
   285	    recursively against a type which does not contain (the same) type
   286	    variable references. The result is a list of type constrains of
   287	    form 'T is a supertype/subtype of x', where T is a type variable
   288	    present in the template and x is a type without reference to type
   289	    variables present in the template.
   290
   291	    Assume T and S are type variables. Now the following results can be
   292	    calculated (read as '(template, actual) --> result'):
   293
   294	      (T, X)            -->  T :> X
   295	      (X[T], X[Y])      -->  T <: Y and T :> Y
   296	      ((T, T), (X, Y))  -->  T :> X and T :> Y
   297	      ((T, S), (X, Y))  -->  T :> X and S :> Y
   298	      (X[T], Any)       -->  T <: Any and T :> Any
   299
   300	    The constraints are represented as Constraint objects. If skip_neg_op == True,
   301	    then skip adding reverse (polymorphic) constraints (since this is already a call
   302	    to infer such constraints).
   303	    """
   304	    if any(
   305	        get_proper_type(template) == get_proper_type(t)
   306	        and get_proper_type(actual) == get_proper_type(a)
   307	        for (t, a) in reversed(type_state.inferring)
   308	    ):
   309	        return []
   310	    if has_recursive_types(template) or isinstance(get_proper_type(template), Instance):
   311	        # This case requires special care because it may cause infinite recursion.
   312	        # Note that we include Instances because the may be recursive as str(Sequence[str]).
   313	        if not has_type_vars(template):
   314	            # Return early on an empty branch.
   315	            return []
   316	        type_state.inferring.append((template, actual))
   317	        res = _infer_constraints(template, actual, direction, skip_neg_op)
   318	        type_state.inferring.pop()
   319	        return res
   320	    return _infer_constraints(template, actual, direction, skip_neg_op)
   321

   323	def _infer_constraints(
   324	    template: Type, actual: Type, direction: int, skip_neg_op: bool
   325	) -> list[Constraint]:
   326	    orig_template = template
   327	    template = get_proper_type(template)
   328	    actual = get_proper_type(actual)
   329
   330	    # Type inference shouldn't be affected by whether union types have been simplified.
   331	    # We however keep any ErasedType items, so that the caller will see it when using
   332	    # checkexpr.has_erased_component().
   333	    if isinstance(template, UnionType):
   334	        template = mypy.typeops.make_simplified_union(template.items, keep_erased=True)
   335	    if isinstance(actual, UnionType):
   336	        actual = mypy.typeops.make_simplified_union(actual.items, keep_erased=True)
   337
   338	    # Ignore Any types from the type suggestion engine to avoid them
   339	    # causing us to infer Any in situations where a better job could
   340	    # be done otherwise. (This can produce false positives but that
   341	    # doesn't really matter because it is all heuristic anyway.)
   342	    if isinstance(actual, AnyType) and actual.type_of_any == TypeOfAny.suggestion_engine:
   343	        return []
   344
   345	    # type[A | B] is always represented as type[A] | type[B] internally.
   346	    # This makes our constraint solver choke on type[T] <: type[A] | type[B],
   347	    # solving T as generic meet(A, B) which is often `object`. Force unwrap such unions
   348	    # if both sides are type[...] or unions thereof. See `testTypeVarType` test
   349	    type_type_unwrapped = False
   350	    if _is_type_type(template) and _is_type_type(actual):
   351	        type_type_unwrapped = True
   352	        template = _unwrap_type_type(template)
   353	        actual = _unwrap_type_type(actual)
   354
   355	    # If the template is simply a type variable, emit a Constraint directly.
   356	    # We need to handle this case before handling Unions for two reasons:
   357	    #  1. "T <: Union[U1, U2]" is not equivalent to "T <: U1 or T <: U2",
   358	    #     because T can itself be a union (notably, Union[U1, U2] itself).
   359	    #  2. "T :> Union[U1, U2]" is logically equivalent to "T :> U1 and
   360	    #     T :> U2", but they are not equivalent to the constraint solver,
   361	    #     which never introduces new Union types (it uses join() instead).
   362	    if isinstance(template, TypeVarType):
   363	        return [Constraint(template, direction, actual)]
   364
   365	    if (
   366	        isinstance(actual, TypeVarType)
   367	        and not actual.id.is_meta_var()
   368	        and direction == SUPERTYPE_OF
   369	    ):
   370	        # Unless template is also a type variable (or a union that contains one), using the upper
   371	        # bound for inference will usually give better result for actual that is a type variable.
   372	        if not isinstance(template, UnionType) or not any(
   373	            isinstance(t, TypeVarType) for t in template.items
   374	        ):
   375	            actual = get_proper_type(actual.upper_bound)
   376
   377	    # Now handle the case of either template or actual being a Union.
   378	    # For a Union to be a subtype of another type, every item of the Union
   379	    # must be a subtype of that type, so concatenate the constraints.
   380	    if direction == SUBTYPE_OF and isinstance(template, UnionType):
   381	        res = []
   382	        for t_item in template.items:
   383	            res.extend(infer_constraints(t_item, actual, direction))
   384	        return res
   385	    if direction == SUPERTYPE_OF and isinstance(actual, UnionType):
   386	        res = []
   387	        for a_item in actual.items:
   388	            # `orig_template` has to be preserved intact in case it's recursive.
   389	            # If we unwrapped ``type[...]`` previously, wrap the item back again,
   390	            # as ``type[...]`` can't be removed from `orig_template`.
   391	            if type_type_unwrapped:
   392	                a_item = TypeType.make_normalized(a_item)
   393	            res.extend(infer_constraints(orig_template, a_item, direction))
   394	        return res
   395
   396	    # Now the potential subtype is known not to be a Union or a type
   397	    # variable that we are solving for. In that case, for a Union to
   398	    # be a supertype of the potential subtype, some item of the Union
   399	    # must be a supertype of it.
   400	    if direction == SUBTYPE_OF and isinstance(actual, UnionType):
   401	        # If some of items is not a complete type, disregard that.
   402	        items = simplify_away_incomplete_types(actual.items)
   403	        # We infer constraints eagerly -- try to find constraints for a type
   404	        # variable if possible. This seems to help with some real-world
   405	        # use cases.
   406	        return any_constraints(
   407	            [infer_constraints_if_possible(template, a_item, direction) for a_item in items],
   408	            eager=True,
   409	        )
   410	    if direction == SUPERTYPE_OF and isinstance(template, UnionType):
   411	        # When the template is a union, we are okay with leaving some
   412	        # type variables indeterminate. This helps with some special
   413	        # cases, though this isn't very principled.
   414	        result = any_constraints(
   415	            [
   416	                infer_constraints_if_possible(t_item, actual, direction)
   417	                for t_item in template.items
   418	            ],
   419	            eager=isinstance(actual, AnyType),
   420	        )
   421	        if result:
   422	            return result
   423	        elif has_recursive_types(template) and not has_recursive_types(actual):
   424	            return handle_recursive_union(template, actual, direction)
   425	        return []
   426
   427	    # Remaining cases are handled by ConstraintBuilderVisitor.
   428	    return template.accept(ConstraintBuilderVisitor(actual, direction, skip_neg_op))

   431	def _is_type_type(tp: ProperType) -> TypeGuard[TypeType | UnionType]:
   432	    """Is ``tp`` a ``type[...]`` or a union thereof?
   433
   434	    ``Type[A | B]`` is internally represented as ``type[A] | type[B]``, and this
   435	    troubles the solver sometimes.
   436	    """
   437	    return (
   438	        isinstance(tp, TypeType)
   439	        or isinstance(tp, UnionType)
   440	        and all(isinstance(get_proper_type(o), TypeType) for o in tp.items)
   441	    )

   444	def _unwrap_type_type(tp: TypeType | UnionType) -> ProperType:
   445	    """Extract the inner type from ``type[...]`` expression or a union thereof."""
   446	    if isinstance(tp, TypeType):
   447	        return tp.item
   448	    return UnionType.make_union([cast(TypeType, get_proper_type(o)).item for o in tp.items])

   451	def infer_constraints_if_possible(
   452	    template: Type, actual: Type, direction: int
   453	) -> list[Constraint] | None:
   454	    """Like infer_constraints, but return None if the input relation is
   455	    known to be unsatisfiable, for example if template=List[T] and actual=int.
   456	    (In this case infer_constraints would return [], just like it would for
   457	    an automatically satisfied relation like template=List[T] and actual=object.)
   458	    """
   459	    if direction == SUBTYPE_OF and not mypy.subtypes.is_subtype(erase_typevars(template), actual):
   460	        return None
   461	    if direction == SUPERTYPE_OF and not mypy.subtypes.is_subtype(
   462	        actual, erase_typevars(template)
   463	    ):
   464	        return None
   465	    if (
   466	        direction == SUPERTYPE_OF
   467	        and isinstance(template, TypeVarType)
   468	        and not mypy.subtypes.is_subtype(actual, erase_typevars(template.upper_bound))
   469	    ):
   470	        # This is not caught by the above branch because of the erase_typevars() call,
   471	        # that would return 'Any' for a type variable.
   472	        return None
   473	    return infer_constraints(template, actual, direction)

   476	def select_trivial(options: Sequence[list[Constraint] | None]) -> list[list[Constraint]]:
   477	    """Select only those lists where each item is a constraint against Any."""
   478	    res = []
   479	    for option in options:
   480	        if option is None:
   481	            continue
   482	        if all(isinstance(get_proper_type(c.target), AnyType) for c in option):
   483	            res.append(option)
   484	    return res

   487	def merge_with_any(constraint: Constraint) -> Constraint:
   488	    """Transform a constraint target into a union with given Any type."""
   489	    target = constraint.target
   490	    if is_union_with_any(target):
   491	        # Do not produce redundant unions.
   492	        return constraint
   493	    # TODO: if we will support multiple sources Any, use this here instead.
   494	    any_type = AnyType(TypeOfAny.implementation_artifact)
   495	    return Constraint(
   496	        constraint.origin_type_var,
   497	        constraint.op,
   498	        UnionType.make_union([target, any_type], target.line, target.column),
   499	    )

   502	def handle_recursive_union(template: UnionType, actual: Type, direction: int) -> list[Constraint]:
   503	    # This is a hack to special-case things like Union[T, Inst[T]] in recursive types. Although
   504	    # it is quite arbitrary, it is a relatively common pattern, so we should handle it well.
   505	    # This function may be called when inferring against such union resulted in different
   506	    # constraints for each item. Normally we give up in such case, but here we instead split
   507	    # the union in two parts, and try inferring sequentially.
   508	    non_type_var_items = [t for t in template.items if not isinstance(t, TypeVarType)]
   509	    type_var_items = [t for t in template.items if isinstance(t, TypeVarType)]
   510	    return infer_constraints(
   511	        UnionType.make_union(non_type_var_items), actual, direction
   512	    ) or infer_constraints(UnionType.make_union(type_var_items), actual, direction)

   515	def any_constraints(options: list[list[Constraint] | None], *, eager: bool) -> list[Constraint]:
   516	    """Deduce what we can from a collection of constraint lists.
   517
   518	    It's a given that at least one of the lists must be satisfied. A
   519	    None element in the list of options represents an unsatisfiable
   520	    constraint and is ignored.  Ignore empty constraint lists if eager
   521	    is true -- they are always trivially satisfiable.
   522	    """
   523	    if eager:
   524	        valid_options = [option for option in options if option]
   525	    else:
   526	        valid_options = [option for option in options if option is not None]
   527
   528	    if not valid_options:
   529	        return []
   530
   531	    if len(valid_options) == 1:
   532	        return valid_options[0]
   533
   534	    if all(is_same_constraints(valid_options[0], c) for c in valid_options[1:]):
   535	        # Multiple sets of constraints that are all the same. Just pick any one of them.
   536	        return valid_options[0]
   537
   538	    if all(is_similar_constraints(valid_options[0], c) for c in valid_options[1:]):
   539	        # All options have same structure. In this case we can merge-in trivial
   540	        # options (i.e. those that only have Any) and try again.
   541	        # TODO: More generally, if a given (variable, direction) pair appears in
   542	        # every option, combine the bounds with meet/join always, not just for Any.
   543	        trivial_options = select_trivial(valid_options)
   544	        if trivial_options and len(trivial_options) < len(valid_options):
   545	            merged_options = []
   546	            for option in valid_options:
   547	                if option in trivial_options:
   548	                    continue
   549	                merged_options.append([merge_with_any(c) for c in option])
   550	            return any_constraints(list(merged_options), eager=eager)
   551
   552	    # If normal logic didn't work, try excluding trivially unsatisfiable constraint (due to
   553	    # upper bounds) from each option, and comparing them again.
   554	    filtered_options = [filter_satisfiable(o) for o in options]
   555	    if filtered_options != options:
   556	        return any_constraints(filtered_options, eager=eager)
   557
   558	    # Try harder: if that didn't work, try to strip typevars that aren't meta vars.
   559	    # Note this is what we would always do, but unfortunately some callers may not
   560	    # set the meta var status correctly (for historical reasons), so we use this as
   561	    # a fallback only.
   562	    filtered_options = [exclude_non_meta_vars(o) for o in options]
   563	    if filtered_options != options:
   564	        return any_constraints(filtered_options, eager=eager)
   565
   566	    # Otherwise, there are either no valid options or multiple, inconsistent valid
   567	    # options. Give up and deduce nothing.
   568	    return []

   571	def filter_satisfiable(option: list[Constraint] | None) -> list[Constraint] | None:
   572	    """Keep only constraints that can possibly be satisfied.
   573
   574	    Currently, we filter out constraints where target is not a subtype of the upper bound.
   575	    Since those can be never satisfied. We may add more cases in future if it improves type
   576	    inference.
   577	    """
   578	    if not option:
   579	        return option
   580
   581	    satisfiable = []
   582	    for c in option:
   583	        if isinstance(c.origin_type_var, TypeVarType) and c.origin_type_var.values:
   584	            if any(
   585	                mypy.subtypes.is_subtype(c.target, value) for value in c.origin_type_var.values
   586	            ):
   587	                satisfiable.append(c)
   588	        elif mypy.subtypes.is_subtype(c.target, c.origin_type_var.upper_bound):
   589	            satisfiable.append(c)
   590	    if not satisfiable:
   591	        return None
   592	    return satisfiable

   595	def exclude_non_meta_vars(option: list[Constraint] | None) -> list[Constraint] | None:
   596	    # If we had an empty list, keep it intact
   597	    if not option:
   598	        return option
   599	    # However, if none of the options actually references meta vars, better remove
   600	    # this constraint entirely.
   601	    return [c for c in option if c.type_var.is_meta_var()] or None

   604	def is_same_constraints(x: list[Constraint], y: list[Constraint]) -> bool:
   605	    for c1 in x:
   606	        if not any(is_same_constraint(c1, c2) for c2 in y):
   607	            return False
   608	    for c1 in y:
   609	        if not any(is_same_constraint(c1, c2) for c2 in x):
   610	            return False
   611	    return True

   614	def is_same_constraint(c1: Constraint, c2: Constraint) -> bool:
   615	    # Ignore direction when comparing constraints against Any.
   616	    skip_op_check = isinstance(get_proper_type(c1.target), AnyType) and isinstance(
   617	        get_proper_type(c2.target), AnyType
   618	    )
   619	    return (
   620	        c1.type_var == c2.type_var
   621	        and (c1.op == c2.op or skip_op_check)
   622	        and mypy.subtypes.is_same_type(c1.target, c2.target)
   623	    )

   626	def is_similar_constraints(x: list[Constraint], y: list[Constraint]) -> bool:
   627	    """Check that two lists of constraints have similar structure.
   628
   629	    This means that each list has same type variable plus direction pairs (i.e we
   630	    ignore the target). Except for constraints where target is Any type, there
   631	    we ignore direction as well.
   632	    """
   633	    return _is_similar_constraints(x, y) and _is_similar_constraints(y, x)

   636	def _is_similar_constraints(x: list[Constraint], y: list[Constraint]) -> bool:
   637	    """Check that every constraint in the first list has a similar one in the second.
   638
   639	    See docstring above for definition of similarity.
   640	    """
   641	    for c1 in x:
   642	        has_similar = False
   643	        for c2 in y:
   644	            # Ignore direction when either constraint is against Any.
   645	            skip_op_check = isinstance(get_proper_type(c1.target), AnyType) or isinstance(
   646	                get_proper_type(c2.target), AnyType
   647	            )
   648	            if c1.type_var == c2.type_var and (c1.op == c2.op or skip_op_check):
   649	                has_similar = True
   650	                break
   651	        if not has_similar:
   652	            return False
   653	    return True

   681	class ConstraintBuilderVisitor(TypeVisitor[list[Constraint]]):
   682	    """Visitor class for inferring type constraints."""
   683
   684	    # The type that is compared against a template
   685	    # TODO: The value may be None. Is that actually correct?
   686	    actual: ProperType
   687
   688	    def __init__(self, actual: ProperType, direction: int, skip_neg_op: bool) -> None:
   689	        # Direction must be SUBTYPE_OF or SUPERTYPE_OF.
   690	        self.actual = actual
   691	        self.direction = direction
   692	        # Whether to skip polymorphic inference (involves inference in opposite direction)
   693	        # this is used to prevent infinite recursion when both template and actual are
   694	        # generic callables.
   695	        self.skip_neg_op = skip_neg_op
   696
   697	    # Trivial leaf types
   698
   699	    def visit_unbound_type(self, template: UnboundType) -> list[Constraint]:
   700	        return []
   701
   702	    def visit_any(self, template: AnyType) -> list[Constraint]:
   703	        return []
   704
   705	    def visit_none_type(self, template: NoneType) -> list[Constraint]:
   706	        return []
   707
   708	    def visit_uninhabited_type(self, template: UninhabitedType) -> list[Constraint]:
   709	        return []
   710
   711	    def visit_erased_type(self, template: ErasedType) -> list[Constraint]:
   712	        return []
   713
   714	    def visit_deleted_type(self, template: DeletedType) -> list[Constraint]:
   715	        return []
   716
   717	    def visit_literal_type(self, template: LiteralType) -> list[Constraint]:
   718	        return []
   719
   720	    # Errors
   721
   722	    def visit_partial_type(self, template: PartialType) -> list[Constraint]:
   723	        # We can't do anything useful with a partial type here.
   724	        assert False, "Internal error"
   725
   726	    # Non-trivial leaf type
   727
   728	    def visit_type_var(self, template: TypeVarType) -> list[Constraint]:
   729	        assert False, (
   730	            "Unexpected TypeVarType in ConstraintBuilderVisitor"
   731	            " (should have been handled in infer_constraints)"
   732	        )
   733
   734	    def visit_param_spec(self, template: ParamSpecType) -> list[Constraint]:
   735	        # Can't infer ParamSpecs from component values (only via Callable[P, T]).
   736	        return []
   737
   738	    def visit_type_var_tuple(self, template: TypeVarTupleType) -> list[Constraint]:
   739	        raise NotImplementedError
   740
   741	    def visit_unpack_type(self, template: UnpackType) -> list[Constraint]:
   742	        raise RuntimeError("Mypy bug: unpack should be handled at a higher level.")
   743
   744	    def visit_parameters(self, template: Parameters) -> list[Constraint]:
   745	        # Constraining Any against C[P] turns into infer_against_any([P], Any)
   746	        if isinstance(self.actual, AnyType):
   747	            return self.infer_against_any(template.arg_types, self.actual)
   748	        if type_state.infer_polymorphic and isinstance(self.actual, Parameters):
   749	            # For polymorphic inference we need to be able to infer secondary constraints
   750	            # in situations like [x: T] <: P <: [x: int].
   751	            return infer_callable_arguments_constraints(template, self.actual, self.direction)
   752	        if type_state.infer_polymorphic and isinstance(self.actual, ParamSpecType):
   753	            # Similar for [x: T] <: Q <: Concatenate[int, P].
   754	            return infer_callable_arguments_constraints(
   755	                template, self.actual.prefix, self.direction
   756	            )
   757	        # There also may be unpatched types after a user error, simply ignore them.
   758	        return []
   759
   760	    # Non-leaf types
   761
   762	    def visit_instance(self, template: Instance) -> list[Constraint]:
   763	        original_actual = actual = self.actual
   764	        res: list[Constraint] = []
   765	        if isinstance(actual, (CallableType, Overloaded)) and template.type.is_protocol:
   766	            if "__call__" in template.type.protocol_members:
   767	                # Special case: a generic callback protocol
   768	                if not any(template == t for t in template.type.inferring):
   769	                    template.type.inferring.append(template)
   770	                    call = mypy.subtypes.find_member(
   771	                        "__call__", template, actual, is_operator=True
   772	                    )
   773	                    assert call is not None
   774	                    if (
   775	                        self.direction == SUPERTYPE_OF
   776	                        and mypy.subtypes.is_subtype(actual, erase_typevars(call))
   777	                        or self.direction == SUBTYPE_OF
   778	                        and mypy.subtypes.is_subtype(erase_typevars(call), actual)
   779	                    ):
   780	                        res.extend(infer_constraints(call, actual, self.direction))
   781	                    template.type.inferring.pop()
   782	        if isinstance(actual, CallableType) and actual.fallback is not None:
   783	            if (
   784	                actual.is_type_obj()
   785	                and template.type.is_protocol
   786	                and self.direction == SUPERTYPE_OF
   787	            ):
   788	                ret_type = get_proper_type(actual.ret_type)
   789	                if isinstance(ret_type, TupleType):
   790	                    ret_type = mypy.typeops.tuple_fallback(ret_type)
   791	                if isinstance(ret_type, Instance):
   792	                    res.extend(
   793	                        self.infer_constraints_from_protocol_members(
   794	                            ret_type, template, ret_type, template, class_obj=True
   795	                        )
   796	                    )
   797	            actual = actual.fallback
   798	        if isinstance(actual, TypeType) and template.type.is_protocol:
   799	            if self.direction == SUPERTYPE_OF:
   800	                a_item = actual.item
   801	                if isinstance(a_item, Instance):
   802	                    res.extend(
   803	                        self.infer_constraints_from_protocol_members(
   804	                            a_item, template, a_item, template, class_obj=True
   805	                        )
   806	                    )
   807	                # Infer constraints for Type[T] via metaclass of T when it makes sense.
   808	                if isinstance(a_item, TypeVarType):
   809	                    a_item = get_proper_type(a_item.upper_bound)
   810	                if isinstance(a_item, Instance) and a_item.type.metaclass_type:
   811	                    res.extend(
   812	                        self.infer_constraints_from_protocol_members(
   813	                            a_item.type.metaclass_type, template, actual, template
   814	                        )
   815	                    )
   816
   817	        if isinstance(actual, Overloaded) and actual.fallback is not None:
   818	            actual = actual.fallback
   819	        if isinstance(actual, TypedDictType):
   820	            actual = actual.as_anonymous().fallback
   821	        if isinstance(actual, LiteralType):
   822	            actual = actual.fallback
   823	        if isinstance(actual, Instance):
   824	            instance = actual
   825	            erased = erase_typevars(template)
   826	            assert isinstance(erased, Instance)  # type: ignore[misc]
   827	            # We always try nominal inference if possible,
   828	            # it is much faster than the structural one.
   829	            if self.direction == SUBTYPE_OF and template.type.has_base(instance.type.fullname):
   830	                mapped = map_instance_to_supertype(template, instance.type)
   831	                tvars = mapped.type.defn.type_vars
   832
   833	                if instance.type.has_type_var_tuple_type:
   834	                    # Variadic types need special handling to map each type argument to
   835	                    # the correct corresponding type variable.
   836	                    assert instance.type.type_var_tuple_prefix is not None
   837	                    assert instance.type.type_var_tuple_suffix is not None
   838	                    prefix_len = instance.type.type_var_tuple_prefix
   839	                    suffix_len = instance.type.type_var_tuple_suffix
   840	                    tvt = instance.type.defn.type_vars[prefix_len]
   841	                    assert isinstance(tvt, TypeVarTupleType)
   842	                    fallback = tvt.tuple_fallback
   843	                    i_prefix, i_middle, i_suffix = split_with_prefix_and_suffix(
   844	                        instance.args, prefix_len, suffix_len
   845	                    )
   846	                    m_prefix, m_middle, m_suffix = split_with_prefix_and_suffix(
   847	                        mapped.args, prefix_len, suffix_len
   848	                    )
   849	                    instance_args = i_prefix + (TupleType(list(i_middle), fallback),) + i_suffix
   850	                    mapped_args = m_prefix + (TupleType(list(m_middle), fallback),) + m_suffix
   851	                else:
   852	                    mapped_args = mapped.args
   853	                    instance_args = instance.args
   854
   855	                # N.B: We use zip instead of indexing because the lengths might have
   856	                # mismatches during daemon reprocessing.
   857	                for tvar, mapped_arg, instance_arg in zip(tvars, mapped_args, instance_args):
   858	                    if isinstance(tvar, TypeVarType):
   859	                        # The constraints for generic type parameters depend on variance.
   860	                        # Include constraints from both directions if invariant.
   861	                        if tvar.variance != CONTRAVARIANT:
   862	                            res.extend(infer_constraints(mapped_arg, instance_arg, self.direction))
   863	                        if tvar.variance != COVARIANT:
   864	                            res.extend(
   865	                                infer_constraints(mapped_arg, instance_arg, neg_op(self.direction))
   866	                            )
   867	                    elif isinstance(tvar, ParamSpecType) and isinstance(mapped_arg, ParamSpecType):
   868	                        prefix = mapped_arg.prefix
   869	                        if isinstance(instance_arg, Parameters):
   870	                            # No such thing as variance for ParamSpecs, consider them invariant
   871	                            # TODO: constraints between prefixes using
   872	                            # infer_callable_arguments_constraints()
   873	                            suffix: Type = instance_arg.copy_modified(
   874	                                instance_arg.arg_types[len(prefix.arg_types) :],
   875	                                instance_arg.arg_kinds[len(prefix.arg_kinds) :],
   876	                                instance_arg.arg_names[len(prefix.arg_names) :],
   877	                            )
   878	                            res.append(Constraint(mapped_arg, SUBTYPE_OF, suffix))
   879	                            res.append(Constraint(mapped_arg, SUPERTYPE_OF, suffix))
   880	                        elif isinstance(instance_arg, ParamSpecType):
   881	                            suffix = instance_arg.copy_modified(
   882	                                prefix=Parameters(
   883	                                    instance_arg.prefix.arg_types[len(prefix.arg_types) :],
   884	                                    instance_arg.prefix.arg_kinds[len(prefix.arg_kinds) :],
   885	                                    instance_arg.prefix.arg_names[len(prefix.arg_names) :],
   886	                                )
   887	                            )
   888	                            res.append(Constraint(mapped_arg, SUBTYPE_OF, suffix))
   889	                            res.append(Constraint(mapped_arg, SUPERTYPE_OF, suffix))
   890	                    elif isinstance(tvar, TypeVarTupleType):
   891	                        # Handle variadic type variables covariantly for consistency.
   892	                        res.extend(infer_constraints(mapped_arg, instance_arg, self.direction))
   893
   894	                return res
   895	            elif self.direction == SUPERTYPE_OF and instance.type.has_base(template.type.fullname):
   896	                mapped = map_instance_to_supertype(instance, template.type)
   897	                tvars = template.type.defn.type_vars
   898	                if template.type.has_type_var_tuple_type:
   899	                    # Variadic types need special handling to map each type argument to
   900	                    # the correct corresponding type variable.
   901	                    assert template.type.type_var_tuple_prefix is not None
   902	                    assert template.type.type_var_tuple_suffix is not None
   903	                    prefix_len = template.type.type_var_tuple_prefix
   904	                    suffix_len = template.type.type_var_tuple_suffix
   905	                    tvt = template.type.defn.type_vars[prefix_len]
   906	                    assert isinstance(tvt, TypeVarTupleType)
   907	                    fallback = tvt.tuple_fallback
   908	                    t_prefix, t_middle, t_suffix = split_with_prefix_and_suffix(
   909	                        template.args, prefix_len, suffix_len
   910	                    )
   911	                    m_prefix, m_middle, m_suffix = split_with_prefix_and_suffix(
   912	                        mapped.args, prefix_len, suffix_len
   913	                    )
   914	                    template_args = t_prefix + (TupleType(list(t_middle), fallback),) + t_suffix
   915	                    mapped_args = m_prefix + (TupleType(list(m_middle), fallback),) + m_suffix
   916	                else:
   917	                    mapped_args = mapped.args
   918	                    template_args = template.args
   919	                # N.B: We use zip instead of indexing because the lengths might have
   920	                # mismatches during daemon reprocessing.
   921	                for tvar, mapped_arg, template_arg in zip(tvars, mapped_args, template_args):
   922	                    if isinstance(tvar, TypeVarType):
   923	                        # The constraints for generic type parameters depend on variance.
   924	                        # Include constraints from both directions if invariant.
   925	                        if tvar.variance != CONTRAVARIANT:
   926	                            res.extend(infer_constraints(template_arg, mapped_arg, self.direction))
   927	                        if tvar.variance != COVARIANT:
   928	                            res.extend(
   929	                                infer_constraints(template_arg, mapped_arg, neg_op(self.direction))
   930	                            )
   931	                    elif isinstance(tvar, ParamSpecType) and isinstance(
   932	                        template_arg, ParamSpecType
   933	                    ):
   934	                        prefix = template_arg.prefix
   935	                        if isinstance(mapped_arg, Parameters):
   936	                            # No such thing as variance for ParamSpecs, consider them invariant
   937	                            # TODO: constraints between prefixes using
   938	                            # infer_callable_arguments_constraints()
   939	                            suffix = mapped_arg.copy_modified(
   940	                                mapped_arg.arg_types[len(prefix.arg_types) :],
   941	                                mapped_arg.arg_kinds[len(prefix.arg_kinds) :],
   942	                                mapped_arg.arg_names[len(prefix.arg_names) :],
   943	                            )
   944	                            res.append(Constraint(template_arg, SUBTYPE_OF, suffix))
   945	                            res.append(Constraint(template_arg, SUPERTYPE_OF, suffix))
   946	                        elif isinstance(mapped_arg, ParamSpecType):
   947	                            suffix = mapped_arg.copy_modified(
   948	                                prefix=Parameters(
   949	                                    mapped_arg.prefix.arg_types[len(prefix.arg_types) :],
   950	                                    mapped_arg.prefix.arg_kinds[len(prefix.arg_kinds) :],
   951	                                    mapped_arg.prefix.arg_names[len(prefix.arg_names) :],
   952	                                )
   953	                            )
   954	                            res.append(Constraint(template_arg, SUBTYPE_OF, suffix))
   955	                            res.append(Constraint(template_arg, SUPERTYPE_OF, suffix))
   956	                    elif isinstance(tvar, TypeVarTupleType):
   957	                        # Consider variadic type variables to be invariant.
   958	                        res.extend(infer_constraints(template_arg, mapped_arg, SUBTYPE_OF))
   959	                        res.extend(infer_constraints(template_arg, mapped_arg, SUPERTYPE_OF))
   960	                return res
   961	            if (
   962	                template.type.is_protocol
   963	                and self.direction == SUPERTYPE_OF
   964	                and
   965	                # We avoid infinite recursion for structural subtypes by checking
   966	                # whether this type already appeared in the inference chain.
   967	                # This is a conservative way to break the inference cycles.
   968	                # It never produces any "false" constraints but gives up soon
   969	                # on purely structural inference cycles, see #3829.
   970	                # Note that we use is_protocol_implementation instead of is_subtype
   971	                # because some type may be considered a subtype of a protocol
   972	                # due to _promote, but still not implement the protocol.
   973	                not any(template == t for t in reversed(template.type.inferring))
   974	                and mypy.subtypes.is_protocol_implementation(instance, erased, skip=["__call__"])
   975	            ):
   976	                template.type.inferring.append(template)
   977	                res.extend(
   978	                    self.infer_constraints_from_protocol_members(
   979	                        instance, template, original_actual, template
   980	                    )
   981	                )
   982	                template.type.inferring.pop()
   983	                return res
   984	            elif (
   985	                instance.type.is_protocol
   986	                and self.direction == SUBTYPE_OF
   987	                and
   988	                # We avoid infinite recursion for structural subtypes also here.
   989	                not any(instance == i for i in reversed(instance.type.inferring))
   990	                and mypy.subtypes.is_protocol_implementation(erased, instance, skip=["__call__"])
   991	            ):
   992	                instance.type.inferring.append(instance)
   993	                res.extend(
   994	                    self.infer_constraints_from_protocol_members(
   995	                        instance, template, template, instance
   996	                    )
   997	                )
   998	                instance.type.inferring.pop()
   999	                return res
  1000	        if res:
  1001	            return res
  1002
  1003	        if isinstance(actual, AnyType):
  1004	            return self.infer_against_any(template.args, actual)
  1005	        if (
  1006	            isinstance(actual, TupleType)
  1007	            and is_named_instance(template, TUPLE_LIKE_INSTANCE_NAMES)
  1008	            and self.direction == SUPERTYPE_OF
  1009	        ):
  1010	            for item in actual.items:
  1011	                if isinstance(item, UnpackType):
  1012	                    unpacked = get_proper_type(item.type)
  1013	                    if isinstance(unpacked, TypeVarTupleType):
  1014	                        # Cannot infer anything for T from [T, ...] <: *Ts
  1015	                        continue
  1016	                    assert (
  1017	                        isinstance(unpacked, Instance)
  1018	                        and unpacked.type.fullname == "builtins.tuple"
  1019	                    )
  1020	                    item = unpacked.args[0]
  1021	                cb = infer_constraints(template.args[0], item, SUPERTYPE_OF)
  1022	                res.extend(cb)
  1023	            return res
  1024	        elif isinstance(actual, TupleType) and self.direction == SUPERTYPE_OF:
  1025	            return infer_constraints(template, mypy.typeops.tuple_fallback(actual), self.direction)
  1026	        elif isinstance(actual, TypeVarType):
  1027	            if not actual.values and not actual.id.is_meta_var():
  1028	                return infer_constraints(template, actual.upper_bound, self.direction)
  1029	            return []
  1030	        elif isinstance(actual, ParamSpecType):
  1031	            return infer_constraints(template, actual.upper_bound, self.direction)
  1032	        elif isinstance(actual, TypeVarTupleType):
  1033	            raise NotImplementedError
  1034	        else:
  1035	            return []

  1037	    def infer_constraints_from_protocol_members(
  1038	        self,
  1039	        instance: Instance,
  1040	        template: Instance,
  1041	        subtype: Type,
  1042	        protocol: Instance,
  1043	        class_obj: bool = False,
  1044	    ) -> list[Constraint]:
  1045	        """Infer constraints for situations where either 'template' or 'instance' is a protocol.
  1046
  1047	        The 'protocol' is the one of two that is an instance of protocol type, 'subtype'
  1048	        is the type used to bind self during inference. Currently, we just infer constrains for
  1049	        every protocol member type (both ways for settable members).
  1050	        """
  1051	        res = []
  1052	        for member in protocol.type.protocol_members:
  1053	            inst = mypy.subtypes.find_member(member, instance, subtype, class_obj=class_obj)
  1054	            temp = mypy.subtypes.find_member(member, template, subtype)
  1055	            if inst is None or temp is None:
  1056	                if member == "__call__":
  1057	                    continue
  1058	                return []  # See #11020
  1059	            # The above is safe since at this point we know that 'instance' is a subtype
  1060	            # of (erased) 'template', therefore it defines all protocol members
  1061	            if class_obj:
  1062	                # For class objects we must only infer constraints if possible, otherwise it
  1063	                # can lead to confusion between class and instance, for example StrEnum is
  1064	                # Iterable[str] for an instance, but Iterable[StrEnum] for a class object.
  1065	                if not mypy.subtypes.is_subtype(
  1066	                    inst, erase_typevars(temp), ignore_pos_arg_names=True
  1067	                ):
  1068	                    continue
  1069	            # This exception matches the one in typeops.py, see PR #14121 for context.

  1247	    def visit_tuple_type(self, template: TupleType) -> list[Constraint]:
  1248	        actual = self.actual
  1249	        unpack_index = find_unpack_in_list(template.items)
  1250	        is_varlength_tuple = (
  1251	            isinstance(actual, Instance) and actual.type.fullname == "builtins.tuple"

  1254	        if isinstance(actual, TupleType) or is_varlength_tuple:
  1255	            res: list[Constraint] = []
  1256	            if unpack_index is not None:
  1257	                if is_varlength_tuple:
  1258	                    # Variadic tuple can be only a supertype of a tuple type, but even if
  1259	                    # direction is opposite, inferring something may give better error messages.
  1260	                    unpack_type = template.items[unpack_index]

  1262	                    unpacked_type = get_proper_type(unpack_type.type)

  1264	                        res = [
  1265	                            Constraint(type_var=unpacked_type, op=self.direction, target=actual)
  1266	                        ]

  1268	                        assert (
  1269	                            isinstance(unpacked_type, Instance)
  1270	                            and unpacked_type.type.fullname == "builtins.tuple"
  1271	                        )

  1276	                            # This one we just handled above.
  1277	                            continue

  1279	                        res.extend(infer_constraints(ti, actual.args[0], self.direction))

  1283	                    unpack_constraints = build_constraints_for_simple_unpack(

  1293	                    # has an unpack. We can infer something if actual unpack is a variadic tuple.
  1294	                    # Tuple[T, S, U] <: tuple[X, *tuple[Y, ...], Z] => T <: X, S <: Y, U <: Z.

  1302	                            tuple(template.items), a_prefix_len, a_suffix_len
  1303	                        )

  1311	                                res.extend(
  1312	                                    infer_constraints(tm, a_unpacked.args[0], self.direction)
  1313	                                )
  1314	                    else:
  1315	                        actual_items = ()
  1316	                        template_items = ()

  1320	            else:

  1322
  1323	            # Cases above will return if actual wasn't a TupleType.
  1324	            assert isinstance(actual, TupleType)

  1331	                    return res + infer_constraints(

  1339	        elif isinstance(actual, AnyType):

  1422	    items = overloaded.items
  1423	    for item in items:
  1424	        # Return type may be indeterminate in the template, so ignore it when performing a
  1425	        # subtype check.

  1427	            item,

  1431	            ignore_return=True,

  1477	def repack_callable_args(callable: CallableType, tuple_type: TypeInfo) -> list[Type]:
  1478	    """Present callable with star unpack in a normalized form.
  1479
  1480	    Since positional arguments cannot follow star argument, they are packed in a suffix,
  1481	    while prefix is represented as individual positional args. We want to put all in a single
  1482	    list with unpack in the middle, and prefix/suffix on the sides (as they would appear
  1483	    in e.g. a TupleType).
  1484	    """
  1485	    if ARG_STAR not in callable.arg_kinds:
  1486	        return callable.arg_types
  1487	    star_index = callable.arg_kinds.index(ARG_STAR)
  1488	    arg_types = callable.arg_types[:star_index]
  1489	    star_type = callable.arg_types[star_index]
  1490	    suffix_types = []
  1491	    if not isinstance(star_type, UnpackType):
  1492	        # Re-normalize *args: X -> *args: *tuple[X, ...]
  1493	        star_type = UnpackType(Instance(tuple_type, [star_type]))
  1494	    else:
  1495	        tp = get_proper_type(star_type.type)
  1496	        if isinstance(tp, TupleType):
  1497	            assert isinstance(tp.items[0], UnpackType)
  1498	            star_type = tp.items[0]
  1499	            suffix_types = tp.items[1:]
  1500	    return arg_types + [star_type] + suffix_types

  1503	def build_constraints_for_simple_unpack(
  1504	    template_args: list[Type], actual_args: list[Type], direction: int
  1505	) -> list[Constraint]:
  1506	    """Infer constraints between two lists of types with variadic items.
  1507
  1508	    This function is only supposed to be called when a variadic item is present in templates.
  1509	    If there is no variadic item the actuals, we simply use split_with_prefix_and_suffix()
  1510	    and infer prefix <: prefix, suffix <: suffix, variadic <: middle. If there is a variadic
  1511	    item in the actuals we need to be more careful, only common prefix/suffix can generate
  1512	    constraints, also we can only infer constraints for variadic template item, if template
  1513	    prefix/suffix are shorter that actual ones, otherwise there may be partial overlap
  1514	    between variadic items, for example if template prefix is longer:
  1515
  1516	        templates: T1, T2, Ts, Ts, Ts, ...
  1517	        actuals:   A1, As, As, As, ...
  1518
  1519	    Note: this function can only be called for builtin variadic constructors: Tuple and Callable.
  1520	    For instances, you should first find correct type argument mapping.
  1521	    """
  1522	    template_unpack = find_unpack_in_list(template_args)
  1523	    assert template_unpack is not None
  1524	    template_prefix = template_unpack
  1525	    template_suffix = len(template_args) - template_prefix - 1
  1526
  1527	    t_unpack = None
  1528	    res = []
  1529
  1530	    actual_unpack = find_unpack_in_list(actual_args)
  1531	    if actual_unpack is None:
  1532	        t_unpack = template_args[template_unpack]
  1533	        if template_prefix + template_suffix > len(actual_args):
  1534	            # These can't be subtypes of each-other, return fast.
  1535	            assert isinstance(t_unpack, UnpackType)
  1536	            if isinstance(t_unpack.type, TypeVarTupleType):
  1537	                # Set TypeVarTuple to empty to improve error messages.
  1538	                return [
  1539	                    Constraint(
  1540	                        t_unpack.type, direction, TupleType([], t_unpack.type.tuple_fallback)
  1541	                    )
  1542	                ]
  1543	            else:
  1544	                return []
  1545	        common_prefix = template_prefix
  1546	        common_suffix = template_suffix
  1547	    else:

  1549	        actual_suffix = len(actual_args) - actual_prefix - 1
  1550	        common_prefix = min(template_prefix, actual_prefix)
  1551	        common_suffix = min(template_suffix, actual_suffix)
  1552	        if actual_prefix >= template_prefix and actual_suffix >= template_suffix:
  1553	            # This is the only case where we can guarantee there will be no partial overlap
  1554	            # (note however partial overlap is OK for variadic tuples, it is handled below).
  1555	            t_unpack = template_args[template_unpack]
  1556
  1557	    # Handle constraints from prefixes/suffixes first.
  1558	    start, middle, end = split_with_prefix_and_suffix(
  1559	        tuple(actual_args), common_prefix, common_suffix
  1560	    )
  1561	    for t, a in zip(template_args[:common_prefix], start):
  1562	        res.extend(infer_constraints(t, a, direction))
  1563	    if common_suffix:
  1564	        for t, a in zip(template_args[-common_suffix:], end):
  1565	            res.extend(infer_constraints(t, a, direction))
  1566
  1567	    if t_unpack is not None:
  1568	        # Add constraint(s) for variadic item when possible.
  1569	        assert isinstance(t_unpack, UnpackType)
  1570	        tp = get_proper_type(t_unpack.type)
  1571	        if isinstance(tp, Instance) and tp.type.fullname == "builtins.tuple":
  1572	            # Homogeneous case *tuple[T, ...] <: [X, Y, Z, ...].
  1573	            for a in middle:
  1574	                # TODO: should we use union instead of join here?
  1575	                if not isinstance(a, UnpackType):
  1576	                    res.extend(infer_constraints(tp.args[0], a, direction))
  1577	                else:
  1578	                    a_tp = get_proper_type(a.type)

  1580	                    if isinstance(a_tp, Instance) and a_tp.type.fullname == "builtins.tuple":
  1581	                        res.extend(infer_constraints(tp.args[0], a_tp.args[0], direction))
  1582	        elif isinstance(tp, TypeVarTupleType):
  1583	            res.append(Constraint(tp, direction, TupleType(list(middle), tp.tuple_fallback)))
  1584	    elif actual_unpack is not None:
  1585	        # A special case for a variadic tuple unpack, we simply infer T <: X from
  1586	        # Tuple[..., *tuple[T, ...], ...] <: Tuple[..., *tuple[X, ...], ...].

  1588	        assert isinstance(actual_unpack_type, UnpackType)
  1589	        a_unpacked = get_proper_type(actual_unpack_type.type)
  1590	        if isinstance(a_unpacked, Instance) and a_unpacked.type.fullname == "builtins.tuple":

  1592	            assert isinstance(t_unpack, UnpackType)
  1593	            tp = get_proper_type(t_unpack.type)
  1594	            if isinstance(tp, Instance) and tp.type.fullname == "builtins.tuple":
  1595	                res.extend(infer_constraints(tp.args[0], a_unpacked.args[0], direction))

  1599	def infer_directed_arg_constraints(left: Type, right: Type, direction: int) -> list[Constraint]:
  1600	    """Infer constraints between two arguments using direction between original callables."""
  1601	    if isinstance(left, (ParamSpecType, UnpackType)) or isinstance(
  1602	        right, (ParamSpecType, UnpackType)
  1603	    ):
  1604	        # This avoids bogus constraints like T <: P.args
  1605	        # TODO: can we infer something useful for *T vs P?
  1606	        return []
  1607	    if direction == SUBTYPE_OF:
  1608	        # We invert direction to account for argument contravariance.
  1609	        return infer_constraints(left, right, neg_op(direction))
  1610	    else:
  1611	        return infer_constraints(right, left, neg_op(direction))

  1614	def infer_callable_arguments_constraints(
  1615	    template: NormalizedCallableType | Parameters,
  1616	    actual: NormalizedCallableType | Parameters,
  1617	    direction: int,
  1618	) -> list[Constraint]:
  1619	    """Infer constraints between argument types of two callables.
  1620
  1621	    This function essentially extracts four steps from are_parameters_compatible() in
  1622	    subtypes.py that involve subtype checks between argument types. We keep the argument
  1623	    matching logic, but ignore various strictness flags present there, and checks that
  1624	    do not involve subtyping. Then in place of every subtype check we put an infer_constraints()
  1625	    call for the same types.
  1626	    """
  1627	    res = []
  1628	    if direction == SUBTYPE_OF:
  1629	        left, right = template, actual
  1630	    else:
  1631	        left, right = actual, template
  1632	    left_star = left.var_arg()
  1633	    left_star2 = left.kw_arg()
  1634	    right_star = right.var_arg()
  1635	    right_star2 = right.kw_arg()
  1636
  1637	    # Numbering of steps below matches the one in are_parameters_compatible() for convenience.
  1638	    # Phase 1a: compare star vs star arguments.
  1639	    if left_star is not None and right_star is not None:
  1640	        res.extend(infer_directed_arg_constraints(left_star.typ, right_star.typ, direction))
  1641	    if left_star2 is not None and right_star2 is not None:
  1642	        res.extend(infer_directed_arg_constraints(left_star2.typ, right_star2.typ, direction))
  1643
  1644	    # Phase 1b: compare left args with corresponding non-star right arguments.
  1645	    for right_arg in right.formal_arguments():
  1646	        left_arg = mypy.typeops.callable_corresponding_argument(left, right_arg)
  1647	        if left_arg is None:
  1648	            continue
  1649	        res.extend(infer_directed_arg_constraints(left_arg.typ, right_arg.typ, direction))
  1650
  1651	    # Phase 1c: compare left args with right *args.
  1652	    if right_star is not None:
  1653	        right_by_position = right.try_synthesizing_arg_from_vararg(None)
  1654	        assert right_by_position is not None
  1655	        i = right_star.pos
  1656	        assert i is not None
  1657	        while i < len(left.arg_kinds) and left.arg_kinds[i].is_positional():
  1658	            left_by_position = left.argument_by_position(i)
  1659	            assert left_by_position is not None
  1660	            res.extend(
  1661	                infer_directed_arg_constraints(
  1662	                    left_by_position.typ, right_by_position.typ, direction
  1663	                )
  1664	            )
  1665	            i += 1
  1666
  1667	    # Phase 1d: compare left args with right **kwargs.
  1668	    if right_star2 is not None:
  1669	        right_names = {name for name in right.arg_names if name is not None}
  1670	        left_only_names = set()
  1671	        for name, kind in zip(left.arg_names, left.arg_kinds):
  1672	            if name is None or kind.is_star() or name in right_names:
  1673	                continue
  1674	            left_only_names.add(name)
  1675
  1676	        right_by_name = right.try_synthesizing_arg_from_kwarg(None)
  1677	        assert right_by_name is not None
  1678	        for name in left_only_names:
  1679	            left_by_name = left.argument_by_name(name)
  1680	            assert left_by_name is not None
  1681	            res.extend(
  1682	                infer_directed_arg_constraints(left_by_name.typ, right_by_name.typ, direction)
  1683	            )
  1684	    return res

  1688	    """For each ParamSpec remove all imprecise constraints, if at least one precise available."""
  1689	    have_precise = set()
  1690	    for c in cs:
  1691	        if not isinstance(c.origin_type_var, ParamSpecType):
  1692	            continue
  1693	        if (
  1694	            isinstance(c.target, ParamSpecType)
  1695	            or isinstance(c.target, Parameters)

  1702	            new_cs.append(c)
  1703	        if not isinstance(c.target, Parameters) or not c.target.imprecise_arg_kinds:
  1704	            new_cs.append(c)
