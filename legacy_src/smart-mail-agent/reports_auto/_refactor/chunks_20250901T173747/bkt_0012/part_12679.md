    51	    the variable could not be solved.

    54	    pick UninhabitedType as the value of the type variable. If strict=False, pick AnyType.

    56	    free type variables in solutions (these require special care when applying). Otherwise,
    57	    use a simplified algorithm that just solves each type variable individually if possible.

    63	        return [], []

   129	def solve_with_dependent(
   130	    vars: list[TypeVarId],
   131	    constraints: list[Constraint],
   132	    original_vars: list[TypeVarId],
   133	    originals: dict[TypeVarId, TypeVarLikeType],
   134	) -> tuple[Solutions, list[TypeVarLikeType]]:
   135	    """Solve set of constraints that may depend on each other, like T <: List[S].
   136
   137	    The whole algorithm consists of five steps:
   138	      * Propagate via linear constraints and use secondary constraints to get transitive closure
   139	      * Find dependencies between type variables, group them in SCCs, and sort topologically
   140	      * Check that all SCC are intrinsically linear, we can't solve (express) T <: List[T]
   141	      * Variables in leaf SCCs that don't have constant bounds are free (choose one per SCC)
   142	      * Solve constraints iteratively starting from leaves, updating bounds after each step.
   143	    """
   144	    graph, lowers, uppers = transitive_closure(vars, constraints)
   145
   146	    dmap = compute_dependencies(vars, graph, lowers, uppers)
   147	    sccs = list(strongly_connected_components(set(vars), dmap))
   148	    if not all(check_linear(scc, lowers, uppers) for scc in sccs):
   149	        return {}, []
   150	    raw_batches = list(topsort(prepare_sccs(sccs, dmap)))
   151
   152	    free_vars = []
   153	    free_solutions = {}
   154	    for scc in raw_batches[0]:
   155	        # If there are no bounds on this SCC, then the only meaningful solution we can
   156	        # express, is that each variable is equal to a new free variable. For example,
   157	        # if we have T <: S, S <: U, we deduce: T = S = U = <free>.
   158	        if all(not lowers[tv] and not uppers[tv] for tv in scc):
   159	            best_free = choose_free([originals[tv] for tv in scc], original_vars)
   160	            if best_free:
   161	                # TODO: failing to choose may cause leaking type variables,
   162	                # we need to fail gracefully instead.
   163	                free_vars.append(best_free.id)
   164	                free_solutions[best_free.id] = best_free
   165
   166	    # Update lowers/uppers with free vars, so these can now be used
   167	    # as valid solutions.
   168	    for l, u in graph:
   169	        if l in free_vars:
   170	            lowers[u].add(free_solutions[l])
   171	        if u in free_vars:
   172	            uppers[l].add(free_solutions[u])
   173
   174	    # Flatten the SCCs that are independent, we can solve them together,
   175	    # since we don't need to update any targets in between.
   176	    batches = []
   177	    for batch in raw_batches:
   178	        next_bc = []
   179	        for scc in batch:
   180	            next_bc.extend(list(scc))
   181	        batches.append(next_bc)
   182
   183	    solutions: dict[TypeVarId, Type | None] = {}
   184	    for flat_batch in batches:
   185	        res = solve_iteratively(flat_batch, graph, lowers, uppers)
   186	        solutions.update(res)
   187	    return solutions, [free_solutions[tv] for tv in free_vars]

   190	def solve_iteratively(
   191	    batch: list[TypeVarId], graph: Graph, lowers: Bounds, uppers: Bounds
   192	) -> Solutions:
   193	    """Solve transitive closure sequentially, updating upper/lower bounds after each step.
   194
   195	    Transitive closure is represented as a linear graph plus lower/upper bounds for each
   196	    type variable, see transitive_closure() docstring for details.
   197
   198	    We solve for type variables that appear in `batch`. If a bound is not constant (i.e. it
   199	    looks like T :> F[S, ...]), we substitute solutions found so far in the target F[S, ...]
   200	    after solving the batch.
   201
   202	    Importantly, after solving each variable in a batch, we move it from linear graph to
   203	    upper/lower bounds, this way we can guarantee consistency of solutions (see comment below
   204	    for an example when this is important).
   205	    """
   206	    solutions = {}
   207	    s_batch = set(batch)
   208	    while s_batch:
   209	        for tv in sorted(s_batch, key=lambda x: x.raw_id):
   210	            if lowers[tv] or uppers[tv]:
   211	                solvable_tv = tv
   212	                break
   213	        else:
   214	            break
   215	        # Solve each solvable type variable separately.
   216	        s_batch.remove(solvable_tv)
   217	        result = solve_one(lowers[solvable_tv], uppers[solvable_tv])
   218	        solutions[solvable_tv] = result
   219	        if result is None:
   220	            # TODO: support backtracking lower/upper bound choices and order within SCCs.
   221	            # (will require switching this function from iterative to recursive).
   222	            continue
   223
   224	        # Update the (transitive) bounds from graph if there is a solution.
   225	        # This is needed to guarantee solutions will never contradict the initial
   226	        # constraints. For example, consider {T <: S, T <: A, S :> B} with A :> B.
   227	        # If we would not update the uppers/lowers from graph, we would infer T = A, S = B
   228	        # which is not correct.
   229	        for l, u in graph.copy():
   230	            if l == u:
   231	                continue
   232	            if l == solvable_tv:
   233	                lowers[u].add(result)
   234	                graph.remove((l, u))
   235	            if u == solvable_tv:
   236	                uppers[l].add(result)
   237	                graph.remove((l, u))
   238
   239	    # We can update uppers/lowers only once after solving the whole SCC,
   240	    # since uppers/lowers can't depend on type variables in the SCC
   241	    # (and we would reject such SCC as non-linear and therefore not solvable).
   242	    subs = {tv: s for (tv, s) in solutions.items() if s is not None}
   243	    for tv in lowers:
   244	        lowers[tv] = {expand_type(lt, subs) for lt in lowers[tv]}
   245	    for tv in uppers:
   246	        uppers[tv] = {expand_type(ut, subs) for ut in uppers[tv]}
   247	    return solutions

   259	def solve_one(lowers: Iterable[Type], uppers: Iterable[Type]) -> Type | None:
   260	    """Solve constraints by finding by using meets of upper bounds, and joins of lower bounds."""
   261
   262	    candidate: Type | None = None
   263
   264	    # Filter out previous results of failed inference, they will only spoil the current pass...
   265	    new_uppers = []
   266	    for u in uppers:
   267	        pu = get_proper_type(u)
   268	        if not isinstance(pu, UninhabitedType) or not pu.ambiguous:
   269	            new_uppers.append(u)
   270	    uppers = new_uppers
   271
   272	    # ...unless this is the only information we have, then we just pass it on.
   273	    lowers = list(lowers)
   274	    if not uppers and not lowers:
   275	        candidate = UninhabitedType()
   276	        candidate.ambiguous = True
   277	        return candidate
   278
   279	    bottom: Type | None = None
   280	    top: Type | None = None
   281
   282	    # Process each bound separately, and calculate the lower and upper
   283	    # bounds based on constraints. Note that we assume that the constraint
   284	    # targets do not have constraint references.
   285	    if type_state.infer_unions and lowers:
   286	        # This deviates from the general mypy semantics because
   287	        # recursive types are union-heavy in 95% of cases.
   288	        # Retain `None` when no bottoms were provided to avoid bogus `Never` inference.
   289	        bottom = UnionType.make_union(lowers)
   290	    else:
   291	        # The order of lowers is non-deterministic.
   292	        # We attempt to sort lowers because joins are non-associative. For instance:
   293	        # join(join(int, str), int | str) == join(object, int | str) == object
   294	        # join(int, join(str, int | str)) == join(int, int | str)    == int | str
   295	        # Note that joins in theory should be commutative, but in practice some bugs mean this is
   296	        # also a source of non-deterministic type checking results.
   297	        sorted_lowers = sorted(lowers, key=_join_sorted_key)
   298	        if sorted_lowers:
   299	            bottom = join_type_list(sorted_lowers)
   300
   301	    for target in uppers:
   302	        if top is None:
   303	            top = target
   304	        else:
   305	            top = meet_types(top, target)
   306
   307	    p_top = get_proper_type(top)
   308	    p_bottom = get_proper_type(bottom)
   309	    if isinstance(p_top, AnyType) or isinstance(p_bottom, AnyType):
   310	        source_any = top if isinstance(p_top, AnyType) else bottom
   311	        assert isinstance(source_any, ProperType) and isinstance(source_any, AnyType)
   312	        return AnyType(TypeOfAny.from_another_any, source_any=source_any)
   313	    elif bottom is None:
   314	        if top:
   315	            candidate = top
   316	        else:
   317	            # No constraints for type variable
   318	            return None
   319	    elif top is None:
   320	        candidate = bottom
   321	    elif is_subtype(bottom, top):
   322	        candidate = bottom
   323	    else:
   324	        candidate = None
   325	    return candidate

   328	def choose_free(
   329	    scc: list[TypeVarLikeType], original_vars: list[TypeVarId]
   330	) -> TypeVarLikeType | None:
   331	    """Choose the best solution for an SCC containing only type variables.
   332
   333	    This is needed to preserve e.g. the upper bound in a situation like this:
   334	        def dec(f: Callable[[T], S]) -> Callable[[T], S]: ...
   335
   336	        @dec
   337	        def test(x: U) -> U: ...
   338
   339	    where U <: A.
   340	    """
   341
   342	    if len(scc) == 1:
   343	        # Fast path, choice is trivial.
   344	        return scc[0]
   345
   346	    common_upper_bound = meet_type_list([t.upper_bound for t in scc])
   347	    common_upper_bound_p = get_proper_type(common_upper_bound)
   348	    # We include None for when strict-optional is disabled.
   349	    if isinstance(common_upper_bound_p, (UninhabitedType, NoneType)):
   350	        # This will cause to infer Never, which is better than a free TypeVar
   351	        # that has an upper bound Never.
   352	        return None
   353
   354	    values: list[Type] = []
   355	    for tv in scc:
   356	        if isinstance(tv, TypeVarType) and tv.values:
   357	            if values:
   358	                # It is too tricky to support multiple TypeVars with values
   359	                # within the same SCC.
   360	                return None
   361	            values = tv.values.copy()
   362
   363	    if values and not is_trivial_bound(common_upper_bound_p):
   364	        # If there are both values and upper bound present, we give up,
   365	        # since type variables having both are not supported.
   366	        return None
   367
   368	    # For convenience with current type application machinery, we use a stable
   369	    # choice that prefers the original type variables (not polymorphic ones) in SCC.
   370	    best = min(scc, key=lambda x: (x.id not in original_vars, x.id.raw_id))
   371	    if isinstance(best, TypeVarType):
   372	        return best.copy_modified(values=values, upper_bound=common_upper_bound)
   373	    if is_trivial_bound(common_upper_bound_p, allow_tuple=True):
   374	        # TODO: support more cases for ParamSpecs/TypeVarTuples
   375	        return best
   376	    return None

   402	def transitive_closure(
   403	    tvars: list[TypeVarId], constraints: list[Constraint]
   404	) -> tuple[Graph, Bounds, Bounds]:
   405	    """Find transitive closure for given constraints on type variables.
   406
   407	    Transitive closure gives maximal set of lower/upper bounds for each type variable,
   408	    such that we cannot deduce any further bounds by chaining other existing bounds.
   409
   410	    The transitive closure is represented by:
   411	      * A set of lower and upper bounds for each type variable, where only constant and
   412	        non-linear terms are included in the bounds.
   413	      * A graph of linear constraints between type variables (represented as a set of pairs)
   414	    Such separation simplifies reasoning, and allows an efficient and simple incremental
   415	    transitive closure algorithm that we use here.
   416
   417	    For example if we have initial constraints [T <: S, S <: U, U <: int], the transitive
   418	    closure is given by:
   419	      * {} <: T <: {int}
   420	      * {} <: S <: {int}
   421	      * {} <: U <: {int}
   422	      * {T <: S, S <: U, T <: U}
   423	    """
   424	    uppers: Bounds = defaultdict(set)
   425	    lowers: Bounds = defaultdict(set)
   426	    graph: Graph = {(tv, tv) for tv in tvars}
   427
   428	    remaining = set(constraints)
   429	    while remaining:
   430	        c = remaining.pop()
   431	        # Note that ParamSpec constraint P <: Q may be considered linear only if Q has no prefix,
   432	        # for cases like P <: Concatenate[T, Q] we should consider this non-linear and put {P} and
   433	        # {T, Q} into separate SCCs. Similarly, Ts <: Tuple[*Us] considered linear, while
   434	        # Ts <: Tuple[*Us, U] is non-linear.
   435	        is_linear, target_id = find_linear(c)
   436	        if is_linear and target_id in tvars:
   437	            assert target_id is not None
   438	            if c.op == SUBTYPE_OF:
   439	                lower, upper = c.type_var, target_id
   440	            else:
   441	                lower, upper = target_id, c.type_var
   442	            if (lower, upper) in graph:
   443	                continue
   444	            graph |= {
   445	                (l, u) for l in tvars for u in tvars if (l, lower) in graph and (upper, u) in graph
   446	            }
   447	            for u in tvars:
   448	                if (upper, u) in graph:
   449	                    lowers[u] |= lowers[lower]
   450	            for l in tvars:
   451	                if (l, lower) in graph:
   452	                    uppers[l] |= uppers[upper]
   453	            for lt in lowers[lower]:
   454	                for ut in uppers[upper]:
   455	                    add_secondary_constraints(remaining, lt, ut)
   456	        elif c.op == SUBTYPE_OF:
   457	            if c.target in uppers[c.type_var]:
   458	                continue
   459	            for l in tvars:
   460	                if (l, c.type_var) in graph:
   461	                    uppers[l].add(c.target)
   462	            for lt in lowers[c.type_var]:
   463	                add_secondary_constraints(remaining, lt, c.target)
   464	        else:
   465	            assert c.op == SUPERTYPE_OF
   466	            if c.target in lowers[c.type_var]:
   467	                continue
   468	            for u in tvars:
   469	                if (c.type_var, u) in graph:
   470	                    lowers[u].add(c.target)
   471	            for ut in uppers[c.type_var]:
   472	                add_secondary_constraints(remaining, c.target, ut)

   476	def add_secondary_constraints(cs: set[Constraint], lower: Type, upper: Type) -> None:
   477	    """Add secondary constraints inferred between lower and upper (in place)."""
   478	    if isinstance(get_proper_type(upper), UnionType) and isinstance(
   479	        get_proper_type(lower), UnionType

   483	        # To avoid this, just skip them for now.
   484	        return
   485	    # TODO: what if secondary constraints result in inference against polymorphic actual?

   490	def compute_dependencies(
   491	    tvars: list[TypeVarId], graph: Graph, lowers: Bounds, uppers: Bounds
   492	) -> dict[TypeVarId, list[TypeVarId]]:
   493	    """Compute dependencies between type variables induced by constraints.
   494
   495	    If we have a constraint like T <: List[S], we say that T depends on S, since
   496	    we will need to solve for S first before we can solve for T.
   497	    """
   498	    res = {}
   499	    for tv in tvars:
   500	        deps = set()
   501	        for lt in lowers[tv]:
   502	            deps |= get_vars(lt, tvars)
   503	        for ut in uppers[tv]:
   504	            deps |= get_vars(ut, tvars)
   505	        for other in tvars:
   506	            if other == tv:
   507	                continue
   508	            if (tv, other) in graph or (other, tv) in graph:
   509	                deps.add(other)
   510	        res[tv] = list(deps)
   511	    return res

   514	def check_linear(scc: set[TypeVarId], lowers: Bounds, uppers: Bounds) -> bool:
   515	    """Check there are only linear constraints between type variables in SCC.
   516
   517	    Linear are constraints like T <: S (while T <: F[S] are non-linear).
   518	    """
   519	    for tv in scc:
   520	        if any(get_vars(lt, list(scc)) for lt in lowers[tv]):
   521	            return False
   522	        if any(get_vars(ut, list(scc)) for ut in uppers[tv]):
   523	            return False
   524	    return True

   527	def skip_reverse_union_constraints(cs: list[Constraint]) -> list[Constraint]:
   528	    """Avoid ambiguities for constraints inferred from unions during polymorphic inference.
   529
   530	    Polymorphic inference implicitly relies on assumption that a reverse of a linear constraint
   531	    is a linear constraint. This is however not true in presence of union types, for example
   532	    T :> Union[S, int] vs S <: T. Trying to solve such constraints would be detected ambiguous
   533	    as (T, S) form a non-linear SCC. However, simply removing the linear part results in a valid
   534	    solution T = Union[S, int], S = <free>. A similar scenario is when we get T <: Union[T, int],
   535	    such constraints carry no information, and will equally confuse linearity check.
   536
   537	    TODO: a cleaner solution may be to avoid inferring such constraints in first place, but
   538	    this would require passing around a flag through all infer_constraints() calls.
   539	    """
   540	    reverse_union_cs = set()
   541	    for c in cs:
   542	        p_target = get_proper_type(c.target)
   543	        if isinstance(p_target, UnionType):
   544	            for item in p_target.items:
   545	                if isinstance(item, TypeVarType):
   546	                    if item == c.origin_type_var and c.op == SUBTYPE_OF:
   547	                        reverse_union_cs.add(c)
   548	                        continue
   549	                    # These two forms are semantically identical, but are different from
   550	                    # the point of view of Constraint.__eq__().
   551	                    reverse_union_cs.add(Constraint(item, neg_op(c.op), c.origin_type_var))
   552	                    reverse_union_cs.add(Constraint(c.origin_type_var, c.op, item))
   553	    return [c for c in cs if c not in reverse_union_cs]

   556	def get_vars(target: Type, vars: list[TypeVarId]) -> set[TypeVarId]:
   557	    """Find type variables for which we are solving in a target type."""
   558	    return {tv.id for tv in get_all_type_vars(target)} & set(vars)

   561	def pre_validate_solutions(
   562	    solutions: list[Type | None],
   563	    original_vars: Sequence[TypeVarLikeType],
   564	    constraints: list[Constraint],
   565	) -> list[Type | None]:
   566	    """Check is each solution satisfies the upper bound of the corresponding type variable.
   567
   568	    If it doesn't satisfy the bound, check if bound itself satisfies all constraints, and
   569	    if yes, use it instead as a fallback solution.
   570	    """
   571	    new_solutions: list[Type | None] = []
   572	    for t, s in zip(original_vars, solutions):
   573	        if is_callable_protocol(t.upper_bound):
   574	            # This is really ad-hoc, but a proper fix would be much more complex,
   575	            # and otherwise this may cause crash in a relatively common scenario.
   576	            new_solutions.append(s)
   577	            continue
   578	        if s is not None and not is_subtype(s, t.upper_bound):
   579	            bound_satisfies_all = True
   580	            for c in constraints:
   581	                if c.op == SUBTYPE_OF and not is_subtype(t.upper_bound, c.target):
   582	                    bound_satisfies_all = False
   583	                    break
   584	                if c.op == SUPERTYPE_OF and not is_subtype(c.target, t.upper_bound):
   585	                    bound_satisfies_all = False
   586	                    break
   587	            if bound_satisfies_all:
   588	                new_solutions.append(t.upper_bound)
   589	                continue
   590	        new_solutions.append(s)
   591	    return new_solutions
   592

   594	def is_callable_protocol(t: Type) -> bool:
   595	    proper_t = get_proper_type(t)

   597	        return "__call__" in proper_t.type.protocol_members
   598	    return False

   239	            and self.state[-1] in (STATE_ARGUMENT_LIST, STATE_ARGUMENT_TYPE)
   240	        ):
   241	            if self.state[-1] == STATE_ARGUMENT_TYPE:
   242	                self.arg_type = self.accumulator
   243	                self.state.pop()
   244	            else:
   245	                self.arg_name = self.accumulator
   246	            self.accumulator = ""
   247	            self.state.append(STATE_ARGUMENT_DEFAULT)
   248
   249	        elif (
   250	            token.type == tokenize.OP
   251	            and token.string in (",", ")")
   252	            and self.state[-1]
   253	            in (STATE_ARGUMENT_LIST, STATE_ARGUMENT_DEFAULT, STATE_ARGUMENT_TYPE)
   254	        ):
   255	            if self.state[-1] == STATE_ARGUMENT_DEFAULT:
   256	                self.arg_default = self.accumulator
   257	                self.state.pop()
   258	            elif self.state[-1] == STATE_ARGUMENT_TYPE:
   259	                self.arg_type = self.accumulator
   260	                self.state.pop()
   261	            elif self.state[-1] == STATE_ARGUMENT_LIST:
   262	                if self.accumulator == "*":
   263	                    if self.keyword_only is not None:
   264	                        # Error condition: cannot have * twice
   265	                        self.reset()
   266	                        return
   267	                    self.keyword_only = len(self.args)
   268	                    self.accumulator = ""
   269	                else:
   270	                    if self.accumulator.startswith("*"):
   271	                        self.keyword_only = len(self.args) + 1
   272	                    self.arg_name = self.accumulator
   273	                    if not (
   274	                        token.string == ")" and self.accumulator.strip() == ""
   275	                    ) and not _ARG_NAME_RE.match(self.arg_name):
   276	                        # Invalid argument name.
   277	                        self.reset()
   278	                        return
   279
   280	            if token.string == ")":
   281	                if (
   282	                    self.state[-1] == STATE_ARGUMENT_LIST
   283	                    and self.keyword_only is not None
   284	                    and self.keyword_only == len(self.args)
   285	                    and not self.arg_name
   286	                ):
   287	                    # Error condition: * must be followed by arguments
   288	                    self.reset()
   289	                    return
   290	                self.state.pop()
   291
   292	            # arg_name is empty when there are no args. e.g. func()
   293	            if self.arg_name:
   294	                if self.arg_type and not is_valid_type(self.arg_type):
   295	                    # wrong type, use Any
   296	                    self.args.append(
   297	                        ArgSig(name=self.arg_name, type=None, default=bool(self.arg_default))
   298	                    )
   299	                else:
   300	                    self.args.append(
   301	                        ArgSig(
   302	                            name=self.arg_name, type=self.arg_type, default=bool(self.arg_default)
   303	                        )
   304	                    )
   305	            self.arg_name = ""
   306	            self.arg_type = None
   307	            self.arg_default = None
   308	            self.accumulator = ""
   309	        elif (
   310	            token.type == tokenize.OP
   311	            and token.string == "/"
   312	            and self.state[-1] == STATE_ARGUMENT_LIST
   313	        ):
   314	            if token.string == "/":
   315	                if self.pos_only is not None or self.keyword_only is not None or not self.args:
   316	                    # Error cases:
   317	                    # - / shows up more than once
   318	                    # - / shows up after *
   319	                    # - / shows up before any arguments
   320	                    self.reset()
   321	                    return
   322	                self.pos_only = len(self.args)
   323	                self.state.append(STATE_ARGUMENT_TYPE)
   324	                self.accumulator = ""
   325
   326	        elif token.type == tokenize.OP and token.string == "->" and self.state[-1] == STATE_INIT:
   327	            self.accumulator = ""
   328	            self.state.append(STATE_RETURN_VALUE)
   329
   330	        # ENDMAKER is necessary for python 3.4 and 3.5.
   331	        elif token.type in (tokenize.NEWLINE, tokenize.ENDMARKER) and self.state[-1] in (
   332	            STATE_INIT,
   333	            STATE_RETURN_VALUE,
   334	        ):
   335	            if self.state[-1] == STATE_RETURN_VALUE:
   336	                if not is_valid_type(self.accumulator):
   337	                    self.reset()
   338	                    return
   339	                self.ret_type = self.accumulator
   340	                self.accumulator = ""
   341	                self.state.pop()
   342
   343	            if self.found:
   344	                self.signatures.append(
   345	                    FunctionSig(name=self.function_name, args=self.args, ret_type=self.ret_type)
   346	                )
   347	                self.found = False
   348	            self.args = []
   349	            self.ret_type = "Any"
   350	            # Leave state as INIT.
   351	        else:
   352	            self.accumulator += token.string
   353
   354	    def reset(self) -> None:
   355	        self.state = [STATE_INIT]
   356	        self.args = []
   357	        self.found = False
   358	        self.accumulator = ""

   360	    def get_signatures(self) -> list[FunctionSig]:
   361	        """Return sorted copy of the list of signatures found so far."""
   362
   363	        def has_arg(name: str, signature: FunctionSig) -> bool:
   364	            return any(x.name == name for x in signature.args)
   365
   366	        def args_kwargs(signature: FunctionSig) -> bool:
   367	            return has_arg("*args", signature) and has_arg("**kwargs", signature)
   368
   369	        # Move functions with (*args, **kwargs) in their signature to last place.
   370	        return sorted(self.signatures, key=lambda x: 1 if args_kwargs(x) else 0)

   373	def infer_sig_from_docstring(docstr: str | None, name: str) -> list[FunctionSig] | None:
   374	    """Convert function signature to list of FunctionSig
   375
   376	    Look for function signatures of function in docstring. Signature is a string of
   377	    the format <function_name>(<signature>) -> <return type> or perhaps without
   378	    the return type.
   379
   380	    Returns empty list, when no signature is found, one signature in typical case,
   381	    multiple signatures, if docstring specifies multiple signatures for overload functions.
   382	    Return None if the docstring is empty.
   383
   384	    Arguments:
   385	        * docstr: docstring
   386	        * name: name of function for which signatures are to be found
   387	    """
   388	    if not (isinstance(docstr, str) and docstr):
   389	        return None
   390
   391	    state = DocStringParser(name)
   392	    # Return all found signatures, even if there is a parse error after some are found.
   393	    with contextlib.suppress(tokenize.TokenError):
   394	        try:
   395	            tokens = tokenize.tokenize(io.BytesIO(docstr.encode("utf-8")).readline)
   396	            for token in tokens:
   397	                state.add_token(token)
   398	        except IndentationError:
   399	            return None
   400	    sigs = state.get_signatures()
   401
   402	    def is_unique_args(sig: FunctionSig) -> bool:
   403	        """return true if function argument names are unique"""
   404	        return len(sig.args) == len({arg.name for arg in sig.args})
   405
   406	    # Return only signatures that have unique argument names. Mypy fails on non-unique arg names.
   407	    return [sig for sig in sigs if is_unique_args(sig)]
