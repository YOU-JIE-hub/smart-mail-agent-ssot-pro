        first_line: int = 0
   141	        # Is the file empty?
   142	        empty: bool = True
   143	        # Parenthesis (and bracket) nesting level.
   144	        nesting: int = 0
   145
   146	        assert self.text is not None
   147	        tokgen = generate_tokens(self.text)
   148	        for toktype, ttext, (slineno, _), (elineno, _), ltext in tokgen:
   149	            if self.show_tokens:                # pragma: debugging
   150	                print("%10s %5s %-20r %r" % (
   151	                    tokenize.tok_name.get(toktype, toktype),
   152	                    nice_pair((slineno, elineno)), ttext, ltext,
   153	                ))
   154	            if toktype == token.INDENT:
   155	                indent += 1
   156	            elif toktype == token.DEDENT:
   157	                indent -= 1
   158	            elif toktype == token.OP:
   159	                if ttext == ":" and nesting == 0:
   160	                    should_exclude = (
   161	                        self.excluded.intersection(range(first_line, elineno + 1))
   162	                    )
   163	                    if not excluding and should_exclude:
   164	                        # Start excluding a suite.  We trigger off of the colon
   165	                        # token so that the #pragma comment will be recognized on
   166	                        # the same line as the colon.
   167	                        self.excluded.add(elineno)
   168	                        exclude_indent = indent
   169	                        excluding = True
   170	                elif ttext in "([{":
   171	                    nesting += 1
   172	                elif ttext in ")]}":
   173	                    nesting -= 1
   174	            elif toktype == token.NEWLINE:
   175	                if first_line and elineno != first_line:
   176	                    # We're at the end of a line, and we've ended on a
   177	                    # different line than the first line of the statement,
   178	                    # so record a multi-line range.
   179	                    for l in range(first_line, elineno+1):
   180	                        self._multiline[l] = first_line
   181	                first_line = 0
   182
   183	            if ttext.strip() and toktype != tokenize.COMMENT:
   184	                # A non-white-space token.
   185	                empty = False
   186	                if not first_line:
   187	                    # The token is not white space, and is the first in a statement.
   188	                    first_line = slineno
   189	                    # Check whether to end an excluded suite.
   190	                    if excluding and indent <= exclude_indent:
   191	                        excluding = False
   192	                    if excluding:
   193	                        self.excluded.add(elineno)
   194
   195	        # Find the starts of the executable statements.
   196	        if not empty:
   197	            byte_parser = ByteParser(self.text, filename=self.filename)
   198	            self.raw_statements.update(byte_parser._find_statements())
   199
   200	        self.excluded = self.first_lines(self.excluded)
   201
   202	        # AST lets us find classes, docstrings, and decorator-affected
   203	        # functions and classes.
   204	        assert self._ast_root is not None
   205	        for node in ast.walk(self._ast_root):
   206	            # Find docstrings.
   207	            if isinstance(node, (ast.ClassDef, ast.FunctionDef, ast.AsyncFunctionDef, ast.Module)):
   208	                if node.body:
   209	                    first = node.body[0]
   210	                    if (
   211	                        isinstance(first, ast.Expr)
   212	                        and isinstance(first.value, ast.Constant)
   213	                        and isinstance(first.value.value, str)
   214	                    ):
   215	                        self.raw_docstrings.update(
   216	                            range(first.lineno, cast(int, first.end_lineno) + 1)
   217	                        )
   218	            # Exclusions carry from decorators and signatures to the bodies of
   219	            # functions and classes.
   220	            if isinstance(node, (ast.ClassDef, ast.FunctionDef, ast.AsyncFunctionDef)):
   221	                first_line = min((d.lineno for d in node.decorator_list), default=node.lineno)
   222	                if self.excluded.intersection(range(first_line, node.lineno + 1)):
   223	                    self.excluded.update(range(first_line, cast(int, node.end_lineno) + 1))
   224

   279	        to the first line of multi-line statements.

   287	    def _analyze_ast(self) -> None:
   288	        """Run the AstArcAnalyzer and save its results.
   289
   290	        `_all_arcs` is the set of arcs in the code.
   291
   292	        """
   293	        assert self._ast_root is not None
   294	        aaa = AstArcAnalyzer(self.filename, self._ast_root, self.raw_statements, self._multiline)
   295	        aaa.analyze()
   296	        arcs = aaa.arcs
   297	        if env.PYBEHAVIOR.exit_through_with:
   298	            self._with_jump_fixers = aaa.with_jump_fixers()
   299	            if self._with_jump_fixers:
   300	                arcs = self.fix_with_jumps(arcs)
   301
   302	        self._all_arcs = set()
   303	        for l1, l2 in arcs:
   304	            fl1 = self.first_line(l1)
   305	            fl2 = self.first_line(l2)
   306	            if fl1 != fl2:
   307	                self._all_arcs.add((fl1, fl2))
   308
   309	        self._missing_arc_fragments = aaa.missing_arc_fragments

   311	    def fix_with_jumps(self, arcs: Iterable[TArc]) -> set[TArc]:
   312	        """Adjust arcs to fix jumps leaving `with` statements.
   313
   314	        Consider this code:
   315
   316	            with open("/tmp/test", "w") as f1:
   317	                a = 2
   318	                b = 3
   319	            print(4)
   320
   321	        In 3.10+, we get traces for lines 1, 2, 3, 1, 4.  But we want to present
   322	        it to the user as if it had been 1, 2, 3, 4.  The arc 3->1 should be
   323	        replaced with 3->4, and 1->4 should be removed.
   324
   325	        For this code, the fixers dict is {(3, 1): ((1, 4), (3, 4))}.  The key
   326	        is the actual measured arc from the end of the with block back to the
   327	        start of the with-statement.  The values are start_next (the with
   328	        statement to the next statement after the with), and end_next (the end
   329	        of the with-statement to the next statement after the with).
   330
   331	        With nested with-statements, we have to trace through a few levels to
   332	        correct a longer chain of arcs.
   333
   334	        """
   335	        to_remove = set()
   336	        to_add = set()
   337	        for arc in arcs:
   338	            if arc in self._with_jump_fixers:
   339	                end0 = arc[0]
   340	                to_remove.add(arc)
   341	                start_next, end_next = self._with_jump_fixers[arc]
   342	                while start_next in self._with_jump_fixers:
   343	                    to_remove.add(start_next)
   344	                    start_next, end_next = self._with_jump_fixers[start_next]
   345	                    to_remove.add(end_next)
   346	                to_add.add((end0, end_next[1]))
   347	                to_remove.add(start_next)
   348	        arcs = (set(arcs) | to_add) - to_remove
   349	        return arcs

   353	        """Get a count of exits from that each line.
   354
   355	        Excluded lines are excluded.
   356
   357	        """
   358	        exit_counts: dict[TLineNo, int] = collections.defaultdict(int)

   384	            self._analyze_ast()
   385	            assert self._missing_arc_fragments is not None

   387	        fragment_pairs = self._missing_arc_fragments.get((start, end), [(None, None)])

   389	        msgs = []

   411	class ByteParser:
   412	    """Parse bytecode to understand the structure of code."""
   413
   414	    def __init__(
   415	        self,
   416	        text: str,
   417	        code: CodeType | None = None,
   418	        filename: str | None = None,
   419	    ) -> None:
   420	        self.text = text
   421	        if code is not None:
   422	            self.code = code
   423	        else:
   424	            assert filename is not None
   425	            # We only get here if earlier ast parsing succeeded, so no need to
   426	            # catch errors.
   427	            self.code = compile(text, filename, "exec", dont_inherit=True)
   428
   429	    def child_parsers(self) -> Iterable[ByteParser]:
   430	        """Iterate over all the code objects nested within this one.
   431
   432	        The iteration includes `self` as its first value.
   433
   434	        We skip code objects named `__annotate__` since they are deferred
   435	        annotations that usually are never run.  If there are errors in the
   436	        annotations, they will be caught by type checkers or other tools that
   437	        use annotations.
   438
   439	        """
   440	        return (
   441	            ByteParser(self.text, code=c)
   442	            for c in code_objects(self.code)
   443	            if c.co_name != "__annotate__"
   444	        )
   445
   446	    def _line_numbers(self) -> Iterable[TLineNo]:
   447	        """Yield the line numbers possible in this code object.
   448
   449	        Uses co_lnotab described in Python/compile.c to find the
   450	        line numbers.  Produces a sequence: l0, l1, ...
   451	        """
   452	        if hasattr(self.code, "co_lines"):
   453	            # PYVERSIONS: new in 3.10
   454	            for _, _, line in self.code.co_lines():
   455	                if line:
   456	                    yield line
   457	        else:
   458	            # Adapted from dis.py in the standard library.
   459	            byte_increments = self.code.co_lnotab[0::2]
   460	            line_increments = self.code.co_lnotab[1::2]
   461
   462	            last_line_num: TLineNo | None = None
   463	            line_num = self.code.co_firstlineno
   464	            byte_num = 0
   465	            for byte_incr, line_incr in zip(byte_increments, line_increments):
   466	                if byte_incr:
   467	                    if line_num != last_line_num:
   468	                        yield line_num
   469	                        last_line_num = line_num
   470	                    byte_num += byte_incr
   471	                if line_incr >= 0x80:
   472	                    line_incr -= 0x100
   473	                line_num += line_incr
   474	            if line_num != last_line_num:
   475	                yield line_num
   476
   477	    def _find_statements(self) -> Iterable[TLineNo]:
   478	        """Find the statements in `self.code`.
   479
   480	        Produce a sequence of line numbers that start statements.  Recurses
   481	        into all code objects reachable from `self.code`.
   482
   483	        """
   484	        for bp in self.child_parsers():
   485	            # Get all of the lineno information from this code.
   486	            yield from bp._line_numbers()

   495	    """The information needed to start an arc.
   496
   497	    `lineno` is the line number the arc starts from.
   498
   499	    `cause` is an English text fragment used as the `missing_cause_msg` for
   500	    AstArcAnalyzer.missing_arc_fragments.  It will be used to describe why an

   517	    "line 1 didn't jump to line 3 because the condition on line 1 was always true."

   519	    The second would be used for the arc from 1 to 2, creating a message like
   520	    "line 1 didn't jump to line 2 because the condition on line 1 was never true."

   650	def is_constant_test_expr(node: ast.AST) -> tuple[bool, bool]:
   651	    """Is this a compile-time constant test expression?
   652
   653	    We don't try to mimic all of CPython's optimizations.  We just have to
   654	    handle the kinds of constant expressions people might actually use.
   655
   656	    """
   657	    if isinstance(node, ast.Constant):
   658	        return True, bool(node.value)
   659	    elif isinstance(node, ast.Name):
   660	        if node.id in ["True", "False", "None", "__debug__"]:
   661	            return True, eval(node.id)  # pylint: disable=eval-used
   662	    elif isinstance(node, ast.UnaryOp) and isinstance(node.op, ast.Not):
   663	        is_constant, val = is_constant_test_expr(node.operand)
   664	        return is_constant, not val
   665	    elif isinstance(node, ast.BoolOp):
   666	        rets = [is_constant_test_expr(v) for v in node.values]
   667	        is_constant = all(is_const for is_const, _ in rets)
   668	        if is_constant:
   669	            op = any if isinstance(node.op, ast.Or) else all
   670	            return True, op(v for _, v in rets)
   671	    return False, False

   674	class AstArcAnalyzer:
   675	    """Analyze source text with an AST to find executable code paths.
   676
   677	    The .analyze() method does the work, and populates these attributes:
   678
   679	    `arcs`: a set of (from, to) pairs of the the arcs possible in the code.
   680
   681	    `missing_arc_fragments`: a dict mapping (from, to) arcs to lists of
   682	    message fragments explaining why the arc is missing from execution::
   683
   684	        { (start, end): [(missing_cause_msg, action_msg), ...], }
   685
   686	    For an arc starting from line 17, they should be usable to form complete
   687	    sentences like: "Line 17 didn't {action_msg} because {missing_cause_msg}".
   688
   689	    NOTE: Starting in July 2024, I've been whittling this down to only report
   690	    arc that are part of true branches.  It's not clear how far this work will
   691	    go.
   692
   693	    """
   694
   695	    def __init__(
   696	        self,
   697	        filename: str,
   698	        root_node: ast.AST,
   699	        statements: set[TLineNo],
   700	        multiline: dict[TLineNo, TLineNo],
   701	    ) -> None:
   702	        self.filename = filename

   734	        """Examine the AST tree from `self.root_node` to determine possible arcs."""
   735	        for node in ast.walk(self.root_node):

   741	    def with_jump_fixers(self) -> dict[TArc, tuple[TArc, TArc]]:
   742	        """Get a dict with data for fixing jumps out of with statements.
   743
   744	        Returns a dict.  The keys are arcs leaving a with-statement by jumping
   745	        back to its start.  The values are pairs: first, the arc from the start
   746	        to the next statement, then the arc that exits the with without going
   747	        to the start.
   748
   749	        """
   750	        fixers = {}
   751	        with_nexts = {
   752	            arc
   753	            for arc in self.arcs
   754	            if arc[0] in self.all_with_starts and arc not in self.with_entries
   755	        }
   756	        for start in self.all_with_starts:
   757	            nexts = {arc[1] for arc in with_nexts if arc[0] == start}
   758	            if not nexts:
   759	                continue
   760	            assert len(nexts) == 1, f"Expected one arc, got {nexts} with {start = }"
   761	            nxt = nexts.pop()
   762	            ends = {arc[0] for arc in self.with_exits if arc[1] == start}
   763	            for end in ends:
   764	                fixers[(end, start)] = ((start, nxt), (end, nxt))
   765	        return fixers

   889	    def node_exits(self, node: ast.AST) -> set[ArcStart]:
   890	        """Find the set of arc starts that exit this node.
   891
   892	        Return a set of ArcStarts, exits from this node to the next. Because a
   893	        node represents an entire sub-tree (including its children), the exits
   894	        from a node can be arbitrarily complex::
   895
   896	            if something(1):
   897	                if other(2):
   898	                    doit(3)
   899	                else:
   900	                    doit(5)
   901
   902	        There are three exits from line 1: they start at lines 1, 3 and 5.
   903	        There are two exits from line 2: lines 3 and 5.
   904
   905	        """
   906	        node_name = node.__class__.__name__
   907	        handler = cast(
   908	            Optional[Callable[[ast.AST], set[ArcStart]]],
   909	            getattr(self, f"_handle__{node_name}", None),
   910	        )
   911	        if handler is not None:
   912	            arc_starts = handler(node)
   913	        else:
   914	            # No handler: either it's something that's ok to default (a simple
   915	            # statement), or it's something we overlooked.
   916	            if env.TESTING:
   917	                if node_name not in self.OK_TO_DEFAULT:
   918	                    raise RuntimeError(f"*** Unhandled: {node}")        # pragma: only failure
   919
   920	            # Default for simple statements: one exit from this node.
   921	            arc_starts = {ArcStart(self.line_for_node(node))}
   922	        return arc_starts

   924	    def process_body(
   925	        self,
   926	        body: Sequence[ast.AST],
   927	        from_start: ArcStart | None = None,
   928	        prev_starts: set[ArcStart] | None = None,
   929	    ) -> set[ArcStart]:
   930	        """Process the body of a compound statement.
   931
   932	        `body` is the body node to process.
   933
   934	        `from_start` is a single `ArcStart` that starts an arc into this body.
   935	        `prev_starts` is a set of ArcStarts that can all be the start of arcs
   936	        into this body.  Only one of `from_start` and `prev_starts` should be
   937	        given.
   938
   939	        Records arcs within the body by calling `self.add_arc`.
   940
   941	        Returns a set of ArcStarts, the exits from this body.
   942
   943	        """
   944	        if prev_starts is None:
   945	            if from_start is None:
   946	                prev_starts = set()
   947	            else:
   948	                prev_starts = {from_start}
   949	        else:
   950	            assert from_start is None
   951
   952	        # Loop over the nodes in the body, making arcs from each one's exits to
   953	        # the next node.
   954	        for body_node in body:
   955	            lineno = self.line_for_node(body_node)
   956	            if lineno not in self.statements:
   957	                maybe_body_node = self.find_non_missing_node(body_node)
   958	                if maybe_body_node is None:
   959	                    continue
   960	                body_node = maybe_body_node
   961	                lineno = self.line_for_node(body_node)
   962	            for prev_start in prev_starts:
   963	                self.add_arc(prev_start.lineno, lineno, prev_start.cause)
   964	            prev_starts = self.node_exits(body_node)
   965	        return prev_starts

   971	        to find a child node in self.statements.

   975	        """

   977	        # means we can avoid a function call in the 99.9999% case of not

    23	def _phys_tokens(toks: TokenInfos) -> TokenInfos:
    24	    """Return all physical tokens, even line continuations.
    25
    26	    tokenize.generate_tokens() doesn't return a token for the backslash that
    27	    continues lines.  This wrapper provides those tokens so that we can
    28	    re-create a faithful representation of the original source.
    29
    30	    Returns the same values as generate_tokens()
    31
    32	    """
    33	    last_line: str | None = None
    34	    last_lineno = -1
    35	    last_ttext: str = ""
    36	    for ttype, ttext, (slineno, scol), (elineno, ecol), ltext in toks:
    37	        if last_lineno != elineno:
    38	            if last_line and last_line.endswith("\\\n"):
    39	                # We are at the beginning of a new line, and the last line
    40	                # ended with a backslash.  We probably have to inject a
    41	                # backslash token into the stream. Unfortunately, there's more
    42	                # to figure out.  This code::
    43	                #
    44	                #   usage = """\
    45	                #   HEY THERE
    46	                #   """
    47	                #
    48	                # triggers this condition, but the token text is::
    49	                #
    50	                #   '"""\\\nHEY THERE\n"""'
    51	                #
    52	                # so we need to figure out if the backslash is already in the
    53	                # string token or not.
    54	                inject_backslash = True
    55	                if last_ttext.endswith("\\"):
    56	                    inject_backslash = False
    57	                elif ttype == token.STRING:

    61	                        #
    62	                        #   a = ["aaa",\ # there may be zero or more blanks between "," and "\".
    63	                        #        "bbb \

    65	                        #
    66	                        inject_backslash = True
    67	                    else:
    68	                        # It's a multi-line string and the first line ends with
    69	                        # a backslash, so we don't need to inject another.
    70	                        inject_backslash = False
    71	                elif env.PYBEHAVIOR.fstring_syntax and ttype == token.FSTRING_MIDDLE:
    72	                    inject_backslash = False
    73	                if inject_backslash:
    74	                    # Figure out what column the backslash is in.
    75	                    ccol = len(last_line.split("\n")[-2]) - 1

    91	Dynamic Context Switchers
    92	=========================
    93
    94	.. versionadded:: 5.0
    95
    96	Dynamic context switcher plugins implement the
    97	:meth:`~coverage.CoveragePlugin.dynamic_context` method to dynamically compute
    98	the context label for each measured frame.
    99
   100	Computed context labels are useful when you want to group measured data without
   101	modifying the source code.
   102
   103	For example, you could write a plugin that checks `frame.f_code` to inspect
   104	the currently executed method, and set the context label to a fully qualified
   105	method name if it's an instance method of `unittest.TestCase` and the method
   106	name starts with 'test'.  Such a plugin would provide basic coverage grouping
   107	by test and could be used with test runners that have no built-in coveragepy
   108	support.

   111	register your dynamic context switcher.

   136
   137	        Plug-in type: file tracer.
   138
   139	        Every Python source file is offered to your plug-in to give it a chance
   140	        to take responsibility for tracing the file.  If your plug-in can
   141	        handle the file, it should return a :class:`FileTracer` object.
   142	        Otherwise return None.
   143
   144	        There is no way to register your plug-in for particular files.
   145	        Instead, this method is invoked for all  files as they are executed,
   146	        and the plug-in decides whether it can trace the file or not.
   147	        Be prepared for `filename` to refer to all kinds of files that have
   148	        nothing to do with your plug-in.
   149

   151	        broad categories of behavior for a plug-in, depending on the kind of
   152	        files your plug-in supports:
   153

   155	          converted into a distinct Python file.  Your plug-in is invoked with
   156	          the Python file name, and it maps it back to its original source
   157	          file.

   160	          Python file.  In this case, your plug-in implements
   161	          :meth:`FileTracer.dynamic_source_filename` to provide the actual

   166	        paths, be sure to take this into account.

   352	    #: The kind of region, like `"function"` or `"class"`. Must be one of the

   113	        plugin_name = f"{self.current_module}.{plugin.__class__.__name__}"
