  1642	def cjkFragSplit(frags, maxWidths, calcBounds, encoding='utf8'):
  1643	    '''This attempts to be wordSplit for frags using the dumb algorithm'''
  1644	    U = []  #get a list of single glyphs with their widths etc etc
  1645	    for f in frags:
  1646	        text = f.text
  1647	        if isBytes(text):
  1648	            text = text.decode(encoding)
  1649	        if text:
  1650	            U.extend([cjkU(t,f,encoding) for t in text])
  1651	        else:
  1652	            U.append(cjkU(text,f,encoding))
  1653	    lines = []
  1654	    i = widthUsed = lineStartPos = 0
  1655	    maxWidth = maxWidths[0]
  1656	    nU = len(U)
  1657	    while i<nU:
  1658	        u = U[i]
  1659	        i += 1
  1660	        w = u.width
  1661	        if hasattr(w,'normalizedValue'):
  1662	            w._normalizer = maxWidth
  1663	            w = w.normalizedValue(maxWidth)
  1664	        widthUsed += w
  1665	        lineBreak = hasattr(u.frag,'lineBreak')
  1666	        endLine = (widthUsed>maxWidth + _FUZZ and widthUsed>0) or lineBreak
  1667	        if endLine:
  1668	            extraSpace = maxWidth - widthUsed
  1669	            if not lineBreak:
  1670	                if ord(u)<0x3000:
  1671	                    # we appear to be inside a non-Asian script section.
  1672	                    # (this is a very crude test but quick to compute).
  1673	                    # This is likely to be quite rare so the speed of the
  1674	                    # code below is hopefully not a big issue.  The main
  1675	                    # situation requiring this is that a document title
  1676	                    # with an english product name in it got cut.
  1677
  1678
  1679	                    # we count back and look for
  1680	                    #  - a space-like character
  1681	                    #  - reversion to Kanji (which would be a good split point)
  1682	                    #  - in the worst case, roughly half way back along the line
  1683	                    limitCheck = (lineStartPos+i)>>1        #(arbitrary taste issue)
  1684	                    for j in range(i-1,limitCheck,-1):
  1685	                        uj = U[j]
  1686	                        if uj and category(uj)=='Zs' or ord(uj)>=0x3000:
  1687	                            k = j+1
  1688	                            if k<i:
  1689	                                j = k+1
  1690	                                extraSpace += sum(U[ii].width for ii in range(j,i))
  1691	                                w = U[k].width
  1692	                                u = U[k]
  1693	                                i = j
  1694	                                break

  1696	                #we are pushing this character back, but
  1697	                #the most important of the Japanese typography rules
  1698	                #if this character cannot start a line, wrap it up to this line so it hangs
  1699	                #in the right margin. We won't do two or more though - that's unlikely and
  1700	                #would result in growing ugliness.
  1701	                #and increase the extra space
  1702	                #bug fix contributed by Alexander Vasilenko <alexs.vasilenko@gmail.com>
  1703	                if u not in ALL_CANNOT_START and i>lineStartPos+1:
  1704	                    #otherwise we need to push the character back
  1705	                    #the i>lineStart+1 condition ensures progress
  1706	                    i -= 1
  1707	                    extraSpace += w
  1708	            lines.append(makeCJKParaLine(U[lineStartPos:i],maxWidth,widthUsed,extraSpace,lineBreak,calcBounds))
  1709	            try:
  1710	                maxWidth = maxWidths[len(lines)]
  1711	            except IndexError:
  1712	                maxWidth = maxWidths[-1]  # use the last one
  1713
  1714	            lineStartPos = i
  1715	            widthUsed = 0
  1716
  1717	    #any characters left?
  1718	    if widthUsed > 0:
  1719	        lines.append(makeCJKParaLine(U[lineStartPos:],maxWidth,widthUsed,maxWidth-widthUsed,False,calcBounds))

    31	def _convnum(s, unit=1, allowRelative=True):
    32	    if s[0] in ('+','-') and allowRelative:
    33	        try:
    34	            return ('relative',int(s)*unit)
    35	        except ValueError:
    36	            return ('relative',float(s)*unit)
    37	    else:
    38	        try:
    39	            return int(s)*unit
    40	        except ValueError:
    41	            return float(s)*unit

    43	def _num(s, unit=1, allowRelative=True,
    44	        _unit_map = {'i':inch,'in':inch,'pt':1,'cm':cm,'mm':mm,'pica':pica },
    45	        _re_unit = re.compile(r'^\s*(.*)(i|in|cm|mm|pt|pica)\s*$'),
    46	        ):
    47	    """Convert a string like '10cm' to an int or float (in points).
    48	       The default unit is point, but optionally you can use other
    49	       default units like mm.
    50	    """
    51	    m = _re_unit.match(s)
    52	    if m:
    53	        unit = _unit_map[m.group(2)]
    54	        s = m.group(1)
    55	    return _convnum(s,unit,allowRelative)

    71	def _numpct(s,unit=1,allowRelative=False):
    72	    if s.endswith('%'):
    73	        return _PCT(_convnum(s[:-1],allowRelative=allowRelative))
    74	    else:

    77	class _PCT(float):
    78	    def __new__(cls,v):
    79	        self = float.__new__(cls,v*0.01)
    80	        self._normalizer = 1.0
    81	        self._value = v
    82	        return self
    83
    84	    def normalizedValue(self,normalizer):
    85	        if not normalizer:
    86	            normaliser = self._normalizer
    87	        r = _PCT(normalizer*self._value)
    88	        r._value = self._value
    89	        r._normalizer = normalizer
    90	        return r
    91
    92	    def __copy__(self):
    93	        r = _PCT(float(self))
    94	        r._value = self._value
    95	        r._normalizer = normalizer
    96	        return r
    97
    98	    def __deepcopy__(self,mem):
    99	        return self.__copy__()

   106	class _ExValidate:
   107	    '''class for syntax checking attributes
   108	    '''
   109	    def __init__(self,tag,attr):
   110	        self.tag = tag
   111	        self.attr = attr
   112
   113	    def invalid(self,s):
   114	        raise ValueError('<%s> invalid value %r for attribute %s' % (self.tag,s,self.attr))
   115
   116	    def validate(self, parser,s):
   117	        raise ValueError('abstract method called')
   118	        return s
   119
   120	    def __call__(self, parser, s):
   121	        try:
   122	            return self.validate(parser, s)
   123	        except:
   124	            self.invalid(s)
