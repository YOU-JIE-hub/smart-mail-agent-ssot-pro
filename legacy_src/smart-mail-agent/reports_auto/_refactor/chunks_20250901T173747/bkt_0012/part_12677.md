   108	# Snapshot representation of a symbol table node or type. The representation is
   109	# opaque -- the only supported operations are comparing for equality and
   110	# hashing (latter for type snapshots only). Snapshots can contain primitive
   111	# objects, nested tuples, lists and dictionaries and primitive objects (type

   124	def compare_symbol_table_snapshots(
   125	    name_prefix: str, snapshot1: dict[str, SymbolSnapshot], snapshot2: dict[str, SymbolSnapshot]
   126	) -> set[str]:
   127	    """Return names that are different in two snapshots of a symbol table.
   128
   129	    Only shallow (intra-module) differences are considered. References to things defined
   130	    outside the module are compared based on the name of the target only.
   131
   132	    Recurse into class symbol tables (if the class is defined in the target module).
   133
   134	    Return a set of fully-qualified names (e.g., 'mod.func' or 'mod.Class.method').
   135	    """
   136	    # Find names only defined only in one version.
   137	    names1 = {f"{name_prefix}.{name}" for name in snapshot1}
   138	    names2 = {f"{name_prefix}.{name}" for name in snapshot2}
   139	    triggers = names1 ^ names2
   140
   141	    # Look for names defined in both versions that are different.
   142	    for name in set(snapshot1.keys()) & set(snapshot2.keys()):
   143	        item1 = snapshot1[name]
   144	        item2 = snapshot2[name]
   145	        kind1 = item1[0]
   146	        kind2 = item2[0]
   147	        item_name = f"{name_prefix}.{name}"
   148	        if kind1 != kind2:
   149	            # Different kind of node in two snapshots -> trivially different.
   150	            triggers.add(item_name)
   151	        elif kind1 == "TypeInfo":
   152	            if item1[:-1] != item2[:-1]:
   153	                # Record major difference (outside class symbol tables).
   154	                triggers.add(item_name)
   155	            # Look for differences in nested class symbol table entries.
   156	            assert isinstance(item1[-1], dict)
   157	            assert isinstance(item2[-1], dict)
   158	            triggers |= compare_symbol_table_snapshots(item_name, item1[-1], item2[-1])
   159	        else:
   160	            # Shallow node (no interesting internal structure). Just use equality.
   161	            if snapshot1[name] != snapshot2[name]:
   162	                triggers.add(item_name)
   163
   164	    return triggers

   206	            result[name] = (
   207	                "ParamSpec",

   116	def merge_asts(
   117	    old: MypyFile, old_symbols: SymbolTable, new: MypyFile, new_symbols: SymbolTable
   118	) -> None:
   119	    """Merge a new version of a module AST to a previous version.
   120
   121	    The main idea is to preserve the identities of externally visible
   122	    nodes in the old AST (that have a corresponding node in the new AST).
   123	    All old node state (outside identity) will come from the new AST.
   124
   125	    When this returns, 'old' will refer to the merged AST, but 'new_symbols'
   126	    will be the new symbol table. 'new' and 'old_symbols' will no longer be
   127	    valid.
   128	    """
   129	    assert new.fullname == old.fullname
   130	    # Find the mapping from new to old node identities for all nodes
   131	    # whose identities should be preserved.
   132	    replacement_map = replacement_map_from_symbol_table(
   133	        old_symbols, new_symbols, prefix=old.fullname
   134	    )
   135	    # Also replace references to the new MypyFile node.
   136	    replacement_map[new] = old
   137	    # Perform replacements to everywhere within the new AST (not including symbol
   138	    # tables).
   139	    node = replace_nodes_in_ast(new, replacement_map)
   140	    assert node is old
   141	    # Also replace AST node references in the *new* symbol table (we'll
   142	    # continue to use the new symbol table since it has all the new definitions
   143	    # that have no correspondence in the old AST).
   144	    replace_nodes_in_symbol_table(new_symbols, replacement_map)

   147	def replacement_map_from_symbol_table(
   148	    old: SymbolTable, new: SymbolTable, prefix: str
   149	) -> dict[SymbolNode, SymbolNode]:
   150	    """Create a new-to-old object identity map by comparing two symbol table revisions.
   151
   152	    Both symbol tables must refer to revisions of the same module id. The symbol tables
   153	    are compared recursively (recursing into nested class symbol tables), but only within
   154	    the given module prefix. Don't recurse into other modules accessible through the symbol
   155	    table.
   156	    """
   157	    replacements: dict[SymbolNode, SymbolNode] = {}
   158	    for name, node in old.items():
   159	        if name in new and (
   160	            node.kind == MDEF or node.node and get_prefix(node.node.fullname) == prefix
   161	        ):
   162	            new_node = new[name]
   163	            if (
   164	                type(new_node.node) == type(node.node)
   165	                and new_node.node
   166	                and node.node
   167	                and new_node.node.fullname == node.node.fullname
   168	                and new_node.kind == node.kind
   169	            ):
   170	                replacements[new_node.node] = node.node
   171	                if isinstance(node.node, TypeInfo) and isinstance(new_node.node, TypeInfo):
   172	                    type_repl = replacement_map_from_symbol_table(
   173	                        node.node.names, new_node.node.names, prefix
   174	                    )
   175	                    replacements.update(type_repl)
   176	                    if node.node.special_alias and new_node.node.special_alias:
   177	                        replacements[new_node.node.special_alias] = node.node.special_alias
   178	    return replacements

     1	"""Generate fine-grained dependencies for AST nodes, for use in the daemon mode.
     2
     3	Dependencies are stored in a map from *triggers* to *sets of affected locations*.
     4
     5	A trigger is a string that represents a program property that has changed, such
     6	as the signature of a specific function. Triggers are written as '<...>' (angle
     7	brackets). When a program property changes, we determine the relevant trigger(s)
     8	and all affected locations. The latter are stale and will have to be reprocessed.
     9
    10	An affected location is a string than can refer to a *target* (a non-nested
    11	function or method, or a module top level), a class, or a trigger (for
    12	recursively triggering other triggers).
    13
    14	Here's an example representation of a simple dependency map (in format
    15	"<trigger> -> locations"):
    16
    17	  <m.A.g> -> m.f
    18	  <m.A> -> <m.f>, m.A, m.f
    19
    20	Assuming 'A' is a class, this means that
    21
    22	1) if a property of 'm.A.g', such as the signature, is changed, we need
    23	   to process target (function) 'm.f'
    24
    25	2) if the MRO or other significant property of class 'm.A' changes, we
    26	   need to process target 'm.f', the entire class 'm.A', and locations
    27	   triggered by trigger '<m.f>' (this explanation is a bit simplified;
    28	   see below for more details).
    29
    30	The triggers to fire are determined using mypy.server.astdiff.
    31
    32	Examples of triggers:
    33
    34	* '<mod.x>' represents a module attribute/function/class. If any externally
    35	  visible property of 'x' changes, this gets fired. For changes within
    36	  classes, only "big" changes cause the class to be triggered (such as a
    37	  change in MRO). Smaller changes, such as changes to some attributes, don't
    38	  trigger the entire class.
    39	* '<mod.Cls.x>' represents the type and kind of attribute/method 'x' of
    40	  class 'mod.Cls'. This can also refer to an attribute inherited from a
    41	  base class (relevant if it's accessed through a value of type 'Cls'
    42	  instead of the base class type).
    43	* '<package.mod>' represents the existence of module 'package.mod'. This
    44	  gets triggered if 'package.mod' is created or deleted, or if it gets
    45	  changed into something other than a module.
    46
    47	Examples of locations:
    48
    49	* 'mod' is the top level of module 'mod' (doesn't include any function bodies,
    50	  but includes class bodies not nested within a function).
    51	* 'mod.f' is function 'f' in module 'mod' (module-level variables aren't separate
    52	  locations but are included in the module top level). Functions also include
    53	  any nested functions and classes -- such nested definitions aren't separate
    54	  locations, for simplicity of implementation.
    55	* 'mod.Cls.f' is method 'f' of 'mod.Cls'. Non-method attributes aren't locations.
    56	* 'mod.Cls' represents each method in class 'mod.Cls' + the top-level of the
    57	  module 'mod'. (To simplify the implementation, there is no location that only
    58	  includes the body of a class without the entire surrounding module top level.)
    59	* Trigger '<...>' as a location is an indirect way of referring to all
    60	  locations triggered by the trigger. These indirect locations keep the
    61	  dependency map smaller and easier to manage.
    62
    63	Triggers can be triggered by program changes such as these:
    64
    65	* Addition or deletion of an attribute (or module).
    66	* Change of the kind of thing a name represents (such as a change from a function
    67	  to a class).

   198	    """Get dependencies of a target -- don't recursive into nested targets."""
   199	    # TODO: Add tests for this function.
   200	    visitor = DependencyVisitor(type_map, python_version, module_tree.alias_deps)
   201	    with visitor.scope.module_scope(module_id):
   202	        if isinstance(target, MypyFile):
   203	            # Only get dependencies of the top-level of the module. Don't recurse into
   204	            # functions.
   205	            for defn in target.defs:
   206	                # TODO: Recurse into top-level statements and class bodies but skip functions.
   207	                if not isinstance(defn, (ClassDef, Decorator, FuncDef, OverloadedFuncDef)):
   208	                    defn.accept(visitor)
   209	        elif isinstance(target, FuncBase) and target.info:
   210	            # It's a method.
   211	            # TODO: Methods in nested classes.
   212	            with visitor.scope.class_scope(target.info):
   213	                target.accept(visitor)
   214	        else:
   215	            target.accept(visitor)
   216	    return visitor.map

   219	class DependencyVisitor(TraverserVisitor):
   220	    def __init__(
   221	        self,
   222	        type_map: dict[Expression, Type],
   223	        python_version: tuple[int, int],
   224	        alias_deps: defaultdict[str, set[str]],
   225	        options: Options | None = None,
   226	    ) -> None:
   227	        self.scope = Scope()
   228	        self.type_map = type_map
   229	        # This attribute holds a mapping from target to names of type aliases
   230	        # it depends on. These need to be processed specially, since they are
   231	        # only present in expanded form in symbol tables. For example, after:
   232	        #    A = List[int]
   233	        #    x: A
   234	        # The module symbol table will just have a Var `x` with type `List[int]`,
   235	        # and the dependency of `x` on `A` is lost. Therefore the alias dependencies
   236	        # are preserved at alias expansion points in `semanal.py`, stored as an attribute
   237	        # on MypyFile, and then passed here.
   238	        self.alias_deps = alias_deps
   239	        self.map: dict[str, set[str]] = {}
   240	        self.is_class = False
   241	        self.is_package_init_file = False
   242	        self.options = options

   244	    def visit_mypy_file(self, o: MypyFile) -> None:
   245	        with self.scope.module_scope(o.fullname):
   246	            self.is_package_init_file = o.is_package_init_file()
   247	            self.add_type_alias_deps(self.scope.current_target())
   248	            for trigger, targets in o.plugin_deps.items():
   249	                self.map.setdefault(trigger, set()).update(targets)
   250	            super().visit_mypy_file(o)

   252	    def visit_func_def(self, o: FuncDef) -> None:
   253	        with self.scope.function_scope(o):
   254	            target = self.scope.current_target()
   255	            if o.type:
   256	                if self.is_class and isinstance(o.type, FunctionLike):
   257	                    signature: Type = bind_self(o.type)
   258	                else:
   259	                    signature = o.type
   260	                for trigger in self.get_type_triggers(signature):
   261	                    self.add_dependency(trigger)
   262	                    self.add_dependency(trigger, target=make_trigger(target))
   263	            if o.info:
   264	                for base in non_trivial_bases(o.info):
   265	                    # Base class __init__/__new__ doesn't generate a logical
   266	                    # dependency since the override can be incompatible.
   267	                    if not self.use_logical_deps() or o.name not in ("__init__", "__new__"):
   268	                        self.add_dependency(make_trigger(base.fullname + "." + o.name))
   269	            self.add_type_alias_deps(self.scope.current_target())
   270	            super().visit_func_def(o)
   271	            variants = set(o.expanded) - {o}
   272	            for ex in variants:
   273	                if isinstance(ex, FuncDef):
   274	                    super().visit_func_def(ex)

   276	    def visit_decorator(self, o: Decorator) -> None:
   277	        if not self.use_logical_deps():
   278	            # We don't need to recheck outer scope for an overload, only overload itself.
   279	            # Also if any decorator is nested, it is not externally visible, so we don't need to
   280	            # generate dependency.
   281	            if not o.func.is_overload and self.scope.current_function_name() is None:
   282	                self.add_dependency(make_trigger(o.func.fullname))
   283	        else:
   284	            # Add logical dependencies from decorators to the function. For example,
   285	            # if we have
   286	            #     @dec
   287	            #     def func(): ...
   288	            # then if `dec` is unannotated, then it will "spoil" `func` and consequently
   289	            # all call sites, making them all `Any`.
   290	            for d in o.decorators:
   291	                tname: str | None = None
   292	                if isinstance(d, RefExpr) and d.fullname:
   293	                    tname = d.fullname
   294	                if isinstance(d, CallExpr) and isinstance(d.callee, RefExpr) and d.callee.fullname:
   295	                    tname = d.callee.fullname
   296	                if tname is not None:
   297	                    self.add_dependency(make_trigger(tname), make_trigger(o.func.fullname))
   298	        super().visit_decorator(o)

   300	    def visit_class_def(self, o: ClassDef) -> None:
   301	        with self.scope.class_scope(o.info):
   302	            target = self.scope.current_full_target()
   303	            self.add_dependency(make_trigger(target), target)
   304	            old_is_class = self.is_class
   305	            self.is_class = True
   306	            # Add dependencies to type variables of a generic class.
   307	            for tv in o.type_vars:
   308	                self.add_dependency(make_trigger(tv.fullname), target)
   309	            self.process_type_info(o.info)
   310	            super().visit_class_def(o)
   311	            self.is_class = old_is_class

   318	    def process_type_info(self, info: TypeInfo) -> None:
   319	        target = self.scope.current_full_target()
   320	        for base in info.bases:
   321	            self.add_type_dependencies(base, target=target)
   322	        if info.tuple_type:
   323	            self.add_type_dependencies(info.tuple_type, target=make_trigger(target))
   324	        if info.typeddict_type:
   325	            self.add_type_dependencies(info.typeddict_type, target=make_trigger(target))
   326	        if info.declared_metaclass:
   327	            self.add_type_dependencies(info.declared_metaclass, target=make_trigger(target))
   328	        if info.is_protocol:
   329	            for base_info in info.mro[:-1]:
   330	                # We add dependencies from whole MRO to cover explicit subprotocols.
   331	                # For example:
   332	                #
   333	                #     class Super(Protocol):
   334	                #         x: int
   335	                #     class Sub(Super, Protocol):
   336	                #         y: int
   337	                #
   338	                # In this example we add <Super[wildcard]> -> <Sub>, to invalidate Sub if
   339	                # a new member is added to Super.
   340	                self.add_dependency(
   341	                    make_wildcard_trigger(base_info.fullname), target=make_trigger(target)
   342	                )
   343	                # More protocol dependencies are collected in type_state._snapshot_protocol_deps
   344	                # after a full run or update is finished.
   345
   346	        self.add_type_alias_deps(self.scope.current_target())
   347	        for name, node in info.names.items():
   348	            if isinstance(node.node, Var):
   349	                # Recheck Liskov if needed, self definitions are checked in the defining method
   350	                if node.node.is_initialized_in_class and has_user_bases(info):
   351	                    self.add_dependency(make_trigger(info.fullname + "." + name))
   352	                for base_info in non_trivial_bases(info):
   353	                    # If the type of an attribute changes in a base class, we make references
   354	                    # to the attribute in the subclass stale.
   355	                    self.add_dependency(
   356	                        make_trigger(base_info.fullname + "." + name),
   357	                        target=make_trigger(info.fullname + "." + name),
   358	                    )
   359	        for base_info in non_trivial_bases(info):
   360	            for name, node in base_info.names.items():
   361	                if self.use_logical_deps():
   362	                    # Skip logical dependency if an attribute is not overridden. For example,
   363	                    # in case of:
   364	                    #     class Base:
   365	                    #         x = 1
   366	                    #         y = 2
   367	                    #     class Sub(Base):
   368	                    #         x = 3
   369	                    # we skip <Base.y> -> <Child.y>, because even if `y` is unannotated it
   370	                    # doesn't affect precision of Liskov checking.
   371	                    if name not in info.names:
   372	                        continue
   373	                    # __init__ and __new__ can be overridden with different signatures, so no
   374	                    # logical dependency.
   375	                    if name in ("__init__", "__new__"):
   376	                        continue
   377	                self.add_dependency(
   378	                    make_trigger(base_info.fullname + "." + name),
   379	                    target=make_trigger(info.fullname + "." + name),
   380	                )
   381	            if not self.use_logical_deps():
   382	                # These dependencies are only useful for propagating changes --
   383	                # they aren't logical dependencies since __init__ and __new__ can be
   384	                # overridden with a different signature.
   385	                self.add_dependency(
   386	                    make_trigger(base_info.fullname + ".__init__"),
   387	                    target=make_trigger(info.fullname + ".__init__"),
   388	                )
   389	                self.add_dependency(
   390	                    make_trigger(base_info.fullname + ".__new__"),
   391	                    target=make_trigger(info.fullname + ".__new__"),
   392	                )
   393	                # If the set of abstract attributes change, this may invalidate class
   394	                # instantiation, or change the generated error message, since Python checks
   395	                # class abstract status when creating an instance.
   396	                self.add_dependency(
   397	                    make_trigger(base_info.fullname + ".(abstract)"),
   398	                    target=make_trigger(info.fullname + ".__init__"),
   399	                )
   400	                # If the base class abstract attributes change, subclass abstract
   401	                # attributes need to be recalculated.
   402	                self.add_dependency(make_trigger(base_info.fullname + ".(abstract)"))
   403

   431	    def visit_assignment_stmt(self, o: AssignmentStmt) -> None:
   432	        rvalue = o.rvalue
   433	        if isinstance(rvalue, CallExpr) and isinstance(rvalue.analyzed, TypeVarExpr):
   434	            analyzed = rvalue.analyzed
   435	            self.add_type_dependencies(
   436	                analyzed.upper_bound, target=make_trigger(analyzed.fullname)
   437	            )
   438	            for val in analyzed.values:
   439	                self.add_type_dependencies(val, target=make_trigger(analyzed.fullname))
   440	            # We need to re-analyze the definition if bound or value is deleted.
   441	            super().visit_call_expr(rvalue)
   442	        elif isinstance(rvalue, CallExpr) and isinstance(rvalue.analyzed, NamedTupleExpr):
   443	            # Depend on types of named tuple items.
   444	            info = rvalue.analyzed.info
   445	            prefix = f"{self.scope.current_full_target()}.{info.name}"
   446	            for name, symnode in info.names.items():
   447	                if not name.startswith("_") and isinstance(symnode.node, Var):
   448	                    typ = symnode.node.type
   449	                    if typ:
   450	                        self.add_type_dependencies(typ)
   451	                        self.add_type_dependencies(typ, target=make_trigger(prefix))
   452	                        attr_target = make_trigger(f"{prefix}.{name}")
   453	                        self.add_type_dependencies(typ, target=attr_target)
   454	        elif isinstance(rvalue, CallExpr) and isinstance(rvalue.analyzed, TypedDictExpr):
   455	            # Depend on the underlying typeddict type
   456	            info = rvalue.analyzed.info
   457	            assert info.typeddict_type is not None
   458	            prefix = f"{self.scope.current_full_target()}.{info.name}"
   459	            self.add_type_dependencies(info.typeddict_type, target=make_trigger(prefix))
   460	        elif isinstance(rvalue, CallExpr) and isinstance(rvalue.analyzed, EnumCallExpr):
   461	            # Enum values are currently not checked, but for future we add the deps on them
   462	            for name, symnode in rvalue.analyzed.info.names.items():
   463	                if isinstance(symnode.node, Var) and symnode.node.type:
   464	                    self.add_type_dependencies(symnode.node.type)
   465	        elif o.is_alias_def:
   466	            assert len(o.lvalues) == 1
   467	            lvalue = o.lvalues[0]
   468	            assert isinstance(lvalue, NameExpr)
   469	            typ = get_proper_type(self.type_map.get(lvalue))
   470	            if isinstance(typ, FunctionLike) and typ.is_type_obj():
   471	                class_name = typ.type_object().fullname
   472	                self.add_dependency(make_trigger(class_name + ".__init__"))
   473	                self.add_dependency(make_trigger(class_name + ".__new__"))
   474	            if isinstance(rvalue, IndexExpr) and isinstance(rvalue.analyzed, TypeAliasExpr):
   475	                self.add_type_dependencies(rvalue.analyzed.node.target)
   476	            elif typ:
   477	                self.add_type_dependencies(typ)
   478	        else:
   479	            # Normal assignment
   480	            super().visit_assignment_stmt(o)
   481	            for lvalue in o.lvalues:
   482	                self.process_lvalue(lvalue)
   483	            items = o.lvalues + [rvalue]
   484	            for i in range(len(items) - 1):
   485	                lvalue = items[i]
   486	                rvalue = items[i + 1]
   487	                if isinstance(lvalue, TupleExpr):
   488	                    self.add_attribute_dependency_for_expr(rvalue, "__iter__")
   489	            if o.type:
   490	                self.add_type_dependencies(o.type)
   491	        if self.use_logical_deps() and o.unanalyzed_type is None:
   492	            # Special case: for definitions without an explicit type like this:
   493	            #     x = func(...)
   494	            # we add a logical dependency <func> -> <x>, because if `func` is not annotated,
   495	            # then it will make all points of use of `x` unchecked.
   496	            if (
   497	                isinstance(rvalue, CallExpr)
   498	                and isinstance(rvalue.callee, RefExpr)
   499	                and rvalue.callee.fullname
   500	            ):
   501	                fname: str | None = None
   502	                if isinstance(rvalue.callee.node, TypeInfo):
   503	                    # use actual __init__ as a dependency source
   504	                    init = rvalue.callee.node.get("__init__")
   505	                    if init and isinstance(init.node, SYMBOL_FUNCBASE_TYPES):
   506	                        fname = init.node.fullname
   507	                else:
   508	                    fname = rvalue.callee.fullname
   509	                if not fname:
   510	                    return
   511	                for lv in o.lvalues:
   512	                    if isinstance(lv, RefExpr) and lv.fullname and lv.is_new_def:
   513	                        if lv.kind == LDEF:
   514	                            return  # local definitions don't generate logical deps
   515	                        self.add_dependency(make_trigger(fname), make_trigger(lv.fullname))
   516
   517	    def process_lvalue(self, lvalue: Expression) -> None:
   518	        """Generate additional dependencies for an lvalue."""
   519	        if isinstance(lvalue, IndexExpr):
   520	            self.add_operator_method_dependency(lvalue.base, "__setitem__")
   521	        elif isinstance(lvalue, NameExpr):
   522	            if lvalue.kind in (MDEF, GDEF):
   523	                # Assignment to an attribute in the class body, or direct assignment to a
   524	                # global variable.
   525	                lvalue_type = self.get_non_partial_lvalue_type(lvalue)
   526	                type_triggers = self.get_type_triggers(lvalue_type)
   527	                attr_trigger = make_trigger(f"{self.scope.current_full_target()}.{lvalue.name}")
   528	                for type_trigger in type_triggers:
   529	                    self.add_dependency(type_trigger, attr_trigger)
   530	        elif isinstance(lvalue, MemberExpr):
   531	            if self.is_self_member_ref(lvalue) and lvalue.is_new_def:
   532	                node = lvalue.node
   533	                if isinstance(node, Var):
   534	                    info = node.info
   535	                    if info and has_user_bases(info):
   536	                        # Recheck Liskov for self definitions
   537	                        self.add_dependency(make_trigger(info.fullname + "." + lvalue.name))
   538	            if lvalue.kind is None:
   539	                # Reference to a non-module attribute
   540	                if lvalue.expr not in self.type_map:
   541	                    # Unreachable assignment -> not checked so no dependencies to generate.
   542	                    return
   543	                object_type = self.type_map[lvalue.expr]
   544	                lvalue_type = self.get_non_partial_lvalue_type(lvalue)
   545	                type_triggers = self.get_type_triggers(lvalue_type)
   546	                for attr_trigger in self.attribute_triggers(object_type, lvalue.name):
   547	                    for type_trigger in type_triggers:
   548	                        self.add_dependency(type_trigger, attr_trigger)
   549	        elif isinstance(lvalue, TupleExpr):
   550	            for item in lvalue.items:
   551	                self.process_lvalue(item)
   552	        elif isinstance(lvalue, StarExpr):
   553	            self.process_lvalue(lvalue.expr)
   554
   555	    def is_self_member_ref(self, memberexpr: MemberExpr) -> bool:
   556	        """Does memberexpr to refer to an attribute of self?"""
   557	        if not isinstance(memberexpr.expr, NameExpr):
   558	            return False
   559	        node = memberexpr.expr.node
   560	        return isinstance(node, Var) and node.is_self

   562	    def get_non_partial_lvalue_type(self, lvalue: RefExpr) -> Type:
   563	        if lvalue not in self.type_map:
   564	            # Likely a block considered unreachable during type checking.
   565	            return UninhabitedType()
   566	        lvalue_type = get_proper_type(self.type_map[lvalue])
   567	        if isinstance(lvalue_type, PartialType):
   568	            if isinstance(lvalue.node, Var):
   569	                if lvalue.node.type:
   570	                    lvalue_type = get_proper_type(lvalue.node.type)
   571	                else:
   572	                    lvalue_type = UninhabitedType()
   573	            else:
   574	                # Probably a secondary, non-definition assignment that doesn't
   575	                # result in a non-partial type. We won't be able to infer any
   576	                # dependencies from this so just return something. (The first,
   577	                # definition assignment with a partial type is handled
   578	                # differently, in the semantic analyzer.)
   579	                assert not lvalue.is_new_def
   580	                return UninhabitedType()
   581	        return lvalue_type

   586	        method = op_methods[o.op]
   587	        self.add_attribute_dependency_for_expr(o.lvalue, method)

   592	    def visit_for_stmt(self, o: ForStmt) -> None:
   593	        super().visit_for_stmt(o)
   594	        if not o.is_async:
   595	            # __getitem__ is only used if __iter__ is missing but for simplicity we
   596	            # just always depend on both.
   597	            self.add_attribute_dependency_for_expr(o.expr, "__iter__")
   598	            self.add_attribute_dependency_for_expr(o.expr, "__getitem__")
   599	            if o.inferred_iterator_type:
   600	                self.add_attribute_dependency(o.inferred_iterator_type, "__next__")
   601	        else:
   602	            self.add_attribute_dependency_for_expr(o.expr, "__aiter__")
   603	            if o.inferred_iterator_type:
   604	                self.add_attribute_dependency(o.inferred_iterator_type, "__anext__")
   605
   606	        self.process_lvalue(o.index)
   607	        if isinstance(o.index, TupleExpr):
   608	            # Process multiple assignment to index variables.
   609	            item_type = o.inferred_item_type
   610	            if item_type:
   611	                # This is similar to above.
   612	                self.add_attribute_dependency(item_type, "__iter__")
   613	                self.add_attribute_dependency(item_type, "__getitem__")
   614	        if o.index_type:
   615	            self.add_type_dependencies(o.index_type)
