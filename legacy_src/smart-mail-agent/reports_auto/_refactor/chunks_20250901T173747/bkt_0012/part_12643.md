   653	    def update(
   654	        self,
   655	        other_data: CoverageData,
   656	        map_path: Callable[[str], str] | None = None,
   657	    ) -> None:
   658	        """Update this data with data from another :class:`CoverageData`.
   659
   660	        If `map_path` is provided, it's a function that re-map paths to match
   661	        the local machine's.  Note: `map_path` is None only when called
   662	        directly from the test suite.
   663
   664	        """
   665	        if self._debug.should("dataop"):
   666	            self._debug.write("Updating with data from {!r}".format(
   667	                getattr(other_data, "_filename", "???"),
   668	            ))
   669	        if self._has_lines and other_data._has_arcs:
   670	            raise DataError("Can't combine branch coverage data with statement data")
   671	        if self._has_arcs and other_data._has_lines:
   672	            raise DataError("Can't combine statement coverage data with branch data")
   673
   674	        map_path = map_path or (lambda p: p)
   675
   676	        # Force the database we're writing to to exist before we start nesting contexts.
   677	        self._start_using()
   678
   679	        # Collector for all arcs, lines and tracers
   680	        other_data.read()
   681	        with other_data._connect() as con:
   682	            # Get files data.
   683	            with con.execute("select path from file") as cur:
   684	                files = {path: map_path(path) for (path,) in cur}
   685
   686	            # Get contexts data.
   687	            with con.execute("select context from context") as cur:
   688	                contexts = [context for (context,) in cur]
   689
   690	            # Get arc data.
   691	            with con.execute(
   692	                "select file.path, context.context, arc.fromno, arc.tono " +
   693	                "from arc " +
   694	                "inner join file on file.id = arc.file_id " +
   695	                "inner join context on context.id = arc.context_id",
   696	            ) as cur:
   697	                arcs = [
   698	                    (files[path], context, fromno, tono)
   699	                    for (path, context, fromno, tono) in cur
   700	                ]
   701
   702	            # Get line data.
   703	            with con.execute(
   704	                "select file.path, context.context, line_bits.numbits " +
   705	                "from line_bits " +
   706	                "inner join file on file.id = line_bits.file_id " +
   707	                "inner join context on context.id = line_bits.context_id",
   708	            ) as cur:
   709	                lines: dict[tuple[str, str], bytes] = {}
   710	                for path, context, numbits in cur:
   711	                    key = (files[path], context)
   712	                    if key in lines:
   713	                        numbits = numbits_union(lines[key], numbits)
   714	                    lines[key] = numbits
   715
   716	            # Get tracer data.
   717	            with con.execute(
   718	                "select file.path, tracer " +
   719	                "from tracer " +
   720	                "inner join file on file.id = tracer.file_id",
   721	            ) as cur:
   722	                tracers = {files[path]: tracer for (path, tracer) in cur}
   723
   724	        with self._connect() as con:
   725	            assert con.con is not None
   726	            con.con.isolation_level = "IMMEDIATE"
   727
   728	            # Get all tracers in the DB. Files not in the tracers are assumed
   729	            # to have an empty string tracer. Since Sqlite does not support
   730	            # full outer joins, we have to make two queries to fill the
   731	            # dictionary.
   732	            with con.execute("select path from file") as cur:
   733	                this_tracers = {path: "" for path, in cur}
   734	            with con.execute(
   735	                "select file.path, tracer from tracer " +
   736	                "inner join file on file.id = tracer.file_id",
   737	            ) as cur:
   738	                this_tracers.update({
   739	                    map_path(path): tracer
   740	                    for path, tracer in cur
   741	                })
   742
   743	            # Create all file and context rows in the DB.
   744	            con.executemany_void(
   745	                "insert or ignore into file (path) values (?)",
   746	                ((file,) for file in files.values()),
   747	            )
   748	            with con.execute("select id, path from file") as cur:
   749	                file_ids = {path: id for id, path in cur}
   750	            self._file_map.update(file_ids)
   751	            con.executemany_void(
   752	                "insert or ignore into context (context) values (?)",
   753	                ((context,) for context in contexts),
   754	            )
   755	            with con.execute("select id, context from context") as cur:
   756	                context_ids = {context: id for id, context in cur}
   757
   758	            # Prepare tracers and fail, if a conflict is found.
   759	            # tracer_paths is used to ensure consistency over the tracer data
   760	            # and tracer_map tracks the tracers to be inserted.
   761	            tracer_map = {}
   762	            for path in files.values():
   763	                this_tracer = this_tracers.get(path)
   764	                other_tracer = tracers.get(path, "")
   765	                # If there is no tracer, there is always the None tracer.
   766	                if this_tracer is not None and this_tracer != other_tracer:
   767	                    raise DataError(
   768	                        "Conflicting file tracer name for '{}': {!r} vs {!r}".format(
   769	                            path, this_tracer, other_tracer,
   770	                        ),
   771	                    )
   772	                tracer_map[path] = other_tracer
   773
   774	            # Prepare arc and line rows to be inserted by converting the file
   775	            # and context strings with integer ids. Then use the efficient
   776	            # `executemany()` to insert all rows at once.
   777
   778	            if arcs:
   779	                self._choose_lines_or_arcs(arcs=True)
   780
   781	                arc_rows = (
   782	                    (file_ids[file], context_ids[context], fromno, tono)
   783	                    for file, context, fromno, tono in arcs
   784	                )
   785
   786	                # Write the combined data.
   787	                con.executemany_void(
   788	                    "insert or ignore into arc " +
   789	                    "(file_id, context_id, fromno, tono) values (?, ?, ?, ?)",
   790	                    arc_rows,
   791	                )
   792
   793	            if lines:
   794	                self._choose_lines_or_arcs(lines=True)
   795
   796	                for (file, context), numbits in lines.items():
   797	                    with con.execute(
   798	                        "select numbits from line_bits where file_id = ? and context_id = ?",
   799	                        (file_ids[file], context_ids[context]),
   800	                    ) as cur:
   801	                        existing = list(cur)
   802	                    if existing:
   803	                        lines[(file, context)] = numbits_union(numbits, existing[0][0])
   804
   805	                con.executemany_void(
   806	                    "insert or replace into line_bits " +
   807	                    "(file_id, context_id, numbits) values (?, ?, ?)",
   808	                    [
   809	                        (file_ids[file], context_ids[context], numbits)
   810	                        for (file, context), numbits in lines.items()
   811	                    ],
   812	                )
   813
   814	            con.executemany_void(
   815	                "insert or ignore into tracer (file_id, tracer) values (?, ?)",
   816	                ((file_ids[filename], tracer) for filename, tracer in tracer_map.items()),
   817	            )
   818
   819	        if not self._no_disk:
   820	            # Update all internal cache data.
   821	            self._reset()
   822	            self.read()

  1084	def filename_suffix(suffix: str | bool | None) -> str | None:
  1085	    """Compute a filename suffix for a data file.
  1086
  1087	    If `suffix` is a string or None, simply return it. If `suffix` is True,
  1088	    then build a suffix incorporating the hostname, process id, and a random
  1089	    number.
  1090
  1091	    Returns a string or None.
  1092
  1093	    """
  1094	    if suffix is True:
  1095	        # If data_suffix was a simple true value, then make a suffix with
  1096	        # plenty of distinguishing information.  We do this here in
  1097	        # `save()` at the last minute so that the pid will be correct even
  1098	        # if the process forks.
  1099	        die = random.Random(os.urandom(8))
  1100	        letters = string.ascii_uppercase + string.ascii_lowercase
  1101	        rolls = "".join(die.choice(letters) for _ in range(6))
  1102	        suffix = f"{socket.gethostname()}.{os.getpid()}.X{rolls}x"
  1103	    elif suffix is False:
  1104	        suffix = None
  1105	    return suffix

    80	class Templite:
    81	    """A simple template renderer, for a nano-subset of Django syntax.
    82
    83	    Supported constructs are extended variable access::
    84
    85	        {{var.modifier.modifier|filter|filter}}
    86
    87	    loops::
    88
    89	        {% for var in list %}...{% endfor %}
    90
    91	    and ifs::
    92
    93	        {% if var %}...{% endif %}
    94
    95	    Comments are within curly-hash markers::
    96
    97	        {# This will be ignored #}
    98
    99	    Lines between `{% joined %}` and `{% endjoined %}` will have lines stripped
   100	    and joined.  Be careful, this could join words together!
   101
   102	    Any of these constructs can have a hyphen at the end (`-}}`, `-%}`, `-#}`),
   103	    which will collapse the white space following the tag.
   104
   105	    Construct a Templite with the template text, then use `render` against a
   106	    dictionary context to create a finished string::
   107
   108	        templite = Templite('''
   109	            <h1>Hello {{name|upper}}!</h1>
   110	            {% for topic in topics %}
   111	                <p>You are interested in {{topic}}.</p>
   112	            {% endif %}
   113	            ''',
   114	            {"upper": str.upper},
   115	        )
   116	        text = templite.render({
   117	            "name": "Ned",
   118	            "topics": ["Python", "Geometry", "Juggling"],
   119	        })
   120
   121	    """
   122	    def __init__(self, text: str, *contexts: dict[str, Any]) -> None:
   123	        """Construct a Templite with the given `text`.
   124
   125	        `contexts` are dictionaries of values to use for future renderings.
   126	        These are good for filters and global values.
   127
   128	        """
   129	        self.context = {}
   130	        for context in contexts:
   131	            self.context.update(context)
   132
   133	        self.all_vars: set[str] = set()
   134	        self.loop_vars: set[str] = set()
   135
   136	        # We construct a function in source form, then compile it and hold onto
   137	        # it, and execute it to render the template.
   138	        code = CodeBuilder()
   139
   140	        code.add_line("def render_function(context, do_dots):")
   141	        code.indent()

   144	        code.add_line("append_result = result.append")
   145	        code.add_line("extend_result = result.extend")
   146	        code.add_line("to_str = str")

   148	        buffered: list[str] = []

   150	        def flush_output() -> None:
   151	            """Force `buffered` to the code builder."""
   152	            if len(buffered) == 1:
   153	                code.add_line("append_result(%s)" % buffered[0])
   154	            elif len(buffered) > 1:
   155	                code.add_line("extend_result([%s])" % ", ".join(buffered))
   156	            del buffered[:]

   158	        ops_stack = []
   159
   160	        # Split the text to form a list of tokens.
   161	        tokens = re.split(r"(?s)({{.*?}}|{%.*?%}|{#.*?#})", text)
   162
   163	        squash = in_joined = False
   164
   165	        for token in tokens:
   166	            if token.startswith("{"):
   167	                start, end = 2, -2
   168	                squash = (token[-3] == "-")
   169	                if squash:
   170	                    end = -3
   171
   172	                if token.startswith("{#"):
   173	                    # Comment: ignore it and move on.
   174	                    continue
   175	                elif token.startswith("{{"):
   176	                    # An expression to evaluate.
   177	                    expr = self._expr_code(token[start:end].strip())

   179	                else:
   180	                    # token.startswith("{%")
   181	                    # Action tag: split into words and parse further.
   182	                    flush_output()

   184	                    words = token[start:end].strip().split()

   187	                        if len(words) != 2:

   189	                        ops_stack.append("if")

   191	                        code.indent()
   192	                    elif words[0] == "for":
   193	                        # A loop: iterate over expression result.
   194	                        if len(words) != 4 or words[2] != "in":
   195	                            self._syntax_error("Don't understand for", token)
   196	                        ops_stack.append("for")
   197	                        self._variable(words[1], self.loop_vars)
   198	                        code.add_line(
   199	                            f"for c_{words[1]} in {self._expr_code(words[3])}:",
   200	                        )
   201	                        code.indent()
   202	                    elif words[0] == "joined":
   203	                        ops_stack.append("joined")
   204	                        in_joined = True
   205	                    elif words[0].startswith("end"):
   206	                        # Endsomething.  Pop the ops stack.
   207	                        if len(words) != 1:
   208	                            self._syntax_error("Don't understand end", token)
   209	                        end_what = words[0][3:]
   210	                        if not ops_stack:
   211	                            self._syntax_error("Too many ends", token)
   212	                        start_what = ops_stack.pop()
   213	                        if start_what != end_what:
   214	                            self._syntax_error("Mismatched end tag", end_what)
   215	                        if end_what == "joined":
   216	                            in_joined = False
   217	                        else:
   218	                            code.dedent()
   219	                    else:
   220	                        self._syntax_error("Don't understand tag", words[0])
   221	            else:
   222	                # Literal content.  If it isn't empty, output it.
   223	                if in_joined:

   225	                elif squash:
   226	                    token = token.lstrip()

   228	                    buffered.append(repr(token))

   231	            self._syntax_error("Unmatched action tag", ops_stack[-1])

   236	            vars_code.add_line(f"c_{var_name} = context[{var_name!r}]")

   238	        code.add_line("return ''.join(result)")
   239	        code.dedent()
   240	        self._render_function = cast(
   241	            Callable[
   242	                [dict[str, Any], Callable[..., Any]],
   243	                str,
   244	            ],
   245	            code.get_globals()["render_function"],
   246	        )

   252	            code = self._expr_code(pipes[0])
   253	            for func in pipes[1:]:
   254	                self._variable(func, self.all_vars)
