   452	    def assertDoctypeHandled(self, doctype_fragment: str) -> None:
   453	        """Assert that a given doctype string is handled correctly."""
   454	        doctype_str, soup = self._document_with_doctype(doctype_fragment)
   455
   456	        # Make sure a Doctype object was created.
   457	        doctype = soup.contents[0]
   458	        assert doctype.__class__ == Doctype
   459	        assert doctype == doctype_fragment
   460	        assert soup.encode("utf8")[: len(doctype_str)] == doctype_str
   461
   462	        # Make sure that the doctype was correctly associated with the
   463	        # parse tree and that the rest of the document parsed.
   464	        assert soup.p is not None
   465	        assert soup.p.contents[0] == "foo"

   467	    def _document_with_doctype(
   468	        self, doctype_fragment: str, doctype_string: str = "DOCTYPE"
   469	    ) -> Tuple[bytes, BeautifulSoup]:
   470	        """Generate and parse a document with the given doctype."""
   471	        doctype = "<!%s %s>" % (doctype_string, doctype_fragment)

   473	        soup = self.soup(markup)
   474	        return doctype.encode("utf8"), soup
