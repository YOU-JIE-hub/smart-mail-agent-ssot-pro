   161	def analyze_member_access(
   162	    name: str,
   163	    typ: Type,
   164	    context: Context,
   165	    *,
   166	    is_lvalue: bool,
   167	    is_super: bool,
   168	    is_operator: bool,
   169	    original_type: Type,
   170	    chk: TypeCheckerSharedApi,
   171	    override_info: TypeInfo | None = None,
   172	    in_literal_context: bool = False,
   173	    self_type: Type | None = None,
   174	    module_symbol_table: SymbolTable | None = None,
   175	    no_deferral: bool = False,
   176	    is_self: bool = False,
   177	    rvalue: Expression | None = None,
   178	    suppress_errors: bool = False,
   179	) -> Type:
   180	    """Return the type of attribute 'name' of 'typ'.
   181
   182	    The actual implementation is in '_analyze_member_access' and this docstring
   183	    also applies to it.
   184
   185	    This is a general operation that supports various different variations:
   186
   187	      1. lvalue or non-lvalue access (setter or getter access)
   188	      2. supertype access when using super() (is_super == True and
   189	         'override_info' should refer to the supertype)
   190
   191	    'original_type' is the most precise inferred or declared type of the base object
   192	    that we have available. When looking for an attribute of 'typ', we may perform
   193	    recursive calls targeting the fallback type, and 'typ' may become some supertype
   194	    of 'original_type'. 'original_type' is always preserved as the 'typ' type used in
   195	    the initial, non-recursive call. The 'self_type' is a component of 'original_type'
   196	    to which generic self should be bound (a narrower type that has a fallback to instance).
   197	    Currently, this is used only for union types.
   198
   199	    'module_symbol_table' is passed to this function if 'typ' is actually a module,
   200	    and we want to keep track of the available attributes of the module (since they
   201	    are not available via the type object directly)
   202
   203	    'rvalue' can be provided optionally to infer better setter type when is_lvalue is True,
   204	    most notably this helps for descriptors with overloaded __set__() method.
   205
   206	    'suppress_errors' will skip any logic that is only needed to generate error messages.
   207	    Note that this more of a performance optimization, one should not rely on this to not
   208	    show any messages, as some may be show e.g. by callbacks called here,
   209	    use msg.filter_errors(), if needed.
   210	    """
   211	    mx = MemberContext(
   212	        is_lvalue=is_lvalue,
   213	        is_super=is_super,
   214	        is_operator=is_operator,
   215	        original_type=original_type,
   216	        context=context,
   217	        chk=chk,
   218	        self_type=self_type,
   219	        module_symbol_table=module_symbol_table,
   220	        no_deferral=no_deferral,
   221	        is_self=is_self,
   222	        rvalue=rvalue,
   223	        suppress_errors=suppress_errors,
   224	    )
   225	    result = _analyze_member_access(name, typ, mx, override_info)
   226	    possible_literal = get_proper_type(result)
   227	    if (
   228	        in_literal_context
   229	        and isinstance(possible_literal, Instance)
   230	        and possible_literal.last_known_value is not None
   231	    ):
   232	        return possible_literal.last_known_value
   233	    else:
   234	        return result
   235

   237	def _analyze_member_access(
   238	    name: str, typ: Type, mx: MemberContext, override_info: TypeInfo | None = None
   239	) -> Type:
   240	    typ = get_proper_type(typ)
   241	    if isinstance(typ, Instance):
   242	        return analyze_instance_member_access(name, typ, mx, override_info)
   243	    elif isinstance(typ, AnyType):
   244	        # The base object has dynamic type.
   245	        return AnyType(TypeOfAny.from_another_any, source_any=typ)
   246	    elif isinstance(typ, UnionType):
   247	        return analyze_union_member_access(name, typ, mx)
   248	    elif isinstance(typ, FunctionLike) and typ.is_type_obj():
   249	        return analyze_type_callable_member_access(name, typ, mx)
   250	    elif isinstance(typ, TypeType):
   251	        return analyze_type_type_member_access(name, typ, mx, override_info)
   252	    elif isinstance(typ, TupleType):
   253	        # Actually look up from the fallback instance type.
   254	        return _analyze_member_access(name, tuple_fallback(typ), mx, override_info)
   255	    elif isinstance(typ, (LiteralType, FunctionLike)):
   256	        # Actually look up from the fallback instance type.
   257	        return _analyze_member_access(name, typ.fallback, mx, override_info)
   258	    elif isinstance(typ, TypedDictType):
   259	        return analyze_typeddict_access(name, typ, mx, override_info)
   260	    elif isinstance(typ, NoneType):
   261	        return analyze_none_member_access(name, typ, mx)
   262	    elif isinstance(typ, TypeVarLikeType):
   263	        if isinstance(typ, TypeVarType) and typ.values:
   264	            return _analyze_member_access(
   265	                name, make_simplified_union(typ.values), mx, override_info
   266	            )
   267	        return _analyze_member_access(name, typ.upper_bound, mx, override_info)
   268	    elif isinstance(typ, DeletedType):
   269	        if not mx.suppress_errors:
   270	            mx.msg.deleted_as_rvalue(typ, mx.context)
   271	        return AnyType(TypeOfAny.from_error)
   272	    return report_missing_attribute(mx.original_type, typ, name, mx)

   275	def may_be_awaitable_attribute(
   276	    name: str, typ: Type, mx: MemberContext, override_info: TypeInfo | None = None
   277	) -> bool:
   278	    """Check if the given type has the attribute when awaited."""
   279	    if mx.chk.checking_missing_await:
   280	        # Avoid infinite recursion.
   281	        return False
   282	    with mx.chk.checking_await_set(), mx.msg.filter_errors() as local_errors:
   283	        aw_type = mx.chk.get_precise_awaitable_type(typ, local_errors)
   284	        if aw_type is None:
   285	            return False
   286	        _ = _analyze_member_access(
   287	            name, aw_type, mx.copy_modified(self_type=aw_type), override_info
   288	        )
   289	        return not local_errors.has_new_errors()

   292	def report_missing_attribute(
   293	    original_type: Type,
   294	    typ: Type,
   295	    name: str,
   296	    mx: MemberContext,
   297	    override_info: TypeInfo | None = None,
   298	) -> Type:
   299	    if mx.suppress_errors:
   300	        return AnyType(TypeOfAny.from_error)
   301	    error_code = mx.msg.has_no_attr(original_type, typ, name, mx.context, mx.module_symbol_table)
   302	    if not mx.msg.prefer_simple_messages():
   303	        if may_be_awaitable_attribute(name, typ, mx, override_info):
   304	            mx.msg.possible_missing_await(mx.context, error_code)
   305	    return AnyType(TypeOfAny.from_error)

   312	def analyze_instance_member_access(
   313	    name: str, typ: Instance, mx: MemberContext, override_info: TypeInfo | None
   314	) -> Type:
   315	    info = typ.type
   316	    if override_info:
   317	        info = override_info
   318
   319	    method = info.get_method(name)
   320
   321	    if name == "__init__" and not mx.is_super and not info.is_final:
   322	        if not method or not method.is_final:
   323	            # Accessing __init__ in statically typed code would compromise
   324	            # type safety unless used via super() or the method/class is final.
   325	            mx.fail(message_registry.CANNOT_ACCESS_INIT)
   326	            return AnyType(TypeOfAny.from_error)
   327
   328	    # The base object has an instance type.
   329
   330	    if (
   331	        state.find_occurrences
   332	        and info.name == state.find_occurrences[0]
   333	        and name == state.find_occurrences[1]
   334	        and not mx.suppress_errors
   335	    ):
   336	        mx.msg.note("Occurrence of '{}.{}'".format(*state.find_occurrences), mx.context)
   337
   338	    # Look up the member. First look up the method dictionary.
   339	    if method and not isinstance(method, Decorator):
   340	        if mx.is_super and not mx.suppress_errors:
   341	            validate_super_call(method, mx)
   342
   343	        if method.is_property:
   344	            assert isinstance(method, OverloadedFuncDef)
   345	            getter = method.items[0]
   346	            assert isinstance(getter, Decorator)
   347	            if mx.is_lvalue and getter.var.is_settable_property:
   348	                mx.chk.warn_deprecated(method.setter, mx.context)
   349	            return analyze_var(name, getter.var, typ, mx)
   350
   351	        if mx.is_lvalue and not mx.suppress_errors:
   352	            mx.msg.cant_assign_to_method(mx.context)
   353	        if not isinstance(method, OverloadedFuncDef):
   354	            signature = function_type(method, mx.named_type("builtins.function"))
   355	        else:
   356	            if method.type is None:
   357	                # Overloads may be not ready if they are decorated. Handle this in same
   358	                # manner as we would handle a regular decorated function: defer if possible.
   359	                if not mx.no_deferral and method.items:
   360	                    mx.not_ready_callback(method.name, mx.context)
   361	                return AnyType(TypeOfAny.special_form)
   362	            assert isinstance(method.type, Overloaded)
   363	            signature = method.type
   364	        if not mx.preserve_type_var_ids:
   365	            signature = freshen_all_functions_type_vars(signature)
   366	        if not method.is_static:
   367	            if isinstance(method, (FuncDef, OverloadedFuncDef)) and method.is_trivial_self:
   368	                signature = bind_self_fast(signature, mx.self_type)
   369	            else:
   370	                signature = check_self_arg(
   371	                    signature, mx.self_type, method.is_class, mx.context, name, mx.msg
   372	                )
   373	                signature = bind_self(signature, mx.self_type, is_classmethod=method.is_class)
   374	        # TODO: should we skip these steps for static methods as well?
   375	        # Since generic static methods should not be allowed.
   376	        typ = map_instance_to_supertype(typ, method.info)
   377	        member_type = expand_type_by_instance(signature, typ)
   378	        freeze_all_type_vars(member_type)
   379	        return member_type
   380	    else:
   381	        # Not a method.
   382	        return analyze_member_var_access(name, typ, info, mx)

   385	def validate_super_call(node: FuncBase, mx: MemberContext) -> None:
   386	    unsafe_super = False
   387	    if isinstance(node, FuncDef) and node.is_trivial_body:
   388	        unsafe_super = True
   389	    elif isinstance(node, OverloadedFuncDef):
   390	        if node.impl:
   391	            impl = node.impl if isinstance(node.impl, FuncDef) else node.impl.func
   392	            unsafe_super = impl.is_trivial_body
   393	        elif not node.is_property and node.items:
   394	            assert isinstance(node.items[0], Decorator)
   395	            unsafe_super = node.items[0].func.is_trivial_body
   396	    if unsafe_super:
   397	        mx.msg.unsafe_super(node.name, node.info.name, mx.context)

   400	def analyze_type_callable_member_access(name: str, typ: FunctionLike, mx: MemberContext) -> Type:
   401	    # Class attribute.
   402	    # TODO super?
   403	    ret_type = typ.items[0].ret_type
   404	    assert isinstance(ret_type, ProperType)
   405	    if isinstance(ret_type, TupleType):
   406	        ret_type = tuple_fallback(ret_type)
   407	    if isinstance(ret_type, TypedDictType):
   408	        ret_type = ret_type.fallback
   409	    if isinstance(ret_type, Instance):
   410	        if not mx.is_operator:
   411	            # When Python sees an operator (eg `3 == 4`), it automatically translates that
   412	            # into something like `int.__eq__(3, 4)` instead of `(3).__eq__(4)` as an
   413	            # optimization.
   414	            #
   415	            # While it normally it doesn't matter which of the two versions are used, it
   416	            # does cause inconsistencies when working with classes. For example, translating
   417	            # `int == int` to `int.__eq__(int)` would not work since `int.__eq__` is meant to
   418	            # compare two int _instances_. What we really want is `type(int).__eq__`, which
   419	            # is meant to compare two types or classes.
   420	            #
   421	            # This check makes sure that when we encounter an operator, we skip looking up
   422	            # the corresponding method in the current instance to avoid this edge case.
   423	            # See https://github.com/python/mypy/pull/1787 for more info.
   424	            # TODO: do not rely on same type variables being present in all constructor overloads.
   425	            result = analyze_class_attribute_access(
   426	                ret_type, name, mx, original_vars=typ.items[0].variables, mcs_fallback=typ.fallback
   427	            )
   428	            if result:
   429	                return result
   430	        # Look up from the 'type' type.
   431	        return _analyze_member_access(name, typ.fallback, mx)
   432	    else:
   433	        assert False, f"Unexpected type {ret_type!r}"

   436	def analyze_type_type_member_access(
   437	    name: str, typ: TypeType, mx: MemberContext, override_info: TypeInfo | None
   438	) -> Type:
   439	    # Similar to analyze_type_callable_attribute_access.
   440	    item = None
   441	    fallback = mx.named_type("builtins.type")
   442	    if isinstance(typ.item, Instance):
   443	        item = typ.item
   444	    elif isinstance(typ.item, AnyType):
   445	        with mx.msg.filter_errors():
   446	            return _analyze_member_access(name, fallback, mx, override_info)
   447	    elif isinstance(typ.item, TypeVarType):
   448	        upper_bound = get_proper_type(typ.item.upper_bound)
   449	        if isinstance(upper_bound, Instance):
   450	            item = upper_bound
   451	        elif isinstance(upper_bound, UnionType):
   452	            return _analyze_member_access(
   453	                name,
   454	                TypeType.make_normalized(upper_bound, line=typ.line, column=typ.column),
   455	                mx,
   456	                override_info,
   457	            )
   458	        elif isinstance(upper_bound, TupleType):
   459	            item = tuple_fallback(upper_bound)
   460	        elif isinstance(upper_bound, AnyType):
   461	            with mx.msg.filter_errors():
   462	                return _analyze_member_access(name, fallback, mx, override_info)
   463	    elif isinstance(typ.item, TupleType):
   464	        item = tuple_fallback(typ.item)
   465	    elif isinstance(typ.item, FunctionLike) and typ.item.is_type_obj():
   466	        item = typ.item.fallback
   467	    elif isinstance(typ.item, TypeType):
   468	        # Access member on metaclass object via Type[Type[C]]
   469	        if isinstance(typ.item.item, Instance):
   470	            item = typ.item.item.type.metaclass_type
   471	    ignore_messages = False
   472
   473	    if item is not None:
   474	        fallback = item.type.metaclass_type or fallback
   475
   476	    if item and not mx.is_operator:
   477	        # See comment above for why operators are skipped
   478	        result = analyze_class_attribute_access(
   479	            item, name, mx, mcs_fallback=fallback, override_info=override_info
   480	        )
   481	        if result:
   482	            if not (isinstance(get_proper_type(result), AnyType) and item.type.fallback_to_any):
   483	                return result
   484	            else:
   485	                # We don't want errors on metaclass lookup for classes with Any fallback
   486	                ignore_messages = True
   487
   488	    with mx.msg.filter_errors(filter_errors=ignore_messages):
   489	        return _analyze_member_access(name, fallback, mx, override_info)
   490

   492	def analyze_union_member_access(name: str, typ: UnionType, mx: MemberContext) -> Type:
   493	    with mx.msg.disable_type_names():
   494	        results = []
   495	        for subtype in typ.relevant_items():
   496	            # Self types should be bound to every individual item of a union.
   497	            item_mx = mx.copy_modified(self_type=subtype)
   498	            results.append(_analyze_member_access(name, subtype, item_mx))
   499	    return make_simplified_union(results)

   516	def analyze_member_var_access(
   517	    name: str, itype: Instance, info: TypeInfo, mx: MemberContext
   518	) -> Type:
   519	    """Analyse attribute access that does not target a method.
   520
   521	    This is logically part of analyze_member_access and the arguments are similar.
   522
   523	    original_type is the type of E in the expression E.var
   524	    """
   525	    # It was not a method. Try looking up a variable.
   526	    node = info.get(name)
   527	    v = node.node if node else None
   528
   529	    mx.chk.warn_deprecated(v, mx.context)
   530
   531	    vv = v
   532	    is_trivial_self = False
   533	    if isinstance(vv, Decorator):
   534	        # The associated Var node of a decorator contains the type.
   535	        v = vv.var
   536	        is_trivial_self = vv.func.is_trivial_self and not vv.decorators
   537	        if mx.is_super and not mx.suppress_errors:
   538	            validate_super_call(vv.func, mx)
   539	    if isinstance(v, FuncDef):
   540	        assert False, "Did not expect a function"
   541	    if isinstance(v, MypyFile):
   542	        mx.chk.module_refs.add(v.fullname)
   543
   544	    if isinstance(vv, (TypeInfo, TypeAlias, MypyFile, TypeVarLikeExpr)):
   545	        # If the associated variable is a TypeInfo synthesize a Var node for
   546	        # the purposes of type checking.  This enables us to type check things
   547	        # like accessing class attributes on an inner class. Similar we allow
   548	        # using qualified type aliases in runtime context. For example:
   549	        #     class C:
   550	        #         A = List[int]
   551	        #     x = C.A() <- this is OK
   552	        typ = mx.chk.expr_checker.analyze_static_reference(vv, mx.context, mx.is_lvalue)
   553	        v = Var(name, type=typ)
   554	        v.info = info
   555
   556	    if isinstance(v, Var):
   557	        implicit = info[name].implicit
   558
   559	        # An assignment to final attribute is always an error,
   560	        # independently of types.
   561	        if mx.is_lvalue and not mx.chk.get_final_context():
   562	            check_final_member(name, info, mx.msg, mx.context)
   563
   564	        return analyze_var(name, v, itype, mx, implicit=implicit, is_trivial_self=is_trivial_self)
   565	    elif (
   566	        not v
   567	        and name not in ["__getattr__", "__setattr__", "__getattribute__"]
   568	        and not mx.is_operator
   569	        and mx.module_symbol_table is None
   570	    ):
   571	        # Above we skip ModuleType.__getattr__ etc. if we have a
   572	        # module symbol table, since the symbol table allows precise
   573	        # checking.
   574	        if not mx.is_lvalue:
   575	            for method_name in ("__getattribute__", "__getattr__"):
   576	                method = info.get_method(method_name)
   577
   578	                # __getattribute__ is defined on builtins.object and returns Any, so without
   579	                # the guard this search will always find object.__getattribute__ and conclude
   580	                # that the attribute exists
   581	                if method and method.info.fullname != "builtins.object":
   582	                    bound_method = analyze_decorator_or_funcbase_access(
   583	                        defn=method, itype=itype, name=method_name, mx=mx
   584	                    )
   585	                    typ = map_instance_to_supertype(itype, method.info)
   586	                    getattr_type = get_proper_type(expand_type_by_instance(bound_method, typ))
   587	                    if isinstance(getattr_type, CallableType):
   588	                        result = getattr_type.ret_type
   589	                    else:
   590	                        result = getattr_type
   591
   592	                    # Call the attribute hook before returning.
   593	                    fullname = f"{method.info.fullname}.{name}"
   594	                    hook = mx.chk.plugin.get_attribute_hook(fullname)
   595	                    if hook:
   596	                        result = hook(
   597	                            AttributeContext(
   598	                                get_proper_type(mx.original_type),
   599	                                result,
   600	                                mx.is_lvalue,
   601	                                mx.context,
   602	                                mx.chk,
   603	                            )
   604	                        )
   605	                    return result
   606	        else:
   607	            setattr_meth = info.get_method("__setattr__")
   608	            if setattr_meth and setattr_meth.info.fullname != "builtins.object":
   609	                bound_type = analyze_decorator_or_funcbase_access(
   610	                    defn=setattr_meth,
   611	                    itype=itype,
   612	                    name="__setattr__",
   613	                    mx=mx.copy_modified(is_lvalue=False),
   614	                )
   615	                typ = map_instance_to_supertype(itype, setattr_meth.info)
   616	                setattr_type = get_proper_type(expand_type_by_instance(bound_type, typ))
   617	                if isinstance(setattr_type, CallableType) and len(setattr_type.arg_types) > 0:
   618	                    return setattr_type.arg_types[-1]
   619
   620	    if itype.type.fallback_to_any:
   621	        return AnyType(TypeOfAny.special_form)
   622
   623	    # Could not find the member.
   624	    if itype.extra_attrs and name in itype.extra_attrs.attrs:
   625	        # For modules use direct symbol table lookup.
   626	        if not itype.extra_attrs.mod_name:
   627	            return itype.extra_attrs.attrs[name]
   628
   629	    if mx.is_super and not mx.suppress_errors:
   630	        mx.msg.undefined_in_superclass(name, mx.context)
   631	        return AnyType(TypeOfAny.from_error)
   632	    else:
   633	        ret = report_missing_attribute(mx.original_type, itype, name, mx)
   634	        # Avoid paying double jeopardy if we can't find the member due to --no-implicit-reexport
   635	        if (
   636	            mx.module_symbol_table is not None
   637	            and name in mx.module_symbol_table
   638	            and not mx.module_symbol_table[name].module_public
   639	        ):
   640	            v = mx.module_symbol_table[name].node
   641	            e = NameExpr(name)
   642	            e.set_line(mx.context)
   643	            e.node = v
   644	            return mx.chk.expr_checker.analyze_ref_expr(e, lvalue=mx.is_lvalue)
   645	        return ret

   648	def check_final_member(name: str, info: TypeInfo, msg: MessageBuilder, ctx: Context) -> None:
   649	    """Give an error if the name being assigned was declared as final."""
   650	    for base in info.mro:
   651	        sym = base.names.get(name)
   652	        if sym and is_final_node(sym.node):
   653	            msg.cant_assign_to_final(name, attr_assign=True, ctx=ctx)

   656	def analyze_descriptor_access(descriptor_type: Type, mx: MemberContext) -> Type:
   657	    """Type check descriptor access.
   658
   659	    Arguments:
   660	        descriptor_type: The type of the descriptor attribute being accessed
   661	            (the type of ``f`` in ``a.f`` when ``f`` is a descriptor).
   662	        mx: The current member access context.
   663	    Return:
   664	        The return type of the appropriate ``__get__/__set__`` overload for the descriptor.
   665	    """
   666	    instance_type = get_proper_type(mx.self_type)
   667	    orig_descriptor_type = descriptor_type
   668	    descriptor_type = get_proper_type(descriptor_type)
   669
   670	    if isinstance(descriptor_type, UnionType):
   671	        # Map the access over union types
   672	        return make_simplified_union(
   673	            [analyze_descriptor_access(typ, mx) for typ in descriptor_type.items]
   674	        )
   675	    elif not isinstance(descriptor_type, Instance):
   676	        return orig_descriptor_type
   677
   678	    if not mx.is_lvalue and not descriptor_type.type.has_readable_member("__get__"):
   679	        return orig_descriptor_type
   680
   681	    # We do this check first to accommodate for descriptors with only __set__ method.
   682	    # If there is no __set__, we type-check that the assigned value matches
   683	    # the return type of __get__. This doesn't match the python semantics,
   684	    # (which allow you to override the descriptor with any value), but preserves
   685	    # the type of accessing the attribute (even after the override).
   686	    if mx.is_lvalue and descriptor_type.type.has_readable_member("__set__"):
   687	        return analyze_descriptor_assign(descriptor_type, mx)
   688
   689	    if mx.is_lvalue and not descriptor_type.type.has_readable_member("__get__"):
   690	        # This turned out to be not a descriptor after all.
   691	        return orig_descriptor_type
   692
   693	    dunder_get = descriptor_type.type.get_method("__get__")
   694	    if dunder_get is None:
   695	        mx.fail(
   696	            message_registry.DESCRIPTOR_GET_NOT_CALLABLE.format(
   697	                descriptor_type.str_with_options(mx.msg.options)
   698	            )
   699	        )
   700	        return AnyType(TypeOfAny.from_error)
   701
   702	    bound_method = analyze_decorator_or_funcbase_access(
   703	        defn=dunder_get,
   704	        itype=descriptor_type,
   705	        name="__get__",
   706	        mx=mx.copy_modified(self_type=descriptor_type),
   707	    )
   708
   709	    typ = map_instance_to_supertype(descriptor_type, dunder_get.info)
   710	    dunder_get_type = expand_type_by_instance(bound_method, typ)
   711
   712	    if isinstance(instance_type, FunctionLike) and instance_type.is_type_obj():
   713	        owner_type = instance_type.items[0].ret_type
   714	        instance_type = NoneType()
   715	    elif isinstance(instance_type, TypeType):
   716	        owner_type = instance_type.item
   717	        instance_type = NoneType()
   718	    else:
   719	        owner_type = instance_type
   720
   721	    callable_name = mx.chk.expr_checker.method_fullname(descriptor_type, "__get__")
   722	    dunder_get_type = mx.chk.expr_checker.transform_callee_type(
   723	        callable_name,
   724	        dunder_get_type,
   725	        [
   726	            TempNode(instance_type, context=mx.context),
   727	            TempNode(TypeType.make_normalized(owner_type), context=mx.context),
   728	        ],
   729	        [ARG_POS, ARG_POS],
   730	        mx.context,
   731	        object_type=descriptor_type,
   732	    )
   733
   734	    _, inferred_dunder_get_type = mx.chk.expr_checker.check_call(
   735	        dunder_get_type,
   736	        [
   737	            TempNode(instance_type, context=mx.context),
   738	            TempNode(TypeType.make_normalized(owner_type), context=mx.context),
   739	        ],
   740	        [ARG_POS, ARG_POS],
   741	        mx.context,
   742	        object_type=descriptor_type,
   743	        callable_name=callable_name,
   744	    )
   745
   746	    mx.chk.check_deprecated(dunder_get, mx.context)
   747	    mx.chk.warn_deprecated_overload_item(
   748	        dunder_get, mx.context, target=inferred_dunder_get_type, selftype=descriptor_type
   749	    )
   750
   751	    inferred_dunder_get_type = get_proper_type(inferred_dunder_get_type)
   752	    if isinstance(inferred_dunder_get_type, AnyType):
   753	        # check_call failed, and will have reported an error
   754	        return inferred_dunder_get_type
   755
   756	    if not isinstance(inferred_dunder_get_type, CallableType):
   757	        mx.fail(
   758	            message_registry.DESCRIPTOR_GET_NOT_CALLABLE.format(
   759	                descriptor_type.str_with_options(mx.msg.options)
   760	            )
   761	        )
   762	        return AnyType(TypeOfAny.from_error)
   763
   764	    return inferred_dunder_get_type.ret_type

   767	def analyze_descriptor_assign(descriptor_type: Instance, mx: MemberContext) -> Type:
   768	    instance_type = get_proper_type(mx.self_type)
   769	    dunder_set = descriptor_type.type.get_method("__set__")
   770	    if dunder_set is None:
   771	        mx.fail(
   772	            message_registry.DESCRIPTOR_SET_NOT_CALLABLE.format(
   773	                descriptor_type.str_with_options(mx.msg.options)
   774	            ).value
   775	        )
   776	        return AnyType(TypeOfAny.from_error)
   777
   778	    bound_method = analyze_decorator_or_funcbase_access(
   779	        defn=dunder_set,
   780	        itype=descriptor_type,
   781	        name="__set__",
   782	        mx=mx.copy_modified(is_lvalue=False, self_type=descriptor_type),
   783	    )
   784	    typ = map_instance_to_supertype(descriptor_type, dunder_set.info)
   785	    dunder_set_type = expand_type_by_instance(bound_method, typ)
   786
   787	    callable_name = mx.chk.expr_checker.method_fullname(descriptor_type, "__set__")
   788	    rvalue = mx.rvalue or TempNode(AnyType(TypeOfAny.special_form), context=mx.context)
   789	    dunder_set_type = mx.chk.expr_checker.transform_callee_type(
   790	        callable_name,
   791	        dunder_set_type,
   792	        [TempNode(instance_type, context=mx.context), rvalue],
   793	        [ARG_POS, ARG_POS],
   794	        mx.context,
   795	        object_type=descriptor_type,
   796	    )
   797
   798	    # For non-overloaded setters, the result should be type-checked like a regular assignment.
   799	    # Hence, we first only try to infer the type by using the rvalue as type context.
   800	    type_context = rvalue
   801	    with mx.msg.filter_errors():
   802	        _, inferred_dunder_set_type = mx.chk.expr_checker.check_call(
   803	            dunder_set_type,
   804	            [TempNode(instance_type, context=mx.context), type_context],
   805	            [ARG_POS, ARG_POS],
   806	            mx.context,
   807	            object_type=descriptor_type,
   808	            callable_name=callable_name,
   809	        )
   810
   811	    # And now we in fact type check the call, to show errors related to wrong arguments
   812	    # count, etc., replacing the type context for non-overloaded setters only.
   813	    inferred_dunder_set_type = get_proper_type(inferred_dunder_set_type)
   814	    if isinstance(inferred_dunder_set_type, CallableType):
   815	        type_context = TempNode(AnyType(TypeOfAny.special_form), context=mx.context)
   816	    mx.chk.expr_checker.check_call(
   817	        dunder_set_type,
   818	        [TempNode(instance_type, context=mx.context), type_context],
   819	        [ARG_POS, ARG_POS],
   820	        mx.context,
   821	        object_type=descriptor_type,
   822	        callable_name=callable_name,
   823	    )
   824
   825	    # Search for possible deprecations:
   826	    mx.chk.check_deprecated(dunder_set, mx.context)
   827	    mx.chk.warn_deprecated_overload_item(
   828	        dunder_set, mx.context, target=inferred_dunder_set_type, selftype=descriptor_type
   829	    )
   830
   831	    # In the following cases, a message already will have been recorded in check_call.
   832	    if (not isinstance(inferred_dunder_set_type, CallableType)) or (
   833	        len(inferred_dunder_set_type.arg_types) < 2
   834	    ):
   835	        return AnyType(TypeOfAny.from_error)
   836	    return inferred_dunder_set_type.arg_types[1]

   839	def is_instance_var(var: Var) -> bool:
   840	    """Return if var is an instance variable according to PEP 526."""
   841	    return (
   842	        # check the type_info node is the var (not a decorated function, etc.)
   843	        var.name in var.info.names
   844	        and var.info.names[var.name].node is var
   845	        and not var.is_classvar
   846	        # variables without annotations are treated as classvar
   847	        and not var.is_inferred
   848	    )

   851	def analyze_var(
   852	    name: str,
   853	    var: Var,
   854	    itype: Instance,
   855	    mx: MemberContext,
   856	    *,
   857	    implicit: bool = False,
   858	    is_trivial_self: bool = False,
   859	) -> Type:
   860	    """Analyze access to an attribute via a Var node.
   861
   862	    This is conceptually part of analyze_member_access and the arguments are similar.
   863	    itype is the instance type in which attribute should be looked up
   864	    original_type is the type of E in the expression E.var
   865	    if implicit is True, the original Var was created as an assignment to self
   866	    if is_trivial_self is True, we can use fast path for bind_self().
   867	    """
   868	    # Found a member variable.
   869	    original_itype = itype
   870	    itype = map_instance_to_supertype(itype, var.info)
   871	    if var.is_settable_property and mx.is_lvalue:
   872	        typ: Type | None = var.setter_type
   873	        if typ is None and var.is_ready:
   874	            # Existing synthetic properties may not set setter type. Fall back to getter.
   875	            typ = var.type
   876	    else:
   877	        typ = var.type
   878	    if typ:
   879	        if isinstance(typ, PartialType):
   880	            return mx.chk.handle_partial_var_type(typ, mx.is_lvalue, var, mx.context)
   881	        if mx.is_lvalue and not mx.suppress_errors:
   882	            if var.is_property and not var.is_settable_property:
   883	                mx.msg.read_only_property(name, itype.type, mx.context)
   884	            if var.is_classvar:
   885	                mx.msg.cant_assign_to_classvar(name, mx.context)
   886	        # This is the most common case for variables, so start with this.
   887	        result = expand_without_binding(typ, var, itype, original_itype, mx)
   888
   889	        # A non-None value indicates that we should actually bind self for this variable.
   890	        call_type: ProperType | None = None
   891	        if var.is_initialized_in_class and (not is_instance_var(var) or mx.is_operator):
   892	            typ = get_proper_type(typ)
   893	            if isinstance(typ, FunctionLike) and not typ.is_type_obj():
   894	                call_type = typ
   895	            elif var.is_property:
   896	                deco_mx = mx.copy_modified(original_type=typ, self_type=typ, is_lvalue=False)
   897	                call_type = get_proper_type(_analyze_member_access("__call__", typ, deco_mx))
   898	            else:
   899	                call_type = typ
   900
   901	        # Bound variables with callable types are treated like methods
   902	        # (these are usually method aliases like __rmul__ = __mul__).
   903	        if isinstance(call_type, FunctionLike) and not call_type.is_type_obj():
   904	            if mx.is_lvalue and not var.is_property and not mx.suppress_errors:
   905	                mx.msg.cant_assign_to_method(mx.context)
   906
   907	        # Bind the self type for each callable component (when needed).
   908	        if call_type and not var.is_staticmethod:
   909	            bound_items = []
   910	            for ct in call_type.items if isinstance(call_type, UnionType) else [call_type]:
   911	                p_ct = get_proper_type(ct)
   912	                if isinstance(p_ct, FunctionLike) and (not p_ct.bound() or var.is_property):
   913	                    item = expand_and_bind_callable(p_ct, var, itype, name, mx, is_trivial_self)
   914	                else:
   915	                    item = expand_without_binding(ct, var, itype, original_itype, mx)
   916	                bound_items.append(item)
   917	            result = UnionType.make_union(bound_items)
   918	    else:
   919	        if not var.is_ready and not mx.no_deferral:
   920	            mx.not_ready_callback(var.name, mx.context)
   921	        # Implicit 'Any' type.
   922	        result = AnyType(TypeOfAny.special_form)
   923	    fullname = f"{var.info.fullname}.{name}"
   924	    hook = mx.chk.plugin.get_attribute_hook(fullname)
   925	    if result and not (implicit or var.info.is_protocol and is_instance_var(var)):
   926	        result = analyze_descriptor_access(result, mx)
   927	    if hook:
   928	        result = hook(
   929	            AttributeContext(
   930	                get_proper_type(mx.original_type), result, mx.is_lvalue, mx.context, mx.chk
   931	            )
   932	        )
   933	    return result

   936	def expand_without_binding(
   937	    typ: Type, var: Var, itype: Instance, original_itype: Instance, mx: MemberContext
   938	) -> Type:
   939	    if not mx.preserve_type_var_ids:
   940	        typ = freshen_all_functions_type_vars(typ)
   941	    typ = expand_self_type_if_needed(typ, mx, var, original_itype)
   942	    expanded = expand_type_by_instance(typ, itype)
   943	    freeze_all_type_vars(expanded)
   944	    return expanded

   947	def expand_and_bind_callable(
   948	    functype: FunctionLike,
   949	    var: Var,
   950	    itype: Instance,
   951	    name: str,
   952	    mx: MemberContext,
   953	    is_trivial_self: bool,
   954	) -> Type:
   955	    if not mx.preserve_type_var_ids:
   956	        functype = freshen_all_functions_type_vars(functype)
   957	    typ = get_proper_type(expand_self_type(var, functype, mx.original_type))
   958	    assert isinstance(typ, FunctionLike)
   959	    if is_trivial_self:
   960	        typ = bind_self_fast(typ, mx.self_type)
   961	    else:
   962	        typ = check_self_arg(typ, mx.self_type, var.is_classmethod, mx.context, name, mx.msg)
   963	        typ = bind_self(typ, mx.self_type, var.is_classmethod)
   964	    expanded = expand_type_by_instance(typ, itype)
   965	    freeze_all_type_vars(expanded)
   966	    if not var.is_property:
   967	        return expanded
   968	    # TODO: a decorated property can result in Overloaded here.
   969	    assert isinstance(expanded, CallableType)
   970	    if var.is_settable_property and mx.is_lvalue and var.setter_type is not None:
   971	        # TODO: use check_call() to infer better type, same as for __set__().
   972	        if not expanded.arg_types:
   973	            # This can happen when accessing invalid property from its own body,
   974	            # error will be reported elsewhere.
   975	            return AnyType(TypeOfAny.from_error)
   976	        return expanded.arg_types[0]
   977	    else:
   978	        return expanded.ret_type

   981	def expand_self_type_if_needed(
   982	    t: Type, mx: MemberContext, var: Var, itype: Instance, is_class: bool = False
   983	) -> Type:
   984	    """Expand special Self type in a backwards compatible manner.
   985
   986	    This should ensure that mixing old-style and new-style self-types work
   987	    seamlessly. Also, re-bind new style self-types in subclasses if needed.
   988	    """
   989	    original = get_proper_type(mx.self_type)
   990	    if not (mx.is_self or mx.is_super):
   991	        repl = mx.self_type
   992	        if is_class:
   993	            if isinstance(original, TypeType):
   994	                repl = original.item
   995	            elif isinstance(original, CallableType):
   996	                # Problematic access errors should have been already reported.
   997	                repl = erase_typevars(original.ret_type)
   998	            else:
   999	                repl = itype
  1000	        return expand_self_type(var, t, repl)
  1001	    elif supported_self_type(
  1002	        # Support compatibility with plain old style T -> T and Type[T] -> T only.
  1003	        get_proper_type(mx.self_type),
  1004	        allow_instances=False,
  1005	        allow_callable=False,
  1006	    ):
  1007	        repl = mx.self_type
  1008	        if is_class and isinstance(original, TypeType):
  1009	            repl = original.item
  1010	        return expand_self_type(var, t, repl)
  1011	    elif (
  1012	        mx.is_self
  1013	        and itype.type != var.info
  1014	        # If an attribute with Self-type was defined in a supertype, we need to
  1015	        # rebind the Self type variable to Self type variable of current class...
  1016	        and itype.type.self_type is not None
  1017	        # ...unless `self` has an explicit non-trivial annotation.
  1018	        and itype == mx.chk.scope.active_self_type()
  1019	    ):
  1020	        return expand_self_type(var, t, itype.type.self_type)
  1021	    else:
  1022	        return t

  1025	def check_self_arg(
  1026	    functype: FunctionLike,
  1027	    dispatched_arg_type: Type,
  1028	    is_classmethod: bool,
  1029	    context: Context,
  1030	    name: str,
  1031	    msg: MessageBuilder,
  1032	) -> FunctionLike:
  1033	    """Check that an instance has a valid type for a method with annotated 'self'.
  1034
  1035	    For example if the method is defined as:
  1036	        class A:
  1037	            def f(self: S) -> T: ...
  1038	    then for 'x.f' we check that type(x) <: S. If the method is overloaded, we select
  1039	    only overloads items that satisfy this requirement. If there are no matching
  1040	    overloads, an error is generated.
  1041	    """
  1042	    items = functype.items
  1043	    if not items:
  1044	        return functype
  1045	    new_items = []
  1046	    if is_classmethod:
  1047	        dispatched_arg_type = TypeType.make_normalized(dispatched_arg_type)
  1048
  1049	    for item in items:
  1050	        if not item.arg_types or item.arg_kinds[0] not in (ARG_POS, ARG_STAR):
  1051	            # No positional first (self) argument (*args is okay).
  1052	            msg.no_formal_self(name, item, context)
  1053	            # This is pretty bad, so just return the original signature if
  1054	            # there is at least one such error.
  1055	            return functype
  1056	        else:
  1057	            selfarg = get_proper_type(item.arg_types[0])
  1058	            # This matches similar special-casing in bind_self(), see more details there.
  1059	            self_callable = name == "__call__" and isinstance(selfarg, CallableType)
  1060	            if self_callable or subtypes.is_subtype(
  1061	                dispatched_arg_type,
  1062	                # This level of erasure matches the one in checker.check_func_def(),
  1063	                # better keep these two checks consistent.
  1064	                erase_typevars(erase_to_bound(selfarg)),
  1065	                # This is to work around the fact that erased ParamSpec and TypeVarTuple
  1066	                # callables are not always compatible with non-erased ones both ways.
  1067	                always_covariant=any(
  1068	                    not isinstance(tv, TypeVarType) for tv in get_all_type_vars(selfarg)
  1069	                ),
  1070	                ignore_pos_arg_names=True,
  1071	            ):
  1072	                new_items.append(item)
  1073	            elif isinstance(selfarg, ParamSpecType):
  1074	                # TODO: This is not always right. What's the most reasonable thing to do here?
  1075	                new_items.append(item)
  1076	            elif isinstance(selfarg, TypeVarTupleType):
  1077	                raise NotImplementedError
  1078	    if not new_items:
  1079	        # Choose first item for the message (it may be not very helpful for overloads).
  1080	        msg.incompatible_self_argument(
  1081	            name, dispatched_arg_type, items[0], is_classmethod, context
  1082	        )
  1083	        return functype
  1084	    if len(new_items) == 1:
  1085	        return new_items[0]
  1086	    return Overloaded(new_items)

  1089	def analyze_class_attribute_access(
  1090	    itype: Instance,
  1091	    name: str,
  1092	    mx: MemberContext,
  1093	    *,
  1094	    mcs_fallback: Instance,
  1095	    override_info: TypeInfo | None = None,
  1096	    original_vars: Sequence[TypeVarLikeType] | None = None,
  1097	) -> Type | None:
  1098	    """Analyze access to an attribute on a class object.
  1099
  1100	    itype is the return type of the class object callable, original_type is the type
  1101	    of E in the expression E.var, original_vars are type variables of the class callable
  1102	    (for generic classes).
  1103	    """
  1104	    info = itype.type
  1105	    if override_info:
  1106	        info = override_info
  1107
  1108	    fullname = f"{info.fullname}.{name}"
  1109	    hook = mx.chk.plugin.get_class_attribute_hook(fullname)
  1110
  1111	    node = info.get(name)
  1112	    if not node:
  1113	        if itype.extra_attrs and name in itype.extra_attrs.attrs:
  1114	            # For modules use direct symbol table lookup.
  1115	            if not itype.extra_attrs.mod_name:
  1116	                return itype.extra_attrs.attrs[name]
  1117	        if info.fallback_to_any or info.meta_fallback_to_any:
  1118	            return apply_class_attr_hook(mx, hook, AnyType(TypeOfAny.special_form))
  1119	        return None
  1120
  1121	    if (
  1122	        isinstance(node.node, Var)
  1123	        and not node.node.is_classvar
  1124	        and not hook
  1125	        and mcs_fallback.type.get(name)
  1126	    ):
  1127	        # If the same attribute is declared on the metaclass and the class but with different types,
  1128	        # and the attribute on the class is not a ClassVar,
  1129	        # the type of the attribute on the metaclass should take priority
  1130	        # over the type of the attribute on the class,
  1131	        # when the attribute is being accessed from the class object itself.
  1132	        #
  1133	        # Return `None` here to signify that the name should be looked up
  1134	        # on the class object itself rather than the instance.
  1135	        return None
  1136
  1137	    mx.chk.warn_deprecated(node.node, mx.context)
  1138
  1139	    is_decorated = isinstance(node.node, Decorator)
  1140	    is_method = is_decorated or isinstance(node.node, FuncBase)
  1141	    if mx.is_lvalue and not mx.suppress_errors:
  1142	        if is_method:
  1143	            mx.msg.cant_assign_to_method(mx.context)
  1144	        if isinstance(node.node, TypeInfo):
  1145	            mx.fail(message_registry.CANNOT_ASSIGN_TO_TYPE)
  1146
  1147	    # Refuse class attribute access if slot defined
  1148	    if info.slots and name in info.slots:
  1149	        mx.fail(message_registry.CLASS_VAR_CONFLICTS_SLOTS.format(name))
  1150
  1151	    # If a final attribute was declared on `self` in `__init__`, then it
  1152	    # can't be accessed on the class object.
  1153	    if node.implicit and isinstance(node.node, Var) and node.node.is_final:
  1154	        mx.fail(message_registry.CANNOT_ACCESS_FINAL_INSTANCE_ATTR.format(node.node.name))
  1155
  1156	    # An assignment to final attribute on class object is also always an error,
  1157	    # independently of types.
  1158	    if mx.is_lvalue and not mx.chk.get_final_context():
  1159	        check_final_member(name, info, mx.msg, mx.context)
  1160
  1161	    if info.is_enum and not (mx.is_lvalue or is_decorated or is_method):
  1162	        enum_class_attribute_type = analyze_enum_class_attribute_access(itype, name, mx)
  1163	        if enum_class_attribute_type:
  1164	            return apply_class_attr_hook(mx, hook, enum_class_attribute_type)
  1165
  1166	    t = node.type
  1167	    if t:
  1168	        if isinstance(t, PartialType):
  1169	            symnode = node.node
  1170	            assert isinstance(symnode, Var)
  1171	            return apply_class_attr_hook(
  1172	                mx, hook, mx.chk.handle_partial_var_type(t, mx.is_lvalue, symnode, mx.context)
  1173	            )
  1174
  1175	        # Find the class where method/variable was defined.
  1176	        if isinstance(node.node, Decorator):
  1177	            super_info: TypeInfo | None = node.node.var.info
  1178	        elif isinstance(node.node, (Var, SYMBOL_FUNCBASE_TYPES)):
  1179	            super_info = node.node.info
  1180	        else:
  1181	            super_info = None
  1182
  1183	        # Map the type to how it would look as a defining class. For example:
  1184	        #     class C(Generic[T]): ...
  1185	        #     class D(C[Tuple[T, S]]): ...
  1186	        #     D[int, str].method()
  1187	        # Here itype is D[int, str], isuper is C[Tuple[int, str]].
  1188	        if not super_info:
  1189	            isuper = None
  1190	        else:
  1191	            isuper = map_instance_to_supertype(itype, super_info)
  1192
  1193	        if isinstance(node.node, Var):
  1194	            assert isuper is not None
  1195	            # Check if original variable type has type variables. For example:
  1196	            #     class C(Generic[T]):
  1197	            #         x: T
  1198	            #     C.x  # Error, ambiguous access
  1199	            #     C[int].x  # Also an error, since C[int] is same as C at runtime
  1200	            # Exception is Self type wrapped in ClassVar, that is safe.
  1201	            def_vars = set(node.node.info.defn.type_vars)
  1202	            if not node.node.is_classvar and node.node.info.self_type:
  1203	                def_vars.add(node.node.info.self_type)
  1204	            # TODO: should we include ParamSpec etc. here (i.e. use get_all_type_vars)?
  1205	            typ_vars = set(get_type_vars(t))
  1206	            if def_vars & typ_vars:
  1207	                # Exception: access on Type[...], including first argument of class methods is OK.
  1208	                if not isinstance(get_proper_type(mx.original_type), TypeType) or node.implicit:
  1209	                    if node.node.is_classvar:
  1210	                        message = message_registry.GENERIC_CLASS_VAR_ACCESS
  1211	                    else:
  1212	                        message = message_registry.GENERIC_INSTANCE_VAR_CLASS_ACCESS
  1213	                    mx.fail(message)
  1214	            t = expand_self_type_if_needed(t, mx, node.node, itype, is_class=True)
  1215	            # Erase non-mapped variables, but keep mapped ones, even if there is an error.
  1216	            # In the above example this means that we infer following types:
  1217	            #     C.x -> Any
  1218	            #     C[int].x -> int
  1219	            t = erase_typevars(expand_type_by_instance(t, isuper), {tv.id for tv in def_vars})
  1220
  1221	        is_classmethod = (is_decorated and cast(Decorator, node.node).func.is_class) or (
  1222	            isinstance(node.node, SYMBOL_FUNCBASE_TYPES) and node.node.is_class
  1223	        )
  1224	        t = get_proper_type(t)
  1225	        is_trivial_self = False
  1226	        if isinstance(node.node, Decorator):
  1227	            # Use fast path if there are trivial decorators like @classmethod or @property
  1228	            is_trivial_self = node.node.func.is_trivial_self and not node.node.decorators
  1229	        elif isinstance(node.node, (FuncDef, OverloadedFuncDef)):
  1230	            is_trivial_self = node.node.is_trivial_self
  1231	        if isinstance(t, FunctionLike) and is_classmethod and not is_trivial_self:
  1232	            t = check_self_arg(t, mx.self_type, False, mx.context, name, mx.msg)
  1233	        result = add_class_tvars(
  1234	            t,
  1235	            isuper,
  1236	            is_classmethod,
  1237	            mx,
  1238	            original_vars=original_vars,
  1239	            is_trivial_self=is_trivial_self,
  1240	        )
  1241	        # __set__ is not called on class objects.
  1242	        if not mx.is_lvalue:
  1243	            result = analyze_descriptor_access(result, mx)
  1244
  1245	        return apply_class_attr_hook(mx, hook, result)
  1246	    elif isinstance(node.node, Var):
  1247	        mx.not_ready_callback(name, mx.context)
  1248	        return AnyType(TypeOfAny.special_form)
  1249
  1250	    if isinstance(node.node, (TypeInfo, TypeAlias, MypyFile, TypeVarLikeExpr)):
  1251	        # TODO: should we apply class plugin here (similar to instance access)?
  1252	        return mx.chk.expr_checker.analyze_static_reference(node.node, mx.context, mx.is_lvalue)
  1253
  1254	    if is_decorated:
  1255	        assert isinstance(node.node, Decorator)
  1256	        if node.node.type:
  1257	            return apply_class_attr_hook(mx, hook, node.node.type)
  1258	        else:
  1259	            mx.not_ready_callback(name, mx.context)
  1260	            return AnyType(TypeOfAny.from_error)
  1261	    else:
  1262	        assert isinstance(node.node, SYMBOL_FUNCBASE_TYPES)
  1263	        typ = function_type(node.node, mx.named_type("builtins.function"))
  1264	        # Note: if we are accessing class method on class object, the cls argument is bound.
  1265	        # Annotated and/or explicit class methods go through other code paths above, for
  1266	        # unannotated implicit class methods we do this here.
  1267	        if node.node.is_class:
  1268	            typ = bind_self_fast(typ)
  1269	        return apply_class_attr_hook(mx, hook, typ)

  1348	def add_class_tvars(
  1349	    t: ProperType,
  1350	    isuper: Instance | None,
  1351	    is_classmethod: bool,
  1352	    mx: MemberContext,
  1353	    original_vars: Sequence[TypeVarLikeType] | None = None,
  1354	    is_trivial_self: bool = False,
  1355	) -> Type:
  1356	    """Instantiate type variables during analyze_class_attribute_access,
  1357	    e.g T and Q in the following:
  1358
  1359	    class A(Generic[T]):
  1360	        @classmethod
  1361	        def foo(cls: Type[Q]) -> Tuple[T, Q]: ...
  1362
  1363	    class B(A[str]): pass
  1364	    B.foo()
  1365
  1366	    Args:
  1367	        t: Declared type of the method (or property)
  1368	        isuper: Current instance mapped to the superclass where method was defined, this
  1369	            is usually done by map_instance_to_supertype()
  1370	        is_classmethod: True if this method is decorated with @classmethod
  1371	        original_vars: Type variables of the class callable on which the method was accessed
  1372	        is_trivial_self: if True, we can use fast path for bind_self().
  1373	    Returns:
  1374	        Expanded method type with added type variables (when needed).
  1375	    """
  1376	    # TODO: verify consistency between Q and T
  1377
  1378	    # We add class type variables if the class method is accessed on class object
  1379	    # without applied type arguments, this matches the behavior of __init__().
  1380	    # For example (continuing the example in docstring):
  1381	    #     A       # The type of callable is def [T] () -> A[T], _not_ def () -> A[Any]
  1382	    #     A[int]  # The type of callable is def () -> A[int]
  1383	    # and
  1384	    #     A.foo       # The type is generic def [T] () -> Tuple[T, A[T]]
  1385	    #     A[int].foo  # The type is non-generic def () -> Tuple[int, A[int]]
  1386	    #
  1387	    # This behaviour is useful for defining alternative constructors for generic classes.
  1388	    # To achieve such behaviour, we add the class type variables that are still free
  1389	    # (i.e. appear in the return type of the class object on which the method was accessed).
  1390	    if isinstance(t, CallableType):
  1391	        tvars = original_vars if original_vars is not None else []
  1392	        if not mx.preserve_type_var_ids:
  1393	            t = freshen_all_functions_type_vars(t)
  1394	        if is_classmethod:
  1395	            if is_trivial_self:
  1396	                t = bind_self_fast(t, mx.self_type)
  1397	            else:
  1398	                t = bind_self(t, mx.self_type, is_classmethod=True)
  1399	        if isuper is not None:
  1400	            t = expand_type_by_instance(t, isuper)
  1401	        freeze_all_type_vars(t)
  1402	        return t.copy_modified(variables=list(tvars) + list(t.variables))
  1403	    elif isinstance(t, Overloaded):
  1404	        return Overloaded(
  1405	            [
  1406	                cast(
  1407	                    CallableType,
  1408	                    add_class_tvars(item, isuper, is_classmethod, mx, original_vars=original_vars),
  1409	                )
  1410	                for item in t.items
  1411	            ]
  1412	        )
  1413	    if isuper is not None:
  1414	        t = expand_type_by_instance(t, isuper)
  1415	    return t

  1424	    See: https://github.com/python/mypy/issues/10409

  1444	def bind_self_fast(method: F, original_type: Type | None = None) -> F:
  1445	    """Return a copy of `method`, with the type of its first parameter (usually
  1446	    self or cls) bound to original_type.
  1447
  1448	    This is a faster version of mypy.typeops.bind_self() that can be used for methods
  1449	    with trivial self/cls annotations.
  1450	    """
  1451	    if isinstance(method, Overloaded):
  1452	        items = [bind_self_fast(c, original_type) for c in method.items]
  1453	        return cast(F, Overloaded(items))
  1454	    assert isinstance(method, CallableType)
  1455	    func: CallableType = method
  1456	    if not func.arg_types:
  1457	        # Invalid method, return something.
  1458	        return method
  1459	    if func.arg_kinds[0] in (ARG_STAR, ARG_STAR2):
  1460	        # See typeops.py for details.
  1461	        return method
  1462	    original_type = get_proper_type(original_type)
  1463	    if isinstance(original_type, CallableType) and original_type.is_type_obj():
  1464	        original_type = TypeType.make_normalized(original_type.ret_type)
  1465	    res = func.copy_modified(
  1466	        arg_types=func.arg_types[1:],
  1467	        arg_kinds=func.arg_kinds[1:],
  1468	        arg_names=func.arg_names[1:],
  1469	        is_bound=True,
  1470	    )
  1471	    return cast(F, res)

  1474	def has_operator(typ: Type, op_method: str, named_type: Callable[[str], Instance]) -> bool:
  1475	    """Does type have operator with the given name?
  1476
  1477	    Note: this follows the rules for operator access, in particular:
  1478	    * __getattr__ is not considered
  1479	    * for class objects we only look in metaclass
  1480	    * instance level attributes (i.e. extra_attrs) are not considered
  1481	    """
  1482	    # This is much faster than analyze_member_access, and so using
  1483	    # it first as a filter is important for performance. This is mostly relevant
  1484	    # in situations where we can't expect that method is likely present,
  1485	    # e.g. for __OP__ vs __rOP__.
  1486	    typ = get_proper_type(typ)
  1487
  1488	    if isinstance(typ, TypeVarLikeType):
  1489	        typ = typ.values_or_bound()
  1490	    if isinstance(typ, AnyType):
  1491	        return True
  1492	    if isinstance(typ, UnionType):
  1493	        return all(has_operator(x, op_method, named_type) for x in typ.relevant_items())
  1494	    if isinstance(typ, FunctionLike) and typ.is_type_obj():
  1495	        return typ.fallback.type.has_readable_member(op_method)
  1496	    if isinstance(typ, TypeType):
  1497	        # Type[Union[X, ...]] is always normalized to Union[Type[X], ...],
  1498	        # so we don't need to care about unions here, but we need to care about
  1499	        # Type[T], where upper bound of T is a union.

   378	        capture pattern.

   388	            unpack = types[unpack_index]

   390	            unpacked = get_proper_type(unpack.type)

   392	            assert isinstance(unpacked, Instance) and unpacked.type.fullname == "builtins.tuple"
   393	            if star_pos is None:

   395	                new_types = types[:unpack_index]

   397	                new_types += types[unpack_index + 1 :]

   400	                tuple([UnpackType(unpacked) if isinstance(t, UnpackType) else t for t in types]),

   404	            new_middle = []

   407	                # the whole tuple "slice".

   410	                else:

   417	            star_length = len(types) - num_patterns
   418	            new_types.append(make_simplified_union(types[star_pos : star_pos + star_length]))
   419	            new_types += types[star_pos + star_length :]

   489	                object_type = self.chk.named_type("builtins.object")
   490	                rest_type = self.chk.named_generic_type(
   491	                    "builtins.dict", [object_type, object_type]
   492	                )
   493
   494	            captures[o.rest] = rest_type
   495
   496	        if can_match:
   497	            # We can't narrow the type here, as Mapping key is invariant.
   498	            new_type = self.type_context[-1]
   499	        else:
   500	            new_type = UninhabitedType()
   501	        return PatternType(new_type, current_type, captures)

   503	    def get_mapping_item_type(
   504	        self, pattern: MappingPattern, mapping_type: Type, key: Expression
   505	    ) -> Type | None:
   506	        mapping_type = get_proper_type(mapping_type)
   507	        if isinstance(mapping_type, TypedDictType):
   508	            with self.msg.filter_errors() as local_errors:
   509	                result: Type | None = self.chk.expr_checker.visit_typeddict_index_expr(
   510	                    mapping_type, key
   511	                )[0]
   512	                has_local_errors = local_errors.has_new_errors()
   513	            # If we can't determine the type statically fall back to treating it as a normal
   514	            # mapping
   515	            if has_local_errors:
   516	                with self.msg.filter_errors() as local_errors:
   517	                    result = self.get_simple_mapping_item_type(pattern, mapping_type, key)
   518
   519	                    if local_errors.has_new_errors():
   520	                        result = None
   521	        else:
   522	            with self.msg.filter_errors():
   523	                result = self.get_simple_mapping_item_type(pattern, mapping_type, key)
   524	        return result

   526	    def get_simple_mapping_item_type(
   527	        self, pattern: MappingPattern, mapping_type: Type, key: Expression
   528	    ) -> Type:
   529	        result, _ = self.chk.expr_checker.check_method_call_by_name(
   530	            "__getitem__", mapping_type, [key], [ARG_POS], pattern
   531	        )
   532	        return result

   534	    def visit_class_pattern(self, o: ClassPattern) -> PatternType:
   535	        current_type = get_proper_type(self.type_context[-1])
   536
   537	        #
   538	        # Check class type
   539	        #
   540	        type_info = o.class_ref.node
   541	        if type_info is None:
   542	            return PatternType(AnyType(TypeOfAny.from_error), AnyType(TypeOfAny.from_error), {})
   543	        if isinstance(type_info, TypeAlias) and not type_info.no_args:
   544	            self.msg.fail(message_registry.CLASS_PATTERN_GENERIC_TYPE_ALIAS, o)
   545	            return self.early_non_match()
   546	        if isinstance(type_info, TypeInfo):
   547	            typ: Type = fill_typevars_with_any(type_info)
   548	        elif isinstance(type_info, TypeAlias):
   549	            typ = type_info.target
   550	        elif (
   551	            isinstance(type_info, Var)
   552	            and type_info.type is not None
   553	            and isinstance(get_proper_type(type_info.type), AnyType)
   554	        ):
   555	            typ = type_info.type
   556	        else:
   557	            if isinstance(type_info, Var) and type_info.type is not None:
   558	                name = type_info.type.str_with_options(self.options)
   559	            else:
   560	                name = type_info.name
   561	            self.msg.fail(message_registry.CLASS_PATTERN_TYPE_REQUIRED.format(name), o)
   562	            return self.early_non_match()
   563
   564	        new_type, rest_type = self.chk.conditional_types_with_intersection(
   565	            current_type, [get_type_range(typ)], o, default=current_type
   566	        )
   567	        if is_uninhabited(new_type):
   568	            return self.early_non_match()
   569	        # TODO: Do I need this?
   570	        narrowed_type = narrow_declared_type(current_type, new_type)
   571
   572	        #
   573	        # Convert positional to keyword patterns
   574	        #
   575	        keyword_pairs: list[tuple[str | None, Pattern]] = []
   576	        match_arg_set: set[str] = set()
   577
   578	        captures: dict[Expression, Type] = {}
   579
   580	        if len(o.positionals) != 0:
   581	            if self.should_self_match(typ):
   582	                if len(o.positionals) > 1:
   583	                    self.msg.fail(message_registry.CLASS_PATTERN_TOO_MANY_POSITIONAL_ARGS, o)
   584	                pattern_type = self.accept(o.positionals[0], narrowed_type)
   585	                if not is_uninhabited(pattern_type.type):
   586	                    return PatternType(
   587	                        pattern_type.type,
   588	                        join_types(rest_type, pattern_type.rest_type),
   589	                        pattern_type.captures,
   590	                    )
   591	                captures = pattern_type.captures
   592	            else:
   593	                with self.msg.filter_errors() as local_errors:
   594	                    match_args_type = analyze_member_access(
   595	                        "__match_args__",
   596	                        typ,
   597	                        o,
   598	                        is_lvalue=False,
   599	                        is_super=False,
   600	                        is_operator=False,
   601	                        original_type=typ,
   602	                        chk=self.chk,
   603	                    )
   604	                    has_local_errors = local_errors.has_new_errors()
   605	                if has_local_errors:
   606	                    self.msg.fail(
   607	                        message_registry.MISSING_MATCH_ARGS.format(
   608	                            typ.str_with_options(self.options)
   609	                        ),
   610	                        o,
   611	                    )
   612	                    return self.early_non_match()
