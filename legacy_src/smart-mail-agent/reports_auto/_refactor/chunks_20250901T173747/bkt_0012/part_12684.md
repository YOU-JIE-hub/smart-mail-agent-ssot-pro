   514	        arg_types: list[list[Type]],
   515	    ) -> str:
   516	        args: list[str] = []
   517	        for i in range(len(arg_types)):
   518	            for kind, name, typ in zip(arg_kinds[i], arg_names[i], arg_types[i]):
   519	                arg = self.format_type(None, typ)
   520	                if kind == ARG_STAR:
   521	                    arg = "*" + arg
   522	                elif kind == ARG_STAR2:
   523	                    arg = "**" + arg
   524	                elif kind.is_named():
   525	                    if name:
   526	                        arg = f"{name}={arg}"
   527	            args.append(arg)
   528	        return f"({', '.join(args)})"

   530	    def find_node(self, key: str) -> tuple[str, str, FuncDef]:
   531	        """From a target name, return module/target names and the func def.
   532
   533	        The 'key' argument can be in one of two formats:
   534	        * As the function full name, e.g., package.module.Cls.method
   535	        * As the function location as file and line separated by column,
   536	          e.g., path/to/file.py:42
   537	        """
   538	        # TODO: Also return OverloadedFuncDef -- currently these are ignored.
   539	        node: SymbolNode | None = None
   540	        if ":" in key:
   541	            # A colon might be part of a drive name on Windows (like `C:/foo/bar`)
   542	            # and is also used as a delimiter between file path and lineno.
   543	            # If a colon is there for any of those reasons, it must be a file+line
   544	            # reference.
   545	            platform_key_count = 2 if sys.platform == "win32" else 1
   546	            if key.count(":") > platform_key_count:
   547	                raise SuggestionFailure(
   548	                    "Malformed location for function: {}. Must be either"
   549	                    " package.module.Class.method or path/to/file.py:line".format(key)
   550	                )
   551	            file, line = key.rsplit(":", 1)
   552	            if not line.isdigit():
   553	                raise SuggestionFailure(f"Line number must be a number. Got {line}")
   554	            line_number = int(line)
   555	            modname, node = self.find_node_by_file_and_line(file, line_number)
   556	            tail = node.fullname[len(modname) + 1 :]  # add one to account for '.'
   557	        else:
   558	            target = split_target(self.fgmanager.graph, key)
   559	            if not target:
   560	                raise SuggestionFailure(f"Cannot find module for {key}")
   561	            modname, tail = target
   562	            node = self.find_node_by_module_and_name(modname, tail)
   563
   564	        if isinstance(node, Decorator):
   565	            node = self.extract_from_decorator(node)
   566	            if not node:
   567	                raise SuggestionFailure(f"Object {key} is a decorator we can't handle")
   568
   569	        if not isinstance(node, FuncDef):
   570	            raise SuggestionFailure(f"Object {key} is not a function")
   571
   572	        return modname, tail, node

   574	    def find_node_by_module_and_name(self, modname: str, tail: str) -> SymbolNode | None:
   575	        """Find symbol node by module id and qualified name.
   576
   577	        Raise SuggestionFailure if can't find one.
   578	        """
   579	        tree = self.ensure_loaded(self.fgmanager.graph[modname])
   580
   581	        # N.B. This is reimplemented from update's lookup_target
   582	        # basically just to produce better error messages.
   583
   584	        names: SymbolTable = tree.names
   585
   586	        # Look through any classes
   587	        components = tail.split(".")
   588	        for i, component in enumerate(components[:-1]):
   589	            if component not in names:
   590	                raise SuggestionFailure(
   591	                    "Unknown class {}.{}".format(modname, ".".join(components[: i + 1]))
   592	                )
   593	            node: SymbolNode | None = names[component].node
   594	            if not isinstance(node, TypeInfo):
   595	                raise SuggestionFailure(
   596	                    "Object {}.{} is not a class".format(modname, ".".join(components[: i + 1]))
   597	                )
   598	            names = node.names
   599
   600	        # Look for the actual function/method
   601	        funcname = components[-1]
   602	        if funcname not in names:
   603	            key = modname + "." + tail
   604	            raise SuggestionFailure(
   605	                "Unknown {} {}".format("method" if len(components) > 1 else "function", key)
   606	            )
   607	        return names[funcname].node

   609	    def find_node_by_file_and_line(self, file: str, line: int) -> tuple[str, SymbolNode]:
   610	        """Find symbol node by path to file and line number.
   611
   612	        Find the first function declared *before or on* the line number.
   613
   614	        Return module id and the node found. Raise SuggestionFailure if can't find one.
   615	        """
   616	        if not any(file.endswith(ext) for ext in PYTHON_EXTENSIONS):
   617	            raise SuggestionFailure("Source file is not a Python file")
   618	        try:
   619	            modname, _ = self.finder.crawl_up(os.path.normpath(file))
   620	        except InvalidSourceList as e:
   621	            raise SuggestionFailure("Invalid source file name: " + file) from e
   622	        if modname not in self.graph:
   623	            raise SuggestionFailure("Unknown module: " + modname)
   624	        # We must be sure about any edits in this file as this might affect the line numbers.
   625	        tree = self.ensure_loaded(self.fgmanager.graph[modname], force=True)
   626	        node: SymbolNode | None = None
   627	        closest_line: int | None = None
   628	        # TODO: Handle nested functions.
   629	        for _, sym, _ in tree.local_definitions():
   630	            if isinstance(sym.node, (FuncDef, Decorator)):
   631	                sym_line = sym.node.line
   632	            # TODO: add support for OverloadedFuncDef.
   633	            else:
   634	                continue
   635
   636	            # We want the closest function above the specified line
   637	            if sym_line <= line and (closest_line is None or sym_line > closest_line):
   638	                closest_line = sym_line
   639	                node = sym.node
   640	        if not node:
   641	            raise SuggestionFailure(f"Cannot find a function at line {line}")
   642	        return modname, node

   644	    def extract_from_decorator(self, node: Decorator) -> FuncDef | None:
   645	        for dec in node.decorators:
   646	            typ = None
   647	            if isinstance(dec, RefExpr) and isinstance(dec.node, (Var, FuncDef)):
   648	                typ = get_proper_type(dec.node.type)
   649	            elif (
   650	                isinstance(dec, CallExpr)
   651	                and isinstance(dec.callee, RefExpr)
   652	                and isinstance(dec.callee.node, (Decorator, FuncDef, Var))
   653	                and isinstance((call_tp := get_proper_type(dec.callee.node.type)), CallableType)
   654	            ):
   655	                typ = get_proper_type(call_tp.ret_type)
   656
   657	            if isinstance(typ, Instance):
   658	                call_method = typ.type.get_method("__call__")
   659	                if isinstance(call_method, FuncDef) and isinstance(call_method.type, FunctionLike):
   660	                    typ = bind_self(call_method.type, None)
   661
   662	            if not isinstance(typ, FunctionLike):
   663	                return None
   664	            for ct in typ.items:
   665	                if not (
   666	                    len(ct.arg_types) == 1
   667	                    and isinstance(ct.arg_types[0], TypeVarType)
   668	                    and ct.arg_types[0] == ct.ret_type
   669	                ):
   670	                    return None
   671
   672	        return node.func

   751	    def score_type(self, t: Type, arg_pos: bool) -> int:
   752	        """Generate a score for a type that we use to pick which type to use.
   753
   754	        Lower is better, prefer non-union/non-any types. Don't penalize optionals.
   755	        """
   756	        t = get_proper_type(t)
   757	        if isinstance(t, AnyType):
   758	            return 20
   759	        if arg_pos and isinstance(t, NoneType):
   760	            return 20
   761	        if isinstance(t, UnionType):
   762	            if any(isinstance(get_proper_type(x), AnyType) for x in t.items):
   763	                return 20
   764	            if any(has_any_type(x) for x in t.items):
   765	                return 15
   766	            if not is_overlapping_none(t):
   767	                return 10
   768	        if isinstance(t, CallableType) and (has_any_type(t) or is_tricky_callable(t)):
   769	            return 10
   770	        return 0
   771
   772	    def score_callable(self, t: CallableType) -> int:
   773	        return sum(self.score_type(x, arg_pos=True) for x in t.arg_types) + self.score_type(
   774	            t.ret_type, arg_pos=False
   775	        )

   778	def any_score_type(ut: Type, arg_pos: bool) -> float:
   779	    """Generate a very made up number representing the Anyness of a type.
   780
   781	    Higher is better, 1.0 is max
   782	    """
   783	    t = get_proper_type(ut)
   784	    if isinstance(t, AnyType) and t.type_of_any != TypeOfAny.suggestion_engine:
   785	        return 0
   786	    if isinstance(t, NoneType) and arg_pos:
   787	        return 0.5
   788	    if isinstance(t, UnionType):
   789	        if any(isinstance(get_proper_type(x), AnyType) for x in t.items):
   790	            return 0.5
   791	        if any(has_any_type(x) for x in t.items):
   792	            return 0.25
   793	    if isinstance(t, CallableType) and is_tricky_callable(t):
   794	        return 0.5
   795	    if has_any_type(t):
   796	        return 0.5
   797
   798	    return 1.0

   801	def any_score_callable(t: CallableType, is_method: bool, ignore_return: bool) -> float:
   802	    # Ignore the first argument of methods
   803	    scores = [any_score_type(x, arg_pos=True) for x in t.arg_types[int(is_method) :]]
   804	    # Return type counts twice (since it spreads type information), unless it is
   805	    # None in which case it does not count at all. (Though it *does* still count
   806	    # if there are no arguments.)
   807	    if not isinstance(get_proper_type(t.ret_type), NoneType) or not scores:
   808	        ret = 1.0 if ignore_return else any_score_type(t.ret_type, arg_pos=False)
   809	        scores += [ret, ret]
   810
   811	    return sum(scores) / len(scores)

   822	    # TODO: Probably a lot
   823	    def __init__(self, module: str | None, graph: Graph, options: Options) -> None:
   824	        super().__init__(options=options)

   828	    def visit_any(self, t: AnyType) -> str:

   830	            return t.missing_import_name

   834	    def visit_instance(self, t: Instance) -> str:
   835	        s = t.type.fullname or t.type.name or None
   836	        if s is None:
   837	            return "<???>"
   838
   839	        mod_obj = split_target(self.graph, s)
   840	        assert mod_obj

   843	        # If a class is imported into the current module, rewrite the reference

   845	        # inserting redundant imports when a type has been reexported.
   846	        if self.module:

   848	            tree = self.graph[self.module].tree
   849	            if tree and parts[0] in tree.names:
   850	                mod = self.module

   852	        if (mod, obj) == ("builtins", "tuple"):
   853	            mod, obj = "typing", "Tuple[" + t.args[0].accept(self) + ", ...]"
   854	        elif t.args:
   855	            obj += f"[{self.list_str(t.args)}]"

   857	        if mod_obj == ("builtins", "unicode"):
   858	            return "Text"
   859	        elif mod == "builtins":
   860	            return obj
   861	        else:
   862	            delim = "." if "." not in obj else ":"
   863	            return mod + delim + obj

   868	            if fallback_name != "builtins.tuple":

   941	def refine_type(ti: Type, si: Type) -> Type:
   942	    """Refine `ti` by replacing Anys in it with information taken from `si`
   943
   944	    This basically works by, when the types have the same structure,
   945	    traversing both of them in parallel and replacing Any on the left
   946	    with whatever the type on the right is. If the types don't have the
   947	    same structure (or aren't supported), the left type is chosen.
   948
   949	    For example:
   950	      refine(Any, T) = T,  for all T
   951	      refine(float, int) = float
   952	      refine(List[Any], List[int]) = List[int]
   953	      refine(Dict[int, Any], Dict[Any, int]) = Dict[int, int]
   954	      refine(Tuple[int, Any], Tuple[Any, int]) = Tuple[int, int]
   955
   956	      refine(Callable[[Any], Any], Callable[[int], int]) = Callable[[int], int]
   957	      refine(Callable[..., int], Callable[[int, float], Any]) = Callable[[int, float], int]
   958
   959	      refine(Optional[Any], int) = Optional[int]
   960	      refine(Optional[Any], Optional[int]) = Optional[int]
   961	      refine(Optional[Any], Union[int, str]) = Optional[Union[int, str]]
   962	      refine(Optional[List[Any]], List[int]) = List[int]
   963
   964	    """
   965	    t = get_proper_type(ti)
   966	    s = get_proper_type(si)
   967
   968	    if isinstance(t, AnyType):
   969	        # If s is also an Any, we return if it is a missing_import Any
   970	        return t if isinstance(s, AnyType) and t.missing_import_name else s
   971
   972	    if isinstance(t, Instance) and isinstance(s, Instance) and t.type == s.type:
   973	        return t.copy_modified(args=[refine_type(ta, sa) for ta, sa in zip(t.args, s.args)])
   974
   975	    if (
   976	        isinstance(t, TupleType)
   977	        and isinstance(s, TupleType)
   978	        and t.partial_fallback == s.partial_fallback
   979	        and len(t.items) == len(s.items)
   980	    ):
   981	        return t.copy_modified(items=[refine_type(ta, sa) for ta, sa in zip(t.items, s.items)])
   982
   983	    if isinstance(t, CallableType) and isinstance(s, CallableType):
   984	        return refine_callable(t, s)
   985
   986	    if isinstance(t, UnionType):
   987	        return refine_union(t, s)
   988
   989	    # TODO: Refining of builtins.tuple, Type?
   990
   991	    return t

   994	def refine_union(t: UnionType, s: ProperType) -> Type:
   995	    """Refine a union type based on another type.
   996
   997	    This is done by refining every component of the union against the
   998	    right hand side type (or every component of its union if it is
   999	    one). If an element of the union is successfully refined, we drop it
  1000	    from the union in favor of the refined versions.
  1001	    """
  1002	    # Don't try to do any union refining if the types are already the
  1003	    # same.  This prevents things like refining Optional[Any] against
  1004	    # itself and producing None.
  1005	    if t == s:
  1006	        return t
  1007
  1008	    rhs_items = s.items if isinstance(s, UnionType) else [s]
  1009
  1010	    new_items = []
  1011	    for lhs in t.items:
  1012	        refined = False
  1013	        for rhs in rhs_items:
  1014	            new = refine_type(lhs, rhs)
  1015	            if new != lhs:
  1016	                new_items.append(new)
  1017	                refined = True
  1018	        if not refined:
  1019	            new_items.append(lhs)
  1020
  1021	    # Turn strict optional on when simplifying the union since we
  1022	    # don't want to drop Nones.
  1023	    with state.strict_optional_set(True):
  1024	        return make_simplified_union(new_items)

  1027	def refine_callable(t: CallableType, s: CallableType) -> CallableType:
  1028	    """Refine a callable based on another.
  1029
  1030	    See comments for refine_type.
  1031	    """
  1032	    if t.fallback != s.fallback:
  1033	        return t
  1034

   448	    """Parse a list of lines that represent a sequence of test items."""
   449
   450	    lines = ["", "[case " + name + "]"] + raw_data.split("\n")
   451	    ret: list[TestItem] = []
   452	    data: list[str] = []

   457	    i = 0
   458	    i0 = 0
   459	    while i < len(lines):
   460	        s = lines[i].strip()
   461
   462	        if lines[i].startswith("[") and s.endswith("]"):
   463	            if id:
   464	                data = collapse_line_continuation(data)
   465	                data = strip_list(data)
   466	                ret.append(TestItem(id, arg, data, i0 + 1, i))

   230	    for s in a:
   231	        prefix = os.sep
   232	        ss = s
   233	        for p in prefix, prefix.replace(os.sep, "/"):
   234	            if p != "/" and p != "//" and p != "\\" and p != "\\\\":

   240	        res.append(re.sub("\\r$", "", ss))

   278	def retry_on_error(func: Callable[[], Any], max_wait: float = 1.0) -> None:
   279	    """Retry callback with exponential backoff when it raises OSError.
   280
   281	    If the function still generates an error after max_wait seconds, propagate
   282	    the exception.
   283
   284	    This can be effective against random file system operation failures on
   285	    Windows.
   286	    """
   287	    t0 = time.time()
   288	    wait_time = 0.01
   289	    while True:
   290	        try:
   291	            func()
   292	            return
   293	        except OSError:
   294	            wait_time = min(wait_time * 2, t0 + max_wait - time.time())
   295	            if wait_time <= 0.01:
   296	                # Done enough waiting, the error seems persistent.
   297	                raise
   298	            time.sleep(wait_time)

   301	def good_repr(obj: object) -> str:
   302	    if isinstance(obj, str):
   303	        if obj.count("\n") > 1:
   304	            bits = ["'''\\"]
   305	            for line in obj.split("\n"):
   306	                # force repr to use ' not ", then cut it off
   307	                bits.append(repr('"' + line)[2:-1])
   308	            bits[-1] += "'''"
   309	            return "\n".join(bits)
   310	    return repr(obj)

   332	def parse_options(
   333	    program_text: str, testcase: DataDrivenTestCase, incremental_step: int
   334	) -> Options:
   335	    """Parse comments like '# flags: --foo' in a test case."""
   336	    options = Options()
   337	    flags = re.search("# flags: (.*)$", program_text, flags=re.MULTILINE)
   338	    if incremental_step > 1:
   339	        flags2 = re.search(f"# flags{incremental_step}: (.*)$", program_text, flags=re.MULTILINE)
   340	        if flags2:
   341	            flags = flags2
   342
   343	    if flags:
   344	        flag_list = flags.group(1).split()
   345	        flag_list.append("--no-site-packages")  # the tests shouldn't need an installed Python
   346	        targets, options = process_options(flag_list, require_targets=False)
   347	        if targets:
   348	            # TODO: support specifying targets via the flags pragma
   349	            raise RuntimeError("Specifying targets via the flags pragma is not supported.")
   350	        if "--show-error-codes" not in flag_list:
   351	            options.hide_error_codes = True
   352	    else:
   353	        flag_list = []
   354	        options = Options()
   355	        options.error_summary = False
   356	        options.hide_error_codes = True
   357	        options.force_union_syntax = True
   358
   359	    # Allow custom python version to override testfile_pyversion.
   360	    if all(flag.split("=")[0] != "--python-version" for flag in flag_list):
   361	        options.python_version = testfile_pyversion(testcase.file)
   362
   363	    if testcase.config.getoption("--mypy-verbose"):
   364	        options.verbosity = testcase.config.getoption("--mypy-verbose")
   365
   366	    return options
