   826	class StubgenPythonSuite(DataSuite):
   827	    """Data-driven end-to-end test cases that generate stub files.
   828
   829	    You can use these magic test case name suffixes:
   830
   831	    *_semanal
   832	        Run semantic analysis (slow as this uses real stubs -- only use
   833	        when necessary)
   834	    *_import
   835	        Import module and perform runtime introspection (in the current
   836	        process!)
   837
   838	    You can use these magic comments:
   839
   840	    # flags: --some-stubgen-option ...
   841	        Specify custom stubgen options
   842
   843	    # modules: module1 module2 ...
   844	        Specify which modules to output (by default only 'main')
   845	    """
   846
   847	    required_out_section = True
   848	    base_path = "."
   849	    files = ["stubgen.test"]
   850
   851	    @unittest.skipIf(sys.platform == "win32", "clean up fails on Windows")
   852	    def run_case(self, testcase: DataDrivenTestCase) -> None:
   853	        with local_sys_path_set():
   854	            self.run_case_inner(testcase)
   855
   856	    def run_case_inner(self, testcase: DataDrivenTestCase) -> None:
   857	        extra = []  # Extra command-line args
   858	        mods = []  # Module names to process
   859	        source = "\n".join(testcase.input)
   860	        for file, content in testcase.files + [("./main.py", source)]:
   861	            # Strip ./ prefix and .py suffix.
   862	            mod = file[2:-3].replace("/", ".")
   863	            if mod.endswith(".__init__"):
   864	                mod, _, _ = mod.rpartition(".")
   865	            mods.append(mod)
   866	            if "-p " not in source:
   867	                extra.extend(["-m", mod])
   868	            with open(file, "w") as f:
   869	                f.write(content)
   870
   871	        options = self.parse_flags(source, extra)
   872	        if sys.version_info < options.pyversion:
   873	            pytest.skip()
   874	        modules = self.parse_modules(source)
   875	        out_dir = "out"
   876	        try:
   877	            try:
   878	                if testcase.name.endswith("_inspect"):
   879	                    options.inspect = True
   880	                else:
   881	                    if not testcase.name.endswith("_import"):
   882	                        options.no_import = True
   883	                    if not testcase.name.endswith("_semanal"):
   884	                        options.parse_only = True
   885
   886	                generate_stubs(options)
   887	                a: list[str] = []
   888	                for module in modules:
   889	                    fnam = module_to_path(out_dir, module)
   890	                    self.add_file(fnam, a, header=len(modules) > 1)
   891	            except CompileError as e:
   892	                a = e.messages
   893	            assert_string_arrays_equal(
   894	                testcase.output, a, f"Invalid output ({testcase.file}, line {testcase.line})"
   895	            )
   896	        finally:
   897	            for mod in mods:
   898	                if mod in sys.modules:
   899	                    del sys.modules[mod]
   900	            shutil.rmtree(out_dir)
   901
   902	    def parse_flags(self, program_text: str, extra: list[str]) -> Options:
   903	        flags = re.search("# flags: (.*)$", program_text, flags=re.MULTILINE)
   904	        pyversion = None
   905	        if flags:
   906	            flag_list = flags.group(1).split()
   907	            for i, flag in enumerate(flag_list):
   908	                if flag.startswith("--python-version="):
   909	                    pyversion = flag.split("=", 1)[1]
   910	                    del flag_list[i]
   911	                    break
   912	        else:
   913	            flag_list = []
   914	        options = parse_options(flag_list + extra)
   915	        if pyversion:
   916	            # A hack to allow testing old python versions with new language constructs
   917	            # This should be rarely used in general as stubgen output should not be version-specific
   918	            major, minor = pyversion.split(".", 1)
   919	            options.pyversion = (int(major), int(minor))
   920	        if "--verbose" not in flag_list:
   921	            options.quiet = True
   922	        else:
   923	            options.verbose = True
   924	        return options

   926	    def parse_modules(self, program_text: str) -> list[str]:
   927	        modules = re.search("# modules: (.*)$", program_text, flags=re.MULTILINE)
   928	        if modules:
   929	            return modules.group(1).split()
   930	        else:
   931	            return ["main"]

   933	    def add_file(self, path: str, result: list[str], header: bool) -> None:
   934	        if not os.path.exists(path):
   935	            result.append("<%s was not generated>" % path.replace("\\", "/"))
   936	            return
   937	        if header:
   938	            result.append(f"# {path[4:]}")
   939	        with open(path, encoding="utf8") as file:
   940	            result.extend(file.read().splitlines())

   164	    with use_tmp_dir(TEST_MODULE_NAME) as tmp_dir:

   189	        outerr.getvalue()
   190	        # remove cwd as it's not available from outside
   191	        .replace(os.path.realpath(tmp_dir) + os.sep, "").replace(tmp_dir + os.sep, "")
   192	    )
   193	    return filtered_output, filtered_outerr

   209	def collect_cases(fn: Callable[..., Iterator[Case]]) -> Callable[..., None]:
   210	    """run_stubtest used to be slow, so we used this decorator to combine cases.
   211
   212	    If you're reading this and bored, feel free to refactor this and make it more like
   213	    other mypy tests.
   214
   215	    """
   216
   217	    def test(*args: Any, **kwargs: Any) -> None:
   218	        cases = list(fn(*args, **kwargs))
   219	        expected_errors = set()
   220	        for c in cases:
   221	            if c.error is None:
   222	                continue
   223	            expected_error = c.error
   224	            if expected_error == "":
   225	                expected_error = TEST_MODULE_NAME
   226	            elif not expected_error.startswith(f"{TEST_MODULE_NAME}."):
   227	                expected_error = f"{TEST_MODULE_NAME}.{expected_error}"
   228	            assert expected_error not in expected_errors, (
   229	                "collect_cases merges cases into a single stubtest invocation; we already "
   230	                "expect an error for {}".format(expected_error)
   231	            )
   232	            expected_errors.add(expected_error)
   233	        output = run_stubtest(
   234	            stub="\n\n".join(textwrap.dedent(c.stub.lstrip("\n")) for c in cases),
   235	            runtime="\n\n".join(textwrap.dedent(c.runtime.lstrip("\n")) for c in cases),
   236	            options=["--generate-allowlist"],
   237	        )
   238
   239	        actual_errors = set(output.splitlines())
   240	        if actual_errors != expected_errors:
   241	            output = run_stubtest(
   242	                stub="\n\n".join(textwrap.dedent(c.stub.lstrip("\n")) for c in cases),
   243	                runtime="\n\n".join(textwrap.dedent(c.runtime.lstrip("\n")) for c in cases),
   244	                options=[],
   245	            )
   246	            assert actual_errors == expected_errors, output
   247
   248	    return test
