   109	class Attribute:
   110	    """The value of an attr.ib() call."""
   111
   112	    def __init__(
   113	        self,
   114	        name: str,
   115	        alias: str | None,
   116	        info: TypeInfo,
   117	        has_default: bool,
   118	        init: bool,
   119	        kw_only: bool,
   120	        converter: Converter | None,
   121	        context: Context,
   122	        init_type: Type | None,
   123	    ) -> None:
   124	        self.name = name
   125	        self.alias = alias
   126	        self.info = info
   127	        self.has_default = has_default
   128	        self.init = init
   129	        self.kw_only = kw_only
   130	        self.converter = converter
   131	        self.context = context
   132	        self.init_type = init_type
   133
   134	    def argument(self, ctx: mypy.plugin.ClassDefContext) -> Argument:
   135	        """Return this attribute as an argument to __init__."""
   136	        assert self.init
   137	        init_type: Type | None = None
   138	        if self.converter:
   139	            if self.converter.init_type:
   140	                init_type = self.converter.init_type
   141	                if init_type and self.init_type and self.converter.ret_type:
   142	                    # The converter return type should be the same type as the attribute type.
   143	                    # Copy type vars from attr type to converter.
   144	                    converter_vars = get_type_vars(self.converter.ret_type)
   145	                    init_vars = get_type_vars(self.init_type)
   146	                    if converter_vars and len(converter_vars) == len(init_vars):
   147	                        variables = {
   148	                            binder.id: arg for binder, arg in zip(converter_vars, init_vars)
   149	                        }
   150	                        init_type = expand_type(init_type, variables)
   151	            else:
   152	                ctx.api.fail("Cannot determine __init__ type from converter", self.context)
   153	                init_type = AnyType(TypeOfAny.from_error)
   154	        else:  # There is no converter, the init type is the normal type.
   155	            init_type = self.init_type or self.info[self.name].type
   156
   157	        unannotated = False
   158	        if init_type is None:
   159	            unannotated = True
   160	            # Convert type not set to Any.
   161	            init_type = AnyType(TypeOfAny.unannotated)
   162	        else:
   163	            proper_type = get_proper_type(init_type)
   164	            if isinstance(proper_type, AnyType):
   165	                if proper_type.type_of_any == TypeOfAny.unannotated:
   166	                    unannotated = True
   167
   168	        if unannotated and ctx.api.options.disallow_untyped_defs:
   169	            # This is a compromise.  If you don't have a type here then the
   170	            # __init__ will be untyped. But since the __init__ is added it's
   171	            # pointing at the decorator. So instead we also show the error in the
   172	            # assignment, which is where you would fix the issue.
   173	            node = self.info[self.name].node
   174	            assert node is not None
   175	            ctx.api.msg.need_annotation_for_var(node, self.context)
   176
   177	        if self.kw_only:
   178	            arg_kind = ARG_NAMED_OPT if self.has_default else ARG_NAMED
   179	        else:
   180	            arg_kind = ARG_OPT if self.has_default else ARG_POS
   181
   182	        # Attrs removes leading underscores when creating the __init__ arguments.
   183	        name = self.alias or self.name.lstrip("_")
   184	        return Argument(Var(name, init_type), init_type, None, arg_kind)

   187	        """Serialize this object so it can be saved and restored."""
   188	        return {
   189	            "name": self.name,
   190	            "alias": self.alias,
   191	            "has_default": self.has_default,
   192	            "init": self.init,
   193	            "kw_only": self.kw_only,
   194	            "has_converter": self.converter is not None,
   195	            "converter_init_type": (
   196	                self.converter.init_type.serialize()
   197	                if self.converter and self.converter.init_type
   198	                else None
   199	            ),
   200	            "context_line": self.context.line,
   201	            "context_column": self.context.column,
   202	            "init_type": self.init_type.serialize() if self.init_type else None,

   210	        raw_init_type = data["init_type"]
   211	        init_type = deserialize_and_fixup_type(raw_init_type, api) if raw_init_type else None
   212	        raw_converter_init_type = data["converter_init_type"]
   213	        converter_init_type = (
   214	            deserialize_and_fixup_type(raw_converter_init_type, api)
   215	            if raw_converter_init_type
   216	            else None
   217	        )
   218
   219	        return Attribute(
   220	            data["name"],
   221	            data["alias"],
   222	            info,
   223	            data["has_default"],

   225	            data["kw_only"],
   226	            Converter(converter_init_type) if data["has_converter"] else None,

   240	def _determine_eq_order(ctx: mypy.plugin.ClassDefContext) -> bool:
   241	    """
   242	    Validate the combination of *cmp*, *eq*, and *order*. Derive the effective
   243	    value of order.
   244	    """
   245	    cmp = _get_decorator_optional_bool_argument(ctx, "cmp")
   246	    eq = _get_decorator_optional_bool_argument(ctx, "eq")
   247	    order = _get_decorator_optional_bool_argument(ctx, "order")
   248
   249	    if cmp is not None and any((eq is not None, order is not None)):
   250	        ctx.api.fail('Don\'t mix "cmp" with "eq" and "order"', ctx.reason)
   251
   252	    # cmp takes precedence due to bw-compatibility.
   253	    if cmp is not None:
   254	        return cmp
   255
   256	    # If left None, equality is on and ordering mirrors equality.
   257	    if eq is None:
   258	        eq = True
   259
   260	    if order is None:
   261	        order = eq
   262
   263	    if eq is False and order is True:
   264	        ctx.api.fail("eq must be True if order is True", ctx.reason)
   265
   266	    return order

   269	def _get_decorator_optional_bool_argument(
   270	    ctx: mypy.plugin.ClassDefContext, name: str, default: bool | None = None
   271	) -> bool | None:
   272	    """Return the Optional[bool] argument for the decorator.
   273
   274	    This handles both @decorator(...) and @decorator.
   275	    """
   276	    if isinstance(ctx.reason, CallExpr):
   277	        attr_value = _get_argument(ctx.reason, name)
   278	        if attr_value:
   279	            if isinstance(attr_value, NameExpr):
   280	                if attr_value.fullname == "builtins.True":
   281	                    return True
   282	                if attr_value.fullname == "builtins.False":
   283	                    return False
   284	                if attr_value.fullname == "builtins.None":
   285	                    return None
   286	            ctx.api.fail(
   287	                f'"{name}" argument must be a True, False, or None literal',
   288	                ctx.reason,
   289	                code=LITERAL_REQ,
   290	            )
   291	            return default
   292	        return default
   293	    else:
   294	        return default

   307	def attr_class_maker_callback(
   308	    ctx: mypy.plugin.ClassDefContext,
   309	    auto_attribs_default: bool | None = False,
   310	    frozen_default: bool = False,
   311	    slots_default: bool = False,
   312	) -> bool:
   313	    """Add necessary dunder methods to classes decorated with attr.s.
   314
   315	    attrs is a package that lets you define classes without writing dull boilerplate code.
   316
   317	    At a quick glance, the decorator searches the class body for assignments of `attr.ib`s (or
   318	    annotated variables if auto_attribs=True), then depending on how the decorator is called,
   319	    it will add an __init__ or all the compare methods.
   320	    For frozen=True it will turn the attrs into properties.
   321
   322	    Hashability will be set according to https://www.attrs.org/en/stable/hashing.html.
   323
   324	    See https://www.attrs.org/en/stable/how-does-it-work.html for information on how attrs works.
   325
   326	    If this returns False, some required metadata was not ready yet, and we need another
   327	    pass.
   328	    """
   329	    with state.strict_optional_set(ctx.api.options.strict_optional):
   330	        # This hook is called during semantic analysis, but it uses a bunch of
   331	        # type-checking ops, so it needs the strict optional set properly.
   332	        return attr_class_maker_callback_impl(
   333	            ctx, auto_attribs_default, frozen_default, slots_default
   334	        )

   337	def attr_class_maker_callback_impl(
   338	    ctx: mypy.plugin.ClassDefContext,
   339	    auto_attribs_default: bool | None,
   340	    frozen_default: bool,
   341	    slots_default: bool,
   342	) -> bool:
   343	    info = ctx.cls.info
   344
   345	    init = _get_decorator_bool_argument(ctx, "init", True)
   346	    frozen = _get_frozen(ctx, frozen_default)
   347	    order = _determine_eq_order(ctx)
   348	    slots = _get_decorator_bool_argument(ctx, "slots", slots_default)
   349
   350	    auto_attribs = _get_decorator_optional_bool_argument(ctx, "auto_attribs", auto_attribs_default)
   351	    kw_only = _get_decorator_bool_argument(ctx, "kw_only", False)
   352	    match_args = _get_decorator_bool_argument(ctx, "match_args", True)
   353
   354	    for super_info in ctx.cls.info.mro[1:-1]:
   355	        if "attrs_tag" in super_info.metadata and "attrs" not in super_info.metadata:
   356	            # Super class is not ready yet. Request another pass.
   357	            return False
   358
   359	    attributes = _analyze_class(ctx, auto_attribs, kw_only)
   360
   361	    # Check if attribute types are ready.
   362	    for attr in attributes:
   363	        node = info.get(attr.name)
   364	        if node is None:
   365	            # This name is likely blocked by some semantic analysis error that
   366	            # should have been reported already.
   367	            _add_empty_metadata(info)
   368	            return True
   369
   370	    _add_attrs_magic_attribute(ctx, [(attr.name, info[attr.name].type) for attr in attributes])
   371	    if slots:
   372	        _add_slots(ctx, attributes)
   373	    if match_args and ctx.api.options.python_version[:2] >= (3, 10):
   374	        # `.__match_args__` is only added for python3.10+, but the argument
   375	        # exists for earlier versions as well.
   376	        _add_match_args(ctx, attributes)
   377
   378	    # Save the attributes so that subclasses can reuse them.
   379	    ctx.cls.info.metadata["attrs"] = {
   380	        "attributes": [attr.serialize() for attr in attributes],
   381	        "frozen": frozen,
   382	    }
   383
   384	    adder = MethodAdder(ctx)
   385	    # If  __init__ is not being generated, attrs still generates it as __attrs_init__ instead.
   386	    _add_init(ctx, attributes, adder, "__init__" if init else ATTRS_INIT_NAME)
   387
   388	    if order:
   389	        _add_order(ctx, adder)
   390	    if frozen:
   391	        _make_frozen(ctx, attributes)
   392	        # Frozen classes are hashable by default, even if inheriting from non-frozen ones.
   393	        hashable: bool | None = _get_decorator_bool_argument(
   394	            ctx, "hash", True
   395	        ) and _get_decorator_bool_argument(ctx, "unsafe_hash", True)
   396	    else:
   397	        hashable = _get_decorator_optional_bool_argument(ctx, "unsafe_hash")
   398	        if hashable is None:  # unspecified
   399	            hashable = _get_decorator_optional_bool_argument(ctx, "hash")
   400
   401	    eq = _get_decorator_optional_bool_argument(ctx, "eq")
   402	    has_own_hash = "__hash__" in ctx.cls.info.names
   403
   404	    if has_own_hash or (hashable is None and eq is False):
   405	        pass  # Do nothing.
   406	    elif hashable:
   407	        # We copy the `__hash__` signature from `object` to make them hashable.
   408	        ctx.cls.info.names["__hash__"] = ctx.cls.info.mro[-1].names["__hash__"]
   409	    else:
   410	        _remove_hashability(ctx)
   411
   412	    return True

   426	def _analyze_class(
   427	    ctx: mypy.plugin.ClassDefContext, auto_attribs: bool | None, kw_only: bool
   428	) -> list[Attribute]:
   429	    """Analyze the class body of an attr maker, its parents, and return the Attributes found.
   430
   431	    auto_attribs=True means we'll generate attributes from type annotations also.
   432	    auto_attribs=None means we'll detect which mode to use.
   433	    kw_only=True means that all attributes created here will be keyword only args in __init__.
   434	    """
   435	    own_attrs: dict[str, Attribute] = {}
   436	    if auto_attribs is None:
   437	        auto_attribs = _detect_auto_attribs(ctx)
   438
   439	    # Walk the body looking for assignments and decorators.
   440	    for stmt in ctx.cls.defs.body:
   441	        if isinstance(stmt, AssignmentStmt):
   442	            for attr in _attributes_from_assignment(ctx, stmt, auto_attribs, kw_only):
   443	                # When attrs are defined twice in the same body we want to use the 2nd definition
   444	                # in the 2nd location. So remove it from the OrderedDict.
   445	                # Unless it's auto_attribs in which case we want the 2nd definition in the
   446	                # 1st location.
   447	                if not auto_attribs and attr.name in own_attrs:
   448	                    del own_attrs[attr.name]
   449	                own_attrs[attr.name] = attr
   450	        elif isinstance(stmt, Decorator):
   451	            _cleanup_decorator(stmt, own_attrs)
   452
   453	    for attribute in own_attrs.values():
   454	        # Even though these look like class level assignments we want them to look like
   455	        # instance level assignments.
   456	        if attribute.name in ctx.cls.info.names:
   457	            node = ctx.cls.info.names[attribute.name].node
   458	            if isinstance(node, PlaceholderNode):
   459	                # This node is not ready yet.
   460	                continue
   461	            assert isinstance(node, Var)
   462	            node.is_initialized_in_class = False
   463
   464	    # Traverse the MRO and collect attributes from the parents.
   465	    taken_attr_names = set(own_attrs)
   466	    super_attrs = []
   467	    for super_info in ctx.cls.info.mro[1:-1]:
   468	        if "attrs" in super_info.metadata:
   469	            # Each class depends on the set of attributes in its attrs ancestors.
   470	            ctx.api.add_plugin_dependency(make_wildcard_trigger(super_info.fullname))
   471
   472	            for data in super_info.metadata["attrs"]["attributes"]:
   473	                # Only add an attribute if it hasn't been defined before.  This
   474	                # allows for overwriting attribute definitions by subclassing.
   475	                if data["name"] not in taken_attr_names:
   476	                    a = Attribute.deserialize(super_info, data, ctx.api)
   477	                    a.expand_typevar_from_subtype(ctx.cls.info)
   478	                    super_attrs.append(a)
   479	                    taken_attr_names.add(a.name)
   480	    attributes = super_attrs + list(own_attrs.values())
   481
   482	    # Check the init args for correct default-ness.  Note: This has to be done after all the
   483	    # attributes for all classes have been read, because subclasses can override parents.
   484	    last_default = False
   485
   486	    for i, attribute in enumerate(attributes):
   487	        if not attribute.init:
   488	            continue
   489
   490	        if attribute.kw_only:
   491	            # Keyword-only attributes don't care whether they are default or not.
   492	            continue
   493
   494	        # If the issue comes from merging different classes, report it
   495	        # at the class definition point.
   496	        context = attribute.context if i >= len(super_attrs) else ctx.cls
   497
   498	        if not attribute.has_default and last_default:
   499	            ctx.api.fail("Non-default attributes not allowed after default attributes.", context)
   500	        last_default |= attribute.has_default
   501
   502	    return attributes

   510	def _detect_auto_attribs(ctx: mypy.plugin.ClassDefContext) -> bool:
   511	    """Return whether auto_attribs should be enabled or disabled.
   512
   513	    It's disabled if there are any unannotated attribs()
   514	    """
   515	    for stmt in ctx.cls.defs.body:
   516	        if isinstance(stmt, AssignmentStmt):
   517	            for lvalue in stmt.lvalues:
   518	                lvalues, rvalues = _parse_assignments(lvalue, stmt)
   519
   520	                if len(lvalues) != len(rvalues):
   521	                    # This means we have some assignment that isn't 1 to 1.
   522	                    # It can't be an attrib.
   523	                    continue
   524
   525	                for lhs, rvalue in zip(lvalues, rvalues):
   526	                    # Check if the right hand side is a call to an attribute maker.
   527	                    if (
   528	                        isinstance(rvalue, CallExpr)
   529	                        and isinstance(rvalue.callee, RefExpr)
   530	                        and rvalue.callee.fullname in attr_attrib_makers
   531	                        and not stmt.new_syntax
   532	                    ):
   533	                        # This means we have an attrib without an annotation and so
   534	                        # we can't do auto_attribs=True
   535	                        return False
   536	    return True

   539	def _attributes_from_assignment(
   540	    ctx: mypy.plugin.ClassDefContext, stmt: AssignmentStmt, auto_attribs: bool, kw_only: bool
   541	) -> Iterable[Attribute]:
   542	    """Return Attribute objects that are created by this assignment.
   543
   544	    The assignments can look like this:
   545	        x = attr.ib()
   546	        x = y = attr.ib()
   547	        x, y = attr.ib(), attr.ib()
   548	    or if auto_attribs is enabled also like this:
   549	        x: type
   550	        x: type = default_value
   551	        x: type = attr.ib(...)
   552	    """
   553	    for lvalue in stmt.lvalues:
   554	        lvalues, rvalues = _parse_assignments(lvalue, stmt)
   555
   556	        if len(lvalues) != len(rvalues):
   557	            # This means we have some assignment that isn't 1 to 1.
   558	            # It can't be an attrib.
   559	            continue
   560
   561	        for lhs, rvalue in zip(lvalues, rvalues):
   562	            # Check if the right hand side is a call to an attribute maker.
   563	            if (
   564	                isinstance(rvalue, CallExpr)
   565	                and isinstance(rvalue.callee, RefExpr)
   566	                and rvalue.callee.fullname in attr_attrib_makers
   567	            ):
   568	                attr = _attribute_from_attrib_maker(ctx, auto_attribs, kw_only, lhs, rvalue, stmt)
   569	                if attr:
   570	                    yield attr
   571	            elif auto_attribs and stmt.type and stmt.new_syntax and not is_class_var(lhs):
   572	                yield _attribute_from_auto_attrib(ctx, kw_only, lhs, rvalue, stmt)

   575	def _cleanup_decorator(stmt: Decorator, attr_map: dict[str, Attribute]) -> None:
   576	    """Handle decorators in class bodies.
   577
   578	    `x.default` will set a default value on x
   579	    `x.validator` and `x.default` will get removed to avoid throwing a type error.
   580	    """
   581	    remove_me = []
   582	    for func_decorator in stmt.decorators:
   583	        if (
   584	            isinstance(func_decorator, MemberExpr)
   585	            and isinstance(func_decorator.expr, NameExpr)
   586	            and func_decorator.expr.name in attr_map
   587	        ):
   588	            if func_decorator.name == "default":
   589	                attr_map[func_decorator.expr.name].has_default = True
   590
   591	            if func_decorator.name in ("default", "validator"):
   592	                # These are decorators on the attrib object that only exist during
   593	                # class creation time.  In order to not trigger a type error later we
   594	                # just remove them.  This might leave us with a Decorator with no
   595	                # decorators (Emperor's new clothes?)
   596	                # TODO: It would be nice to type-check these rather than remove them.
   597	                #       default should be Callable[[], T]
   598	                #       validator should be Callable[[Any, 'Attribute', T], Any]
   599	                #       where T is the type of the attribute.
   600	                remove_me.append(func_decorator)
   601	    for dec in remove_me:
   602	        stmt.decorators.remove(dec)

   621	def _attribute_from_attrib_maker(
   622	    ctx: mypy.plugin.ClassDefContext,
   623	    auto_attribs: bool,
   624	    kw_only: bool,
   625	    lhs: NameExpr,
   626	    rvalue: CallExpr,
   627	    stmt: AssignmentStmt,
   628	) -> Attribute | None:
   629	    """Return an Attribute from the assignment or None if you can't make one."""
   630	    if auto_attribs and not stmt.new_syntax:
   631	        # auto_attribs requires an annotation on *every* attr.ib.
   632	        assert lhs.node is not None
   633	        ctx.api.msg.need_annotation_for_var(lhs.node, stmt)
   634	        return None
   635
   636	    if len(stmt.lvalues) > 1:
   637	        ctx.api.fail("Too many names for one attribute", stmt)
   638	        return None
   639
   640	    # This is the type that belongs in the __init__ method for this attrib.
   641	    init_type = stmt.type
   642
   643	    # Read all the arguments from the call.
   644	    init = _get_bool_argument(ctx, rvalue, "init", True)
   645	    # Note: If the class decorator says kw_only=True the attribute is ignored.
   646	    # See https://github.com/python-attrs/attrs/issues/481 for explanation.
   647	    kw_only |= _get_bool_argument(ctx, rvalue, "kw_only", False)
   648
   649	    # TODO: Check for attr.NOTHING
   650	    attr_has_default = bool(_get_argument(rvalue, "default"))
   651	    attr_has_factory = bool(_get_argument(rvalue, "factory"))
   652
   653	    if attr_has_default and attr_has_factory:
   654	        ctx.api.fail('Can\'t pass both "default" and "factory".', rvalue)
   655	    elif attr_has_factory:
   656	        attr_has_default = True
   657
   658	    # If the type isn't set through annotation but is passed through `type=` use that.
   659	    type_arg = _get_argument(rvalue, "type")
   660	    if type_arg and not init_type:
   661	        try:
   662	            un_type = expr_to_unanalyzed_type(type_arg, ctx.api.options, ctx.api.is_stub_file)
   663	        except TypeTranslationError:
   664	            ctx.api.fail("Invalid argument to type", type_arg)
   665	        else:
   666	            init_type = ctx.api.anal_type(un_type)
   667	            if init_type and isinstance(lhs.node, Var) and not lhs.node.type:
   668	                # If there is no annotation, add one.
   669	                lhs.node.type = init_type
   670	                lhs.is_inferred_def = False
   671
   672	    # Note: convert is deprecated but works the same as converter.
   673	    converter = _get_argument(rvalue, "converter")
   674	    convert = _get_argument(rvalue, "convert")
   675	    if convert and converter:
   676	        ctx.api.fail('Can\'t pass both "convert" and "converter".', rvalue)
   677	    elif convert:
   678	        ctx.api.fail("convert is deprecated, use converter", rvalue)
   679	        converter = convert
   680	    converter_info = _parse_converter(ctx, converter)
   681
   682	    # Custom alias might be defined:
   683	    alias = None
   684	    alias_expr = _get_argument(rvalue, "alias")
   685	    if alias_expr:
   686	        alias = ctx.api.parse_str_literal(alias_expr)
   687	        if alias is None:
   688	            ctx.api.fail(
   689	                '"alias" argument to attrs field must be a string literal',
   690	                rvalue,
   691	                code=LITERAL_REQ,
   692	            )
   693	    name = unmangle(lhs.name)
   694	    return Attribute(
   695	        name, alias, ctx.cls.info, attr_has_default, init, kw_only, converter_info, stmt, init_type
   696	    )

   699	def _parse_converter(
   700	    ctx: mypy.plugin.ClassDefContext, converter_expr: Expression | None
   701	) -> Converter | None:
   702	    """Return the Converter object from an Expression."""
   703	    # TODO: Support complex converters, e.g. lambdas, calls, etc.
   704	    if not converter_expr:
   705	        return None
   706	    converter_info = Converter()
   707	    if (
   708	        isinstance(converter_expr, CallExpr)
   709	        and isinstance(converter_expr.callee, RefExpr)
   710	        and converter_expr.callee.fullname in attr_optional_converters
   711	        and converter_expr.args
   712	        and converter_expr.args[0]
   713	    ):
   714	        # Special handling for attr.converters.optional(type)
   715	        # We extract the type and add make the init_args Optional in Attribute.argument
   716	        converter_expr = converter_expr.args[0]
   717	        is_attr_converters_optional = True
   718	    else:
   719	        is_attr_converters_optional = False
   720
   721	    converter_type: Type | None = None
   722	    if isinstance(converter_expr, RefExpr) and converter_expr.node:
   723	        if isinstance(converter_expr.node, FuncDef):
   724	            if converter_expr.node.type and isinstance(converter_expr.node.type, FunctionLike):
   725	                converter_type = converter_expr.node.type
   726	            else:  # The converter is an unannotated function.
   727	                converter_info.init_type = AnyType(TypeOfAny.unannotated)
   728	                return converter_info
   729	        elif isinstance(converter_expr.node, OverloadedFuncDef) and is_valid_overloaded_converter(
   730	            converter_expr.node
   731	        ):
   732	            converter_type = converter_expr.node.type
   733	        elif isinstance(converter_expr.node, TypeInfo):
   734	            converter_type = type_object_type(converter_expr.node, ctx.api.named_type)
   735	    elif (
   736	        isinstance(converter_expr, IndexExpr)
   737	        and isinstance(converter_expr.analyzed, TypeApplication)
   738	        and isinstance(converter_expr.base, RefExpr)
   739	        and isinstance(converter_expr.base.node, TypeInfo)
   740	    ):
   741	        # The converter is a generic type.
   742	        converter_type = type_object_type(converter_expr.base.node, ctx.api.named_type)
   743	        if isinstance(converter_type, CallableType):
   744	            converter_type = apply_generic_arguments(
   745	                converter_type,
   746	                converter_expr.analyzed.types,
   747	                ctx.api.msg.incompatible_typevar_value,
   748	                converter_type,
   749	            )
   750	        else:
   751	            converter_type = None
   752
   753	    if isinstance(converter_expr, LambdaExpr):
   754	        # TODO: should we send a fail if converter_expr.min_args > 1?
   755	        converter_info.init_type = AnyType(TypeOfAny.unannotated)
   756	        return converter_info
   757
   758	    if not converter_type:
   759	        # Signal that we have an unsupported converter.
   760	        ctx.api.fail(
   761	            "Unsupported converter, only named functions, types and lambdas are currently "
   762	            "supported",
   763	            converter_expr,
   764	        )
   765	        converter_info.init_type = AnyType(TypeOfAny.from_error)
   766	        return converter_info
   767
   768	    converter_type = get_proper_type(converter_type)
   769	    if isinstance(converter_type, CallableType) and converter_type.arg_types:
   770	        converter_info.init_type = converter_type.arg_types[0]
   771	        if not is_attr_converters_optional:
   772	            converter_info.ret_type = converter_type.ret_type
   773	    elif isinstance(converter_type, Overloaded):
   774	        types: list[Type] = []
   775	        for item in converter_type.items:
   776	            # Walk the overloads looking for methods that can accept one argument.
   777	            num_arg_types = len(item.arg_types)
   778	            if not num_arg_types:
   779	                continue
   780	            if num_arg_types > 1 and any(kind == ARG_POS for kind in item.arg_kinds[1:]):
   781	                continue
   782	            types.append(item.arg_types[0])
   783	        # Make a union of all the valid types.
   784	        if types:
   785	            converter_info.init_type = make_simplified_union(types)
   786
   787	    if is_attr_converters_optional and converter_info.init_type:
   788	        # If the converter was attr.converter.optional(type) then add None to
   789	        # the allowed init_type.
   790	        converter_info.init_type = UnionType.make_union([converter_info.init_type, NoneType()])
   791
   792	    return converter_info
   793

   803	    lvalue: Expression, stmt: AssignmentStmt
   804	) -> tuple[list[NameExpr], list[Expression]]:
   805	    """Convert a possibly complex assignment expression into lists of lvalues and rvalues."""
   806	    lvalues: list[NameExpr] = []
   807	    rvalues: list[Expression] = []
   808	    if isinstance(lvalue, (TupleExpr, ListExpr)):
   809	        if all(isinstance(item, NameExpr) for item in lvalue.items):
   810	            lvalues = cast(list[NameExpr], lvalue.items)
   811	        if isinstance(stmt.rvalue, (TupleExpr, ListExpr)):
   812	            rvalues = stmt.rvalue.items
   813	    elif isinstance(lvalue, NameExpr):
   814	        lvalues = [lvalue]
   815	        rvalues = [stmt.rvalue]
   816	    return lvalues, rvalues

   819	def _add_order(ctx: mypy.plugin.ClassDefContext, adder: MethodAdder) -> None:
   820	    """Generate all the ordering methods for this class."""
   821	    bool_type = ctx.api.named_type("builtins.bool")
   822	    object_type = ctx.api.named_type("builtins.object")
   823	    # Make the types be:
   824	    #    AT = TypeVar('AT')
   825	    #    def __lt__(self: AT, other: AT) -> bool
   826	    # This way comparisons with subclasses will work correctly.
   827	    fullname = f"{ctx.cls.info.fullname}.{SELF_TVAR_NAME}"
   828	    tvd = TypeVarType(
   829	        SELF_TVAR_NAME,
   830	        fullname,
   831	        # Namespace is patched per-method below.
   832	        id=TypeVarId(-1, namespace=""),
   833	        values=[],
   834	        upper_bound=object_type,
   835	        default=AnyType(TypeOfAny.from_omitted_generics),
   836	    )
   837	    self_tvar_expr = TypeVarExpr(
   838	        SELF_TVAR_NAME, fullname, [], object_type, AnyType(TypeOfAny.from_omitted_generics)
   839	    )
   840	    ctx.cls.info.names[SELF_TVAR_NAME] = SymbolTableNode(MDEF, self_tvar_expr)
   841
   842	    for method in ["__lt__", "__le__", "__gt__", "__ge__"]:
   843	        namespace = f"{ctx.cls.info.fullname}.{method}"
   844	        tvd = tvd.copy_modified(id=TypeVarId(tvd.id.raw_id, namespace=namespace))
   845	        args = [Argument(Var("other", tvd), tvd, None, ARG_POS)]
   846	        adder.add_method(method, args, bool_type, self_type=tvd, tvd=tvd)

   849	def _make_frozen(ctx: mypy.plugin.ClassDefContext, attributes: list[Attribute]) -> None:
   850	    """Turn all the attributes into properties to simulate frozen classes."""
   851	    for attribute in attributes:
   852	        if attribute.name in ctx.cls.info.names:
   853	            # This variable belongs to this class so we can modify it.
   854	            node = ctx.cls.info.names[attribute.name].node
   855	            if not isinstance(node, Var):
   856	                # The superclass attribute was overridden with a non-variable.
   857	                # No need to do anything here, override will be verified during
   858	                # type checking.
   859	                continue
   860	            node.is_property = True
   861	        else:
   862	            # This variable belongs to a super class so create new Var so we
   863	            # can modify it.
   864	            var = Var(attribute.name, attribute.init_type)
   865	            var.info = ctx.cls.info
   866	            var._fullname = f"{ctx.cls.info.fullname}.{var.name}"
   867	            ctx.cls.info.names[var.name] = SymbolTableNode(MDEF, var)
   868	            var.is_property = True

   871	def _add_init(
   872	    ctx: mypy.plugin.ClassDefContext,
   873	    attributes: list[Attribute],
   874	    adder: MethodAdder,
   875	    method_name: Literal["__init__", "__attrs_init__"],
   876	) -> None:
   877	    """Generate an __init__ method for the attributes and add it to the class."""
   878	    # Convert attributes to arguments with kw_only arguments at the end of
   879	    # the argument list
   880	    pos_args = []
   881	    kw_only_args = []
   882	    sym_table = ctx.cls.info.names
   883	    for attribute in attributes:
   884	        if not attribute.init:
   885	            continue
   886	        if attribute.kw_only:
   887	            kw_only_args.append(attribute.argument(ctx))
   888	        else:
   889	            pos_args.append(attribute.argument(ctx))
   890
   891	        # If the attribute is Final, present in `__init__` and has
   892	        # no default, make sure it doesn't error later.
   893	        if not attribute.has_default and attribute.name in sym_table:
   894	            sym_node = sym_table[attribute.name].node
   895	            if isinstance(sym_node, Var) and sym_node.is_final:
   896	                sym_node.final_set_in_init = True
   897	    args = pos_args + kw_only_args
   898	    if all(
   899	        # We use getattr rather than instance checks because the variable.type
   900	        # might be wrapped into a Union or some other type, but even non-Any
   901	        # types reliably track the fact that the argument was not annotated.
   902	        getattr(arg.variable.type, "type_of_any", None) == TypeOfAny.unannotated
   903	        for arg in args
   904	    ):
   905	        # This workaround makes --disallow-incomplete-defs usable with attrs,
   906	        # but is definitely suboptimal as a long-term solution.
   907	        # See https://github.com/python/mypy/issues/5954 for discussion.
   908	        for a in args:
   909	            a.variable.type = AnyType(TypeOfAny.implementation_artifact)
   910	            a.type_annotation = AnyType(TypeOfAny.implementation_artifact)
   911	    adder.add_method(method_name, args, NoneType())

  1057	def _get_expanded_attr_types(
  1058	    ctx: mypy.plugin.FunctionSigContext,
  1059	    typ: ProperType,
  1060	    display_typ: ProperType,
  1061	    parent_typ: ProperType,
  1062	) -> list[Mapping[str, Type]] | None:
  1063	    """
  1064	    For a given type, determine what attrs classes it can be: for each class, return the field types.
  1065	    For generic classes, the field types are expanded.
  1066	    If the type contains Any or a non-attrs type, returns None; in the latter case, also reports an error.
  1067	    """
  1068	    if isinstance(typ, AnyType):
  1069	        return None
  1070	    elif isinstance(typ, UnionType):
  1071	        ret: list[Mapping[str, Type]] | None = []
  1072	        for item in typ.relevant_items():
  1073	            item = get_proper_type(item)
  1074	            item_types = _get_expanded_attr_types(ctx, item, item, parent_typ)
  1075	            if ret is not None and item_types is not None:
  1076	                ret += item_types
  1077	            else:
  1078	                ret = None  # but keep iterating to emit all errors
  1079	        return ret
  1080	    elif isinstance(typ, TypeVarType):
  1081	        return _get_expanded_attr_types(
  1082	            ctx, get_proper_type(typ.upper_bound), display_typ, parent_typ
  1083	        )
  1084	    elif isinstance(typ, Instance):
  1085	        init_func = _get_attrs_init_type(typ)
  1086	        if init_func is None:
  1087	            _fail_not_attrs_class(ctx, display_typ, parent_typ)
  1088	            return None
  1089	        init_func = expand_type_by_instance(init_func, typ)
  1090	        # [1:] to skip the self argument of AttrClass.__init__
  1091	        field_names = cast(list[str], init_func.arg_names[1:])
  1092	        field_types = init_func.arg_types[1:]
  1093	        return [dict(zip(field_names, field_types))]
  1094	    else:
  1095	        _fail_not_attrs_class(ctx, display_typ, parent_typ)
  1096	        return None

   101	def find_shallow_matching_overload_item(overload: Overloaded, call: CallExpr) -> CallableType:
   102	    """Perform limited lookup of a matching overload item.
   103
   104	    Full overload resolution is only supported during type checking, but plugins
   105	    sometimes need to resolve overloads. This can be used in some such use cases.
   106
   107	    Resolve overloads based on these things only:
   108
   109	    * Match using argument kinds and names
   110	    * If formal argument has type None, only accept the "None" expression in the callee
   111	    * If formal argument has type Literal[True] or Literal[False], only accept the
   112	      relevant bool literal
   113
   114	    Return the first matching overload item, or the last one if nothing matches.
   115	    """
   116	    for item in overload.items[:-1]:
   117	        ok = True
   118	        mapped = map_actuals_to_formals(
   119	            call.arg_kinds,
   120	            call.arg_names,
   121	            item.arg_kinds,
   122	            item.arg_names,
   123	            lambda i: AnyType(TypeOfAny.special_form),
   124	        )
   125
   126	        # Look for extra actuals
   127	        matched_actuals = set()
   128	        for actuals in mapped:
   129	            matched_actuals.update(actuals)
   130	        if any(i not in matched_actuals for i in range(len(call.args))):
   131	            ok = False
   132
   133	        for arg_type, kind, actuals in zip(item.arg_types, item.arg_kinds, mapped):
   134	            if kind.is_required() and not actuals:
   135	                # Missing required argument
   136	                ok = False
   137	                break
   138	            elif actuals:
   139	                args = [call.args[i] for i in actuals]
   140	                arg_type = get_proper_type(arg_type)
   141	                arg_none = any(isinstance(arg, NameExpr) and arg.name == "None" for arg in args)
   142	                if isinstance(arg_type, NoneType):
   143	                    if not arg_none:
   144	                        ok = False
   145	                        break
   146	                elif (
   147	                    arg_none
   148	                    and not is_overlapping_none(arg_type)
   149	                    and not (
   150	                        isinstance(arg_type, Instance)
   151	                        and arg_type.type.fullname == "builtins.object"
   152	                    )
   153	                    and not isinstance(arg_type, AnyType)
   154	                ):
   155	                    ok = False

   158	                    if not any(parse_bool(arg) == arg_type.value for arg in args):
   159	                        ok = False
   160	                        break

    58	        t = get_proper_type(t)
    59	        # Every type can be converted from itself (obviously).
    60	        allowed_types.append(t)
    61	        if isinstance(t, Instance):
    62	            unboxed = _find_simplecdata_base_arg(t, api)
    63	            if unboxed is not None:
    64	                # If _SimpleCData appears in tp's (direct or indirect) bases, its type argument
    65	                # specifies the type's "unboxed" version, which can always be converted back to
    66	                # the original "boxed" type.
    67	                allowed_types.append(unboxed)
    68
    69	                if t.type.has_base("ctypes._PointerLike"):
    70	                    # Pointer-like _SimpleCData subclasses can also be converted from
    71	                    # an int or None.
    72	                    allowed_types.append(api.named_generic_type("builtins.int", []))
    73	                    allowed_types.append(NoneType())
    74
    75	    return make_simplified_union(allowed_types)

    78	def _autounboxed_cdata(tp: Type) -> ProperType:
    79	    """Get the auto-unboxed version of a CData type, if applicable.
    80
    81	    For *direct* _SimpleCData subclasses, the only type argument of _SimpleCData in the bases list
    82	    is returned.
    83	    For all other CData types, including indirect _SimpleCData subclasses, tp is returned as-is.
    84	    """
    85	    tp = get_proper_type(tp)
    86
    87	    if isinstance(tp, UnionType):
    88	        return make_simplified_union([_autounboxed_cdata(t) for t in tp.items])
    89	    elif isinstance(tp, Instance):
    90	        for base in tp.type.bases:
    91	            if base.type.fullname == "_ctypes._SimpleCData":
    92	                # If tp has _SimpleCData as a direct base class,
    93	                # the auto-unboxed type is the single type argument of the _SimpleCData type.
    94	                assert len(base.args) == 1
    95	                return get_proper_type(base.args[0])
    96	    # If tp is not a concrete type, or if there is no _SimpleCData in the bases,
    97	    # the type is not auto-unboxed.
    98	    return tp

   111	def array_constructor_callback(ctx: mypy.plugin.FunctionContext) -> Type:
   112	    """Callback to provide an accurate signature for the ctypes.Array constructor."""
   113	    # Extract the element type from the constructor's return type, i. e. the type of the array
   114	    # being constructed.
   115	    et = _get_array_element_type(ctx.default_return_type)
   116	    if et is not None:
   117	        allowed = _autoconvertible_to_cdata(et, ctx.api)
   118	        assert (
   119	            len(ctx.arg_types) == 1
   120	        ), "The stub of the ctypes.Array constructor should have a single vararg parameter"
   121	        for arg_num, (arg_kind, arg_type) in enumerate(zip(ctx.arg_kinds[0], ctx.arg_types[0]), 1):
   122	            if arg_kind == nodes.ARG_POS and not is_subtype(arg_type, allowed):
   123	                ctx.api.msg.fail(
   124	                    "Array constructor argument {} of type {}"
   125	                    " is not convertible to the array element type {}".format(
   126	                        arg_num,
   127	                        format_type(arg_type, ctx.api.options),
   128	                        format_type(et, ctx.api.options),
   129	                    ),
   130	                    ctx.context,
   131	                )
   132	            elif arg_kind == nodes.ARG_STAR:
   133	                ty = ctx.api.named_generic_type("typing.Iterable", [allowed])
   134	                if not is_subtype(arg_type, ty):
   135	                    it = ctx.api.named_generic_type("typing.Iterable", [et])
   136	                    ctx.api.msg.fail(
   137	                        "Array constructor argument {} of type {}"
   138	                        " is not convertible to the array element type {}".format(
   139	                            arg_num,
   140	                            format_type(arg_type, ctx.api.options),
   141	                            format_type(it, ctx.api.options),
   142	                        ),
   143	                        ctx.context,
   144	                    )
   145
   146	    return ctx.default_return_type
