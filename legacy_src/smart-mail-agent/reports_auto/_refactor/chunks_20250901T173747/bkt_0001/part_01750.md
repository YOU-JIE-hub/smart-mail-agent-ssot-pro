   181	    data is a sequence of series

   183	    where x & y are ints or floats

   196	    #find all line segments

   201	        if not ds: continue
   202	        n = len(ds)

   417	    def _bezierArcSegmentCCW(self, cx,cy, rx,ry, theta0, theta1):
   418	        """compute the control points for a bezier arc with theta1-theta0 <= 90.
   419	        Points are computed for an arc with angle theta increasing in the
   420	        counter-clockwise (CCW) direction.  returns a tuple with starting point
   421	        and 3 control points of a cubic bezier curve for the curvto opertator"""
   422
   423	        # Requires theta1 - theta0 <= 90 for a good approximation
   424	        assert abs(theta1 - theta0) <= 90
   425	        cos0 = cos(pi*theta0/180.0)
   426	        sin0 = sin(pi*theta0/180.0)
   427	        x0 = cx + rx*cos0
   428	        y0 = cy + ry*sin0
   429
   430	        cos1 = cos(pi*theta1/180.0)
   431	        sin1 = sin(pi*theta1/180.0)
   432
   433	        x3 = cx + rx*cos1
   434	        y3 = cy + ry*sin1
   435
   436	        dx1 = -rx * sin0
   437	        dy1 = ry * cos0
   438
   439	        #from pdfgeom
   440	        halfAng = pi*(theta1-theta0)/(2.0 * 180.0)
   441	        k = abs(4.0 / 3.0 * (1.0 - cos(halfAng) ) /(sin(halfAng)) )
   442	        x1 = x0 + dx1 * k
   443	        y1 = y0 + dy1 * k
   444
   445	        dx2 = -rx * sin1
   446	        dy2 = ry * cos1
   447
   448	        x2 = x3 - dx2 * k
   449	        y2 = y3 - dy2 * k
   450	        return ((x0,y0), ((x1,y1), (x2,y2), (x3,y3)) )

   452	    def bezierArcCCW(self, cx,cy, rx,ry, theta0, theta1):
   453	        """return a set of control points for Bezier approximation to an arc
   454	        with angle increasing counter clockwise. No requirement on (theta1-theta0) <= 90
   455	        However, it must be true that theta1-theta0 > 0."""
   456
   457	        # I believe this is also clockwise
   458	        # pretty much just like Robert Kern's pdfgeom.BezierArc
   459	        angularExtent = theta1 - theta0
   460	        # break down the arc into fragments of <=90 degrees
   461	        if abs(angularExtent) <= 90.0:  # we just need one fragment
   462	            angleList = [(theta0,theta1)]
   463	        else:
   464	            Nfrag = int( ceil( abs(angularExtent)/90.) )
   465	            fragAngle = float(angularExtent)/ Nfrag  # this could be negative
   466	            angleList = []
   467	            for ii in range(Nfrag):
   468	                a = theta0 + ii * fragAngle
   469	                b = a + fragAngle # hmm.. is I wonder if this is precise enought
   470	                angleList.append((a,b))
   471
   472	        ctrlpts = []
   473	        for (a,b) in angleList:
   474	            if not ctrlpts: # first time
   475	                [(x0,y0), pts] = self._bezierArcSegmentCCW(cx,cy, rx,ry, a,b)
   476	                ctrlpts.append(pts)
   477	            else:
   478	                [(tmpx,tmpy), pts] = self._bezierArcSegmentCCW(cx,cy, rx,ry, a,b)
   479	                ctrlpts.append(pts)
   480	        return ((x0,y0), ctrlpts)

   588	        """add closed path ellipse with center cx,cy and axes rx,ry: counter-clockwise orientation
   589	        (remember y-axis increases downward) """
   590	        self.pathBegin()
   591	        # first segment
   592	        x0 = cx + rx   # (x0,y0) start pt
   593	        y0 = cy
   594
   595	        x3 = cx        # (x3,y3) end pt of arc
   596	        y3 = cy-ry
   597
   598	        x1 = cx+rx
   599	        y1 = cy-ry*BEZIER_ARC_MAGIC
   600
   601	        x2 = x3 + rx*BEZIER_ARC_MAGIC
   602	        y2 = y3
   603	        self.moveTo(x0, y0)
   604	        self.curveTo(x1,y1,x2,y2,x3,y3)
   605	        # next segment
   606	        x0 = x3
   607	        y0 = y3
   608
   609	        x3 = cx-rx
   610	        y3 = cy
   611
   612	        x1 = cx-rx*BEZIER_ARC_MAGIC
   613	        y1 = cy-ry
   614
   615	        x2 = x3
   616	        y2 = cy- ry*BEZIER_ARC_MAGIC
   617	        self.curveTo(x1,y1,x2,y2,x3,y3)
   618	        # next segment
   619	        x0 = x3
   620	        y0 = y3
   621
   622	        x3 = cx
   623	        y3 = cy+ry
   624
   625	        x1 = cx-rx
   626	        y1 = cy+ry*BEZIER_ARC_MAGIC
   627
   628	        x2 = cx -rx*BEZIER_ARC_MAGIC
   629	        y2 = cy+ry
   630	        self.curveTo(x1,y1,x2,y2,x3,y3)
   631	        #last segment
   632	        x0 = x3
   633	        y0 = y3
   634
   635	        x3 = cx+rx
   636	        y3 = cy
   637
   638	        x1 = cx+rx*BEZIER_ARC_MAGIC
   639	        y1 = cy+ry
   640

   357	    def roundRect(self, x1,y1, x2,y2, rx=8, ry=8):
   358	        """Draw a rounded rectangle between x1,y1, and x2,y2,
   359	        with corners inset as ellipses with x radius rx and y radius ry.
   360	        These should have x1<x2, y1<y2, rx>0, and ry>0."""
   361	        # Path is drawn in counter-clockwise direction
   362
   363	        x1, x2 = min(x1,x2), max(x1, x2) # from piddle.py
   364	        y1, y2 = min(y1,y2), max(y1, y2)
   365
   366	        # Note: arcto command draws a line from current point to beginning of arc
   367	        # save current matrix, translate to center of ellipse, scale by rx ry, and draw
   368	        # a circle of unit radius in counterclockwise dir, return to original matrix
   369	        # arguments are (cx, cy, rx, ry, startAngle, endAngle)
   370	        ellipsePath = 'matrix currentmatrix %s %s translate %s %s scale 0 0 1 %s %s arc setmatrix'
   371
   372	        # choice between newpath and moveTo beginning of arc
   373	        # go with newpath for precision, does this violate any assumptions in code???
   374	        rr = ['newpath'] # Round Rect code path
   375	        a = rr.append
   376	        # upper left corner ellipse is first
   377	        a(ellipsePath % (x1+rx, y1+ry, rx, -ry, 90, 180))
   378	        a(ellipsePath % (x1+rx, y2-ry, rx, -ry, 180, 270))
   379	        a(ellipsePath % (x2-rx, y2-ry, rx, -ry, 270, 360))
   380	        a(ellipsePath % (x2-rx, y1+ry, rx, -ry, 0,  90) )
   381	        a('closepath')
   382
   383	        self._fillAndStroke(rr)

   394	    def drawArc(self, x1,y1, x2,y2, startAng=0, extent=360, fromcenter=0):
   395	        """Draw a partial ellipse inscribed within the rectangle x1,y1,x2,y2,
   396	        starting at startAng degrees and covering extent degrees.   Angles
   397	        start with 0 to the right (+x) and increase counter-clockwise.
   398	        These should have x1<x2 and y1<y2."""
   399	        #calculate centre of ellipse
   400	        #print "x1,y1,x2,y2,startAng,extent,fromcenter", x1,y1,x2,y2,startAng,extent,fromcenter
   401	        cx, cy = (x1+x2)/2.0, (y1+y2)/2.0
   402	        rx, ry = (x2-x1)/2.0, (y2-y1)/2.0
   403
   404	        codeline = self._genArcCode(x1, y1, x2, y2, startAng, extent)
   405
   406	        startAngleRadians = math.pi*startAng/180.0
   407	        extentRadians = math.pi*extent/180.0
   408	        endAngleRadians = startAngleRadians + extentRadians
   409
   410	        codelineAppended = 0
   411
   412	        # fill portion
   413
   414	        if self._fillColor != None:
   415	            self.setColor(self._fillColor)
   416	            self.code_append(codeline)
   417	            codelineAppended = 1
   418	            if self._strokeColor!=None: self.code_append('gsave')
   419	            self.lineTo(cx,cy)
   420	            self.code_append('eofill')
   421	            if self._strokeColor!=None: self.code_append('grestore')
   422
   423	        # stroke portion
   424	        if self._strokeColor != None:
   425	            # this is a bit hacked up.  There is certainly a better way...
   426	            self.setColor(self._strokeColor)
   427	            (startx, starty) = (cx+rx*math.cos(startAngleRadians), cy+ry*math.sin(startAngleRadians))
   428	            if not codelineAppended:
   429	                self.code_append(codeline)
   430	            if fromcenter:
   431	                # move to center
   432	                self.lineTo(cx,cy)
   433	                self.lineTo(startx, starty)
   434	                self.code_append('closepath')
   435	            self.code_append('stroke')

   437	    def _genArcCode(self, x1, y1, x2, y2, startAng, extent):
   438	        "Calculate the path for an arc inscribed in rectangle defined by (x1,y1),(x2,y2)"
   439	        #calculate semi-minor and semi-major axes of ellipse
   440	        xScale = abs((x2-x1)/2.0)
   441	        yScale = abs((y2-y1)/2.0)
   442	        #calculate centre of ellipse
   443	        x, y = (x1+x2)/2.0, (y1+y2)/2.0
   444
   445	        codeline = 'matrix currentmatrix %s %s translate %s %s scale 0 0 1 %s %s %s setmatrix'
   446
   447	        if extent >= 0:
   448	            arc='arc'
   449	        else:
   450	            arc='arcn'
   451	        data = (x,y, xScale, yScale, startAng, startAng+extent, arc)
   452
   453	        return codeline % data

   731	        """This is the recursive method called for each node
   732	        in the tree"""
   733	        self._canvas.comment('begin node %r'%node)
   734	        color = self._canvas._color
   735	        if not (isinstance(node, Path) and node.isClipPath):

   741	        self.applyStateChanges(deltas, {})
