  1639	        and left.value in (0, 1)

  1645	    if isinstance(right, mypy.types.TypedDictType) and mypy.types.is_named_instance(
  1646	        left, "builtins.dict"
  1647	    ):
  1648	        # Special case checks against TypedDicts
  1649	        return True

  1651	    with mypy.state.state.strict_optional_set(True):
  1652	        return mypy.subtypes.is_subtype(left, right)

  1655	def get_mypy_type_of_runtime_value(runtime: Any) -> mypy.types.Type | None:
  1656	    """Returns a mypy type object representing the type of ``runtime``.
  1657
  1658	    Returns None if we can't find something that works.
  1659
  1660	    """
  1661	    if runtime is None:
  1662	        return mypy.types.NoneType()
  1663	    if isinstance(runtime, property):
  1664	        # Give up on properties to avoid issues with things that are typed as attributes.
  1665	        return None
  1666
  1667	    def anytype() -> mypy.types.AnyType:
  1668	        return mypy.types.AnyType(mypy.types.TypeOfAny.unannotated)
  1669
  1670	    if isinstance(
  1671	        runtime,
  1672	        (types.FunctionType, types.BuiltinFunctionType, types.MethodType, types.BuiltinMethodType),
  1673	    ):
  1674	        builtins = get_stub("builtins")
  1675	        assert builtins is not None
  1676	        type_info = builtins.names["function"].node
  1677	        assert isinstance(type_info, nodes.TypeInfo)
  1678	        fallback = mypy.types.Instance(type_info, [anytype()])
  1679	        signature = safe_inspect_signature(runtime)
  1680	        if signature:
  1681	            arg_types = []
  1682	            arg_kinds = []
  1683	            arg_names = []
  1684	            for arg in signature.parameters.values():
  1685	                arg_types.append(anytype())
  1686	                arg_names.append(
  1687	                    None if arg.kind == inspect.Parameter.POSITIONAL_ONLY else arg.name
  1688	                )
  1689	                no_default = arg.default is inspect.Parameter.empty
  1690	                if arg.kind == inspect.Parameter.POSITIONAL_ONLY:
  1691	                    arg_kinds.append(nodes.ARG_POS if no_default else nodes.ARG_OPT)
  1692	                elif arg.kind == inspect.Parameter.POSITIONAL_OR_KEYWORD:
  1693	                    arg_kinds.append(nodes.ARG_POS if no_default else nodes.ARG_OPT)
  1694	                elif arg.kind == inspect.Parameter.KEYWORD_ONLY:
  1695	                    arg_kinds.append(nodes.ARG_NAMED if no_default else nodes.ARG_NAMED_OPT)
  1696	                elif arg.kind == inspect.Parameter.VAR_POSITIONAL:
  1697	                    arg_kinds.append(nodes.ARG_STAR)
  1698	                elif arg.kind == inspect.Parameter.VAR_KEYWORD:
  1699	                    arg_kinds.append(nodes.ARG_STAR2)
  1700	                else:
  1701	                    raise AssertionError
  1702	        else:
  1703	            arg_types = [anytype(), anytype()]
  1704	            arg_kinds = [nodes.ARG_STAR, nodes.ARG_STAR2]
  1705	            arg_names = [None, None]
  1706
  1707	        return mypy.types.CallableType(
  1708	            arg_types,
  1709	            arg_kinds,
  1710	            arg_names,
  1711	            ret_type=anytype(),
  1712	            fallback=fallback,
  1713	            is_ellipsis_args=True,
  1714	        )
  1715
  1716	    # Try and look up a stub for the runtime object
  1717	    stub = get_stub(type(runtime).__module__)
  1718	    if stub is None:
  1719	        return None
  1720	    type_name = type(runtime).__name__
  1721	    if type_name not in stub.names:
  1722	        return None
  1723	    type_info = stub.names[type_name].node
  1724	    if isinstance(type_info, nodes.Var):
  1725	        return type_info.type
  1726	    if not isinstance(type_info, nodes.TypeInfo):
  1727	        return None
  1728
  1729	    if isinstance(runtime, tuple):
  1730	        # Special case tuples so we construct a valid mypy.types.TupleType
  1731	        optional_items = [get_mypy_type_of_runtime_value(v) for v in runtime]
  1732	        items = [(i if i is not None else anytype()) for i in optional_items]
  1733	        fallback = mypy.types.Instance(type_info, [anytype()])
  1734	        return mypy.types.TupleType(items, fallback)
  1735
  1736	    fallback = mypy.types.Instance(type_info, [anytype() for _ in type_info.type_vars])
  1737
  1738	    value: bool | int | str
  1739	    if isinstance(runtime, enum.Enum) and isinstance(runtime.name, str):
  1740	        value = runtime.name
  1741	    elif isinstance(runtime, bytes):
  1742	        value = bytes_to_human_readable_repr(runtime)
  1743	    elif isinstance(runtime, (bool, int, str)):
  1744	        value = runtime
  1745	    else:
  1746	        return fallback
  1747
  1748	    return mypy.types.LiteralType(value=value, fallback=fallback)

  1759	def build_stubs(modules: list[str], options: Options, find_submodules: bool = False) -> list[str]:
  1760	    """Uses mypy to construct stub objects for the given modules.
  1761
  1762	    This sets global state that ``get_stub`` can access.
  1763
  1764	    Returns all modules we might want to check. If ``find_submodules`` is False, this is equal
  1765	    to ``modules``.
  1766
  1767	    :param modules: List of modules to build stubs for.
  1768	    :param options: Mypy options for finding and building stubs.
  1769	    :param find_submodules: Whether to attempt to find submodules of the given modules as well.
  1770
  1771	    """
  1772	    data_dir = mypy.build.default_data_dir()
  1773	    search_path = mypy.modulefinder.compute_search_paths([], options, data_dir)
  1774	    find_module_cache = mypy.modulefinder.FindModuleCache(
  1775	        search_path, fscache=None, options=options
  1776	    )
  1777
  1778	    all_modules = []
  1779	    sources = []
  1780	    for module in modules:
  1781	        all_modules.append(module)
  1782	        if not find_submodules:
  1783	            module_path = find_module_cache.find_module(module)
  1784	            if not isinstance(module_path, str):
  1785	                # test_module will yield an error later when it can't find stubs
  1786	                continue
  1787	            sources.append(mypy.modulefinder.BuildSource(module_path, module, None))
  1788	        else:
  1789	            found_sources = find_module_cache.find_modules_recursive(module)
  1790	            sources.extend(found_sources)
  1791	            # find submodules via mypy
  1792	            all_modules.extend(s.module for s in found_sources if s.module not in all_modules)
  1793	            # find submodules via pkgutil
  1794	            try:
  1795	                runtime = silent_import_module(module)
  1796	                all_modules.extend(
  1797	                    m.name
  1798	                    for m in pkgutil.walk_packages(runtime.__path__, runtime.__name__ + ".")
  1799	                    if m.name not in all_modules
  1800	                )
  1801	            except KeyboardInterrupt:
  1802	                raise
  1803	            except BaseException:
  1804	                pass
  1805
  1806	    if sources:
  1807	        try:
  1808	            res = mypy.build.build(sources=sources, options=options)
  1809	        except mypy.errors.CompileError as e:
  1810	            raise StubtestFailure(f"failed mypy compile:\n{e}") from e
  1811	        if res.errors:
  1812	            raise StubtestFailure("mypy build errors:\n" + "\n".join(res.errors))
  1813
  1814	        global _all_stubs
  1815	        _all_stubs = res.files
  1816
  1817	    return all_modules

  1825	def get_typeshed_stdlib_modules(
  1826	    custom_typeshed_dir: str | None, version_info: tuple[int, int] | None = None
  1827	) -> set[str]:
  1828	    """Returns a list of stdlib modules in typeshed (for current Python version)."""
  1829	    stdlib_py_versions = mypy.modulefinder.load_stdlib_py_versions(custom_typeshed_dir)
  1830	    if version_info is None:
  1831	        version_info = sys.version_info[0:2]
  1832
  1833	    def exists_in_version(module: str) -> bool:
  1834	        assert version_info is not None
  1835	        parts = module.split(".")
  1836	        for i in range(len(parts), 0, -1):
  1837	            current_module = ".".join(parts[:i])
  1838	            if current_module in stdlib_py_versions:
  1839	                minver, maxver = stdlib_py_versions[current_module]
  1840	                return version_info >= minver and (maxver is None or version_info <= maxver)
  1841	        return False
  1842
  1843	    if custom_typeshed_dir:
  1844	        typeshed_dir = Path(custom_typeshed_dir)
  1845	    else:
  1846	        typeshed_dir = Path(mypy.build.default_data_dir()) / "typeshed"
  1847	    stdlib_dir = typeshed_dir / "stdlib"
  1848
  1849	    modules: set[str] = set()
  1850	    for path in stdlib_dir.rglob("*.pyi"):
  1851	        if path.stem == "__init__":
  1852	            path = path.parent
  1853	        module = ".".join(path.relative_to(stdlib_dir).parts[:-1] + (path.stem,))
  1854	        if exists_in_version(module):
  1855	            modules.add(module)
  1856	    return modules

  1859	def get_importable_stdlib_modules() -> set[str]:
  1860	    """Return all importable stdlib modules at runtime."""
  1861	    all_stdlib_modules: AbstractSet[str]
  1862	    if sys.version_info >= (3, 10):
  1863	        all_stdlib_modules = sys.stdlib_module_names
  1864	    else:
  1865	        all_stdlib_modules = set(sys.builtin_module_names)
  1866	        modules_by_finder: defaultdict[importlib.machinery.FileFinder, set[str]] = defaultdict(set)
  1867	        for m in pkgutil.iter_modules():
  1868	            if isinstance(m.module_finder, importlib.machinery.FileFinder):
  1869	                modules_by_finder[m.module_finder].add(m.name)
  1870	        for finder, module_group in modules_by_finder.items():
  1871	            if (
  1872	                "site-packages" not in Path(finder.path).parts
  1873	                # if "_queue" is present, it's most likely the module finder
  1874	                # for stdlib extension modules;
  1875	                # if "queue" is present, it's most likely the module finder
  1876	                # for pure-Python stdlib modules.
  1877	                # In either case, we'll want to add all the modules that the finder has to offer us.
  1878	                # This is a bit hacky, but seems to work well in a cross-platform way.
  1879	                and {"_queue", "queue"} & module_group
  1880	            ):
  1881	                all_stdlib_modules.update(module_group)
  1882
  1883	    importable_stdlib_modules: set[str] = set()
  1884	    for module_name in all_stdlib_modules:
  1885	        if module_name in ANNOYING_STDLIB_MODULES:
  1886	            continue
  1887
  1888	        try:
  1889	            runtime = silent_import_module(module_name)
  1890	        except ImportError:
  1891	            continue
  1892	        else:
  1893	            importable_stdlib_modules.add(module_name)
  1894
  1895	        try:
  1896	            # some stdlib modules (e.g. `nt`) don't have __path__ set...
  1897	            runtime_path = runtime.__path__
  1898	            runtime_name = runtime.__name__
  1899	        except AttributeError:
  1900	            continue
  1901
  1902	        for submodule in pkgutil.walk_packages(runtime_path, runtime_name + "."):
  1903	            submodule_name = submodule.name
  1904
  1905	            # There are many annoying *.__main__ stdlib modules,
  1906	            # and including stubs for them isn't really that useful anyway:
  1907	            # tkinter.__main__ opens a tkinter windows; unittest.__main__ raises SystemExit; etc.
  1908	            #
  1909	            # The idlelib.* submodules are similarly annoying in opening random tkinter windows,
  1910	            # and we're unlikely to ever add stubs for idlelib in typeshed
  1911	            # (see discussion in https://github.com/python/typeshed/pull/9193)
  1912	            #
  1913	            # test.* modules do weird things like raising exceptions in __del__ methods,
  1914	            # leading to unraisable exceptions being logged to the terminal
  1915	            # as a warning at the end of the stubtest run
  1916	            if submodule_name.endswith(".__main__") or submodule_name.startswith(
  1917	                ("idlelib.", "test.")
  1918	            ):
  1919	                continue
  1920
  1921	            try:
  1922	                silent_import_module(submodule_name)
  1923	            except KeyboardInterrupt:
  1924	                raise
  1925	            # importing multiprocessing.popen_forkserver on Windows raises AttributeError...
  1926	            # some submodules also appear to raise SystemExit as well on some Python versions
  1927	            # (not sure exactly which)
  1928	            except BaseException:
  1929	                continue
  1930	            else:
  1931	                importable_stdlib_modules.add(submodule_name)
  1932
  1933	    return importable_stdlib_modules

  1970	def test_stubs(args: _Arguments, use_builtins_fixtures: bool = False) -> int:
  1971	    """This is stubtest! It's time to test the stubs!"""
  1972	    # Load the allowlist. This is a series of strings corresponding to Error.object_desc
  1973	    # Values in the dict will store whether we used the allowlist entry or not.
  1974	    allowlist = {
  1975	        entry: False
  1976	        for allowlist_file in args.allowlist
  1977	        for entry in get_allowlist_entries(allowlist_file)
  1978	    }
  1979	    allowlist_regexes = {entry: re.compile(entry) for entry in allowlist}
  1980
  1981	    # If we need to generate an allowlist, we store Error.object_desc for each error here.
  1982	    generated_allowlist = set()
  1983
  1984	    modules = args.modules
  1985	    if args.check_typeshed:
  1986	        if args.modules:
  1987	            print(
  1988	                _style("error:", color="red", bold=True),
  1989	                "cannot pass both --check-typeshed and a list of modules",
  1990	            )
  1991	            return 1
  1992	        typeshed_modules = get_typeshed_stdlib_modules(args.custom_typeshed_dir)
  1993	        runtime_modules = get_importable_stdlib_modules()
  1994	        modules = sorted((typeshed_modules | runtime_modules) - ANNOYING_STDLIB_MODULES)
  1995
  1996	    if not modules:
  1997	        print(_style("error:", color="red", bold=True), "no modules to check")
  1998	        return 1
  1999
  2000	    options = Options()
  2001	    options.incremental = False
  2002	    options.custom_typeshed_dir = args.custom_typeshed_dir
  2003	    if options.custom_typeshed_dir:
  2004	        options.abs_custom_typeshed_dir = os.path.abspath(options.custom_typeshed_dir)
  2005	    options.config_file = args.mypy_config_file
  2006	    options.use_builtins_fixtures = use_builtins_fixtures
  2007	    options.show_traceback = args.show_traceback
  2008	    options.pdb = args.pdb
  2009
  2010	    if options.config_file:
  2011
  2012	        def set_strict_flags() -> None:  # not needed yet
  2013	            return
  2014
  2015	        parse_config_file(options, set_strict_flags, options.config_file, sys.stdout, sys.stderr)
  2016
  2017	    def error_callback(msg: str) -> typing.NoReturn:
  2018	        print(_style("error:", color="red", bold=True), msg)
  2019	        sys.exit(1)
  2020
  2021	    def warning_callback(msg: str) -> None:
  2022	        print(_style("warning:", color="yellow", bold=True), msg)
  2023
  2024	    options.process_error_codes(error_callback=error_callback)
  2025	    options.process_incomplete_features(
  2026	        error_callback=error_callback, warning_callback=warning_callback
  2027	    )
  2028	    options.process_strict_bytes()
  2029
  2030	    try:
  2031	        modules = build_stubs(modules, options, find_submodules=not args.check_typeshed)
  2032	    except StubtestFailure as stubtest_failure:
  2033	        print(
  2034	            _style("error:", color="red", bold=True),
  2035	            f"not checking stubs due to {stubtest_failure}",
  2036	        )
  2037	        return 1
  2038
  2039	    exit_code = 0
  2040	    error_count = 0
  2041	    for module in modules:
  2042	        for error in test_module(module):
  2043	            # Filter errors
  2044	            if args.ignore_missing_stub and error.is_missing_stub():
  2045	                continue
  2046	            if args.ignore_positional_only and error.is_positional_only_related():
  2047	                continue
  2048	            if error.object_desc in allowlist:
  2049	                allowlist[error.object_desc] = True
  2050	                continue
  2051	            is_allowlisted = False
  2052	            for w in allowlist:
  2053	                if allowlist_regexes[w].fullmatch(error.object_desc):
  2054	                    allowlist[w] = True
  2055	                    is_allowlisted = True
  2056	                    break
  2057	            if is_allowlisted:
  2058	                continue
  2059
  2060	            # We have errors, so change exit code, and output whatever necessary
  2061	            exit_code = 1
  2062	            if args.generate_allowlist:
  2063	                generated_allowlist.add(error.object_desc)
  2064	                continue
  2065	            print(error.get_description(concise=args.concise))
  2066	            error_count += 1
  2067
  2068	    # Print unused allowlist entries
  2069	    if not args.ignore_unused_allowlist:
  2070	        for w in allowlist:
  2071	            # Don't consider an entry unused if it regex-matches the empty string
  2072	            # This lets us allowlist errors that don't manifest at all on some systems
  2073	            if not allowlist[w] and not allowlist_regexes[w].fullmatch(""):
  2074	                exit_code = 1
  2075	                error_count += 1
  2076	                print(f"note: unused allowlist entry {w}")
  2077
  2078	    # Print the generated allowlist
  2079	    if args.generate_allowlist:
  2080	        for e in sorted(generated_allowlist):
  2081	            print(e)
  2082	        exit_code = 0
  2083	    elif not args.concise:
  2084	        if error_count:
  2085	            print(
  2086	                _style(
  2087	                    f"Found {error_count} error{plural_s(error_count)}"
  2088	                    f" (checked {len(modules)} module{plural_s(modules)})",
  2089	                    color="red",
  2090	                    bold=True,
  2091	                )
  2092	            )
  2093	        else:
  2094	            print(
  2095	                _style(
  2096	                    f"Success: no issues found in {len(modules)} module{plural_s(modules)}",
  2097	                    color="green",
  2098	                    bold=True,
  2099	                )
  2100	            )
  2101
  2102	    return exit_code

    63	def walk_packages(
    64	    inspect: ModuleInspect, packages: list[str], verbose: bool = False
    65	) -> Iterator[str]:
    66	    """Iterates through all packages and sub-packages in the given list.
    67
    68	    This uses runtime imports (in another process) to find both Python and C modules.
    69	    For Python packages we simply pass the __path__ attribute to pkgutil.walk_packages() to
    70	    get the content of the package (all subpackages and modules).  However, packages in C
    71	    extensions do not have this attribute, so we have to roll out our own logic: recursively
    72	    find all modules imported in the package that have matching names.

    91	            yield from walk_packages(inspect, prop.subpackages, verbose)

   184	def remove_misplaced_type_comments(source: str | bytes) -> str | bytes:
   185	    """Remove comments from source that could be understood as misplaced type comments.
   186
   187	    Normal comments may look like misplaced type comments, and since they cause blocking
   188	    parse errors, we want to avoid them.
   189	    """
   190	    if isinstance(source, bytes):
   191	        # This gives us a 1-1 character code mapping, so it's roundtrippable.
   192	        text = source.decode("latin1")
   193	    else:
   194	        text = source
   195
   196	    # Remove something that looks like a variable type comment but that's by itself
   197	    # on a line, as it will often generate a parse error (unless it's # type: ignore).
   198	    text = re.sub(r'^[ \t]*# +type: +["\'a-zA-Z_].*$', "", text, flags=re.MULTILINE)
   199
   200	    # Remove something that looks like a function type comment after docstring,
   201	    # which will result in a parse error.
   202	    text = re.sub(r'""" *\n[ \t\n]*# +type: +\(.*$', '"""\n', text, flags=re.MULTILINE)
   203	    text = re.sub(r"''' *\n[ \t\n]*# +type: +\(.*$", "'''\n", text, flags=re.MULTILINE)
   204
   205	    # Remove something that looks like a badly formed function type comment.
   206	    text = re.sub(r"^[ \t]*# +type: +\([^()]+(\)[ \t]*)?$", "", text, flags=re.MULTILINE)
   207
   208	    if isinstance(source, bytes):
   209	        return text.encode("latin1")
   210	    else:
   211	        return text

   452	class ImportTracker:
   453	    """Record necessary imports during stub generation."""
   454
   455	    def __init__(self) -> None:
   456	        # module_for['foo'] has the module name where 'foo' was imported from, or None if
   457	        # 'foo' is a module imported directly;
   458	        # direct_imports['foo'] is the module path used when the name 'foo' was added to the
   459	        # namespace.
   460	        # reverse_alias['foo'] is the name that 'foo' had originally when imported with an
   461	        # alias; examples
   462	        #     'from pkg import mod'      ==> module_for['mod'] == 'pkg'
   463	        #     'from pkg import mod as m' ==> module_for['m'] == 'pkg'
   464	        #                                ==> reverse_alias['m'] == 'mod'
   465	        #     'import pkg.mod as m'      ==> module_for['m'] == None
   466	        #                                ==> reverse_alias['m'] == 'pkg.mod'
   467	        #     'import pkg.mod'           ==> module_for['pkg'] == None
   468	        #                                ==> module_for['pkg.mod'] == None
   469	        #                                ==> direct_imports['pkg'] == 'pkg.mod'
   470	        #                                ==> direct_imports['pkg.mod'] == 'pkg.mod'
   471	        self.module_for: dict[str, str | None] = {}
   472	        self.direct_imports: dict[str, str] = {}
   473	        self.reverse_alias: dict[str, str] = {}
   474
   475	        # required_names is the set of names that are actually used in a type annotation
   476	        self.required_names: set[str] = set()
   477
   478	        # Names that should be reexported if they come from another module
   479	        self.reexports: set[str] = set()
   480
   481	    def add_import_from(
   482	        self, module: str, names: list[tuple[str, str | None]], require: bool = False
   483	    ) -> None:
   484	        for name, alias in names:
   485	            if alias:
   486	                # 'from {module} import {name} as {alias}'
   487	                self.module_for[alias] = module
   488	                self.reverse_alias[alias] = name
   489	            else:
   490	                # 'from {module} import {name}'
   491	                self.module_for[name] = module
   492	                self.reverse_alias.pop(name, None)
   493	            if require:
   494	                self.require_name(alias or name)
   495	            self.direct_imports.pop(alias or name, None)
   496
   497	    def add_import(self, module: str, alias: str | None = None, require: bool = False) -> None:
   498	        if alias:
   499	            # 'import {module} as {alias}'
   500	            assert "." not in alias  # invalid syntax
   501	            self.module_for[alias] = None
   502	            self.reverse_alias[alias] = module
   503	            if require:
   504	                self.required_names.add(alias)
   505	        else:
   506	            # 'import {module}'
   507	            name = module
   508	            if require:
   509	                self.required_names.add(name)
   510	            # add module and its parent packages
   511	            while name:
   512	                self.module_for[name] = None
   513	                self.direct_imports[name] = module
   514	                self.reverse_alias.pop(name, None)
   515	                name = name.rpartition(".")[0]
   516
   517	    def require_name(self, name: str) -> None:
   518	        while name not in self.direct_imports and "." in name:
   519	            name = name.rsplit(".", 1)[0]
   520	        self.required_names.add(name)
   521
   522	    def reexport(self, name: str) -> None:
   523	        """Mark a given non qualified name as needed in __all__.
   524
   525	        This means that in case it comes from a module, it should be
   526	        imported with an alias even if the alias is the same as the name.
   527	        """
   528	        self.require_name(name)
   529	        self.reexports.add(name)
   530
   531	    def import_lines(self) -> list[str]:
   532	        """The list of required import lines (as strings with python code).
   533
   534	        In order for a module be included in this output, an identifier must be both
   535	        'required' via require_name() and 'imported' via add_import_from()
   536	        or add_import()
   537	        """
   538	        result = []
   539
   540	        # To summarize multiple names imported from a same module, we collect those
   541	        # in the `module_map` dictionary, mapping a module path to the list of names that should
   542	        # be imported from it. the names can also be alias in the form 'original as alias'
   543	        module_map: Mapping[str, list[str]] = defaultdict(list)
   544
   545	        for name in sorted(
   546	            self.required_names,
   547	            key=lambda n: (self.reverse_alias[n], n) if n in self.reverse_alias else (n, ""),
   548	        ):
   549	            # If we haven't seen this name in an import statement, ignore it
   550	            if name not in self.module_for:
   551	                continue
   552
   553	            m = self.module_for[name]
   554	            if m is not None:
   555	                # This name was found in a from ... import ...
   556	                # Collect the name in the module_map

   126	    left: Type,
   127	    right: Type,

   134	    ignore_promotions: bool = False,

   137	    """Is 'left' subtype of 'right'?
   138
   139	    Also consider Any to be a subtype of any type, and vice versa. This
   140	    recursively applies to components of composite types (List[int] is subtype
   141	    of List[Any], for example).
   142
   143	    type_parameter_checker is used to check the type parameters (for example,
   144	    A with B in is_subtype(C[A], C[B]). The default checks for subtype relation
   145	    between the type arguments (e.g., A and B), taking the variance of the
   146	    type var into account.
   147	    """
   148	    if subtype_context is None:
   149	        subtype_context = SubtypeContext(
   150	            ignore_type_params=ignore_type_params,
   151	            ignore_pos_arg_names=ignore_pos_arg_names,
   152	            ignore_declared_variance=ignore_declared_variance,
   153	            always_covariant=always_covariant,
   154	            ignore_promotions=ignore_promotions,
   155	            options=options,
   156	        )
   157	    else:
   158	        assert not any(
   159	            {
   160	                ignore_type_params,
   161	                ignore_pos_arg_names,
   162	                ignore_declared_variance,
   163	                always_covariant,
   164	                ignore_promotions,
   165	                options,
   166	            }
   167	        ), "Don't pass both context and individual flags"
   168	    if type_state.is_assumed_subtype(left, right):
   169	        return True
   170	    if mypy.typeops.is_recursive_pair(left, right):
   171	        # This case requires special care because it may cause infinite recursion.
   172	        # Our view on recursive types is known under a fancy name of iso-recursive mu-types.
   173	        # Roughly this means that a recursive type is defined as an alias where right hand side
   174	        # can refer to the type as a whole, for example:

   182	        # check that get_proper_type(left) <: get_proper_type(right). On the example above,

   221	    if type_state.is_assumed_proper_subtype(left, right):
   222	        return True
   223	    if mypy.typeops.is_recursive_pair(left, right):
   224	        # Same as for non-proper subtype, see detailed comment there for explanation.
   225	        with pop_on_exit(type_state.get_assumptions(is_proper=True), left, right):
   226	            return _is_subtype(left, right, subtype_context, proper_subtype=True)
   227	    return _is_subtype(left, right, subtype_context, proper_subtype=True)

   256	def is_same_type(
   257	    a: Type, b: Type, ignore_promotions: bool = True, subtype_context: SubtypeContext | None = None
   258	) -> bool:
   259	    """Are these types proper subtypes of each other?
   260
   261	    This means types may have different representation (e.g. an alias, or
   262	    a non-simplified union) but are semantically exchangeable in all contexts.
   263	    """
   264	    # First, use fast path for some common types. This is performance-critical.
   265	    if (
   266	        type(a) is Instance
   267	        and type(b) is Instance
   268	        and a.type == b.type
   269	        and len(a.args) == len(b.args)
   270	        and a.last_known_value is b.last_known_value
   271	    ):
   272	        return all(is_same_type(x, y) for x, y in zip(a.args, b.args))
   273	    elif isinstance(a, TypeVarType) and isinstance(b, TypeVarType) and a.id == b.id:
   274	        return True
   275
   276	    # Note that using ignore_promotions=True (default) makes types like int and int64
   277	    # considered not the same type (which is the case at runtime).
   278	    # Also Union[bool, int] (if it wasn't simplified before) will be different
   279	    # from plain int, etc.
   280	    return is_proper_subtype(
   281	        a, b, ignore_promotions=ignore_promotions, subtype_context=subtype_context
   282	    ) and is_proper_subtype(
   283	        b, a, ignore_promotions=ignore_promotions, subtype_context=subtype_context
   284	    )
   285
   286
   287	# This is a common entry point for subtyping checks (both proper and non-proper).
   288	# Never call this private function directly, use the public versions.
   289	def _is_subtype(
   290	    left: Type, right: Type, subtype_context: SubtypeContext, proper_subtype: bool
   291	) -> bool:
   292	    subtype_context.check_context(proper_subtype)
   293	    orig_right = right
   294	    orig_left = left
   295	    left = get_proper_type(left)
   296	    right = get_proper_type(right)
   297
   298	    # Note: Unpack type should not be a subtype of Any, since it may represent
   299	    # multiple types. This should always go through the visitor, to check arity.
   300	    if (
   301	        not proper_subtype
   302	        and isinstance(right, (AnyType, UnboundType, ErasedType))
   303	        and not isinstance(left, UnpackType)
   304	    ):
   305	        # TODO: should we consider all types proper subtypes of UnboundType and/or
   306	        # ErasedType as we do for non-proper subtyping.
   307	        return True
   308
   309	    if isinstance(right, UnionType) and not isinstance(left, UnionType):
   310	        # Normally, when 'left' is not itself a union, the only way
   311	        # 'left' can be a subtype of the union 'right' is if it is a
   312	        # subtype of one of the items making up the union.
   313	        if proper_subtype:
   314	            is_subtype_of_item = any(
   315	                is_proper_subtype(orig_left, item, subtype_context=subtype_context)
   316	                for item in right.items
   317	            )
   318	        else:
   319	            is_subtype_of_item = any(
   320	                is_subtype(orig_left, item, subtype_context=subtype_context)
   321	                for item in right.items
   322	            )
   323	        # Recombine rhs literal types, to make an enum type a subtype
   324	        # of a union of all enum items as literal types. Only do it if
   325	        # the previous check didn't succeed, since recombining can be
   326	        # expensive.
   327	        # `bool` is a special case, because `bool` is `Literal[True, False]`.
   328	        if (
   329	            not is_subtype_of_item
   330	            and isinstance(left, Instance)
   331	            and (left.type.is_enum or left.type.fullname == "builtins.bool")
   332	        ):
   333	            right = UnionType(
   334	                mypy.typeops.try_contracting_literals_in_union(flatten_nested_unions(right.items))
   335	            )
   336	            if proper_subtype:
   337	                is_subtype_of_item = any(
   338	                    is_proper_subtype(orig_left, item, subtype_context=subtype_context)
   339	                    for item in right.items
   340	                )
   341	            else:
   342	                is_subtype_of_item = any(
   343	                    is_subtype(orig_left, item, subtype_context=subtype_context)
   344	                    for item in right.items
   345	                )
   346	        # However, if 'left' is a type variable T, T might also have
   347	        # an upper bound which is itself a union. This case will be
   348	        # handled below by the SubtypeVisitor. We have to check both
   349	        # possibilities, to handle both cases like T <: Union[T, U]
   350	        # and cases like T <: B where B is the upper bound of T and is
   351	        # a union. (See #2314.)
   352	        if not isinstance(left, TypeVarType):
   353	            return is_subtype_of_item
   354	        elif is_subtype_of_item:
   355	            return True
   356	        # otherwise, fall through
   357	    return left.accept(SubtypeVisitor(orig_right, subtype_context, proper_subtype))

   360	def check_type_parameter(
   361	    left: Type, right: Type, variance: int, proper_subtype: bool, subtype_context: SubtypeContext
   362	) -> bool:
   363	    # It is safe to consider empty collection literals and similar as covariant, since
   364	    # such type can't be stored in a variable, see checker.is_valid_inferred_type().
   365	    if variance == INVARIANT:
   366	        p_left = get_proper_type(left)
   367	        if isinstance(p_left, UninhabitedType) and p_left.ambiguous:
   368	            variance = COVARIANT
   369	    # If variance hasn't been inferred yet, we are lenient and default to
   370	    # covariance. This shouldn't happen often, but it's very difficult to
   371	    # avoid these cases altogether.
   372	    if variance == COVARIANT or variance == VARIANCE_NOT_READY:
   373	        if proper_subtype:
   374	            return is_proper_subtype(left, right, subtype_context=subtype_context)
   375	        else:
   376	            return is_subtype(left, right, subtype_context=subtype_context)
   377	    elif variance == CONTRAVARIANT:
   378	        if proper_subtype:
   379	            return is_proper_subtype(right, left, subtype_context=subtype_context)
   380	        else:
   381	            return is_subtype(right, left, subtype_context=subtype_context)
   382	    else:
   383	        if proper_subtype:
   384	            # We pass ignore_promotions=False because it is a default for subtype checks.
   385	            # The actual value will be taken from the subtype_context, and it is whatever
   386	            # the original caller passed.
   387	            return is_same_type(
   388	                left, right, ignore_promotions=False, subtype_context=subtype_context
   389	            )
   390	        else:
   391	            return is_equivalent(left, right, subtype_context=subtype_context)

   394	class SubtypeVisitor(TypeVisitor[bool]):
   395	    def __init__(self, right: Type, subtype_context: SubtypeContext, proper_subtype: bool) -> None:
   396	        self.right = get_proper_type(right)
   397	        self.orig_right = right
   398	        self.proper_subtype = proper_subtype
   399	        self.subtype_context = subtype_context
   400	        self.options = subtype_context.options
   401	        self._subtype_kind = SubtypeVisitor.build_subtype_kind(subtype_context, proper_subtype)
   402
   403	    @staticmethod
   404	    def build_subtype_kind(subtype_context: SubtypeContext, proper_subtype: bool) -> SubtypeKind:
   405	        return (
   406	            state.strict_optional,
   407	            proper_subtype,
   408	            subtype_context.ignore_type_params,
   409	            subtype_context.ignore_pos_arg_names,
   410	            subtype_context.ignore_declared_variance,
   411	            subtype_context.always_covariant,
   412	            subtype_context.ignore_promotions,
   413	            subtype_context.erase_instances,
   414	            subtype_context.keep_erased_types,
   415	        )
   416
   417	    def _is_subtype(self, left: Type, right: Type) -> bool:
   418	        if self.proper_subtype:
   419	            return is_proper_subtype(left, right, subtype_context=self.subtype_context)
   420	        return is_subtype(left, right, subtype_context=self.subtype_context)
   421
   422	    def _all_subtypes(self, lefts: Iterable[Type], rights: Iterable[Type]) -> bool:
   423	        return all(self._is_subtype(li, ri) for (li, ri) in zip(lefts, rights))
   424
   425	    # visit_x(left) means: is left (which is an instance of X) a subtype of right?
   426
   427	    def visit_unbound_type(self, left: UnboundType) -> bool:
   428	        # This can be called if there is a bad type annotation. The result probably
   429	        # doesn't matter much but by returning True we simplify these bad types away
   430	        # from unions, which could filter out some bogus messages.
   431	        return True
   432
   433	    def visit_any(self, left: AnyType) -> bool:
   434	        return isinstance(self.right, AnyType) if self.proper_subtype else True
   435
   436	    def visit_none_type(self, left: NoneType) -> bool:
   437	        if state.strict_optional:
   438	            if isinstance(self.right, NoneType) or is_named_instance(
   439	                self.right, "builtins.object"
   440	            ):
   441	                return True
   442	            if isinstance(self.right, Instance) and self.right.type.is_protocol:
   443	                members = self.right.type.protocol_members
   444	                # None is compatible with Hashable (and other similar protocols). This is
   445	                # slightly sloppy since we don't check the signature of "__hash__".
   446	                # None is also compatible with `SupportsStr` protocol.
   447	                return not members or all(member in ("__hash__", "__str__") for member in members)
   448	            return False
   449	        else:
   450	            return True
   451
   452	    def visit_uninhabited_type(self, left: UninhabitedType) -> bool:
   453	        return True
   454
   455	    def visit_erased_type(self, left: ErasedType) -> bool:
   456	        # This may be encountered during type inference. The result probably doesn't
   457	        # matter much.
   458	        # TODO: it actually does matter, figure out more principled logic about this.
   459	        return not self.subtype_context.keep_erased_types
   460
   461	    def visit_deleted_type(self, left: DeletedType) -> bool:
   462	        return True
   463
   464	    def visit_instance(self, left: Instance) -> bool:
   465	        if left.type.fallback_to_any and not self.proper_subtype:
   466	            # NOTE: `None` is a *non-subclassable* singleton, therefore no class
   467	            # can by a subtype of it, even with an `Any` fallback.
   468	            # This special case is needed to treat descriptors in classes with
   469	            # dynamic base classes correctly, see #5456.
   470	            return not isinstance(self.right, NoneType)
   471	        right = self.right
   472	        if isinstance(right, TupleType) and right.partial_fallback.type.is_enum:
   473	            return self._is_subtype(left, mypy.typeops.tuple_fallback(right))
   474	        if isinstance(right, TupleType):
   475	            if len(right.items) == 1:
   476	                # Non-normalized Tuple type (may be left after semantic analysis
   477	                # because semanal_typearg visitor is not a type translator).
   478	                item = right.items[0]
   479	                if isinstance(item, UnpackType):
   480	                    unpacked = get_proper_type(item.type)
   481	                    if isinstance(unpacked, Instance):
   482	                        return self._is_subtype(left, unpacked)
   483	            if left.type.has_base(right.partial_fallback.type.fullname):
   484	                if not self.proper_subtype:
   485	                    # Special cases to consider:
   486	                    #   * Plain tuple[Any, ...] instance is a subtype of all tuple types.
   487	                    #   * Foo[*tuple[Any, ...]] (normalized) instance is a subtype of all
   488	                    #     tuples with fallback to Foo (e.g. for variadic NamedTuples).
   489	                    mapped = map_instance_to_supertype(left, right.partial_fallback.type)
   490	                    if is_erased_instance(mapped):
   491	                        if (
   492	                            mapped.type.fullname == "builtins.tuple"
   493	                            or mapped.type.has_type_var_tuple_type
   494	                        ):
   495	                            return True
   496	            return False
   497	        if isinstance(right, TypeVarTupleType):
   498	            # tuple[Any, ...] is like Any in the world of tuples (see special case above).
   499	            if left.type.has_base("builtins.tuple"):
   500	                mapped = map_instance_to_supertype(left, right.tuple_fallback.type)
   501	                if isinstance(get_proper_type(mapped.args[0]), AnyType):
   502	                    return not self.proper_subtype
   503	        if isinstance(right, Instance):
   504	            if type_state.is_cached_subtype_check(self._subtype_kind, left, right):
   505	                return True
   506	            if type_state.is_cached_negative_subtype_check(self._subtype_kind, left, right):
   507	                return False
   508	            if not self.subtype_context.ignore_promotions and not right.type.is_protocol:
   509	                for base in left.type.mro:
   510	                    if base._promote and any(
   511	                        self._is_subtype(p, self.right) for p in base._promote
   512	                    ):
   513	                        type_state.record_subtype_cache_entry(self._subtype_kind, left, right)
   514	                        return True
   515	                # Special case: Low-level integer types are compatible with 'int'. We can't
   516	                # use promotions, since 'int' is already promoted to low-level integer types,
   517	                # and we can't have circular promotions.
   518	                if left.type.alt_promote and left.type.alt_promote.type is right.type:
   519	                    return True
   520	            rname = right.type.fullname
   521	            # Always try a nominal check if possible,
   522	            # there might be errors that a user wants to silence *once*.
   523	            # NamedTuples are a special case, because `NamedTuple` is not listed
   524	            # in `TypeInfo.mro`, so when `(a: NamedTuple) -> None` is used,
   525	            # we need to check for `is_named_tuple` property
   526	            if (
   527	                left.type.has_base(rname)
   528	                or rname == "builtins.object"
   529	                or (
   530	                    rname in TYPED_NAMEDTUPLE_NAMES
   531	                    and any(l.is_named_tuple for l in left.type.mro)
   532	                )
   533	            ) and not self.subtype_context.ignore_declared_variance:
   534	                # Map left type to corresponding right instances.
   535	                t = map_instance_to_supertype(left, right.type)
   536	                if self.subtype_context.erase_instances:
   537	                    erased = erase_type(t)
   538	                    assert isinstance(erased, Instance)
   539	                    t = erased
   540	                nominal = True
   541	                if right.type.has_type_var_tuple_type:
   542	                    # For variadic instances we simply find the correct type argument mappings,
   543	                    # all the heavy lifting is done by the tuple subtyping.
   544	                    assert right.type.type_var_tuple_prefix is not None
   545	                    assert right.type.type_var_tuple_suffix is not None
   546	                    prefix = right.type.type_var_tuple_prefix
   547	                    suffix = right.type.type_var_tuple_suffix
   548	                    tvt = right.type.defn.type_vars[prefix]
   549	                    assert isinstance(tvt, TypeVarTupleType)
   550	                    fallback = tvt.tuple_fallback
   551	                    left_prefix, left_middle, left_suffix = split_with_prefix_and_suffix(
   552	                        t.args, prefix, suffix
   553	                    )
   554	                    right_prefix, right_middle, right_suffix = split_with_prefix_and_suffix(
   555	                        right.args, prefix, suffix
   556	                    )
   557	                    left_args = (
   558	                        left_prefix + (TupleType(list(left_middle), fallback),) + left_suffix
   559	                    )
   560	                    right_args = (
   561	                        right_prefix + (TupleType(list(right_middle), fallback),) + right_suffix
   562	                    )
   563	                    if not self.proper_subtype and is_erased_instance(t):
   564	                        return True
   565	                    if len(left_args) != len(right_args):
   566	                        return False
   567	                    type_params = zip(left_args, right_args, right.type.defn.type_vars)
   568	                else:
   569	                    type_params = zip(t.args, right.args, right.type.defn.type_vars)
   570	                if not self.subtype_context.ignore_type_params:
   571	                    tried_infer = False
   572	                    for lefta, righta, tvar in type_params:
   573	                        if isinstance(tvar, TypeVarType):
   574	                            if tvar.variance == VARIANCE_NOT_READY and not tried_infer:
   575	                                infer_class_variances(right.type)
   576	                                tried_infer = True
   577	                            if (
   578	                                self.subtype_context.always_covariant
   579	                                and tvar.variance == INVARIANT
   580	                            ):
   581	                                variance = COVARIANT
   582	                            else:
   583	                                variance = tvar.variance
   584	                            if not check_type_parameter(
   585	                                lefta, righta, variance, self.proper_subtype, self.subtype_context
   586	                            ):
   587	                                nominal = False
   588	                        else:
   589	                            # TODO: everywhere else ParamSpecs are handled as invariant.
   590	                            if not check_type_parameter(
   591	                                lefta, righta, COVARIANT, self.proper_subtype, self.subtype_context
   592	                            ):
   593	                                nominal = False
   594	                if nominal:
   595	                    type_state.record_subtype_cache_entry(self._subtype_kind, left, right)
   596	                else:
   597	                    type_state.record_negative_subtype_cache_entry(self._subtype_kind, left, right)
   598	                return nominal
   599	            if right.type.is_protocol and is_protocol_implementation(
   600	                left, right, proper_subtype=self.proper_subtype, options=self.options
   601	            ):
   602	                return True
   603	            # We record negative cache entry here, and not in the protocol check like we do for
   604	            # positive cache, to avoid accidentally adding a type that is not a structural
   605	            # subtype, but is a nominal subtype (involving type: ignore override).
   606	            type_state.record_negative_subtype_cache_entry(self._subtype_kind, left, right)
   607	            return False
   608	        if isinstance(right, TypeType):
   609	            item = right.item
   610	            if isinstance(item, TupleType):
   611	                item = mypy.typeops.tuple_fallback(item)
   612	            # TODO: this is a bit arbitrary, we should only skip Any-related cases.
   613	            if not self.proper_subtype:
   614	                if is_named_instance(left, "builtins.type"):
   615	                    return self._is_subtype(TypeType(AnyType(TypeOfAny.special_form)), right)
   616	                if left.type.is_metaclass():
   617	                    if isinstance(item, AnyType):
   618	                        return True
   619	                    if isinstance(item, Instance):
   620	                        return is_named_instance(item, "builtins.object")
   621	        if isinstance(right, LiteralType) and left.last_known_value is not None:
   622	            return self._is_subtype(left.last_known_value, right)
   623	        if isinstance(right, FunctionLike):
   624	            # Special case: Instance can be a subtype of Callable / Overloaded.
   625	            call = find_member("__call__", left, left, is_operator=True)
   626	            if call:
   627	                return self._is_subtype(call, right)
   628	            return False
   629	        else:
   630	            return False
   631
   632	    def visit_type_var(self, left: TypeVarType) -> bool:
   633	        right = self.right
   634	        if isinstance(right, TypeVarType) and left.id == right.id:
   635	            # Fast path for most common case.
   636	            if left.upper_bound == right.upper_bound:
   637	                return True
   638	            # Corner case for self-types in classes generic in type vars
   639	            # with value restrictions.
   640	            if left.id.is_self():
   641	                return True
   642	            return self._is_subtype(left.upper_bound, right.upper_bound)
   643	        if left.values and self._is_subtype(UnionType.make_union(left.values), right):
   644	            return True
   645	        return self._is_subtype(left.upper_bound, self.right)
   646
   647	    def visit_param_spec(self, left: ParamSpecType) -> bool:
   648	        right = self.right
   649	        if (
   650	            isinstance(right, ParamSpecType)
   651	            and right.id == left.id
   652	            and right.flavor == left.flavor
   653	        ):
   654	            return self._is_subtype(left.prefix, right.prefix)
   655	        if isinstance(right, Parameters) and are_trivial_parameters(right):
   656	            return True
   657	        return self._is_subtype(left.upper_bound, self.right)
   658
   659	    def visit_type_var_tuple(self, left: TypeVarTupleType) -> bool:
   660	        right = self.right
   661	        if isinstance(right, TypeVarTupleType) and right.id == left.id:
   662	            return left.min_len >= right.min_len
   663	        return self._is_subtype(left.upper_bound, self.right)
   664
   665	    def visit_unpack_type(self, left: UnpackType) -> bool:
   666	        # TODO: Ideally we should not need this (since it is not a real type).
   667	        # Instead callers (upper level types) should handle it when it appears in type list.
   668	        if isinstance(self.right, UnpackType):
   669	            return self._is_subtype(left.type, self.right.type)
   670	        if isinstance(self.right, Instance) and self.right.type.fullname == "builtins.object":
   671	            return True
   672	        return False
   673
   674	    def visit_parameters(self, left: Parameters) -> bool:
   675	        if isinstance(self.right, Parameters):
   676	            return are_parameters_compatible(
   677	                left,
   678	                self.right,
   679	                is_compat=self._is_subtype,
   680	                # TODO: this should pass the current value, but then couple tests fail.
   681	                is_proper_subtype=False,
   682	                ignore_pos_arg_names=self.subtype_context.ignore_pos_arg_names,
   683	            )
   684	        elif isinstance(self.right, Instance):
   685	            return self.right.type.fullname == "builtins.object"
   686	        else:
   687	            return False
   688
   689	    def visit_callable_type(self, left: CallableType) -> bool:
   690	        right = self.right
   691	        if isinstance(right, CallableType):
   692	            if left.type_guard is not None and right.type_guard is not None:
   693	                if not self._is_subtype(left.type_guard, right.type_guard):
   694	                    return False
   695	            elif left.type_is is not None and right.type_is is not None:
   696	                # For TypeIs we have to check both ways; it is unsafe to pass
   697	                # a TypeIs[Child] when a TypeIs[Parent] is expected, because
   698	                # if the narrower returns False, we assume that the narrowed value is
   699	                # *not* a Parent.
   700	                if not self._is_subtype(left.type_is, right.type_is) or not self._is_subtype(
   701	                    right.type_is, left.type_is
   702	                ):
   703	                    return False
   704	            elif right.type_guard is not None and left.type_guard is None:
   705	                # This means that one function has `TypeGuard` and other does not.
