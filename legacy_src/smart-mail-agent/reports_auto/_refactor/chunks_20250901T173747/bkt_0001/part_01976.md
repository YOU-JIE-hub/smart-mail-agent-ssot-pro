    66	        if line[0:16] == 'StartCharMetrics':
    67	            inMetrics = 1
    68	        elif line[0:14] == 'EndCharMetrics':
    69	            inMetrics = 0
    70	        elif inMetrics:
    71	            chunks = line.split(';')
    72	            chunks = [chunk.strip() for chunk in chunks]
    73	            cidChunk, widthChunk, nameChunk = chunks[0:3]
    74
    75	            # character ID
    76	            l, r = cidChunk.split()
    77	            assert l == 'C', 'bad line in font file %s' % line
    78	            cid = int(r)
    79
    80	            # width
    81	            l, r = widthChunk.split()
    82	            assert l == 'WX', 'bad line in font file %s' % line
    83	            try:
    84	                width = int(r)
    85	            except ValueError:
    86	                width = float(r)
    87
    88	            # name
    89	            l, r = nameChunk.split()
    90	            assert l == 'N', 'bad line in font file %s' % line
    91	            name = r
    92
    93	            glyphLevel.append((cid, width, name))
    94
    95	    # pass 2 font info
    96	    inHeader = 0
    97	    for line in lines:
    98	        if line[0:16] == 'StartFontMetrics':
    99	            inHeader = 1
   100	        if line[0:16] == 'StartCharMetrics':
   101	            inHeader = 0
   102	        elif inHeader:
   103	            if line[0:7] == 'Comment': pass
   104	            try:
   105	                left, right = line.split(' ',1)
   106	            except:
   107	                raise ValueError("Header information error in afm %s: line='%s'" % (afmFileName, line))
   108	            try:
   109	                right = int(right)
   110	            except:
   111	                pass
   112	            topLevel[left] = right
   113
   114
   115	    return (topLevel, glyphLevel)

   271	        """
   272	        Return a compact list of the code points differing between two encodings
   273
   274	        This is in the Adobe format: list of
   275	           [[b1, name1, name2, name3],
   276	           [b2, name4]]
   277
   278	        where b1...bn is the starting code point, and the glyph names following
   279	        are assigned consecutive code points.
   280
   281	        """
   282
   283	        ranges = []
   284	        curRange = None

   286	            glyph = self.vector[i]
   287	            if glyph==otherEnc.vector[i]:

   289	                    ranges.append(curRange)
   290	                    curRange = []
   291	            else:
   292	                if curRange:
   293	                    curRange.append(glyph)
   294	                elif glyph:
   295	                    curRange = [i, glyph]

     7	    allowedTypes = strTypes if isinstance(strTypes, tuple) else (strTypes,)

     9	    for x in pattern_sequence:

    16	class PDFPattern(PDFObject):
    17	    __RefOnly__ = 1
    18	    def __init__(self, pattern_sequence, **keywordargs):
    19	        """
    20	        Description of a kind of PDF object using a pattern.
    21
    22	        Pattern sequence should contain strings, singletons of form [string] or
    23	        PDFPatternIf objects.
    24	        Strings are literal strings to be used in the object.
    25	        Singletons are names of keyword arguments to include.
    26	        PDFpatternIf objects allow some conditionality.
    27	        Keyword arguments can be non-instances which are substituted directly in string conversion,
    28	        or they can be object instances in which case they should be pdfdoc.* style
    29	        objects with a x.format(doc) method.
    30	        Keyword arguments may be set on initialization or subsequently using __setitem__, before format.
    31	        "constant object" instances can also be inserted in the patterns.
    32	        """
    33	        _patternSequenceCheck(pattern_sequence)
    34	        self.pattern = pattern_sequence
    35	        self.arguments = keywordargs
    36
    37	    def __setitem__(self, item, value):
    38	        self.arguments[item] = value
    39
    40	    def __getitem__(self, item):
    41	        return self.arguments[item]
    42
    43	    def eval(self,L):
    44	        arguments = self.arguments
    45	        document = self.__document
    46	        for x in L:
    47	            if isinstance(x,strTypes):
    48	                yield pdfdocEnc(x)
    49	            elif isinstance(x,PDFObject):
    50	                yield x.format(document)
    51	            elif isinstance(x,PDFPatternIf):
    52	                result = list(self.eval(x.cond))
    53	                cond = result and result[0]
    54	                for z in self.eval(x.thenPart if cond else x.elsePart):
    55	                    yield z
    56	            else:
    57	                name = x[0]
    58	                value = arguments.get(name, None)
    59	                if value is None:
    60	                    raise ValueError("%s value not defined" % ascii(name))
    61	                if isinstance(value,PDFObject):
    62	                    yield format(value,document)
    63	                elif isinstance(value,strTypes):
    64	                    yield pdfdocEnc(value)
    65	                else:
    66	                    yield pdfdocEnc(str(value))
    67
    68	    def format(self, document):
    69	        self.__document = document
    70	        try:
    71	            return b"".join(self.eval(self.pattern))
    72	        finally:
    73	            del self.__document
    74
    75	    def clone(self):
    76	        c = object.__new__(self.__class__)
    77	        c.pattern = self.pattern
    78	        c.arguments = self.arguments
    79	        return c

    82	    '''cond will be evaluated as [cond] in PDFpattern eval.
    83	    It should evaluate to a list with value 0/1 etc etc.
    84	    thenPart is a list to be evaluated if the cond evaulates true,
    85	    elsePart is the false sequence.
    86	    '''
    87	    def __init__(self,cond,thenPart=[],elsePart=[]):
    88	        if not isinstance(cond,list): cond = [cond]

   219	# Read data from the JPEG file. We should probably be using PIL to
   220	# get this information for us -- but this way is more fun!
   221	# Returns (width, height, color components) as a triple
   222	# This is based on Thomas Merz's code from GhostScript (viewjpeg.ps)
   223	def readJPEGInfo(image):
   224	    "Read width, height and number of components from open JPEG file."
   225
   226	    import struct
   227	    from reportlab.pdfbase.pdfdoc import PDFError
   228
   229	    #Acceptable JPEG Markers:
   230	    #  SROF0=baseline, SOF1=extended sequential or SOF2=progressive
   231	    validMarkers = [0xC0, 0xC1, 0xC2]
   232
   233	    #JPEG markers without additional parameters
   234	    noParamMarkers = \
   235	        [ 0xD0, 0xD1, 0xD2, 0xD3, 0xD4, 0xD5, 0xD6, 0xD7, 0xD8, 0x01 ]
   236
   237	    #Unsupported JPEG Markers
   238	    unsupportedMarkers = \
   239	        [ 0xC3, 0xC5, 0xC6, 0xC7, 0xC8, 0xC9, 0xCA, 0xCB, 0xCD, 0xCE, 0xCF ]
   240
   241	    #read JPEG marker segments until we find SOFn marker or EOF
   242	    dpi = (72,72)
   243	    done = 0
   244	    while not done:
   245	        x = struct.unpack('B', image.read(1))
   246	        if x[0] == 0xFF:                    #found marker
   247	            x = struct.unpack('B', image.read(1))
   248	            #print('marker=%2x' % x[0])
   249	            if x[0] in validMarkers:
   250	                image.seek(2, 1)            #skip segment length
   251	                x = struct.unpack('B', image.read(1)) #data precision
   252	                if x[0] != 8:
   253	                    raise PDFError('JPEG must have 8 bits per component')
   254	                y = struct.unpack('BB', image.read(2))
   255	                height = (y[0] << 8) + y[1]
   256	                y = struct.unpack('BB', image.read(2))
   257	                width =  (y[0] << 8) + y[1]
   258	                y = struct.unpack('B', image.read(1))
   259	                color =  y[0]
   260	                return width, height, color, dpi
   261	            elif x[0]==0xE0:
   262	                x = struct.unpack('BB', image.read(2))
   263	                n = (x[0] << 8) + x[1] - 2
   264	                x = image.read(n)
