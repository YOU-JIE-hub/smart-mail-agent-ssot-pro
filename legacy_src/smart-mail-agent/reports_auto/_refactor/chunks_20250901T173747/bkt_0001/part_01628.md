  6245	    def lookup(
  6246	        self, name: str, ctx: Context, suppress_errors: bool = False
  6247	    ) -> SymbolTableNode | None:
  6248	        """Look up an unqualified (no dots) name in all active namespaces.
  6249
  6250	        Note that the result may contain a PlaceholderNode. The caller may
  6251	        want to defer in that case.
  6252
  6253	        Generate an error if the name is not defined unless suppress_errors
  6254	        is true or the current namespace is incomplete. In the latter case
  6255	        defer.
  6256	        """
  6257	        implicit_name = False
  6258	        # 1a. Name declared using 'global x' takes precedence
  6259	        if name in self.global_decls[-1]:
  6260	            if name in self.globals:
  6261	                return self.globals[name]
  6262	            if not suppress_errors:
  6263	                self.name_not_defined(name, ctx)
  6264	            return None
  6265	        # 1b. Name declared using 'nonlocal x' takes precedence
  6266	        if name in self.nonlocal_decls[-1]:
  6267	            for table in reversed(self.locals[:-1]):
  6268	                if table is not None and name in table:
  6269	                    return table[name]
  6270	            if not suppress_errors:
  6271	                self.name_not_defined(name, ctx)
  6272	            return None
  6273	        # 2a. Class attributes (if within class definition)
  6274	        if self.type and not self.is_func_scope() and name in self.type.names:
  6275	            node = self.type.names[name]
  6276	            if not node.implicit:
  6277	                if self.is_active_symbol_in_class_body(node.node):
  6278	                    return node
  6279	            else:
  6280	                # Defined through self.x assignment
  6281	                implicit_name = True
  6282	                implicit_node = node
  6283	        # 2b. Class attributes __qualname__ and __module__
  6284	        if self.type and not self.is_func_scope() and name in {"__qualname__", "__module__"}:
  6285	            return SymbolTableNode(MDEF, Var(name, self.str_type()))
  6286	        # 3. Local (function) scopes
  6287	        for table in reversed(self.locals):
  6288	            if table is not None and name in table:
  6289	                return table[name]
  6290
  6291	        # 4. Current file global scope
  6292	        if name in self.globals:
  6293	            return self.globals[name]
  6294	        # 5. Builtins
  6295	        b = self.globals.get("__builtins__", None)
  6296	        if b:
  6297	            assert isinstance(b.node, MypyFile)
  6298	            table = b.node.names
  6299	            if name in table:
  6300	                if len(name) > 1 and name[0] == "_" and name[1] != "_":
  6301	                    if not suppress_errors:
  6302	                        self.name_not_defined(name, ctx)
  6303	                    return None
  6304	                node = table[name]
  6305	                return node
  6306	        # Give up.
  6307	        if not implicit_name and not suppress_errors:
  6308	            self.name_not_defined(name, ctx)
  6309	        else:
  6310	            if implicit_name:
  6311	                return implicit_node
  6312	        return None

  6314	    def is_active_symbol_in_class_body(self, node: SymbolNode | None) -> bool:
  6315	        """Can a symbol defined in class body accessed at current statement?
  6316
  6317	        Only allow access to class attributes textually after
  6318	        the definition, so that it's possible to fall back to the
  6319	        outer scope. Example:
  6320
  6321	            class X: ...
  6322
  6323	            class C:
  6324	                X = X  # Initializer refers to outer scope
  6325
  6326	        Nested classes are an exception, since we want to support
  6327	        arbitrary forward references in type annotations. Also, we
  6328	        allow forward references to type aliases to support recursive
  6329	        types.
  6330	        """
  6331	        # TODO: Forward reference to name imported in class body is not
  6332	        #       caught.
  6333	        if self.statement is None:
  6334	            # Assume it's fine -- don't have enough context to check
  6335	            return True
  6336	        if (
  6337	            node is None
  6338	            or self.is_textually_before_statement(node)
  6339	            or not self.is_defined_in_current_module(node.fullname)
  6340	        ):
  6341	            return True
  6342	        if self.is_type_like(node):
  6343	            # Allow forward references to classes/type aliases (see docstring), but
  6344	            # a forward reference should never shadow an existing regular reference.
  6345	            if node.name not in self.globals:
  6346	                return True
  6347	            global_node = self.globals[node.name]
  6348	            if not self.is_textually_before_class(global_node.node):
  6349	                return True
  6350	            return not self.is_type_like(global_node.node)
  6351	        return False
  6352
  6353	    def is_type_like(self, node: SymbolNode | None) -> bool:
  6354	        return isinstance(node, (TypeInfo, TypeAlias)) or (
  6355	            isinstance(node, PlaceholderNode) and node.becomes_typeinfo
  6356	        )

  6358	    def is_textually_before_statement(self, node: SymbolNode) -> bool:
  6359	        """Check if a node is defined textually before the current statement
  6360
  6361	        Note that decorated functions' line number are the same as
  6362	        the top decorator.
  6363	        """
  6364	        assert self.statement
  6365	        line_diff = self.statement.line - node.line
  6366
  6367	        # The first branch handles reference an overloaded function variant inside itself,
  6368	        # this is a corner case where mypy technically deviates from runtime name resolution,
  6369	        # but it is fine because we want an overloaded function to be treated as a single unit.
  6370	        if self.is_overloaded_item(node, self.statement):
  6371	            return False
  6372	        elif isinstance(node, Decorator) and not node.is_overload:
  6373	            return line_diff > len(node.original_decorators)
  6374	        else:
  6375	            return line_diff > 0

  6377	    def is_textually_before_class(self, node: SymbolNode | None) -> bool:
  6378	        """Similar to above, but check if a node is defined before current class."""
  6379	        assert self.type is not None
  6380	        if node is None:
  6381	            return False
  6382	        return node.line < self.type.defn.line

  6384	    def is_overloaded_item(self, node: SymbolNode, statement: Statement) -> bool:
  6385	        """Check whether the function belongs to the overloaded variants"""
  6386	        if isinstance(node, OverloadedFuncDef) and isinstance(statement, FuncDef):
  6387	            in_items = statement in {
  6388	                item.func if isinstance(item, Decorator) else item for item in node.items
  6389	            }
  6390	            in_impl = node.impl is not None and (
  6391	                (isinstance(node.impl, Decorator) and statement is node.impl.func)
  6392	                or statement is node.impl
  6393	            )

  6395	        return False

  6402	    def lookup_qualified(
  6403	        self, name: str, ctx: Context, suppress_errors: bool = False
  6404	    ) -> SymbolTableNode | None:
  6405	        """Lookup a qualified name in all activate namespaces.
  6406
  6407	        Note that the result may contain a PlaceholderNode. The caller may
  6408	        want to defer in that case.
  6409
  6410	        Generate an error if the name is not defined unless suppress_errors
  6411	        is true or the current namespace is incomplete. In the latter case
  6412	        defer.
  6413	        """
  6414	        if "." not in name:
  6415	            # Simple case: look up a short name.
  6416	            return self.lookup(name, ctx, suppress_errors=suppress_errors)
  6417	        parts = name.split(".")
  6418	        namespace = self.cur_mod_id
  6419	        sym = self.lookup(parts[0], ctx, suppress_errors=suppress_errors)
  6420	        if sym:
  6421	            for i in range(1, len(parts)):
  6422	                node = sym.node
  6423	                part = parts[i]
  6424	                if isinstance(node, TypeInfo):
  6425	                    nextsym = node.get(part)
  6426	                elif isinstance(node, MypyFile):
  6427	                    nextsym = self.get_module_symbol(node, part)
  6428	                    namespace = node.fullname
  6429	                elif isinstance(node, PlaceholderNode):
  6430	                    return sym
  6431	                elif isinstance(node, TypeAlias) and node.no_args:
  6432	                    assert isinstance(node.target, ProperType)
  6433	                    if isinstance(node.target, Instance):
  6434	                        nextsym = node.target.type.get(part)
  6435	                    else:
  6436	                        nextsym = None
  6437	                else:
  6438	                    if isinstance(node, Var):
  6439	                        typ = get_proper_type(node.type)
  6440	                        if isinstance(typ, AnyType):
  6441	                            # Allow access through Var with Any type without error.
  6442	                            return self.implicit_symbol(sym, name, parts[i:], typ)
  6443	                    # This might be something like valid `P.args` or invalid `P.__bound__` access.
  6444	                    # Important note that `ParamSpecExpr` is also ignored in other places.
  6445	                    # See https://github.com/python/mypy/pull/13468
  6446	                    if isinstance(node, ParamSpecExpr) and part in ("args", "kwargs"):
  6447	                        return None
  6448	                    # Lookup through invalid node, such as variable or function
  6449	                    nextsym = None
  6450	                if not nextsym or nextsym.module_hidden:
  6451	                    if not suppress_errors:
  6452	                        self.name_not_defined(name, ctx, namespace=namespace)
  6453	                    return None
  6454	                sym = nextsym
  6455	        return sym

  6467	    def get_module_symbol(self, node: MypyFile, name: str) -> SymbolTableNode | None:

  6472	        module = node.fullname
  6473	        names = node.names
  6474	        sym = names.get(name)
  6475	        if not sym:
  6476	            fullname = module + "." + name

  6546	    def lookup_fully_qualified_or_none(self, fullname: str) -> SymbolTableNode | None:
  6547	        """Lookup a fully qualified name that refers to a module-level definition.
  6548
  6549	        Don't assume that the name is defined. This happens in the global namespace --
  6550	        the local module namespace is ignored. This does not dereference indirect
  6551	        refs.
  6552
  6553	        Note that this can't be used for names nested in class namespaces.
  6554	        """
  6555	        # TODO: unify/clean-up/simplify lookup methods, see #4157.
  6556	        module, name = fullname.rsplit(".", maxsplit=1)
  6557
  6558	        if module in self.modules:
  6559	            # If the module exists, look up the name in the module.
  6560	            # This is the common case.
  6561	            filenode = self.modules[module]
  6562	            result = filenode.names.get(name)
  6563	            if result is None and self.is_incomplete_namespace(module):
  6564	                # TODO: More explicit handling of incomplete refs?
  6565	                self.record_incomplete_ref()
  6566	            return result
  6567	        else:
  6568	            # Else, try to find the longest prefix of the module name that is in the modules dictionary.
  6569	            splitted_modules = fullname.split(".")
  6570	            names = []
  6571

  6701	    def add_symbol_table_node(
  6702	        self,
  6703	        name: str,
  6704	        symbol: SymbolTableNode,
  6705	        context: Context | None = None,
  6706	        can_defer: bool = True,
  6707	        escape_comprehensions: bool = False,
  6708	        no_progress: bool = False,
  6709	        type_param: bool = False,
  6710	    ) -> bool:
  6711	        """Add symbol table node to the currently active symbol table.
  6712
  6713	        Return True if we actually added the symbol, or False if we refused
  6714	        to do so (because something is not ready or it was a no-op).
  6715
  6716	        Generate an error if there is an invalid redefinition.
  6717
  6718	        If context is None, unconditionally add node, since we can't report
  6719	        an error. Note that this is used by plugins to forcibly replace nodes!
  6720
  6721	        TODO: Prevent plugins from replacing nodes, as it could cause problems?
  6722
  6723	        Args:
  6724	            name: short name of symbol
  6725	            symbol: Node to add
  6726	            can_defer: if True, defer current target if adding a placeholder
  6727	            context: error context (see above about None value)
  6728	        """
  6729	        names = self.current_symbol_table(
  6730	            escape_comprehensions=escape_comprehensions, type_param=type_param
  6731	        )
  6732	        existing = names.get(name)
  6733	        if isinstance(symbol.node, PlaceholderNode) and can_defer:
  6734	            if context is not None:
  6735	                self.process_placeholder(name, "name", context)
  6736	            else:
  6737	                # see note in docstring describing None contexts
  6738	                self.defer()
  6739	        if (
  6740	            existing is not None
  6741	            and context is not None
  6742	            and not is_valid_replacement(existing, symbol)
  6743	        ):
  6744	            # There is an existing node, so this may be a redefinition.
  6745	            # If the new node points to the same node as the old one,
  6746	            # or if both old and new nodes are placeholders, we don't
  6747	            # need to do anything.
  6748	            old = existing.node
  6749	            new = symbol.node
  6750	            if isinstance(new, PlaceholderNode):
  6751	                # We don't know whether this is okay. Let's wait until the next iteration.
  6752	                return False
  6753	            if not is_same_symbol(old, new):
  6754	                if isinstance(new, (FuncDef, Decorator, OverloadedFuncDef, TypeInfo)):
  6755	                    self.add_redefinition(names, name, symbol)
  6756	                if not (isinstance(new, (FuncDef, Decorator)) and self.set_original_def(old, new)):
  6757	                    self.name_already_defined(name, context, existing)
  6758	        elif name not in self.missing_names[-1] and "*" not in self.missing_names[-1]:
  6759	            names[name] = symbol
  6760	            if not no_progress:
  6761	                self.progress = True
  6762	            return True
  6763	        return False

  6765	    def add_redefinition(self, names: SymbolTable, name: str, symbol: SymbolTableNode) -> None:
  6766	        """Add a symbol table node that reflects a redefinition as a function or a class.
  6767
  6768	        Redefinitions need to be added to the symbol table so that they can be found
  6769	        through AST traversal, but they have dummy names of form 'name-redefinition[N]',
  6770	        where N ranges over 2, 3, ... (omitted for the first redefinition).
  6771
  6772	        Note: we always store redefinitions independently of whether they are valid or not
  6773	        (so they will be semantically analyzed), the caller should give an error for invalid
  6774	        redefinitions (such as e.g. variable redefined as a class).
  6775	        """
  6776	        i = 1
  6777	        # Don't serialize redefined nodes. They are likely to have
  6778	        # busted internal references which can cause problems with
  6779	        # serialization and they can't have any external references to
  6780	        # them.
  6781	        symbol.no_serialize = True
  6782	        while True:
  6783	            if i == 1:
  6784	                new_name = f"{name}-redefinition"
  6785	            else:
  6786	                new_name = f"{name}-redefinition{i}"
  6787	            existing = names.get(new_name)
  6788	            if existing is None:
  6789	                names[new_name] = symbol
  6790	                return
  6791	            elif existing.node is symbol.node:
  6792	                # Already there
  6793	                return
  6794	            i += 1

  6926	    def defer(self, debug_context: Context | None = None, force_progress: bool = False) -> None:
  6927	        """Defer current analysis target to be analyzed again.
  6928
  6929	        This must be called if something in the current target is
  6930	        incomplete or has a placeholder node. However, this must *not*
  6931	        be called during the final analysis iteration! Instead, an error
  6932	        should be generated. Often 'process_placeholder' is a good
  6933	        way to either defer or generate an error.
  6934
  6935	        NOTE: Some methods, such as 'anal_type', 'mark_incomplete' and

  6939	        assert not self.final_iteration, "Must not defer during final iteration"
  6940	        if force_progress:

  6945	            # similar situations (recursive named tuples etc).
  6946	            self.progress = True
  6947	        self.deferred = True
  6948	        # Store debug info for this deferral.
  6949	        line = (
  6950	            debug_context.line if debug_context else self.statement.line if self.statement else -1

  7093	        self, escape_comprehensions: bool = False, type_param: bool = False
  7094	    ) -> SymbolTable:
  7095	        if type_param and self.scope_stack[-1] == SCOPE_ANNOTATION:
  7096	            n = self.locals[-1]
  7097	            assert n is not None
  7098	            return n
  7099	        elif self.is_func_scope():
  7100	            if self.scope_stack[-1] == SCOPE_ANNOTATION:
  7101	                n = self.locals[-2]
  7102	            else:
  7103	                n = self.locals[-1]
  7104	            assert n is not None
  7105	            if escape_comprehensions:
  7106	                assert len(self.locals) == len(self.scope_stack)
  7107	                # Retrieve the symbol table from the enclosing non-comprehension scope.
  7108	                for i, scope_type in enumerate(reversed(self.scope_stack)):
  7109	                    if scope_type != SCOPE_COMPREHENSION:
  7110	                        if i == len(self.locals) - 1:  # The last iteration.
  7111	                            # The caller of the comprehension is in the global space.
  7112	                            names = self.globals
  7113	                        else:
  7114	                            names_candidate = self.locals[-1 - i]
  7115	                            assert (
  7116	                                names_candidate is not None
  7117	                            ), "Escaping comprehension from invalid scope"
  7118	                            names = names_candidate
  7119	                        break
  7120	                else:
  7121	                    assert False, "Should have at least one non-comprehension scope"
  7122	            else:
  7123	                names = n
  7124	            assert names is not None
  7125	        elif self.type is not None:
  7126	            names = self.type.names
  7127	        else:
  7128	            names = self.globals
  7129	        return names

  7223	    def in_checked_function(self) -> bool:
  7224	        """Should we type-check the current function?
  7225
  7226	        - Yes if --check-untyped-defs is set.
  7227	        - Yes outside functions.
  7228	        - Yes in annotated functions.
  7229	        - No otherwise.
  7230	        """
  7231	        if self.options.check_untyped_defs or not self.function_stack:
  7232	            return True
  7233
  7234	        current_index = len(self.function_stack) - 1
  7235	        while current_index >= 0:
  7236	            current_func = self.function_stack[current_index]
  7237	            if not isinstance(current_func, LambdaExpr):
  7238	                return not current_func.is_dynamic()
  7239
  7240	            # Special case, `lambda` inherits the "checked" state from its parent.
  7241	            # Because `lambda` itself cannot be annotated.
  7242	            # `lambdas` can be deeply nested, so we try to find at least one other parent.
  7243	            current_index -= 1
  7244
  7245	        # This means that we only have a stack of `lambda` functions,
  7246	        # no regular functions.
  7247	        return True

  7396	        self,
  7397	        typ: Type,
  7398	        *,
  7399	        tvar_scope: TypeVarLikeScope | None = None,
  7400	        allow_tuple_literal: bool = False,
  7401	        allow_unbound_tvars: bool = False,
  7402	        allow_placeholder: bool = False,
  7403	        allow_typed_dict_special_forms: bool = False,
  7404	        allow_final: bool = False,
  7405	        allow_param_spec_literals: bool = False,
  7406	        allow_unpack: bool = False,
  7407	        report_invalid_types: bool = True,
  7408	        prohibit_self_type: str | None = None,
  7409	        prohibit_special_class_field_types: str | None = None,
  7410	        allow_type_any: bool = False,
  7411	    ) -> Type | None:
  7412	        """Semantically analyze a type.
  7413
  7414	        Args:
  7415	            typ: Type to analyze (if already analyzed, this is a no-op)
  7416	            allow_placeholder: If True, may return PlaceholderType if
  7417	                encountering an incomplete definition
  7418
  7419	        Return None only if some part of the type couldn't be bound *and* it
  7420	        referred to an incomplete namespace or definition. In this case also
  7421	        defer as needed. During a final iteration this won't return None;
  7422	        instead report an error if the type can't be analyzed and return
  7423	        AnyType.
  7424
  7425	        In case of other errors, report an error message and return AnyType.
  7426
  7427	        NOTE: The caller shouldn't defer even if this returns None or a
  7428	              placeholder type.
  7429	        """
  7430	        has_self_type = find_self_type(
  7431	            typ, lambda name: self.lookup_qualified(name, typ, suppress_errors=True)
  7432	        )
  7433	        if has_self_type and self.type and prohibit_self_type is None:
  7434	            self.setup_self_type()
  7435	        a = self.type_analyzer(
  7436	            tvar_scope=tvar_scope,
  7437	            allow_unbound_tvars=allow_unbound_tvars,
  7438	            allow_tuple_literal=allow_tuple_literal,
  7439	            allow_placeholder=allow_placeholder,
  7440	            allow_typed_dict_special_forms=allow_typed_dict_special_forms,
  7441	            allow_final=allow_final,
  7442	            allow_param_spec_literals=allow_param_spec_literals,
  7443	            allow_unpack=allow_unpack,
  7444	            report_invalid_types=report_invalid_types,
  7445	            prohibit_self_type=prohibit_self_type,
  7446	            prohibit_special_class_field_types=prohibit_special_class_field_types,
  7447	            allow_type_any=allow_type_any,
  7448	        )
  7449	        tag = self.track_incomplete_refs()
  7450	        typ = typ.accept(a)
  7451	        if self.found_incomplete_ref(tag):
  7452	            # Something could not be bound yet.
  7453	            return None
  7454	        self.add_type_alias_deps(a.aliases_used)

  7529	    def parse_dataclass_transform_spec(self, call: CallExpr) -> DataclassTransformSpec:
  7530	        """Build a DataclassTransformSpec from the arguments passed to the given call to
  7531	        typing.dataclass_transform."""
  7532	        parameters = DataclassTransformSpec()
  7533	        for name, value in zip(call.arg_names, call.args):
  7534	            # Skip any positional args. Note that any such args are invalid, but we can rely on

  7537	                continue

  7539	            # field_specifiers is currently the only non-boolean argument; check for it first so
  7540	            # so the rest of the block can fail through to handling booleans

  7543	                    value

  7547	            boolean = require_bool_literal_argument(self, value, name)
  7548	            if boolean is None:
  7549	                continue

  7551	            if name == "eq_default":

  7553	            elif name == "order_default":
  7554	                parameters.order_default = boolean
  7555	            elif name == "kw_only_default":
  7556	                parameters.kw_only_default = boolean
  7557	            elif name == "frozen_default":
  7558	                parameters.frozen_default = boolean
  7559	            else:
  7560	                self.fail(f'Unrecognized dataclass_transform parameter "{name}"', call)
