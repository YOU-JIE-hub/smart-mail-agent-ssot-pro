    72	    sender: str, subject: str, content: str, attachments: list[str]

    75	    score = 0
    76	    reasons: list[str] = []

    78	    sender = (sender or "").strip().lower()

    90	    kw = r.get("keywords", {})
    91	    for k, s in kw.items():
    92	        if k in subject_u or k in content_u:
    93	            score += int(s)
    94	            reasons.append(f"keyword:{k}")
    95

    97	    w = r.get("weights", {})

   100	    susp_tlds = {t.lower() for t in r.get("suspicious_tlds", [])}
