    37	    return left.accept(RTSubtypeVisitor(right))

    40	class RTSubtypeVisitor(RTypeVisitor[bool]):
    41	    """Is left a runtime subtype of right?
    42
    43	    A few special cases such as right being 'object' are handled in
    44	    is_runtime_subtype and don't need to be covered here.

    47	    def __init__(self, right: RType) -> None:
    48	        self.right = right

    50	    def visit_rinstance(self, left: RInstance) -> bool:
    51	        return is_subtype(left, self.right)
    52
    53	    def visit_runion(self, left: RUnion) -> bool:
    54	        return not self.right.is_unboxed and is_subtype(left, self.right)
    55
    56	    def visit_rprimitive(self, left: RPrimitive) -> bool:
    57	        if is_short_int_rprimitive(left) and is_int_rprimitive(self.right):
    58	            return True
    59	        if is_bit_rprimitive(left) and is_bool_rprimitive(self.right):
    60	            return True
    61	        return left is self.right
