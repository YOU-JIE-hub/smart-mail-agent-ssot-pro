   251	def validate_basetemp(path: str) -> str:
   252	    # GH 7119
   253	    msg = "basetemp must not be empty, the current working directory or any parent directory of it"
   254
   255	    # empty path
   256	    if not path:
   257	        raise argparse.ArgumentTypeError(msg)
   258
   259	    def is_ancestor(base: Path, query: Path) -> bool:
   260	        """Return whether query is an ancestor of base."""
   261	        if base == query:
   262	            return True
   263	        return query in base.parents
   264
   265	    # check if path is an ancestor of cwd
   266	    if is_ancestor(Path.cwd(), Path(path).absolute()):
   267	        raise argparse.ArgumentTypeError(msg)

   269	    # check symlinks for ancestors
   270	    if is_ancestor(Path.cwd().resolve(), Path(path).resolve()):
   271	        raise argparse.ArgumentTypeError(msg)
   272
   273	    return path

   276	def wrap_session(
   277	    config: Config, doit: Callable[[Config, Session], int | ExitCode | None]
   278	) -> int | ExitCode:
   279	    """Skeleton command line program."""
   280	    session = Session.from_config(config)
   281	    session.exitstatus = ExitCode.OK
   282	    initstate = 0
   283	    try:
   284	        try:
   285	            config._do_configure()
   286	            initstate = 1
   287	            config.hook.pytest_sessionstart(session=session)
   288	            initstate = 2
   289	            session.exitstatus = doit(config, session) or 0
   290	        except UsageError:
   291	            session.exitstatus = ExitCode.USAGE_ERROR
   292	            raise
   293	        except Failed:
   294	            session.exitstatus = ExitCode.TESTS_FAILED
   295	        except (KeyboardInterrupt, exit.Exception):
   296	            excinfo = _pytest._code.ExceptionInfo.from_current()
   297	            exitstatus: int | ExitCode = ExitCode.INTERRUPTED
   298	            if isinstance(excinfo.value, exit.Exception):
   299	                if excinfo.value.returncode is not None:
   300	                    exitstatus = excinfo.value.returncode
   301	                if initstate < 2:
   302	                    sys.stderr.write(f"{excinfo.typename}: {excinfo.value.msg}\n")
   303	            config.hook.pytest_keyboard_interrupt(excinfo=excinfo)
   304	            session.exitstatus = exitstatus
   305	        except BaseException:
   306	            session.exitstatus = ExitCode.INTERNAL_ERROR
   307	            excinfo = _pytest._code.ExceptionInfo.from_current()
   308	            try:
   309	                config.notify_exception(excinfo, config.option)
   310	            except exit.Exception as exc:
   311	                if exc.returncode is not None:
   312	                    session.exitstatus = exc.returncode
   313	                sys.stderr.write(f"{type(exc).__name__}: {exc}\n")
   314	            else:
   315	                if isinstance(excinfo.value, SystemExit):
   316	                    sys.stderr.write("mainloop: caught unexpected SystemExit!\n")
   317
   318	    finally:
   319	        # Explicitly break reference cycle.
   320	        excinfo = None  # type: ignore
   321	        os.chdir(session.startpath)
   322	        if initstate >= 2:
   323	            try:
   324	                config.hook.pytest_sessionfinish(
   325	                    session=session, exitstatus=session.exitstatus
   326	                )
   327	            except exit.Exception as exc:
   328	                if exc.returncode is not None:
   329	                    session.exitstatus = exc.returncode
   330	                sys.stderr.write(f"{type(exc).__name__}: {exc}\n")
   331	        config._ensure_unconfigure()
   332	    return session.exitstatus
   333

   335	def pytest_cmdline_main(config: Config) -> int | ExitCode:
   336	    return wrap_session(config, _main)

   339	def _main(config: Config, session: Session) -> int | ExitCode | None:
   340	    """Default command line protocol for initialization, session,
   341	    running tests and reporting."""
   342	    config.hook.pytest_collection(session=session)
   343	    config.hook.pytest_runtestloop(session=session)
   344
   345	    if session.testsfailed:
   346	        return ExitCode.TESTS_FAILED
   347	    elif session.testscollected == 0:
   348	        return ExitCode.NO_TESTS_COLLECTED
   349	    return None

   352	def pytest_collection(session: Session) -> None:
   353	    session.perform_collect()

   356	def pytest_runtestloop(session: Session) -> bool:
   357	    if session.testsfailed and not session.config.option.continue_on_collection_errors:
   358	        raise session.Interrupted(
   359	            f"{session.testsfailed} error{'s' if session.testsfailed != 1 else ''} during collection"
   360	        )
   361
   362	    if session.config.option.collectonly:
   363	        return True
   364
   365	    for i, item in enumerate(session.items):
   366	        nextitem = session.items[i + 1] if i + 1 < len(session.items) else None
   367	        item.config.hook.pytest_runtest_protocol(item=item, nextitem=nextitem)
   368	        if session.shouldfail:
   369	            raise session.Failed(session.shouldfail)
   370	        if session.shouldstop:
   371	            raise session.Interrupted(session.shouldstop)
   372	    return True

   375	def _in_venv(path: Path) -> bool:
   376	    """Attempt to detect if ``path`` is the root of a Virtual Environment by
   377	    checking for the existence of the pyvenv.cfg file.
   378
   379	    [https://peps.python.org/pep-0405/]
   380
   381	    For regression protection we also check for conda environments that do not include pyenv.cfg yet --
   382	    https://github.com/conda/conda/issues/13337 is the conda issue tracking adding pyenv.cfg.
   383
   384	    Checking for the `conda-meta/history` file per https://github.com/pytest-dev/pytest/issues/12652#issuecomment-2246336902.
   385
   386	    """
   387	    try:
   388	        return (
   389	            path.joinpath("pyvenv.cfg").is_file()
   390	            or path.joinpath("conda-meta", "history").is_file()
   391	        )
   392	    except OSError:
   393	        return False

   396	def pytest_ignore_collect(collection_path: Path, config: Config) -> bool | None:
   397	    if collection_path.name == "__pycache__":
   398	        return True
   399
   400	    ignore_paths = config._getconftest_pathlist(
   401	        "collect_ignore", path=collection_path.parent
   402	    )
   403	    ignore_paths = ignore_paths or []
   404	    excludeopt = config.getoption("ignore")
   405	    if excludeopt:
   406	        ignore_paths.extend(absolutepath(x) for x in excludeopt)
   407
   408	    if collection_path in ignore_paths:
   409	        return True
   410
   411	    ignore_globs = config._getconftest_pathlist(
   412	        "collect_ignore_glob", path=collection_path.parent
   413	    )
   414	    ignore_globs = ignore_globs or []
   415	    excludeglobopt = config.getoption("ignore_glob")
   416	    if excludeglobopt:
   417	        ignore_globs.extend(absolutepath(x) for x in excludeglobopt)
   418
   419	    if any(fnmatch.fnmatch(str(collection_path), str(glob)) for glob in ignore_globs):
   420	        return True
   421
   422	    allow_in_venv = config.getoption("collect_in_virtualenv")
   423	    if not allow_in_venv and _in_venv(collection_path):
   424	        return True
   425
   426	    if collection_path.is_dir():
   427	        norecursepatterns = config.getini("norecursedirs")
   428	        if any(fnmatch_ex(pat, collection_path) for pat in norecursepatterns):
   429	            return True
   430
   431	    return None

   441	    deselect_prefixes = tuple(config.getoption("deselect") or [])
   442	    if not deselect_prefixes:
   443	        return

   445	    remaining = []
   446	    deselected = []

   452
   453	    if deselected:
   454	        config.hook.pytest_deselected(items=deselected)
   455	        items[:] = remaining

   496	class Dir(nodes.Directory):
   497	    """Collector of files in a file system directory.
   498
   499	    .. versionadded:: 8.0
   500
   501	    .. note::
   502
   503	        Python directories with an `__init__.py` file are instead collected by
   504	        :class:`~pytest.Package` by default. Both are :class:`~pytest.Directory`
   505	        collectors.
   506	    """
   507
   508	    @classmethod
   509	    def from_parent(  # type: ignore[override]
   510	        cls,
   511	        parent: nodes.Collector,
   512	        *,
   513	        path: Path,
   514	    ) -> Self:
   515	        """The public constructor.
   516
   517	        :param parent: The parent collector of this Dir.
   518	        :param path: The directory's path.
   519	        :type path: pathlib.Path
   520	        """
   521	        return super().from_parent(parent=parent, path=path)

   523	    def collect(self) -> Iterable[nodes.Item | nodes.Collector]:
   524	        config = self.config
   525	        col: nodes.Collector | None
   526	        cols: Sequence[nodes.Collector]
   527	        ihook = self.ihook
   528	        for direntry in scandir(self.path):
   529	            if direntry.is_dir():
   530	                path = Path(direntry.path)
   531	                if not self.session.isinitpath(path, with_parents=True):
   532	                    if ihook.pytest_ignore_collect(collection_path=path, config=config):
   533	                        continue
   534	                col = ihook.pytest_collect_directory(path=path, parent=self)
   535	                if col is not None:
   536	                    yield col
   537
   538	            elif direntry.is_file():
   539	                path = Path(direntry.path)
   540	                if not self.session.isinitpath(path):
   541	                    if ihook.pytest_ignore_collect(collection_path=path, config=config):
   542	                        continue
   543	                cols = ihook.pytest_collect_file(file_path=path, parent=self)
   544	                yield from cols

   667	        self,
   668	        path: str | os.PathLike[str],
   669	        *,
   670	        with_parents: bool = False,
   671	    ) -> bool:
   672	        """Is path an initial path?
   673
   674	        An initial path is a path explicitly given to pytest on the command
   675	        line.
   676
   677	        :param with_parents:
   678	            If set, also return True if the path is a parent of an initial path.
   679
   680	        .. versionchanged:: 8.0
   681	            Added the ``with_parents`` parameter.
   682	        """
   683	        # Optimization: Path(Path(...)) is much slower than isinstance.
   684	        path_ = path if isinstance(path, Path) else Path(path)
   685	        if with_parents:
   686	            return path_ in self._initialpaths_with_parents
   687	        else:
   688	            return path_ in self._initialpaths

   690	    def gethookproxy(self, fspath: os.PathLike[str]) -> pluggy.HookRelay:
   691	        # Optimization: Path(Path(...)) is much slower than isinstance.
   692	        path = fspath if isinstance(fspath, Path) else Path(fspath)
   693	        pm = self.config.pluginmanager
   694	        # Check if we have the common case of running
   695	        # hooks with all conftest.py files.
   696	        my_conftestmodules = pm._getconftestmodules(path)
   697	        remove_mods = pm._conftest_plugins.difference(my_conftestmodules)
   698	        proxy: pluggy.HookRelay
   699	        if remove_mods:
   700	            # One or more conftests are not in use at this path.
   701	            proxy = PathAwareHookProxy(FSHookProxy(pm, remove_mods))  # type: ignore[arg-type,assignment]
   702	        else:
   703	            # All plugins are active for this fspath.
   704	            proxy = self.config.hook
   705	        return proxy

   707	    def _collect_path(
   708	        self,
   709	        path: Path,
   710	        path_cache: dict[Path, Sequence[nodes.Collector]],
   711	    ) -> Sequence[nodes.Collector]:
   712	        """Create a Collector for the given path.
   713
   714	        `path_cache` makes it so the same Collectors are returned for the same
   715	        path.
   716	        """
   717	        if path in path_cache:
   718	            return path_cache[path]
   719
   720	        if path.is_dir():
   721	            ihook = self.gethookproxy(path.parent)
   722	            col: nodes.Collector | None = ihook.pytest_collect_directory(
   723	                path=path, parent=self
   724	            )
   725	            cols: Sequence[nodes.Collector] = (col,) if col is not None else ()
   726
   727	        elif path.is_file():
   728	            ihook = self.gethookproxy(path)
   729	            cols = ihook.pytest_collect_file(file_path=path, parent=self)
   730
   731	        else:
   732	            # Broken symlink or invalid/missing file.
   733	            cols = ()
   734
   735	        path_cache[path] = cols
   736	        return cols

   748	    def perform_collect(
   749	        self, args: Sequence[str] | None = None, genitems: bool = True
   750	    ) -> Sequence[nodes.Item | nodes.Collector]:
   751	        """Perform the collection phase for this session.
   752
   753	        This is called by the default :hook:`pytest_collection` hook
   754	        implementation; see the documentation of this hook for more details.
   755	        For testing purposes, it may also be called directly on a fresh
   756	        ``Session``.

   758	        This function normally recursively expands any collectors collected
   759	        from the session to their items, and only items are returned. For
   760	        testing purposes, this may be suppressed by passing ``genitems=False``,
   761	        in which case the return value contains these collectors unexpanded,
   762	        and ``session.items`` is empty.

   765	            args = self.config.args

   768	        self.trace.root.indent += 1

   770	        hook = self.config.hook
   771
   772	        self._notfound = []
   773	        self._initial_parts = []
   774	        self._collection_cache = {}
   775	        self.items = []
   776	        items: Sequence[nodes.Item | nodes.Collector] = self.items
   777	        try:
   778	            initialpaths: list[Path] = []
   779	            initialpaths_with_parents: list[Path] = []
   780	            for arg in args:
   781	                collection_argument = resolve_collection_argument(
   782	                    self.config.invocation_params.dir,
   783	                    arg,
   784	                    as_pypath=self.config.option.pyargs,
   785	                )

   787	                initialpaths.append(collection_argument.path)
   788	                initialpaths_with_parents.append(collection_argument.path)
   789	                initialpaths_with_parents.extend(collection_argument.path.parents)
   790	            self._initialpaths = frozenset(initialpaths)
   791	            self._initialpaths_with_parents = frozenset(initialpaths_with_parents)
   792
   793	            rep = collect_one_node(self)
   794	            self.ihook.pytest_collectreport(report=rep)
   795	            self.trace.root.indent -= 1
   796	            if self._notfound:
   797	                errors = []
   798	                for arg, collectors in self._notfound:
   799	                    if collectors:
   800	                        errors.append(
   801	                            f"not found: {arg}\n(no match in any of {collectors!r})"
   802	                        )
   803	                    else:
   804	                        errors.append(f"found no collectors for {arg}")

   808	            if not genitems:

   844	        # This is a cache for the root directories of the initial paths.
   845	        # We can't use collection_cache for Session because of its special
   846	        # role as the bootstrapping collector.

   853	            self.trace.root.indent += 1

   855	            argpath = collection_argument.path

   857	            module_name = collection_argument.module_name

   860	            if argpath.is_dir():
   861	                assert not names, f"invalid arg {(argpath, names)!r}"

   865	            #   /a/b/c.py -> [/, /a, /a/b, /a/b/c.py]
   866	            if module_name is None:

   868	                for path in argpath.parents:

   874	                # name. Paths beyond the package hierarchy are not included.
   875	                module_name_parts = module_name.split(".")
   876	                for i, path in enumerate(argpath.parents, 2):

   878	                        break

   882	            # and discarding all nodes which don't match the level's part.
   883	            any_matched_in_initial_part = False
   884	            notfound_collectors = []

   886	                (self, [*paths, *names])

   888	            while work:

   892	                if not matchparts:

   895	                    continue

   897	                # Should have been matched by now, discard.
   898	                if not isinstance(matchnode, nodes.Collector):
   899	                    continue

   902	                # Collecting Session (self) is done directly to avoid endless
   903	                # recursion to this function.

   905	                if isinstance(matchnode, Session):
   906	                    assert isinstance(matchparts[0], Path)
   907	                    subnodes = matchnode._collect_path(matchparts[0], path_cache)
   908	                else:

   910	                    # times on the command line should not be deduplicated.
   911	                    handle_dupes = not (
   912	                        len(matchparts) == 1
   913	                        and isinstance(matchparts[0], Path)
   914	                        and matchparts[0].is_file()
   915	                    )
   916	                    rep, duplicate = self._collect_one_node(matchnode, handle_dupes)
   917	                    if not duplicate and not rep.passed:
   918	                        # Report collection failures here to avoid failing to
   919	                        # run some test specified in the command line because
   920	                        # the module could not be imported (#134).

   922	                    if not rep.passed:

   927	                any_matched_in_collector = False
   928	                for node in reversed(subnodes):
   929	                    # Path part e.g. `/a/b/` in `/a/b/test_file.py::TestIt::test_it`.
   930	                    if isinstance(matchparts[0], Path):
   931	                        is_match = node.path == matchparts[0]
   932	                        if sys.platform == "win32" and not is_match:
   933	                            # In case the file paths do not match, fallback to samefile() to
   934	                            # account for short-paths on Windows (#11895).

   936	                            # We don't want to match links to the current node,
   937	                            # otherwise we would match the same file more than once (#12039).
   938	                            is_match = same_file and (
   939	                                os.path.islink(node.path)
   940	                                == os.path.islink(matchparts[0])
   941	                            )
   942
   943	                    # Name part e.g. `TestIt` in `/a/b/test_file.py::TestIt::test_it`.
   944	                    else:
   945	                        # TODO: Remove parametrized workaround once collection structure contains
   946	                        # parametrization.
   947	                        is_match = (
   948	                            node.name == matchparts[0]
   949	                            or node.name.split("[")[0] == matchparts[0]
   950	                        )
   951	                    if is_match:
   952	                        work.append((node, matchparts[1:]))
   953	                        any_matched_in_collector = True

   955	                if not any_matched_in_collector:
   956	                    notfound_collectors.append(matchnode)

   958	            if not any_matched_in_initial_part:

   960	                self._notfound.append((report_arg, notfound_collectors))

   962	            self.trace.root.indent -= 1

   155
   156	    Given a list of names, matches any substring of one of these names. The
   157	    string inclusion check is case-insensitive.
   158
   159	    Will match on the name of colitem, including the names of its parents.
   160	    Only matches names of items which are either a :class:`Class` or a
   161	    :class:`Function`.

   164	    any item, as well as names directly assigned to test functions.

   172	    def from_item(cls, item: Item) -> KeywordMatcher:
   173	        mapped_names = set()
   174
   175	        # Add the names of the current item and any parent items,
   176	        # except the Session and root Directory's which are not
   177	        # interesting for matching.
   178	        import pytest
   179
   180	        for node in item.listchain():
   181	            if isinstance(node, pytest.Session):
   182	                continue

   184	                node.parent, pytest.Session

   186	                continue
   187	            mapped_names.add(node.name)

   190	        mapped_names.update(item.listextrakeywords())

   192	        # Add the names attached to the current function through direct assignment.
   193	        function_obj = getattr(item, "function", None)
   194	        if function_obj:
   195	            mapped_names.update(function_obj.__dict__)

   197	        # Add the markers to the keywords as we no longer handle them correctly.
   198	        mapped_names.update(mark.name for mark in item.iter_markers())

    60	def resolve(name: str) -> object:
    61	    # Simplified from zope.dottedname.
    62	    parts = name.split(".")
    63
    64	    used = parts.pop(0)
    65	    found: object = __import__(used)
    66	    for part in parts:
    67	        used += "." + part
    68	        try:
    69	            found = getattr(found, part)
    70	        except AttributeError:
    71	            pass
    72	        else:
    73	            continue
    74	        # We use explicit un-nesting of the handling block in order
    75	        # to avoid nested exceptions.
    76	        try:
    77	            __import__(used)
    78	        except ImportError as ex:
    79	            expected = str(ex).split()[-1]
    80	            if expected == used:
    81	                raise
    82	            else:
    83	                raise ImportError(f"import error in {used}: {ex}") from ex
