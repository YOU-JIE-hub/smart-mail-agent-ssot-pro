  1405	class Instance(ProperType):
  1406	    """An instance type of form C[T1, ..., Tn].
  1407
  1408	    The list of type variables may be empty.
  1409
  1410	    Several types have fallbacks to `Instance`, because in Python everything is an object
  1411	    and this concept is impossible to express without intersection types. We therefore use
  1412	    fallbacks for all "non-special" (like UninhabitedType, ErasedType etc) types.
  1413	    """
  1414
  1415	    __slots__ = ("type", "args", "invalid", "type_ref", "last_known_value", "_hash", "extra_attrs")
  1416
  1417	    def __init__(
  1418	        self,
  1419	        typ: mypy.nodes.TypeInfo,
  1420	        args: Sequence[Type],
  1421	        line: int = -1,
  1422	        column: int = -1,
  1423	        *,
  1424	        last_known_value: LiteralType | None = None,
  1425	        extra_attrs: ExtraAttrs | None = None,
  1426	    ) -> None:
  1427	        super().__init__(line, column)
  1428	        self.type = typ
  1429	        self.args = tuple(args)
  1430	        self.type_ref: str | None = None
  1431
  1432	        # True if recovered after incorrect number of type arguments error
  1433	        self.invalid = False
  1434
  1435	        # This field keeps track of the underlying Literal[...] value associated with
  1436	        # this instance, if one is known.
  1437	        #
  1438	        # This field is set whenever possible within expressions, but is erased upon
  1439	        # variable assignment (see erasetype.remove_instance_last_known_values) unless
  1440	        # the variable is declared to be final.
  1441	        #
  1442	        # For example, consider the following program:
  1443	        #
  1444	        #     a = 1
  1445	        #     b: Final[int] = 2
  1446	        #     c: Final = 3
  1447	        #     print(a + b + c + 4)
  1448	        #
  1449	        # The 'Instance' objects associated with the expressions '1', '2', '3', and '4' will
  1450	        # have last_known_values of type Literal[1], Literal[2], Literal[3], and Literal[4]
  1451	        # respectively. However, the Instance object assigned to 'a' and 'b' will have their
  1452	        # last_known_value erased: variable 'a' is mutable; variable 'b' was declared to be
  1453	        # specifically an int.
  1454	        #
  1455	        # Or more broadly, this field lets this Instance "remember" its original declaration
  1456	        # when applicable. We want this behavior because we want implicit Final declarations
  1457	        # to act pretty much identically with constants: we should be able to replace any
  1458	        # places where we use some Final variable with the original value and get the same
  1459	        # type-checking behavior. For example, we want this program:
  1460	        #
  1461	        #    def expects_literal(x: Literal[3]) -> None: pass
  1462	        #    var: Final = 3
  1463	        #    expects_literal(var)
  1464	        #
  1465	        # ...to type-check in the exact same way as if we had written the program like this:
  1466	        #
  1467	        #    def expects_literal(x: Literal[3]) -> None: pass
  1468	        #    expects_literal(3)
  1469	        #
  1470	        # In order to make this work (especially with literal types), we need var's type
  1471	        # (an Instance) to remember the "original" value.
  1472	        #
  1473	        # Preserving this value within expressions is useful for similar reasons.
  1474	        #
  1475	        # Currently most of mypy will ignore this field and will continue to treat this type like
  1476	        # a regular Instance. We end up using this field only when we are explicitly within a
  1477	        # Literal context.
  1478	        self.last_known_value = last_known_value
  1479
  1480	        # Cached hash value
  1481	        self._hash = -1
  1482
  1483	        # Additional attributes defined per instance of this type. For example modules
  1484	        # have different attributes per instance of types.ModuleType.
  1485	        self.extra_attrs = extra_attrs
  1486
  1487	    def accept(self, visitor: TypeVisitor[T]) -> T:
  1488	        return visitor.visit_instance(self)
  1489
  1490	    def __hash__(self) -> int:
  1491	        if self._hash == -1:
  1492	            self._hash = hash((self.type, self.args, self.last_known_value, self.extra_attrs))
  1493	        return self._hash
  1494
  1495	    def __eq__(self, other: object) -> bool:
  1496	        if not isinstance(other, Instance):
  1497	            return NotImplemented
  1498	        return (
  1499	            self.type == other.type
  1500	            and self.args == other.args
  1501	            and self.last_known_value == other.last_known_value
  1502	            and self.extra_attrs == other.extra_attrs
  1503	        )
  1504
  1505	    def serialize(self) -> JsonDict | str:
  1506	        assert self.type is not None
  1507	        type_ref = self.type.fullname
  1508	        if not self.args and not self.last_known_value:
  1509	            return type_ref
  1510	        data: JsonDict = {
  1511	            ".class": "Instance",
  1512	            "type_ref": type_ref,
  1513	            "args": [arg.serialize() for arg in self.args],
  1514	        }
  1515	        if self.last_known_value is not None:
  1516	            data["last_known_value"] = self.last_known_value.serialize()
  1517	        data["extra_attrs"] = self.extra_attrs.serialize() if self.extra_attrs else None
  1518	        return data

  1521	    def deserialize(cls, data: JsonDict | str) -> Instance:
  1522	        if isinstance(data, str):
  1523	            inst = Instance(NOT_READY, [])
  1524	            inst.type_ref = data
  1525	            return inst
  1526	        assert data[".class"] == "Instance"
  1527	        args: list[Type] = []
  1528	        if "args" in data:
  1529	            args_list = data["args"]
  1530	            assert isinstance(args_list, list)
  1531	            args = [deserialize_type(arg) for arg in args_list]

  1535	            inst.last_known_value = LiteralType.deserialize(data["last_known_value"])
  1536	        if data.get("extra_attrs") is not None:

  2417
  2418	    items: list[Type]
  2419	    partial_fallback: Instance
  2420	    implicit: bool
  2421
  2422	    def __init__(
  2423	        self,
  2424	        items: list[Type],
  2425	        fallback: Instance,
  2426	        line: int = -1,
  2427	        column: int = -1,
  2428	        implicit: bool = False,

  2436	        if self.can_be_any_bool():
  2437	            # Corner case: it is a `NamedTuple` with `__bool__` method defined.
  2438	            # It can be anything: both `True` and `False`.
  2439	            return True
  2440	        return self.length() > 0
  2441

  2443	        if self.can_be_any_bool():
  2444	            # Corner case: it is a `NamedTuple` with `__bool__` method defined.
  2445	            # It can be anything: both `True` and `False`.
  2446	            return True
  2447	        if self.length() == 0:
  2448	            return True
  2449	        if self.length() > 1:
  2450	            return False
  2451	        # Special case tuple[*Ts] may or may not be false.
  2452	        item = self.items[0]
  2453	        if not isinstance(item, UnpackType):
  2454	            return False
  2455	        if not isinstance(item.type, TypeVarTupleType):
  2456	            # Non-normalized tuple[int, ...] can be false.
  2457	            return True
  2458	        return item.type.min_len == 0

  2464	            and self.partial_fallback.type.names.get("__bool__")

  2490	    def deserialize(cls, data: JsonDict) -> TupleType:

  2493	            [deserialize_type(t) for t in data["items"]],
  2494	            Instance.deserialize(data["partial_fallback"]),
  2495	            implicit=data["implicit"],
  2496	        )

  2498	    def copy_modified(
  2499	        self, *, fallback: Instance | None = None, items: list[Type] | None = None
  2500	    ) -> TupleType:
  2501	        if fallback is None:
  2502	            fallback = self.partial_fallback
  2503	        if items is None:
  2504	            items = self.items
  2505	        return TupleType(items, fallback, self.line, self.column)

  2507	    def slice(
  2508	        self, begin: int | None, end: int | None, stride: int | None, *, fallback: Instance | None
  2509	    ) -> TupleType | None:
  2510	        if fallback is None:
  2511	            fallback = self.partial_fallback
  2512
  2513	        if stride == 0:
  2514	            return None
  2515
  2516	        if any(isinstance(t, UnpackType) for t in self.items):
  2517	            total = len(self.items)
  2518	            unpack_index = find_unpack_in_list(self.items)
  2519	            assert unpack_index is not None
  2520	            if begin is None and end is None:
  2521	                # We special-case this to support reversing variadic tuples.
  2522	                # General support for slicing is tricky, so we handle only simple cases.
  2523	                if stride == -1:
  2524	                    slice_items = self.items[::-1]
  2525	                elif stride is None or stride == 1:
  2526	                    slice_items = self.items
  2527	                else:
  2528	                    return None
  2529	            elif (begin is None or unpack_index >= begin >= 0) and (
  2530	                end is not None and unpack_index >= end >= 0
  2531	            ):
  2532	                # Start and end are in the prefix, everything works in this case.
  2533	                slice_items = self.items[begin:end:stride]
  2534	            elif (begin is not None and unpack_index - total < begin < 0) and (
  2535	                end is None or unpack_index - total < end < 0
  2536	            ):
  2537	                # Start and end are in the suffix, everything works in this case.
  2538	                slice_items = self.items[begin:end:stride]
  2539	            elif (begin is None or unpack_index >= begin >= 0) and (
  2540	                end is None or unpack_index - total < end < 0
  2541	            ):
  2542	                # Start in the prefix, end in the suffix, we can support only trivial strides.
  2543	                if stride is None or stride == 1:
  2544	                    slice_items = self.items[begin:end:stride]
  2545	                else:
  2546	                    return None
  2547	            elif (begin is not None and unpack_index - total < begin < 0) and (
  2548	                end is not None and unpack_index >= end >= 0
  2549	            ):
  2550	                # Start in the suffix, end in the prefix, we can support only trivial strides.
  2551	                if stride is None or stride == -1:
  2552	                    slice_items = self.items[begin:end:stride]
  2553	                else:
  2554	                    return None
  2555	            else:
  2556	                # TODO: there some additional cases we can support for homogeneous variadic
  2557	                # items, we can "eat away" finite number of items.
  2558	                return None
  2559	        else:
  2560	            slice_items = self.items[begin:end:stride]
  2561	        return TupleType(slice_items, fallback, self.line, self.column, self.implicit)

  2813	class LiteralType(ProperType):
  2814	    """The type of a Literal instance. Literal[Value]
  2815
  2816	    A Literal always consists of:
  2817
  2818	    1. A native Python object corresponding to the contained inner value
  2819	    2. A fallback for this Literal. The fallback also corresponds to the

  2821

  2823	    'LiteralType(value=42, fallback=instance_of_int)'
  2824

  2826	    represented as `LiteralType(value="RED", fallback=instance_of_color)'.
  2827	    """

  2833	    ) -> None:
  2834	        super().__init__(line, column)
  2835	        self.value = value
  2836	        self.fallback = fallback
  2837	        self._hash = -1  # Cached hash value

  2839	    # NOTE: Enum types are always truthy by default, but this can be changed
