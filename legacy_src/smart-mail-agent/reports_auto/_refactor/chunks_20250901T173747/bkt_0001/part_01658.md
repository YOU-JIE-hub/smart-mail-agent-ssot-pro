   127	def instantiate_generator_class(builder: IRBuilder) -> Value:
   128	    fitem = builder.fn_info.fitem
   129	    generator_reg = builder.add(Call(builder.fn_info.generator_class.ir.ctor, [], fitem.line))
   130
   131	    if builder.fn_info.can_merge_generator_and_env_classes():
   132	        # Set the generator instance to the initial state (zero).
   133	        zero = Integer(0)
   134	        builder.add(SetAttr(generator_reg, NEXT_LABEL_ATTR_NAME, zero, fitem.line))
   135	    else:
   136	        # Get the current environment register. If the current function is nested, then the
   137	        # generator class gets instantiated from the callable class' '__call__' method, and hence
   138	        # we use the callable class' environment register. Otherwise, we use the original
   139	        # function's environment register.
   140	        if builder.fn_info.is_nested:
   141	            curr_env_reg = builder.fn_info.callable_class.curr_env_reg
   142	        else:
   143	            curr_env_reg = builder.fn_info.curr_env_reg
   144
   145	        # Set the generator class' environment attribute to point at the environment class
   146	        # defined in the current scope.
   147	        builder.add(SetAttr(generator_reg, ENV_ATTR_NAME, curr_env_reg, fitem.line))
   148
   149	        # Set the generator instance's environment to the initial state (zero).
   150	        zero = Integer(0)
   151	        builder.add(SetAttr(curr_env_reg, NEXT_LABEL_ATTR_NAME, zero, fitem.line))
   152	    return generator_reg

   223	class LowLevelIRBuilder:
   224	    """A "low-level" IR builder class.
   225
   226	    LowLevelIRBuilder provides core abstractions we use for constructing
   227	    IR as well as a number of higher-level ones (accessing attributes,
   228	    calling functions and methods, and coercing between types, for
   229	    example).
   230
   231	    The core principle of the low-level IR builder is that all of its
   232	    facilities operate solely on the mypyc IR level and not the mypy AST
   233	    level---it has *no knowledge* of mypy types or expressions.
   234
   235	    The mypyc.irbuilder.builder.IRBuilder class wraps an instance of this
   236	    class and provides additional functionality to transform mypy AST nodes
   237	    to IR.
   238	    """
   239
   240	    def __init__(self, errors: Errors | None, options: CompilerOptions) -> None:
   241	        self.errors = errors
   242	        self.options = options
   243	        self.args: list[Register] = []
   244	        self.blocks: list[BasicBlock] = []
   245	        # Stack of except handler entry blocks
   246	        self.error_handlers: list[BasicBlock | None] = [None]
   247	        # Values that we need to keep alive as long as we have borrowed
   248	        # temporaries. Use flush_keep_alives() to mark the end of the live range.
   249	        self.keep_alives: list[Value] = []
   250
   251	    def set_module(self, module_name: str, module_path: str) -> None:
   252	        """Set the name and path of the current module."""
   253	        self.module_name = module_name
   254	        self.module_path = module_path

   256	    # Basic operations
   257
   258	    def add(self, op: Op) -> Value:
   259	        """Add an op."""
   260	        assert not self.blocks[-1].terminated, "Can't add to finished block"
   261	        self.blocks[-1].ops.append(op)
   262	        return op
   263
   264	    def goto(self, target: BasicBlock) -> None:
   265	        """Add goto to a basic block."""
   266	        if not self.blocks[-1].terminated:
   267	            self.add(Goto(target))
   268
   269	    def activate_block(self, block: BasicBlock) -> None:
   270	        """Add a basic block and make it the active one (target of adds)."""
   271	        if self.blocks:
   272	            assert self.blocks[-1].terminated
   273
   274	        block.error_handler = self.error_handlers[-1]
   275	        self.blocks.append(block)
   276
   277	    def goto_and_activate(self, block: BasicBlock) -> None:
   278	        """Add goto a block and make it the active block."""
   279	        self.goto(block)
   280	        self.activate_block(block)
   281
   282	    def keep_alive(self, values: list[Value], *, steal: bool = False) -> None:
   283	        self.add(KeepAlive(values, steal=steal))
   284
   285	    def push_error_handler(self, handler: BasicBlock | None) -> None:
   286	        self.error_handlers.append(handler)
   287
   288	    def pop_error_handler(self) -> BasicBlock | None:
   289	        return self.error_handlers.pop()

   291	    def self(self) -> Register:
   292	        """Return reference to the 'self' argument.
   293
   294	        This only works in a method.
   295	        """
   296	        return self.args[0]

   299	        if self.keep_alives:
   300	            self.add(KeepAlive(self.keep_alives.copy()))
   301	            self.keep_alives = []

   310	    def box(self, src: Value) -> Value:
   311	        if src.type.is_unboxed:
   312	            if isinstance(src, Integer) and is_tagged(src.type):
   313	                return self.add(LoadLiteral(src.value >> 1, rtype=object_rprimitive))
   314	            return self.add(Box(src))
   315	        else:
   316	            return src

   318	    def unbox_or_cast(
   319	        self, src: Value, target_type: RType, line: int, *, can_borrow: bool = False
   320	    ) -> Value:
   321	        if target_type.is_unboxed:
   322	            return self.add(Unbox(src, target_type, line))
   323	        else:
   324	            if can_borrow:
   325	                self.keep_alives.append(src)
   326	            return self.add(Cast(src, target_type, line, borrow=can_borrow))

   328	    def coerce(
   329	        self,
   330	        src: Value,
   331	        target_type: RType,
   332	        line: int,
   333	        force: bool = False,
   334	        *,
   335	        can_borrow: bool = False,
   336	    ) -> Value:
   337	        """Generate a coercion/cast from one type to other (only if needed).
   338
   339	        For example, int -> object boxes the source int; int -> int emits nothing;
   340	        object -> int unboxes the object. All conversions preserve object value.
   341
   342	        If force is true, always generate an op (even if it is just an assignment) so
   343	        that the result will have exactly target_type as the type.
   344
   345	        Returns the register with the converted value (may be same as src).
   346	        """
   347	        src_type = src.type
   348	        if src_type.is_unboxed and not target_type.is_unboxed:
   349	            # Unboxed -> boxed
   350	            return self.box(src)
   351	        if (src_type.is_unboxed and target_type.is_unboxed) and not is_runtime_subtype(
   352	            src_type, target_type
   353	        ):
   354	            if (
   355	                isinstance(src, Integer)
   356	                and is_short_int_rprimitive(src_type)
   357	                and is_fixed_width_rtype(target_type)
   358	            ):
   359	                value = src.numeric_value()
   360	                if not check_native_int_range(target_type, value):
   361	                    self.error(f'Value {value} is out of range for "{target_type}"', line)
   362	                return Integer(src.value >> 1, target_type)
   363	            elif is_int_rprimitive(src_type) and is_fixed_width_rtype(target_type):
   364	                return self.coerce_int_to_fixed_width(src, target_type, line)
   365	            elif is_fixed_width_rtype(src_type) and is_int_rprimitive(target_type):
   366	                return self.coerce_fixed_width_to_int(src, line)
   367	            elif is_short_int_rprimitive(src_type) and is_fixed_width_rtype(target_type):
   368	                return self.coerce_short_int_to_fixed_width(src, target_type, line)
   369	            elif (
   370	                isinstance(src_type, RPrimitive)
   371	                and isinstance(target_type, RPrimitive)
   372	                and src_type.is_native_int
   373	                and target_type.is_native_int
   374	                and src_type.size == target_type.size
   375	                and src_type.is_signed == target_type.is_signed
   376	            ):
   377	                # Equivalent types
   378	                return src
   379	            elif (is_bool_rprimitive(src_type) or is_bit_rprimitive(src_type)) and is_tagged(
   380	                target_type
   381	            ):
   382	                shifted = self.int_op(
   383	                    bool_rprimitive, src, Integer(1, bool_rprimitive), IntOp.LEFT_SHIFT
   384	                )
   385	                return self.add(Extend(shifted, target_type, signed=False))
   386	            elif (
   387	                is_bool_rprimitive(src_type) or is_bit_rprimitive(src_type)
   388	            ) and is_fixed_width_rtype(target_type):
   389	                return self.add(Extend(src, target_type, signed=False))
   390	            elif isinstance(src, Integer) and is_float_rprimitive(target_type):
   391	                if is_tagged(src_type):
   392	                    return Float(float(src.value // 2))
   393	                return Float(float(src.value))
   394	            elif is_tagged(src_type) and is_float_rprimitive(target_type):
   395	                return self.int_to_float(src, line)
   396	            elif (
   397	                isinstance(src_type, RTuple)
   398	                and isinstance(target_type, RTuple)
   399	                and len(src_type.types) == len(target_type.types)
   400	            ):
   401	                # Coerce between two tuple types by coercing each item separately
   402	                values = []
   403	                for i in range(len(src_type.types)):
   404	                    v = None
   405	                    if isinstance(src, TupleSet):
   406	                        item = src.items[i]
   407	                        # We can't reuse register values, since they can be modified.
   408	                        if not isinstance(item, Register):
   409	                            v = item
   410	                    if v is None:
   411	                        v = TupleGet(src, i)
   412	                        self.add(v)
   413	                    values.append(v)
   414	                return self.add(
   415	                    TupleSet(
   416	                        [self.coerce(v, t, line) for v, t in zip(values, target_type.types)], line
   417	                    )
   418	                )
   419	            # To go between any other unboxed types, we go through a boxed
   420	            # in-between value, for simplicity.
   421	            tmp = self.box(src)
   422	            return self.unbox_or_cast(tmp, target_type, line)
   423	        if (not src_type.is_unboxed and target_type.is_unboxed) or not is_subtype(
   424	            src_type, target_type
   425	        ):
   426	            return self.unbox_or_cast(src, target_type, line, can_borrow=can_borrow)
   427	        elif force:
   428	            tmp = Register(target_type)
   429	            self.add(Assign(tmp, src))
   430	            return tmp
   431	        return src

   433	    def coerce_int_to_fixed_width(self, src: Value, target_type: RType, line: int) -> Value:
   434	        assert is_fixed_width_rtype(target_type), target_type
   435	        assert isinstance(target_type, RPrimitive)
   436
   437	        res = Register(target_type)
   438
   439	        fast, slow, end = BasicBlock(), BasicBlock(), BasicBlock()
   440
   441	        check = self.check_tagged_short_int(src, line)
   442	        self.add(Branch(check, fast, slow, Branch.BOOL))
   443
   444	        self.activate_block(fast)
   445
   446	        size = target_type.size
   447	        if size < int_rprimitive.size:
   448	            # Add a range check when the target type is smaller than the source type
   449	            fast2, fast3 = BasicBlock(), BasicBlock()
   450	            upper_bound = 1 << (size * 8 - 1)
   451	            if not target_type.is_signed:
   452	                upper_bound *= 2
   453	            check2 = self.add(ComparisonOp(src, Integer(upper_bound, src.type), ComparisonOp.SLT))
   454	            self.add(Branch(check2, fast2, slow, Branch.BOOL))
   455	            self.activate_block(fast2)
   456	            if target_type.is_signed:
   457	                lower_bound = -upper_bound
   458	            else:
   459	                lower_bound = 0
   460	            check3 = self.add(ComparisonOp(src, Integer(lower_bound, src.type), ComparisonOp.SGE))
   461	            self.add(Branch(check3, fast3, slow, Branch.BOOL))
   462	            self.activate_block(fast3)
   463	            tmp = self.int_op(
   464	                c_pyssize_t_rprimitive,
   465	                src,
   466	                Integer(1, c_pyssize_t_rprimitive),
   467	                IntOp.RIGHT_SHIFT,
   468	                line,
   469	            )
   470	            tmp = self.add(Truncate(tmp, target_type))
   471	        else:
   472	            if size > int_rprimitive.size:
   473	                tmp = self.add(Extend(src, target_type, signed=True))
   474	            else:
   475	                tmp = src
   476	            tmp = self.int_op(target_type, tmp, Integer(1, target_type), IntOp.RIGHT_SHIFT, line)
   477
   478	        self.add(Assign(res, tmp))
   479	        self.goto(end)
   480
   481	        self.activate_block(slow)
   482	        if is_int64_rprimitive(target_type) or (
   483	            is_int32_rprimitive(target_type) and size == int_rprimitive.size
   484	        ):
   485	            # Slow path calls a library function that handles more complex logic
   486	            ptr = self.int_op(
   487	                pointer_rprimitive, src, Integer(1, pointer_rprimitive), IntOp.XOR, line
   488	            )
   489	            ptr2 = Register(c_pointer_rprimitive)
   490	            self.add(Assign(ptr2, ptr))
   491	            if is_int64_rprimitive(target_type):
   492	                conv_op = int_to_int64_op
   493	            else:
   494	                conv_op = int_to_int32_op
   495	            tmp = self.call_c(conv_op, [ptr2], line)
   496	            self.add(Assign(res, tmp))
   497	            self.add(KeepAlive([src]))
   498	            self.goto(end)
   499	        elif is_int32_rprimitive(target_type):
   500	            # Slow path just always generates an OverflowError
   501	            self.call_c(int32_overflow, [], line)
   502	            self.add(Unreachable())
   503	        elif is_int16_rprimitive(target_type):
   504	            # Slow path just always generates an OverflowError
   505	            self.call_c(int16_overflow, [], line)
   506	            self.add(Unreachable())
   507	        elif is_uint8_rprimitive(target_type):
   508	            # Slow path just always generates an OverflowError
   509	            self.call_c(uint8_overflow, [], line)
   510	            self.add(Unreachable())
   511	        else:
   512	            assert False, target_type
   513
   514	        self.activate_block(end)
   515	        return res
   516

   525	    def coerce_fixed_width_to_int(self, src: Value, line: int) -> Value:
   526	        if (
   527	            (is_int32_rprimitive(src.type) and PLATFORM_SIZE == 8)
   528	            or is_int16_rprimitive(src.type)
   529	            or is_uint8_rprimitive(src.type)
   530	        ):
   531	            # Simple case -- just sign extend and shift.
   532	            extended = self.add(Extend(src, c_pyssize_t_rprimitive, signed=src.type.is_signed))
   533	            return self.int_op(
   534	                int_rprimitive,
   535	                extended,
   536	                Integer(1, c_pyssize_t_rprimitive),
   537	                IntOp.LEFT_SHIFT,
   538	                line,
   539	            )
   540
   541	        assert is_fixed_width_rtype(src.type)
   542	        assert isinstance(src.type, RPrimitive)
   543	        src_type = src.type
   544
   545	        res = Register(int_rprimitive)
   546
   547	        fast, fast2, slow, end = BasicBlock(), BasicBlock(), BasicBlock(), BasicBlock()
   548
   549	        c1 = self.add(ComparisonOp(src, Integer(MAX_SHORT_INT, src_type), ComparisonOp.SLE))
   550	        self.add(Branch(c1, fast, slow, Branch.BOOL))
   551
   552	        self.activate_block(fast)
   553	        c2 = self.add(ComparisonOp(src, Integer(MIN_SHORT_INT, src_type), ComparisonOp.SGE))
   554	        self.add(Branch(c2, fast2, slow, Branch.BOOL))
   555
   556	        self.activate_block(slow)
   557	        if is_int64_rprimitive(src_type):
   558	            conv_op = int64_to_int_op
   559	        elif is_int32_rprimitive(src_type):
   560	            assert PLATFORM_SIZE == 4
   561	            conv_op = ssize_t_to_int_op
   562	        else:
   563	            assert False, src_type
   564	        x = self.call_c(conv_op, [src], line)
   565	        self.add(Assign(res, x))
   566	        self.goto(end)
   567
   568	        self.activate_block(fast2)
   569	        if int_rprimitive.size < src_type.size:
   570	            tmp = self.add(Truncate(src, c_pyssize_t_rprimitive))
   571	        else:
   572	            tmp = src
   573	        s = self.int_op(int_rprimitive, tmp, Integer(1, tmp.type), IntOp.LEFT_SHIFT, line)
   574	        self.add(Assign(res, s))
   575	        self.goto(end)
   576
   577	        self.activate_block(end)
   578	        return res

   580	    def coerce_nullable(self, src: Value, target_type: RType, line: int) -> Value:
   581	        """Generate a coercion from a potentially null value."""
   582	        if src.type.is_unboxed == target_type.is_unboxed and (
   583	            (target_type.is_unboxed and is_runtime_subtype(src.type, target_type))
   584	            or (not target_type.is_unboxed and is_subtype(src.type, target_type))
   585	        ):
   586	            return src
   587
   588	        target = Register(target_type)
   589
   590	        valid, invalid, out = BasicBlock(), BasicBlock(), BasicBlock()
   591	        self.add(Branch(src, invalid, valid, Branch.IS_ERROR))
   592
   593	        self.activate_block(valid)
   594	        coerced = self.coerce(src, target_type, line)
   595	        self.add(Assign(target, coerced, line))
   596	        self.goto(out)
   597
   598	        self.activate_block(invalid)
   599	        error = self.add(LoadErrorValue(target_type))
   600	        self.add(Assign(target, error, line))
   601
   602	        self.goto_and_activate(out)
   603	        return target

   607	    def get_attr(
   608	        self, obj: Value, attr: str, result_type: RType, line: int, *, borrow: bool = False
   609	    ) -> Value:
   610	        """Get a native or Python attribute of an object."""
   611	        if (
   612	            isinstance(obj.type, RInstance)
   613	            and obj.type.class_ir.is_ext_class
   614	            and obj.type.class_ir.has_attr(attr)
   615	        ):
   616	            op = GetAttr(obj, attr, line, borrow=borrow)
   617	            # For non-refcounted attribute types, the borrow might be
   618	            # disabled even if requested, so don't check 'borrow'.
   619	            if op.is_borrowed:
   620	                self.keep_alives.append(obj)
   621	            return self.add(op)
   622	        elif isinstance(obj.type, RUnion):
   623	            return self.union_get_attr(obj, obj.type, attr, result_type, line)
   624	        else:
   625	            return self.py_get_attr(obj, attr, line)

   629	    ) -> Value:

   631
   632	        def get_item_attr(value: Value) -> Value:
   633	            return self.get_attr(value, attr, result_type, line)

   635	        return self.decompose_union_helper(obj, rtype, result_type, get_item_attr, line)

   647	    def isinstance_helper(self, obj: Value, class_irs: list[ClassIR], line: int) -> Value:
   648	        """Fast path for isinstance() that checks against a list of native classes."""
   649	        if not class_irs:
   650	            return self.false()
   651	        ret = self.isinstance_native(obj, class_irs[0], line)
   652	        for class_ir in class_irs[1:]:
   653
   654	            def other() -> Value:
   655	                return self.isinstance_native(obj, class_ir, line)
   656
   657	            ret = self.shortcircuit_helper("or", bool_rprimitive, lambda: ret, other, line)
   658	        return ret

   660	    def get_type_of_obj(self, obj: Value, line: int) -> Value:
   661	        ob_type_address = self.add(GetElementPtr(obj, PyObject, "ob_type", line))
   662	        ob_type = self.add(LoadMem(object_rprimitive, ob_type_address))
   663	        self.add(KeepAlive([obj]))
   664	        return ob_type
   665
   666	    def type_is_op(self, obj: Value, type_obj: Value, line: int) -> Value:
   667	        typ = self.get_type_of_obj(obj, line)
   668	        return self.add(ComparisonOp(typ, type_obj, ComparisonOp.EQ, line))

   670	    def isinstance_native(self, obj: Value, class_ir: ClassIR, line: int) -> Value:
   671	        """Fast isinstance() check for a native class.
   672
   673	        If there are three or fewer concrete (non-trait) classes among the class
   674	        and all its children, use even faster type comparison checks `type(obj)
   675	        is typ`.
   676	        """
   677	        concrete = all_concrete_classes(class_ir)
   678	        if concrete is None or len(concrete) > FAST_ISINSTANCE_MAX_SUBCLASSES + 1:
   679	            return self.primitive_op(
   680	                fast_isinstance_op, [obj, self.get_native_type(class_ir)], line
   681	            )
   682	        if not concrete:
   683	            # There can't be any concrete instance that matches this.
   684	            return self.false()
   685	        type_obj = self.get_native_type(concrete[0])
   686	        ret = self.type_is_op(obj, type_obj, line)
   687	        for c in concrete[1:]:
   688
   689	            def other() -> Value:
   690	                return self.type_is_op(obj, self.get_native_type(c), line)
   691
   692	            ret = self.shortcircuit_helper("or", bool_rprimitive, lambda: ret, other, line)
   693	        return ret

   697	    def _construct_varargs(
   698	        self,
   699	        args: Sequence[tuple[Value, ArgKind, str | None]],
   700	        line: int,
   701	        *,
   702	        has_star: bool,
   703	        has_star2: bool,
   704	    ) -> tuple[Value | None, Value | None]:
   705	        """Construct *args and **kwargs from a collection of arguments
   706
   707	        This is pretty complicated, and almost all of the complication here stems from
   708	        one of two things (but mostly the second):
   709	          * The handling of ARG_STAR/ARG_STAR2. We want to create as much of the args/kwargs
   710	            values in one go as we can, so we collect values until our hand is forced, and
   711	            then we emit creation of the list/tuple, and expand it from there if needed.
   712
   713	          * Support potentially nullable argument values. This has very narrow applicability,
   714	            as this will never be done by our compiled Python code, but is critically used
   715	            by gen_glue_method when generating glue methods to mediate between the function
   716	            signature of a parent class and its subclasses.
   717
   718	            For named-only arguments, this is quite simple: if it is
   719	            null, don't put it in the dict.
   720
   721	            For positional-or-named arguments, things are much more complicated.
   722	              * First, anything that was passed as a positional arg
   723	                must be forwarded along as a positional arg. It *must
   724	                not* be converted to a named arg. This is because mypy
   725	                does not enforce that positional-or-named arguments
   726	                have the same name in subclasses, and it is not
   727	                uncommon for code to have different names in
   728	                subclasses (a bunch of mypy's visitors do this, for
   729	                example!). This is arguably a bug in both mypy and code doing
   730	                this, and they ought to be using positional-only arguments, but
   731	                positional-only arguments are new and ugly.
   732
   733	              * On the flip side, we're willing to accept the
   734	                infelicity of sometimes turning an argument that was
   735	                passed by keyword into a positional argument. It's wrong,
   736	                but it's very marginal, and avoiding it would require passing
   737	                a bitmask of which arguments were named with every function call,
   738	                or something similar.
   739	                (See some discussion of this in testComplicatedArgs)
   740
   741	            Thus, our strategy for positional-or-named arguments is to
   742	            always pass them as positional, except in the one
   743	            situation where we can not, and where we can be absolutely
   744	            sure they were passed by name: when an *earlier*
   745	            positional argument was missing its value.
   746
   747	            This means that if we have a method `f(self, x: int=..., y: object=...)`:
   748	              * x and y present:      args=(x, y), kwargs={}
   749	              * x present, y missing: args=(x,),   kwargs={}
   750	              * x missing, y present: args=(),     kwargs={'y': y}
   751
   752	            To implement this, when we have multiple optional
   753	            positional arguments, we maintain a flag in a register
   754	            that tracks whether an argument has been missing, and for
   755	            each such optional argument (except the first), we check
   756	            the flag to determine whether to append the argument to
   757	            the *args list or add it to the **kwargs dict. What a
   758	            mess!
   759
   760	            This is what really makes everything here such a tangle;
   761	            otherwise the *args and **kwargs code could be separated.
   762
   763	        The arguments has_star and has_star2 indicate whether the target function
   764	        takes an ARG_STAR and ARG_STAR2 argument, respectively.
   765	        (These will always be true when making a pycall, and be based
   766	        on the actual target signature for a native call.)
   767	        """
   768
   769	        star_result: Value | None = None
   770	        star2_result: Value | None = None
   771	        # We aggregate values that need to go into *args and **kwargs
   772	        # in these lists. Once all arguments are processed (in the
   773	        # happiest case), or we encounter an ARG_STAR/ARG_STAR2 or a
   774	        # nullable arg, then we create the list and/or dict.
   775	        star_values: list[Value] = []
   776	        star2_keys: list[Value] = []
   777	        star2_values: list[Value] = []
   778
   779	        seen_empty_reg: Register | None = None
   780
   781	        for value, kind, name in args:
   782	            if kind == ARG_STAR:
   783	                if star_result is None:
   784	                    star_result = self.new_list_op(star_values, line)
   785	                self.primitive_op(list_extend_op, [star_result, value], line)
   786	            elif kind == ARG_STAR2:
   787	                if star2_result is None:
   788	                    star2_result = self._create_dict(star2_keys, star2_values, line)
   789
   790	                self.call_c(dict_update_in_display_op, [star2_result, value], line=line)
   791	            else:
   792	                nullable = kind.is_optional()
   793	                maybe_pos = kind.is_positional() and has_star
   794	                maybe_named = kind.is_named() or (kind.is_optional() and name and has_star2)
   795
   796	                # If the argument is nullable, we need to create the
   797	                # relevant args/kwargs objects so that we can
   798	                # conditionally modify them.
   799	                if nullable:
   800	                    if maybe_pos and star_result is None:
   801	                        star_result = self.new_list_op(star_values, line)
   802	                    if maybe_named and star2_result is None:
   803	                        star2_result = self._create_dict(star2_keys, star2_values, line)
   804
   805	                # Easy cases: just collect the argument.
   806	                if maybe_pos and star_result is None:
   807	                    star_values.append(value)
   808	                    continue
   809
   810	                if maybe_named and star2_result is None:
   811	                    assert name is not None
   812	                    key = self.load_str(name)
   813	                    star2_keys.append(key)
   814	                    star2_values.append(value)
   815	                    continue
   816
   817	                # OK, anything that is nullable or *after* a nullable arg needs to be here
   818	                # TODO: We could try harder to avoid creating basic blocks in the common case
   819	                new_seen_empty_reg = seen_empty_reg
   820
   821	                out = BasicBlock()
   822	                if nullable:
   823	                    # If this is the first nullable positional arg we've seen, create
   824	                    # a register to track whether anything has been null.
   825	                    # (We won't *check* the register until the next argument, though.)
   826	                    if maybe_pos and not seen_empty_reg:
   827	                        new_seen_empty_reg = Register(bool_rprimitive)
   828	                        self.add(Assign(new_seen_empty_reg, self.false(), line))
   829
   830	                    skip = BasicBlock() if maybe_pos else out
   831	                    keep = BasicBlock()
   832	                    self.add(Branch(value, skip, keep, Branch.IS_ERROR))
   833	                    self.activate_block(keep)
   834
   835	                # If this could be positional or named and we /might/ have seen a missing
   836	                # positional arg, then we need to compile *both* a positional and named
   837	                # version! What a pain!
   838	                if maybe_pos and maybe_named and seen_empty_reg:
   839	                    pos_block, named_block = BasicBlock(), BasicBlock()
   840	                    self.add(Branch(seen_empty_reg, named_block, pos_block, Branch.BOOL))
   841	                else:
   842	                    pos_block = named_block = BasicBlock()
   843	                    self.goto(pos_block)
   844
   845	                if maybe_pos:
   846	                    self.activate_block(pos_block)
   847	                    assert star_result
   848	                    self.translate_special_method_call(
   849	                        star_result, "append", [value], result_type=None, line=line
   850	                    )
   851	                    self.goto(out)
   852
   853	                if maybe_named and (not maybe_pos or seen_empty_reg):
   854	                    self.activate_block(named_block)
   855	                    assert name is not None
   856	                    key = self.load_str(name)
   857	                    assert star2_result
   858	                    self.translate_special_method_call(
   859	                        star2_result, "__setitem__", [key, value], result_type=None, line=line
   860	                    )
   861	                    self.goto(out)
   862
   863	                if nullable and maybe_pos and new_seen_empty_reg:
   864	                    assert skip is not out
   865	                    self.activate_block(skip)
   866	                    self.add(Assign(new_seen_empty_reg, self.true(), line))
   867	                    self.goto(out)
   868
   869	                self.activate_block(out)
   870
   871	                seen_empty_reg = new_seen_empty_reg
   872
   873	        assert not (star_result or star_values) or has_star
   874	        assert not (star2_result or star2_values) or has_star2
   875	        if has_star:
   876	            # If we managed to make it this far without creating a
   877	            # *args list, then we can directly create a
   878	            # tuple. Otherwise create the tuple from the list.
   879	            if star_result is None:
   880	                star_result = self.new_tuple(star_values, line)
   881	            else:
   882	                star_result = self.primitive_op(list_tuple_op, [star_result], line)
   883	        if has_star2 and star2_result is None:
   884	            star2_result = self._create_dict(star2_keys, star2_values, line)
   885
   886	        return star_result, star2_result

   888	    def py_call(
   889	        self,
   890	        function: Value,
   891	        arg_values: list[Value],
   892	        line: int,
   893	        arg_kinds: list[ArgKind] | None = None,
   894	        arg_names: Sequence[str | None] | None = None,
   895	    ) -> Value:
   896	        """Call a Python function (non-native and slow).
   897
   898	        Use py_call_op or py_call_with_kwargs_op for Python function call.
   899	        """
   900	        result = self._py_vector_call(function, arg_values, line, arg_kinds, arg_names)
   901	        if result is not None:
   902	            return result
   903
   904	        # If all arguments are positional, we can use py_call_op.
   905	        if arg_kinds is None or all(kind == ARG_POS for kind in arg_kinds):
   906	            return self.call_c(py_call_op, [function] + arg_values, line)
   907
   908	        # Otherwise fallback to py_call_with_kwargs_op.
   909	        assert arg_names is not None
   910
   911	        pos_args_tuple, kw_args_dict = self._construct_varargs(
   912	            list(zip(arg_values, arg_kinds, arg_names)), line, has_star=True, has_star2=True
   913	        )
   914	        assert pos_args_tuple and kw_args_dict
   915
   916	        return self.call_c(py_call_with_kwargs_op, [function, pos_args_tuple, kw_args_dict], line)

   918	    def _py_vector_call(
   919	        self,
   920	        function: Value,
   921	        arg_values: list[Value],
   922	        line: int,
   923	        arg_kinds: list[ArgKind] | None = None,
   924	        arg_names: Sequence[str | None] | None = None,
   925	    ) -> Value | None:
   926	        """Call function using the vectorcall API if possible.
   927
   928	        Return the return value if successful. Return None if a non-vectorcall
   929	        API should be used instead.
   930	        """
   931	        # We can do this if all args are positional or named (no *args or **kwargs, not optional).
   932	        if arg_kinds is None or all(
   933	            not kind.is_star() and not kind.is_optional() for kind in arg_kinds
   934	        ):
   935	            if arg_values:
   936	                # Create a C array containing all arguments as boxed values.
   937	                coerced_args = [self.coerce(arg, object_rprimitive, line) for arg in arg_values]
   938	                arg_ptr = self.setup_rarray(object_rprimitive, coerced_args, object_ptr=True)
   939	            else:
   940	                arg_ptr = Integer(0, object_pointer_rprimitive)
   941	            num_pos = num_positional_args(arg_values, arg_kinds)
   942	            keywords = self._vectorcall_keywords(arg_names)
   943	            value = self.call_c(
   944	                py_vectorcall_op,
   945	                [function, arg_ptr, Integer(num_pos, c_size_t_rprimitive), keywords],
   946	                line,
   947	            )
   948	            if arg_values:
   949	                # Make sure arguments won't be freed until after the call.
   950	                # We need this because RArray doesn't support automatic
   951	                # memory management.
   952	                self.add(KeepAlive(coerced_args))
   953	            return value
   954	        return None

   956	    def _vectorcall_keywords(self, arg_names: Sequence[str | None] | None) -> Value:
   957	        """Return a reference to a tuple literal with keyword argument names.
   958
   959	        Return null pointer if there are no keyword arguments.
   960	        """
   961	        if arg_names:
   962	            kw_list = [name for name in arg_names if name is not None]
   963	            if kw_list:
   964	                return self.add(LoadLiteral(tuple(kw_list), object_rprimitive))
   965	        return Integer(0, object_rprimitive)

  1033	    def call(
  1034	        self,
  1035	        decl: FuncDecl,
  1036	        args: Sequence[Value],
  1037	        arg_kinds: list[ArgKind],
  1038	        arg_names: Sequence[str | None],
  1039	        line: int,
  1040	        *,
  1041	        bitmap_args: list[Register] | None = None,
  1042	    ) -> Value:
  1043	        """Call a native function.
  1044
  1045	        If bitmap_args is given, they override the values of (some) of the bitmap
  1046	        arguments used to track the presence of values for certain arguments. By
  1047	        default, the values of the bitmap arguments are inferred from args.
  1048	        """
  1049	        # Normalize args to positionals.
  1050	        args = self.native_args_to_positional(
  1051	            args, arg_kinds, arg_names, decl.sig, line, bitmap_args=bitmap_args
  1052	        )
  1053	        return self.add(Call(decl, args, line))

  1055	    def native_args_to_positional(
  1056	        self,
  1057	        args: Sequence[Value],
  1058	        arg_kinds: list[ArgKind],
  1059	        arg_names: Sequence[str | None],
  1060	        sig: FuncSignature,
  1061	        line: int,
  1062	        *,
  1063	        bitmap_args: list[Register] | None = None,
  1064	    ) -> list[Value]:
  1065	        """Prepare arguments for a native call.
  1066
  1067	        Given args/kinds/names and a target signature for a native call, map
  1068	        keyword arguments to their appropriate place in the argument list,
  1069	        fill in error values for unspecified default arguments,
  1070	        package arguments that will go into *args/**kwargs into a tuple/dict,
  1071	        and coerce arguments to the appropriate type.
  1072	        """
  1073
  1074	        sig_args = sig.args
  1075	        n = sig.num_bitmap_args
  1076	        if n:
  1077	            sig_args = sig_args[:-n]
  1078
  1079	        sig_arg_kinds = [arg.kind for arg in sig_args]
  1080	        sig_arg_names = [arg.name for arg in sig_args]
  1081
  1082	        concrete_kinds = [concrete_arg_kind(arg_kind) for arg_kind in arg_kinds]
  1083	        formal_to_actual = map_actuals_to_formals(
  1084	            concrete_kinds,
  1085	            arg_names,
  1086	            sig_arg_kinds,
  1087	            sig_arg_names,
  1088	            lambda n: AnyType(TypeOfAny.special_form),
  1089	        )
  1090
  1091	        # First scan for */** and construct those
  1092	        has_star = has_star2 = False
  1093	        star_arg_entries = []
  1094	        for lst, arg in zip(formal_to_actual, sig_args):
  1095	            if arg.kind.is_star():
  1096	                star_arg_entries.extend([(args[i], arg_kinds[i], arg_names[i]) for i in lst])
  1097	            has_star = has_star or arg.kind == ARG_STAR
  1098	            has_star2 = has_star2 or arg.kind == ARG_STAR2
  1099
  1100	        star_arg, star2_arg = self._construct_varargs(
  1101	            star_arg_entries, line, has_star=has_star, has_star2=has_star2
  1102	        )
  1103
  1104	        # Flatten out the arguments, loading error values for default
  1105	        # arguments, constructing tuples/dicts for star args, and
  1106	        # coercing everything to the expected type.
  1107	        output_args: list[Value] = []
  1108	        for lst, arg in zip(formal_to_actual, sig_args):
  1109	            if arg.kind == ARG_STAR:
  1110	                assert star_arg
  1111	                output_arg = star_arg
  1112	            elif arg.kind == ARG_STAR2:
  1113	                assert star2_arg
  1114	                output_arg = star2_arg
  1115	            elif not lst:
  1116	                if is_fixed_width_rtype(arg.type):
  1117	                    output_arg = Integer(0, arg.type)
  1118	                elif is_float_rprimitive(arg.type):
  1119	                    output_arg = Float(0.0)
  1120	                else:
  1121	                    output_arg = self.add(LoadErrorValue(arg.type, is_borrowed=True))
  1122	            else:
  1123	                base_arg = args[lst[0]]
  1124
  1125	                if arg_kinds[lst[0]].is_optional():
  1126	                    output_arg = self.coerce_nullable(base_arg, arg.type, line)
  1127	                else:
  1128	                    output_arg = self.coerce(base_arg, arg.type, line)
  1129
  1130	            output_args.append(output_arg)
  1131
  1132	        for i in reversed(range(n)):
  1133	            if bitmap_args and i < len(bitmap_args):
  1134	                # Use override provided by caller
  1135	                output_args.append(bitmap_args[i])
  1136	                continue
  1137	            # Infer values of bitmap args
  1138	            bitmap = 0
  1139	            c = 0
  1140	            for lst, arg in zip(formal_to_actual, sig_args):
  1141	                if arg.kind.is_optional() and arg.type.error_overlap:
  1142	                    if i * BITMAP_BITS <= c < (i + 1) * BITMAP_BITS:
  1143	                        if lst:
  1144	                            bitmap |= 1 << (c & (BITMAP_BITS - 1))
  1145	                    c += 1
  1146	            output_args.append(Integer(bitmap, bitmap_rprimitive))
  1147
  1148	        return output_args

  1151	        self,

  1153	        name: str,

  1157	        arg_kinds: list[ArgKind] | None = None,
  1158	        arg_names: list[str | None] | None = None,
  1159	        can_borrow: bool = False,
  1160	    ) -> Value:
  1161	        """Generate either a native or Python method call."""
  1162	        # If we have *args, then fallback to Python method call.
  1163	        if arg_kinds is not None and any(kind.is_star() for kind in arg_kinds):
  1164	            return self.py_method_call(base, name, arg_values, line, arg_kinds, arg_names)
  1165
  1166	        # If the base type is one of ours, do a MethodCall
  1167	        if (
  1168	            isinstance(base.type, RInstance)
  1169	            and base.type.class_ir.is_ext_class
  1170	            and not base.type.class_ir.builtin_base
  1171	        ):
  1172	            if base.type.class_ir.has_method(name):
  1173	                decl = base.type.class_ir.method_decl(name)
  1174	                if arg_kinds is None:
  1175	                    assert arg_names is None, "arg_kinds not present but arg_names is"
  1176	                    arg_kinds = [ARG_POS for _ in arg_values]
  1177	                    arg_names = [None for _ in arg_values]
  1178	                else:
  1179	                    assert arg_names is not None, "arg_kinds present but arg_names is not"
  1180
  1181	                # Normalize args to positionals.
  1182	                assert decl.bound_sig
  1183	                arg_values = self.native_args_to_positional(
  1184	                    arg_values, arg_kinds, arg_names, decl.bound_sig, line
  1185	                )
  1186	                return self.add(MethodCall(base, name, arg_values, line))
  1187	            elif base.type.class_ir.has_attr(name):
  1188	                function = self.add(GetAttr(base, name, line))
  1189	                return self.py_call(
  1190	                    function, arg_values, line, arg_kinds=arg_kinds, arg_names=arg_names
  1191	                )
  1192
  1193	        elif isinstance(base.type, RUnion):
  1194	            return self.union_method_call(
  1195	                base, base.type, name, arg_values, result_type, line, arg_kinds, arg_names
  1196	            )
  1197
  1198	        # Try to do a special-cased method call
  1199	        if not arg_kinds or arg_kinds == [ARG_POS] * len(arg_values):
  1200	            target = self.translate_special_method_call(
  1201	                base, name, arg_values, result_type, line, can_borrow=can_borrow
  1202	            )
  1203	            if target:
  1204	                return target
  1205
  1206	        # Fall back to Python method call
  1207	        return self.py_method_call(base, name, arg_values, line, arg_kinds, arg_names)

  1311	    def binary_op(self, lreg: Value, rreg: Value, op: str, line: int) -> Value:
  1312	        """Perform a binary operation.
  1313
  1314	        Generate specialized operations based on operand types, with a fallback
  1315	        to generic operations.
  1316	        """
  1317	        ltype = lreg.type
  1318	        rtype = rreg.type
  1319
  1320	        # Special case tuple comparison here so that nested tuples can be supported
  1321	        if isinstance(ltype, RTuple) and isinstance(rtype, RTuple) and op in ("==", "!="):
  1322	            return self.compare_tuples(lreg, rreg, op, line)
  1323
  1324	        # Special case == and != when we can resolve the method call statically
  1325	        if op in ("==", "!="):
  1326	            value = self.translate_eq_cmp(lreg, rreg, op, line)
  1327	            if value is not None:
  1328	                return value
  1329
  1330	        # Special case various ops
  1331	        if op in ("is", "is not"):
  1332	            return self.translate_is_op(lreg, rreg, op, line)
  1333	        # TODO: modify 'str' to use same interface as 'compare_bytes' as it avoids
  1334	        # call to PyErr_Occurred()
  1335	        if is_str_rprimitive(ltype) and is_str_rprimitive(rtype) and op in ("==", "!="):
  1336	            return self.compare_strings(lreg, rreg, op, line)
  1337	        if is_bytes_rprimitive(ltype) and is_bytes_rprimitive(rtype) and op in ("==", "!="):
  1338	            return self.compare_bytes(lreg, rreg, op, line)
  1339	        if is_bool_rprimitive(ltype) and is_bool_rprimitive(rtype) and op in BOOL_BINARY_OPS:
  1340	            if op in ComparisonOp.signed_ops:
  1341	                return self.bool_comparison_op(lreg, rreg, op, line)
  1342	            else:
  1343	                return self.bool_bitwise_op(lreg, rreg, op[0], line)
  1344	        if isinstance(rtype, RInstance) and op in ("in", "not in"):
  1345	            return self.translate_instance_contains(rreg, lreg, op, line)
  1346	        if is_fixed_width_rtype(ltype):
  1347	            if op in FIXED_WIDTH_INT_BINARY_OPS:
  1348	                op = op.removesuffix("=")
  1349	                if op != "//":
  1350	                    op_id = int_op_to_id[op]
  1351	                else:
  1352	                    op_id = IntOp.DIV
  1353	                if is_bool_rprimitive(rtype) or is_bit_rprimitive(rtype):
  1354	                    rreg = self.coerce(rreg, ltype, line)
  1355	                    rtype = ltype
  1356	                if is_fixed_width_rtype(rtype) or is_tagged(rtype):
  1357	                    return self.fixed_width_int_op(ltype, lreg, rreg, op_id, line)
  1358	                if isinstance(rreg, Integer):
  1359	                    return self.fixed_width_int_op(
  1360	                        ltype, lreg, self.coerce(rreg, ltype, line), op_id, line
  1361	                    )
  1362	            elif op in ComparisonOp.signed_ops:
  1363	                if is_int_rprimitive(rtype):
  1364	                    rreg = self.coerce_int_to_fixed_width(rreg, ltype, line)
  1365	                elif is_bool_rprimitive(rtype) or is_bit_rprimitive(rtype):
  1366	                    rreg = self.coerce(rreg, ltype, line)
  1367	                op_id = ComparisonOp.signed_ops[op]
  1368	                if is_fixed_width_rtype(rreg.type):
  1369	                    return self.comparison_op(lreg, rreg, op_id, line)
  1370	                if isinstance(rreg, Integer):
  1371	                    return self.comparison_op(lreg, self.coerce(rreg, ltype, line), op_id, line)
  1372	        elif is_fixed_width_rtype(rtype):
  1373	            if op in FIXED_WIDTH_INT_BINARY_OPS:
  1374	                op = op.removesuffix("=")
  1375	                if op != "//":
  1376	                    op_id = int_op_to_id[op]
  1377	                else:
  1378	                    op_id = IntOp.DIV
  1379	                if isinstance(lreg, Integer):
  1380	                    return self.fixed_width_int_op(
  1381	                        rtype, self.coerce(lreg, rtype, line), rreg, op_id, line
  1382	                    )
  1383	                if is_tagged(ltype):
  1384	                    return self.fixed_width_int_op(rtype, lreg, rreg, op_id, line)
  1385	                if is_bool_rprimitive(ltype) or is_bit_rprimitive(ltype):
  1386	                    lreg = self.coerce(lreg, rtype, line)
  1387	                    return self.fixed_width_int_op(rtype, lreg, rreg, op_id, line)
  1388	            elif op in ComparisonOp.signed_ops:
  1389	                if is_int_rprimitive(ltype):
  1390	                    lreg = self.coerce_int_to_fixed_width(lreg, rtype, line)
  1391	                elif is_bool_rprimitive(ltype) or is_bit_rprimitive(ltype):
  1392	                    lreg = self.coerce(lreg, rtype, line)
  1393	                op_id = ComparisonOp.signed_ops[op]
  1394	                if isinstance(lreg, Integer):
  1395	                    return self.comparison_op(self.coerce(lreg, rtype, line), rreg, op_id, line)
  1396	                if is_fixed_width_rtype(lreg.type):
  1397	                    return self.comparison_op(lreg, rreg, op_id, line)
  1398
  1399	        if is_float_rprimitive(ltype) or is_float_rprimitive(rtype):
  1400	            if isinstance(lreg, Integer):
  1401	                lreg = Float(float(lreg.numeric_value()))
  1402	            elif isinstance(rreg, Integer):
  1403	                rreg = Float(float(rreg.numeric_value()))
  1404	            elif is_int_rprimitive(lreg.type):
  1405	                lreg = self.int_to_float(lreg, line)
  1406	            elif is_int_rprimitive(rreg.type):
  1407	                rreg = self.int_to_float(rreg, line)
  1408	            if is_float_rprimitive(lreg.type) and is_float_rprimitive(rreg.type):
  1409	                if op in float_comparison_op_to_id:
  1410	                    return self.compare_floats(lreg, rreg, float_comparison_op_to_id[op], line)
  1411	                if op.endswith("="):
  1412	                    base_op = op[:-1]
  1413	                else:
  1414	                    base_op = op
  1415	                if base_op in float_op_to_id:
  1416	                    return self.float_op(lreg, rreg, base_op, line)
  1417
  1418	        dunder_op = self.dunder_op(lreg, rreg, op, line)
  1419	        if dunder_op:
  1420	            return dunder_op
  1421
  1422	        primitive_ops_candidates = binary_ops.get(op, [])
  1423	        target = self.matching_primitive_op(primitive_ops_candidates, [lreg, rreg], line)
  1424	        assert target, "Unsupported binary operation: %s" % op
  1425	        return target

  1427	    def dunder_op(self, lreg: Value, rreg: Value | None, op: str, line: int) -> Value | None:
  1428	        """
  1429	        Dispatch a dunder method if applicable.
  1430	        For example for `a + b` it will use `a.__add__(b)` which can lead to higher performance
  1431	        due to the fact that the method could be already compiled and optimized instead of going
  1432	        all the way through `PyNumber_Add(a, b)` python api (making a jump into the python DL).
  1433	        """
  1434	        ltype = lreg.type
  1435	        if not isinstance(ltype, RInstance):
  1436	            return None
  1437
  1438	        method_name = op_methods.get(op) if rreg else unary_op_methods.get(op)
  1439	        if method_name is None:
  1440	            return None
  1441
  1442	        if not ltype.class_ir.has_method(method_name):
  1443	            return None
  1444
  1445	        decl = ltype.class_ir.method_decl(method_name)
  1446	        if not rreg and len(decl.sig.args) != 1:
  1447	            return None
  1448
  1449	        if rreg and (len(decl.sig.args) != 2 or not is_subtype(rreg.type, decl.sig.args[1].type)):
  1450	            return None
  1451
  1452	        if rreg and is_subtype(not_implemented_op.type, decl.sig.ret_type):
  1453	            # If the method is able to return NotImplemented, we should not optimize it.
  1454	            # We can just let go so it will be handled through the python api.
  1455	            return None
  1456
  1457	        args = [rreg] if rreg else []
  1458	        return self.gen_method_call(lreg, method_name, args, decl.sig.ret_type, line)

  1460	    def check_tagged_short_int(self, val: Value, line: int, negated: bool = False) -> Value:

  1473	        """Compare two strings"""
  1474	        compare_result = self.call_c(unicode_compare, [lhs, rhs], line)
  1475	        error_constant = Integer(-1, c_int_rprimitive, line)
  1476	        compare_error_check = self.add(
  1477	            ComparisonOp(compare_result, error_constant, ComparisonOp.EQ, line)
  1478	        )
  1479	        exception_check, propagate, final_compare = BasicBlock(), BasicBlock(), BasicBlock()
  1480	        branch = Branch(compare_error_check, exception_check, final_compare, Branch.BOOL)

  1492	        self.activate_block(propagate)

  1504	    def compare_tuples(self, lhs: Value, rhs: Value, op: str, line: int = -1) -> Value:
  1505	        """Compare two tuples item by item"""
  1506	        # type cast to pass mypy check
  1507	        assert isinstance(lhs.type, RTuple) and isinstance(rhs.type, RTuple)
  1508	        equal = True if op == "==" else False
  1509	        result = Register(bool_rprimitive)
  1510	        # empty tuples
  1511	        if len(lhs.type.types) == 0 and len(rhs.type.types) == 0:
  1512	            self.add(Assign(result, self.true() if equal else self.false(), line))
  1513	            return result
  1514	        length = len(lhs.type.types)
  1515	        false_assign, true_assign, out = BasicBlock(), BasicBlock(), BasicBlock()
  1516	        check_blocks = [BasicBlock() for _ in range(length)]
  1517	        lhs_items = [self.add(TupleGet(lhs, i, line)) for i in range(length)]
  1518	        rhs_items = [self.add(TupleGet(rhs, i, line)) for i in range(length)]
  1519
  1520	        if equal:
  1521	            early_stop, final = false_assign, true_assign
  1522	        else:
  1523	            early_stop, final = true_assign, false_assign
  1524
  1525	        for i in range(len(lhs.type.types)):
  1526	            if i != 0:
  1527	                self.activate_block(check_blocks[i])
  1528	            lhs_item = lhs_items[i]
  1529	            rhs_item = rhs_items[i]
  1530	            compare = self.binary_op(lhs_item, rhs_item, op, line)
  1531	            # Cast to bool if necessary since most types uses comparison returning a object type
  1532	            # See generic_ops.py for more information
  1533	            if not is_bool_rprimitive(compare.type):
  1534	                compare = self.primitive_op(bool_op, [compare], line)
  1535	            if i < len(lhs.type.types) - 1:
  1536	                branch = Branch(compare, early_stop, check_blocks[i + 1], Branch.BOOL)
  1537	            else:
  1538	                branch = Branch(compare, early_stop, final, Branch.BOOL)
  1539	            # if op is ==, we branch on false, else branch on true
  1540	            branch.negated = equal
  1541	            self.add(branch)
  1542	        self.activate_block(false_assign)
  1543	        self.add(Assign(result, self.false(), line))
  1544	        self.goto(out)
  1545	        self.activate_block(true_assign)
  1546	        self.add(Assign(result, self.true(), line))
  1547	        self.goto_and_activate(out)
  1548	        return result

  1577	    def unary_op(self, value: Value, expr_op: str, line: int) -> Value:
  1578	        typ = value.type
  1579	        if is_bool_rprimitive(typ) or is_bit_rprimitive(typ):
  1580	            if expr_op == "not":
  1581	                return self.unary_not(value, line)
  1582	            if expr_op == "+":
  1583	                return value
  1584	        if is_fixed_width_rtype(typ):
  1585	            if expr_op == "-":
  1586	                # Translate to '0 - x'
  1587	                return self.int_op(typ, Integer(0, typ), value, IntOp.SUB, line)
  1588	            elif expr_op == "~":
  1589	                if typ.is_signed:
  1590	                    # Translate to 'x ^ -1'
  1591	                    return self.int_op(typ, value, Integer(-1, typ), IntOp.XOR, line)
  1592	                else:
  1593	                    # Translate to 'x ^ 0xff...'
  1594	                    mask = (1 << (typ.size * 8)) - 1
  1595	                    return self.int_op(typ, value, Integer(mask, typ), IntOp.XOR, line)
  1596	            elif expr_op == "+":

  1701	        expr_type: RType,
