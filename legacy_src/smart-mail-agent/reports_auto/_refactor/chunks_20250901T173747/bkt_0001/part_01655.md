   615	    def get_assignment_target(
   616	        self, lvalue: Lvalue, line: int = -1, *, for_read: bool = False
   617	    ) -> AssignmentTarget:
   618	        if line == -1:
   619	            line = lvalue.line
   620	        if isinstance(lvalue, NameExpr):
   621	            # If we are visiting a decorator, then the SymbolNode we really want to be looking at
   622	            # is the function that is decorated, not the entire Decorator node itself.
   623	            symbol = lvalue.node
   624	            if isinstance(symbol, Decorator):
   625	                symbol = symbol.func
   626	            if symbol is None:
   627	                # Semantic analyzer doesn't create ad-hoc Vars for special forms.
   628	                assert lvalue.is_special_form
   629	                symbol = Var(lvalue.name)
   630	            if not for_read and isinstance(symbol, Var) and symbol.is_cls:
   631	                self.error("Cannot assign to the first argument of classmethod", line)
   632	            if lvalue.kind == LDEF:
   633	                if symbol not in self.symtables[-1]:
   634	                    if isinstance(symbol, Var) and not isinstance(symbol.type, DeletedType):
   635	                        reg_type = self.type_to_rtype(symbol.type)
   636	                    else:
   637	                        reg_type = self.node_type(lvalue)
   638	                    # If the function is a generator function, then first define a new variable
   639	                    # in the current function's environment class. Next, define a target that
   640	                    # refers to the newly defined variable in that environment class. Add the
   641	                    # target to the table containing class environment variables, as well as the
   642	                    # current environment.
   643	                    if self.fn_info.is_generator:
   644	                        return self.add_var_to_env_class(
   645	                            symbol, reg_type, self.fn_info.generator_class, reassign=False
   646	                        )
   647
   648	                    # Otherwise define a new local variable.
   649	                    return self.add_local_reg(symbol, reg_type)
   650	                else:
   651	                    # Assign to a previously defined variable.
   652	                    return self.lookup(symbol)
   653	            elif lvalue.kind == GDEF:
   654	                globals_dict = self.load_globals_dict()
   655	                name = self.load_str(lvalue.name)
   656	                return AssignmentTargetIndex(globals_dict, name)
   657	            else:
   658	                assert False, lvalue.kind
   659	        elif isinstance(lvalue, IndexExpr):
   660	            # Indexed assignment x[y] = e
   661	            base = self.accept(lvalue.base)
   662	            index = self.accept(lvalue.index)
   663	            return AssignmentTargetIndex(base, index)
   664	        elif isinstance(lvalue, MemberExpr):
   665	            # Attribute assignment x.y = e
   666	            can_borrow = self.is_native_attr_ref(lvalue)
   667	            obj = self.accept(lvalue.expr, can_borrow=can_borrow)
   668	            return AssignmentTargetAttr(obj, lvalue.name, can_borrow=can_borrow)
   669	        elif isinstance(lvalue, TupleExpr):
   670	            # Multiple assignment a, ..., b = e
   671	            star_idx: int | None = None
   672	            lvalues = []
   673	            for idx, item in enumerate(lvalue.items):
   674	                targ = self.get_assignment_target(item)
   675	                lvalues.append(targ)
   676	                if isinstance(item, StarExpr):
   677	                    if star_idx is not None:
   678	                        self.error("Two starred expressions in assignment", line)
   679	                    star_idx = idx
   680
   681	            return AssignmentTargetTuple(lvalues, star_idx)
   682
   683	        elif isinstance(lvalue, StarExpr):
   684	            return self.get_assignment_target(lvalue.expr)
   685
   686	        assert False, "Unsupported lvalue: %r" % lvalue

   688	    def read(
   689	        self, target: Value | AssignmentTarget, line: int = -1, can_borrow: bool = False
   690	    ) -> Value:
   691	        if isinstance(target, Value):
   692	            return target
   693	        if isinstance(target, AssignmentTargetRegister):
   694	            return target.register
   695	        if isinstance(target, AssignmentTargetIndex):
   696	            reg = self.gen_method_call(
   697	                target.base, "__getitem__", [target.index], target.type, line
   698	            )
   699	            if reg is not None:
   700	                return reg
   701	            assert False, target.base.type
   702	        if isinstance(target, AssignmentTargetAttr):
   703	            if isinstance(target.obj.type, RInstance) and target.obj.type.class_ir.is_ext_class:
   704	                borrow = can_borrow and target.can_borrow
   705	                return self.add(GetAttr(target.obj, target.attr, line, borrow=borrow))
   706	            else:
   707	                return self.py_get_attr(target.obj, target.attr, line)
   708
   709	        assert False, "Unsupported lvalue: %r" % target

   720	    def assign(self, target: Register | AssignmentTarget, rvalue_reg: Value, line: int) -> None:
   721	        if isinstance(target, Register):
   722	            self.add(Assign(target, self.coerce_rvalue(rvalue_reg, target.type, line)))
   723	        elif isinstance(target, AssignmentTargetRegister):
   724	            rvalue_reg = self.coerce_rvalue(rvalue_reg, target.type, line)
   725	            self.add(Assign(target.register, rvalue_reg))
   726	        elif isinstance(target, AssignmentTargetAttr):
   727	            if isinstance(target.obj_type, RInstance):
   728	                rvalue_reg = self.coerce_rvalue(rvalue_reg, target.type, line)
   729	                self.add(SetAttr(target.obj, target.attr, rvalue_reg, line))
   730	            else:
   731	                key = self.load_str(target.attr)
   732	                boxed_reg = self.builder.box(rvalue_reg)
   733	                self.primitive_op(py_setattr_op, [target.obj, key, boxed_reg], line)
   734	        elif isinstance(target, AssignmentTargetIndex):
   735	            target_reg2 = self.gen_method_call(
   736	                target.base, "__setitem__", [target.index, rvalue_reg], None, line
   737	            )
   738	            assert target_reg2 is not None, target.base.type
   739	        elif isinstance(target, AssignmentTargetTuple):
   740	            if isinstance(rvalue_reg.type, RTuple) and target.star_idx is None:
   741	                rtypes = rvalue_reg.type.types
   742	                assert len(rtypes) == len(target.items)
   743	                for i in range(len(rtypes)):
   744	                    item_value = self.add(TupleGet(rvalue_reg, i, line))
   745	                    self.assign(target.items[i], item_value, line)
   746	            elif (
   747	                is_list_rprimitive(rvalue_reg.type) or is_tuple_rprimitive(rvalue_reg.type)
   748	            ) and target.star_idx is None:
   749	                self.process_sequence_assignment(target, rvalue_reg, line)
   750	            else:
   751	                self.process_iterator_tuple_assignment(target, rvalue_reg, line)
   752	        else:
   753	            assert False, "Unsupported assignment target"

   755	    def coerce_rvalue(self, rvalue: Value, rtype: RType, line: int) -> Value:
   756	        if is_float_rprimitive(rtype) and is_tagged(rvalue.type):
   757	            typename = rvalue.type.short_name()
   758	            if typename == "short_int":
   759	                typename = "int"
   760	            self.error(
   761	                "Incompatible value representations in assignment "
   762	                + f'(expression has type "{typename}", variable has type "float")',
   763	                line,
   764	            )
   765	        return self.coerce(rvalue, rtype, line)

   767	    def process_sequence_assignment(
   768	        self, target: AssignmentTargetTuple, rvalue: Value, line: int
   769	    ) -> None:
   770	        """Process assignment like 'x, y = s', where s is a variable-length list or tuple."""
   771	        # Check the length of sequence.
   772	        expected_len = Integer(len(target.items), c_pyssize_t_rprimitive)
   773	        self.builder.call_c(check_unpack_count_op, [rvalue, expected_len], line)
   774
   775	        # Read sequence items.
   776	        values = []
   777	        for i in range(len(target.items)):
   778	            item = target.items[i]
   779	            index = self.builder.load_int(i)
   780	            if is_list_rprimitive(rvalue.type):
   781	                item_value = self.primitive_op(list_get_item_unsafe_op, [rvalue, index], line)
   782	            else:
   783	                item_value = self.builder.gen_method_call(
   784	                    rvalue, "__getitem__", [index], item.type, line
   785	                )
   786	            values.append(item_value)
   787
   788	        # Assign sequence items to the target lvalues.
   789	        for lvalue, value in zip(target.items, values):
   790	            self.assign(lvalue, value, line)

   792	    def process_iterator_tuple_assignment_helper(
   793	        self, litem: AssignmentTarget, ritem: Value, line: int
   794	    ) -> None:
   795	        error_block, ok_block = BasicBlock(), BasicBlock()
   796	        self.add(Branch(ritem, error_block, ok_block, Branch.IS_ERROR))
   797
   798	        self.activate_block(error_block)
   799	        self.add(
   800	            RaiseStandardError(RaiseStandardError.VALUE_ERROR, "not enough values to unpack", line)
   801	        )
   802	        self.add(Unreachable())
   803
   804	        self.activate_block(ok_block)
   805	        self.assign(litem, ritem, line)
   806
   807	    def process_iterator_tuple_assignment(
   808	        self, target: AssignmentTargetTuple, rvalue_reg: Value, line: int
   809	    ) -> None:
   810	        iterator = self.primitive_op(iter_op, [rvalue_reg], line)
   811
   812	        # This may be the whole lvalue list if there is no starred value
   813	        split_idx = target.star_idx if target.star_idx is not None else len(target.items)
   814
   815	        # Assign values before the first starred value
   816	        for litem in target.items[:split_idx]:
   817	            ritem = self.call_c(next_op, [iterator], line)
   818	            error_block, ok_block = BasicBlock(), BasicBlock()
   819	            self.add(Branch(ritem, error_block, ok_block, Branch.IS_ERROR))
   820
   821	            self.activate_block(error_block)
   822	            self.add(
   823	                RaiseStandardError(
   824	                    RaiseStandardError.VALUE_ERROR, "not enough values to unpack", line
   825	                )
   826	            )
   827	            self.add(Unreachable())
   828
   829	            self.activate_block(ok_block)
   830
   831	            self.assign(litem, ritem, line)
   832
   833	        # Assign the starred value and all values after it
   834	        if target.star_idx is not None:
   835	            post_star_vals = target.items[split_idx + 1 :]
   836	            iter_list = self.primitive_op(to_list, [iterator], line)
   837	            iter_list_len = self.builtin_len(iter_list, line)
   838	            post_star_len = Integer(len(post_star_vals))
   839	            condition = self.binary_op(post_star_len, iter_list_len, "<=", line)
   840
   841	            error_block, ok_block = BasicBlock(), BasicBlock()
   842	            self.add(Branch(condition, ok_block, error_block, Branch.BOOL))
   843
   844	            self.activate_block(error_block)
   845	            self.add(
   846	                RaiseStandardError(
   847	                    RaiseStandardError.VALUE_ERROR, "not enough values to unpack", line
   848	                )
   849	            )
   850	            self.add(Unreachable())
   851
   852	            self.activate_block(ok_block)
   853
   854	            for litem in reversed(post_star_vals):
   855	                ritem = self.primitive_op(list_pop_last, [iter_list], line)
   856	                self.assign(litem, ritem, line)
   857
   858	            # Assign the starred value
   859	            self.assign(target.items[target.star_idx], iter_list, line)
   860
   861	        # There is no starred value, so check if there are extra values in rhs that
   862	        # have not been assigned.
   863	        else:
   864	            extra = self.call_c(next_op, [iterator], line)
   865	            error_block, ok_block = BasicBlock(), BasicBlock()
   866	            self.add(Branch(extra, ok_block, error_block, Branch.IS_ERROR))
   867
   868	            self.activate_block(error_block)
   869	            self.add(
   870	                RaiseStandardError(
   871	                    RaiseStandardError.VALUE_ERROR, "too many values to unpack", line
   872	                )
   873	            )
   874	            self.add(Unreachable())
   875
   876	            self.activate_block(ok_block)

   944	        if isinstance(target_type, UnionType):

   946	                [self.get_sequence_type_from_type(item) for item in target_type.items]

   956	        elif isinstance(target_type, TupleType):
   957	            # Tuple might have elements of different types.

   959	            if len(rtypes) == 1:

    19	def setup_callable_class(builder: IRBuilder) -> None:
    20	    """Generate an (incomplete) callable class representing a function.
    21
    22	    This can be a nested function or a function within a non-extension
    23	    class.  Also set up the 'self' variable for that class.
    24
    25	    This takes the most recently visited function and returns a
    26	    ClassIR to represent that function. Each callable class contains
    27	    an environment attribute which points to another ClassIR
    28	    representing the environment class where some of its variables can
    29	    be accessed.
    30
    31	    Note that some methods, such as '__call__', are not yet
    32	    created here. Use additional functions, such as
    33	    add_call_to_callable_class(), to add them.
    34
    35	    Return a newly constructed ClassIR representing the callable
    36	    class for the nested function.
    37	    """
    38	    # Check to see that the name has not already been taken. If so,
    39	    # rename the class. We allow multiple uses of the same function
    40	    # name because this is valid in if-else blocks. Example:
    41	    #
    42	    #     if True:
    43	    #         def foo():          ---->    foo_obj()
    44	    #             return True
    45	    #     else:
    46	    #         def foo():          ---->    foo_obj_0()
    47	    #             return False
    48	    name = base_name = f"{builder.fn_info.namespaced_name()}_obj"
    49	    count = 0
    50	    while name in builder.callable_class_names:
    51	        name = base_name + "_" + str(count)
    52	        count += 1
    53	    builder.callable_class_names.add(name)
    54
    55	    # Define the actual callable class ClassIR, and set its
    56	    # environment to point at the previously defined environment
    57	    # class.
    58	    callable_class_ir = ClassIR(name, builder.module_name, is_generated=True, is_final_class=True)
    59
    60	    # The functools @wraps decorator attempts to call setattr on
    61	    # nested functions, so we create a dict for these nested
    62	    # functions.
    63	    # https://github.com/python/cpython/blob/3.7/Lib/functools.py#L58
    64	    if builder.fn_info.is_nested:
    65	        callable_class_ir.has_dict = True
    66
    67	    # If the enclosing class doesn't contain nested (which will happen if
    68	    # this is a toplevel lambda), don't set up an environment.
    69	    if builder.fn_infos[-2].contains_nested:
    70	        callable_class_ir.attributes[ENV_ATTR_NAME] = RInstance(builder.fn_infos[-2].env_class)
    71	    callable_class_ir.mro = [callable_class_ir]
    72	    builder.fn_info.callable_class = ImplicitClass(callable_class_ir)
    73	    builder.classes.append(callable_class_ir)
    74
    75	    # Add a 'self' variable to the environment of the callable class,
    76	    # and store that variable in a register to be accessed later.
    77	    self_target = builder.add_self_to_env(callable_class_ir)
    78	    builder.fn_info.callable_class.self_reg = builder.read(self_target, builder.fn_info.fitem.line)

    82	    builder: IRBuilder,
    83	    args: list[Register],
    84	    blocks: list[BasicBlock],
    85	    sig: FuncSignature,
    86	    fn_info: FuncInfo,

    92	    given callable class, used to represent that function.

    95	    nargs = len(sig.args) - sig.num_bitmap_args

   102	    )
   103	    fn_info.callable_class.ir.methods["__call__"] = call_fn_ir
   104	    fn_info.callable_class.ir.method_decls["__call__"] = call_fn_decl

   108	def add_get_to_callable_class(builder: IRBuilder, fn_info: FuncInfo) -> None:
   109	    """Generate the '__get__' method for a callable class."""
   110	    line = fn_info.fitem.line
   111	    with builder.enter_method(
   112	        fn_info.callable_class.ir,
   113	        "__get__",
   114	        object_rprimitive,
   115	        fn_info,
   116	        self_type=object_rprimitive,

   119	        builder.add_argument("owner", object_rprimitive)

   122	        # object. If accessed through an object, create a new bound
   123	        # instance method object.
   124	        instance_block, class_block = BasicBlock(), BasicBlock()
   125	        comparison = builder.translate_is_op(
   126	            builder.read(instance), builder.none_object(), "is", line
   127	        )
   128	        builder.add_bool_branch(comparison, class_block, instance_block)
   129
   130	        builder.activate_block(class_block)
   131	        builder.add(Return(builder.self()))
   132
   133	        builder.activate_block(instance_block)
   134	        builder.add(
   135	            Return(builder.call_c(method_new_op, [builder.self(), builder.read(instance)], line))

   139	def instantiate_callable_class(builder: IRBuilder, fn_info: FuncInfo) -> Value:
   140	    """Create an instance of a callable class for a function.
   141
   142	    Calls to the function will actually call this instance.
   143
   144	    Note that fn_info refers to the function being assigned, whereas
   145	    builder.fn_info refers to the function encapsulating the function
   146	    being turned into a callable class.
   147	    """
   148	    fitem = fn_info.fitem
   149	    func_reg = builder.add(Call(fn_info.callable_class.ir.ctor, [], fitem.line))
   150
   151	    # Set the environment attribute of the callable class to point at
   152	    # the environment class defined in the callable class' immediate
   153	    # outer scope. Note that there are three possible environment
   154	    # class registers we may use. This depends on what the encapsulating
   155	    # (parent) function is:
   156	    #
   157	    # - A nested function: the callable class is instantiated
   158	    #   from the current callable class' '__call__' function, and hence
   159	    #   the callable class' environment register is used.
   160	    # - A generator function: the callable class is instantiated
   161	    #   from the '__next__' method of the generator class, and hence the
   162	    #   environment of the generator class is used.
   163	    # - Regular function: we use the environment of the original function.
   164	    curr_env_reg = None
   165	    if builder.fn_info.is_generator:
   166	        curr_env_reg = builder.fn_info.generator_class.curr_env_reg
   167	    elif builder.fn_info.is_nested:
   168	        curr_env_reg = builder.fn_info.callable_class.curr_env_reg
   169	    elif builder.fn_info.contains_nested:
   170	        curr_env_reg = builder.fn_info.curr_env_reg
   171	    if curr_env_reg:
   172	        builder.add(SetAttr(func_reg, ENV_ATTR_NAME, curr_env_reg, fitem.line))
   173	    return func_reg

    87	def transform_class_def(builder: IRBuilder, cdef: ClassDef) -> None:
    88	    """Create IR for a class definition.
    89
    90	    This can generate both extension (native) and non-extension
    91	    classes.  These are generated in very different ways. In the
    92	    latter case we construct a Python type object at runtime by doing
    93	    the equivalent of "type(name, bases, dict)" in IR. Extension

    95	    mypyc.codegen.emitclass.

   100	        builder.error("Nested class definitions not supported", cdef.line)
   101	        return

   106	    # classes aren't necessarily populated yet at

   108	    if any(ir.base_mro[i].base != ir.base_mro[i + 1] for i in range(len(ir.base_mro) - 1)):

   116	                        parent.fullname
   117	                    ),

   196	            # TODO: Do we need interpreted glue methods? Maybe not?

   294	        if not (is_class_var(lvalue) or stmt.is_final_def):
   295	            return
   296	        typ = self.builder.load_native_type_object(self.cdef.fullname)
   297	        value = self.builder.accept(stmt.rvalue)
   298	        self.builder.primitive_op(
   299	            py_setattr_op, [typ, self.builder.load_str(lvalue.name), value], stmt.line
   300	        )
   301	        if self.builder.non_function_scope() and stmt.is_final_def:
   302	            self.builder.init_final_static(lvalue, value, self.cdef.name)
   303
   304	    def finalize(self, ir: ClassIR) -> None:
   305	        attrs_with_defaults, default_assignments = find_attr_initializers(
   306	            self.builder, self.cdef, self.skip_attr_default
   307	        )
   308	        ir.attrs_with_defaults.update(attrs_with_defaults)
   309	        generate_attr_defaults_init(self.builder, self.cdef, default_assignments)
   310	        create_ne_from_eq(self.builder, self.cdef)

   313	class DataClassBuilder(ExtClassBuilder):
   314	    # controls whether an __annotations__ attribute should be added to the class
   315	    # __dict__.  This is not desirable for attrs classes where auto_attribs is
   316	    # disabled, as attrs will reject it.
   317	    add_annotations_to_dict = True
   318
   319	    def __init__(self, builder: IRBuilder, cdef: ClassDef) -> None:
   320	        super().__init__(builder, cdef)
   321	        self.non_ext = self.create_non_ext_info()
   322
   323	    def create_non_ext_info(self) -> NonExtClassInfo:
   324	        """Set up a NonExtClassInfo to track dataclass attributes.
   325
   326	        In addition to setting up a normal extension class for dataclasses,
   327	        we also collect its class attributes like a non-extension class so
   328	        that we can hand them to the dataclass decorator.
   329	        """
   330	        return NonExtClassInfo(
   331	            self.builder.call_c(dict_new_op, [], self.cdef.line),
   332	            self.builder.add(TupleSet([], self.cdef.line)),
   333	            self.builder.call_c(dict_new_op, [], self.cdef.line),
   334	            self.builder.add(LoadAddress(type_object_op.type, type_object_op.src, self.cdef.line)),
   335	        )
   336
   337	    def skip_attr_default(self, name: str, stmt: AssignmentStmt) -> bool:
   338	        return stmt.type is not None
   339
   340	    def get_type_annotation(self, stmt: AssignmentStmt) -> TypeInfo | None:
   341	        # We populate __annotations__ because dataclasses uses it to determine
   342	        # which attributes to compute on.
   343	        ann_type = get_proper_type(stmt.type)
   344	        if isinstance(ann_type, Instance):
   345	            return ann_type.type
   346	        return None
   347
   348	    def add_attr(self, lvalue: NameExpr, stmt: AssignmentStmt) -> None:
   349	        add_non_ext_class_attr_ann(
   350	            self.builder, self.non_ext, lvalue, stmt, self.get_type_annotation
   351	        )
   352	        add_non_ext_class_attr(
   353	            self.builder, self.non_ext, lvalue, stmt, self.cdef, self.attrs_to_cache
   354	        )
   355	        super().add_attr(lvalue, stmt)
   356
   357	    def finalize(self, ir: ClassIR) -> None:
   358	        """Generate code to finish instantiating a dataclass.
   359
   360	        This works by replacing all of the attributes on the class
   361	        (which will be descriptors) with whatever they would be in a
   362	        non-extension class, calling dataclass, then switching them back.
   363
   364	        The resulting class is an extension class and instances of it do not
   365	        have a __dict__ (unless something else requires it).
   366	        All methods written explicitly in the source are compiled and
   367	        may be called through the vtable while the methods generated
   368	        by dataclasses are interpreted and may not be.
   369
   370	        (If we just called dataclass without doing this, it would think that all
   371	        of the descriptors for our attributes are default values and generate an
   372	        incorrect constructor. We need to do the switch so that dataclass gets the
   373	        appropriate defaults.)
   374	        """
   375	        super().finalize(ir)
   376	        assert self.type_obj
   377	        add_dunders_to_non_ext_dict(
   378	            self.builder, self.non_ext, self.cdef.line, self.add_annotations_to_dict
   379	        )
   380	        dec = self.builder.accept(
   381	            next(d for d in self.cdef.decorators if is_dataclass_decorator(d))
   382	        )
   383	        dataclass_type_val = self.builder.load_str(dataclass_type(self.cdef) or "unknown")
   384	        self.builder.call_c(
   385	            dataclass_sleight_of_hand,
   386	            [dec, self.type_obj, self.non_ext.dict, self.non_ext.anns, dataclass_type_val],
   387	            self.cdef.line,
   388	        )
