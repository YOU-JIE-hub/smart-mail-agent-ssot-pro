   100	def fontify(pytext, searchfrom = 0, searchto = None):
   101	    if searchto is None:
   102	        searchto = len(pytext)
   103	    # Cache a few attributes for quicker reference.
   104	    search = matchRE.search
   105	    idSearch = idRE.search

   107	    tags = []
   108	    tags_append = tags.append
   109	    commentTag = 'comment'
   110	    stringTag = 'string'
   111	    keywordTag = 'keyword'
   112	    identifierTag = 'identifier'
   113
   114	    start = 0
   115	    end = searchfrom
   116	    while 1:
   117	        m = search(pytext, end)
   118	        if m is None:
   119	            break   # EXIT LOOP
   120	        start = m.start()
   121	        if start >= searchto:
   122	            break   # EXIT LOOP
   123	        match = m.group(0)
   124	        end = start + len(match)
   125	        c = match[0]
   126	        if c not in "#'\"":
   127	            # Must have matched a keyword.
   128	            if start != searchfrom:
   129	                # there's still a redundant char before and after it, strip!
   130	                match = match[1:-1]
   131	                start = start + 1
   132	            else:
   133	                # this is the first keyword in the text.
   134	                # Only a space at the end.
   135	                match = match[:-1]
   136	            end = end - 1

   148	        elif c == "#":
   149	            tags_append((commentTag, start, end, None))
   150	        else:
   151	            tags_append((stringTag, start, end, None))

     9	class ArcIV:
    10		'''
    11		performs 'ArcIV' Stream Encryption of S using key
    12		Based on what is widely thought to be RSA's ArcIV algorithm.
    13		It produces output streams that are identical.
    14
    15		NB there is no separate decoder arciv(arciv(s,key),key) == s
    16		'''
    17		def __init__(self,key):
    18			self._key = key
    19			self.reset()
    20
    21		def reset(self):
    22			'''restore the cipher to it's start state'''
    23			#Initialize private key, k With the values of the key mod 256.
    24			#and sbox With numbers 0 - 255. Then compute sbox
    25			key = self._key
    26			if isUnicode(key): key = key.encode('utf8')
    27			sbox = list(range(256))
    28			k = list(range(256))
    29			lk = len(key)
    30			for i in sbox:
    31				k[i] = key[i % lk] % 256
    32
    33			#Re-order sbox using the private key, k.
    34			#Iterating each element of sbox re-calculate the counter j
    35			#Then interchange the elements sbox[a] & sbox[b]
    36			j = 0
    37			for i in range(256):
    38				j = (j+sbox[i]+k[i]) % 256
    39				sbox[i], sbox[j] = sbox[j], sbox[i]
    40			self._sbox, self._i, self._j = sbox, 0, 0
    41
    42		def _encode(self, B):
    43			'''
    44			return the list of encoded bytes of B, B might be a string or a
    45			list of integers between 0 <= i <= 255
    46			'''

   157	    """This represents colors using the CMYK (cyan, magenta, yellow, black)
   158	    model commonly used in professional printing.  This is implemented
   159	    as a derived class so that renderers which only know about RGB "see it"
   160	    as an RGB color through its 'red','green' and 'blue' attributes, according
   161	    to an approximate function.
   162
   163	    The RGB approximation is worked out when the object in constructed, so
   164	    the color attributes should not be changed afterwards.
   165
   166	    Extra attributes may be attached to the class to support specific ink models,
   167	    and renderers may look for these."""
   168
   169	    _scale = 1.0
   170	    def __init__(self, cyan=0, magenta=0, yellow=0, black=0,
   171	                spotName=None, density=1, knockout=None, alpha=1):
   172	        """
   173	        Initialize with four colors in range [0-1]. the optional
   174	        spotName, density & knockout may be of use to specific renderers.
   175	        spotName is intended for use as an identifier to the renderer not client programs.
   176	        density is used to modify the overall amount of ink.
   177	        knockout is a renderer dependent option that determines whether the applied colour
   178	        knocksout (removes) existing colour; None means use the global default.
   179	        """
   180	        self.cyan = cyan
   181	        self.magenta = magenta
   182	        self.yellow = yellow
   183	        self.black = black
   184	        self.spotName = spotName
   185	        self.density = max(min(density,1),0)    # force into right range
   186	        self.knockout = knockout
   187	        self.alpha = alpha
   188
   189	        # now work out the RGB approximation. override
   190	        self.red, self.green, self.blue = cmyk2rgb( (cyan, magenta, yellow, black) )
   191
   192	        if density<1:
   193	            #density adjustment of rgb approximants, effectively mix with white
   194	            r, g, b = self.red, self.green, self.blue
   195	            r = density*(r-1)+1
   196	            g = density*(g-1)+1
   197	            b = density*(b-1)+1
   198	            self.red, self.green, self.blue = (r,g,b)
   199
   200	    def __repr__(self):
   201	        return "%s(%s%s%s%s%s)" % (self.__class__.__name__,
   202	            fp_str(self.cyan, self.magenta, self.yellow, self.black).replace(' ',','),
   203	            (self.spotName and (',spotName='+repr(self.spotName)) or ''),
   204	            (self.density!=1 and (',density='+fp_str(self.density)) or ''),
   205	            (self.knockout is not None and (',knockout=%d' % self.knockout) or ''),
   206	            (self.alpha is not None and (',alpha=%s' % self.alpha) or ''),
   207	            )
   208
   209	    def fader(self, n, reverse=False):
   210	        '''return n colors based on density fade
   211	        *NB* note this dosen't reach density zero'''
   212	        scale = self._scale
   213	        dd = scale/float(n)
   214	        L = [self.clone(density=scale - i*dd) for i in range(n)]
   215	        if reverse: L.reverse()
   216	        return L

   220	        """obvious way to compare colours
   221	        Comparing across the two color models is of limited use.
   222	        >>> cmp(CMYKColor(0,0,0,1),None)
   223	        -1
   224	        >>> cmp(CMYKColor(0,0,0,1),_CMYK_black)
   225	        0
   226	        >>> cmp(PCMYKColor(0,0,0,100),_CMYK_black)
   227	        0
   228	        >>> cmp(CMYKColor(0,0,0,1),Color(0,0,1)),Color(0,0,0).rgba()==CMYKColor(0,0,0,1).rgba()
   229	        (-1, True)
   230	        """
   231	        return self.cyan, self.magenta, self.yellow, self.black, self.density, self.spotName, self.alpha

   234	        return isinstance(other,CMYKColor)

   249	        if not D:
   250	            for n,v in getAllNamedColors().items():
   251	                if isinstance(v,CMYKColor):
   252	                    t = v.cyan,v.magenta,v.yellow,v.black

   303	def cmyk2rgb(cmyk,density=1):
   304	    "Convert from a CMYK color tuple to an RGB color tuple"
   305	    c,m,y,k = cmyk
   306	    # From the Adobe Postscript Ref. Manual 2nd ed.
   307	    r = 1.0 - min(1.0, c + k)
   308	    g = 1.0 - min(1.0, m + k)
   309	    b = 1.0 - min(1.0, y + k)
   310	    return (r,g,b)

   312	def rgb2cmyk(r,g,b):
   313	    '''one way to get cmyk from rgb'''
   314	    c = 1 - r
   315	    m = 1 - g
   316	    y = 1 - b
   317	    k = min(c,m,y)
   318	    c = min(1,max(0,c-k))
   319	    m = min(1,max(0,m-k))
   320	    y = min(1,max(0,y-k))
   321	    k = min(1,max(0,k))
   322	    return (c,m,y,k)

   324	def color2bw(colorRGB):
   325	    "Transform an RGB color to a black and white equivalent."
   326
   327	    col = colorRGB
   328	    r, g, b, a = col.red, col.green, col.blue, col.alpha
   329	    n = (r + g + b) / 3.0
   330	    bwColorRGB = Color(n, n, n, a)
   331	    return bwColorRGB

   333	def HexColor(val, htmlOnly=False, hasAlpha=False):
   334	    """This function converts a hex string, or an actual integer number,
   335	    into the corresponding color.  E.g., in "#AABBCC" or 0xAABBCC,
   336	    AA is the red, BB is the green, and CC is the blue (00-FF).
   337
   338	    An alpha value can also be given in the form #AABBCCDD or 0xAABBCCDD where
   339	    DD is the alpha value if hasAlpha is True.
   340
   341	    For completeness I assume that #aabbcc or 0xaabbcc are hex numbers
   342	    otherwise a pure integer is converted as decimal rgb.  If htmlOnly is true,
   343	    only the #aabbcc form is allowed.
   344
   345	    >>> HexColor('#ffffff')
   346	    Color(1,1,1,1)
   347	    >>> HexColor('#FFFFFF')
   348	    Color(1,1,1,1)
   349	    >>> HexColor('0xffffff')
   350	    Color(1,1,1,1)
   351	    >>> HexColor('16777215')
   352	    Color(1,1,1,1)
   353
   354	    An '0x' or '#' prefix is required for hex (as opposed to decimal):
   355
   356	    >>> HexColor('ffffff')
   357	    Traceback (most recent call last):
   358	    ValueError: invalid literal for int() with base 10: 'ffffff'
   359
   360	    >>> HexColor('#FFFFFF', htmlOnly=True)
   361	    Color(1,1,1,1)
   362	    >>> HexColor('0xffffff', htmlOnly=True)
   363	    Traceback (most recent call last):
   364	    ValueError: not a hex string
   365	    >>> HexColor('16777215', htmlOnly=True)
   366	    Traceback (most recent call last):
   367	    ValueError: not a hex string
   368
   369	    """ #" for emacs
   370
   371	    if isStr(val):
   372	        val = asNative(val)
   373	        b = 10
   374	        if val[:1] == '#':
   375	            val = val[1:]
   376	            b = 16
   377	            if len(val) == 8:
   378	                alpha = True
   379	        else:
   380	            if htmlOnly:
   381	                raise ValueError('not a hex string')
   382	            if val[:2].lower() == '0x':
   383	                b = 16
   384	                val = val[2:]
   385	                if len(val) == 8:
   386	                    alpha = True
   387	        val = int(val,b)
   388	    if hasAlpha:
   389	        return Color(((val>>24)&0xFF)/255.0,((val>>16)&0xFF)/255.0,((val>>8)&0xFF)/255.0,(val&0xFF)/255.0)
   390	    return Color(((val>>16)&0xFF)/255.0,((val>>8)&0xFF)/255.0,(val&0xFF)/255.0)

   392	def linearlyInterpolatedColor(c0, c1, x0, x1, x):
   393	    """
   394	    Linearly interpolates colors. Can handle RGB, CMYK and PCMYK
   395	    colors - give ValueError if colours aren't the same.
   396	    Doesn't currently handle 'Spot Color Interpolation'.
   397	    """
   398
   399	    if c0.__class__ != c1.__class__:
   400	        raise ValueError("Color classes must be the same for interpolation!\nGot %r and %r'"%(c0,c1))
   401	    if x1<x0:
   402	        x0,x1,c0,c1 = x1,x0,c1,c0 # normalized so x1>x0
   403	    if x<x0-1e-8 or x>x1+1e-8: # fudge factor for numerical problems
   404	        raise ValueError("Can't interpolate: x=%f is not between %f and %f!" % (x,x0,x1))
   405	    if x<=x0:
   406	        return c0
   407	    elif x>=x1:
   408	        return c1
   409
   410	    cname = c0.__class__.__name__
   411	    dx = float(x1-x0)
   412	    x = x-x0
   413
   414	    if cname == 'Color': # RGB
   415	        r = c0.red+x*(c1.red - c0.red)/dx
   416	        g = c0.green+x*(c1.green- c0.green)/dx
   417	        b = c0.blue+x*(c1.blue - c0.blue)/dx
   418	        a = c0.alpha+x*(c1.alpha - c0.alpha)/dx
   419	        return Color(r,g,b,alpha=a)
   420	    elif cname == 'CMYKColor':
   421	        if cmykDistance(c0,c1)<1e-8:
   422	            #colors same do density and preserve spotName if any
   423	            assert c0.spotName == c1.spotName, "Identical cmyk, but different spotName"
   424	            c = c0.cyan
   425	            m = c0.magenta
   426	            y = c0.yellow
   427	            k = c0.black
   428	            d = c0.density+x*(c1.density - c0.density)/dx
   429	            a = c0.alpha+x*(c1.alpha - c0.alpha)/dx
   430	            return CMYKColor(c,m,y,k, density=d, spotName=c0.spotName, alpha=a)
   431	        elif cmykDistance(c0,_CMYK_white)<1e-8:
   432	            #special c0 is white
   433	            c = c1.cyan
   434	            m = c1.magenta
   435	            y = c1.yellow
   436	            k = c1.black
   437	            d = x*c1.density/dx
   438	            a = x*c1.alpha/dx
   439	            return CMYKColor(c,m,y,k, density=d, spotName=c1.spotName, alpha=a)
   440	        elif cmykDistance(c1,_CMYK_white)<1e-8:
   441	            #special c1 is white
   442	            c = c0.cyan
   443	            m = c0.magenta
   444	            y = c0.yellow
   445	            k = c0.black
   446	            d = x*c0.density/dx
   447	            d = c0.density*(1-x/dx)
   448	            a = c0.alpha*(1-x/dx)
   449	            return PCMYKColor(c,m,y,k, density=d, spotName=c0.spotName, alpha=a)
   450	        else:
   451	            c = c0.cyan+x*(c1.cyan - c0.cyan)/dx
   452	            m = c0.magenta+x*(c1.magenta - c0.magenta)/dx
   453	            y = c0.yellow+x*(c1.yellow - c0.yellow)/dx
   454	            k = c0.black+x*(c1.black - c0.black)/dx
   455	            d = c0.density+x*(c1.density - c0.density)/dx
   456	            a = c0.alpha+x*(c1.alpha - c0.alpha)/dx
   457	            return CMYKColor(c,m,y,k, density=d, alpha=a)
   458	    elif cname == 'PCMYKColor':
   459	        if cmykDistance(c0,c1)<1e-8:
   460	            #colors same do density and preserve spotName if any
   461	            assert c0.spotName == c1.spotName, "Identical cmyk, but different spotName"
   462	            c = c0.cyan
   463	            m = c0.magenta
   464	            y = c0.yellow
   465	            k = c0.black
   466	            d = c0.density+x*(c1.density - c0.density)/dx
   467	            a = c0.alpha+x*(c1.alpha - c0.alpha)/dx
   468	            return PCMYKColor(c*100,m*100,y*100,k*100, density=d*100,
   469	                              spotName=c0.spotName, alpha=100*a)
   470	        elif cmykDistance(c0,_CMYK_white)<1e-8:
   471	            #special c0 is white
   472	            c = c1.cyan
   473	            m = c1.magenta
   474	            y = c1.yellow
   475	            k = c1.black
   476	            d = x*c1.density/dx
   477	            a = x*c1.alpha/dx
   478	            return PCMYKColor(c*100,m*100,y*100,k*100, density=d*100,
   479	                              spotName=c1.spotName, alpha=a*100)
   480	        elif cmykDistance(c1,_CMYK_white)<1e-8:
   481	            #special c1 is white
   482	            c = c0.cyan
   483	            m = c0.magenta
   484	            y = c0.yellow
   485	            k = c0.black
   486	            d = x*c0.density/dx
   487	            d = c0.density*(1-x/dx)
   488	            a = c0.alpha*(1-x/dx)
   489	            return PCMYKColor(c*100,m*100,y*100,k*100, density=d*100,
   490	                              spotName=c0.spotName, alpha=a*100)
   491	        else:
   492	            c = c0.cyan+x*(c1.cyan - c0.cyan)/dx
   493	            m = c0.magenta+x*(c1.magenta - c0.magenta)/dx
   494	            y = c0.yellow+x*(c1.yellow - c0.yellow)/dx
   495	            k = c0.black+x*(c1.black - c0.black)/dx
   496	            d = c0.density+x*(c1.density - c0.density)/dx
   497	            a = c0.alpha+x*(c1.alpha - c0.alpha)/dx
   498	            return PCMYKColor(c*100,m*100,y*100,k*100, density=d*100, alpha=a*100)
   499	    else:
   500	        raise ValueError("Can't interpolate: Unknown color class %s!" % cname)

   694	def colorDistance(col1, col2):
   695	    """Returns a number between 0 and root(3) stating how similar
   696	    two colours are - distance in r,g,b, space.  Only used to find
   697	    names for things."""
   698	    return math.sqrt(
   699	            (col1.red - col2.red)**2 +
   700	            (col1.green - col2.green)**2 +
   701	            (col1.blue - col2.blue)**2
   702	            )

   751	def hue2rgb(m1, m2, h):
   752	    if h<0: h += 1
   753	    if h>1: h -= 1
   754	    if h*6<1: return m1+(m2-m1)*h*6
   755	    if h*2<1: return m2
   756	    if h*3<2: return m1+(m2-m1)*(4-6*h)
   757	    return m1

   759	def hsl2rgb(h, s, l):
   760	    if l<=0.5:
   761	        m2 = l*(s+1)
   762	    else:
   763	        m2 = l+s-l*s
   764	    m1 = l*2-m2
   765	    return hue2rgb(m1, m2, h+1./3),hue2rgb(m1, m2, h),hue2rgb(m1, m2, h-1./3)

   768	_re_css = re.compile(r'^\s*(pcmyk|cmyk|rgb|hsl)(a|)\s*\(\s*([^)]*)\)\s*$')
   769	class cssParse:
   770	    def pcVal(self,v):
   771	        v = v.strip()
   772	        try:
   773	            c=float(v[:-1])
   774	            c=min(100,max(0,c))/100.
   775	        except:
   776	            raise ValueError('bad percentage argument value %r in css color %r' % (v,self.s))
   777	        return c
   778
   779	    def rgbPcVal(self,v):
   780	        return int(self.pcVal(v)*255+0.5)/255.
   781
   782	    def rgbVal(self,v):
   783	        v = v.strip()
   784	        try:
   785	            c=float(v)
   786	            if 0<c<=1: c *= 255
   787	            return int(min(255,max(0,c)))/255.
   788	        except:
   789	            raise ValueError('bad argument value %r in css color %r' % (v,self.s))
   790
   791	    def hueVal(self,v):
   792	        v = v.strip()
   793	        try:
   794	            c=float(v)
   795	            return ((c%360+360)%360)/360.
   796	        except:
   797	            raise ValueError('bad hue argument value %r in css color %r' % (v,self.s))
   798
   799	    def alphaVal(self,v,c=1,n='alpha'):
   800	        try:
   801	            a = float(v)
   802	            return min(c,max(0,a))
   803	        except:
   804	            raise ValueError('bad %s argument value %r in css color %r' % (n,v,self.s))
   805
   806	    _n_c = dict(pcmyk=(4,100,True,False),cmyk=(4,1,True,False),hsl=(3,1,False,True),rgb=(3,1,False,False))
   807
   808	    def __call__(self,s):
   809	        n = _re_css.match(s)
   810	        if not n: return
   811	        self.s = s
   812	        b,c,cmyk,hsl = self._n_c[n.group(1)]
   813	        ha = n.group(2)
   814	        n = n.group(3).split(',')   #strip parens and split on comma
   815	        if len(n)!=(b+(ha and 1 or 0)):
   816	            raise ValueError('css color %r has wrong number of components' % s)
   817	        if ha:
   818	            n,a = n[:b],self.alphaVal(n[b],c)
   819	        else:
   820	            a = c
   821
   822	        if cmyk:
   823	            C = self.alphaVal(n[0],c,'cyan')
   824	            M = self.alphaVal(n[1],c,'magenta')
   825	            Y = self.alphaVal(n[2],c,'yellow')
   826	            K = self.alphaVal(n[3],c,'black')
   827	            return (c>1 and PCMYKColor or CMYKColor)(C,M,Y,K,alpha=a)
   828	        else:
   829	            if hsl:
   830	                R,G,B= hsl2rgb(self.hueVal(n[0]),self.pcVal(n[1]),self.pcVal(n[2]))
   831	            else:
   832	                R,G,B = list(map('%' in n[0] and self.rgbPcVal or self.rgbVal,n))
   833
   834	            return Color(R,G,B,a)

   842	    security holes from tie to time.  In April 2023 we are creating explicit, "dumb" parsing code to
   843	    replace this.  Acceptable patterns are

   862	        if isinstance(arg,Color): return arg

   867	        elif isStr(arg):

   908	                G.update(self._G)

   998	def fade(aSpotColor, percentages):
   999	    """Waters down spot colors and returns a list of new ones
  1000
  1001	    e.g fade(myColor, [100,80,60,40,20]) returns a list of five colors
  1002	    """
  1003	    out = []
  1004	    for percent in percentages:
  1005	        frac = percent * 0.01   #assume they give us numbers from 0 to 100
  1006	        newCyan = frac * aSpotColor.cyan
  1007	        newMagenta = frac * aSpotColor.magenta
  1008	        newYellow = frac * aSpotColor.yellow
  1009	        newBlack = frac * aSpotColor.black
  1010	        newDensity = frac * aSpotColor.density
  1011	        newSpot = CMYKColor(    newCyan, newMagenta, newYellow, newBlack,
  1012	                            spotName = aSpotColor.spotName,
  1013	                            density = newDensity)
  1014	        out.append(newSpot)
  1015	    return out
