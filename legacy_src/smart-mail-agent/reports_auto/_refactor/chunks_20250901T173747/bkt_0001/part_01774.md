  2535	#       < sup ="pts"] [rise="pts"]> < /sup > - superscript

    98	    1) absolute values go through unchanged

   555	    def _elementWidth(self,v,s):
   556	        if isinstance(v,(list,tuple)):
   557	            w = 0
   558	            for e in v:
   559	                ew = self._elementWidth(e,s)

   563	        elif isinstance(v,Flowable):
   564	            if v._fixedWidth:

   566	                if hasattr(v, 'drawWidth') and isinstance(v.drawWidth,(int,float)): return v.drawWidth
   567	            if hasattr(v,'__styledWrap__'): #very experimental
   568	                try:
   569	                    return getattr(v,'__styledWrap__')(s)[0]
   570	                except:
   571	                    pass
   572	        # Even if something is fixedWidth, the attribute to check is not
   573	        # necessarily consistent (cf. Image.drawWidth).  Therefore, we'll
   574	        # be extra-careful and fall through to this code if necessary.
   575	        if hasattr(v, 'minWidth'):

   587	                return 0
