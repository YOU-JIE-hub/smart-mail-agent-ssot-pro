  2526	# The ParaFormatter will be able to format the following
  2527	# tags:
  2528	#       < /b > - bold
  2529	#       < /i > - italics
  2530	#       < u [color="red"] [width="pts"] [offset="pts"]> < /u > - underline
  2531	#           width and offset can be empty meaning use existing canvas line width
  2532	#           or with an f/F suffix regarded as a fraction of the font size
  2533	#       < strike > < /strike > - strike through has the same parameters as underline
  2534	#       < super [size="pts"] [rise="pts"]> < /super > - superscript
  2535	#       < sup ="pts"] [rise="pts"]> < /sup > - superscript
  2536	#       < sub ="pts"] [rise="pts"]> < /sub > - subscript
  2537	#       <font name=fontfamily/fontname color=colorname size=float>
  2538	#        <span name=fontfamily/fontname color=colorname backcolor=colorname size=float style=stylename>
  2539	#       < bullet > </bullet> - bullet text (at head of para only)
  2540	#       <onDraw name=callable label="a label"/>
  2541	#       <index [name="callablecanvasattribute"] label="a label"/>
  2542	#       <link>link text</link>
  2543	#           attributes of links
  2544	#               size/fontSize/uwidth/uoffset=num
  2545	#               name/face/fontName=name
  2546	#               fg/textColor/color/ucolor=color
  2547	#               backcolor/backColor/bgcolor=color
  2548	#               dest/destination/target/href/link=target
  2549	#               underline=bool turn on underline
  2550	#       <a>anchor text</a>
  2551	#           attributes of anchors
  2552	#               fontSize=num
  2553	#               fontName=name
  2554	#               fg/textColor/color=color
  2555	#               backcolor/backColor/bgcolor=color
  2556	#               href=href
  2557	#       <a name="anchorpoint"/>
  2558	#       <unichar name="unicode character name"/>
  2559	#       <unichar value="unicode code point"/>
  2560	#       <img src="path" width="1in" height="1in" valign="bottom"/>
  2561	#               width="w%" --> fontSize*w/100   idea from Roberto Alsina
  2562	#               height="h%" --> linewidth*h/100 <ralsina@netmanagers.com.ar>
  2563	#       <greek> - </greek>
  2564	#       <nobr> ... </nobr> turn off word breaking and hyphenation
