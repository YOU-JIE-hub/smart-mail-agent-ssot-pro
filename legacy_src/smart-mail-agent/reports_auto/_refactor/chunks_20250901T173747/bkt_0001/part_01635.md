   367	        obj = getattr(r, attr)
   368	        if isinstance(obj, types.ModuleType):
   369	            return False

   509	    stub: nodes.TypeInfo, runtime: MaybeMissing[type[Any]], object_path: list[str]
   510	) -> Iterator[Error]:
   511	    if stub.is_type_check_only:
   512	        # This type only exists in stubs, we only check that the runtime part
   513	        # is missing. Other checks are not required.
   514	        if not isinstance(runtime, Missing):
   515	            yield Error(
   516	                object_path,
   517	                'is marked as "@type_check_only", but also exists at runtime',
   518	                stub,
   519	                runtime,
   520	                stub_desc=repr(stub),
   521	            )
   522	        return
   523
   524	    if isinstance(runtime, Missing):
   525	        yield Error(object_path, "is not present at runtime", stub, runtime, stub_desc=repr(stub))
   526	        return
   527	    if not isinstance(runtime, type):
   528	        # Yes, some runtime objects can be not types, no way to tell mypy about that.
   529	        yield Error(object_path, "is not a type", stub, runtime, stub_desc=repr(stub))  # type: ignore[unreachable]
   530	        return
   531
   532	    yield from _verify_final(stub, runtime, object_path)
   533	    is_runtime_typeddict = stub.typeddict_type is not None and is_typeddict(runtime)
   534	    yield from _verify_metaclass(
   535	        stub, runtime, object_path, is_runtime_typeddict=is_runtime_typeddict
   536	    )
   537
   538	    # Check everything already defined on the stub class itself (i.e. not inherited)
   539	    #
   540	    # Filter out non-identifier names, as these are (hopefully always?) whacky/fictional things
   541	    # (like __mypy-replace or __mypy-post_init, etc.) that don't exist at runtime,
   542	    # and exist purely for internal mypy reasons
   543	    to_check = {name for name in stub.names if name.isidentifier()}
   544	    # Check all public things on the runtime class
   545	    to_check.update(
   546	        m for m in vars(runtime) if not is_probably_private(m) and m not in IGNORABLE_CLASS_DUNDERS
   547	    )
   548	    # Special-case the __init__ method for Protocols and the __new__ method for TypedDicts
   549	    #
   550	    # TODO: On Python <3.11, __init__ methods on Protocol classes
   551	    # are silently discarded and replaced.
   552	    # However, this is not the case on Python 3.11+.
   553	    # Ideally, we'd figure out a good way of validating Protocol __init__ methods on 3.11+.
   554	    if stub.is_protocol:
   555	        to_check.discard("__init__")
   556	    if is_runtime_typeddict:
   557	        to_check.discard("__new__")
   558
   559	    for entry in sorted(to_check):
   560	        mangled_entry = entry
   561	        if entry.startswith("__") and not entry.endswith("__"):
   562	            mangled_entry = f"_{stub.name.lstrip('_')}{entry}"
   563	        stub_to_verify = next((t.names[entry].node for t in stub.mro if entry in t.names), MISSING)
   564	        assert stub_to_verify is not None
   565	        try:
   566	            try:
   567	                runtime_attr = getattr(runtime, mangled_entry)
   568	            except AttributeError:
   569	                runtime_attr = inspect.getattr_static(runtime, mangled_entry, MISSING)
   570	        except Exception:
   571	            # Catch all exceptions in case the runtime raises an unexpected exception
   572	            # from __getattr__ or similar.
   573	            continue
   574
   575	        # If it came from the metaclass, consider the runtime_attr to be MISSING
   576	        # for a more accurate message
   577	        if (
   578	            runtime_attr is not MISSING
   579	            and type(runtime) is not runtime
   580	            and getattr(runtime_attr, "__objclass__", None) is type(runtime)
   581	        ):
   582	            runtime_attr = MISSING
   583
   584	        # __setattr__ and __delattr__ on object are a special case,
   585	        # so if we only have these methods inherited from there, pretend that
   586	        # we don't have them. See python/typeshed#7385.
   587	        if (
   588	            entry in ("__setattr__", "__delattr__")
   589	            and runtime_attr is not MISSING
   590	            and runtime is not object
   591	            and getattr(runtime_attr, "__objclass__", None) is object
   592	        ):
   593	            runtime_attr = MISSING
   594
   595	        # Do not error for an object missing from the stub
   596	        # If the runtime object is a types.WrapperDescriptorType object
   597	        # and has a non-special dunder name.
   598	        # The vast majority of these are false positives.
   599	        if not (
   600	            isinstance(stub_to_verify, Missing)
   601	            and isinstance(runtime_attr, types.WrapperDescriptorType)
   602	            and is_dunder(mangled_entry, exclude_special=True)
   603	        ):
   604	            yield from verify(stub_to_verify, runtime_attr, object_path + [entry])
   605

   607	def _static_lookup_runtime(object_path: list[str]) -> MaybeMissing[Any]:
   608	    static_runtime = importlib.import_module(object_path[0])
   609	    for entry in object_path[1:]:
   610	        try:
   611	            static_runtime = inspect.getattr_static(static_runtime, entry)
   612	        except AttributeError:
   613	            # This can happen with mangled names, ignore for now.
   614	            # TODO: pass more information about ancestors of nodes/objects to verify, so we don't
   615	            # have to do this hacky lookup. Would be useful in several places.
   616	            return MISSING
   617	    return static_runtime

   620	def _verify_static_class_methods(
   621	    stub: nodes.FuncBase, runtime: Any, static_runtime: MaybeMissing[Any], object_path: list[str]
   622	) -> Iterator[str]:
   623	    if stub.name in ("__new__", "__init_subclass__", "__class_getitem__"):
   624	        # Special cased by Python, so don't bother checking
   625	        return
   626	    if inspect.isbuiltin(runtime):
   627	        # The isinstance checks don't work reliably for builtins, e.g. datetime.datetime.now, so do
   628	        # something a little hacky that seems to work well
   629	        probably_class_method = isinstance(getattr(runtime, "__self__", None), type)
   630	        if probably_class_method and not stub.is_class:
   631	            yield "runtime is a classmethod but stub is not"
   632	        if not probably_class_method and stub.is_class:
   633	            yield "stub is a classmethod but runtime is not"
   634	        return
   635
   636	    if static_runtime is MISSING:
   637	        return
   638
   639	    if isinstance(static_runtime, classmethod) and not stub.is_class:
   640	        yield "runtime is a classmethod but stub is not"
   641	    if not isinstance(static_runtime, classmethod) and stub.is_class:
   642	        yield "stub is a classmethod but runtime is not"
   643	    if isinstance(static_runtime, staticmethod) and not stub.is_static:
   644	        yield "runtime is a staticmethod but stub is not"
   645	    if not isinstance(static_runtime, staticmethod) and stub.is_static:
   646	        yield "stub is a staticmethod but runtime is not"

   649	def _verify_arg_name(
   650	    stub_arg: nodes.Argument, runtime_arg: inspect.Parameter, function_name: str
   651	) -> Iterator[str]:
   652	    """Checks whether argument names match."""
   653	    # Ignore exact names for most dunder methods
   654	    if is_dunder(function_name, exclude_special=True):
   655	        return
   656
   657	    if (
   658	        stub_arg.variable.name == runtime_arg.name
   659	        or stub_arg.variable.name.removeprefix("__") == runtime_arg.name
   660	    ):
   661	        return
   662
   663	    nonspecific_names = {"object", "args"}
   664	    if runtime_arg.name in nonspecific_names:
   665	        return
   666
   667	    def names_approx_match(a: str, b: str) -> bool:
   668	        a = a.strip("_")
   669	        b = b.strip("_")
   670	        return a.startswith(b) or b.startswith(a) or len(a) == 1 or len(b) == 1
   671
   672	    # Be more permissive about names matching for positional-only arguments
   673	    if runtime_arg.kind == inspect.Parameter.POSITIONAL_ONLY and names_approx_match(
   674	        stub_arg.variable.name, runtime_arg.name
   675	    ):
   676	        return
   677	    # This comes up with namedtuples, so ignore
   678	    if stub_arg.variable.name == "_self":
   679	        return
   680	    yield (
   681	        f'stub argument "{stub_arg.variable.name}" '
   682	        f'differs from runtime argument "{runtime_arg.name}"'
   683	    )

   686	def _verify_arg_default_value(
   687	    stub_arg: nodes.Argument, runtime_arg: inspect.Parameter
   688	) -> Iterator[str]:
   689	    """Checks whether argument default values are compatible."""
   690	    if runtime_arg.default is not inspect.Parameter.empty:
   691	        if stub_arg.kind.is_required():
   692	            yield (
   693	                f'runtime argument "{runtime_arg.name}" '
   694	                "has a default value but stub argument does not"
   695	            )
   696	        else:
   697	            runtime_type = get_mypy_type_of_runtime_value(runtime_arg.default)
   698	            # Fallback to the type annotation type if var type is missing. The type annotation
   699	            # is an UnboundType, but I don't know enough to know what the pros and cons here are.
   700	            # UnboundTypes have ugly question marks following them, so default to var type.
   701	            # Note we do this same fallback when constructing signatures in from_overloadedfuncdef
   702	            stub_type = stub_arg.variable.type or stub_arg.type_annotation
   703	            if isinstance(stub_type, mypy.types.TypeVarType):
   704	                stub_type = stub_type.upper_bound
   705	            if (
   706	                runtime_type is not None
   707	                and stub_type is not None
   708	                # Avoid false positives for marker objects
   709	                and type(runtime_arg.default) is not object
   710	                # And ellipsis
   711	                and runtime_arg.default is not ...
   712	                and not is_subtype_helper(runtime_type, stub_type)
   713	            ):
   714	                yield (
   715	                    f'runtime argument "{runtime_arg.name}" '
   716	                    f"has a default value of type {runtime_type}, "
   717	                    f"which is incompatible with stub argument type {stub_type}"
   718	                )
   719	            if stub_arg.initializer is not None:
   720	                stub_default = evaluate_expression(stub_arg.initializer)
   721	                if (
   722	                    stub_default is not UNKNOWN
   723	                    and stub_default is not ...
   724	                    and runtime_arg.default is not UNREPRESENTABLE
   725	                ):
   726	                    defaults_match = True
   727	                    # We want the types to match exactly, e.g. in case the stub has
   728	                    # True and the runtime has 1 (or vice versa).
   729	                    if type(stub_default) is not type(runtime_arg.default):
   730	                        defaults_match = False
   731	                    else:
   732	                        try:
   733	                            defaults_match = bool(stub_default == runtime_arg.default)
   734	                        except Exception:
   735	                            # Exception can be raised in bool dunder method (e.g. numpy arrays)
   736	                            # At this point, consider the default to be different, it is probably
   737	                            # too complex to put in a stub anyway.
   738	                            defaults_match = False
   739	                    if not defaults_match:
   740	                        yield (
   741	                            f'runtime argument "{runtime_arg.name}" '
   742	                            f"has a default value of {runtime_arg.default!r}, "
   743	                            f"which is different from stub argument default {stub_default!r}"
   744	                        )
   745	    else:
   746	        if stub_arg.kind.is_optional():
   747	            yield (
   748	                f'stub argument "{stub_arg.variable.name}" has a default value '
   749	                f"but runtime argument does not"
   750	            )

   753	def maybe_strip_cls(name: str, args: list[nodes.Argument]) -> list[nodes.Argument]:
   754	    if args and name in ("__init_subclass__", "__class_getitem__"):
   755	        # These are implicitly classmethods. If the stub chooses not to have @classmethod, we
   756	        # should remove the cls argument
   757	        if args[0].variable.name == "cls":
   758	            return args[1:]
   759	    return args

   762	class Signature(Generic[T]):
   763	    def __init__(self) -> None:
   764	        self.pos: list[T] = []
   765	        self.kwonly: dict[str, T] = {}
   766	        self.varpos: T | None = None
   767	        self.varkw: T | None = None
   768
   769	    def __str__(self) -> str:
   770	        def get_name(arg: Any) -> str:
   771	            if isinstance(arg, inspect.Parameter):
   772	                return arg.name
   773	            if isinstance(arg, nodes.Argument):
   774	                return arg.variable.name
   775	            raise AssertionError
   776
   777	        def get_type(arg: Any) -> str | None:
   778	            if isinstance(arg, inspect.Parameter):
   779	                return None
   780	            if isinstance(arg, nodes.Argument):
   781	                return str(arg.variable.type or arg.type_annotation)
   782	            raise AssertionError
   783
   784	        def has_default(arg: Any) -> bool:
   785	            if isinstance(arg, inspect.Parameter):
   786	                return arg.default is not inspect.Parameter.empty
   787	            if isinstance(arg, nodes.Argument):
   788	                return arg.kind.is_optional()
   789	            raise AssertionError
   790
   791	        def get_desc(arg: Any) -> str:
   792	            arg_type = get_type(arg)
   793	            return (
   794	                get_name(arg)
   795	                + (f": {arg_type}" if arg_type else "")
   796	                + (" = ..." if has_default(arg) else "")
   797	            )
   798
   799	        kw_only = sorted(self.kwonly.values(), key=lambda a: (has_default(a), get_name(a)))
   800	        ret = "def ("
   801	        ret += ", ".join(
   802	            [get_desc(arg) for arg in self.pos]
   803	            + (["*" + get_name(self.varpos)] if self.varpos else (["*"] if self.kwonly else []))
   804	            + [get_desc(arg) for arg in kw_only]
   805	            + (["**" + get_name(self.varkw)] if self.varkw else [])
   806	        )
   807	        ret += ")"
   808	        return ret

   810	    @staticmethod
   811	    def from_funcitem(stub: nodes.FuncItem) -> Signature[nodes.Argument]:
   812	        stub_sig: Signature[nodes.Argument] = Signature()
   813	        stub_args = maybe_strip_cls(stub.name, stub.arguments)
   814	        for stub_arg in stub_args:
   815	            if stub_arg.kind.is_positional():
   816	                stub_sig.pos.append(stub_arg)
   817	            elif stub_arg.kind.is_named():
   818	                stub_sig.kwonly[stub_arg.variable.name] = stub_arg
   819	            elif stub_arg.kind == nodes.ARG_STAR:
   820	                stub_sig.varpos = stub_arg
   821	            elif stub_arg.kind == nodes.ARG_STAR2:
   822	                stub_sig.varkw = stub_arg
   823	            else:
   824	                raise AssertionError
   825	        return stub_sig

   827	    @staticmethod
   828	    def from_inspect_signature(signature: inspect.Signature) -> Signature[inspect.Parameter]:
   829	        runtime_sig: Signature[inspect.Parameter] = Signature()
   830	        for runtime_arg in signature.parameters.values():
   831	            if runtime_arg.kind in (
   832	                inspect.Parameter.POSITIONAL_ONLY,
   833	                inspect.Parameter.POSITIONAL_OR_KEYWORD,
   834	            ):
   835	                runtime_sig.pos.append(runtime_arg)
   836	            elif runtime_arg.kind == inspect.Parameter.KEYWORD_ONLY:
   837	                runtime_sig.kwonly[runtime_arg.name] = runtime_arg
   838	            elif runtime_arg.kind == inspect.Parameter.VAR_POSITIONAL:
   839	                runtime_sig.varpos = runtime_arg
   840	            elif runtime_arg.kind == inspect.Parameter.VAR_KEYWORD:
   841	                runtime_sig.varkw = runtime_arg
   842	            else:
   843	                raise AssertionError
   844	        return runtime_sig

   846	    @staticmethod
   847	    def from_overloadedfuncdef(stub: nodes.OverloadedFuncDef) -> Signature[nodes.Argument]:
   848	        """Returns a Signature from an OverloadedFuncDef.
   849
   850	        If life were simple, to verify_overloadedfuncdef, we'd just verify_funcitem for each of its
   851	        items. Unfortunately, life isn't simple and overloads are pretty deceitful. So instead, we
   852	        try and combine the overload's items into a single signature that is compatible with any
   853	        lies it might try to tell.
   854
   855	        """
   856	        # For most dunder methods, just assume all args are positional-only
   857	        assume_positional_only = is_dunder(stub.name, exclude_special=True)
   858
   859	        all_args: dict[str, list[tuple[nodes.Argument, int]]] = {}
   860	        for func in map(_resolve_funcitem_from_decorator, stub.items):
   861	            assert func is not None, "Failed to resolve decorated overload"
   862	            args = maybe_strip_cls(stub.name, func.arguments)
   863	            for index, arg in enumerate(args):
   864	                # For positional-only args, we allow overloads to have different names for the same
   865	                # argument. To accomplish this, we just make up a fake index-based name.
   866	                name = (
   867	                    f"__{index}"
   868	                    if arg.variable.name.startswith("__")
   869	                    or arg.pos_only
   870	                    or assume_positional_only
   871	                    or arg.variable.name.strip("_") == "self"
   872	                    else arg.variable.name
   873	                )
   874	                all_args.setdefault(name, []).append((arg, index))
   875
   876	        def get_position(arg_name: str) -> int:
   877	            # We just need this to return the positional args in the correct order.
   878	            return max(index for _, index in all_args[arg_name])
   879
   880	        def get_type(arg_name: str) -> mypy.types.ProperType:
   881	            with mypy.state.state.strict_optional_set(True):
   882	                all_types = [
   883	                    arg.variable.type or arg.type_annotation for arg, _ in all_args[arg_name]
   884	                ]
   885	                return mypy.typeops.make_simplified_union([t for t in all_types if t])
   886
   887	        def get_kind(arg_name: str) -> nodes.ArgKind:
   888	            kinds = {arg.kind for arg, _ in all_args[arg_name]}
   889	            if nodes.ARG_STAR in kinds:
   890	                return nodes.ARG_STAR
   891	            if nodes.ARG_STAR2 in kinds:
   892	                return nodes.ARG_STAR2
   893	            # The logic here is based on two tenets:
   894	            # 1) If an arg is ever optional (or unspecified), it is optional
   895	            # 2) If an arg is ever positional, it is positional
   896	            is_opt = (
   897	                len(all_args[arg_name]) < len(stub.items)
   898	                or nodes.ARG_OPT in kinds
   899	                or nodes.ARG_NAMED_OPT in kinds
   900	            )
   901	            is_pos = nodes.ARG_OPT in kinds or nodes.ARG_POS in kinds
   902	            if is_opt:
   903	                return nodes.ARG_OPT if is_pos else nodes.ARG_NAMED_OPT
   904	            return nodes.ARG_POS if is_pos else nodes.ARG_NAMED
   905
   906	        sig: Signature[nodes.Argument] = Signature()
   907	        for arg_name in sorted(all_args, key=get_position):
   908	            # example_arg_name gives us a real name (in case we had a fake index-based name)
   909	            example_arg_name = all_args[arg_name][0][0].variable.name
   910	            arg = nodes.Argument(
   911	                nodes.Var(example_arg_name, get_type(arg_name)),
   912	                type_annotation=None,
   913	                initializer=None,
   914	                kind=get_kind(arg_name),
   915	                pos_only=all(arg.pos_only for arg, _ in all_args[arg_name]),
   916	            )
   917	            if arg.kind.is_positional():
   918	                sig.pos.append(arg)
   919	            elif arg.kind.is_named():
   920	                sig.kwonly[arg.variable.name] = arg
   921	            elif arg.kind == nodes.ARG_STAR:
   922	                sig.varpos = arg
   923	            elif arg.kind == nodes.ARG_STAR2:
   924	                sig.varkw = arg
   925	            else:
   926	                raise AssertionError
   927	        return sig

   930	def _verify_signature(
   931	    stub: Signature[nodes.Argument], runtime: Signature[inspect.Parameter], function_name: str
   932	) -> Iterator[str]:
   933	    # Check positional arguments match up
   934	    for stub_arg, runtime_arg in zip(stub.pos, runtime.pos):
   935	        yield from _verify_arg_name(stub_arg, runtime_arg, function_name)
   936	        yield from _verify_arg_default_value(stub_arg, runtime_arg)
   937	        if (
   938	            runtime_arg.kind == inspect.Parameter.POSITIONAL_ONLY
   939	            and not stub_arg.pos_only
   940	            and not stub_arg.variable.name.startswith("__")
   941	            and stub_arg.variable.name.strip("_") != "self"
   942	            and not is_dunder(function_name, exclude_special=True)  # noisy for dunder methods
   943	        ):
   944	            yield (
   945	                f'stub argument "{stub_arg.variable.name}" should be positional-only '
   946	                f'(add "/", e.g. "{runtime_arg.name}, /")'
   947	            )
   948	        if (
   949	            runtime_arg.kind != inspect.Parameter.POSITIONAL_ONLY
   950	            and (stub_arg.pos_only or stub_arg.variable.name.startswith("__"))
   951	            and not runtime_arg.name.startswith("__")
   952	            and stub_arg.variable.name.strip("_") != "self"
   953	            and not is_dunder(function_name, exclude_special=True)  # noisy for dunder methods
   954	        ):
   955	            yield (
   956	                f'stub argument "{stub_arg.variable.name}" should be positional or keyword '
   957	                '(remove "/")'
   958	            )
   959
   960	    # Check unmatched positional args
   961	    if len(stub.pos) > len(runtime.pos):
   962	        # There are cases where the stub exhaustively lists out the extra parameters the function
   963	        # would take through *args. Hence, a) if runtime accepts *args, we don't check whether the
   964	        # runtime has all of the stub's parameters, b) below, we don't enforce that the stub takes
   965	        # *args, since runtime logic may prevent arbitrary arguments from actually being accepted.
   966	        if runtime.varpos is None:
   967	            for stub_arg in stub.pos[len(runtime.pos) :]:
   968	                # If the variable is in runtime.kwonly, it's just mislabelled as not a
   969	                # keyword-only argument
   970	                if stub_arg.variable.name not in runtime.kwonly:
   971	                    msg = f'runtime does not have argument "{stub_arg.variable.name}"'
   972	                    if runtime.varkw is not None:
   973	                        msg += ". Maybe you forgot to make it keyword-only in the stub?"
   974	                    yield msg
   975	                else:
   976	                    yield f'stub argument "{stub_arg.variable.name}" is not keyword-only'
   977	            if stub.varpos is not None:
   978	                yield f'runtime does not have *args argument "{stub.varpos.variable.name}"'
   979	    elif len(stub.pos) < len(runtime.pos):
   980	        for runtime_arg in runtime.pos[len(stub.pos) :]:
   981	            if runtime_arg.name not in stub.kwonly:
   982	                if not _is_private_parameter(runtime_arg):
   983	                    yield f'stub does not have argument "{runtime_arg.name}"'
   984	            else:
   985	                yield f'runtime argument "{runtime_arg.name}" is not keyword-only'
   986
   987	    # Checks involving *args
   988	    if len(stub.pos) <= len(runtime.pos) or runtime.varpos is None:
   989	        if stub.varpos is None and runtime.varpos is not None:
   990	            yield f'stub does not have *args argument "{runtime.varpos.name}"'
   991	        if stub.varpos is not None and runtime.varpos is None:
   992	            yield f'runtime does not have *args argument "{stub.varpos.variable.name}"'
   993
   994	    # Check keyword-only args
   995	    for arg in sorted(set(stub.kwonly) & set(runtime.kwonly)):
   996	        stub_arg, runtime_arg = stub.kwonly[arg], runtime.kwonly[arg]
   997	        yield from _verify_arg_name(stub_arg, runtime_arg, function_name)
   998	        yield from _verify_arg_default_value(stub_arg, runtime_arg)
   999
  1000	    # Check unmatched keyword-only args
  1001	    if runtime.varkw is None or not set(runtime.kwonly).issubset(set(stub.kwonly)):
  1002	        # There are cases where the stub exhaustively lists out the extra parameters the function
  1003	        # would take through **kwargs. Hence, a) if runtime accepts **kwargs (and the stub hasn't
  1004	        # exhaustively listed out params), we don't check whether the runtime has all of the stub's
  1005	        # parameters, b) below, we don't enforce that the stub takes **kwargs, since runtime logic
  1006	        # may prevent arbitrary keyword arguments from actually being accepted.
  1007	        for arg in sorted(set(stub.kwonly) - set(runtime.kwonly)):
  1008	            if arg in {runtime_arg.name for runtime_arg in runtime.pos}:
  1009	                # Don't report this if we've reported it before
  1010	                if arg not in {runtime_arg.name for runtime_arg in runtime.pos[len(stub.pos) :]}:
  1011	                    yield f'runtime argument "{arg}" is not keyword-only'
  1012	            else:
  1013	                yield f'runtime does not have argument "{arg}"'
  1014	    for arg in sorted(set(runtime.kwonly) - set(stub.kwonly)):
  1015	        if arg in {stub_arg.variable.name for stub_arg in stub.pos}:
  1016	            # Don't report this if we've reported it before
  1017	            if not (
  1018	                runtime.varpos is None
  1019	                and arg in {stub_arg.variable.name for stub_arg in stub.pos[len(runtime.pos) :]}
  1020	            ):
  1021	                yield f'stub argument "{arg}" is not keyword-only'
  1022	        else:
  1023	            if not _is_private_parameter(runtime.kwonly[arg]):
  1024	                yield f'stub does not have argument "{arg}"'
  1025
  1026	    # Checks involving **kwargs
  1027	    if stub.varkw is None and runtime.varkw is not None:
  1028	        # As mentioned above, don't enforce that the stub takes **kwargs.
  1029	        # Also check against positional parameters, to avoid a nitpicky message when an argument
  1030	        # isn't marked as keyword-only
  1031	        stub_pos_names = {stub_arg.variable.name for stub_arg in stub.pos}
  1032	        # Ideally we'd do a strict subset check, but in practice the errors from that aren't useful
  1033	        if not set(runtime.kwonly).issubset(set(stub.kwonly) | stub_pos_names):
  1034	            yield f'stub does not have **kwargs argument "{runtime.varkw.name}"'
  1035	    if stub.varkw is not None and runtime.varkw is None:
  1036	        yield f'runtime does not have **kwargs argument "{stub.varkw.variable.name}"'
  1037

  1047	@verify.register(nodes.FuncItem)
  1048	def verify_funcitem(
  1049	    stub: nodes.FuncItem, runtime: MaybeMissing[Any], object_path: list[str]
  1050	) -> Iterator[Error]:
  1051	    if isinstance(runtime, Missing):
  1052	        yield Error(object_path, "is not present at runtime", stub, runtime)
  1053	        return
  1054
  1055	    if not is_probably_a_function(runtime):
  1056	        yield Error(object_path, "is not a function", stub, runtime)
  1057	        if not callable(runtime):
  1058	            return
  1059
  1060	    # Look the object up statically, to avoid binding by the descriptor protocol
  1061	    static_runtime = _static_lookup_runtime(object_path)
  1062
  1063	    if isinstance(stub, nodes.FuncDef):
  1064	        for error_text in _verify_abstract_status(stub, runtime):
  1065	            yield Error(object_path, error_text, stub, runtime)
  1066	        for error_text in _verify_final_method(stub, runtime, static_runtime):
  1067	            yield Error(object_path, error_text, stub, runtime)
  1068
  1069	    for message in _verify_static_class_methods(stub, runtime, static_runtime, object_path):
  1070	        yield Error(object_path, "is inconsistent, " + message, stub, runtime)
  1071
  1072	    signature = safe_inspect_signature(runtime)
  1073	    runtime_is_coroutine = inspect.iscoroutinefunction(runtime)
  1074
  1075	    if signature:
  1076	        stub_sig = Signature.from_funcitem(stub)
  1077	        runtime_sig = Signature.from_inspect_signature(signature)
  1078	        runtime_sig_desc = describe_runtime_callable(signature, is_async=runtime_is_coroutine)
  1079	        stub_desc = str(stub_sig)
  1080	    else:
  1081	        runtime_sig_desc, stub_desc = None, None
  1082
  1083	    # Don't raise an error if the stub is a coroutine, but the runtime isn't.
  1084	    # That results in false positives.
  1085	    # See https://github.com/python/typeshed/issues/7344
  1086	    if runtime_is_coroutine and not stub.is_coroutine:
  1087	        yield Error(
  1088	            object_path,
  1089	            'is an "async def" function at runtime, but not in the stub',
  1090	            stub,
  1091	            runtime,
  1092	            stub_desc=stub_desc,
  1093	            runtime_desc=runtime_sig_desc,
  1094	        )
  1095
  1096	    if not signature:
  1097	        return
  1098
  1099	    for message in _verify_signature(stub_sig, runtime_sig, function_name=stub.name):
  1100	        yield Error(
  1101	            object_path,
  1102	            "is inconsistent, " + message,
  1103	            stub,
  1104	            runtime,
  1105	            runtime_desc=runtime_sig_desc,
  1106	        )

  1215	            )

  1218	            "is inconsistent, " + message,
  1219	            stub,

  1321	def _resolve_funcitem_from_decorator(dec: nodes.OverloadPart) -> nodes.FuncItem | None:
  1322	    """Returns a FuncItem that corresponds to the output of the decorator.
  1323
  1324	    Returns None if we can't figure out what that would be. For convenience, this function also
  1325	    accepts FuncItems.
  1326	    """
  1327	    if isinstance(dec, nodes.FuncItem):
  1328	        return dec
  1329	    if dec.func.is_property:
  1330	        return None
  1331
  1332	    def apply_decorator_to_funcitem(
  1333	        decorator: nodes.Expression, func: nodes.FuncItem
  1334	    ) -> nodes.FuncItem | None:
  1335	        if (
  1336	            isinstance(decorator, nodes.CallExpr)
  1337	            and isinstance(decorator.callee, nodes.RefExpr)
  1338	            and decorator.callee.fullname in mypy.types.DEPRECATED_TYPE_NAMES
  1339	        ):
  1340	            return func
  1341	        if not isinstance(decorator, nodes.RefExpr):
  1342	            return None
  1343	        if not decorator.fullname:
  1344	            # Happens with namedtuple
  1345	            return None
  1346	        if (
  1347	            decorator.fullname in ("builtins.staticmethod", "abc.abstractmethod")
  1348	            or decorator.fullname in mypy.types.OVERLOAD_NAMES
  1349	            or decorator.fullname in mypy.types.OVERRIDE_DECORATOR_NAMES
  1350	            or decorator.fullname in mypy.types.FINAL_DECORATOR_NAMES
  1351	        ):
  1352	            return func
  1353	        if decorator.fullname == "builtins.classmethod":
  1354	            if func.arguments[0].variable.name not in ("cls", "mcs", "metacls"):
  1355	                raise StubtestFailure(
  1356	                    f"unexpected class argument name {func.arguments[0].variable.name!r} "
  1357	                    f"in {dec.fullname}"
  1358	                )
  1359	            # FuncItem is written so that copy.copy() actually works, even when compiled
  1360	            ret = copy.copy(func)
  1361	            # Remove the cls argument, since it's not present in inspect.signature of classmethods
  1362	            ret.arguments = ret.arguments[1:]
  1363	            return ret
  1364	        # Just give up on any other decorators. After excluding properties, we don't run into
  1365	        # anything else when running on typeshed's stdlib.
  1366	        return None
  1367
  1368	    func: nodes.FuncItem = dec.func
  1369	    for decorator in dec.original_decorators:
  1370	        resulting_func = apply_decorator_to_funcitem(decorator, func)
  1371	        if resulting_func is None:
  1372	            return None
  1373	        func = resulting_func
  1374	    return func

  1411	    stub: nodes.TypeAlias, runtime: MaybeMissing[Any], object_path: list[str]
  1412	) -> Iterator[Error]:
  1413	    stub_target = mypy.types.get_proper_type(stub.target)
  1414	    stub_desc = f"Type alias for {stub_target}"
  1415	    if isinstance(runtime, Missing):

  1417	        return
  1418	    runtime_origin = get_origin(runtime) or runtime
  1419	    if isinstance(stub_target, mypy.types.Instance):
  1420	        if not isinstance(runtime_origin, type):
  1421	            yield Error(
  1422	                object_path,

  1424	                stub,
  1425	                runtime,
  1426	                stub_desc=stub_desc,
  1427	            )
  1428	            return
  1429
  1430	        stub_origin = stub_target.type
  1431	        # Do our best to figure out the fullname of the runtime object...
  1432	        runtime_name: object
  1433	        try:
  1434	            runtime_name = runtime_origin.__qualname__
  1435	        except AttributeError:
  1436	            runtime_name = getattr(runtime_origin, "__name__", MISSING)
  1437	        if isinstance(runtime_name, str):
  1438	            runtime_module: object = getattr(runtime_origin, "__module__", MISSING)
  1439	            if isinstance(runtime_module, str):
  1440	                if runtime_module == "collections.abc" or (
  1441	                    runtime_module == "re" and runtime_name in {"Match", "Pattern"}
  1442	                ):
  1443	                    runtime_module = "typing"
  1444	                runtime_fullname = f"{runtime_module}.{runtime_name}"
  1445	                if re.fullmatch(rf"_?{re.escape(stub_origin.fullname)}", runtime_fullname):
  1446	                    # Okay, we're probably fine.
  1447	                    return
  1448
  1449	        # Okay, either we couldn't construct a fullname
  1450	        # or the fullname of the stub didn't match the fullname of the runtime.
  1451	        # Fallback to a full structural check of the runtime vis-a-vis the stub.
  1452	        yield from verify(stub_origin, runtime_origin, object_path)
  1453	        return
  1454	    if isinstance(stub_target, mypy.types.UnionType):
  1455	        # complain if runtime is not a Union or UnionType
  1456	        if runtime_origin is not Union and (

  1465	                object_path, "is not a subclass of tuple", stub, runtime, stub_desc=stub_desc

  1467	        # could check Tuple contents here...

  1471	            yield Error(
  1472	                object_path, "is not a type alias for Callable", stub, runtime, stub_desc=stub_desc
  1473	            )
  1474	        # could check Callable contents here...
  1475	        return
  1476	    if isinstance(stub_target, mypy.types.AnyType):

  1595	def safe_inspect_signature(runtime: Any) -> inspect.Signature | None:
  1596	    try:
  1597	        try:
  1598	            return inspect.signature(runtime)
  1599	        except ValueError:
  1600	            if (
  1601	                hasattr(runtime, "__text_signature__")
  1602	                and "<unrepresentable>" in runtime.__text_signature__
  1603	            ):
  1604	                # Try to fix up the signature. Workaround for
  1605	                # https://github.com/python/cpython/issues/87233
  1606	                sig = runtime.__text_signature__.replace("<unrepresentable>", "...")
  1607	                sig = inspect._signature_fromstr(inspect.Signature, runtime, sig)  # type: ignore[attr-defined]
  1608	                assert isinstance(sig, inspect.Signature)
  1609	                new_params = [
  1610	                    (
  1611	                        parameter.replace(default=UNREPRESENTABLE)
  1612	                        if parameter.default is ...
  1613	                        else parameter
  1614	                    )
  1615	                    for parameter in sig.parameters.values()
  1616	                ]
  1617	                return sig.replace(parameters=new_params)
  1618	            else:
  1619	                raise
  1620	    except Exception:
  1621	        # inspect.signature throws ValueError all the time
  1622	        # catch RuntimeError because of https://bugs.python.org/issue39504
  1623	        # catch TypeError because of https://github.com/python/typeshed/pull/5762
  1624	        # catch AttributeError because of inspect.signature(_curses.window.border)
  1625	        return None
