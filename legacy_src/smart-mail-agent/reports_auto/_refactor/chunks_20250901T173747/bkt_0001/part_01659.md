  1755	            and value.type.class_ir.is_ext_class
  1756	            and value.type.class_ir.has_method("__bool__")
  1757	        ):
  1758	            # Directly call the __bool__ method on classes that have it.
  1759	            result = self.gen_method_call(value, "__bool__", [], bool_rprimitive, value.line)
  1760	        elif is_float_rprimitive(value.type):
  1761	            result = self.compare_floats(value, Float(0.0), FloatComparisonOp.NEQ, value.line)
  1762	        else:
  1763	            value_type = optional_value_type(value.type)
  1764	            if value_type is not None:
  1765	                not_none = self.translate_is_op(value, self.none_object(), "is not", value.line)
  1766	                always_truthy = False
  1767	                if isinstance(value_type, RInstance):
  1768	                    # check whether X.__bool__ is always just the default (object.__bool__)
  1769	                    if not value_type.class_ir.has_method(
  1770	                        "__bool__"
  1771	                    ) and value_type.class_ir.is_method_final("__bool__"):
  1772	                        always_truthy = True

  1775	                    result = not_none

  1782	                    self.activate_block(true)

  1784	                    # type to change even if it is a subtype.
  1785	                    remaining = self.unbox_or_cast(value, value_type, value.line)
  1786	                    as_bool = self.bool_value(remaining)

  1788	                    self.goto(end)
  1789	                    self.activate_block(false)

  1792	                    self.activate_block(end)
  1793	            else:

  1824	    def call_c(
  1825	        self,
  1826	        desc: CFunctionDescription,
  1827	        args: list[Value],
  1828	        line: int,
  1829	        result_type: RType | None = None,
  1830	    ) -> Value:
  1831	        """Call function using C/native calling convention (not a Python callable)."""
  1832	        # Handle void function via singleton RVoid instance
  1833	        coerced = []
  1834	        # Coerce fixed number arguments
  1835	        for i in range(min(len(args), len(desc.arg_types))):
  1836	            formal_type = desc.arg_types[i]
  1837	            arg = args[i]
  1838	            arg = self.coerce(arg, formal_type, line)
  1839	            coerced.append(arg)
  1840	        # Reorder args if necessary
  1841	        if desc.ordering is not None:
  1842	            assert desc.var_arg_type is None
  1843	            coerced = [coerced[i] for i in desc.ordering]
  1844	        # Coerce any var_arg
  1845	        var_arg_idx = -1
  1846	        if desc.var_arg_type is not None:
  1847	            var_arg_idx = len(desc.arg_types)
  1848	            for i in range(len(desc.arg_types), len(args)):
  1849	                arg = args[i]
  1850	                arg = self.coerce(arg, desc.var_arg_type, line)
  1851	                coerced.append(arg)
  1852	        # Add extra integer constant if any
  1853	        for item in desc.extra_int_constants:
  1854	            val, typ = item
  1855	            extra_int_constant = Integer(val, typ, line)
  1856	            coerced.append(extra_int_constant)
  1857	        error_kind = desc.error_kind
  1858	        if error_kind == ERR_NEG_INT:
  1859	            # Handled with an explicit comparison
  1860	            error_kind = ERR_NEVER
  1861	        target = self.add(
  1862	            CallC(
  1863	                desc.c_function_name,
  1864	                coerced,
  1865	                desc.return_type,
  1866	                desc.steals,
  1867	                desc.is_borrowed,
  1868	                error_kind,
  1869	                line,
  1870	                var_arg_idx,
  1871	                is_pure=desc.is_pure,
  1872	            )
  1873	        )
  1874	        if desc.is_borrowed:
  1875	            # If the result is borrowed, force the arguments to be
  1876	            # kept alive afterwards, as otherwise the result might be
  1877	            # immediately freed, at the risk of a dangling pointer.
  1878	            for arg in coerced:
  1879	                if not isinstance(arg, (Integer, LoadLiteral)):
  1880	                    self.keep_alives.append(arg)
  1881	        if desc.error_kind == ERR_NEG_INT:
  1882	            comp = ComparisonOp(target, Integer(0, desc.return_type, line), ComparisonOp.SGE, line)
  1883	            comp.error_kind = ERR_FALSE
  1884	            self.add(comp)
  1885
  1886	        if desc.truncated_type is None:
  1887	            result = target
  1888	        else:
  1889	            truncate = self.add(Truncate(target, desc.truncated_type))
  1890	            result = truncate
  1891	        if result_type and not is_runtime_subtype(result.type, result_type):
  1892	            if is_none_rprimitive(result_type):
  1893	                # Special case None return. The actual result may actually be a bool
  1894	                # and so we can't just coerce it.
  1895	                result = self.none()
  1896	            else:
  1897	                result = self.coerce(target, result_type, line, can_borrow=desc.is_borrowed)
  1898	        return result

   201	            # being visited to note where it was first visited.

   205	        # Check if child is nested within fdef (possibly indirectly
   206	        # within multiple nested functions).

   209	        parent = self.nested_funcs[child]
   210	        return parent == fitem or self.is_parent(fitem, parent)

   213	        # Find the function where the symbol was (likely) first declared,
   214	        # and mark is as a non-local symbol within that function.
   215	        func = self.symbols_to_funcs[symbol]

   260	def prepare_class_def(
   261	    path: str,
   262	    module_name: str,
   263	    cdef: ClassDef,
   264	    errors: Errors,
   265	    mapper: Mapper,
   266	    options: CompilerOptions,
   267	) -> None:
   268	    """Populate the interface-level information in a class IR.
   269
   270	    This includes attribute and method declarations, and the MRO, among other things, but
   271	    method bodies are generated in a later pass.
   272	    """
   273
   274	    ir = mapper.type_to_ir[cdef.info]
   275	    info = cdef.info
   276
   277	    attrs = get_mypyc_attrs(cdef)
   278	    if attrs.get("allow_interpreted_subclasses") is True:
   279	        ir.allow_interpreted_subclasses = True
   280	    if attrs.get("serializable") is True:
   281	        # Supports copy.copy and pickle (including subclasses)
   282	        ir._serializable = True
   283
   284	    # Check for subclassing from builtin types
   285	    for cls in info.mro:
   286	        # Special case exceptions and dicts
   287	        # XXX: How do we handle *other* things??
   288	        if cls.fullname == "builtins.BaseException":
   289	            ir.builtin_base = "PyBaseExceptionObject"
   290	        elif cls.fullname == "builtins.dict":
   291	            ir.builtin_base = "PyDictObject"
   292	        elif cls.fullname.startswith("builtins."):
   293	            if not can_subclass_builtin(cls.fullname):
   294	                # Note that if we try to subclass a C extension class that
   295	                # isn't in builtins, bad things will happen and we won't
   296	                # catch it here! But this should catch a lot of the most
   297	                # common pitfalls.
   298	                errors.error(
   299	                    "Inheriting from most builtin types is unimplemented", path, cdef.line
   300	                )
   301	                errors.note(
   302	                    "Potential workaround: @mypy_extensions.mypyc_attr(native_class=False)",
   303	                    path,
   304	                    cdef.line,
   305	                )
   306	                errors.note(
   307	                    "https://mypyc.readthedocs.io/en/stable/native_classes.html#defining-non-native-classes",
   308	                    path,
   309	                    cdef.line,
   310	                )
   311
   312	    # Set up the parent class
   313	    bases = [mapper.type_to_ir[base.type] for base in info.bases if base.type in mapper.type_to_ir]
   314	    if len(bases) > 1 and any(not c.is_trait for c in bases) and bases[0].is_trait:
   315	        # If the first base is a non-trait, don't ever error here. While it is correct
   316	        # to error if a trait comes before the next non-trait base (e.g. non-trait, trait,
   317	        # non-trait), it's pointless, confusing noise from the bigger issue: multiple
   318	        # inheritance is *not* supported.
   319	        errors.error("Non-trait base must appear first in parent list", path, cdef.line)
   320	    ir.traits = [c for c in bases if c.is_trait]
   321
   322	    mro = []  # All mypyc base classes
   323	    base_mro = []  # Non-trait mypyc base classes
   324	    for cls in info.mro:
   325	        if cls not in mapper.type_to_ir:
   326	            if cls.fullname != "builtins.object":
   327	                ir.inherits_python = True
   328	            continue
   329	        base_ir = mapper.type_to_ir[cls]
   330	        if not base_ir.is_trait:
   331	            base_mro.append(base_ir)
   332	        mro.append(base_ir)
   333
   334	        if cls.defn.removed_base_type_exprs or not base_ir.is_ext_class:
   335	            ir.inherits_python = True
   336
   337	    base_idx = 1 if not ir.is_trait else 0
   338	    if len(base_mro) > base_idx:
   339	        ir.base = base_mro[base_idx]
   340	    ir.mro = mro
   341	    ir.base_mro = base_mro
   342
   343	    prepare_methods_and_attributes(cdef, ir, path, module_name, errors, mapper, options)
   344	    prepare_init_method(cdef, ir, module_name, mapper)
   345
   346	    for base in bases:
   347	        if base.children is not None:
   348	            base.children.append(ir)
   349
   350	    if is_dataclass(cdef):
   351	        ir.is_augmented = True
