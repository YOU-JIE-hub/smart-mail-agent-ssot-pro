    80	class Templite:
    81	    """A simple template renderer, for a nano-subset of Django syntax.
    82
    83	    Supported constructs are extended variable access::
    84
    85	        {{var.modifier.modifier|filter|filter}}
    86
    87	    loops::
    88
    89	        {% for var in list %}...{% endfor %}
    90
    91	    and ifs::
    92
    93	        {% if var %}...{% endif %}
    94
    95	    Comments are within curly-hash markers::
    96
    97	        {# This will be ignored #}
    98
    99	    Lines between `{% joined %}` and `{% endjoined %}` will have lines stripped
   100	    and joined.  Be careful, this could join words together!
   101
   102	    Any of these constructs can have a hyphen at the end (`-}}`, `-%}`, `-#}`),
   103	    which will collapse the white space following the tag.
   104
   105	    Construct a Templite with the template text, then use `render` against a
   106	    dictionary context to create a finished string::
   107
   108	        templite = Templite('''
   109	            <h1>Hello {{name|upper}}!</h1>
   110	            {% for topic in topics %}
   111	                <p>You are interested in {{topic}}.</p>
   112	            {% endif %}
   113	            ''',
   114	            {"upper": str.upper},
   115	        )
   116	        text = templite.render({
   117	            "name": "Ned",
   118	            "topics": ["Python", "Geometry", "Juggling"],
   119	        })
   120
   121	    """
   122	    def __init__(self, text: str, *contexts: dict[str, Any]) -> None:
   123	        """Construct a Templite with the given `text`.
   124
   125	        `contexts` are dictionaries of values to use for future renderings.
   126	        These are good for filters and global values.
   127
   128	        """
   129	        self.context = {}
   130	        for context in contexts:
   131	            self.context.update(context)
   132
   133	        self.all_vars: set[str] = set()
   134	        self.loop_vars: set[str] = set()
   135
   136	        # We construct a function in source form, then compile it and hold onto
   137	        # it, and execute it to render the template.
   138	        code = CodeBuilder()
   139
   140	        code.add_line("def render_function(context, do_dots):")
   141	        code.indent()
   142	        vars_code = code.add_section()
   143	        code.add_line("result = []")
   144	        code.add_line("append_result = result.append")
   145	        code.add_line("extend_result = result.extend")
   146	        code.add_line("to_str = str")
   147
   148	        buffered: list[str] = []
   149
   150	        def flush_output() -> None:
   151	            """Force `buffered` to the code builder."""
   152	            if len(buffered) == 1:
   153	                code.add_line("append_result(%s)" % buffered[0])
   154	            elif len(buffered) > 1:
   155	                code.add_line("extend_result([%s])" % ", ".join(buffered))
   156	            del buffered[:]
   157
   158	        ops_stack = []
   159
   160	        # Split the text to form a list of tokens.
   161	        tokens = re.split(r"(?s)({{.*?}}|{%.*?%}|{#.*?#})", text)
   162
   163	        squash = in_joined = False
   164
   165	        for token in tokens:
   166	            if token.startswith("{"):
   167	                start, end = 2, -2
   168	                squash = (token[-3] == "-")
   169	                if squash:
   170	                    end = -3
   171
   172	                if token.startswith("{#"):
   173	                    # Comment: ignore it and move on.
   174	                    continue
   175	                elif token.startswith("{{"):
   176	                    # An expression to evaluate.
   177	                    expr = self._expr_code(token[start:end].strip())
   178	                    buffered.append("to_str(%s)" % expr)
   179	                else:
   180	                    # token.startswith("{%")
   181	                    # Action tag: split into words and parse further.
   182	                    flush_output()
   183
   184	                    words = token[start:end].strip().split()
   185	                    if words[0] == "if":

   187	                        if len(words) != 2:

   194	                        if len(words) != 4 or words[2] != "in":
   195	                            self._syntax_error("Don't understand for", token)
   196	                        ops_stack.append("for")
   197	                        self._variable(words[1], self.loop_vars)
   198	                        code.add_line(
   199	                            f"for c_{words[1]} in {self._expr_code(words[3])}:",
   200	                        )
   201	                        code.indent()
   202	                    elif words[0] == "joined":
   203	                        ops_stack.append("joined")
   204	                        in_joined = True
   205	                    elif words[0].startswith("end"):
   206	                        # Endsomething.  Pop the ops stack.
   207	                        if len(words) != 1:
   208	                            self._syntax_error("Don't understand end", token)
   209	                        end_what = words[0][3:]
   210	                        if not ops_stack:
   211	                            self._syntax_error("Too many ends", token)
   212	                        start_what = ops_stack.pop()
   213	                        if start_what != end_what:
   214	                            self._syntax_error("Mismatched end tag", end_what)
   215	                        if end_what == "joined":
   216	                            in_joined = False
   217	                        else:
   218	                            code.dedent()
   219	                    else:
   220	                        self._syntax_error("Don't understand tag", words[0])
   221	            else:
   222	                # Literal content.  If it isn't empty, output it.
   223	                if in_joined:

   225	                elif squash:
   226	                    token = token.lstrip()
   227	                if token:
   228	                    buffered.append(repr(token))

   231	            self._syntax_error("Unmatched action tag", ops_stack[-1])

   236	            vars_code.add_line(f"c_{var_name} = context[{var_name!r}]")

   238	        code.add_line("return ''.join(result)")
   239	        code.dedent()

   241	            Callable[
   242	                [dict[str, Any], Callable[..., Any]],
   243	                str,
   244	            ],
   245	            code.get_globals()["render_function"],
   246	        )

   250	        if "|" in expr:

   252	            code = self._expr_code(pipes[0])
   253	            for func in pipes[1:]:
   254	                self._variable(func, self.all_vars)

    79	        """Generate a Cobertura-compatible XML report for `morfs`.
    80
    81	        `morfs` is a list of modules or file names.

    83	        `outfile` is a file object to write the XML to.

    85	        """
    86	        # Initial setup.
    87	        outfile = outfile or sys.stdout
    88	        has_arcs = self.coverage.get_data().has_arcs()

    90	        # Create the DOM that will store the data.

    95	        # Write header stuff.

   105	        # Call xml_file for each file in the data.

   107	            self.xml_file(fr, analysis, has_arcs)

   109	        xsources = self.xml_out.createElement("sources")
   110	        xcoverage.appendChild(xsources)
   111
   112	        # Populate the XML DOM with the source info.
   113	        for path in human_sorted(self.source_paths):
   114	            xsource = self.xml_out.createElement("source")
   115	            appendChild(xsources, xsource)
   116	            txt = self.xml_out.createTextNode(path)
   117	            appendChild(xsource, txt)
   118
   119	        lnum_tot, lhits_tot = 0, 0
   120	        bnum_tot, bhits_tot = 0, 0
   121
   122	        xpackages = self.xml_out.createElement("packages")
   123	        xcoverage.appendChild(xpackages)
   124
   125	        # Populate the XML DOM with the package info.
   126	        for pkg_name, pkg_data in human_sorted_items(self.packages.items()):
   127	            xpackage = self.xml_out.createElement("package")
   128	            appendChild(xpackages, xpackage)
   129	            xclasses = self.xml_out.createElement("classes")
   130	            appendChild(xpackage, xclasses)
   131	            for _, class_elt in human_sorted_items(pkg_data.elements.items()):
   132	                appendChild(xclasses, class_elt)
   133	            xpackage.setAttribute("name", pkg_name.replace(os.sep, "."))
   134	            xpackage.setAttribute("line-rate", rate(pkg_data.hits, pkg_data.lines))
   135	            if has_arcs:
   136	                branch_rate = rate(pkg_data.br_hits, pkg_data.branches)
   137	            else:
   138	                branch_rate = "0"
   139	            xpackage.setAttribute("branch-rate", branch_rate)
   140	            xpackage.setAttribute("complexity", "0")

   142	            lhits_tot += pkg_data.hits
   143	            lnum_tot += pkg_data.lines
   144	            bhits_tot += pkg_data.br_hits
   145	            bnum_tot += pkg_data.branches
   146
   147	        xcoverage.setAttribute("lines-valid", str(lnum_tot))
   148	        xcoverage.setAttribute("lines-covered", str(lhits_tot))
   149	        xcoverage.setAttribute("line-rate", rate(lhits_tot, lnum_tot))
   150	        if has_arcs:
   151	            xcoverage.setAttribute("branches-valid", str(bnum_tot))
   152	            xcoverage.setAttribute("branches-covered", str(bhits_tot))
   153	            xcoverage.setAttribute("branch-rate", rate(bhits_tot, bnum_tot))
   154	        else:
   155	            xcoverage.setAttribute("branches-covered", "0")

   161	        outfile.write(serialize_xml(self.xml_out))

   171	    def xml_file(self, fr: FileReporter, analysis: Analysis, has_arcs: bool) -> None:
   172	        """Add to the XML report for a single file."""
   173
   174	        if self.config.skip_empty:
   175	            if analysis.numbers.n_statements == 0:
   176	                return
   177
   178	        # Create the "lines" and "package" XML elements, which
   179	        # are populated later.  Note that a package == a directory.
   180	        filename = fr.filename.replace("\\", "/")
   181	        for source_path in self.source_paths:
   182	            if not self.config.relative_files:
   183	                source_path = files.canonical_filename(source_path)
   184	            if filename.startswith(source_path.replace("\\", "/") + "/"):
   185	                rel_name = filename[len(source_path)+1:]
   186	                break
   187	        else:
   188	            rel_name = fr.relative_filename().replace("\\", "/")
   189	            self.source_paths.add(fr.filename[:-len(rel_name)].rstrip(r"\/"))
   190
   191	        dirname = os.path.dirname(rel_name) or "."
   192	        dirname = "/".join(dirname.split("/")[:self.config.xml_package_depth])
   193	        package_name = dirname.replace("/", ".")
   194
   195	        package = self.packages.setdefault(package_name, PackageData({}, 0, 0, 0, 0))
   196
   197	        xclass: xml.dom.minidom.Element = self.xml_out.createElement("class")
   198
   199	        appendChild(xclass, self.xml_out.createElement("methods"))
   200
   201	        xlines = self.xml_out.createElement("lines")
   202	        appendChild(xclass, xlines)
   203
   204	        xclass.setAttribute("name", os.path.relpath(rel_name, dirname))
   205	        xclass.setAttribute("filename", rel_name.replace("\\", "/"))
   206	        xclass.setAttribute("complexity", "0")
   207
   208	        branch_stats = analysis.branch_stats()
   209	        missing_branch_arcs = analysis.missing_branch_arcs()
   210
   211	        # For each statement, create an XML "line" element.
   212	        for line in sorted(analysis.statements):
   213	            xline = self.xml_out.createElement("line")
   214	            xline.setAttribute("number", str(line))
   215
   216	            # Q: can we get info about the number of times a statement is
   217	            # executed?  If so, that should be recorded here.
   218	            xline.setAttribute("hits", str(int(line not in analysis.missing)))
   219
   220	            if has_arcs:
   221	                if line in branch_stats:
   222	                    total, taken = branch_stats[line]
   223	                    xline.setAttribute("branch", "true")
   224	                    xline.setAttribute(
   225	                        "condition-coverage",
   226	                        "%d%% (%d/%d)" % (100*taken//total, taken, total),
   227	                    )
   228	                if line in missing_branch_arcs:
   229	                    annlines = ["exit" if b < 0 else str(b) for b in missing_branch_arcs[line]]
   230	                    xline.setAttribute("missing-branches", ",".join(annlines))
   231	            appendChild(xlines, xline)
   232
   233	        class_lines = len(analysis.statements)
   234	        class_hits = class_lines - len(analysis.missing)
   235
   236	        if has_arcs:
   237	            class_branches = sum(t for t, k in branch_stats.values())
   238	            missing_branches = sum(t - k for t, k in branch_stats.values())
   239	            class_br_hits = class_branches - missing_branches
   240	        else:
   241	            class_branches = 0
   242	            class_br_hits = 0
   243
   244	        # Finalize the statistics that are collected in the XML DOM.
   245	        xclass.setAttribute("line-rate", rate(class_hits, class_lines))
   246	        if has_arcs:
   247	            branch_rate = rate(class_br_hits, class_branches)
   248	        else:
   249	            branch_rate = "0"
   250	        xclass.setAttribute("branch-rate", branch_rate)
   251
   252	        package.elements[rel_name] = xclass
   253	        package.hits += class_hits
   254	        package.lines += class_lines
   255	        package.br_hits += class_br_hits
   256	        package.branches += class_branches
