  2111	    def report_protocol_problems(
  2112	        self,
  2113	        subtype: Instance | TupleType | TypedDictType | TypeType | CallableType,
  2114	        supertype: Instance,
  2115	        context: Context,
  2116	        *,
  2117	        parent_error: ErrorInfo,
  2118	    ) -> None:
  2119	        """Report possible protocol conflicts between 'subtype' and 'supertype'.
  2120
  2121	        This includes missing members, incompatible types, and incompatible
  2122	        attribute flags, such as settable vs read-only or class variable vs
  2123	        instance variable.
  2124	        """
  2125	        OFFSET = 4  # Four spaces, so that notes will look like this:
  2126	        # note: 'Cls' is missing following 'Proto' members:
  2127	        # note:     method, attr
  2128	        MAX_ITEMS = 2  # Maximum number of conflicts, missing members, and overloads shown
  2129	        # List of special situations where we don't want to report additional problems
  2130	        exclusions: dict[type, list[str]] = {
  2131	            TypedDictType: ["typing.Mapping"],
  2132	            TupleType: ["typing.Iterable", "typing.Sequence"],
  2133	        }
  2134	        if supertype.type.fullname in exclusions.get(type(subtype), []):
  2135	            return
  2136	        if any(isinstance(tp, UninhabitedType) for tp in get_proper_types(supertype.args)):
  2137	            # We don't want to add notes for failed inference (e.g. Iterable[Never]).
  2138	            # This will be only confusing a user even more.
  2139	            return
  2140
  2141	        class_obj = False
  2142	        is_module = False
  2143	        skip = []
  2144	        if isinstance(subtype, TupleType):
  2145	            subtype = subtype.partial_fallback
  2146	        elif isinstance(subtype, TypedDictType):
  2147	            subtype = subtype.fallback
  2148	        elif isinstance(subtype, TypeType):
  2149	            if not isinstance(subtype.item, Instance):
  2150	                return
  2151	            class_obj = True
  2152	            subtype = subtype.item
  2153	        elif isinstance(subtype, CallableType):
  2154	            if subtype.is_type_obj():
  2155	                ret_type = get_proper_type(subtype.ret_type)
  2156	                if isinstance(ret_type, TupleType):
  2157	                    ret_type = ret_type.partial_fallback
  2158	                if not isinstance(ret_type, Instance):
  2159	                    return
  2160	                class_obj = True
  2161	                subtype = ret_type
  2162	            else:
  2163	                subtype = subtype.fallback
  2164	                skip = ["__call__"]
  2165	        if subtype.extra_attrs and subtype.extra_attrs.mod_name:
  2166	            is_module = True
  2167
  2168	        # Report missing members
  2169	        missing = get_missing_protocol_members(subtype, supertype, skip=skip)
  2170	        if (
  2171	            missing
  2172	            and (len(missing) < len(supertype.type.protocol_members) or missing == ["__call__"])
  2173	            and len(missing) <= MAX_ITEMS
  2174	        ):
  2175	            if missing == ["__call__"] and class_obj:
  2176	                self.note(
  2177	                    '"{}" has constructor incompatible with "__call__" of "{}"'.format(
  2178	                        subtype.type.name, supertype.type.name
  2179	                    ),
  2180	                    context,
  2181	                    parent_error=parent_error,
  2182	                )
  2183	            else:
  2184	                self.note(
  2185	                    '"{}" is missing following "{}" protocol member{}:'.format(
  2186	                        subtype.type.name, supertype.type.name, plural_s(missing)
  2187	                    ),
  2188	                    context,
  2189	                    parent_error=parent_error,
  2190	                )
  2191	                self.note(", ".join(missing), context, offset=OFFSET, parent_error=parent_error)
  2192	        elif len(missing) > MAX_ITEMS or len(missing) == len(supertype.type.protocol_members):
  2193	            # This is an obviously wrong type: too many missing members
  2194	            return
  2195
  2196	        # Report member type conflicts
  2197	        conflict_types = get_conflict_protocol_types(
  2198	            subtype, supertype, class_obj=class_obj, options=self.options
  2199	        )
  2200	        if conflict_types and (
  2201	            not is_subtype(subtype, erase_type(supertype), options=self.options)
  2202	            or not subtype.type.defn.type_vars
  2203	            or not supertype.type.defn.type_vars
  2204	            # Always show detailed message for ParamSpec
  2205	            or subtype.type.has_param_spec_type
  2206	            or supertype.type.has_param_spec_type
  2207	        ):
  2208	            type_name = format_type(subtype, self.options, module_names=True)
  2209	            self.note(
  2210	                f"Following member(s) of {type_name} have conflicts:",
  2211	                context,
  2212	                parent_error=parent_error,
  2213	            )
  2214	            for name, got, exp, is_lvalue in conflict_types[:MAX_ITEMS]:
  2215	                exp = get_proper_type(exp)
  2216	                got = get_proper_type(got)
  2217	                setter_suffix = " setter type" if is_lvalue else ""
  2218	                if (
  2219	                    not isinstance(exp, (CallableType, Overloaded))
  2220	                    or not isinstance(got, (CallableType, Overloaded))
  2221	                    # If expected type is a type object, it means it is a nested class.
  2222	                    # Showing constructor signature in errors would be confusing in this case,
  2223	                    # since we don't check the signature, only subclassing of type objects.
  2224	                    or exp.is_type_obj()
  2225	                ):
  2226	                    self.note(
  2227	                        "{}: expected{} {}, got {}".format(
  2228	                            name,
  2229	                            setter_suffix,
  2230	                            *format_type_distinctly(exp, got, options=self.options),
  2231	                        ),
  2232	                        context,
  2233	                        offset=OFFSET,
  2234	                        parent_error=parent_error,
  2235	                    )
  2236	                    if is_lvalue and is_subtype(got, exp, options=self.options):
  2237	                        self.note(
  2238	                            "Setter types should behave contravariantly",
  2239	                            context,
  2240	                            offset=OFFSET,
  2241	                            parent_error=parent_error,
  2242	                        )
  2243	                else:
  2244	                    self.note(
  2245	                        "Expected{}:".format(setter_suffix),
  2246	                        context,
  2247	                        offset=OFFSET,
  2248	                        parent_error=parent_error,
  2249	                    )
  2250	                    if isinstance(exp, CallableType):
  2251	                        self.note(
  2252	                            pretty_callable(exp, self.options, skip_self=class_obj or is_module),
  2253	                            context,
  2254	                            offset=2 * OFFSET,
  2255	                            parent_error=parent_error,
  2256	                        )
  2257	                    else:
  2258	                        assert isinstance(exp, Overloaded)
  2259	                        self.pretty_overload(
  2260	                            exp,
  2261	                            context,
  2262	                            2 * OFFSET,
  2263	                            parent_error=parent_error,
  2264	                            skip_self=class_obj or is_module,
  2265	                        )
  2266	                    self.note("Got:", context, offset=OFFSET, parent_error=parent_error)
  2267	                    if isinstance(got, CallableType):
  2268	                        self.note(
  2269	                            pretty_callable(got, self.options, skip_self=class_obj or is_module),
  2270	                            context,
  2271	                            offset=2 * OFFSET,
  2272	                            parent_error=parent_error,
  2273	                        )
  2274	                    else:
  2275	                        assert isinstance(got, Overloaded)
  2276	                        self.pretty_overload(
  2277	                            got,
  2278	                            context,
  2279	                            2 * OFFSET,
  2280	                            parent_error=parent_error,
  2281	                            skip_self=class_obj or is_module,
  2282	                        )
  2283	            self.print_more(conflict_types, context, OFFSET, MAX_ITEMS, code=parent_error.code)
  2284
  2285	        # Report flag conflicts (i.e. settable vs read-only etc.)
  2286	        conflict_flags = get_bad_protocol_flags(subtype, supertype, class_obj=class_obj)
  2287	        for name, subflags, superflags in conflict_flags[:MAX_ITEMS]:
  2288	            if not class_obj and IS_CLASSVAR in subflags and IS_CLASSVAR not in superflags:
  2289	                self.note(
  2290	                    "Protocol member {}.{} expected instance variable, got class variable".format(
  2291	                        supertype.type.name, name
  2292	                    ),
  2293	                    context,
  2294	                    parent_error=parent_error,
  2295	                )
  2296	            if not class_obj and IS_CLASSVAR in superflags and IS_CLASSVAR not in subflags:
  2297	                self.note(
  2298	                    "Protocol member {}.{} expected class variable, got instance variable".format(
  2299	                        supertype.type.name, name
  2300	                    ),
  2301	                    context,
  2302	                    parent_error=parent_error,
  2303	                )
  2304	            if IS_SETTABLE in superflags and IS_SETTABLE not in subflags:
  2305	                self.note(
  2306	                    "Protocol member {}.{} expected settable variable,"
  2307	                    " got read-only attribute".format(supertype.type.name, name),
  2308	                    context,
  2309	                    parent_error=parent_error,
  2310	                )
  2311	            if IS_CLASS_OR_STATIC in superflags and IS_CLASS_OR_STATIC not in subflags:
  2312	                self.note(
  2313	                    "Protocol member {}.{} expected class or static method".format(
  2314	                        supertype.type.name, name
  2315	                    ),
  2316	                    context,
  2317	                    parent_error=parent_error,
  2318	                )
  2319	            if (
  2320	                class_obj
  2321	                and IS_VAR in superflags
  2322	                and (IS_VAR in subflags and IS_CLASSVAR not in subflags)

  2551	def format_type_inner(
  2552	    typ: Type,
  2553	    verbosity: int,

  2555	    fullnames: set[str] | None,

  2557	) -> str:

  2560
  2561	    Args:

  2563	      fullnames: a set of names that should be printed in full

  2566	    def format(typ: Type) -> str:

  2571
  2572	    def format_union_items(types: Sequence[Type]) -> list[str]:

  2574	        if len(formatted) > MAX_UNION_ITEMS and verbosity == 0:
  2575	            more = len(formatted) - MAX_UNION_ITEMS // 2

  2577	        else:
  2578	            more = 0
  2579	        if more:
  2580	            formatted.append(f"<{more} more items>")
  2581	        if any(format(typ) == "None" for typ in types):
  2582	            formatted.append("None")
  2583	        return formatted

  2585	    def format_union(types: Sequence[Type]) -> str:
  2586	        return " | ".join(format_union_items(types))
  2587
  2588	    def format_literal_value(typ: LiteralType) -> str:
  2589	        if typ.is_enum_literal():

  2598	        else:

  2600	                type_str = typ.alias.fullname

  2602	                type_str = typ.alias.name

  2617	                return f'{base_str} "{itype.extra_attrs.mod_name}"'
  2618	            return base_str

  2631	        elif itype.type.fullname == "builtins.tuple":

  2633	            return f"tuple[{item_type_str}, ...]"

  2636	            return f"{base_str}[{format_list(itype.args)}]"
  2637	    elif isinstance(typ, UnpackType):

  2643	        fullname = scoped_type_var_name(typ)
  2644	        if verbosity >= 2 or (fullnames and fullname in fullnames):
  2645	            return fullname
  2646	        return typ.name
  2647	    elif isinstance(typ, TypeVarTupleType):
  2648	        # This is similar to non-generic instance types.
  2649	        fullname = scoped_type_var_name(typ)

  2652	        return typ.name
  2653	    elif isinstance(typ, ParamSpecType):

  2738	    elif isinstance(typ, UninhabitedType):

  2751	            elif func.type_is is not None:

   214	    def find_module_via_source_set(self, id: str) -> ModuleSearchResult | None:
   215	        """Fast path to find modules by looking through the input sources
   216
   217	        This is only used when --fast-module-lookup is passed on the command line."""
   218	        if not self.source_set:
   219	            return None
   220
   221	        p = self.source_set.source_modules.get(id, None)
   222	        if p and self.fscache.isfile(p):
   223	            # We need to make sure we still have __init__.py all the way up
   224	            # otherwise we might have false positives compared to slow path
   225	            # in case of deletion of init files, which is covered by some tests.
   226	            # TODO: are there some combination of flags in which this check should be skipped?
   227	            d = os.path.dirname(p)
   228	            for _ in range(id.count(".")):
   229	                if not any(
   230	                    self.fscache.isfile(os_path_join(d, "__init__" + x)) for x in PYTHON_EXTENSIONS
   231	                ):
   232	                    return None
   233	                d = os.path.dirname(d)
   234	            return p
   235
   236	        idx = id.rfind(".")
   237	        if idx != -1:
   238	            # When we're looking for foo.bar.baz and can't find a matching module
   239	            # in the source set, look up for a foo.bar module.
   240	            parent = self.find_module_via_source_set(id[:idx])
   241	            if parent is None or not isinstance(parent, str):
   242	                return None
   243
   244	            basename, ext = os.path.splitext(parent)
   245	            if not any(parent.endswith("__init__" + x) for x in PYTHON_EXTENSIONS) and (
   246	                ext in PYTHON_EXTENSIONS and not self.fscache.isdir(basename)
   247	            ):
   248	                # If we do find such a *module* (and crucially, we don't want a package,
   249	                # hence the filtering out of __init__ files, and checking for the presence
   250	                # of a folder with a matching name), then we can be pretty confident that
   251	                # 'baz' will either be a top-level variable in foo.bar, or will not exist.
   252	                #
   253	                # Either way, spelunking in other search paths for another 'foo.bar.baz'
   254	                # module should be avoided because:
   255	                #  1. in the unlikely event that one were found, it's highly likely that
   256	                #     it would be unrelated to the source being typechecked and therefore
   257	                #     more likely to lead to erroneous results
   258	                #  2. as described in _find_module, in some cases the search itself could
   259	                #  potentially waste significant amounts of time
   260	                return ModuleNotFoundReason.NOT_FOUND
   261	        return None

   263	    def find_lib_path_dirs(self, id: str, lib_path: tuple[str, ...]) -> PackageDirs:
   264	        """Find which elements of a lib_path have the directory a module needs to exist."""
   265	        components = id.split(".")
   266	        dir_chain = os.sep.join(components[:-1])  # e.g., 'foo/bar'
   267
   268	        dirs = []
   269	        for pathitem in self.get_toplevel_possibilities(lib_path, components[0]):
   270	            # e.g., '/usr/lib/python3.4/foo/bar'
   271	            if dir_chain:
   272	                dir = os_path_join(pathitem, dir_chain)
   273	            else:
   274	                dir = pathitem
   275	            if self.fscache.isdir(dir):
   276	                dirs.append((dir, True))
   277	        return dirs

   279	    def get_toplevel_possibilities(self, lib_path: tuple[str, ...], id: str) -> list[str]:
   280	        """Find which elements of lib_path could contain a particular top-level module.
   281
   282	        In practice, almost all modules can be routed to the correct entry in
   283	        lib_path by looking at just the first component of the module name.
   284
   285	        We take advantage of this by enumerating the contents of all of the
   286	        directories on the lib_path and building a map of which entries in
   287	        the lib_path could contain each potential top-level module that appears.
   288	        """
   289
   290	        if lib_path in self.initial_components:
   291	            return self.initial_components[lib_path].get(id, [])
   292
   293	        # Enumerate all the files in the directories on lib_path and produce the map
   294	        components: dict[str, list[str]] = {}
   295	        for dir in lib_path:
   296	            try:
   297	                contents = self.fscache.listdir(dir)
   298	            except OSError:
   299	                contents = []
   300	            # False positives are fine for correctness here, since we will check
   301	            # precisely later, so we only look at the root of every filename without
   302	            # any concern for the exact details.
   303	            for name in contents:
   304	                name = os.path.splitext(name)[0]
   305	                components.setdefault(name, []).append(dir)
   306
   307	        self.initial_components[lib_path] = components
   308	        return components.get(id, [])
