def parse_eval(p):
    txt = Path(p).read_text(encoding="utf-8")
    m_acc = re.search(r'Accuracy\s*=\s*([\d\.]+)', txt)
    m_mf1 = re.search(r'MacroF1\s*=\s*([\d\.]+)', txt)
    per = {}
    for lab in ["biz_quote","complaint","other","policy_qa","profile_update","tech_support"]:
        m = re.search(rf'^{lab}:\s*P=([\d\.]+)\s*R=([\d\.]+)\s*F1=([\d\.]+).*$', txt, re.M)
        if m: per[lab] = tuple(map(float, m.groups()))
    return (float(m_acc.group(1)) if m_acc else None,
            float(m_mf1.group(1)) if m_mf1 else None,
            per, txt)

def top_confusions(conf_path, k=6):
    rows=[]
    with open(conf_path, "r", encoding="utf-8") as f:
        labs=f.readline().strip().split("\t")[1:]
        mat=[list(map(int, ln.strip().split("\t")[1:])) for ln in f if ln.strip()]

        for j,lj in enumerate(labs):
            if i!=j and mat[i][j]>0:
                rows.append((mat[i][j], f"{li} → {lj}", mat[i][j]))
    rows.sort(reverse=True)
    return [f"{lab} ({cnt})" for _,lab,cnt in rows[:k]]

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
import argparse, json, sys
from pathlib import Path
import numpy as np

# ----------------- utils -----------------
INFO_TERMS = [
    "api docs","sdk","integration","roadmap","case study","deck","video",
    "API 文件","SDK","整合","路線圖","案例","簡報","影片"
]
FAIL_TERMS = [
    "error","fail","cannot","timeout","429","5xx","saml","ntp",
    "錯誤","失敗","無法","逾時","429","5xx","SAML","NTP"
]
def rules_guard(text:str)->bool:
    t = (text or "").lower()
    info_hit = any(w.lower() in t for w in INFO_TERMS)
    fail_hit = any(w.lower() in t for w in FAIL_TERMS)
    return (info_hit and not fail_hit)  # True → 擋掉 tech_support

def pick_text(o:dict)->str:
    # 優先 text / content / raw；否則 subject+body
    for k in ("text","content","raw"):
        if k in o and o[k]: return str(o[k])
    subj = o.get("subject","") or o.get("title","")
    body = o.get("body","") or o.get("message","")
    t = (subj+"\n"+body).strip()
    return t if t else json.dumps(o, ensure_ascii=False)

def pick_label(o:dict):
    for k in ("label","intent","y","category"):
        if k in o and o[k]: return str(o[k])
    return None

def softmax(x:np.ndarray, axis=-1):
    x = np.asarray(x, dtype=np.float64)
    x = x - np.max(x, axis=axis, keepdims=True)
    e = np.exp(x)
    s = e.sum(axis=axis, keepdims=True)
    s[s==0] = 1.0
    return e / s

# ----------------- main -----------------
def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--model", required=True, help="path to intent *.pkl (Pipeline)")
    ap.add_argument("--test",  required=True, help="jsonl test file")
    ap.add_argument("--threshold", type=float, default=0.52)
    ap.add_argument("--margin",    type=float, default=0.10)
    ap.add_argument("--only-tech", dest="only_tech", action="store_true")
    ap.add_argument("--only_tech", dest="only_tech", action="store_true")
    ap.add_argument("--rules-guard", action="store_true")
    ap.add_argument("--out_prefix", default="reports_auto/external_fallback")
    ap.add_argument("--scan", action="store_true")
    args = ap.parse_args()

    # load model (Pipeline)
    import pickle
    mdl = pickle.load(open(args.model, "rb"))

    # read jsonl
    rows = []
    labels_set = set()
    with open(args.test,"r",encoding="utf-8") as f:
        for ln in f:
            o = json.loads(ln)
            x = pick_text(o)
            y = pick_label(o)
            if y is not None: labels_set.add(y)
            rows.append((x,y,o))
    X = [x for x,_,_ in rows]
    Y = [y for _,y,_ in rows if y is not None]
    has_label = all(y is not None for _,y,_ in rows)

    # predict base
    y_hat = mdl.predict(X)
    classes = list(getattr(mdl, "classes_", getattr(mdl.named_steps.get("clf", None), "classes_", [])))

    # prob or decision scores
    if hasattr(mdl, "predict_proba"):
        P = mdl.predict_proba(X)
    else:
        # LinearSVC / 沒機率：用 decision_function → softmax 成機率感
        df = mdl.decision_function(X)
        if df.ndim == 1:  # binary 轉 2col
            df = np.vstack([-df, df]).T
        P = softmax(df, axis=1)
        if not classes:
            # 從最末分類器拿
            clf = getattr(mdl, "named_steps", {}).get("clf", None)
            classes = list(getattr(clf, "classes_", []))
    if not classes:
        raise SystemExit("[FATAL] 無法取得 classes_")

    # fallback
    TECH = "tech_support"
    y_hat_fb = []
    changed = 0
    for i,(x,raw_pred) in enumerate(zip(X, y_hat)):
        pi = P[i]
        order = np.argsort(pi)[::-1]
        p1, c1 = float(pi[order[0]]), classes[order[0]]
        p2, c2 = (float(pi[order[1]]), classes[order[1]]) if len(order)>1 else (0.0, None)
        pred = raw_pred

        # 規則先
        if args.rules_guard and pred == TECH and rules_guard(x):
            pred = "other"

        # 門檻/邊際
        only = args.only_tech
        if (not only) or (pred==TECH or c1==TECH):
            if p1 < args.threshold or (p1 - p2) < args.margin:
                if pred == TECH:
                    pred = "other"

        if pred != raw_pred: changed += 1
        y_hat_fb.append(pred)

    # 評估
    from collections import Counter
    def eval_simple(y_true, y_pred, labels):
        acc = np.mean([a==b for a,b in zip(y_true,y_pred)]) if y_true else 0.0
        cm = Counter(zip(y_true,y_pred))
        by = {lab:{"tp":0,"fp":0,"fn":0} for lab in labels}
        for (yt,yp),c in cm.items():
            if yt==yp: by[yt]["tp"] += c
            else:      by[yp]["fp"] += c; by[yt]["fn"] += c
        def prf(d):
            tp,fp,fn = d["tp"],d["fp"],d["fn"]
            prec = tp/(tp+fp) if (tp+fp)>0 else 0.0
            rec  = tp/(tp+fn) if (tp+fn)>0 else 0.0
            f1   = (2*prec*rec/(prec+rec)) if (prec+rec)>0 else 0.0
            return prec,rec,f1
        macro = np.mean([prf(by[lab])[2] for lab in labels]) if labels else 0.0
        return acc, by, cm, macro

    out = Path(args.out_prefix)
    out.parent.mkdir(parents=True, exist_ok=True)

    def dump_eval(suffix, y_pred):
        eval_txt = out.with_name(out.name + "_eval"+suffix+".txt")
        conf_tsv = out.with_name(out.name + "_confusion"+suffix+".tsv")
        errs_tsv = out.with_name(out.name + "_errors"+suffix+".tsv")
        if has_label:
            labels = sorted(labels_set)
            acc, by, cm, macro = eval_simple(Y, y_pred, labels)
            with open(eval_txt,"w",encoding="utf-8") as fo:
                fo.write(f"[BASE]\n" if suffix=="_base" else f"[FALLBACK] threshold={args.threshold} margin={args.margin} only_tech={args.only_tech} rules_guard={args.rules_guard}\n")
                fo.write(f"accuracy={acc:.3f}\n")
                fo.write(f"MacroF1={macro:.3f}\n")
                if suffix != "_base":
                    fo.write(f"changed={changed}\n")
            # confusion
            with open(conf_tsv,"w",encoding="utf-8") as fo:
                fo.write("y_true\ty_pred\tcount\n")
                for (yt,yp),c in sorted(cm.items()):
                    fo.write(f"{yt}\t{yp}\t{c}\n")
            # errors
            with open(errs_tsv,"w",encoding="utf-8") as fo:
                fo.write("y_true\ty_pred\tp1\tmargin\tsubject_or_head\ttext\n")
                for (x,y,o),pred,pi in zip(rows, y_pred, P):
                    if y is None or y==pred: continue
                    order=np.argsort(pi)[::-1]
                    p1 = float(pi[order[0]]) if len(order)>0 else 0.0
                    p2 = float(pi[order[1]]) if len(order)>1 else 0.0
                    subj = o.get("subject","") or o.get("title","")
                    san_subj = (subj or "").replace("\t"," ").replace("\n"," ")
                    san_text = pick_text(o).replace("\t"," ").replace("\n"," ")
                    fo.write(f"{y}\t{pred}\t{p1:.3f}\t{(p1-p2):.3f}\t{san_subj}\t{san_text}\n")
        else:
            with open(eval_txt,"w",encoding="utf-8") as fo:
                fo.write("NO_GROUND_TRUTH\n")
        return eval_txt, conf_tsv, errs_tsv

    base_eval = dump_eval("_base", y_hat)
    fb_eval   = dump_eval("",      y_hat_fb)

    # 可選掃描
    if args.scan and has_label:
        scan_file = out.with_name(out.name+"_scan.tsv")
        labels = sorted(labels_set)
        with open(scan_file,"w",encoding="utf-8") as fo:
            fo.write("T\tM\tAccuracy\tMacroF1\tchanged\n")
            for T in np.linspace(0.50,0.70,11):
                for M in (0.00,0.05,0.10,0.15,0.20):
                    y_tmp=[]; ch=0
                    for i,(x,raw_pred) in enumerate(zip(X,y_hat)):
                        pi = P[i]; order = np.argsort(pi)[::-1]
                        p1, c1 = float(pi[order[0]]), classes[order[0]]
                        p2     = float(pi[order[1]]) if len(order)>1 else 0.0
                        pred = raw_pred
                        if args.rules_guard and pred=="tech_support" and rules_guard(x):
                            pred="other"
                        if (not args.only_tech) or (pred=="tech_support" or c1=="tech_support"):
                            if p1<T or (p1-p2)<M:
                                if pred=="tech_support": pred="other"
                        if pred!=raw_pred: ch+=1
                        y_tmp.append(pred)
                    acc,_,_,macro = eval_simple(Y,y_tmp,labels)
                    fo.write(f"{T:.2f}\t{M:.2f}\t{acc:.3f}\t{macro:.3f}\t{ch}\n")

    print("[OUT]", base_eval, fb_eval)
```

```python
    raise SystemExit("[FATAL] model has no classes_")

def softmax(x, axis=1):
    x = x - np.max(x, axis=axis, keepdims=True)
    e = np.exp(x)
    return e / np.sum(e, axis=axis, keepdims=True)

def get_proba(pipe, X, n_labels):
    clf = None
    try: clf = pipe.named_steps.get("clf", pipe.steps[-1][1])
    except Exception: pass
    if clf is not None and hasattr(clf, "predict_proba"):
        # 如果有 feats，就只 transform 一次
        if hasattr(pipe,"named_steps") and "feats" in pipe.named_steps:
            F = pipe.named_steps["feats"].transform(X)
            return clf.predict_proba(F)
        return pipe.predict_proba(X)
    # 回退：decision_function -> softmax
    scores = pipe.decision_function(X)
    if scores.ndim == 1: scores = np.stack([-scores, scores], axis=1)
    return softmax(scores, axis=1)

def apply_fallback(probs, labels, p1_thr, margin, policy_lock):
    top = probs.argmax(axis=1)
    p_sorted = np.sort(probs, axis=1)[:, ::-1]
    p1 = p_sorted[:,0]
    p2 = p_sorted[:,1] if probs.shape[1] > 1 else np.zeros_like(p1)
    yp = top.copy()
    other_idx = labels.index("other") if "other" in labels else None
    if other_idx is None:
        return [labels[i] for i in yp]
    for i in range(len(yp)):
        if policy_lock and labels[yp[i]] == "policy_qa":
            continue
        if (p1[i] < p1_thr) or ((p1[i] - p2[i]) < margin):
            yp[i] = other_idx
    return [labels[i] for i in yp]

def evaluate(Y, YP, labels):
    lab2i = {lab:i for i,lab in enumerate(labels)}
    cm = np.zeros((len(labels), len(labels)), dtype=int)
    for yt, yp in zip(Y, YP):
        if yt in lab2i and yp in lab2i:
            cm[lab2i[yt], lab2i[yp]] += 1
    acc = np.trace(cm)/cm.sum() if cm.sum()>0 else 0.0
    out={}
    for i,lab in enumerate(labels):
        tp=cm[i,i]; fp=cm[:,i].sum()-tp; fn=cm[i,:].sum()-tp
        P=tp/(tp+fp) if (tp+fp)>0 else 0.0
        R=tp/(tp+fn) if (tp+fn)>0 else 0.0
        F=(2*P*R/(P+R)) if (P+R)>0 else 0.0
        out[lab]=(P,R,F,tp,fp,fn)
    macro=float(np.mean([v[2] for v in out.values()])) if out else 0.0
    return acc, macro, out, cm

    P = get_proba(pipe, X, len(labels))
    base = [labels[i] for i in P.argmax(axis=1)]
    fall = apply_fallback(P, labels, p1, margin, lock)

    # write preds
    Path(args.out_preds).parent.mkdir(parents=True, exist_ok=True)
    with open(args.out_preds, "w", encoding="utf-8") as f:
        for r,b,y in zip(rows, base, fall):
```

```python
def collect_labels(p: Path) -> List[str]:
    labs=set()

def to_item(text: str, spans: List[Dict[str,Any]], tok, max_len: int, label_list: List[str]):
    enc = tok(text, truncation=True, max_length=max_len, return_offsets_mapping=True, return_special_tokens_mask=True)

    texts, spans = [], []
    for o in load_jsonl(Path(a.silver)): texts.append(o["text"]); spans.append(o.get("spans",[]))
    ds = DS(texts, spans, tok, a.max_len, labels)
    model = AutoModelForTokenClassification.from_pretrained(a.base_model, num_labels=len(labels, local_files_only=bool(int(os.environ.get('SMA_LOCAL_ONLY','0')))), id2label=id2, label2id={l:i for i,l in id2.items()})
    collator = DataCollatorForTokenClassification(tok)
```

# Smart Mail Agent — AI + RPA 整合專案
