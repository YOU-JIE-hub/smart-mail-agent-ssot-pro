  7377	    @contextmanager
  7378	    def enter_partial_types(
  7379	        self, *, is_function: bool = False, is_class: bool = False
  7380	    ) -> Iterator[None]:
  7381	        """Enter a new scope for collecting partial types.
  7382
  7383	        Also report errors for (some) variables which still have partial
  7384	        types, i.e. we couldn't infer a complete type.
  7385	        """
  7386	        is_local = (self.partial_types and self.partial_types[-1].is_local) or is_function
  7387	        self.partial_types.append(PartialTypeScope({}, is_function, is_local))
  7388	        yield
  7389
  7390	        # Don't complain about not being able to infer partials if it is
  7391	        # at the toplevel (with allow_untyped_globals) or if it is in an
  7392	        # untyped function being checked with check_untyped_defs.
  7393	        permissive = (self.options.allow_untyped_globals and not is_local) or (
  7394	            self.options.check_untyped_defs and self.dynamic_funcs and self.dynamic_funcs[-1]
  7395	        )
  7396
  7397	        partial_types, _, _ = self.partial_types.pop()
  7398	        if not self.current_node_deferred:
  7399	            for var, context in partial_types.items():
  7400	                # If we require local partial types, there are a few exceptions where
  7401	                # we fall back to inferring just "None" as the type from a None initializer:
  7402	                #
  7403	                # 1. If all happens within a single function this is acceptable, since only
  7404	                #    the topmost function is a separate target in fine-grained incremental mode.
  7405	                #    We primarily want to avoid "splitting" partial types across targets.
  7406	                #
  7407	                # 2. A None initializer in the class body if the attribute is defined in a base
  7408	                #    class is fine, since the attribute is already defined and it's currently okay
  7409	                #    to vary the type of an attribute covariantly. The None type will still be
  7410	                #    checked for compatibility with base classes elsewhere. Without this exception
  7411	                #    mypy could require an annotation for an attribute that already has been
  7412	                #    declared in a base class, which would be bad.
  7413	                allow_none = (
  7414	                    not self.options.local_partial_types
  7415	                    or is_function
  7416	                    or (is_class and self.is_defined_in_base_class(var))
  7417	                )
  7418	                if (
  7419	                    allow_none
  7420	                    and isinstance(var.type, PartialType)
  7421	                    and var.type.type is None
  7422	                    and not permissive
  7423	                ):
  7424	                    var.type = NoneType()
  7425	                else:
  7426	                    if var not in self.partial_reported and not permissive:
  7427	                        self.msg.need_annotation_for_var(var, context, self.options.python_version)
  7428	                        self.partial_reported.add(var)
  7429	                    if var.type:
  7430	                        fixed = fixup_partial_type(var.type)
  7431	                        var.invalid_partial_type = fixed != var.type
  7432	                        var.type = fixed

  7482	    def find_partial_types_in_all_scopes(
  7483	        self, var: Var
  7484	    ) -> tuple[bool, bool, dict[Var, Context] | None]:
  7485	        """Look for partial type scope containing variable.
  7486
  7487	        Return tuple (is the scope active, is the scope a local scope, scope).
  7488	        """
  7489	        for scope in reversed(self.partial_types):
  7490	            if var in scope.map:
  7491	                # All scopes within the outermost function are active. Scopes out of
  7492	                # the outermost function are inactive to allow local reasoning (important
  7493	                # for fine-grained incremental mode).

  7498	                    # as if --local-partial-types is always on (because it used to be like this).
  7499	                    disallow_other_scopes = True

  7501	                scope_active = (
  7502	                    not disallow_other_scopes or scope.is_local == self.partial_types[-1].is_local
  7503	                )

  7505	        return False, False, None

  7600	    def conditional_types_with_intersection(
  7601	        self,
  7602	        expr_type: Type,
  7603	        type_ranges: list[TypeRange] | None,
  7604	        ctx: Context,
  7605	        default: Type | None = None,
  7606	    ) -> tuple[Type | None, Type | None]:
  7607	        initial_types = conditional_types(expr_type, type_ranges, default)
  7608	        # For some reason, doing "yes_map, no_map = conditional_types_to_typemaps(...)"
  7609	        # doesn't work: mypyc will decide that 'yes_map' is of type None if we try.
  7610	        yes_type: Type | None = initial_types[0]
  7611	        no_type: Type | None = initial_types[1]
  7612
  7613	        if not isinstance(get_proper_type(yes_type), UninhabitedType) or type_ranges is None:
  7614	            return yes_type, no_type
  7615
  7616	        # If conditional_types was unable to successfully narrow the expr_type
  7617	        # using the type_ranges and concluded if-branch is unreachable, we try
  7618	        # computing it again using a different algorithm that tries to generate
  7619	        # an ad-hoc intersection between the expr_type and the type_ranges.
  7620	        proper_type = get_proper_type(expr_type)
  7621	        if isinstance(proper_type, UnionType):
  7622	            possible_expr_types = get_proper_types(proper_type.relevant_items())
  7623	        else:
  7624	            possible_expr_types = [proper_type]
  7625
  7626	        possible_target_types = []
  7627	        for tr in type_ranges:
  7628	            item = get_proper_type(tr.item)
  7629	            if isinstance(item, (Instance, NoneType)):
  7630	                possible_target_types.append(item)
  7631	        if not possible_target_types:
  7632	            return yes_type, no_type
  7633
  7634	        out = []
  7635	        errors: list[tuple[str, str]] = []
  7636	        for v in possible_expr_types:
  7637	            if not isinstance(v, Instance):
  7638	                return yes_type, no_type
  7639	            for t in possible_target_types:
  7640	                if isinstance(t, NoneType):
  7641	                    errors.append((f'"{v.type.name}" and "NoneType"', '"NoneType" is final'))
  7642	                    continue
  7643	                intersection = self.intersect_instances((v, t), errors)
  7644	                if intersection is None:
  7645	                    continue
  7646	                out.append(intersection)
  7647	        if not out:
  7648	            # Only report errors if no element in the union worked.
  7649	            if self.should_report_unreachable_issues():
  7650	                for types, reason in errors:
  7651	                    self.msg.impossible_intersection(types, reason, ctx)
  7652	            return UninhabitedType(), expr_type
  7653	        new_yes_type = make_simplified_union(out)
  7654	        return new_yes_type, expr_type

  7668	    def get_isinstance_type(self, expr: Expression) -> list[TypeRange] | None:
  7669	        if isinstance(expr, OpExpr) and expr.op == "|":
  7670	            left = self.get_isinstance_type(expr.left)
  7671	            if left is None and is_literal_none(expr.left):
  7672	                left = [TypeRange(NoneType(), is_upper_bound=False)]
  7673	            right = self.get_isinstance_type(expr.right)
  7674	            if right is None and is_literal_none(expr.right):
  7675	                right = [TypeRange(NoneType(), is_upper_bound=False)]
  7676	            if left is None or right is None:
  7677	                return None
  7678	            return left + right
  7679	        all_types = get_proper_types(flatten_types(self.lookup_type(expr)))
  7680	        types: list[TypeRange] = []
  7681	        for typ in all_types:
  7682	            if isinstance(typ, FunctionLike) and typ.is_type_obj():
  7683	                # If a type is generic, `isinstance` can only narrow its variables to Any.
  7684	                any_parameterized = fill_typevars_with_any(typ.type_object())
  7685	                # Tuples may have unattended type variables among their items
  7686	                if isinstance(any_parameterized, TupleType):
  7687	                    erased_type = erase_typevars(any_parameterized)
  7688	                else:
  7689	                    erased_type = any_parameterized
  7690	                types.append(TypeRange(erased_type, is_upper_bound=False))
  7691	            elif isinstance(typ, TypeType):
  7692	                # Type[A] means "any type that is a subtype of A" rather than "precisely type A"
  7693	                # we indicate this by setting is_upper_bound flag
  7694	                is_upper_bound = True
  7695	                if isinstance(typ.item, NoneType):
  7696	                    # except for Type[None], because "'NoneType' is not an acceptable base type"
  7697	                    is_upper_bound = False
  7698	                types.append(TypeRange(typ.item, is_upper_bound=is_upper_bound))
  7699	            elif isinstance(typ, Instance) and typ.type.fullname == "builtins.type":
  7700	                object_type = Instance(typ.type.mro[-1], [])
  7701	                types.append(TypeRange(object_type, is_upper_bound=True))
  7702	            elif isinstance(typ, Instance) and typ.type.fullname == "types.UnionType" and typ.args:
  7703	                types.append(TypeRange(UnionType(typ.args), is_upper_bound=False))
  7704	            elif isinstance(typ, AnyType):
  7705	                types.append(TypeRange(typ, is_upper_bound=False))
  7706	            else:  # we didn't see an actual type, but rather a variable with unknown value
  7707	                return None
  7708	        if not types:
  7709	            # this can happen if someone has empty tuple as 2nd argument to isinstance
  7710	            # strictly speaking, we should return UninhabitedType but for simplicity we will simply
  7711	            # refuse to do any type inference for now
  7712	            return None
  7713	        return types

  7715	    def is_literal_enum(self, n: Expression) -> bool:
  7716	        """Returns true if this expression (with the given type context) is an Enum literal.
  7717
  7718	        For example, if we had an enum:
  7719
  7720	            class Foo(Enum):
  7721	                A = 1
  7722	                B = 2
  7723
  7724	        ...and if the expression 'Foo' referred to that enum within the current type context,
  7725	        then the expression 'Foo.A' would be a literal enum. However, if we did 'a = Foo.A',
  7726	        then the variable 'a' would *not* be a literal enum.
  7727
  7728	        We occasionally special-case expressions like 'Foo.A' and treat them as a single primitive
  7729	        unit for the same reasons we sometimes treat 'True', 'False', or 'None' as a single
  7730	        primitive unit.
  7731	        """
  7732	        if not isinstance(n, MemberExpr) or not isinstance(n.expr, NameExpr):
  7733	            return False
  7734
  7735	        parent_type = self.lookup_type_or_none(n.expr)
  7736	        member_type = self.lookup_type_or_none(n)
  7737	        if member_type is None or parent_type is None:
  7738	            return False
  7739
  7740	        parent_type = get_proper_type(parent_type)
  7741	        member_type = get_proper_type(coerce_to_literal(member_type))
  7742	        if not isinstance(parent_type, FunctionLike) or not isinstance(member_type, LiteralType):
  7743	            return False
  7744
  7745	        if not parent_type.is_type_obj():
  7746	            return False
  7747
  7748	        return (
  7749	            member_type.is_enum_literal()
  7750	            and member_type.fallback.type == parent_type.type_object()
  7751	        )

  7753	    def add_any_attribute_to_type(self, typ: Type, name: str) -> Type:
  7754	        """Inject an extra attribute with Any type using fallbacks."""
  7755	        orig_typ = typ

  7929	def conditional_types(
  7930	    current_type: Type, proposed_type_ranges: list[TypeRange] | None, default: Type | None = None
  7931	) -> tuple[Type | None, Type | None]:
  7932	    """Takes in the current type and a proposed type of an expression.
  7933
  7934	    Returns a 2-tuple: The first element is the proposed type, if the expression
  7935	    can be the proposed type. The second element is the type it would hold
  7936	    if it was not the proposed type, if any. UninhabitedType means unreachable.
  7937	    None means no new information can be inferred. If default is set it is returned
  7938	    instead."""
  7939	    if proposed_type_ranges:
  7940	        if len(proposed_type_ranges) == 1:
  7941	            target = proposed_type_ranges[0].item
  7942	            target = get_proper_type(target)
  7943	            if isinstance(target, LiteralType) and (
  7944	                target.is_enum_literal() or isinstance(target.value, bool)
  7945	            ):
  7946	                enum_name = target.fallback.type.fullname
  7947	                current_type = try_expanding_sum_type_to_union(current_type, enum_name)
  7948	        proposed_items = [type_range.item for type_range in proposed_type_ranges]
  7949	        proposed_type = make_simplified_union(proposed_items)
  7950	        if isinstance(proposed_type, AnyType):
  7951	            # We don't really know much about the proposed type, so we shouldn't
  7952	            # attempt to narrow anything. Instead, we broaden the expr to Any to
  7953	            # avoid false positives
  7954	            return proposed_type, default
  7955	        elif not any(
  7956	            type_range.is_upper_bound for type_range in proposed_type_ranges
  7957	        ) and is_proper_subtype(current_type, proposed_type, ignore_promotions=True):
  7958	            # Expression is always of one of the types in proposed_type_ranges
  7959	            return default, UninhabitedType()
  7960	        elif not is_overlapping_types(current_type, proposed_type, ignore_promotions=True):
  7961	            # Expression is never of any type in proposed_type_ranges
  7962	            return UninhabitedType(), default
  7963	        else:
  7964	            # we can only restrict when the type is precise, not bounded
  7965	            proposed_precise_type = UnionType.make_union(
  7966	                [
  7967	                    type_range.item
  7968	                    for type_range in proposed_type_ranges
  7969	                    if not type_range.is_upper_bound
  7970	                ]
  7971	            )
  7972	            remaining_type = restrict_subtype_away(current_type, proposed_precise_type)
  7973	            return proposed_type, remaining_type
  7974	    else:
  7975	        # An isinstance check, but we don't understand the type
  7976	        return current_type, default

  8099	def and_conditional_maps(m1: TypeMap, m2: TypeMap, use_meet: bool = False) -> TypeMap:
  8100	    """Calculate what information we can learn from the truth of (e1 and e2)
  8101	    in terms of the information that we can learn from the truth of e1 and
  8102	    the truth of e2.
  8103	    """
  8104
  8105	    if m1 is None or m2 is None:
  8106	        # One of the conditions can never be true.
  8107	        return None
  8108	    # Both conditions can be true; combine the information. Anything
  8109	    # we learn from either conditions' truth is valid. If the same
  8110	    # expression's type is refined by both conditions, we somewhat
  8111	    # arbitrarily give precedence to m2 unless m1 value is Any.
  8112	    # In the future, we could use an intersection type or meet_types().
  8113	    result = m2.copy()
  8114	    m2_keys = {literal_hash(n2) for n2 in m2}
  8115	    for n1 in m1:
  8116	        if literal_hash(n1) not in m2_keys or isinstance(get_proper_type(m1[n1]), AnyType):
  8117	            result[n1] = m1[n1]
  8118	    if use_meet:
  8119	        # For now, meet common keys only if specifically requested.
  8120	        # This is currently used for tuple types narrowing, where having
  8121	        # a precise result is important.
  8122	        for n1 in m1:
  8123	            for n2 in m2:
  8124	                if literal_hash(n1) == literal_hash(n2):
  8125	                    result[n1] = meet_types(m1[n1], m2[n2])
  8126	    return result

  8129	def or_conditional_maps(m1: TypeMap, m2: TypeMap, coalesce_any: bool = False) -> TypeMap:
  8130	    """Calculate what information we can learn from the truth of (e1 or e2)
  8131	    in terms of the information that we can learn from the truth of e1 and
  8132	    the truth of e2. If coalesce_any is True, consider Any a supertype when
  8133	    joining restrictions.
  8134	    """
  8135
  8136	    if m1 is None:
  8137	        return m2
  8138	    if m2 is None:
  8139	        return m1
  8140	    # Both conditions can be true. Combine information about
  8141	    # expressions whose type is refined by both conditions. (We do not
  8142	    # learn anything about expressions whose type is refined by only
  8143	    # one condition.)
  8144	    result: dict[Expression, Type] = {}
  8145	    for n1 in m1:
  8146	        for n2 in m2:
  8147	            if literal_hash(n1) == literal_hash(n2):
  8148	                if coalesce_any and isinstance(get_proper_type(m1[n1]), AnyType):
  8149	                    result[n1] = m1[n1]
  8150	                else:
  8151	                    result[n1] = make_simplified_union([m1[n1], m2[n2]])
  8152	    return result

  8155	def reduce_conditional_maps(
  8156	    type_maps: list[tuple[TypeMap, TypeMap]], use_meet: bool = False
  8157	) -> tuple[TypeMap, TypeMap]:
  8158	    """Reduces a list containing pairs of if/else TypeMaps into a single pair.
  8159
  8160	    We "and" together all of the if TypeMaps and "or" together the else TypeMaps. So
  8161	    for example, if we had the input:
  8162
  8163	        [
  8164	            ({x: TypeIfX, shared: TypeIfShared1}, {x: TypeElseX, shared: TypeElseShared1}),
  8165	            ({y: TypeIfY, shared: TypeIfShared2}, {y: TypeElseY, shared: TypeElseShared2}),
  8166	        ]
  8167
  8168	    ...we'd return the output:
  8169
  8170	        (
  8171	            {x: TypeIfX,   y: TypeIfY,   shared: PseudoIntersection[TypeIfShared1, TypeIfShared2]},
  8172	            {shared: Union[TypeElseShared1, TypeElseShared2]},
  8173	        )
  8174
  8175	    ...where "PseudoIntersection[X, Y] == Y" because mypy actually doesn't understand intersections
  8176	    yet, so we settle for just arbitrarily picking the right expr's type.
  8177
  8178	    We only retain the shared expression in the 'else' case because we don't actually know
  8179	    whether x was refined or y was refined -- only just that one of the two was refined.
  8180	    """
  8181	    if len(type_maps) == 0:
  8182	        return {}, {}
  8183	    elif len(type_maps) == 1:
  8184	        return type_maps[0]
  8185	    else:
  8186	        final_if_map, final_else_map = type_maps[0]
  8187	        for if_map, else_map in type_maps[1:]:
  8188	            final_if_map = and_conditional_maps(final_if_map, if_map, use_meet=use_meet)
  8189	            final_else_map = or_conditional_maps(final_else_map, else_map)
  8190
  8191	        return final_if_map, final_else_map

  8255	    """Expand a generic callable using all combinations of type variables' values/bounds."""
  8256	    for tv in c.variables:

  8258	        # type variable that can have other type variables in the upper bound.
  8259	        if tv.id.is_self():
  8260	            c = expand_type(c, {tv.id: tv.upper_bound}).copy_modified(
  8261	                variables=[v for v in c.variables if not v.id.is_self()]
  8262	            )
  8263	            break

  8266	        # Fast path.
  8267	        return [c]

  8269	    tvar_values = []
  8270	    for tvar in c.variables:
  8271	        if isinstance(tvar, TypeVarType) and tvar.values:
  8272	            tvar_values.append(tvar.values)
  8273	        else:

  8294	    1.  s's parameters are partially overlapping with t's. i.e. there are calls that are
  8295	        valid for both signatures.
  8296	    2.  for these common calls, some of t's parameters types are wider that s's.
  8297	    3.  s's return type is NOT a subset of t's.
  8298
  8299	    Note that we use subset rather than subtype relationship in these checks because:
  8300	    * Overload selection happens at runtime, not statically.
  8301	    * This results in more lenient behavior.
  8302	    This can cause false negatives (e.g. if overloaded function returns an externally
  8303	    visible attribute with invariant type), but such situations are rare. In general,
  8304	    overloads in Python are generally unsafe, so we intentionally try to avoid giving
  8305	    non-actionable errors (see more details in comments below).
  8306
  8307	    Assumes that 'signature' appears earlier in the list of overload
  8308	    alternatives then 'other' and that their argument counts are overlapping.
  8309	    """
  8310	    # Try detaching callables from the containing class so that all TypeVars
  8311	    # are treated as being free, i.e. the signature is as seen from inside the class,
  8312	    # where "self" is not yet bound to anything.
  8313	    signature = detach_callable(signature, class_type_vars)
  8314	    other = detach_callable(other, class_type_vars)
  8315
  8316	    # Note: We repeat this check twice in both directions compensate for slight
  8317	    # asymmetries in 'is_callable_compatible'.
  8318
  8319	    for sig_variant in expand_callable_variants(signature):
  8320	        for other_variant in expand_callable_variants(other):
  8321	            # Using only expanded callables may cause false negatives, we can add
  8322	            # more variants (e.g. using inference between callables) in the future.
  8323	            if is_subset_no_promote(sig_variant.ret_type, other_variant.ret_type):
  8324	                continue
  8325	            if not (
  8326	                is_callable_compatible(
  8327	                    sig_variant,
  8328	                    other_variant,
  8329	                    is_compat=is_overlapping_types_for_overload,
  8330	                    check_args_covariantly=False,
  8331	                    is_proper_subtype=False,
  8332	                    is_compat_return=lambda l, r: not is_subset_no_promote(l, r),
  8333	                    allow_partial_overlap=True,
  8334	                )
  8335	                or is_callable_compatible(
  8336	                    other_variant,
  8337	                    sig_variant,
  8338	                    is_compat=is_overlapping_types_for_overload,
  8339	                    check_args_covariantly=True,
  8340	                    is_proper_subtype=False,
  8341	                    is_compat_return=lambda l, r: not is_subset_no_promote(r, l),
  8342	                    allow_partial_overlap=True,
  8343	                )
  8344	            ):
  8345	                continue
  8346	            # Using the same `allow_partial_overlap` flag as before, can cause false
  8347	            # negatives in case where star argument is used in a catch-all fallback overload.
  8348	            # But again, practicality beats purity here.
  8349	            if not partial_only or not is_callable_compatible(
  8350	                other_variant,
  8351	                sig_variant,
  8352	                is_compat=is_subset_no_promote,
  8353	                check_args_covariantly=True,
  8354	                is_proper_subtype=False,
  8355	                ignore_return=True,
  8356	                allow_partial_overlap=True,
  8357	            ):
  8358	                return True
  8359	    return False

  8539	class DisjointDict(Generic[TKey, TValue]):
  8540	    """An variation of the union-find algorithm/data structure where instead of keeping
  8541	    track of just disjoint sets, we keep track of disjoint dicts -- keep track of multiple
  8542	    Set[Key] -> Set[Value] mappings, where each mapping's keys are guaranteed to be disjoint.
  8543
  8544	    This data structure is currently used exclusively by 'group_comparison_operands' below
  8545	    to merge chains of '==' and 'is' comparisons when two or more chains use the same expression
  8546	    in best-case O(n), where n is the number of operands.
  8547
  8548	    Specifically, the `add_mapping()` function and `items()` functions will take on average
  8549	    O(k + v) and O(n) respectively, where k and v are the number of keys and values we're adding
  8550	    for a given chain. Note that k <= n and v <= n.
  8551
  8552	    We hit these average/best-case scenarios for most user code: e.g. when the user has just
  8553	    a single chain like 'a == b == c == d == ...' or multiple disjoint chains like
  8554	    'a==b < c==d < e==f < ...'. (Note that a naive iterative merging would be O(n^2) for
  8555	    the latter case).
  8556
  8557	    In comparison, this data structure will make 'group_comparison_operands' have a worst-case
  8558	    runtime of O(n*log(n)): 'add_mapping()' and 'items()' are worst-case O(k*log(n) + v) and
  8559	    O(k*log(n)) respectively. This happens only in the rare case where the user keeps repeatedly
  8560	    making disjoint mappings before merging them in a way that persistently dodges the path
  8561	    compression optimization in '_lookup_root_id', which would end up constructing a single
  8562	    tree of height log_2(n). This makes root lookups no longer amoritized constant time when we
  8563	    finally call 'items()'.
  8564	    """
  8565
  8566	    def __init__(self) -> None:
  8567	        # Each key maps to a unique ID
  8568	        self._key_to_id: dict[TKey, int] = {}
  8569
  8570	        # Each id points to the parent id, forming a forest of upwards-pointing trees. If the
  8571	        # current id already is the root, it points to itself. We gradually flatten these trees
  8572	        # as we perform root lookups: eventually all nodes point directly to its root.
  8573	        self._id_to_parent_id: dict[int, int] = {}
  8574
  8575	        # Each root id in turn maps to the set of values.
  8576	        self._root_id_to_values: dict[int, set[TValue]] = {}
  8577
  8578	    def add_mapping(self, keys: set[TKey], values: set[TValue]) -> None:
  8579	        """Adds a 'Set[TKey] -> Set[TValue]' mapping. If there already exists a mapping
  8580	        containing one or more of the given keys, we merge the input mapping with the old one.
  8581
  8582	        Note that the given set of keys must be non-empty -- otherwise, nothing happens.
  8583	        """
  8584	        if not keys:
  8585	            return
  8586
  8587	        subtree_roots = [self._lookup_or_make_root_id(key) for key in keys]
  8588	        new_root = subtree_roots[0]
  8589
  8590	        root_values = self._root_id_to_values[new_root]
  8591	        root_values.update(values)
  8592	        for subtree_root in subtree_roots[1:]:
  8593	            if subtree_root == new_root or subtree_root not in self._root_id_to_values:
  8594	                continue
  8595	            self._id_to_parent_id[subtree_root] = new_root
  8596	            root_values.update(self._root_id_to_values.pop(subtree_root))
  8597
  8598	    def items(self) -> list[tuple[set[TKey], set[TValue]]]:
  8599	        """Returns all disjoint mappings in key-value pairs."""
  8600	        root_id_to_keys: dict[int, set[TKey]] = {}
  8601	        for key in self._key_to_id:
  8602	            root_id = self._lookup_root_id(key)
  8603	            if root_id not in root_id_to_keys:
  8604	                root_id_to_keys[root_id] = set()
  8605	            root_id_to_keys[root_id].add(key)
  8606
  8607	        output = []
  8608	        for root_id, keys in root_id_to_keys.items():
  8609	            output.append((keys, self._root_id_to_values[root_id]))
  8610
  8611	        return output
  8612
  8613	    def _lookup_or_make_root_id(self, key: TKey) -> int:
  8614	        if key in self._key_to_id:
  8615	            return self._lookup_root_id(key)
  8616	        else:
  8617	            new_id = len(self._key_to_id)
  8618	            self._key_to_id[key] = new_id
  8619	            self._id_to_parent_id[new_id] = new_id
  8620	            self._root_id_to_values[new_id] = set()
  8621	            return new_id
  8622
  8623	    def _lookup_root_id(self, key: TKey) -> int:
  8624	        i = self._key_to_id[key]
  8625	        while i != self._id_to_parent_id[i]:
  8626	            # Optimization: make keys directly point to their grandparents to speed up
  8627	            # future traversals. This prevents degenerate trees of height n from forming.
  8628	            new_parent = self._id_to_parent_id[self._id_to_parent_id[i]]
  8629	            self._id_to_parent_id[i] = new_parent
  8630	            i = new_parent
  8631	        return i

  8634	def group_comparison_operands(
  8635	    pairwise_comparisons: Iterable[tuple[str, Expression, Expression]],
  8636	    operand_to_literal_hash: Mapping[int, Key],
  8637	    operators_to_group: set[str],
  8638	) -> list[tuple[str, list[int]]]:
  8639	    """Group a series of comparison operands together chained by any operand
  8640	    in the 'operators_to_group' set. All other pairwise operands are kept in
  8641	    groups of size 2.
  8642
  8643	    For example, suppose we have the input comparison expression:
  8644
  8645	        x0 == x1 == x2 < x3 < x4 is x5 is x6 is not x7 is not x8
  8646
  8647	    If we get these expressions in a pairwise way (e.g. by calling ComparisonExpr's
  8648	    'pairwise()' method), we get the following as input:
  8649
  8650	        [('==', x0, x1), ('==', x1, x2), ('<', x2, x3), ('<', x3, x4),
  8651	         ('is', x4, x5), ('is', x5, x6), ('is not', x6, x7), ('is not', x7, x8)]
  8652
  8653	    If `operators_to_group` is the set {'==', 'is'}, this function will produce
  8654	    the following "simplified operator list":
  8655
  8656	       [("==", [0, 1, 2]), ("<", [2, 3]), ("<", [3, 4]),
  8657	        ("is", [4, 5, 6]), ("is not", [6, 7]), ("is not", [7, 8])]
  8658
  8659	    Note that (a) we yield *indices* to the operands rather then the operand
  8660	    expressions themselves and that (b) operands used in a consecutive chain
  8661	    of '==' or 'is' are grouped together.
  8662
  8663	    If two of these chains happen to contain operands with the same underlying
  8664	    literal hash (e.g. are assignable and correspond to the same expression),
  8665	    we combine those chains together. For example, if we had:
  8666
  8667	        same == x < y == same
  8668
  8669	    ...and if 'operand_to_literal_hash' contained the same values for the indices
  8670	    0 and 3, we'd produce the following output:
  8671
  8672	        [("==", [0, 1, 2, 3]), ("<", [1, 2])]
  8673
  8674	    But if the 'operand_to_literal_hash' did *not* contain an entry, we'd instead
  8675	    default to returning:
  8676
  8677	        [("==", [0, 1]), ("<", [1, 2]), ("==", [2, 3])]
  8678
  8679	    This function is currently only used to assist with type-narrowing refinements
  8680	    and is extracted out to a helper function so we can unit test it.
  8681	    """
  8682	    groups: dict[str, DisjointDict[Key, int]] = {op: DisjointDict() for op in operators_to_group}
  8683
  8684	    simplified_operator_list: list[tuple[str, list[int]]] = []
  8685	    last_operator: str | None = None
  8686	    current_indices: set[int] = set()
  8687	    current_hashes: set[Key] = set()
  8688	    for i, (operator, left_expr, right_expr) in enumerate(pairwise_comparisons):
  8689	        if last_operator is None:
  8690	            last_operator = operator
  8691
  8692	        if current_indices and (operator != last_operator or operator not in operators_to_group):
  8693	            # If some of the operands in the chain are assignable, defer adding it: we might
  8694	            # end up needing to merge it with other chains that appear later.
  8695	            if not current_hashes:
  8696	                simplified_operator_list.append((last_operator, sorted(current_indices)))
  8697	            else:
  8698	                groups[last_operator].add_mapping(current_hashes, current_indices)
  8699	            last_operator = operator
  8700	            current_indices = set()
  8701	            current_hashes = set()
  8702
  8703	        # Note: 'i' corresponds to the left operand index, so 'i + 1' is the
  8704	        # right operand.
  8705	        current_indices.add(i)
  8706	        current_indices.add(i + 1)
  8707
  8708	        # We only ever want to combine operands/combine chains for these operators
  8709	        if operator in operators_to_group:
  8710	            left_hash = operand_to_literal_hash.get(i)
  8711	            if left_hash is not None:
  8712	                current_hashes.add(left_hash)
  8713	            right_hash = operand_to_literal_hash.get(i + 1)
  8714	            if right_hash is not None:
  8715	                current_hashes.add(right_hash)
  8716
  8717	    if last_operator is not None:
  8718	        if not current_hashes:
  8719	            simplified_operator_list.append((last_operator, sorted(current_indices)))
  8720	        else:
  8721	            groups[last_operator].add_mapping(current_hashes, current_indices)
  8722
  8723	    # Now that we know which chains happen to contain the same underlying expressions
  8724	    # and can be merged together, add in this info back to the output.
  8725	    for operator, disjoint_dict in groups.items():
  8726	        for keys, indices in disjoint_dict.items():
  8727	            simplified_operator_list.append((operator, sorted(indices)))
  8728
  8729	    # For stability, reorder list by the first operand index to appear
  8730	    simplified_operator_list.sort(key=lambda item: item[1][0])
  8731	    return simplified_operator_list

   306	    def enclosing_class(self, func: FuncItem | None = None) -> TypeInfo | None:
   307	        """Is there a class *directly* enclosing this function?"""
   308	        func = func or self.current_function()

   310	        index = self.stack.index(func)

from __future__ import annotations

import os
import smtplib
from email.message import EmailMessage

__all__ = ["main", "smtplib"]


def main() -> int:
    need = ["SMTP_USER", "SMTP_PASS", "SMTP_HOST", "SMTP_PORT", "REPLY_TO"]
    env = {k: os.getenv(k) for k in need}
    if any(not env[k] for k in need):
        return 2
    msg = EmailMessage()
    msg["From"] = env["REPLY_TO"]
    msg["To"] = env["REPLY_TO"]
    msg["Subject"] = "Online check"
    msg.set_content("ping")
    try:
        with smtplib.SMTP_SSL(env["SMTP_HOST"], int(env["SMTP_PORT"])) as s:
            s.login(env["SMTP_USER"], env["SMTP_PASS"])
            s.send_message(msg)
        return 0
    except Exception:
        return 1
