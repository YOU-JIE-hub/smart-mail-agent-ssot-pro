   790	class QRPolynomial:
   791	    def __init__(self, num, shift):
   792	        if (len(num) == 0):
   793	            raise Exception(len(num) + "/" + shift)
   794	        offset = 0
   795	        while offset < len(num) and num[offset] == 0:
   796	            offset += 1
   797	        self.num = num[offset:] + [0]*shift
   798
   799	    def get(self, index):
   800	        return self.num[index]
   801
   802	    def getLength(self):
   803	        return len(self.num)
   804
   805	    def multiply(self, e):
   806	        num = [0] * (self.getLength() + e.getLength() - 1);
   807	        for i in range(self.getLength()):
   808	            for j in range(e.getLength()):
   809	                num[i + j] ^= QRMath.gexp(QRMath.glog(self.get(i) ) +
   810	                                          QRMath.glog(e.get(j) ) )
   811	        return QRPolynomial(num, 0);
   812
   813	    def mod(self, e):
   814	        if (self.getLength() < e.getLength()):
   815	            return self;
   816	        ratio = QRMath.glog(self.num[0] ) - QRMath.glog(e.num[0] )
   817	        num = [nn ^ QRMath.gexp(QRMath.glog(en) + ratio)
   818	               for nn,en in zip(self.num, e.num)]
   819	        num += self.num[e.getLength():]
   820	        # recursive call
   821	        return QRPolynomial(num, 0).mod(e);

   382	    '''
   383	    >>> usps = [USPS_4State('01234567094987654321',x).binary for x in ('','01234','012345678','01234567891')]
   384	    >>> print(' '.join(nhex(x) for x in usps))

   387	    0x51 0x65 0x606 0x751
   388	    '''
   389	    hexbytes = nhex(int(value))[2:]
   390	    hexbytes = '0'*(26-len(hexbytes))+hexbytes
   391	    gp = 0x0F35
   392	    fcs = 0x07FF
   393	    data = int(hexbytes[:2],16)<<5
   394	    for b in range(2,8):
   395	        if (fcs ^ data)&0x400:
   396	            fcs = (fcs<<1)^gp
   397	        else:
   398	            fcs = fcs<<1
   399	        fcs &= 0x7ff
   400	        data <<= 1
   401
   402	    for x in range(2,2*13,2):

   407	            else:
   408	                fcs = fcs<<1

   414	    '''reverse unsigned 13 bit number
   415	    >>> print(_ru13(7936), _ru13(31), _ru13(47), _ru13(7808))
   416	    31 7936 7808 47
   417	    '''
   418	    r = 0

   426	    '''create and return table of 13 bit values with N bits on
   427	    >>> T = _initNof13Table(5,1287)
   428	    >>> print(' '.join('T[%d]=%d' % (i, T[i]) for i in (0,1,2,3,4,1271,1272,1284,1285,1286)))

   430	    '''
   431	    T = lenT*[None]
   432	    l = 0
   433	    u = lenT-1
   434	    for c in range(8192):
   435	        bc = 0
   436	        for b in range(13):
   437	            bc += (c&(1<<b))!=0
   438	        if bc!=N: continue

   440	        if r<c: continue    #we already looked at this pair
   441	        if r==c:
   442	            T[u] = c
   443	            u -= 1
   444	        else:
   445	            T[l] = c
   446	            l += 1
   447	            T[l] = r
   448	            l += 1
   449	    assert l==(u+1), 'u+1(%d)!=l(%d) for %d of 13 table' % (u+1,l,N)

    50	def _findMinMaxValue(V, x, default, func, special=None):
    51	    if isSeq(V[0][0]):
    52	        if special:
    53	            f=lambda T,x=x,special=special,func=func: special(T,x,func)
    54	        else:
    55	            f=lambda T,x=x: T[x]
    56	        V=list(map(lambda e,f=f: list(map(f,e)),V))
    57	    V = list(filter(len,[[x for x in x if x is not None] for x in V]))
    58	    if len(V)==0: return default
    59	    return func(list(map(func,V)))

   108	class AxisLineAnnotation:
   109	    '''Create a grid like line using the given user value to draw the line
   110	    kwds may contain
   111	    startOffset if true v is offset from the default grid start position
   112	    endOffset   if true v is offset from the default grid end position
   113	    scaleValue  True/not given --> scale the value
   114	                otherwise use the absolute value
   115	    lo          lowest coordinate to draw default 0
   116	    hi          highest coordinate to draw at default = length
   117	    drawAtLimit True draw line at appropriate limit if its coordinate exceeds the lo, hi range
   118	                False ignore if it's outside the range
   119	    all Line keywords are acceptable
   120	    '''
   121	    def __init__(self,v,**kwds):
   122	        self._v = v
   123	        self._kwds = kwds
   124
   125	    def __call__(self,axis):
   126	        kwds = self._kwds.copy()
   127	        scaleValue = kwds.pop('scaleValue',True)
   128	        endOffset = kwds.pop('endOffset',False)
   129	        startOffset = kwds.pop('startOffset',False)
   130	        if axis.isYAxis:
   131	            offs = axis._x
   132	            d0 = axis._y
   133	        else:
   134	            offs = axis._y
   135	            d0 = axis._x
   136	        s = kwds.pop('start',None)
   137	        e = kwds.pop('end',None)
   138	        if s is None or e is None:
   139	            dim = getattr(getattr(axis,'joinAxis',None),'getGridDims',None)
   140	            if dim and hasattr(dim,'__call__'):
   141	                dim = dim()
   142	            if dim:
   143	                if s is None: s = dim[0]
   144	                if e is None: e = dim[1]
   145	            else:
   146	                if s is None: s = 0
   147	                if e is None: e = 0
   148	        hi = kwds.pop('hi',axis._length)+d0
   149	        lo = kwds.pop('lo',0)+d0
   150	        lo,hi=min(lo,hi),max(lo,hi)
   151	        drawAtLimit = kwds.pop('drawAtLimit',False)
   152	        oaglp = axis._get_line_pos
   153	        if not scaleValue:
   154	            axis._get_line_pos = lambda x: x
   155	        try:
   156	            v = self._v
   157	            if endOffset:
   158	                v = v + hi
   159	            elif startOffset:
   160	                v = v + lo
   161	            func = axis._getLineFunc(s-offs,e-offs,kwds.pop('parent',None))
   162	            if not hasattr(axis,'_tickValues'):
   163	                axis._pseudo_configure()
   164	            d = axis._get_line_pos(v)
   165	            if d<lo or d>hi:
   166	                if not drawAtLimit: return None
   167	                if d<lo:
   168	                    d = lo
   169	                else:
   170	                    d = hi
   171	                axis._get_line_pos = lambda x: d
   172	            L = func(v)
   173	            for k,v in kwds.items():
   174	                setattr(L,k,v)
   175	        finally:
   176	            axis._get_line_pos = oaglp
   177	        return L

   179	class AxisBackgroundAnnotation:
   180	    '''Create a set of coloured bars on the background of a chart using axis ticks as the bar borders
   181	    colors is a set of colors to use for the background bars. A colour of None is just a skip.
   182	    Special effects if you pass a rect or Shaded rect instead.
   183	    '''
   184	    def __init__(self,colors,**kwds):
   185	        self._colors = colors
   186	        self._kwds = kwds
   187
   188	    def __call__(self,axis):
   189	        colors = self._colors
   190	        if not colors: return
   191	        kwds = self._kwds.copy()
   192	        isYAxis = axis.isYAxis
   193	        if isYAxis:
   194	            offs = axis._x
   195	            d0 = axis._y
   196	        else:
   197	            offs = axis._y
   198	            d0 = axis._x
   199	        s = kwds.pop('start',None)
   200	        e = kwds.pop('end',None)
   201	        if s is None or e is None:
   202	            dim = getattr(getattr(axis,'joinAxis',None),'getGridDims',None)
   203	            if dim and hasattr(dim,'__call__'):
   204	                dim = dim()
   205	            if dim:
   206	                if s is None: s = dim[0]
   207	                if e is None: e = dim[1]
   208	            else:
   209	                if s is None: s = 0
   210	                if e is None: e = 0
   211	        if not hasattr(axis,'_tickValues'):
   212	            axis._pseudo_configure()
   213	        tv = getattr(axis,'_tickValues',None)
   214	        if not tv: return
   215	        G = Group()
   216	        ncolors = len(colors)
   217	        v0 = axis._get_line_pos(tv[0])
   218	        for i in range(1,len(tv)):
   219	            v1 = axis._get_line_pos(tv[i])
   220	            c = colors[(i-1)%ncolors]
   221	            if c:
   222	                if isYAxis:
   223	                    y = v0
   224	                    x = s
   225	                    height = v1-v0
   226	                    width = e-s
   227	                else:
   228	                    x = v0
   229	                    y = s
   230	                    width = v1-v0
   231	                    height = e-s
   232	                if isinstance(c,Color):
   233	                    r = Rect(x,y,width,height,fillColor=c,strokeColor=None)
   234	                elif isinstance(c,Rect):
   235	                    r = Rect(x,y,width,height)
   236	                    for k in c.__dict__:
   237	                        if k not in ('x','y','width','height'):
   238	                            setattr(r,k,getattr(c,k))
   239	                elif isinstance(c,ShadedRect):
   240	                    r = ShadedRect(x=x,y=y,width=width,height=height)
   241	                    for k in c.__dict__:
   242	                        if k not in ('x','y','width','height'):
   243	                            setattr(r,k,getattr(c,k))
   244	                G.add(r)
   245	            v0 = v1
   246	        return G

   248	class TickLU:
   249	    '''lookup special cases for tick values'''
   250	    def __init__(self,*T,**kwds):
   251	        self.accuracy = kwds.pop('accuracy',1e-8)
   252	        self.T = T
   253	    def __contains__(self,t):
   254	        accuracy = self.accuracy
   255	        for x,v in self.T:
   256	            if abs(x-t)<accuracy:
   257	                return True
   258	        return False
   259	    def __getitem__(self,t):
   260	        accuracy = self.accuracy
   261	        for x,v in self.T:
   262	            if abs(x-t)<self.accuracy:
   263	                return v
   264	        raise IndexError('cannot locate index %r' % t)
