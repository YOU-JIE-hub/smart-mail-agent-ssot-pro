   406	        if isinstance(stmt.rvalue, CallExpr):

   423	def allocate_class(builder: IRBuilder, cdef: ClassDef) -> Value:
   424	    # OK AND NOW THE FUN PART
   425	    base_exprs = cdef.base_type_exprs + cdef.removed_base_type_exprs
   426	    new_style_type_args = cdef.type_args
   427	    if new_style_type_args:
   428	        bases = [make_generic_base_class(builder, cdef.fullname, new_style_type_args, cdef.line)]
   429	    else:
   430	        bases = []
   431
   432	    if base_exprs or new_style_type_args:
   433	        bases.extend([builder.accept(x) for x in base_exprs])
   434	        tp_bases = builder.new_tuple(bases, cdef.line)
   435	    else:
   436	        tp_bases = builder.add(LoadErrorValue(object_rprimitive, is_borrowed=True))
   437	    modname = builder.load_str(builder.module_name)
   438	    template = builder.add(
   439	        LoadStatic(object_rprimitive, cdef.name + "_template", builder.module_name, NAMESPACE_TYPE)
   440	    )
   441	    # Create the class
   442	    tp = builder.call_c(pytype_from_template_op, [template, tp_bases, modname], cdef.line)
   443	    # Immediately fix up the trait vtables, before doing anything with the class.
   444	    ir = builder.mapper.type_to_ir[cdef.info]
   445	    if not ir.is_trait and not ir.builtin_base:
   446	        builder.add(
   447	            Call(
   448	                FuncDecl(
   449	                    cdef.name + "_trait_vtable_setup",
   450	                    None,
   451	                    builder.module_name,
   452	                    FuncSignature([], bool_rprimitive),
   453	                ),
   454	                [],
   455	                -1,
   456	            )
   457	        )
   458	    # Populate a '__mypyc_attrs__' field containing the list of attrs
   459	    builder.primitive_op(
   460	        py_setattr_op,
   461	        [
   462	            tp,
   463	            builder.load_str("__mypyc_attrs__"),
   464	            create_mypyc_attrs_tuple(builder, builder.mapper.type_to_ir[cdef.info], cdef.line),
   465	        ],
   466	        cdef.line,
   467	    )
   468
   469	    # Save the class
   470	    builder.add(InitStatic(tp, cdef.name, builder.module_name, NAMESPACE_TYPE))
   471
   472	    # Add it to the dict
   473	    builder.primitive_op(
   474	        dict_set_item_op, [builder.load_globals_dict(), builder.load_str(cdef.name), tp], cdef.line
   475	    )
   476
   477	    return tp

   480	def make_generic_base_class(
   481	    builder: IRBuilder, fullname: str, type_args: list[TypeParam], line: int
   482	) -> Value:
   483	    """Construct Generic[...] base class object for a new-style generic class (Python 3.12)."""
   484	    mod = builder.call_c(import_op, [builder.load_str("_typing")], line)
   485	    tvs = create_type_params(builder, mod, type_args, line)
   486	    args = []
   487	    for tv, type_param in zip(tvs, type_args):
   488	        if type_param.kind == TYPE_VAR_TUPLE_KIND:
   489	            # Evaluate *Ts for a TypeVarTuple
   490	            it = builder.primitive_op(iter_op, [tv], line)
   491	            tv = builder.call_c(next_op, [it], line)
   492	        args.append(tv)
   493
   494	    gent = builder.py_get_attr(mod, "Generic", line)
   495	    if len(args) == 1:
   496	        arg = args[0]
   497	    else:
   498	        arg = builder.new_tuple(args, line)
   499
   500	    base = builder.primitive_op(py_get_item_op, [gent, arg], line)
   501	    return base

   512	def populate_non_ext_bases(builder: IRBuilder, cdef: ClassDef) -> Value:
   513	    """Create base class tuple of a non-extension class.
   514
   515	    The tuple is passed to the metaclass constructor.
   516	    """
   517	    is_named_tuple = cdef.info.is_named_tuple
   518	    ir = builder.mapper.type_to_ir[cdef.info]
   519	    bases = []
   520	    for cls in (b.type for b in cdef.info.bases):
   521	        if cls.fullname == "builtins.object":
   522	            continue
   523	        if is_named_tuple and cls.fullname in (
   524	            "typing.Sequence",
   525	            "typing.Iterable",
   526	            "typing.Collection",
   527	            "typing.Reversible",
   528	            "typing.Container",
   529	            "typing.Sized",
   530	        ):
   531	            # HAX: Synthesized base classes added by mypy don't exist at runtime, so skip them.
   532	            #      This could break if they were added explicitly, though...
   533	            continue
   534	        # Add the current class to the base classes list of concrete subclasses
   535	        if cls in builder.mapper.type_to_ir:
   536	            base_ir = builder.mapper.type_to_ir[cls]
   537	            if base_ir.children is not None:
   538	                base_ir.children.append(ir)
   539
   540	        if cls.fullname in MAGIC_TYPED_DICT_CLASSES:
   541	            # HAX: Mypy internally represents TypedDict classes differently from what
   542	            #      should happen at runtime. Replace with something that works.
   543	            module = "typing"
   544	            name = "_TypedDict"
   545	            base = builder.get_module_attr(module, name, cdef.line)
   546	        elif is_named_tuple and cls.fullname == "builtins.tuple":
   547	            name = "_NamedTuple"
   548	            base = builder.get_module_attr("typing", name, cdef.line)
   549	        else:
   550	            cls_module = cls.fullname.rsplit(".", 1)[0]
   551	            if cls_module == builder.current_module:
   552	                base = builder.load_global_str(cls.name, cdef.line)
   553	            else:
   554	                base = builder.load_module_attr_by_fullname(cls.fullname, cdef.line)
   555	        bases.append(base)
   556	        if cls.fullname in MAGIC_TYPED_DICT_CLASSES:
   557	            # The remaining base classes are synthesized by mypy and should be ignored.
   558	            break
   559	    return builder.new_tuple(bases, cdef.line)

   563	    """Find the metaclass of a class from its defs and bases."""
   564	    if cdef.metaclass:

   573	            # We can't easily calculate it generically, so special case it.

   583	def setup_non_ext_dict(
   584	    builder: IRBuilder, cdef: ClassDef, metaclass: Value, bases: Value
   585	) -> Value:
   586	    """Initialize the class dictionary for a non-extension class.
   587
   588	    This class dictionary is passed to the metaclass constructor.
   589	    """
   590	    # Check if the metaclass defines a __prepare__ method, and if so, call it.
   591	    has_prepare = builder.primitive_op(
   592	        py_hasattr_op, [metaclass, builder.load_str("__prepare__")], cdef.line
   593	    )
   594
   595	    non_ext_dict = Register(dict_rprimitive)
   596
   597	    true_block, false_block, exit_block = BasicBlock(), BasicBlock(), BasicBlock()
   598	    builder.add_bool_branch(has_prepare, true_block, false_block)
   599
   600	    builder.activate_block(true_block)
   601	    cls_name = builder.load_str(cdef.name)
   602	    prepare_meth = builder.py_get_attr(metaclass, "__prepare__", cdef.line)
   603	    prepare_dict = builder.py_call(prepare_meth, [cls_name, bases], cdef.line)
   604	    builder.assign(non_ext_dict, prepare_dict, cdef.line)
   605	    builder.goto(exit_block)
   606
   607	    builder.activate_block(false_block)
   608	    builder.assign(non_ext_dict, builder.call_c(dict_new_op, [], cdef.line), cdef.line)
   609	    builder.goto(exit_block)
   610	    builder.activate_block(exit_block)
   611
   612	    return non_ext_dict

   615	def add_non_ext_class_attr_ann(
   616	    builder: IRBuilder,
   617	    non_ext: NonExtClassInfo,
   618	    lvalue: NameExpr,
   619	    stmt: AssignmentStmt,
   620	    get_type_info: Callable[[AssignmentStmt], TypeInfo | None] | None = None,
   621	) -> None:
   622	    """Add a class attribute to __annotations__ of a non-extension class."""
   623	    # FIXME: try to better preserve the special forms and type parameters of generics.
   624	    typ: Value | None = None
   625	    if get_type_info is not None:
   626	        type_info = get_type_info(stmt)
   627	        if type_info:
   628	            # NOTE: Using string type information is similar to using
   629	            # `from __future__ import annotations` in standard python.
   630	            # NOTE: For string types we need to use the fullname since it
   631	            # includes the module. If string type doesn't have the module,
   632	            # @dataclass will try to get the current module and fail since the
   633	            # current module is not in sys.modules.
   634	            if builder.current_module == type_info.module_name and stmt.line < type_info.line:
   635	                typ = builder.load_str(type_info.fullname)
   636	            else:
   637	                typ = load_type(builder, type_info, stmt.unanalyzed_type, stmt.line)
   638
   639	    if typ is None:
   640	        # FIXME: if get_type_info is not provided, don't fall back to stmt.type?
   641	        ann_type = get_proper_type(stmt.type)
   642	        if (
   643	            isinstance(stmt.unanalyzed_type, UnboundType)
   644	            and stmt.unanalyzed_type.original_str_expr is not None
   645	        ):
   646	            # Annotation is a forward reference, so don't attempt to load the actual
   647	            # type and load the string instead.
   648	            #
   649	            # TODO: is it possible to determine whether a non-string annotation is
   650	            # actually a forward reference due to the __annotations__ future?
   651	            typ = builder.load_str(stmt.unanalyzed_type.original_str_expr)
   652	        elif isinstance(ann_type, Instance):
   653	            typ = load_type(builder, ann_type.type, stmt.unanalyzed_type, stmt.line)
   654	        else:
   655	            typ = builder.add(LoadAddress(type_object_op.type, type_object_op.src, stmt.line))
   656
   657	    key = builder.load_str(lvalue.name)
   658	    builder.primitive_op(dict_set_item_op, [non_ext.anns, key, typ], stmt.line)

   688	def find_attr_initializers(
   689	    builder: IRBuilder, cdef: ClassDef, skip: Callable[[str, AssignmentStmt], bool] | None = None
   690	) -> tuple[set[str], list[AssignmentStmt]]:
   691	    """Find initializers of attributes in a class body.
   692
   693	    If provided, the skip arg should be a callable which will return whether
   694	    to skip generating a default for an attribute.  It will be passed the name of
   695	    the attribute and the corresponding AssignmentStmt.
   696	    """
   697	    cls = builder.mapper.type_to_ir[cdef.info]
   698	    if cls.builtin_base:
   699	        return set(), []
   700
   701	    attrs_with_defaults = set()
   702
   703	    # Pull out all assignments in classes in the mro so we can initialize them
   704	    # TODO: Support nested statements
   705	    default_assignments = []
   706	    for info in reversed(cdef.info.mro):
   707	        if info not in builder.mapper.type_to_ir:
   708	            continue
   709	        for stmt in info.defn.defs.body:
   710	            if (
   711	                isinstance(stmt, AssignmentStmt)
   712	                and isinstance(stmt.lvalues[0], NameExpr)
   713	                and not is_class_var(stmt.lvalues[0])
   714	                and not isinstance(stmt.rvalue, TempNode)
   715	            ):
   716	                name = stmt.lvalues[0].name
   717	                if name == "__slots__":
   718	                    continue
   719
   720	                if name == "__deletable__":
   721	                    check_deletable_declaration(builder, cls, stmt.line)
   722	                    continue
   723
   724	                if skip is not None and skip(name, stmt):
   725	                    continue
   726
   727	                attr_type = cls.attr_type(name)
   728
   729	                # If the attribute is initialized to None and type isn't optional,
   730	                # doesn't initialize it to anything (special case for "# type:" comments).
   731	                if isinstance(stmt.rvalue, RefExpr) and stmt.rvalue.fullname == "builtins.None":
   732	                    if (
   733	                        not is_optional_type(attr_type)
   734	                        and not is_object_rprimitive(attr_type)
   735	                        and not is_none_rprimitive(attr_type)
   736	                    ):
   737	                        continue
   738
   739	                attrs_with_defaults.add(name)
   740	                default_assignments.append(stmt)
   741
   742	    return attrs_with_defaults, default_assignments

   745	def generate_attr_defaults_init(
   746	    builder: IRBuilder, cdef: ClassDef, default_assignments: list[AssignmentStmt]
   747	) -> None:
   748	    """Generate an initialization method for default attr values (from class vars)."""
   749	    if not default_assignments:
   750	        return
   751	    cls = builder.mapper.type_to_ir[cdef.info]
   752	    if cls.builtin_base:
   753	        return
   754
   755	    with builder.enter_method(cls, "__mypyc_defaults_setup", bool_rprimitive):
   756	        self_var = builder.self()
   757	        for stmt in default_assignments:
   758	            lvalue = stmt.lvalues[0]
   759	            assert isinstance(lvalue, NameExpr)
   760	            if not stmt.is_final_def and not is_constant(stmt.rvalue):
   761	                builder.warning("Unsupported default attribute value", stmt.rvalue.line)

   764	            val = builder.coerce(builder.accept(stmt.rvalue), attr_type, stmt.line)
   765	            init = SetAttr(self_var, lvalue.name, val, -1)
   766	            init.mark_as_initializer()
   767	            builder.add(init)
   768
   769	        builder.add(Return(builder.true()))

   772	def check_deletable_declaration(builder: IRBuilder, cl: ClassIR, line: int) -> None:
   773	    for attr in cl.deletable:

   786	                        attr, cl.name, base.name

   788	                    line,

   799	def gen_glue_ne_method(builder: IRBuilder, cls: ClassIR, line: int) -> None:
   800	    """Generate a "__ne__" method from a "__eq__" method."""
   801	    func_ir = cls.get_method("__eq__")
   802	    assert func_ir
   803	    eq_sig = func_ir.decl.sig
   804	    strict_typing = builder.options.strict_dunders_typing
   805	    with builder.enter_method(cls, "__ne__", eq_sig.ret_type):
   806	        rhs_type = eq_sig.args[0].type if strict_typing else object_rprimitive
   807	        rhs_arg = builder.add_argument("rhs", rhs_type)
   808	        eqval = builder.add(MethodCall(builder.self(), "__eq__", [rhs_arg], line))
   809
   810	        can_return_not_implemented = is_subtype(not_implemented_op.type, eq_sig.ret_type)
   811	        return_bool = is_subtype(eq_sig.ret_type, bool_rprimitive)
   812
   813	        if not strict_typing or can_return_not_implemented:
   814	            # If __eq__ returns NotImplemented, then __ne__ should also
   815	            not_implemented_block, regular_block = BasicBlock(), BasicBlock()
   816	            not_implemented = builder.add(
   817	                LoadAddress(not_implemented_op.type, not_implemented_op.src, line)
   818	            )
   819	            builder.add(
   820	                Branch(
   821	                    builder.translate_is_op(eqval, not_implemented, "is", line),
   822	                    not_implemented_block,
   823	                    regular_block,
   824	                    Branch.BOOL,
   825	                )
   826	            )
   827	            builder.activate_block(regular_block)
   828	            rettype = bool_rprimitive if return_bool and strict_typing else object_rprimitive
   829	            retval = builder.coerce(builder.unary_op(eqval, "not", line), rettype, line)
   830	            builder.add(Return(retval))
   831	            builder.activate_block(not_implemented_block)
   832	            builder.add(Return(not_implemented))
   833	        else:
   834	            rettype = bool_rprimitive if return_bool and strict_typing else object_rprimitive
   835	            retval = builder.coerce(builder.unary_op(eqval, "not", line), rettype, line)
   836	            builder.add(Return(retval))

   889	def add_dunders_to_non_ext_dict(
   890	    builder: IRBuilder, non_ext: NonExtClassInfo, line: int, add_annotations: bool = True
   891	) -> None:
   892	    if add_annotations:
   893	        # Add __annotations__ to the class dict.

    32	# All possible result types of constant folding
    33	ConstantValue = Union[int, float, complex, str, bytes]
    34	CONST_TYPES: Final = (int, float, complex, str, bytes)
    35
    36
    37	def constant_fold_expr(builder: IRBuilder, expr: Expression) -> ConstantValue | None:
    38	    """Return the constant value of an expression for supported operations.
    39
    40	    Return None otherwise.
    41	    """
    42	    if isinstance(expr, IntExpr):
    43	        return expr.value
    44	    if isinstance(expr, FloatExpr):
    45	        return expr.value
    46	    if isinstance(expr, StrExpr):
    47	        return expr.value
    48	    if isinstance(expr, BytesExpr):
    49	        return bytes_from_str(expr.value)
    50	    if isinstance(expr, ComplexExpr):
    51	        return expr.value
    52	    elif isinstance(expr, NameExpr):
    53	        node = expr.node
    54	        if isinstance(node, Var) and node.is_final:
    55	            final_value = node.final_value
    56	            if isinstance(final_value, (CONST_TYPES)):
    57	                return final_value
    58	    elif isinstance(expr, MemberExpr):
    59	        final = builder.get_final_ref(expr)
    60	        if final is not None:
    61	            fn, final_var, native = final
    62	            if final_var.is_final:
    63	                final_value = final_var.final_value
    64	                if isinstance(final_value, (CONST_TYPES)):
    65	                    return final_value
    66	    elif isinstance(expr, OpExpr):
    67	        left = constant_fold_expr(builder, expr.left)
    68	        right = constant_fold_expr(builder, expr.right)
    69	        if left is not None and right is not None:
    70	            return constant_fold_binary_op_extended(expr.op, left, right)
    71	    elif isinstance(expr, UnaryExpr):
    72	        value = constant_fold_expr(builder, expr.expr)
    73	        if value is not None and not isinstance(value, bytes):
    74	            return constant_fold_unary_op(expr.op, value)
    75	    return None
    76
    77
    78	def constant_fold_binary_op_extended(
    79	    op: str, left: ConstantValue, right: ConstantValue
    80	) -> ConstantValue | None:
    81	    """Like mypy's constant_fold_binary_op(), but includes bytes support.
    82
    83	    mypy cannot use constant folded bytes easily so it's simpler to only support them in mypyc.
    84	    """
    85	    if not isinstance(left, bytes) and not isinstance(right, bytes):
    86	        return constant_fold_binary_op(op, left, right)
    87
    88	    if op == "+" and isinstance(left, bytes) and isinstance(right, bytes):
    89	        return left + right
    90	    elif op == "*" and isinstance(left, bytes) and isinstance(right, int):
    91	        return left * right
    92	    elif op == "*" and isinstance(left, int) and isinstance(right, bytes):
    93	        return left * right
    94
    95	    return None

   112	def transform_name_expr(builder: IRBuilder, expr: NameExpr) -> Value:
   113	    if isinstance(expr.node, TypeVarLikeExpr) and expr.node.is_new_style:
   114	        # Reference to Python 3.12 implicit TypeVar/TupleVarTuple/... object.
   115	        # These are stored in C statics and not visible in Python namespaces.
   116	        return builder.load_type_var(expr.node.name, expr.node.line)
   117	    if expr.node is None:
   118	        builder.add(
   119	            RaiseStandardError(
   120	                RaiseStandardError.RUNTIME_ERROR,
   121	                "mypyc internal error: should be unreachable",
   122	                expr.line,
   123	            )
   124	        )

   127	    if fullname in builtin_names:
   128	        typ, src = builtin_names[fullname]
   129	        return builder.add(LoadAddress(typ, src, expr.line))

   131	    if fullname == "builtins.None":

   134	        return builder.true()
   135	    if fullname == "builtins.False":
   136	        return builder.false()
   137	    if fullname in ("typing.TYPE_CHECKING", "typing_extensions.TYPE_CHECKING"):
   138	        return builder.false()

   140	    math_literal = transform_math_literal(builder, fullname)
   141	    if math_literal is not None:
   142	        return math_literal
   143
   144	    if isinstance(expr.node, Var) and expr.node.is_final:
   145	        value = builder.emit_load_final(
   146	            expr.node,

   148	            expr.name,
   149	            builder.is_native_ref_expr(expr),
   150	            builder.types[expr],

   154	            return value

   156	    if isinstance(expr.node, MypyFile) and expr.node.fullname in builder.imports:
   157	        return builder.load_module(expr.node.fullname)
   158
   159	    # If the expression is locally defined, then read the result from the corresponding
   160	    # assignment target and return it. Otherwise if the expression is a global, load it from
   161	    # the globals dictionary.
   162	    # Except for imports, that currently always happens in the global namespace.
   163	    if expr.kind == LDEF and not (isinstance(expr.node, Var) and expr.node.is_suppressed_import):
   164	        # Try to detect and error when we hit the irritating mypy bug
   165	        # where a local variable is cast to None. (#5423)
   166	        if (
   167	            isinstance(expr.node, Var)

   172	                'Local variable "{}" has inferred type None; add an annotation'.format(

   174	                ),
   175	                expr.node.line,
   176	            )

   178	        # TODO: Behavior currently only defined for Var, FuncDef and MypyFile node types.
   179	        if isinstance(expr.node, MypyFile):
   180	            # Load reference to a module imported inside function from
   181	            # the modules dictionary. It would be closer to Python
   182	            # semantics to access modules imported inside functions

   184	            # AST doesn't include a Var node for the module. We

   187	            obj = builder.primitive_op(
   188	                dict_get_item_op, [mod_dict, builder.load_str(expr.node.fullname)], expr.line
   189	            )
   190	            return obj
   191	        else:
   192	            return builder.read(builder.get_assignment_target(expr, for_read=True), expr.line)
   193
   194	    return builder.load_global(expr)

   348	def translate_method_call(builder: IRBuilder, expr: CallExpr, callee: MemberExpr) -> Value:
   349	    """Generate IR for an arbitrary call of form e.m(...).
   350
   351	    This can also deal with calls to module-level functions.
   352	    """
   353	    if builder.is_native_ref_expr(callee):
   354	        # Call to module-level native function or such
   355	        return translate_call(builder, expr, callee)
   356	    elif (
   357	        isinstance(callee.expr, RefExpr)
   358	        and isinstance(callee.expr.node, TypeInfo)
   359	        and callee.expr.node in builder.mapper.type_to_ir
   360	        and builder.mapper.type_to_ir[callee.expr.node].has_method(callee.name)
   361	        and all(kind in (ARG_POS, ARG_NAMED) for kind in expr.arg_kinds)
   362	    ):

   364	        assert isinstance(callee.expr.node, TypeInfo)
   365	        ir = builder.mapper.type_to_ir[callee.expr.node]
   366	        return call_classmethod(builder, ir, expr, callee)

   368	        # Fall back to a PyCall for non-native module calls
   369	        function = builder.accept(callee)
   370	        args = [builder.accept(arg) for arg in expr.args]
   371	        return builder.py_call(
   372	            function, args, expr.line, arg_kinds=expr.arg_kinds, arg_names=expr.arg_names
   373	        )
   374	    else:
   375	        if isinstance(callee.expr, RefExpr):

   377	            if isinstance(node, Var) and node.is_cls:
   378	                typ = get_proper_type(node.type)
   379	                if isinstance(typ, TypeType) and isinstance(typ.item, Instance):

   381	                    if class_ir and class_ir.is_ext_class and class_ir.has_no_subclasses():
   382	                        # Call a native classmethod via cls that can be statically bound,
   383	                        # since the class has no subclasses.
   384	                        return call_classmethod(builder, class_ir, expr, callee)

   386	        receiver_typ = builder.node_type(callee.expr)
   387
   388	        # If there is a specializer for this method name/type, try calling it.
   389	        # We would return the first successful one.
   390	        val = apply_method_specialization(builder, expr, callee, receiver_typ)
   391	        if val is not None:

   394	        obj = builder.accept(callee.expr)

   402	            expr.arg_kinds,
   403	            expr.arg_names,

   433	def translate_super_method_call(builder: IRBuilder, expr: CallExpr, callee: SuperExpr) -> Value:
   434	    if callee.info is None or (len(callee.call.args) != 0 and len(callee.call.args) != 2):
   435	        return translate_call(builder, expr, callee)
   436
   437	    # We support two-argument super but only when it is super(CurrentClass, self)
   438	    # TODO: We could support it when it is a parent class in many cases?
   439	    if len(callee.call.args) == 2:
   440	        self_arg = callee.call.args[1]
   441	        if (
   442	            not isinstance(self_arg, NameExpr)
   443	            or not isinstance(self_arg.node, Var)
   444	            or not self_arg.node.is_self
   445	        ):
   446	            return translate_call(builder, expr, callee)
   447
   448	        typ_arg = callee.call.args[0]
   449	        if (
   450	            not isinstance(typ_arg, NameExpr)
   451	            or not isinstance(typ_arg.node, TypeInfo)
   452	            or callee.info is not typ_arg.node
   453	        ):
   454	            return translate_call(builder, expr, callee)
   455
   456	    ir = builder.mapper.type_to_ir[callee.info]
   457	    # Search for the method in the mro, skipping ourselves. We
   458	    # determine targets of super calls to native methods statically.
   459	    for base in ir.mro[1:]:
   460	        if callee.name in base.method_decls:
   461	            break
   462	    else:
   463	        if (
   464	            ir.is_ext_class
   465	            and ir.builtin_base is None
   466	            and not ir.inherits_python
   467	            and callee.name == "__init__"
   468	            and len(expr.args) == 0
   469	        ):
   470	            # Call translates to object.__init__(self), which is a
   471	            # no-op, so omit the call.
   472	            return builder.none()
   473	        return translate_call(builder, expr, callee)
   474
   475	    decl = base.method_decl(callee.name)
   476	    arg_values = [builder.accept(arg) for arg in expr.args]
   477	    arg_kinds, arg_names = expr.arg_kinds.copy(), expr.arg_names.copy()
   478
   479	    if decl.kind != FUNC_STATICMETHOD:
   480	        # Grab first argument
   481	        vself: Value = builder.self()
   482	        if decl.kind == FUNC_CLASSMETHOD:
   483	            vself = builder.primitive_op(type_op, [vself], expr.line)
   484	        elif builder.fn_info.is_generator:
   485	            # For generator classes, the self target is the 6th value
   486	            # in the symbol table (which is an ordered dict). This is sort
   487	            # of ugly, but we can't search by name since the 'self' parameter
   488	            # could be named anything, and it doesn't get added to the
   489	            # environment indexes.
   490	            self_targ = list(builder.symtables[-1].values())[6]
   491	            vself = builder.read(self_targ, builder.fn_info.fitem.line)
   492	        arg_values.insert(0, vself)
   493	        arg_kinds.insert(0, ARG_POS)
   494	        arg_names.insert(0, None)
   495
   496	    return builder.builder.call(decl, arg_values, arg_kinds, arg_names, expr.line)

   516	def transform_op_expr(builder: IRBuilder, expr: OpExpr) -> Value:
   517	    if expr.op in ("and", "or"):
   518	        return builder.shortcircuit_expr(expr)
   519
   520	    # Special case for string formatting
   521	    if expr.op == "%" and isinstance(expr.left, (StrExpr, BytesExpr)):
   522	        ret = translate_printf_style_formatting(builder, expr.left, expr.right)
   523	        if ret is not None:
   524	            return ret
   525
   526	    folded = try_constant_fold(builder, expr)
   527	    if folded:
   528	        return folded
   529
   530	    borrow_left = False
   531	    borrow_right = False
   532
   533	    ltype = builder.node_type(expr.left)
   534	    rtype = builder.node_type(expr.right)
   535
   536	    # Special case some int ops to allow borrowing operands.
   537	    if is_int_rprimitive(ltype) and is_int_rprimitive(rtype):
   538	        if expr.op == "//":
   539	            expr = try_optimize_int_floor_divide(expr)
   540	        if expr.op in int_borrow_friendly_op:
   541	            borrow_left = is_borrow_friendly_expr(builder, expr.right)
   542	            borrow_right = True
   543	    elif is_fixed_width_rtype(ltype) and is_fixed_width_rtype(rtype):
   544	        borrow_left = is_borrow_friendly_expr(builder, expr.right)
   545	        borrow_right = True
   546
   547	    left = builder.accept(expr.left, can_borrow=borrow_left)
   548	    right = builder.accept(expr.right, can_borrow=borrow_right)
   549	    return builder.binary_op(left, right, expr.op, expr.line)
   550

   552	def try_optimize_int_floor_divide(expr: OpExpr) -> OpExpr:
   553	    """Replace // with a power of two with a right shift, if possible."""
   554	    if not isinstance(expr.right, IntExpr):
   555	        return expr
   556	    divisor = expr.right.value
   557	    shift = divisor.bit_length() - 1
   558	    if 0 < shift < 28 and divisor == (1 << shift):
   559	        return OpExpr(">>", expr.left, IntExpr(shift))
   560	    return expr

   596	def try_gen_slice_op(builder: IRBuilder, base: Value, index: SliceExpr) -> Value | None:
   597	    """Generate specialized slice op for some index expressions.
   598
   599	    Return None if a specialized op isn't available.
   600
   601	    This supports obj[x:y], obj[:x], and obj[x:] for a few types.
   602	    """
   603	    if index.stride:
   604	        # We can only handle the default stride of 1.
   605	        return None
   606
   607	    if index.begin_index:
   608	        begin_type = builder.node_type(index.begin_index)
   609	    else:
   610	        begin_type = int_rprimitive
   611	    if index.end_index:
   612	        end_type = builder.node_type(index.end_index)
   613	    else:
   614	        end_type = int_rprimitive
   615
   616	    # Both begin and end index must be int (or missing).
   617	    if is_int_rprimitive(begin_type) and is_int_rprimitive(end_type):
   618	        if index.begin_index:
   619	            begin = builder.accept(index.begin_index)
   620	        else:
   621	            begin = builder.load_int(0)
   622	        if index.end_index:
   623	            end = builder.accept(index.end_index)
   624	        else:
   625	            # Replace missing end index with the largest short integer
   626	            # (a sequence can't be longer).
   627	            end = builder.load_int(MAX_SHORT_INT)
   628	        candidates = [list_slice_op, tuple_slice_op, str_slice_op, bytes_slice_op]
   629	        return builder.builder.matching_call_c(candidates, [base, begin, end], index.line)
   630
   631	    return None

   634	def transform_conditional_expr(builder: IRBuilder, expr: ConditionalExpr) -> Value:
   635	    if_body, else_body, next_block = BasicBlock(), BasicBlock(), BasicBlock()
   636
   637	    process_conditional(builder, expr.cond, if_body, else_body)
   638	    expr_type = builder.node_type(expr)
   639	    # Having actual Phi nodes would be really nice here!
   640	    target = Register(expr_type)
   641
   642	    builder.activate_block(if_body)
   643	    true_value = builder.accept(expr.if_expr)
   644	    true_value = builder.coerce(true_value, expr_type, expr.line)
   645	    builder.add(Assign(target, true_value))
   646	    builder.goto(next_block)
   647
   648	    builder.activate_block(else_body)
   649	    false_value = builder.accept(expr.else_expr)
   650	    false_value = builder.coerce(false_value, expr_type, expr.line)
   651	    builder.add(Assign(target, false_value))
   652	    builder.goto(next_block)
   653
   654	    builder.activate_block(next_block)
   655
   656	    return target

   659	def set_literal_values(builder: IRBuilder, items: Sequence[Expression]) -> list[object] | None:
   660	    values: list[object] = []
   661	    for item in items:
   662	        const_value = constant_fold_expr(builder, item)
   663	        if const_value is not None:
   664	            values.append(const_value)
   665	            continue
   666
   667	        if isinstance(item, RefExpr):
   668	            if item.fullname == "builtins.None":
   669	                values.append(None)
   670	            elif item.fullname == "builtins.True":
   671	                values.append(True)
   672	            elif item.fullname == "builtins.False":
   673	                values.append(False)
   674	        elif isinstance(item, TupleExpr):
   675	            tuple_values = set_literal_values(builder, item.items)
   676	            if tuple_values is not None:
   677	                values.append(tuple(tuple_values))
   678
   679	    if len(values) != len(items):
   680	        # Bail if not all items can be converted into values.
   681	        return None
   682	    return values

   702	def transform_comparison_expr(builder: IRBuilder, e: ComparisonExpr) -> Value:
   703	    # x in (...)/[...]
   704	    # x not in (...)/[...]
   705	    first_op = e.operators[0]
   706	    if (
   707	        first_op in ["in", "not in"]
   708	        and len(e.operators) == 1
   709	        and isinstance(e.operands[1], (TupleExpr, ListExpr))
   710	    ):
   711	        items = e.operands[1].items
   712	        n_items = len(items)
   713	        # x in y -> x == y[0] or ... or x == y[n]
   714	        # x not in y -> x != y[0] and ... and x != y[n]
   715	        # 16 is arbitrarily chosen to limit code size
   716	        if 1 < n_items < 16:
   717	            if e.operators[0] == "in":
   718	                bin_op = "or"
   719	                cmp_op = "=="
   720	            else:
   721	                bin_op = "and"
   722	                cmp_op = "!="
   723	            lhs = e.operands[0]
   724	            mypy_file = builder.graph["builtins"].tree
   725	            assert mypy_file is not None
   726	            info = mypy_file.names["bool"].node
   727	            assert isinstance(info, TypeInfo)
   728	            bool_type = Instance(info, [])
   729	            exprs = []
   730	            for item in items:
   731	                expr = ComparisonExpr([cmp_op], [lhs, item])
   732	                builder.types[expr] = bool_type
   733	                exprs.append(expr)
   734
   735	            or_expr: Expression = exprs.pop(0)
   736	            for expr in exprs:
   737	                or_expr = OpExpr(bin_op, or_expr, expr)
   738	                builder.types[or_expr] = bool_type
   739	            return builder.accept(or_expr)
   740	        # x in [y]/(y) -> x == y
   741	        # x not in [y]/(y) -> x != y
   742	        elif n_items == 1:
   743	            if e.operators[0] == "in":
   744	                cmp_op = "=="
   745	            else:
   746	                cmp_op = "!="
   747	            e.operators = [cmp_op]
   748	            e.operands[1] = items[0]
   749	        # x in []/() -> False
   750	        # x not in []/() -> True
   751	        elif n_items == 0:
   752	            if e.operators[0] == "in":
   753	                return builder.false()
   754	            else:
   755	                return builder.true()
   756
   757	    # x in {...}
   758	    # x not in {...}
   759	    if (
   760	        first_op in ("in", "not in")
   761	        and len(e.operators) == 1
   762	        and isinstance(e.operands[1], SetExpr)
   763	    ):
   764	        set_literal = precompute_set_literal(builder, e.operands[1])
   765	        if set_literal is not None:
   766	            lhs = e.operands[0]
   767	            result = builder.builder.primitive_op(
   768	                set_in_op, [builder.accept(lhs), set_literal], e.line, bool_rprimitive
   769	            )
   770	            if first_op == "not in":
   771	                return builder.unary_op(result, "not", e.line)
   772	            return result
   773
   774	    if len(e.operators) == 1:
   775	        # Special some common simple cases
   776	        if first_op in ("is", "is not"):
   777	            right_expr = e.operands[1]
   778	            if isinstance(right_expr, NameExpr) and right_expr.fullname == "builtins.None":
   779	                # Special case 'is None' / 'is not None'.
   780	                return translate_is_none(builder, e.operands[0], negated=first_op != "is")
   781	        left_expr = e.operands[0]
   782	        if is_int_rprimitive(builder.node_type(left_expr)):
   783	            right_expr = e.operands[1]
   784	            if is_int_rprimitive(builder.node_type(right_expr)):
   785	                if first_op in int_borrow_friendly_op:
   786	                    borrow_left = is_borrow_friendly_expr(builder, right_expr)
   787	                    left = builder.accept(left_expr, can_borrow=borrow_left)
   788	                    right = builder.accept(right_expr, can_borrow=True)
   789	                    return builder.binary_op(left, right, first_op, e.line)
   790
   791	    # TODO: Don't produce an expression when used in conditional context
   792	    # All of the trickiness here is due to support for chained conditionals
   793	    # (`e1 < e2 > e3`, etc). `e1 < e2 > e3` is approximately equivalent to
   794	    # `e1 < e2 and e2 > e3` except that `e2` is only evaluated once.
   795	    expr_type = builder.node_type(e)
   796
   797	    # go(i, prev) generates code for `ei opi e{i+1} op{i+1} ... en`,
   798	    # assuming that prev contains the value of `ei`.
   799	    def go(i: int, prev: Value) -> Value:
   800	        if i == len(e.operators) - 1:
   801	            return transform_basic_comparison(
   802	                builder, e.operators[i], prev, builder.accept(e.operands[i + 1]), e.line
   803	            )
   804
   805	        next = builder.accept(e.operands[i + 1])
   806	        return builder.builder.shortcircuit_helper(
   807	            "and",
   808	            expr_type,
   809	            lambda: transform_basic_comparison(builder, e.operators[i], prev, next, e.line),
   810	            lambda: go(i + 1, next),
   811	            e.line,
   812	        )
   813
   814	    return go(0, builder.accept(e.operands[0]))
   815

   822	def transform_basic_comparison(
   823	    builder: IRBuilder, op: str, left: Value, right: Value, line: int
   824	) -> Value:
   825	    if is_fixed_width_rtype(left.type) and op in ComparisonOp.signed_ops:
   826	        if right.type == left.type:
   827	            if left.type.is_signed:
   828	                op_id = ComparisonOp.signed_ops[op]
   829	            else:
   830	                op_id = ComparisonOp.unsigned_ops[op]
   831	            return builder.builder.comparison_op(left, right, op_id, line)
   832	        elif isinstance(right, Integer):
   833	            if left.type.is_signed:
   834	                op_id = ComparisonOp.signed_ops[op]
   835	            else:
   836	                op_id = ComparisonOp.unsigned_ops[op]
   837	            return builder.builder.comparison_op(
   838	                left, builder.coerce(right, left.type, line), op_id, line
   839	            )
   840	    elif (
   841	        is_fixed_width_rtype(right.type)
   842	        and op in ComparisonOp.signed_ops
   843	        and isinstance(left, Integer)
   844	    ):
   845	        if right.type.is_signed:
   846	            op_id = ComparisonOp.signed_ops[op]
   847	        else:
   848	            op_id = ComparisonOp.unsigned_ops[op]
   849	        return builder.builder.comparison_op(
   850	            builder.coerce(left, right.type, line), right, op_id, line
   851	        )
   852
   853	    negate = False
   854	    if op == "is not":
   855	        op, negate = "is", True
   856	    elif op == "not in":
   857	        op, negate = "in", True
   858
   859	    target = builder.binary_op(left, right, op, line)
   860
   861	    if negate:
   862	        target = builder.unary_op(target, "not", line)
   863	    return target

   866	def translate_printf_style_formatting(
   867	    builder: IRBuilder, format_expr: StrExpr | BytesExpr, rhs: Expression
   868	) -> Value | None:
   869	    tokens = tokenizer_printf_style(format_expr.value)
   870	    if tokens is not None:
   871	        literals, format_ops = tokens
   872
   873	        exprs = []
   874	        if isinstance(rhs, TupleExpr):
   875	            exprs = rhs.items
   876	        elif isinstance(rhs, Expression):
   877	            exprs.append(rhs)
   878
   879	        if isinstance(format_expr, BytesExpr):
   880	            substitutions = convert_format_expr_to_bytes(
   881	                builder, format_ops, exprs, format_expr.line
   882	            )
   883	            if substitutions is not None:
   884	                return join_formatted_bytes(builder, literals, substitutions, format_expr.line)
   885	        else:
   886	            substitutions = convert_format_expr_to_str(
   887	                builder, format_ops, exprs, format_expr.line
   888	            )
   889	            if substitutions is not None:
   890	                return join_formatted_strings(builder, literals, substitutions, format_expr.line)
   891
   892	    return None

   980	def _visit_display(
   981	    builder: IRBuilder,
   982	    items: list[Expression],
   983	    constructor_op: Callable[[list[Value], int], Value],
   984	    append_op: PrimitiveDescription,
   985	    extend_op: PrimitiveDescription,
   986	    line: int,
   987	    is_list: bool,
   988	) -> Value:
   989	    accepted_items = []
   990	    for item in items:
   991	        if isinstance(item, StarExpr):
   992	            accepted_items.append((True, builder.accept(item.expr)))
   993	        else:
   994	            accepted_items.append((False, builder.accept(item)))
   995
   996	    result: Value | None = None
   997	    initial_items = []
   998	    for starred, value in accepted_items:
   999	        if result is None and not starred and is_list:
  1000	            initial_items.append(value)
  1001	            continue
  1002
  1003	        if result is None:
  1004	            result = constructor_op(initial_items, line)
  1005
  1006	        builder.primitive_op(extend_op if starred else append_op, [result, value], line)
  1007
  1008	    if result is None:
  1009	        result = constructor_op(initial_items, line)
  1010
  1011	    return result
