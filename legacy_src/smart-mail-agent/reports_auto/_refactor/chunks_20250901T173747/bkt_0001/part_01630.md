   376	def find_dataclass_transform_spec(node: Node | None) -> DataclassTransformSpec | None:
   377	    """
   378	    Find the dataclass transform spec for the given node, if any exists.
   379
   380	    Per PEP 681 (https://peps.python.org/pep-0681/#the-dataclass-transform-decorator), dataclass
   381	    transforms can be specified in multiple ways, including decorator functions and
   382	    metaclasses/base classes. This function resolves the spec from any of these variants.
   383	    """
   384
   385	    # The spec only lives on the function/class definition itself, so we need to unwrap down to that
   386	    # point
   387	    if isinstance(node, CallExpr):
   388	        # Like dataclasses.dataclass, transform-based decorators can be applied either with or
   389	        # without parameters; ie, both of these forms are accepted:
   390	        #
   391	        # @typing.dataclass_transform
   392	        # class Foo: ...
   393	        # @typing.dataclass_transform(eq=True, order=True, ...)
   394	        # class Bar: ...
   395	        #
   396	        # We need to unwrap the call for the second variant.
   397	        node = node.callee
   398
   399	    if isinstance(node, RefExpr):
   400	        node = node.node
   401
   402	    if isinstance(node, Decorator):
   403	        # typing.dataclass_transform usage must always result in a Decorator; it always uses the
   404	        # `@dataclass_transform(...)` syntax and never `@dataclass_transform`
   405	        node = node.func
   406
   407	    if isinstance(node, OverloadedFuncDef):
   408	        # The dataclass_transform decorator may be attached to any single overload, so we must
   409	        # search them all.
   410	        # Note that using more than one decorator is undefined behavior, so we can just take the
   411	        # first that we find.
   412	        for candidate in node.items:
   413	            spec = find_dataclass_transform_spec(candidate)
   414	            if spec is not None:
   415	                return spec
   416	        return find_dataclass_transform_spec(node.impl)
   417
   418	    # For functions, we can directly consult the AST field for the spec
   419	    if isinstance(node, FuncDef):
   420	        return node.dataclass_transform_spec
   421
   422	    if isinstance(node, ClassDef):
   423	        node = node.info
   424	    if isinstance(node, TypeInfo):
   425	        # Search all parent classes to see if any are decorated with `typing.dataclass_transform`
   426	        for base in node.mro[1:]:
   427	            if base.dataclass_transform_spec is not None:
   428	                return base.dataclass_transform_spec
   429
   430	        # Check if there is a metaclass that is decorated with `typing.dataclass_transform`
   431	        #
   432	        # Note that PEP 681 only discusses using a metaclass that is directly decorated with
   433	        # `typing.dataclass_transform`; subclasses thereof should be treated with dataclass
   434	        # semantics rather than as transforms:
   435	        #
   436	        # > If dataclass_transform is applied to a class, dataclass-like semantics will be assumed
   437	        # > for any class that directly or indirectly derives from the decorated class or uses the
   438	        # > decorated class as a metaclass.
   439	        #
   440	        # The wording doesn't make this entirely explicit, but Pyright (the reference
   441	        # implementation for this PEP) only handles directly-decorated metaclasses.
   442	        metaclass_type = node.metaclass_type
   443	        if metaclass_type is not None and metaclass_type.type.dataclass_transform_spec is not None:
   444	            return metaclass_type.type.dataclass_transform_spec
   445
   446	    return None

    91	            if isinstance(base_expr, RefExpr):
    92	                self.api.accept(base_expr)
    93	                if base_expr.fullname in TPDICT_NAMES or self.is_typeddict(base_expr):
    94	                    possible = True
    95	                    if isinstance(base_expr.node, TypeInfo) and base_expr.node.is_final:
    96	                        err = message_registry.CANNOT_INHERIT_FROM_FINAL
    97	                        self.fail(err.format(base_expr.node.name).value, defn, code=err.code)
    98	        if not possible:
    99	            return False, None
   100	        existing_info = None
   101	        if isinstance(defn.analyzed, TypedDictExpr):
   102	            existing_info = defn.analyzed.info
   103
   104	        field_types: dict[str, Type] | None
   105	        if (
   106	            len(defn.base_type_exprs) == 1
   107	            and isinstance(defn.base_type_exprs[0], RefExpr)
   108	            and defn.base_type_exprs[0].fullname in TPDICT_NAMES
   109	        ):
   110	            # Building a new TypedDict
   111	            field_types, statements, required_keys, readonly_keys = (
   112	                self.analyze_typeddict_classdef_fields(defn)
   113	            )
   114	            if field_types is None:
   115	                return True, None  # Defer
   116	            if self.api.is_func_scope() and "@" not in defn.name:
   117	                defn.name += "@" + str(defn.line)
   118	            info = self.build_typeddict_typeinfo(
   119	                defn.name, field_types, required_keys, readonly_keys, defn.line, existing_info
   120	            )
   121	            defn.analyzed = TypedDictExpr(info)
   122	            defn.analyzed.line = defn.line
   123	            defn.analyzed.column = defn.column
   124	            defn.defs.body = statements
   125	            return True, info
   126
   127	        # Extending/merging existing TypedDicts
   128	        typeddict_bases: list[Expression] = []
   129	        typeddict_bases_set = set()
   130	        for expr in defn.base_type_exprs:
   131	            ok, maybe_type_info, _ = self.check_typeddict(expr, None, False)
   132	            if ok and maybe_type_info is not None:
   133	                # expr is a CallExpr
   134	                info = maybe_type_info
   135	                typeddict_bases_set.add(info.fullname)
   136	                typeddict_bases.append(expr)
   137	            elif isinstance(expr, RefExpr) and expr.fullname in TPDICT_NAMES:
   138	                if "TypedDict" not in typeddict_bases_set:
   139	                    typeddict_bases_set.add("TypedDict")
   140	                else:
   141	                    self.fail('Duplicate base class "TypedDict"', defn)
   142	            elif (
   143	                isinstance(expr, RefExpr)
   144	                and self.is_typeddict(expr)
   145	                or isinstance(expr, IndexExpr)
   146	                and self.is_typeddict(expr.base)
   147	            ):
   148	                info = self._parse_typeddict_base(expr, defn)
   149	                if info.fullname not in typeddict_bases_set:
   150	                    typeddict_bases_set.add(info.fullname)
   151	                    typeddict_bases.append(expr)
   152	                else:
   153	                    self.fail(f'Duplicate base class "{info.name}"', defn)
   154	            else:
   155	                self.fail("All bases of a new TypedDict must be TypedDict types", defn)
   156
   157	        field_types = {}
   158	        required_keys = set()
   159	        readonly_keys = set()
   160	        # Iterate over bases in reverse order so that leftmost base class' keys take precedence
   161	        for base in reversed(typeddict_bases):
   162	            self.add_keys_and_types_from_base(
   163	                base, field_types, required_keys, readonly_keys, defn
   164	            )
   165	        (new_field_types, new_statements, new_required_keys, new_readonly_keys) = (
   166	            self.analyze_typeddict_classdef_fields(defn, oldfields=field_types)
   167	        )
   168	        if new_field_types is None:
   169	            return True, None  # Defer
   170	        field_types.update(new_field_types)
   171	        required_keys.update(new_required_keys)
   172	        readonly_keys.update(new_readonly_keys)
   173	        info = self.build_typeddict_typeinfo(
   174	            defn.name, field_types, required_keys, readonly_keys, defn.line, existing_info
   175	        )
   176	        defn.analyzed = TypedDictExpr(info)
   177	        defn.analyzed.line = defn.line
   178	        defn.analyzed.column = defn.column
   179	        defn.defs.body = new_statements
   180	        return True, info

   182	    def add_keys_and_types_from_base(
   183	        self,
   184	        base: Expression,
   185	        field_types: dict[str, Type],
   186	        required_keys: set[str],
   187	        readonly_keys: set[str],
   188	        ctx: Context,
   189	    ) -> None:
   190	        info = self._parse_typeddict_base(base, ctx)
   191	        base_args: list[Type] = []
   192	        if isinstance(base, IndexExpr):
   193	            args = self.analyze_base_args(base, ctx)
   194	            if args is None:
   195	                return
   196	            base_args = args
   197
   198	        assert info.typeddict_type is not None
   199	        base_typed_dict = info.typeddict_type
   200	        base_items = base_typed_dict.items
   201	        valid_items = base_items.copy()
   202
   203	        # Always fix invalid bases to avoid crashes.
   204	        tvars = info.defn.type_vars
   205	        if len(base_args) != len(tvars):
   206	            any_kind = TypeOfAny.from_omitted_generics
   207	            if base_args:
   208	                self.fail(f'Invalid number of type arguments for "{info.name}"', ctx)
   209	                any_kind = TypeOfAny.from_error
   210	            base_args = [AnyType(any_kind) for _ in tvars]
   211
   212	        with state.strict_optional_set(self.options.strict_optional):
   213	            valid_items = self.map_items_to_base(valid_items, tvars, base_args)
   214	        for key in base_items:
   215	            if key in field_types:
   216	                self.fail(TYPEDDICT_OVERRIDE_MERGE.format(key), ctx)
   217
   218	        field_types.update(valid_items)
   219	        required_keys.update(base_typed_dict.required_keys)
   220	        readonly_keys.update(base_typed_dict.readonly_keys)

   222	    def _parse_typeddict_base(self, base: Expression, ctx: Context) -> TypeInfo:
   223	        if isinstance(base, RefExpr):
   224	            if isinstance(base.node, TypeInfo):
   225	                return base.node
   226	            elif isinstance(base.node, TypeAlias):
   227	                # Only old TypeAlias / plain assignment, PEP695 `type` stmt
   228	                # cannot be used as a base class
   229	                target = get_proper_type(base.node.target)
   230	                assert isinstance(target, TypedDictType)
   231	                return target.fallback.type
   232	            else:
   233	                assert False
   234	        elif isinstance(base, IndexExpr):
   235	            assert isinstance(base.base, RefExpr)
   236	            return self._parse_typeddict_base(base.base, ctx)
   237	        else:
   238	            assert isinstance(base, CallExpr)
   239	            assert isinstance(base.analyzed, TypedDictExpr)
   240	            return base.analyzed.info

   242	    def analyze_base_args(self, base: IndexExpr, ctx: Context) -> list[Type] | None:
   243	        """Analyze arguments of base type expressions as types.
   244
   245	        We need to do this, because normal base class processing happens after
   246	        the TypedDict special-casing (plus we get a custom error message).
   247	        """
   248	        base_args = []
   249	        if isinstance(base.index, TupleExpr):
   250	            args = base.index.items
   251	        else:
   252	            args = [base.index]
   253
   254	        for arg_expr in args:
   255	            try:
   256	                type = expr_to_unanalyzed_type(arg_expr, self.options, self.api.is_stub_file)
   257	            except TypeTranslationError:
   258	                self.fail("Invalid TypedDict type argument", ctx)
   259	                return None
   260	            analyzed = self.api.anal_type(
   261	                type,
   262	                allow_typed_dict_special_forms=True,
   263	                allow_placeholder=not self.api.is_func_scope(),
   264	            )
   265	            if analyzed is None:
   266	                return None
   267	            base_args.append(analyzed)
   268	        return base_args

   270	    def map_items_to_base(
   271	        self, valid_items: dict[str, Type], tvars: list[TypeVarLikeType], base_args: list[Type]
   272	    ) -> dict[str, Type]:
   273	        """Map item types to how they would look in their base with type arguments applied.
   274
   275	        Note it is safe to use expand_type() during semantic analysis, because it should never
   276	        (indirectly) call is_subtype().
   277	        """
   278	        mapped_items = {}
   279	        for key in valid_items:
   280	            type_in_base = valid_items[key]
   281	            if not tvars:
   282	                mapped_items[key] = type_in_base
   283	                continue
   284	            # TODO: simple zip can't be used for variadic types.
   285	            mapped_items[key] = expand_type(
   286	                type_in_base, {t.id: a for (t, a) in zip(tvars, base_args)}
   287	            )
   288	        return mapped_items

   290	    def analyze_typeddict_classdef_fields(
   291	        self, defn: ClassDef, oldfields: Collection[str] | None = None
   292	    ) -> tuple[dict[str, Type] | None, list[Statement], set[str], set[str]]:
   293	        """Analyze fields defined in a TypedDict class definition.
   294
   295	        This doesn't consider inherited fields (if any). Also consider totality,
   296	        if given.
   297
   298	        Return tuple with these items:
   299	         * Dict of key -> type (or None if found an incomplete reference -> deferral)
   300	         * List of statements from defn.defs.body that are legally allowed to be a
   301	           part of a TypedDict definition
   302	         * Set of required keys
   303	        """
   304	        fields: dict[str, Type] = {}
   305	        readonly_keys = set[str]()
   306	        required_keys = set[str]()
   307	        statements: list[Statement] = []
   308
   309	        total: bool | None = True
   310	        for key in defn.keywords:
   311	            if key == "total":
   312	                total = require_bool_literal_argument(
   313	                    self.api, defn.keywords["total"], "total", True
   314	                )
   315	                continue
   316	            for_function = ' for "__init_subclass__" of "TypedDict"'
   317	            self.msg.unexpected_keyword_argument_for_function(for_function, key, defn)
   318
   319	        for stmt in defn.defs.body:
   320	            if not isinstance(stmt, AssignmentStmt):
   321	                # Still allow pass or ... (for empty TypedDict's) and docstrings
   322	                if isinstance(stmt, PassStmt) or (
   323	                    isinstance(stmt, ExpressionStmt)
   324	                    and isinstance(stmt.expr, (EllipsisExpr, StrExpr))
   325	                ):
   326	                    statements.append(stmt)
   327	                else:
   328	                    defn.removed_statements.append(stmt)
   329	                    self.fail(TPDICT_CLASS_ERROR, stmt)
   330	            elif len(stmt.lvalues) > 1 or not isinstance(stmt.lvalues[0], NameExpr):
   331	                # An assignment, but an invalid one.
   332	                defn.removed_statements.append(stmt)
   333	                self.fail(TPDICT_CLASS_ERROR, stmt)
   334	            else:
   335	                name = stmt.lvalues[0].name
   336	                if name in (oldfields or []):
   337	                    self.fail(f'Overwriting TypedDict field "{name}" while extending', stmt)
   338	                if name in fields:
   339	                    self.fail(f'Duplicate TypedDict key "{name}"', stmt)
   340	                    continue
   341	                # Append stmt, name, and type in this case...
   342	                statements.append(stmt)
   343
   344	                field_type: Type
   345	                if stmt.unanalyzed_type is None:
   346	                    field_type = AnyType(TypeOfAny.unannotated)
   347	                else:
   348	                    analyzed = self.api.anal_type(
   349	                        stmt.unanalyzed_type,
   350	                        allow_typed_dict_special_forms=True,
   351	                        allow_placeholder=not self.api.is_func_scope(),
   352	                        prohibit_self_type="TypedDict item type",
   353	                        prohibit_special_class_field_types="TypedDict",
   354	                    )
   355	                    if analyzed is None:
   356	                        return None, [], set(), set()  # Need to defer
   357	                    field_type = analyzed
   358	                    if not has_placeholder(analyzed):
   359	                        stmt.type = self.extract_meta_info(analyzed, stmt)[0]
   360
   361	                field_type, required, readonly = self.extract_meta_info(field_type)
   362	                fields[name] = field_type
   363
   364	                if (total or required is True) and required is not False:
   365	                    required_keys.add(name)
   366	                if readonly:
   367	                    readonly_keys.add(name)
   368
   369	                # ...despite possible minor failures that allow further analysis.
   370	                if stmt.type is None or hasattr(stmt, "new_syntax") and not stmt.new_syntax:
   371	                    self.fail(TPDICT_CLASS_ERROR, stmt)
   372	                elif not isinstance(stmt.rvalue, TempNode):
   373	                    # x: int assigns rvalue to TempNode(AnyType())
   374	                    self.fail("Right hand side values are not supported in TypedDict", stmt)
   375
   376	        return fields, statements, required_keys, readonly_keys

   378	    def extract_meta_info(
   379	        self, typ: Type, context: Context | None = None
   380	    ) -> tuple[Type, bool | None, bool]:
   381	        """Unwrap all metadata types."""
   382	        is_required = None  # default, no modification
   383	        readonly = False  # by default all is mutable
   384
   385	        seen_required = False
   386	        seen_readonly = False
   387	        while isinstance(typ, (RequiredType, ReadOnlyType)):
   388	            if isinstance(typ, RequiredType):
   389	                if context is not None and seen_required:
   390	                    self.fail(
   391	                        '"{}" type cannot be nested'.format(
   392	                            "Required[]" if typ.required else "NotRequired[]"
   393	                        ),
   394	                        context,
   395	                        code=codes.VALID_TYPE,
   396	                    )
   397	                is_required = typ.required
   398	                seen_required = True
   399	                typ = typ.item
   400	            if isinstance(typ, ReadOnlyType):
   401	                if context is not None and seen_readonly:
   402	                    self.fail('"ReadOnly[]" type cannot be nested', context, code=codes.VALID_TYPE)
   403	                readonly = True
   404	                seen_readonly = True
   405	                typ = typ.item
   406	        return typ, is_required, readonly

   408	    def check_typeddict(
   409	        self, node: Expression, var_name: str | None, is_func_scope: bool
   410	    ) -> tuple[bool, TypeInfo | None, list[TypeVarLikeType]]:
   411	        """Check if a call defines a TypedDict.
   412
   413	        The optional var_name argument is the name of the variable to
   414	        which this is assigned, if any.
   415
   416	        Return a pair (is it a typed dict, corresponding TypeInfo).
   417
   418	        If the definition is invalid but looks like a TypedDict,
   419	        report errors but return (some) TypeInfo. If some type is not ready,
   420	        return (True, None).
   421	        """
   422	        if not isinstance(node, CallExpr):
   423	            return False, None, []
   424	        call = node
   425	        callee = call.callee
   426	        if not isinstance(callee, RefExpr):
   427	            return False, None, []
   428	        fullname = callee.fullname
   429	        if fullname not in TPDICT_NAMES:
   430	            return False, None, []
   431	        res = self.parse_typeddict_args(call)
   432	        if res is None:
   433	            # This is a valid typed dict, but some type is not ready.
   434	            # The caller should defer this until next iteration.
   435	            return True, None, []
   436	        name, items, types, total, tvar_defs, ok = res
   437	        if not ok:
   438	            # Error. Construct dummy return value.
   439	            if var_name:
   440	                name = var_name
   441	                if is_func_scope:
   442	                    name += "@" + str(call.line)
   443	            else:
   444	                name = var_name = "TypedDict@" + str(call.line)
   445	            info = self.build_typeddict_typeinfo(name, {}, set(), set(), call.line, None)
   446	        else:
   447	            if var_name is not None and name != var_name:
   448	                self.fail(
   449	                    'First argument "{}" to TypedDict() does not match variable name "{}"'.format(
   450	                        name, var_name
   451	                    ),
   452	                    node,
   453	                    code=codes.NAME_MATCH,
   454	                )
   455	            if name != var_name or is_func_scope:
   456	                # Give it a unique name derived from the line number.
   457	                name += "@" + str(call.line)
   458	            required_keys = {
   459	                field
   460	                for (field, t) in zip(items, types)
   461	                if (total or (isinstance(t, RequiredType) and t.required))
   462	                and not (isinstance(t, RequiredType) and not t.required)
   463	            }
   464	            readonly_keys = {
   465	                field for (field, t) in zip(items, types) if isinstance(t, ReadOnlyType)
   466	            }
   467	            types = [  # unwrap Required[T] or ReadOnly[T] to just T
   468	                t.item if isinstance(t, (RequiredType, ReadOnlyType)) else t for t in types
   469	            ]
   470
   471	            # Perform various validations after unwrapping.
   472	            for t in types:
   473	                check_for_explicit_any(
   474	                    t, self.options, self.api.is_typeshed_stub_file, self.msg, context=call
   475	                )
   476	            if self.options.disallow_any_unimported:
   477	                for t in types:
   478	                    if has_any_from_unimported_type(t):
   479	                        self.msg.unimported_type_becomes_any("Type of a TypedDict key", t, call)
   480
   481	            existing_info = None
   482	            if isinstance(node.analyzed, TypedDictExpr):
   483	                existing_info = node.analyzed.info
   484	            info = self.build_typeddict_typeinfo(
   485	                name,
   486	                dict(zip(items, types)),
   487	                required_keys,
   488	                readonly_keys,
   489	                call.line,
   490	                existing_info,
   491	            )
   492	            info.line = node.line
   493	        # Store generated TypeInfo under both names, see semanal_namedtuple for more details.
   494	        if name != var_name or is_func_scope:
   495	            self.api.add_symbol_skip_local(name, info)
   496	        if var_name:
   497	            self.api.add_symbol(var_name, info, node)
   498	        call.analyzed = TypedDictExpr(info)
   499	        call.analyzed.set_line(call)
   500	        return True, info, tvar_defs
   501
   502	    def parse_typeddict_args(
   503	        self, call: CallExpr
   504	    ) -> tuple[str, list[str], list[Type], bool, list[TypeVarLikeType], bool] | None:
   505	        """Parse typed dict call expression.
   506
   507	        Return names, types, totality, was there an error during parsing.
   508	        If some type is not ready, return None.
   509	        """
   510	        # TODO: Share code with check_argument_count in checkexpr.py?
   511	        args = call.args
   512	        if len(args) < 2:
   513	            return self.fail_typeddict_arg("Too few arguments for TypedDict()", call)
   514	        if len(args) > 3:
   515	            return self.fail_typeddict_arg("Too many arguments for TypedDict()", call)
   516	        # TODO: Support keyword arguments
   517	        if call.arg_kinds not in ([ARG_POS, ARG_POS], [ARG_POS, ARG_POS, ARG_NAMED]):
   518	            return self.fail_typeddict_arg("Unexpected arguments to TypedDict()", call)
   519	        if len(args) == 3 and call.arg_names[2] != "total":
   520	            return self.fail_typeddict_arg(
   521	                f'Unexpected keyword argument "{call.arg_names[2]}" for "TypedDict"', call
   522	            )
   523	        if not isinstance(args[0], StrExpr):
   524	            return self.fail_typeddict_arg(
   525	                "TypedDict() expects a string literal as the first argument", call
   526	            )
   527	        if not isinstance(args[1], DictExpr):
   528	            return self.fail_typeddict_arg(
   529	                "TypedDict() expects a dictionary literal as the second argument", call
   530	            )
   531	        total: bool | None = True
   532	        if len(args) == 3:
   533	            total = require_bool_literal_argument(self.api, call.args[2], "total")
   534	            if total is None:
   535	                return "", [], [], True, [], False
   536	        dictexpr = args[1]
   537	        tvar_defs = self.api.get_and_bind_all_tvars([t for k, t in dictexpr.items])
   538	        res = self.parse_typeddict_fields_with_types(dictexpr.items)
   539	        if res is None:
   540	            # One of the types is not ready, defer.
   541	            return None
   542	        items, types, ok = res
   543	        assert total is not None
   544	        return args[0].value, items, types, total, tvar_defs, ok

   546	    def parse_typeddict_fields_with_types(
   547	        self, dict_items: list[tuple[Expression | None, Expression]]
   548	    ) -> tuple[list[str], list[Type], bool] | None:
   549	        """Parse typed dict items passed as pairs (name expression, type expression).
   550
   551	        Return names, types, was there an error. If some type is not ready, return None.
   552	        """
   553	        seen_keys = set()
   554	        items: list[str] = []
   555	        types: list[Type] = []
   556	        for field_name_expr, field_type_expr in dict_items:
   557	            if isinstance(field_name_expr, StrExpr):
   558	                key = field_name_expr.value
   559	                items.append(key)
   560	                if key in seen_keys:
   561	                    self.fail(f'Duplicate TypedDict key "{key}"', field_name_expr)
   562	                seen_keys.add(key)
   563	            else:
   564	                name_context = field_name_expr or field_type_expr
   565	                self.fail_typeddict_arg("Invalid TypedDict() field name", name_context)
   566	                return [], [], False
   567	            try:
   568	                type = expr_to_unanalyzed_type(
   569	                    field_type_expr, self.options, self.api.is_stub_file
   570	                )
   571	            except TypeTranslationError:
   572	                self.fail_typeddict_arg("Use dict literal for nested TypedDict", field_type_expr)
   573	                return [], [], False
   574	            analyzed = self.api.anal_type(
   575	                type,
   576	                allow_typed_dict_special_forms=True,
   577	                allow_placeholder=not self.api.is_func_scope(),
   578	                prohibit_self_type="TypedDict item type",
   579	                prohibit_special_class_field_types="TypedDict",
   580	            )
   581	            if analyzed is None:
   582	                return None
   583	            types.append(analyzed)
   584	        return items, types, True

   592	    def build_typeddict_typeinfo(
   593	        self,
   594	        name: str,
   595	        item_types: dict[str, Type],
   596	        required_keys: set[str],
   597	        readonly_keys: set[str],
   598	        line: int,
   599	        existing_info: TypeInfo | None,
   600	    ) -> TypeInfo:
   601	        # Prefer typing then typing_extensions if available.
   602	        fallback = (
   603	            self.api.named_type_or_none("typing._TypedDict", [])
   604	            or self.api.named_type_or_none("typing_extensions._TypedDict", [])
   605	            or self.api.named_type_or_none("mypy_extensions._TypedDict", [])
   606	        )
   607	        assert fallback is not None
   608	        info = existing_info or self.api.basic_new_typeinfo(name, fallback, line)
   609	        typeddict_type = TypedDictType(item_types, required_keys, readonly_keys, fallback)

    11

    16	  module id and returns fully qualified names of differences (which act as
    17	  triggers).

    20	versions and compare the snapshots. The use of snapshots makes it easy to
    21	compare two versions of the *same* symbol table that is being mutated.
    22

    24

    26	  of the symbol table), it is considered different, of course.

    29	  it is considered different (for example, if a class is replaced with a
    30	  function).

    41	  is different rather than 'mod.Cls' being different).

   108	# Snapshot representation of a symbol table node or type. The representation is
   109	# opaque -- the only supported operations are comparing for equality and
   110	# hashing (latter for type snapshots only). Snapshots can contain primitive
   111	# objects, nested tuples, lists and dictionaries and primitive objects (type
   112	# snapshots are immutable).

   118	SnapshotItem: _TypeAlias = tuple[Union[Primitive, "SnapshotItem"], ...]

   124	def compare_symbol_table_snapshots(
   125	    name_prefix: str, snapshot1: dict[str, SymbolSnapshot], snapshot2: dict[str, SymbolSnapshot]
   126	) -> set[str]:
   127	    """Return names that are different in two snapshots of a symbol table.
   128
   129	    Only shallow (intra-module) differences are considered. References to things defined
   130	    outside the module are compared based on the name of the target only.
   131
   132	    Recurse into class symbol tables (if the class is defined in the target module).
   133
   134	    Return a set of fully-qualified names (e.g., 'mod.func' or 'mod.Class.method').
   135	    """
   136	    # Find names only defined only in one version.
   137	    names1 = {f"{name_prefix}.{name}" for name in snapshot1}
   138	    names2 = {f"{name_prefix}.{name}" for name in snapshot2}
   139	    triggers = names1 ^ names2
   140
   141	    # Look for names defined in both versions that are different.
   142	    for name in set(snapshot1.keys()) & set(snapshot2.keys()):
   143	        item1 = snapshot1[name]
   144	        item2 = snapshot2[name]
   145	        kind1 = item1[0]
   146	        kind2 = item2[0]
   147	        item_name = f"{name_prefix}.{name}"
   148	        if kind1 != kind2:
   149	            # Different kind of node in two snapshots -> trivially different.
   150	            triggers.add(item_name)
   151	        elif kind1 == "TypeInfo":
   152	            if item1[:-1] != item2[:-1]:
   153	                # Record major difference (outside class symbol tables).
   154	                triggers.add(item_name)
   155	            # Look for differences in nested class symbol table entries.
   156	            assert isinstance(item1[-1], dict)
   157	            assert isinstance(item2[-1], dict)
   158	            triggers |= compare_symbol_table_snapshots(item_name, item1[-1], item2[-1])
   159	        else:
   160	            # Shallow node (no interesting internal structure). Just use equality.
   161	            if snapshot1[name] != snapshot2[name]:
   162	                triggers.add(item_name)
   163
   164	    return triggers

   167	def snapshot_symbol_table(name_prefix: str, table: SymbolTable) -> dict[str, SymbolSnapshot]:
   168	    """Create a snapshot description that represents the state of a symbol table.
   169
   170	    The snapshot has a representation based on nested tuples and dicts
   171	    that makes it easy and fast to find differences.
   172
   173	    Only "shallow" state is included in the snapshot -- references to
   174	    things defined in other modules are represented just by the names of
   175	    the targets.
   176	    """
   177	    result: dict[str, SymbolSnapshot] = {}
   178	    for name, symbol in table.items():
   179	        node = symbol.node
   180	        # TODO: cross_ref?
   181	        fullname = node.fullname if node else None
   182	        common = (fullname, symbol.kind, symbol.module_public)
   183	        if isinstance(node, MypyFile):
   184	            # This is a cross-reference to another module.
   185	            # If the reference is busted because the other module is missing,
   186	            # the node will be a "stale_info" TypeInfo produced by fixup,
   187	            # but that doesn't really matter to us here.
   188	            result[name] = ("Moduleref", common)
   189	        elif isinstance(node, TypeVarExpr):
   190	            result[name] = (
   191	                "TypeVar",
   192	                node.variance,
   193	                [snapshot_type(value) for value in node.values],
   194	                snapshot_type(node.upper_bound),
   195	                snapshot_type(node.default),
   196	            )
   197	        elif isinstance(node, TypeAlias):
   198	            result[name] = (
   199	                "TypeAlias",
   200	                snapshot_types(node.alias_tvars),
   201	                node.normalized,
   202	                node.no_args,
   203	                snapshot_optional_type(node.target),
   204	            )
   205	        elif isinstance(node, ParamSpecExpr):
   206	            result[name] = (
   207	                "ParamSpec",
   208	                node.variance,
   209	                snapshot_type(node.upper_bound),
   210	                snapshot_type(node.default),
   211	            )
   212	        elif isinstance(node, TypeVarTupleExpr):
   213	            result[name] = (
   214	                "TypeVarTuple",
   215	                node.variance,
   216	                snapshot_type(node.upper_bound),
   217	                snapshot_type(node.default),
   218	            )
   219	        else:
   220	            assert symbol.kind != UNBOUND_IMPORTED
   221	            if node and get_prefix(node.fullname) != name_prefix:
   222	                # This is a cross-reference to a node defined in another module.
   223	                # Include the node kind (FuncDef, Decorator, TypeInfo, ...), so that we will
   224	                # reprocess when a *new* node is created instead of merging an existing one.
   225	                result[name] = ("CrossRef", common, type(node).__name__)
   226	            else:
   227	                result[name] = snapshot_definition(node, common)
   228	    return result

   352	class SnapshotTypeVisitor(TypeVisitor[SnapshotItem]):
   353	    """Creates a read-only, self-contained snapshot of a type object.
   354
   355	    Properties of a snapshot:
   356
   357	    - Contains (nested) tuples and other immutable primitive objects only.
   358	    - References to AST nodes are replaced with full names of targets.
   359	    - Has no references to mutable or non-primitive objects.
   360	    - Two snapshots represent the same object if and only if they are
   361	      equal.
   362	    - Results must be sortable. It's important that tuples have
   363	      consistent types and can't arbitrarily mix str and None values,
   364	      for example, since they can't be compared.
   365	    """
   366

   368	        return (
   369	            "UnboundType",
   370	            typ.name,
   371	            typ.optional,
   372	            typ.empty_tuple_index,
   373	            snapshot_types(typ.args),
   374	        )

   377	        return snapshot_simple_type(typ)
