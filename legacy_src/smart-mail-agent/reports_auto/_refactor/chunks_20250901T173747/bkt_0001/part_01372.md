   508	@lru_cache(maxsize=1024)
   509	def is_suspiciously_successive_range(
   510	    unicode_range_a: str | None, unicode_range_b: str | None
   511	) -> bool:
   512	    """
   513	    Determine if two Unicode range seen next to each other can be considered as suspicious.
   514	    """
   515	    if unicode_range_a is None or unicode_range_b is None:
   516	        return True
   517
   518	    if unicode_range_a == unicode_range_b:
   519	        return False
   520
   521	    if "Latin" in unicode_range_a and "Latin" in unicode_range_b:
   522	        return False
   523
   524	    if "Emoticons" in unicode_range_a or "Emoticons" in unicode_range_b:
   525	        return False
   526
   527	    # Latin characters can be accompanied with a combining diacritical mark
   528	    # eg. Vietnamese.
   529	    if ("Latin" in unicode_range_a or "Latin" in unicode_range_b) and (
   530	        "Combining" in unicode_range_a or "Combining" in unicode_range_b
   531	    ):
   532	        return False
   533
   534	    keywords_range_a, keywords_range_b = (
   535	        unicode_range_a.split(" "),
   536	        unicode_range_b.split(" "),
   537	    )
   538
   539	    for el in keywords_range_a:
   540	        if el in UNICODE_SECONDARY_RANGE_KEYWORD:
   541	            continue
   542	        if el in keywords_range_b:
   543	            return False
   544

   317	def cp_similarity(iana_name_a: str, iana_name_b: str) -> float:
   318	    if is_multi_byte_encoding(iana_name_a) or is_multi_byte_encoding(iana_name_b):
   319	        return 0.0

   321	    decoder_a = importlib.import_module(f"encodings.{iana_name_a}").IncrementalDecoder
   322	    decoder_b = importlib.import_module(f"encodings.{iana_name_b}").IncrementalDecoder
   323
   324	    id_a: IncrementalDecoder = decoder_a(errors="ignore")
   325	    id_b: IncrementalDecoder = decoder_b(errors="ignore")

   327	    character_match_count: int = 0
   328
   329	    for i in range(255):
   330	        to_be_decoded: bytes = bytes([i])
   331	        if id_a.decode(to_be_decoded) == id_b.decode(to_be_decoded):
   332	            character_match_count += 1

   334	    return character_match_count / 254

   395	            # not the cleanest way to perform that fix but clever enough for now.
   396	            if is_multi_byte_decoder and i > 0:
   397	                chunk_partial_size_chk: int = min(chunk_size, 16)

   399	                if (
   400	                    decoded_payload

   407	                            cut_sequence = sig_payload + cut_sequence

   411	                        if chunk[:chunk_partial_size_chk] in decoded_payload:
   412	                            break

   371	def open_stream(
   372	    filename: str | os.PathLike[str],
   373	    mode: str = "r",
   374	    encoding: str | None = None,
   375	    errors: str | None = "strict",
   376	    atomic: bool = False,
   377	) -> tuple[t.IO[t.Any], bool]:

   379	    filename = os.fspath(filename)
   380
   381	    # Standard streams first. These are simple because they ignore the
   382	    # atomic flag. Use fsdecode to handle Path("-").
   383	    if os.fsdecode(filename) == "-":
   384	        if any(m in mode for m in ["w", "a", "x"]):
   385	            if binary:
   386	                return get_binary_stdout(), False
   387	            return get_text_stdout(encoding=encoding, errors=errors), False
   388	        if binary:
   389	            return get_binary_stdin(), False
   390	        return get_text_stdin(encoding=encoding, errors=errors), False
   391
   392	    # Non-atomic writes directly go out through the regular open functions.
   393	    if not atomic:
   394	        return _wrap_io_open(filename, mode, encoding, errors), True
   395
   396	    # Some usability stuff for atomic writes
   397	    if "a" in mode:
   398	        raise ValueError(
   399	            "Appending to an existing file is not supported, because that"
   400	            " would involve an expensive `copy`-operation to a temporary"
   401	            " file. Open the file in normal `w`-mode and copy explicitly"
   402	            " if that's what you're after."
   403	        )
   404	    if "x" in mode:
   405	        raise ValueError("Use the `overwrite`-parameter instead.")
   406	    if "w" not in mode:
   407	        raise ValueError("Atomic writes only make sense with `w`-mode.")
   408
   409	    # Atomic writes are more complicated.  They work by opening a file
   410	    # as a proxy in the same folder and then using the fdopen
   411	    # functionality to wrap it in a Python file.  Then we wrap it in an
   412	    # atomic file that moves the file over on close.
   413	    import errno
   414	    import random

   416	    try:

   418	    except OSError:

   423	    if binary:
   424	        flags |= getattr(os, "O_BINARY", 0)

   426	    while True:
   427	        tmp_filename = os.path.join(
   428	            os.path.dirname(filename),
   429	            f".__atomic-write{random.randrange(1 << 32):08x}",
   430	        )
   431	        try:
   432	            fd = os.open(tmp_filename, flags, 0o666 if perm is None else perm)
   433	            break

   229	            self.bar_template
   230	            % {
   231	                "label": self.label,
   232	                "bar": self.format_bar(),
   233	                "info": self.info_sep.join(info_bits),
   234	            }
   235	        ).rstrip()

   237	    def render_progress(self) -> None:
   238	        import shutil
   239
   240	        if self.hidden:
   241	            return
   242
   243	        if not self._is_atty:
   244	            # Only output the label once if the output is not a TTY.
   245	            if self._last_line != self.label:
   246	                self._last_line = self.label
   247	                echo(self.label, file=self.file, color=self.color)
   248	            return
   249
   250	        buf = []
   251	        # Update width in case the terminal has been resized
   252	        if self.autowidth:
   253	            old_width = self.width
   254	            self.width = 0
   255	            clutter_length = term_len(self.format_progress_line())
   256	            new_width = max(0, shutil.get_terminal_size().columns - clutter_length)
   257	            if new_width < old_width and self.max_width is not None:
   258	                buf.append(BEFORE_BAR)
   259	                buf.append(" " * self.max_width)
   260	                self.max_width = new_width
   261	            self.width = new_width
   262
   263	        clear_width = self.width
   264	        if self.max_width is not None:
   265	            clear_width = self.max_width
   266
   267	        buf.append(BEFORE_BAR)
   268	        line = self.format_progress_line()
   269	        line_len = term_len(line)
   270	        if self.max_width is None or self.max_width < line_len:
   271	            self.max_width = line_len
   272
   273	        buf.append(line)
   274	        buf.append(" " * (clear_width - line_len))
   275	        line = "".join(buf)
   276	        # Render the line only if it changed.
   277
   278	        if line != self._last_line:
   279	            self._last_line = line
   280	            echo(line, file=self.file, color=self.color, nl=False)
   281	            self.file.flush()

   283	    def make_step(self, n_steps: int) -> None:
   284	        self.pos += n_steps
   285	        if self.length is not None and self.pos >= self.length:
   286	            self.finished = True
   287
   288	        if (time.time() - self.last_eta) < 1.0:
   289	            return
   290
   291	        self.last_eta = time.time()
   292
   293	        # self.avg is a rolling list of length <= 7 of steps where steps are
   294	        # defined as time elapsed divided by the total progress through
   295	        # self.length.
   296	        if self.pos:
   297	            step = (time.time() - self.start) / self.pos
   298	        else:
   299	            step = time.time() - self.start
   300
   301	        self.avg = self.avg[-6:] + [step]
   302
   303	        self.eta_known = self.length is not None
   304
   305	    def update(self, n_steps: int, current_item: V | None = None) -> None:
   306	        """Update the progress bar by advancing a specified number of
   307	        steps, and optionally set the ``current_item`` for this new
   308	        position.
   309
   310	        :param n_steps: Number of steps to advance.
   311	        :param current_item: Optional item to set as ``current_item``
   312	            for the updated position.
   313
   314	        .. versionchanged:: 8.0
   315	            Added the ``current_item`` optional parameter.
   316
   317	        .. versionchanged:: 8.0
   318	            Only render when the number of steps meets the
   319	            ``update_min_steps`` threshold.
   320	        """
   321	        if current_item is not None:
   322	            self.current_item = current_item
   323
   324	        self._completed_intervals += n_steps
   325
   326	        if self._completed_intervals >= self.update_min_steps:
   327	            self.make_step(self._completed_intervals)
   328	            self.render_progress()
   329	            self._completed_intervals = 0
   330
   331	    def finish(self) -> None:
   332	        self.eta_known = False
   333	        self.current_item = None
   334	        self.finished = True

   336	    def generator(self) -> cabc.Iterator[V]:
   337	        """Return a generator which yields the items added to the bar
   338	        during construction, and updates the progress bar *after* the
   339	        yielded block returns.
   340	        """
   341	        # WARNING: the iterator interface for `ProgressBar` relies on
   342	        # this and only works because this is a simple generator which
   343	        # doesn't create or manage additional state. If this function
   344	        # changes, the impact should be evaluated both against
   345	        # `iter(bar)` and `next(bar)`. `next()` in particular may call
   346	        # `self.generator()` repeatedly, and this must remain safe in
   347	        # order for that interface to work.
   348	        if not self.entered:
   349	            raise RuntimeError("You need to use progress bars in a with block.")
   350
   351	        if not self._is_atty:
   352	            yield from self.iter
   353	        else:
   354	            for rv in self.iter:
   355	                self.current_item = rv
   356
   357	                # This allows show_item_func to be updated before the
   358	                # item is processed. Only trigger at the beginning of
   359	                # the update interval.
   360	                if self._completed_intervals == 0:
   361	                    self.render_progress()
   362
   363	                yield rv
   364	                self.update(1)
   365
   366	            self.finish()
   367	            self.render_progress()

   371	    """Decide what method to use for paging through text."""

   374	    # There are no standard streams attached to write to. For example,
