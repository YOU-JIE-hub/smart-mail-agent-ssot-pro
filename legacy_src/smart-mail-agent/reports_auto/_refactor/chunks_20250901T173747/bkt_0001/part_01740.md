    95	    """
    96	    Convert a string into a token type::
    97
    98	        >>> string_to_token('String.Double')

   100	        >>> string_to_token('Token.Literal.Number')

   106

   111	        return s

   115	    for item in s.split('.'):
   116	        node = getattr(node, item)
