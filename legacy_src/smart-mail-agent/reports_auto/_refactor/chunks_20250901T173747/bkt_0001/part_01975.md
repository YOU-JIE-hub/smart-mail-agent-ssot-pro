   388	    def format(self):
   389	        # register the Catalog/INfo and then format the objects one by one until exhausted
   390	        # (possible infinite loop if there is a bug that continually makes new objects/refs...)
   391	        # Prepare encryption
   392	        self.encrypt.prepare(self)
   393	        cat = self.Catalog
   394	        info = self.info
   395	        self.Reference(cat)
   396	        self.Reference(info)
   397	        # register the encryption dictionary if present
   398	        encryptref = None
   399	        encryptinfo = self.encrypt.info()
   400	        if encryptinfo:
   401	            encryptref = self.Reference(encryptinfo)
   402	        # make std fonts (this could be made optional
   403	        counter = 0 # start at first object (object 1 after preincrement)
   404	        ids = [] # the collection of object ids in object number order
   405	        numbertoid = self.numberToId
   406	        idToNV = self.idToObjectNumberAndVersion
   407	        idToOb = self.idToObject
   408	        idToOf = self.idToOffset
   409	        ### note that new entries may be "appended" DURING FORMATTING
   410	        # __accum__ allows objects to know where they are in the file etc etc
   411	        self.__accum__ = File = PDFFile(self._pdfVersion) # output collector
   412	        while True:
   413	            counter += 1 # do next object...
   414	            if counter not in numbertoid: break
   415	            oid = numbertoid[counter]
   416	            #printidToOb
   417	            obj = idToOb[oid]
   418	            IO = PDFIndirectObject(oid, obj)
   419	            # register object number and version
   420	            IOf = IO.format(self)
   421	            # add a comment to the PDF output
   422	            if not rl_config.invariant and rl_config.pdfComments:
   423	                try:
   424	                    classname = obj.__class__.__name__
   425	                except:
   426	                    classname = ascii(obj)
   427	                File.add("%% %s: class %s \n" % (ascii(oid), classname[:50]))
   428	            offset = File.add(IOf)
   429	            idToOf[oid] = offset
   430	            ids.append(oid)
   431	        del self.__accum__
   432	        # sanity checks (must happen AFTER formatting)
   433	        lno = len(numbertoid)
   434	        if counter-1!=lno:
   435	            raise ValueError("counter %s doesn't match number to id dictionary %s" %(counter, lno))
   436	        # now add the xref
   437	        xref = PDFCrossReferenceTable()
   438	        xref.addsection(0, ids)
   439	        xreff = xref.format(self)
   440	        xrefoffset = File.add(xreff)
   441	        # now add the trailer
   442	        trailer = PDFTrailer(
   443	            startxref = xrefoffset,
   444	            Size = lno+1,
   445	            Root = self.Reference(cat),
   446	            Info = self.Reference(info),
   447	            Encrypt = encryptref,
   448	            ID = self.ID(),
   449	            )
   450	        trailerf = trailer.format(self)
   451	        File.add(trailerf)
   452	        for ds in getattr(self,'_digiSigs',[]):
   453	            ds.sign(File)
   454	        # return string format for pdf file
   455	        return File.format(self)

   498	        # don't make references to other references, or non instances, unless they are named!

   512	        # otherwise register the new object

   514	        if name is None:

   525	        self.numberToId[objectcounter] = name
   526	        idToObject[name] = obj

   573	class PDFString(PDFObject):
   574	    unicodeEncValid = False
   575	    def __init__(self, s, escape=1, enc='auto'):
   576	        '''s can be unicode/utf8 or a PDFString
   577	        if escape is true then the output will be passed through escape
   578	        if enc is raw then bytes will be left alone
   579	        if enc is auto we'll try and automatically adapt to utf_16_be/utf_16_le if the
   580	        effective string is not entirely in pdfdoc
   581	        if self.unicodeEncValid unicode will use the specifed encoding
   582	        '''
   583	        if isinstance(s,PDFString):
   584	            self.s = s.s
   585	            self.escape = s.escape

   588	            self.s = s

   590	            self.enc = enc

   592	        s = self.s
   593	        enc = getattr(self,'enc','auto')
   594	        if isBytes(s):
   595	            if enc == 'auto':
   596	                try:
   597	                    if s.startswith(codecs.BOM_UTF16_BE):
   598	                        u = s.decode('utf_16_be')
   599	                    elif s.startswith(codecs.BOM_UTF16_LE):
   600	                        u = s.decode('utf_16_le')
   601	                    else:
   602	                        u = s.decode('utf8')
   603	                    if _checkPdfdoc(u):
   604	                        s = u.encode('pdfdoc')
   605	                    else:
   606	                        s = codecs.BOM_UTF16_BE+u.encode('utf_16_be')
   607	                except:
   608	                    try:
   609	                        s.decode('pdfdoc')
   610	                    except:
   611	                        stderr.write('Error in %s' % (repr(s),))
   612	                        raise
   613	        elif isUnicode(s):
   614	            if enc == 'auto':
   615	                if _checkPdfdoc(s):
   616	                    s = s.encode('pdfdoc')
   617	                else:
   618	                    s = codecs.BOM_UTF16_BE+s.encode('utf_16_be')
   619	            elif self.unicodeEncValid:
   620	                s = s.encode(self.enc)
   621	            else:
   622	                s = codecs.BOM_UTF16_BE+s.encode('utf_16_be')
   623	        else:
   624	            raise ValueError('PDFString argument must be str/unicode not %s' % type(s))
   625
   626	        escape = getattr(self,'escape',1)
   627	        if not isinstance(document.encrypt,NoEncryption):
   628	            s = document.encrypt.encode(s)
   629	            escape = 1
   630	        if escape:
   631	            try:
   632	                es = "(%s)" % escapePDF(s)

   634	                raise ValueError("cannot escape %s %s" % (s, repr(s)))

   639	            return pdfdocEnc(es)

   911	class PDFCrossReferenceSubsection(PDFObject):
   912	    def __init__(self, firstentrynumber, idsequence):
   913	        self.firstentrynumber = firstentrynumber
   914	        self.idsequence = idsequence
   915	    def format(self, document):
   916	        """id sequence should represent contiguous object nums else error. free numbers not supported (yet)"""
   917	        firstentrynumber = self.firstentrynumber
   918	        idsequence = self.idsequence
   919	        entries = list(idsequence)
   920	        nentries = len(idsequence)
   921	        # special case: object number 0 is always free
   922	        taken = {}
   923	        if firstentrynumber==0:
   924	            taken[0] = "standard free entry"
   925	            nentries = nentries+1
   926	            entries.insert(0, "0000000000 65535 f ")
   927	        idToNV = document.idToObjectNumberAndVersion
   928	        idToOffset = document.idToOffset
   929	        lastentrynumber = firstentrynumber+nentries-1
   930	        for id in idsequence:
   931	            (num, version) = idToNV[id]
   932	            if num in taken:
   933	                raise ValueError("object number collision %s %s %s" % (num, repr(id), repr(taken[id])))
   934	            if num>lastentrynumber or num<firstentrynumber:
   935	                raise ValueError("object number %s not in range %s..%s" % (num, firstentrynumber, lastentrynumber))
   936	            # compute position in list
   937	            rnum = num-firstentrynumber
   938	            taken[num] = id
   939	            offset = idToOffset[id]
   940	            entries[num] = '%0.10d %0.5d n ' % (offset, version)
   941	        # now add the initial line
   942	        firstline = "%s %s" % (firstentrynumber, nentries)
   943	        entries.insert(0, firstline)
   944	        # make sure it ends with \n
   945	        entries.append("")
   946	        return pdfdocEnc('\n'.join(entries))

  1213	        printed on a canvas, but instead influence the 'logical' page numbers
  1214	        displayed by PDF viewers. However, when using roman numerals (i, ii,
  1215	        iii...) or page prefixes for appendecies (A.1, A.2...) on the physical

  1217	        displayed by the PDF viewer to match up with the physical numbers. A
  1218	        PDFPageLabel changes the properties of numbering at the page on which it
  1219	        appears (see the class 'PDFPageLabels' for specifying where a PDFPageLabel
  1220	        is associated) and all subsequent pages, until a new PDFPageLabel is
  1221	        encountered.

  1224	        subsequent page labels. 'style' determines the numberings style, arabic,
  1225	        roman, letters; 'start' specifies the starting number; and 'prefix' any
  1226	        prefix to be applied to the page numbers. All these arguments can be left
  1227	        out or set to None.

  1239

  1241	                can be used when numbering style changes to reset the page number back
  1242	                to one, ie from roman to arabic, or from arabic to appendecies. Can be
  1243	                any positive integer or None. I'm not sure what the effect of
  1244	                specifying None is, probably that page numbering continues with the
  1245	                current sequence, I'd have to check the spec to clarify though.
  1246
  1247	        * prefix:
  1248
  1249	            -   A string which is prefixed to the page numbers. Can be used to display
  1250	                appendecies in the format: A.1, A.2, ..., B.1, B.2, ... where a
