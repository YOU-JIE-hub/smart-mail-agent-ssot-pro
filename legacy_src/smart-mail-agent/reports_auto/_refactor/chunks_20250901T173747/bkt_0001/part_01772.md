   337	def _addAttributeNames(m):
   338	    K = list(m.keys())
   339	    for k in K:
   340	        n = m[k][0]
   341	        if n not in m: m[n] = m[k]
   342	        n = n.lower()
   343	        if n not in m: m[n] = m[k]

   352	def _applyAttributes(obj, attr):
   353	    for k, v in attr.items():
   354	        if isinstance(v,(list,tuple)) and v[0]=='relative':
   355	            if hasattr(obj, k):
   356	                v = v[1]+getattr(obj,k)
   357	            else:
   358	                v = v[1]
   359	        setattr(obj,k,v)
