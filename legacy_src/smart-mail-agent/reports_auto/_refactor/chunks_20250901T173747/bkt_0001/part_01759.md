   397	def getArgvDict(**kw):
   398	    ''' Builds a dictionary from its keyword arguments with overrides from sys.argv.
   399	        Attempts to be smart about conversions, but the value can be an instance
   400	        of ArgDictValue to allow specifying a conversion function.
   401	    '''
   402	    def handleValue(v,av,func):
   403	        if func:
   404	            v = func(av)
   405	        else:
   406	            if isStr(v):
   407	                v = av
   408	            elif isinstance(v,float):
   409	                v = float(av)
   410	            elif isinstance(v,int):
   411	                v = int(av)
   412	            elif isinstance(v,list):
   413	                v = list(literal_eval(av),{})
   414	            elif isinstance(v,tuple):
   415	                v = tuple(literal_eval(av),{})
   416	            else:
   417	                raise TypeError("Can't convert string %r to %s" % (av,type(v)))
   418	        return v
   419
   420	    A = sys.argv[1:]
   421	    R = {}
   422	    for k, v in kw.items():
   423	        if isinstance(v,ArgvDictValue):
   424	            v, func = v.value, v.func
   425	        else:
   426	            func = None
   427	        handled = 0
   428	        ke = k+'='
   429	        for a in A:
   430	            if a.startswith(ke):
   431	                av = a[len(ke):]
   432	                A.remove(a)
   433	                R[k] = handleValue(v,av,func)
   434	                handled = 1
   435	                break
   436
   437	        if not handled: R[k] = handleValue(v,v,func)

  1089	def prev_this_next(items):
  1090	    """
  1091	    Loop over a collection with look-ahead and look-back.
  1092
  1093	    From Thomas Guest,
  1094	    http://wordaligned.org/articles/zippy-triples-served-with-python
  1095
  1096	    Seriously useful looping tool (Google "zippy triples")
  1097	    lets you loop a collection and see the previous and next items,
  1098	    which get set to None at the ends.
  1099
  1100	    To be used in layout algorithms where one wants a peek at the
  1101	    next item coming down the pipe.
  1102
  1103	    """
  1104
  1105	    extend = itertools.chain([None], items, [None])
  1106	    prev, this, next = itertools.tee(extend, 3)
  1107	    try:
  1108	        next(this)
  1109	        next(next)
  1110	        next(next)
  1111	    except StopIteration:
  1112	        pass
  1113	    return zip(prev, this, next)

  1115	def commasplit(s):
  1116	    '''
  1117	    Splits the string s at every unescaped comma and returns the result as a list.
  1118	    To escape a comma, double it. Individual items are stripped.
  1119	    To avoid the ambiguity of 3 successive commas to denote a comma at the beginning
  1120	    or end of an item, add a space between the item seperator and the escaped comma.
  1121
  1122	    >>> commasplit(u'a,b,c') == [u'a', u'b', u'c']
  1123	    True
  1124	    >>> commasplit('a,, , b , c    ') == [u'a,', u'b', u'c']
  1125	    True
  1126	    >>> commasplit(u'a, ,,b, c') == [u'a', u',b', u'c']
  1127	    '''
  1128	    if isBytes(s): s = s.decode('utf8')
  1129	    n = len(s)-1
  1130	    s += u' '
  1131	    i = 0
  1132	    r=[u'']
  1133	    while i<=n:
  1134	        if s[i]==u',':
  1135	            if s[i+1]==u',':
  1136	                r[-1]+=u','
  1137	                i += 1
  1138	            else:
  1139	                r[-1] = r[-1].strip()
  1140	                if i!=n: r.append(u'')
  1141	        else:
  1142	            r[-1] += s[i]
  1143	        i+=1
  1144	    r[-1] = r[-1].strip()
  1145	    return r

    51	    """"Simplest possible parser with only the most basic options.
    52
    53	    This defines the line-handling abilities and basic mechanism.
    54	    The class YAMLParser includes capabilities for a fairly rich
    55	    story."""
    56
    57	    def __init__(self):
    58	        self.reset()
    59
    60	    def reset(self):
    61	        self._lineNo = 0
    62	        self._style = 'Normal'  # the default
    63	        self._results = []
    64	        self._buf = []
    65	        self._mode = PLAIN
    66
    67	    def parseFile(self, filename):
    68	        #returns list of objects
    69	        data = open(filename, 'r').readlines()
    70
    71	        for line in data:
    72	            #strip trailing newlines
    73	            self.readLine(line[:-1])
    74	        self.endPara()
    75	        return self._results
    76
    77	    def parseText(self, textBlock):
    78	        "Parses the a possible multi-line text block"
    79	        lines = textBlock.split('\n')
    80	        for line in lines:
    81	            self.readLine(line)
    82	        self.endPara()
    83	        return self._results
    84
    85	    def readLine(self, line):
    86	        #this is the inner loop
    87	        self._lineNo = self._lineNo + 1
    88	        stripped = line.lstrip()
    89	        if len(stripped) == 0:
    90	            if self._mode == PLAIN:
    91	                self.endPara()
    92	            else:  #preformatted, append it
    93	                self._buf.append(line)
    94	        elif line[0]=='.':
    95	            # we have a command of some kind
    96	            self.endPara()
    97	            words = stripped[1:].split()
    98	            cmd, args = words[0], words[1:]
    99
   100	            #is it a parser method?
   101	            if hasattr(self.__class__, cmd):
   102	                #this was very bad; any type error in the method was hidden
   103	                #we have to hack the traceback
   104	                try:
   105	                    getattr(self,cmd)(*args)
   106	                except TypeError as err:
   107	                    sys.stderr.write("Parser method: %s(*%s) %s at line %d\n" % (cmd, args, err, self._lineNo))
   108	                    raise
   109	            else:
   110	                # assume it is a paragraph style -
   111	                # becomes the formatter's problem
   112	                self.endPara()  #end the last one
   113	                words = stripped.split(' ', 1)
   114	                assert len(words)==2, "Style %s but no data at line %d" % (words[0], self._lineNo)
   115	                (styletag, data) = words
   116	                self._style = styletag[1:]
   117	                self._buf.append(data)
   118	        else:
   119	            #we have data, add to para
   120	            self._buf.append(line)
   121
   122	    def endPara(self):
   123	        #ends the current paragraph, or preformatted block
   124
   125	        text = ' '.join(self._buf)
   126	        if text:
   127	            if self._mode == PREFORMATTED:
   128	                #item 3 is list of lines
   129	                self._results.append(('PREFORMATTED', self._style,
   130	                                 '\n'.join(self._buf)))
   131	            else:
   132	                self._results.append(('PARAGRAPH', self._style, text))
   133	        self._buf = []
   134	        self._style = 'Normal'
   135
   136	    def beginPre(self, stylename):
   137	        self._mode = PREFORMATTED
   138	        self._style = stylename
   139
   140	    def endPre(self):
   141	        self.endPara()
   142	        self._mode = PLAIN
   143
   144	    def image(self, filename):
   145	        self.endPara()
   146	        self._results.append(('IMAGE', filename))
