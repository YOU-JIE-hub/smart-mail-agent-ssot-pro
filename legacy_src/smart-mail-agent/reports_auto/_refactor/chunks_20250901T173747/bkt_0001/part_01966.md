    12	def frange(start, end=None, inc=None):
    13	    "A range function, that does accept float increments..."
    14
    15	    if end == None:
    16	        end = start + 0.0
    17	        start = 0.0
    18
    19	    if inc == None:
    20	        inc = 1.0
    21
    22	    L = []
    23	    end = end - inc*0.0001  #to avoid numrical problems
    24	    while 1:
    25	        next = start + len(L) * inc
    26	        if inc > 0 and next >= end:
    27	            break
    28	        elif inc < 0 and next <= end:
    29	            break
    30	        L.append(next)
    31
    32	    return L

   425	def centroid(P):
   426	    '''compute average point of a set of points'''
   427	    cx = 0
   428	    cy = 0
   429	    for x,y in P:
   430	        cx+=x
   431	        cy+=y
   432	    n = float(len(P))

   435	def rotatedEnclosingRect(P, angle, rect):
   436	    '''
   437	    given P a sequence P of x,y coordinate pairs and an angle in degrees
   438	    find the centroid of P and the axis at angle theta through it
   439	    find the extreme points of P wrt axis parallel distance and axis
   440	    orthogonal distance. Then compute the least rectangle that will still
   441	    enclose P when rotated by angle.
   442
   443	    The class R
   444	    '''
   445	    from math import pi, cos, sin
   446	    x0, y0 = centroid(P)
   447	    theta = (angle/180.)*pi
   448	    s,c=sin(theta),cos(theta)
   449	    def parallelAxisDist(xy,s=s,c=c,x0=x0,y0=y0):
   450	        x,y = xy
   451	        return (s*(y-y0)+c*(x-x0))
   452	    def orthogonalAxisDist(xy,s=s,c=c,x0=x0,y0=y0):
   453	        x,y = xy
   454	        return (c*(y-y0)+s*(x-x0))
   455	    L = list(map(parallelAxisDist,P))
   456	    L.sort()
   457	    a0, a1 = L[0], L[-1]
   458	    L = list(map(orthogonalAxisDist,P))
   459	    L.sort()
   460	    b0, b1 = L[0], L[-1]
   461	    rect.x, rect.width = a0, a1-a0
   462	    rect.y, rect.height = b0, b1-b0
   463	    g = Group(transform=(c,s,-s,c,x0,y0))
   464	    g.add(rect)
   465	    return g

   100	def fontify(pytext, searchfrom = 0, searchto = None):
   101	    if searchto is None:
   102	        searchto = len(pytext)
   103	    # Cache a few attributes for quicker reference.
   104	    search = matchRE.search
   105	    idSearch = idRE.search
   106
   107	    tags = []
   108	    tags_append = tags.append
   109	    commentTag = 'comment'
   110	    stringTag = 'string'
   111	    keywordTag = 'keyword'
   112	    identifierTag = 'identifier'
   113
   114	    start = 0
   115	    end = searchfrom
   116	    while 1:
   117	        m = search(pytext, end)
   118	        if m is None:
   119	            break   # EXIT LOOP
   120	        start = m.start()
   121	        if start >= searchto:
   122	            break   # EXIT LOOP
   123	        match = m.group(0)
   124	        end = start + len(match)
   125	        c = match[0]
   126	        if c not in "#'\"":
   127	            # Must have matched a keyword.
   128	            if start != searchfrom:
   129	                # there's still a redundant char before and after it, strip!
   130	                match = match[1:-1]
   131	                start = start + 1
   132	            else:
   133	                # this is the first keyword in the text.
   134	                # Only a space at the end.
   135	                match = match[:-1]
   136	            end = end - 1
   137	            tags_append((keywordTag, start, end, None))
   138	            # If this was a defining keyword, look ahead to the
   139	            # following identifier.
   140	            if match in ["def", "class"]:
   141	                m = idSearch(pytext, end)
   142	                if m is not None:
   143	                    start = m.start()
   144	                    if start == end:
   145	                        match = m.group(0)
   146	                        end = start + len(match)
   147	                        tags_append((identifierTag, start, end, None))
   148	        elif c == "#":
   149	            tags_append((commentTag, start, end, None))
   150	        else:
   151	            tags_append((stringTag, start, end, None))
   152	    return tags

    12		Based on what is widely thought to be RSA's ArcIV algorithm.
    13		It produces output streams that are identical.

    19			self.reset()

    21		def reset(self):
    22			'''restore the cipher to it's start state'''
    23			#Initialize private key, k With the values of the key mod 256.
    24			#and sbox With numbers 0 - 255. Then compute sbox
    25			key = self._key
    26			if isUnicode(key): key = key.encode('utf8')
    27			sbox = list(range(256))
    28			k = list(range(256))
    29			lk = len(key)
    30			for i in sbox:
    31				k[i] = key[i % lk] % 256
    32
    33			#Re-order sbox using the private key, k.
    34			#Iterating each element of sbox re-calculate the counter j
    35			#Then interchange the elements sbox[a] & sbox[b]
    36			j = 0
    37			for i in range(256):
    38				j = (j+sbox[i]+k[i]) % 256
    39				sbox[i], sbox[j] = sbox[j], sbox[i]
    40			self._sbox, self._i, self._j = sbox, 0, 0

    42		def _encode(self, B):
    43			'''
    44			return the list of encoded bytes of B, B might be a string or a
    45			list of integers between 0 <= i <= 255
    46			'''
    47			sbox, i, j = self._sbox, self._i, self._j
    48
    49			C = list(B.encode('utf8')) if isinstance(B,str) else (list(B) if isinstance(B,bytes) else B[:])
    50			n = len(C)
    51			p = 0
    52			while p<n:
    53				#update the variables i, j.
    54				self._i = i = (i + 1) % 256
    55				self._j = j = (j + sbox[i]) % 256
    56				#swap sbox[i] and sbox[j]
    57				sbox[i], sbox[j] = sbox[j], sbox[i]
    58				#overwrite the plaintext with the ciphered byte
    59				C[p] = C[p] ^ sbox[(sbox[i] + sbox[j]) % 256]
    60				p += 1
