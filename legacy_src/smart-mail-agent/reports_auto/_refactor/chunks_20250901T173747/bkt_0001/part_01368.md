   104	        # This is a UTF-8 document that contains bytestrings
   105	        # completely incompatible with UTF-8 (ie. encoded with some other
   106	        # encoding).
   107	        #
   108	        # Since there is no consistent encoding for the document,
   109	        # Unicode, Dammit will eventually encode the document as UTF-8
   110	        # and encode the incompatible characters as REPLACEMENT
   111	        # CHARACTER.
   112	        #
   113	        # If chardet is installed, it will detect that the document
   114	        # can be converted into ISO-8859-1 without errors. This happens
   115	        # to be the wrong encoding, but it is a consistent encoding, so the
   116	        # code we're testing here won't run.
