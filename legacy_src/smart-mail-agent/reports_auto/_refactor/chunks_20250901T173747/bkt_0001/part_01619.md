rformance over generating detailed
   314	        error descriptions.
   315	        """
   316	        if id not in self.results:
   317	            top_level = id.partition(".")[0]
   318	            use_typeshed = True
   319	            if id in self.stdlib_py_versions:
   320	                use_typeshed = self._typeshed_has_version(id)
   321	            elif top_level in self.stdlib_py_versions:
   322	                use_typeshed = self._typeshed_has_version(top_level)
   323	            result, should_cache = self._find_module(id, use_typeshed)
   324	            if should_cache:
   325	                if (
   326	                    not (
   327	                        fast_path or (self.options is not None and self.options.fast_module_lookup)
   328	                    )
   329	                    and result is ModuleNotFoundReason.NOT_FOUND
   330	                    and self._can_find_module_in_parent_dir(id)
   331	                ):
   332	                    self.results[id] = ModuleNotFoundReason.WRONG_WORKING_DIRECTORY
   333	                else:
   334	                    self.results[id] = result
   335	                return self.results[id]
   336	            else:
   337	                return result
   338	        return self.results[id]

   340	    def _typeshed_has_version(self, module: str) -> bool:
   341	        if not self.options:
   342	            return True
   343	        version = typeshed_py_version(self.options)
   344	        min_version, max_version = self.stdlib_py_versions[module]
   345	        return version >= min_version and (max_version is None or version <= max_version)

   347	    def _find_module_non_stub_helper(
   348	        self, id: str, pkg_dir: str
   349	    ) -> OnePackageDir | ModuleNotFoundReason:
   350	        plausible_match = False
   351	        dir_path = pkg_dir
   352	        components = id.split(".")
   353	        for index, component in enumerate(components):
   354	            dir_path = os_path_join(dir_path, component)
   355	            if self.fscache.isfile(os_path_join(dir_path, "py.typed")):
   356	                return os.path.join(pkg_dir, *components[:-1]), index == 0
   357	            elif not plausible_match and (
   358	                self.fscache.isdir(dir_path) or self.fscache.isfile(dir_path + ".py")
   359	            ):
   360	                plausible_match = True
   361	            # If this is not a directory then we can't traverse further into it
   362	            if not self.fscache.isdir(dir_path):
   363	                break
   364	        if plausible_match:
   365	            if self.options:
   366	                module_specific_options = self.options.clone_for_module(id)
   367	                if module_specific_options.follow_untyped_imports:
   368	                    return os.path.join(pkg_dir, *components[:-1]), False
   369	            return ModuleNotFoundReason.FOUND_WITHOUT_TYPE_HINTS
   370	        else:
   371	            return ModuleNotFoundReason.NOT_FOUND

   374	        path, verify = match
   375	        for i in range(1, len(components)):

   377	            if pkg_id not in self.ns_ancestors and self.fscache.isdir(path):
   378	                self.ns_ancestors[pkg_id] = path
   379	            path = os.path.dirname(path)

   381	    def _can_find_module_in_parent_dir(self, id: str) -> bool:
   382	        """Test if a module can be found by checking the parent directories
   383	        of the current working directory.
   384	        """
   385	        working_dir = os.getcwd()
   386	        parent_search = FindModuleCache(
   387	            SearchPaths((), (), (), ()),
   388	            self.fscache,
   389	            self.options,
   390	            stdlib_py_versions=self.stdlib_py_versions,
   391	        )
   392	        while any(is_init_file(file) for file in os.listdir(working_dir)):
   393	            working_dir = os.path.dirname(working_dir)
   394	            parent_search.search_paths = SearchPaths((working_dir,), (), (), ())
   395	            if not isinstance(parent_search._find_module(id, False)[0], ModuleNotFoundReason):
   396	                return True
   397	        return False

   399	    def _find_module(self, id: str, use_typeshed: bool) -> tuple[ModuleSearchResult, bool]:
   400	        """Try to find a module in all available sources.
   401
   402	        Returns:
   403	            ``(result, can_be_cached)`` pair.
   404	        """
   405	        fscache = self.fscache
   406
   407	        # Fast path for any modules in the current source set.
   408	        # This is particularly important when there are a large number of search
   409	        # paths which share the first (few) component(s) due to the use of namespace
   410	        # packages, for instance:
   411	        # foo/
   412	        #    company/
   413	        #        __init__.py
   414	        #        foo/
   415	        # bar/
   416	        #    company/
   417	        #        __init__.py
   418	        #        bar/
   419	        # baz/
   420	        #    company/
   421	        #        __init__.py
   422	        #        baz/
   423	        #
   424	        # mypy gets [foo/company/foo, bar/company/bar, baz/company/baz, ...] as input
   425	        # and computes [foo, bar, baz, ...] as the module search path.
   426	        #
   427	        # This would result in O(n) search for every import of company.*, leading to
   428	        # O(n**2) behavior in load_graph as such imports are unsurprisingly present
   429	        # at least once, and usually many more times than that, in each and every file
   430	        # being parsed.
   431	        #
   432	        # Thankfully, such cases are efficiently handled by looking up the module path
   433	        # via BuildSourceSet.
   434	        p = (
   435	            self.find_module_via_source_set(id)
   436	            if (self.options is not None and self.options.fast_module_lookup)
   437	            else None
   438	        )
   439	        if p:
   440	            return p, True
   441
   442	        # If we're looking for a module like 'foo.bar.baz', it's likely that most of the
   443	        # many elements of lib_path don't even have a subdirectory 'foo/bar'.  Discover
   444	        # that only once and cache it for when we look for modules like 'foo.bar.blah'
   445	        # that will require the same subdirectory.
   446	        components = id.split(".")
   447	        dir_chain = os.sep.join(components[:-1])  # e.g., 'foo/bar'
   448
   449	        # We have two sets of folders so that we collect *all* stubs folders and
   450	        # put them in the front of the search path
   451	        third_party_inline_dirs: PackageDirs = []
   452	        third_party_stubs_dirs: PackageDirs = []
   453	        found_possible_third_party_missing_type_hints = False
   454	        # Third-party stub/typed packages
   455	        candidate_package_dirs = {
   456	            package_dir[0]
   457	            for component in (components[0], components[0] + "-stubs")
   458	            for package_dir in self.find_lib_path_dirs(component, self.search_paths.package_path)
   459	        }
   460	        # Caching FOUND_WITHOUT_TYPE_HINTS is not always safe. That causes issues with
   461	        # typed subpackages in namespace packages.
   462	        can_cache_any_result = True
   463	        for pkg_dir in self.search_paths.package_path:
   464	            if pkg_dir not in candidate_package_dirs:
   465	                continue
   466	            stub_name = components[0] + "-stubs"
   467	            stub_dir = os_path_join(pkg_dir, stub_name)
   468	            if fscache.isdir(stub_dir):
   469	                stub_typed_file = os_path_join(stub_dir, "py.typed")
   470	                stub_components = [stub_name] + components[1:]
   471	                path = os.path.join(pkg_dir, *stub_components[:-1])
   472	                if fscache.isdir(path):
   473	                    if fscache.isfile(stub_typed_file):
   474	                        # Stub packages can have a py.typed file, which must include
   475	                        # 'partial\n' to make the package partial
   476	                        # Partial here means that mypy should look at the runtime
   477	                        # package if installed.
   478	                        if fscache.read(stub_typed_file).decode().strip() == "partial":
   479	                            runtime_path = os_path_join(pkg_dir, dir_chain)
   480	                            third_party_inline_dirs.append((runtime_path, True))
   481	                            # if the package is partial, we don't verify the module, as
   482	                            # the partial stub package may not have a __init__.pyi
   483	                            third_party_stubs_dirs.append((path, False))
   484	                        else:
   485	                            # handle the edge case where people put a py.typed file
   486	                            # in a stub package, but it isn't partial
   487	                            third_party_stubs_dirs.append((path, True))
   488	                    else:
   489	                        third_party_stubs_dirs.append((path, True))
   490	            non_stub_match = self._find_module_non_stub_helper(id, pkg_dir)
   491	            if isinstance(non_stub_match, ModuleNotFoundReason):
   492	                if non_stub_match is ModuleNotFoundReason.FOUND_WITHOUT_TYPE_HINTS:
   493	                    found_possible_third_party_missing_type_hints = True
   494	                    can_cache_any_result = False
   495	            else:
   496	                third_party_inline_dirs.append(non_stub_match)
   497	                self._update_ns_ancestors(components, non_stub_match)
   498
   499	        if self.options and self.options.use_builtins_fixtures:
   500	            # Everything should be in fixtures.
   501	            third_party_inline_dirs.clear()
   502	            third_party_stubs_dirs.clear()
   503	            found_possible_third_party_missing_type_hints = False
   504	        python_mypy_path = self.search_paths.mypy_path + self.search_paths.python_path
   505	        candidate_base_dirs = self.find_lib_path_dirs(id, python_mypy_path)
   506	        if use_typeshed:
   507	            # Search for stdlib stubs in typeshed before installed
   508	            # stubs to avoid picking up backports (dataclasses, for
   509	            # example) when the library is included in stdlib.
   510	            candidate_base_dirs += self.find_lib_path_dirs(id, self.search_paths.typeshed_path)
   511	        candidate_base_dirs += third_party_stubs_dirs + third_party_inline_dirs
   512
   513	        # If we're looking for a module like 'foo.bar.baz', then candidate_base_dirs now
   514	        # contains just the subdirectories 'foo/bar' that actually exist under the
   515	        # elements of lib_path.  This is probably much shorter than lib_path itself.
   516	        # Now just look for 'baz.pyi', 'baz/__init__.py', etc., inside those directories.
   517	        seplast = os.sep + components[-1]  # so e.g. '/baz'
   518	        sepinit = os.sep + "__init__"
   519	        near_misses = []  # Collect near misses for namespace mode (see below).
   520	        for base_dir, verify in candidate_base_dirs:
   521	            base_path = base_dir + seplast  # so e.g. '/usr/lib/python3.4/foo/bar/baz'
   522	            has_init = False
   523	            dir_prefix = base_dir
   524	            for _ in range(len(components) - 1):
   525	                dir_prefix = os.path.dirname(dir_prefix)
   526
   527	            # Stubs-only packages always take precedence over py.typed packages
   528	            path_stubs = f"{base_path}-stubs{sepinit}.pyi"
   529	            if fscache.isfile_case(path_stubs, dir_prefix):
   530	                if verify and not verify_module(fscache, id, path_stubs, dir_prefix):
   531	                    near_misses.append((path_stubs, dir_prefix))
   532	                else:
   533	                    return path_stubs, True
   534
   535	            # Prefer package over module, i.e. baz/__init__.py* over baz.py*.
   536	            for extension in PYTHON_EXTENSIONS:
   537	                path = base_path + sepinit + extension
   538	                if fscache.isfile_case(path, dir_prefix):
   539	                    has_init = True
   540	                    if verify and not verify_module(fscache, id, path, dir_prefix):
   541	                        near_misses.append((path, dir_prefix))
   542	                        continue
   543	                    return path, True
   544
   545	            # In namespace mode, register a potential namespace package
   546	            if self.options and self.options.namespace_packages:
   547	                if (
   548	                    not has_init
   549	                    and fscache.exists_case(base_path, dir_prefix)
   550	                    and not fscache.isfile_case(base_path, dir_prefix)
   551	                ):
   552	                    near_misses.append((base_path, dir_prefix))
   553
   554	            # No package, look for module.
   555	            for extension in PYTHON_EXTENSIONS:
   556	                path = base_path + extension
   557	                if fscache.isfile_case(path, dir_prefix):
   558	                    if verify and not verify_module(fscache, id, path, dir_prefix):
   559	                        near_misses.append((path, dir_prefix))
   560	                        continue
   561	                    return path, True
   562
   563	        # In namespace mode, re-check those entries that had 'verify'.
   564	        # Assume search path entries xxx, yyy and zzz, and we're
   565	        # looking for foo.bar.baz.  Suppose near_misses has:
   566	        #
   567	        # - xxx/foo/bar/baz.py
   568	        # - yyy/foo/bar/baz/__init__.py
   569	        # - zzz/foo/bar/baz.pyi
   570	        #
   571	        # If any of the foo directories has __init__.py[i], it wins.
   572	        # Else, we look for foo/bar/__init__.py[i], etc.  If there are
   573	        # none, the first hit wins.  Note that this does not take into
   574	        # account whether the lowest-level module is a file (baz.py),
   575	        # a package (baz/__init__.py), or a stub file (baz.pyi) -- for
   576	        # these the first one encountered along the search path wins.
   577	        #
   578	        # The helper function highest_init_level() returns an int that
   579	        # indicates the highest level at which a __init__.py[i] file
   580	        # is found; if no __init__ was found it returns 0, if we find
   581	        # only foo/bar/__init__.py it returns 1, and if we have
   582	        # foo/__init__.py it returns 2 (regardless of what's in
   583	        # foo/bar).  It doesn't look higher than that.
   584	        if self.options and self.options.namespace_packages and near_misses:
   585	            levels = [
   586	                highest_init_level(fscache, id, path, dir_prefix)
   587	                for path, dir_prefix in near_misses
   588	            ]
   589	            index = levels.index(max(levels))
   590	            return near_misses[index][0], True
   591
   592	        # Finally, we may be asked to produce an ancestor for an
   593	        # installed package with a py.typed marker that is a
   594	        # subpackage of a namespace package.  We only fess up to these
   595	        # if we would otherwise return "not found".
   596	        ancestor = self.ns_ancestors.get(id)
   597	        if ancestor is not None:
   598	            return ancestor, True
   599
   600	        approved_dist_name = stub_distribution_name(id)
   601	        if approved_dist_name:
   602	            if len(components) == 1:
   603	                return ModuleNotFoundReason.APPROVED_STUBS_NOT_INSTALLED, True
   604	            # If we're a missing submodule of an already installed approved stubs, we don't want to
   605	            # error with APPROVED_STUBS_NOT_INSTALLED, but rather want to return NOT_FOUND.
   606	            for i in range(1, len(components)):
   607	                parent_id = ".".join(components[:i])
   608	                if stub_distribution_name(parent_id) == approved_dist_name:
   609	                    break
   610	            else:
   611	                return ModuleNotFoundReason.APPROVED_STUBS_NOT_INSTALLED, True
   612	            if self.find_module(parent_id) is ModuleNotFoundReason.APPROVED_STUBS_NOT_INSTALLED:
   613	                return ModuleNotFoundReason.APPROVED_STUBS_NOT_INSTALLED, True
   614	            return ModuleNotFoundReason.NOT_FOUND, True
   615
   616	        if found_possible_third_party_missing_type_hints:
   617	            return ModuleNotFoundReason.FOUND_WITHOUT_TYPE_HINTS, can_cache_any_result
   618	        return ModuleNotFoundReason.NOT_FOUND, True
   619
   620	    def find_modules_recursive(self, module: str) -> list[BuildSource]:
   621	        module_path = self.find_module(module, fast_path=True)
   622	        if isinstance(module_path, ModuleNotFoundReason):
   623	            return []
   624	        sources = [BuildSource(module_path, module, None)]
   625
   626	        package_path = None
   627	        if is_init_file(module_path):
   628	            package_path = os.path.dirname(module_path)
   629	        elif self.fscache.isdir(module_path):
   630	            package_path = module_path
   631	        if package_path is None:
   632	            return sources
   633
   634	        # This logic closely mirrors that in find_sources. One small but important difference is
   635	        # that we do not sort names with keyfunc. The recursive call to find_modules_recursive
   636	        # calls find_module, which will handle the preference between packages, pyi and py.
   637	        # Another difference is it doesn't handle nested search paths / package roots.
   638
   639	        seen: set[str] = set()
   640	        names = sorted(self.fscache.listdir(package_path))
   641	        for name in names:
   642	            # Skip certain names altogether
   643	            if name in ("__pycache__", "site-packages", "node_modules") or name.startswith("."):
   644	                continue
   645	            subpath = os_path_join(package_path, name)
   646
   647	            if self.options and matches_exclude(
   648	                subpath, self.options.exclude, self.fscache, self.options.verbosity >= 2
   649	            ):
   650	                continue
   651	            if (
   652	                self.options
   653	                and self.options.exclude_gitignore
   654	                and matches_gitignore(subpath, self.fscache, self.options.verbosity >= 2)
   655	            ):
   656	                continue
   657
   658	            if self.fscache.isdir(subpath):
   659	                # Only recurse into packages
   660	                if (self.options and self.options.namespace_packages) or (
   661	                    self.fscache.isfile(os_path_join(subpath, "__init__.py"))
   662	                    or self.fscache.isfile(os_path_join(subpath, "__init__.pyi"))
   663	                ):
   664	                    seen.add(name)
   665	                    sources.extend(self.find_modules_recursive(module + "." + name))
   666	            else:
   667	                stem, suffix = os.path.splitext(name)
   668	                if stem == "__init__":
   669	                    continue
   670	                if stem not in seen and "." not in stem and suffix in PYTHON_EXTENSIONS:
   671	                    # (If we sorted names by keyfunc) we could probably just make the BuildSource
   672	                    # ourselves, but this ensures compatibility with find_module / the cache
   673	                    seen.add(stem)
   674	                    sources.extend(self.find_modules_recursive(module + "." + stem))
   675	        return sources

   678	def matches_exclude(
   679	    subpath: str, excludes: list[str], fscache: FileSystemCache, verbose: bool
   680	) -> bool:
   681	    if not excludes:
   682	        return False
   683	    subpath_str = os.path.relpath(subpath).replace(os.sep, "/")
   684	    if fscache.isdir(subpath):
   685	        subpath_str += "/"
   686	    for exclude in excludes:
   687	        try:
   688	            if re.search(exclude, subpath_str):
   689	                if verbose:
   690	                    print(
   691	                        f"TRACE: Excluding {subpath_str} (matches pattern {exclude})",
   692	                        file=sys.stderr,
   693	                    )
   694	                return True
   695	        except re.error as e:
   696	            print(
   697	                f"error: The exclude {exclude} is an invalid regular expression, because: {e}"
   698	                + (
   699	                    "\n(Hint: use / as a path separator, even if you're on Windows!)"
   700	                    if "\\" in exclude
   701	                    else ""
   702	                )
   703	                + "\nFor more information on Python's flavor of regex, see:"
   704	                + " https://docs.python.org/3/library/re.html",
   705	                file=sys.stderr,
   706	            )
   707	            sys.exit(2)
   708	    return False

   711	def matches_gitignore(subpath: str, fscache: FileSystemCache, verbose: bool) -> bool:
   712	    dir, _ = os.path.split(subpath)
   713	    for gi_path, gi_spec in find_gitignores(dir):
   714	        relative_path = os.path.relpath(subpath, gi_path)
   715	        if fscache.isdir(relative_path):
   716	            relative_path = relative_path + "/"
   717	        if gi_spec.match_file(relative_path):
   718	            if verbose:
   719	                print(
   720	                    f"TRACE: Excluding {relative_path} (matches .gitignore) in {gi_path}",
   721	                    file=sys.stderr,
   722	                )
   723	            return True
   724	    return False

   727	@functools.lru_cache
   728	def find_gitignores(dir: str) -> list[tuple[str, PathSpec]]:
   729	    parent_dir = os.path.dirname(dir)
   730	    if parent_dir == dir:
   731	        parent_gitignores = []
   732	    else:
   733	        parent_gitignores = find_gitignores(parent_dir)
   734
   735	    gitignore = os.path.join(dir, ".gitignore")
   736	    if os.path.isfile(gitignore):
   737	        with open(gitignore) as f:
   738	            lines = f.readlines()
   739	        try:
   740	            return parent_gitignores + [(dir, PathSpec.from_lines("gitwildmatch", lines))]
   741	        except GitWildMatchPatternError:
   742	            print(f"error: could not parse {gitignore}", file=sys.stderr)
   743	            return parent_gitignores
   744	    return parent_gitignores

   751	def verify_module(fscache: FileSystemCache, id: str, path: str, prefix: str) -> bool:
   752	    """Check that all packages containing id have a __init__ file."""
   753	    if is_init_file(path):
   754	        path = os.path.dirname(path)
   755	    for i in range(id.count(".")):
   756	        path = os.path.dirname(path)
   757	        if not any(
   758	            fscache.isfile_case(os_path_join(path, f"__init__{extension}"), prefix)
   759	            for extension in PYTHON_EXTENSIONS
   760	        ):
   761	            return False
   762	    return True

   765	def highest_init_level(fscache: FileSystemCache, id: str, path: str, prefix: str) -> int:
   766	    """Compute the highest level where an __init__ file is found."""
   767	    if is_init_file(path):
   768	        path = os.path.dirname(path)
   769	    level = 0
   770	    for i in range(id.count(".")):
   771	        path = os.path.dirname(path)
   772	        if any(
   773	            fscache.isfile_case(os_path_join(path, f"__init__{extension}"), prefix)
   774	            for extension in PYTHON_EXTENSIONS
   775	        ):
   776	            level = i + 1
   777	    return level

   106	class ModuleInspect:
   107	    """Perform runtime introspection of modules in a separate process.
   108
   109	    Reuse the process for multiple modules for efficiency. However, if there is an
   110	    error, retry using a fresh process to avoid cross-contamination of state between
   111	    modules.
   112
   113	    We use a separate process to isolate us from many side effects. For example, the
   114	    import of a module may kill the current process, and we want to recover from that.
   115
   116	    Always use in a with statement for proper clean-up:
   117
   118	      with ModuleInspect() as m:
   119	          p = m.get_package_properties('urllib.parse')
   120	    """
   121
   122	    def __init__(self) -> None:
   123	        self._start()
   124
   125	    def _start(self) -> None:
   126	        if sys.platform == "linux":
   127	            ctx = get_context("forkserver")
   128	        else:
   129	            ctx = get_context("spawn")
   130	        self.tasks: Queue[str] = ctx.Queue()
   131	        self.results: Queue[ModuleProperties | str] = ctx.Queue()
   132	        self.proc = ctx.Process(target=worker, args=(self.tasks, self.results, sys.path))
   133	        self.proc.start()
   134	        self.counter = 0  # Number of successful roundtrips
   135
   136	    def close(self) -> None:
   137	        """Free any resources used."""
   138	        self.proc.terminate()
   139
   140	    def get_package_properties(self, package_id: str) -> ModuleProperties:
   141	        """Return some properties of a module/package using runtime introspection.
   142
   143	        Raise InspectError if the target couldn't be imported.
   144	        """
   145	        self.tasks.put(package_id)
   146	        res = self._get_from_queue()
   147	        if res is None:
   148	            # The process died; recover and report error.
   149	            self._start()
   150	            raise InspectError(f"Process died when importing {package_id!r}")
   151	        if isinstance(res, str):
   152	            # Error importing module
   153	            if self.counter > 0:
   154	                # Also try with a fresh process. Maybe one of the previous imports has
   155	                # corrupted some global state.
   156	                self.close()
   157	                self._start()
   158	                return self.get_package_properties(package_id)
   159	            raise InspectError(res)
   160	        self.counter += 1
   161	        return res
   162
   163	    def _get_from_queue(self) -> ModuleProperties | str | None:
   164	        """Get value from the queue.
   165
   166	        Return the value read from the queue, or None if the process unexpectedly died.
   167	        """
   168	        max_iter = 600
   169	        n = 0
   170	        while True:
   171	            if n == max_iter:
   172	                raise RuntimeError("Timeout waiting for subprocess")

   174	                return self.results.get(timeout=0.05)
   175	            except queue.Empty:
   176	                if not self.proc.is_alive():
   177	                    return None
   178	            n += 1

   248	class MypyFile(SymbolNode):
   249	    """The abstract syntax tree of a single source file."""
   250
   251	    __slots__ = (
   252	        "_fullname",
   253	        "path",
   254	        "defs",
   255	        "alias_deps",
   256	        "is_bom",
   257	        "names",
   258	        "imports",
   259	        "ignored_lines",
   260	        "skipped_lines",
   261	        "is_stub",
   262	        "is_cache_skeleton",
   263	        "is_partial_stub_package",
   264	        "plugin_deps",
   265	        "future_import_flags",
   266	        "_is_typeshed_file",
   267	    )
   268
   269	    __match_args__ = ("name", "path", "defs")
   270
   271	    # Fully qualified module name
   272	    _fullname: str
   273	    # Path to the file (empty string if not known)
   274	    path: str
   275	    # Top-level definitions and statements
   276	    defs: list[Statement]
   277	    # Type alias dependencies as mapping from target to set of alias full names
   278	    alias_deps: defaultdict[str, set[str]]
   279	    # Is there a UTF-8 BOM at the start?
   280	    is_bom: bool
   281	    names: SymbolTable
   282	    # All import nodes within the file (also ones within functions etc.)
   283	    imports: list[ImportBase]
   284	    # Lines on which to ignore certain errors when checking.
   285	    # If the value is empty, ignore all errors; otherwise, the list contains all
   286	    # error codes to ignore.
   287	    ignored_lines: dict[int, list[str]]
   288	    # Lines that were skipped during semantic analysis e.g. due to ALWAYS_FALSE, MYPY_FALSE,
   289	    # or platform/version checks. Those lines would not be type-checked.
   290	    skipped_lines: set[int]
   291	    # Is this file represented by a stub file (.pyi)?
   292	    is_stub: bool
   293	    # Is this loaded from the cache and thus missing the actual body of the file?
   294	    is_cache_skeleton: bool
   295	    # Does this represent an __init__.pyi stub with a module __getattr__
   296	    # (i.e. a partial stub package), for such packages we suppress any missing
   297	    # module errors in addition to missing attribute errors.
   298	    is_partial_stub_package: bool
   299	    # Plugin-created dependencies
   300	    plugin_deps: dict[str, set[str]]
   301	    # Future imports defined in this file. Populated during semantic analysis.
   302	    future_import_flags: set[str]
   303	    _is_typeshed_file: bool | None
   304
   305	    def __init__(
   306	        self,
   307	        defs: list[Statement],
   308	        imports: list[ImportBase],
   309	        is_bom: bool = False,
   310	        ignored_lines: dict[int, list[str]] | None = None,
   311	    ) -> None:
   312	        super().__init__()
   313	        self.defs = defs
   314	        self.line = 1  # Dummy line number
   315	        self.column = 0  # Dummy column
   316	        self.imports = imports
   317	        self.is_bom = is_bom
   318	        self.alias_deps = defaultdict(set)
   319	        self.plugin_deps = {}
   320	        if ignored_lines:
   321	            self.ignored_lines = ignored_lines
   322	        else:
   323	            self.ignored_lines = {}
   324	        self.skipped_lines = set()
   325
   326	        self.path = ""
   327	        self.is_stub = False
   328	        self.is_cache_skeleton = False
   329	        self.is_partial_stub_package = False
   330	        self.future_import_flags = set()
   331	        self._is_typeshed_file = None
   332
   333	    def local_definitions(self) -> Iterator[Definition]:
   334	        """Return all definitions within the module (including nested).
   335
   336	        This doesn't include imported definitions.
   337	        """
   338	        return local_definitions(self.names, self.fullname)
   339
   340	    @property
   341	    def name(self) -> str:
   342	        return "" if not self._fullname else self._fullname.split(".")[-1]
   343
   344	    @property
   345	    def fullname(self) -> str:
   346	        return self._fullname
   347
   348	    def accept(self, visitor: NodeVisitor[T]) -> T:
   349	        return visitor.visit_mypy_file(self)
   350
   351	    def is_package_init_file(self) -> bool:
   352	        return len(self.path) != 0 and os.path.basename(self.path).startswith("__init__.")
   353
   354	    def is_future_flag_set(self, flag: str) -> bool:
   355	        return flag in self.future_import_flags
   356
   357	    def is_typeshed_file(self, options: Options) -> bool:
   358	        # Cache result since this is called a lot
   359	        if self._is_typeshed_file is None:
   360	            self._is_typeshed_file = is_typeshed_file(options.abs_custom_typeshed_dir, self.path)
   361	        return self._is_typeshed_file
   362
   363	    def serialize(self) -> JsonDict:
   364	        return {
   365	            ".class": "MypyFile",
   366	            "_fullname": self._fullname,
   367	            "names": self.names.serialize(self._fullname),
   368	            "is_stub": self.is_stub,
   369	            "path": self.path,
   370	            "is_partial_stub_package": self.is_partial_stub_package,
   371	            "future_import_flags": list(self.future_import_flags),
   372	        }
   373
   374	    @classmethod
   375	    def deserialize(cls, data: JsonDict) -> MypyFile:
   376	        assert data[".class"] == "MypyFile", data
   377	        tree = MypyFile([], [])
   378	        tree._fullname = data["_fullname"]
   379	        tree.names = SymbolTable.deserialize(data["names"])
   380	        tree.is_stub = data["is_stub"]
   381	        tree.path = data["path"]
   382	        tree.is_partial_stub_package = data["is_partial_stub_package"]
   383	        tree.is_cache_skeleton = True
   384	        tree.future_import_flags = set(data["future_import_flags"])
   385	        return tree

   531	class OverloadedFuncDef(FuncBase, SymbolNode, Statement):
   532	    """A logical node representing all the variants of a multi-declaration function.
   533
   534	    A multi-declaration function is often an @overload, but can also be a
   535	    @property with a setter and a/or a deleter.
   536
   537	    This node has no explicit representation in the source program.
   538	    Overloaded variants must be consecutive in the source file.
   539	    """
   540
   541	    __slots__ = (
   542	        "items",
   543	        "unanalyzed_items",
   544	        "impl",
   545	        "deprecated",
   546	        "setter_index",
   547	        "_is_trivial_self",
   548	    )
   549
   550	    items: list[OverloadPart]
   551	    unanalyzed_items: list[OverloadPart]
   552	    impl: OverloadPart | None
   553	    deprecated: str | None
   554	    setter_index: int | None
   555
   556	    def __init__(self, items: list[OverloadPart]) -> None:
   557	        super().__init__()
   558	        self.items = items
   559	        self.unanalyzed_items = items.copy()
   560	        self.impl = None
   561	        self.deprecated = None
   562	        self.setter_index = None
   563	        self._is_trivial_self: bool | None = None
   564	        if items:
   565	            # TODO: figure out how to reliably set end position (we don't know the impl here).
   566	            self.set_line(items[0].line, items[0].column)
   567
   568	    @property
   569	    def name(self) -> str:
   570	        if self.items:
   571	            return self.items[0].name
   572	        else:
   573	            # This may happen for malformed overload
   574	            assert self.impl is not None
   575	            return self.impl.name
   576
   577	    @property
   578	    def is_trivial_self(self) -> bool:
   579	        """Check we can use bind_self() fast path for this overload.
   580
   581	        This will return False if at least one overload:
   582	          * Has an explicit self annotation, or Self in signature.
   583	          * Has a non-trivial decorator.
   584	        """
   585	        if self._is_trivial_self is not None:
   586	            return self._is_trivial_self
   587	        for item in self.items:
   588	            if isinstance(item, FuncDef):
   589	                if not item.is_trivial_self:
   590	                    self._is_trivial_self = False
   591	                    return False
   592	            elif item.decorators or not item.func.is_trivial_self:
   593	                self._is_trivial_self = False
   594	                return False
   595	        self._is_trivial_self = True
   596	        return True
   597
   598	    @property
   599	    def setter(self) -> Decorator:
   600	        # Do some consistency checks first.
   601	        first_item = self.items[0]
   602	        assert isinstance(first_item, Decorator)
   603	        assert first_item.var.is_settable_property
   604	        assert self.setter_index is not None
   605	        item = self.items[self.setter_index]
   606	        assert isinstance(item, Decorator)
   607	        return item
   608
   609	    def accept(self, visitor: StatementVisitor[T]) -> T:
   610	        return visitor.visit_overloaded_func_def(self)
   611
   612	    def serialize(self) -> JsonDict:
   613	        return {
   614	            ".class": "OverloadedFuncDef",
   615	            "items": [i.serialize() for i in self.items],
   616	            "type": None if self.type is None else self.type.serialize(),
   617	            "fullname": self._fullname,
   618	            "impl": None if self.impl is None else self.impl.serialize(),
   619	            "flags": get_flags(self, FUNCBASE_FLAGS),
   620	            "deprecated": self.deprecated,
   621	            "setter_index": self.setter_index,
   622	        }
   623
   624	    @classmethod
   625	    def deserialize(cls, data: JsonDict) -> OverloadedFuncDef:
   626	        assert data[".class"] == "OverloadedFuncDef"
   627	        res = OverloadedFuncDef(
   628	            [cast(OverloadPart, SymbolNode.deserialize(d)) for d in data["items"]]
   629	        )
   630	        if data.get("impl") is not None:
   631	            res.impl = cast(OverloadPart, SymbolNode.deserialize(data["impl"]))
   632	            # set line for empty overload items, as not set in __init__
   633	            if len(res.items) > 0:
   634	                res.set_line(res.impl.line)
   635	        if data.get("type") is not None:
   636	            typ = mypy.types.deserialize_type(data["type"])
   637	            assert isinstance(typ, mypy.types.ProperType)
   638	            res.type = typ
   639	        res._fullname = data["fullname"]
   640	        set_flags(res, data["flags"])
   641	        res.deprecated = data["deprecated"]
   642	        res.setter_index = data["setter_index"]
   643	        # NOTE: res.info will be set in the fixup phase.
   644	        return res
   645
   646	    def is_dynamic(self) -> bool:
   647	        return all(item.is_dynamic() for item in self.items)

   731	        "body",  # Body of the function
   732	        "is_overload",  # Is this an overload variant of function with more than
   733	        # one overload variant?
   734	        "is_generator",  # Contains a yield statement?
   735	        "is_coroutine",  # Defined using 'async def' syntax?
   736	        "is_async_generator",  # Is an async def generator?

   738	        "expanded",  # Variants of function with type variables with values expanded

   741	    __deletable__ = ("arguments", "max_pos", "min_args")

   743	    def __init__(
   744	        self,
   745	        arguments: list[Argument] | None = None,
   746	        body: Block | None = None,
   747	        typ: mypy.types.FunctionLike | None = None,
   748	        type_args: list[TypeParam] | None = None,
   749	    ) -> None:
   750	        super().__init__()
   751	        self.arguments = arguments or []
   752	        self.arg_names = [None if arg.pos_only else arg.variable.name for arg in self.arguments]
   753	        self.arg_kinds: list[ArgKind] = [arg.kind for arg in self.arguments]
   754	        self.max_pos: int = self.arg_kinds.count(ARG_POS) + self.arg_kinds.count(ARG_OPT)
   755	        self.type_args: list[TypeParam] | None = type_args
   756	        self.body: Block = body or Block([])
   757	        self.type = typ
   758	        self.unanalyzed_type = typ
   759	        self.is_overload: bool = False
   760	        self.is_generator: bool = False
   761	        self.is_coroutine: bool = False
   762	        self.is_async_generator: bool = False
   763	        self.is_awaitable_coroutine: bool = False
   764	        self.expanded: list[FuncItem] = []

   766	        self.min_args = 0
   767	        for i in range(len(self.arguments)):
   768	            if self.arguments[i] is None and i < self.max_fixed_argc():
   769	                self.min_args = i + 1

   772	        return self.max_pos

   774	    def is_dynamic(self) -> bool:
   775	        return self.type is None

   795	    """Function definition.
   796
   797	    This is a non-lambda function defined using 'def'.
   798	    """
   799
   800	    __slots__ = (
   801	        "_name",
   802	        "is_decorated",
   803	        "is_conditional",
   804	        "abstract_status",
   805	        "original_def",
   806	        "is_trivial_body",
   807	        "is_trivial_self",
   808	        "is_mypy_only",
   809	        # Present only when a function is decorated with @typing.dataclass_transform or similar
   810	        "dataclass_transform_spec",
   811	        "docstring",
   812	        "deprecated",
   813	    )
   814
   815	    __match_args__ = ("name", "arguments", "type", "body")
   816
   817	    # Note that all __init__ args must have default values
   818	    def __init__(
   819	        self,
   820	        name: str = "",  # Function name
   821	        arguments: list[Argument] | None = None,
   822	        body: Block | None = None,
   823	        typ: mypy.types.FunctionLike | None = None,
   824	        type_args: list[TypeParam] | None = None,
   825	    ) -> None:
   826	        super().__init__(arguments, body, typ, type_args)
   827	        self._name = name
   828	        self.is_decorated = False
   829	        self.is_conditional = False  # Defined conditionally (within block)?
   830	        self.abstract_status = NOT_ABSTRACT
   831	        # Is this an abstract method with trivial body?
   832	        # Such methods can't be called via super().
   833	        self.is_trivial_body = False
   834	        # Original conditional definition
   835	        self.original_def: None | FuncDef | Var | Decorator = None
   836	        # Definitions that appear in if TYPE_CHECKING are marked with this flag.
   837	        self.is_mypy_only = False
   838	        self.dataclass_transform_spec: DataclassTransformSpec | None = None
   839	        self.docstring: str | None = None
   840	        self.deprecated: str | None = None
   841	        # This is used to simplify bind_self() logic in trivial cases (which are
   842	        # the majority). In cases where self is not annotated and there are no Self
   843	        # in the signature we can simply drop the first argument.
   844	        self.is_trivial_self = False

   846	    @property
   847	    def name(self) -> str:
   848	        return self._name
   849
   850	    def accept(self, visitor: StatementVisitor[T]) -> T:
   851	        return visitor.visit_func_def(self)
   852
   853	    def serialize(self) -> JsonDict:
   854	        # We're deliberating omitting arguments and storing only arg_names and
   855	        # arg_kinds for space-saving reasons (arguments is not used in later
   856	        # stages of mypy).
   857	        # TODO: After a FuncDef is deserialized, the only time we use `arg_names`
   858	        # and `arg_kinds` is when `type` is None and we need to infer a type. Can
   859	        # we store the inferred type ahead of time?
   860	        return {
   861	            ".class": "FuncDef",
   862	            "name": self._name,
   863	            "fullname": self._fullname,
   864	            "arg_names": self.arg_names,
   865	            "arg_kinds": [int(x.value) for x in self.arg_kinds],
   866	            "type": None if self.type is None else self.type.serialize(),
   867	            "flags": get_flags(self, FUNCDEF_FLAGS),
   868	            "abstract_status": self.abstract_status,
   869	            # TODO: Do we need expanded, original_def?
   870	            "dataclass_transform_spec": (
   871	                None
   872	                if self.dataclass_transform_spec is None
   873	                else self.dataclass_transform_spec.serialize()
   874	            ),
   875	            "deprecated": self.deprecated,
   876	        }
   877
   878	    @classmethod
   879	    def deserialize(cls, data: JsonDict) -> FuncDef:
   880	        assert data[".class"] == "FuncDef"
   881	        body = Block([])
   882	        ret = FuncDef(
   883	            data["name"],
   884	            [],
   885	            body,
   886	            (
   887	                None
   888	                if data["type"] is None
   889	                else cast(mypy.types.FunctionLike, mypy.types.deserialize_type(data["type"]))
   890	            ),
   891	        )
   892	        ret._fullname = data["fullname"]
   893	        set_flags(ret, data["flags"])
   894	        # NOTE: ret.info is set in the fixup phase.
   895	        ret.arg_names = data["arg_names"]
   896	        ret.arg_kinds = [ArgKind(x) for x in data["arg_kinds"]]
   897	        ret.abstract_status = data["abstract_status"]
   898	        ret.dataclass_transform_spec = (
   899	            DataclassTransformSpec.deserialize(data["dataclass_transform_spec"])
   900	            if data["dataclass_transform_spec"] is not None
   901	            else None
   902	        )
   903	        ret.deprecated = data["deprecated"]
   904	        # Leave these uninitialized so that future uses will trigger an error
   905	        del ret.arguments
   906	        del ret.max_pos
   907	        del ret.min_args
   908	        return ret

  1347	    that define new names, and assignments with explicit types ("# type: t"
  1348	    or "x: t [= ...]").

  1354	        "lvalues",

  1369	    # Declared type in a comment, may be None.

  1371	    # Original, not semantically analyzed type in annotation (used for reprocessing)
  1372	    unanalyzed_type: mypy.types.Type | None
