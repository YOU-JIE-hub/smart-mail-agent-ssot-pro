   139	def slot_key(attr: str) -> str:
   140	    """Map dunder method name to sort key.
   141
   142	    Sort reverse operator methods and __delitem__ after others ('x' > '_').
   143	    """
   144	    if (attr.startswith("__r") and attr != "__rshift__") or attr == "__delitem__":
   145	        return "x" + attr
   146	    return attr

   149	def generate_slots(cl: ClassIR, table: SlotTable, emitter: Emitter) -> dict[str, str]:
   150	    fields: dict[str, str] = {}
   151	    generated: dict[str, str] = {}
   152	    # Sort for determinism on Python 3.5
   153	    for name, (slot, generator) in sorted(table.items(), key=lambda x: slot_key(x[0])):
   154	        method_cls = cl.get_method_and_class(name)
   155	        if method_cls and (method_cls[1] == cl or name in ALWAYS_FILL):
   156	            if slot in generated:
   157	                # Reuse previously generated wrapper.
   158	                fields[slot] = generated[slot]
   159	            else:
   160	                # Generate new wrapper.
   161	                name = generator(cl, method_cls[0], emitter)
   162	                fields[slot] = name
   163	                generated[slot] = name
   164
   165	    return fields

   189	def generate_class(cl: ClassIR, module: str, emitter: Emitter) -> None:
   190	    """Generate C code for a class.
   191
   192	    This is the main entry point to the module.
   193	    """
   194	    name = cl.name
   195	    name_prefix = cl.name_prefix(emitter.names)
   196
   197	    setup_name = f"{name_prefix}_setup"
   198	    new_name = f"{name_prefix}_new"
   199	    finalize_name = f"{name_prefix}_finalize"
   200	    members_name = f"{name_prefix}_members"
   201	    getseters_name = f"{name_prefix}_getseters"
   202	    vtable_name = f"{name_prefix}_vtable"
   203	    traverse_name = f"{name_prefix}_traverse"
   204	    clear_name = f"{name_prefix}_clear"
   205	    dealloc_name = f"{name_prefix}_dealloc"
   206	    methods_name = f"{name_prefix}_methods"
   207	    vtable_setup_name = f"{name_prefix}_trait_vtable_setup"
   208
   209	    fields: dict[str, str] = {"tp_name": f'"{name}"'}
   210
   211	    generate_full = not cl.is_trait and not cl.builtin_base
   212	    needs_getseters = cl.needs_getseters or not cl.is_generated or cl.has_dict
   213
   214	    if not cl.builtin_base:
   215	        fields["tp_new"] = new_name
   216
   217	    if generate_full:
   218	        fields["tp_dealloc"] = f"(destructor){name_prefix}_dealloc"
   219	        fields["tp_traverse"] = f"(traverseproc){name_prefix}_traverse"
   220	        fields["tp_clear"] = f"(inquiry){name_prefix}_clear"
   221	    # Populate .tp_finalize and generate a finalize method only if __del__ is defined for this class.
   222	    del_method = next((e.method for e in cl.vtable_entries if e.name == "__del__"), None)
   223	    if del_method:
   224	        fields["tp_finalize"] = f"(destructor){finalize_name}"
   225	    if needs_getseters:
   226	        fields["tp_getset"] = getseters_name
   227	    fields["tp_methods"] = methods_name
   228
   229	    def emit_line() -> None:
   230	        emitter.emit_line()
   231
   232	    emit_line()
   233
   234	    # If the class has a method to initialize default attribute
   235	    # values, we need to call it during initialization.
   236	    defaults_fn = cl.get_method("__mypyc_defaults_setup")
   237
   238	    # If there is a __init__ method, we'll use it in the native constructor.
   239	    init_fn = cl.get_method("__init__")
   240
   241	    # Fill out slots in the type object from dunder methods.
   242	    fields.update(generate_slots(cl, SLOT_DEFS, emitter))
   243
   244	    # Fill out dunder methods that live in tables hanging off the side.
   245	    for table_name, type, slot_defs in SIDE_TABLES:
   246	        slots = generate_slots(cl, slot_defs, emitter)
   247	        if slots:
   248	            table_struct_name = generate_side_table_for_class(cl, table_name, type, slots, emitter)
   249	            fields[f"tp_{table_name}"] = f"&{table_struct_name}"
   250
   251	    richcompare_name = generate_richcompare_wrapper(cl, emitter)
   252	    if richcompare_name:
   253	        fields["tp_richcompare"] = richcompare_name
   254
   255	    # If the class inherits from python, make space for a __dict__
   256	    struct_name = cl.struct_name(emitter.names)
   257	    if cl.builtin_base:
   258	        base_size = f"sizeof({cl.builtin_base})"
   259	    elif cl.is_trait:
   260	        base_size = "sizeof(PyObject)"
   261	    else:
   262	        base_size = f"sizeof({struct_name})"
   263	    # Since our types aren't allocated using type() we need to
   264	    # populate these fields ourselves if we want them to have correct
   265	    # values. PyType_Ready will inherit the offsets from tp_base but
   266	    # that isn't what we want.
   267
   268	    # XXX: there is no reason for the __weakref__ stuff to be mixed up with __dict__
   269	    if cl.has_dict and not has_managed_dict(cl, emitter):
   270	        # __dict__ lives right after the struct and __weakref__ lives right after that
   271	        # TODO: They should get members in the struct instead of doing this nonsense.
   272	        weak_offset = f"{base_size} + sizeof(PyObject *)"
   273	        emitter.emit_lines(
   274	            f"PyMemberDef {members_name}[] = {{",
   275	            f'{{"__dict__", T_OBJECT_EX, {base_size}, 0, NULL}},',
   276	            f'{{"__weakref__", T_OBJECT_EX, {weak_offset}, 0, NULL}},',
   277	            "{0}",
   278	            "};",
   279	        )
   280
   281	        fields["tp_members"] = members_name
   282	        fields["tp_basicsize"] = f"{base_size} + 2*sizeof(PyObject *)"
   283	        if emitter.capi_version < (3, 12):
   284	            fields["tp_dictoffset"] = base_size
   285	            fields["tp_weaklistoffset"] = weak_offset
   286	    else:
   287	        fields["tp_basicsize"] = base_size
   288
   289	    if generate_full:
   290	        # Declare setup method that allocates and initializes an object. type is the
   291	        # type of the class being initialized, which could be another class if there
   292	        # is an interpreted subclass.
   293	        emitter.emit_line(f"static PyObject *{setup_name}(PyTypeObject *type);")
   294	        assert cl.ctor is not None
   295	        emitter.emit_line(native_function_header(cl.ctor, emitter) + ";")
   296
   297	        emit_line()
   298	        init_fn = cl.get_method("__init__")
   299	        generate_new_for_class(cl, new_name, vtable_name, setup_name, init_fn, emitter)
   300	        emit_line()
   301	        generate_traverse_for_class(cl, traverse_name, emitter)
   302	        emit_line()
   303	        generate_clear_for_class(cl, clear_name, emitter)
   304	        emit_line()
   305	        generate_dealloc_for_class(cl, dealloc_name, clear_name, bool(del_method), emitter)
   306	        emit_line()
   307
   308	        if cl.allow_interpreted_subclasses:
   309	            shadow_vtable_name: str | None = generate_vtables(
   310	                cl, vtable_setup_name + "_shadow", vtable_name + "_shadow", emitter, shadow=True
   311	            )
   312	            emit_line()
   313	        else:
   314	            shadow_vtable_name = None
   315	        vtable_name = generate_vtables(cl, vtable_setup_name, vtable_name, emitter, shadow=False)
   316	        emit_line()
   317	    if del_method:
   318	        generate_finalize_for_class(del_method, finalize_name, emitter)
   319	        emit_line()
   320	    if needs_getseters:
   321	        generate_getseter_declarations(cl, emitter)
   322	        emit_line()
   323	        generate_getseters_table(cl, getseters_name, emitter)
   324	        emit_line()
   325
   326	    if cl.is_trait:
   327	        generate_new_for_trait(cl, new_name, emitter)
   328
   329	    generate_methods_table(cl, methods_name, emitter)
   330	    emit_line()
   331
   332	    flags = ["Py_TPFLAGS_DEFAULT", "Py_TPFLAGS_HEAPTYPE", "Py_TPFLAGS_BASETYPE"]
   333	    if generate_full:
   334	        flags.append("Py_TPFLAGS_HAVE_GC")
   335	    if cl.has_method("__call__"):
   336	        fields["tp_vectorcall_offset"] = "offsetof({}, vectorcall)".format(
   337	            cl.struct_name(emitter.names)
   338	        )
   339	        flags.append("_Py_TPFLAGS_HAVE_VECTORCALL")
   340	        if not fields.get("tp_vectorcall"):
   341	            # This is just a placeholder to please CPython. It will be
   342	            # overridden during setup.
   343	            fields["tp_call"] = "PyVectorcall_Call"
   344	    if has_managed_dict(cl, emitter):
   345	        flags.append("Py_TPFLAGS_MANAGED_DICT")
   346	    fields["tp_flags"] = " | ".join(flags)
   347
   348	    emitter.emit_line(f"static PyTypeObject {emitter.type_struct_name(cl)}_template_ = {{")
   349	    emitter.emit_line("PyVarObject_HEAD_INIT(NULL, 0)")
   350	    for field, value in fields.items():
   351	        emitter.emit_line(f".{field} = {value},")
   352	    emitter.emit_line("};")
   353	    emitter.emit_line(
   354	        "static PyTypeObject *{t}_template = &{t}_template_;".format(
   355	            t=emitter.type_struct_name(cl)
   356	        )
   357	    )
   358
   359	    emitter.emit_line()
   360	    if generate_full:
   361	        generate_setup_for_class(
   362	            cl, setup_name, defaults_fn, vtable_name, shadow_vtable_name, emitter
   363	        )
   364	        emitter.emit_line()
   365	        generate_constructor_for_class(cl, cl.ctor, init_fn, setup_name, vtable_name, emitter)
   366	        emitter.emit_line()
   367	    if needs_getseters:
   368	        generate_getseters(cl, emitter)
   369

   379	def generate_object_struct(cl: ClassIR, emitter: Emitter) -> None:
   380	    seen_attrs: set[tuple[str, RType]] = set()
   381	    lines: list[str] = []
   382	    lines += ["typedef struct {", "PyObject_HEAD", "CPyVTableItem *vtable;"]
   383	    if cl.has_method("__call__"):
   384	        lines.append("vectorcallfunc vectorcall;")
   385	    bitmap_attrs = []
   386	    for base in reversed(cl.base_mro):
   387	        if not base.is_trait:
   388	            if base.bitmap_attrs:
   389	                # Do we need another attribute bitmap field?
   390	                if emitter.bitmap_field(len(base.bitmap_attrs) - 1) not in bitmap_attrs:
   391	                    for i in range(0, len(base.bitmap_attrs), BITMAP_BITS):
   392	                        attr = emitter.bitmap_field(i)
   393	                        if attr not in bitmap_attrs:
   394	                            lines.append(f"{BITMAP_TYPE} {attr};")
   395	                            bitmap_attrs.append(attr)
   396	            for attr, rtype in base.attributes.items():
   397	                if (attr, rtype) not in seen_attrs:
   398	                    lines.append(f"{emitter.ctype_spaced(rtype)}{emitter.attr(attr)};")
   399	                    seen_attrs.add((attr, rtype))
   400
   401	                    if isinstance(rtype, RTuple):
   402	                        emitter.declare_tuple_struct(rtype)
   403
   404	    lines.append(f"}} {cl.struct_name(emitter.names)};")
   405	    lines.append("")
   406	    emitter.context.declarations[cl.struct_name(emitter.names)] = HeaderDeclaration(
   407	        lines, is_type=True
   408	    )

   411	def generate_vtables(
   412	    base: ClassIR, vtable_setup_name: str, vtable_name: str, emitter: Emitter, shadow: bool
   413	) -> str:
   414	    """Emit the vtables and vtable setup functions for a class.
   415
   416	    This includes both the primary vtable and any trait implementation vtables.
   417	    The trait vtables go before the main vtable, and have the following layout:
   418	        {
   419	            CPyType_T1,         // pointer to type object
   420	            C_T1_trait_vtable,  // pointer to array of method pointers
   421	            C_T1_offset_table,  // pointer to array of attribute offsets
   422	            CPyType_T2,
   423	            C_T2_trait_vtable,
   424	            C_T2_offset_table,
   425	            ...
   426	        }
   427	    The method implementations are calculated at the end of IR pass, attribute
   428	    offsets are {offsetof(native__C, _x1), offsetof(native__C, _y1), ...}.
   429
   430	    To account for both dynamic loading and dynamic class creation,
   431	    vtables are populated dynamically at class creation time, so we
   432	    emit empty array definitions to store the vtables and a function to
   433	    populate them.
   434
   435	    If shadow is True, generate "shadow vtables" that point to the
   436	    shadow glue methods (which should dispatch via the Python C-API).
   437
   438	    Returns the expression to use to refer to the vtable, which might be
   439	    different than the name, if there are trait vtables.
   440	    """
   441
   442	    def trait_vtable_name(trait: ClassIR) -> str:
   443	        return "{}_{}_trait_vtable{}".format(
   444	            base.name_prefix(emitter.names),
   445	            trait.name_prefix(emitter.names),
   446	            "_shadow" if shadow else "",
   447	        )
   448
   449	    def trait_offset_table_name(trait: ClassIR) -> str:
   450	        return "{}_{}_offset_table".format(
   451	            base.name_prefix(emitter.names), trait.name_prefix(emitter.names)
   452	        )
   453
   454	    # Emit array definitions with enough space for all the entries
   455	    emitter.emit_line(
   456	        "static CPyVTableItem {}[{}];".format(
   457	            vtable_name, max(1, len(base.vtable_entries) + 3 * len(base.trait_vtables))
   458	        )
   459	    )
   460
   461	    for trait, vtable in base.trait_vtables.items():
   462	        # Trait methods entry (vtable index -> method implementation).
   463	        emitter.emit_line(
   464	            f"static CPyVTableItem {trait_vtable_name(trait)}[{max(1, len(vtable))}];"
   465	        )
   466	        # Trait attributes entry (attribute number in trait -> offset in actual struct).
   467	        emitter.emit_line(
   468	            "static size_t {}[{}];".format(
   469	                trait_offset_table_name(trait), max(1, len(trait.attributes))
   470	            )
   471	        )
   472
   473	    # Emit vtable setup function
   474	    emitter.emit_line("static bool")
   475	    emitter.emit_line(f"{NATIVE_PREFIX}{vtable_setup_name}(void)")
   476	    emitter.emit_line("{")
   477
   478	    if base.allow_interpreted_subclasses and not shadow:
   479	        emitter.emit_line(f"{NATIVE_PREFIX}{vtable_setup_name}_shadow();")
   480
   481	    subtables = []
   482	    for trait, vtable in base.trait_vtables.items():
   483	        name = trait_vtable_name(trait)
   484	        offset_name = trait_offset_table_name(trait)
   485	        generate_vtable(vtable, name, emitter, [], shadow)
   486	        generate_offset_table(offset_name, emitter, trait, base)
   487	        subtables.append((trait, name, offset_name))
   488
   489	    generate_vtable(base.vtable_entries, vtable_name, emitter, subtables, shadow)
   490
   491	    emitter.emit_line("return 1;")
   492	    emitter.emit_line("}")
   493
   494	    return vtable_name if not subtables else f"{vtable_name} + {len(subtables) * 3}"
   495

   497	def generate_offset_table(
   498	    trait_offset_table_name: str, emitter: Emitter, trait: ClassIR, cl: ClassIR
   499	) -> None:
   500	    """Generate attribute offset row of a trait vtable."""
   501	    emitter.emit_line(f"size_t {trait_offset_table_name}_scratch[] = {{")
   502	    for attr in trait.attributes:
   503	        emitter.emit_line(f"offsetof({cl.struct_name(emitter.names)}, {emitter.attr(attr)}),")
   504	    if not trait.attributes:
   505	        # This is for msvc.
   506	        emitter.emit_line("0")
   507	    emitter.emit_line("};")
   508	    emitter.emit_line(
   509	        "memcpy({name}, {name}_scratch, sizeof({name}));".format(name=trait_offset_table_name)
   510	    )

   513	def generate_vtable(
   514	    entries: VTableEntries,
   515	    vtable_name: str,
   516	    emitter: Emitter,
   517	    subtables: list[tuple[ClassIR, str, str]],
   518	    shadow: bool,
   519	) -> None:
   520	    emitter.emit_line(f"CPyVTableItem {vtable_name}_scratch[] = {{")
   521	    if subtables:
   522	        emitter.emit_line("/* Array of trait vtables */")
   523	        for trait, table, offset_table in subtables:
   524	            emitter.emit_line(
   525	                "(CPyVTableItem){}, (CPyVTableItem){}, (CPyVTableItem){},".format(
   526	                    emitter.type_struct_name(trait), table, offset_table
   527	                )
   528	            )
   529	        emitter.emit_line("/* Start of real vtable */")
   530
   531	    for entry in entries:
   532	        method = entry.shadow_method if shadow and entry.shadow_method else entry.method
   533	        emitter.emit_line(
   534	            "(CPyVTableItem){}{}{},".format(
   535	                emitter.get_group_prefix(entry.method.decl),
   536	                NATIVE_PREFIX,
   537	                method.cname(emitter.names),
   538	            )
   539	        )
   540
   541	    # msvc doesn't allow empty arrays; maybe allowing them at all is an extension?
   542	    if not entries:
   543	        emitter.emit_line("NULL")
   544	    emitter.emit_line("};")
   545	    emitter.emit_line("memcpy({name}, {name}_scratch, sizeof({name}));".format(name=vtable_name))

   548	def generate_setup_for_class(
   549	    cl: ClassIR,
   550	    func_name: str,
   551	    defaults_fn: FuncIR | None,
   552	    vtable_name: str,
   553	    shadow_vtable_name: str | None,
   554	    emitter: Emitter,
   555	) -> None:
   556	    """Generate a native function that allocates an instance of a class."""
   557	    emitter.emit_line("static PyObject *")
   558	    emitter.emit_line(f"{func_name}(PyTypeObject *type)")
   559	    emitter.emit_line("{")
   560	    emitter.emit_line(f"{cl.struct_name(emitter.names)} *self;")
   561	    emitter.emit_line(f"self = ({cl.struct_name(emitter.names)} *)type->tp_alloc(type, 0);")
   562	    emitter.emit_line("if (self == NULL)")
   563	    emitter.emit_line("    return NULL;")
   564
   565	    if shadow_vtable_name:
   566	        emitter.emit_line(f"if (type != {emitter.type_struct_name(cl)}) {{")
   567	        emitter.emit_line(f"self->vtable = {shadow_vtable_name};")
   568	        emitter.emit_line("} else {")
   569	        emitter.emit_line(f"self->vtable = {vtable_name};")
   570	        emitter.emit_line("}")
   571	    else:
   572	        emitter.emit_line(f"self->vtable = {vtable_name};")
   573
   574	    for i in range(0, len(cl.bitmap_attrs), BITMAP_BITS):
   575	        field = emitter.bitmap_field(i)
   576	        emitter.emit_line(f"self->{field} = 0;")
   577
   578	    if cl.has_method("__call__"):
   579	        name = cl.method_decl("__call__").cname(emitter.names)
   580	        emitter.emit_line(f"self->vectorcall = {PREFIX}{name};")
   581
   582	    for base in reversed(cl.base_mro):
   583	        for attr, rtype in base.attributes.items():
   584	            value = emitter.c_undefined_value(rtype)
   585
   586	            # We don't need to set this field to NULL since tp_alloc() already
   587	            # zero-initializes `self`.
   588	            if value != "NULL":
   589	                emitter.emit_line(rf"self->{emitter.attr(attr)} = {value};")
   590
   591	    # Initialize attributes to default values, if necessary
   592	    if defaults_fn is not None:
   593	        emitter.emit_lines(
   594	            "if ({}{}((PyObject *)self) == 0) {{".format(
   595	                NATIVE_PREFIX, defaults_fn.cname(emitter.names)
   596	            ),
   597	            "Py_DECREF(self);",
   598	            "return NULL;",
   599	            "}",
   600	        )
   601
   602	    emitter.emit_line("return (PyObject *)self;")
   603	    emitter.emit_line("}")

   607	    cl: ClassIR,
   608	    fn: FuncDecl,
   609	    init_fn: FuncIR | None,
   610	    setup_name: str,
   611	    vtable_name: str,
   612	    emitter: Emitter,
   613	) -> None:
   614	    """Generate a native function that allocates and initializes an instance of a class."""
   615	    emitter.emit_line(f"{native_function_header(fn, emitter)}")
   616	    emitter.emit_line("{")
   617	    emitter.emit_line(f"PyObject *self = {setup_name}({emitter.type_struct_name(cl)});")
   618	    emitter.emit_line("if (self == NULL)")
   619	    emitter.emit_line("    return NULL;")
   620	    args = ", ".join(["self"] + [REG_PREFIX + arg.name for arg in fn.sig.args])
   621	    if init_fn is not None:
   622	        emitter.emit_line(
   623	            "char res = {}{}{}({});".format(
   624	                emitter.get_group_prefix(init_fn.decl),
   625	                NATIVE_PREFIX,
   626	                init_fn.cname(emitter.names),
   627	                args,
   628	            )
   629	        )
   630	        emitter.emit_line("if (res == 2) {")
   631	        emitter.emit_line("Py_DECREF(self);")
   632	        emitter.emit_line("return NULL;")
   633	        emitter.emit_line("}")
   634
   635	    # If there is a nontrivial ctor that we didn't define, invoke it via tp_init
   636	    elif len(fn.sig.args) > 1:
   637	        emitter.emit_line(f"int res = {emitter.type_struct_name(cl)}->tp_init({args});")
   638
   639	        emitter.emit_line("if (res < 0) {")
   640	        emitter.emit_line("Py_DECREF(self);")
   641	        emitter.emit_line("return NULL;")
   642	        emitter.emit_line("}")
   643
   644	    emitter.emit_line("return self;")
   645	    emitter.emit_line("}")

   649	    """Generate an init function suitable for use as tp_init.
   650
   651	    tp_init needs to be a function that returns an int, and our
   652	    __init__ methods return a PyObject. Translate NULL to -1,
   653	    everything else to 0.
   654	    """
   655	    func_name = f"{cl.name_prefix(emitter.names)}_init"
   656
   657	    emitter.emit_line("static int")
   658	    emitter.emit_line(f"{func_name}(PyObject *self, PyObject *args, PyObject *kwds)")

   661	        emitter.emit_line(
   662	            "return {}{}(self, args, kwds) != NULL ? 0 : -1;".format(
   663	                PREFIX, init_fn.cname(emitter.names)
   664	            )
   665	        )
   666	    else:
   667	        emitter.emit_line("return 0;")
   668	    emitter.emit_line("}")

   670	    return func_name

   673	def generate_new_for_class(
   674	    cl: ClassIR,
   675	    func_name: str,
   676	    vtable_name: str,
   677	    setup_name: str,
   678	    init_fn: FuncIR | None,
   679	    emitter: Emitter,
   680	) -> None:
   681	    emitter.emit_line("static PyObject *")
   682	    emitter.emit_line(f"{func_name}(PyTypeObject *type, PyObject *args, PyObject *kwds)")
   683	    emitter.emit_line("{")
   684	    # TODO: Check and unbox arguments
   685	    if not cl.allow_interpreted_subclasses:
   686	        emitter.emit_line(f"if (type != {emitter.type_struct_name(cl)}) {{")
   687	        emitter.emit_line(
   688	            'PyErr_SetString(PyExc_TypeError, "interpreted classes cannot inherit from compiled");'
   689	        )
   690	        emitter.emit_line("return NULL;")
   691	        emitter.emit_line("}")

   693	    if not init_fn or cl.allow_interpreted_subclasses or cl.builtin_base or cl.is_serializable():
   694	        # Match Python semantics -- __new__ doesn't call __init__.
   695	        emitter.emit_line(f"return {setup_name}(type);")

   703	            f"PyObject *ret = {PREFIX}{init_fn.cname(emitter.names)}(self, args, kwds);"
   704	        )
   705	        emitter.emit_lines("if (ret == NULL)", "    return NULL;")
   706	        emitter.emit_line("return self;")
   707	    emitter.emit_line("}")

   727	    """Emit function that performs cycle GC traversal of an instance."""
   728	    emitter.emit_line("static int")
   729	    emitter.emit_line(
   730	        f"{func_name}({cl.struct_name(emitter.names)} *self, visitproc visit, void *arg)"
   731	    )

   733	    for base in reversed(cl.base_mro):
   734	        for attr, rtype in base.attributes.items():
   735	            emitter.emit_gc_visit(f"self->{emitter.attr(attr)}", rtype)
   736	    if has_managed_dict(cl, emitter):
   737	        emitter.emit_line("PyObject_VisitManagedDict((PyObject *)self, visit, arg);")

   740	        # __dict__ lives right after the struct and __weakref__ lives right after that
   741	        emitter.emit_gc_visit(
   742	            f"*((PyObject **)((char *)self + sizeof({struct_name})))", object_rprimitive
   743	        )
   744	        emitter.emit_gc_visit(
   745	            f"*((PyObject **)((char *)self + sizeof(PyObject *) + sizeof({struct_name})))",
   746	            object_rprimitive,
   747	        )
   748	    emitter.emit_line("return 0;")
   749	    emitter.emit_line("}")

   799	    del_method: FuncIR, finalize_func_name: str, emitter: Emitter
   800	) -> None:
   801	    emitter.emit_line("static void")
   802	    emitter.emit_line(f"{finalize_func_name}(PyObject *self)")
   803	    emitter.emit_line("{")
   804	    emitter.emit_line("PyObject *type, *value, *traceback;")
   805	    emitter.emit_line("PyErr_Fetch(&type, &value, &traceback);")
   806	    emitter.emit_line(
   807	        "{}{}{}(self);".format(
   808	            emitter.get_group_prefix(del_method.decl),
   809	            NATIVE_PREFIX,
   810	            del_method.cname(emitter.names),
   811	        )
   812	    )
   813	    emitter.emit_line("if (PyErr_Occurred() != NULL) {")
   814	    emitter.emit_line('PyObject *del_str = PyUnicode_FromString("__del__");')
   815	    emitter.emit_line(
   816	        "PyObject *del_method = (del_str == NULL) ? NULL : _PyType_Lookup(Py_TYPE(self), del_str);"
   817	    )
   818	    # CPython interpreter uses PyErr_WriteUnraisable: https://docs.python.org/3/c-api/exceptions.html#c.PyErr_WriteUnraisable
   819	    # However, the message is slightly different due to the way mypyc compiles classes.
   820	    # CPython interpreter prints: Exception ignored in: <function F.__del__ at 0x100aed940>
   821	    # mypyc prints: Exception ignored in: <slot wrapper '__del__' of 'F' objects>
   822	    emitter.emit_line("PyErr_WriteUnraisable(del_method);")
   823	    emitter.emit_line("Py_XDECREF(del_method);")
   824	    emitter.emit_line("Py_XDECREF(del_str);")
   825	    emitter.emit_line("}")
   826	    # PyErr_Restore also clears exception raised in __del__.
   827	    emitter.emit_line("PyErr_Restore(type, value, traceback);")
   828	    emitter.emit_line("}")

   958	def generate_getter(cl: ClassIR, attr: str, rtype: RType, emitter: Emitter) -> None:
   959	    attr_field = emitter.attr(attr)
   960	    emitter.emit_line("static PyObject *")
   961	    emitter.emit_line(
   962	        "{}({} *self, void *closure)".format(
   963	            getter_name(cl, attr, emitter.names), cl.struct_name(emitter.names)
   964	        )
   965	    )
   966	    emitter.emit_line("{")
   967	    attr_expr = f"self->{attr_field}"
   968
   969	    # HACK: Don't consider refcounted values as always defined, since it's possible to
   970	    #       access uninitialized values via 'gc.get_objects()'. Accessing non-refcounted
   971	    #       values is benign.
   972	    always_defined = cl.is_always_defined(attr) and not rtype.is_refcounted
   973
   974	    if not always_defined:
   975	        emitter.emit_undefined_attr_check(rtype, attr_expr, "==", "self", attr, cl, unlikely=True)
   976	        emitter.emit_line("PyErr_SetString(PyExc_AttributeError,")
   977	        emitter.emit_line(f'    "attribute {repr(attr)} of {repr(cl.name)} undefined");')
   978	        emitter.emit_line("return NULL;")
   979	        emitter.emit_line("}")
   980	    emitter.emit_inc_ref(f"self->{attr_field}", rtype)
   981	    emitter.emit_box(f"self->{attr_field}", "retval", rtype, declare_dest=True)
   982	    emitter.emit_line("return retval;")
   983	    emitter.emit_line("}")

   986	def generate_setter(cl: ClassIR, attr: str, rtype: RType, emitter: Emitter) -> None:
   987	    attr_field = emitter.attr(attr)
   988	    emitter.emit_line("static int")
   989	    emitter.emit_line(
   990	        "{}({} *self, PyObject *value, void *closure)".format(
   991	            setter_name(cl, attr, emitter.names), cl.struct_name(emitter.names)
   992	        )
   993	    )
   994	    emitter.emit_line("{")
   995
   996	    deletable = cl.is_deletable(attr)
   997	    if not deletable:
   998	        emitter.emit_line("if (value == NULL) {")
   999	        emitter.emit_line("PyErr_SetString(PyExc_AttributeError,")
  1000	        emitter.emit_line(
  1001	            f'    "{repr(cl.name)} object attribute {repr(attr)} cannot be deleted");'
  1002	        )
  1003	        emitter.emit_line("return -1;")
  1004	        emitter.emit_line("}")
  1005
  1006	    # HACK: Don't consider refcounted values as always defined, since it's possible to
  1007	    #       access uninitialized values via 'gc.get_objects()'. Accessing non-refcounted
  1008	    #       values is benign.
  1009	    always_defined = cl.is_always_defined(attr) and not rtype.is_refcounted
  1010
  1011	    if rtype.is_refcounted:
  1012	        attr_expr = f"self->{attr_field}"
  1013	        if not always_defined:
  1014	            emitter.emit_undefined_attr_check(rtype, attr_expr, "!=", "self", attr, cl)
  1015	        emitter.emit_dec_ref(f"self->{attr_field}", rtype)
  1016	        if not always_defined:
  1017	            emitter.emit_line("}")
  1018
  1019	    if deletable:
  1020	        emitter.emit_line("if (value != NULL) {")
  1021
  1022	    if rtype.is_unboxed:
  1023	        emitter.emit_unbox("value", "tmp", rtype, error=ReturnHandler("-1"), declare_dest=True)
  1024	    elif is_same_type(rtype, object_rprimitive):
  1025	        emitter.emit_line("PyObject *tmp = value;")
  1026	    else:
  1027	        emitter.emit_cast("value", "tmp", rtype, declare_dest=True)
  1028	        emitter.emit_lines("if (!tmp)", "    return -1;")
  1029	    emitter.emit_inc_ref("tmp", rtype)

   108	def generate_native_function(
   109	    fn: FuncIR, emitter: Emitter, source_path: str, module_name: str
   110	) -> None:
   111	    declarations = Emitter(emitter.context)
   112	    names = generate_names_for_ir(fn.arg_regs, fn.blocks)
   113	    body = Emitter(emitter.context, names)
   114	    visitor = FunctionEmitterVisitor(body, declarations, source_path, module_name)
   115
   116	    declarations.emit_line(f"{native_function_header(fn.decl, emitter)} {{")
   117	    body.indent()
   118
   119	    for r in all_values(fn.arg_regs, fn.blocks):
   120	        if isinstance(r.type, RTuple):
   121	            emitter.declare_tuple_struct(r.type)
   122	        if isinstance(r.type, RArray):
   123	            continue  # Special: declared on first assignment
   124
   125	        if r in fn.arg_regs:
   126	            continue  # Skip the arguments
   127
   128	        ctype = emitter.ctype_spaced(r.type)
   129	        init = ""
   130	        declarations.emit_line(
   131	            "{ctype}{prefix}{name}{init};".format(
   132	                ctype=ctype, prefix=REG_PREFIX, name=names[r], init=init
   133	            )
   134	        )
   135
   136	    # Before we emit the blocks, give them all labels
   137	    blocks = fn.blocks
   138	    for i, block in enumerate(blocks):
   139	        block.label = i
   140
   141	    # Find blocks that are never jumped to or are only jumped to from the
   142	    # block directly above it. This allows for more labels and gotos to be
   143	    # eliminated during code generation.
   144	    for block in fn.blocks:
   145	        terminator = block.terminator
   146	        assert isinstance(terminator, ControlOp)
   147
   148	        for target in terminator.targets():
   149	            is_next_block = target.label == block.label + 1
   150
   151	            # Always emit labels for GetAttr error checks since the emit code that
   152	            # generates them will add instructions between the branch and the
   153	            # next label, causing the label to be wrongly removed. A better
   154	            # solution would be to change the IR so that it adds a basic block
   155	            # in between the calls.
   156	            is_problematic_op = isinstance(terminator, Branch) and any(
   157	                isinstance(s, GetAttr) for s in terminator.sources()
   158	            )
   159
   160	            if not is_next_block or is_problematic_op:
   161	                fn.blocks[target.label].referenced = True
   162
   163	    common = frequently_executed_blocks(fn.blocks[0])
   164
   165	    for i in range(len(blocks)):
   166	        block = blocks[i]
   167	        visitor.rare = block not in common
   168	        next_block = None
   169	        if i + 1 < len(blocks):
   170	            next_block = blocks[i + 1]
   171	        body.emit_label(block)
   172	        visitor.next_block = next_block
   173
   174	        ops = block.ops
   175	        visitor.ops = ops
   176	        visitor.op_index = 0
   177	        while visitor.op_index < len(ops):
   178	            ops[visitor.op_index].accept(visitor)
   179	            visitor.op_index += 1
   180
   181	    body.emit_line("}")
   182
   183	    emitter.emit_from_emitter(declarations)
   184	    emitter.emit_from_emitter(body)

   212	    def visit_branch(self, op: Branch) -> None:
   213	        true, false = op.true, op.false
   214	        negated = op.negated
   215	        negated_rare = False
   216	        if true is self.next_block and op.traceback_entry is None:
   217	            # Switch true/false since it avoids an else block.
   218	            true, false = false, true

   222	        neg = "!" if negated else ""
   223	        cond = ""

   225	            expr_result = self.reg(op.value)
   226	            cond = f"{neg}{expr_result}"
   227	        elif op.op == Branch.IS_ERROR:
   228	            typ = op.value.type
   229	            compare = "!=" if negated else "=="
   230	            if isinstance(typ, RTuple):

   233	                    typ, self.reg(op.value), self.c_error_value, compare

   237	        else:

   243	                cond = f"unlikely({cond})"
   244	            else:
   245	                cond = f"likely({cond})"

   253	                self.emit_traceback(op)
   254	                self.emit_lines("goto %s;" % self.label(true), "}")
   255	        else:

   257	            self.emit_traceback(op)

   260	                self.emit_line("goto %s;" % self.label(true))

   461	        src = self.reg(op.src)
   462	        rtype = op.class_type
   463	        cl = rtype.class_ir
   464	        attr_rtype, decl_cl = cl.attr_details(op.attr)
   465	        if cl.get_method(op.attr):
   466	            # Again, use vtable access for properties...
   467	            assert not op.is_init and op.error_kind == ERR_FALSE, "%s %d %d %s" % (
   468	                op.attr,
   469	                op.is_init,
   470	                op.error_kind,
   471	                rtype,
   472	            )
   473	            version = "_TRAIT" if cl.is_trait else ""
   474	            self.emit_line(
   475	                "%s = CPY_SET_ATTR%s(%s, %s, %d, %s, %s, %s); /* %s */"
   476	                % (
   477	                    dest,
   478	                    version,
   479	                    obj,
   480	                    self.emitter.type_struct_name(rtype.class_ir),
   481	                    rtype.setter_index(op.attr),
   482	                    src,
   483	                    rtype.struct_name(self.names),
   484	                    self.ctype(rtype.attr_type(op.attr)),
   485	                    op.attr,
   486	                )
   487	            )
   488	        else:
   489	            # ...and struct access for normal attributes.
   490	            attr_expr = self.get_attr_expr(obj, op, decl_cl)
   491	            if not op.is_init and attr_rtype.is_refcounted:
   492	                # This is not an initialization (where we know that the attribute was
   493	                # previously undefined), so decref the old value.
   494	                always_defined = cl.is_always_defined(op.attr)
   495	                if not always_defined:
   496	                    self.emitter.emit_undefined_attr_check(
   497	                        attr_rtype, attr_expr, "!=", obj, op.attr, cl
   498	                    )
   499	                self.emitter.emit_dec_ref(attr_expr, attr_rtype)
   500	                if not always_defined:
   501	                    self.emitter.emit_line("}")
   502	            elif attr_rtype.error_overlap and not cl.is_always_defined(op.attr):
   503	                # If there is overlap with the error value, update bitmap to mark
   504	                # attribute as defined.
   505	                self.emitter.emit_attr_bitmap_set(src, obj, attr_rtype, cl, op.attr)
   506
   507	            # This steals the reference to src, so we don't need to increment the arg
   508	            self.emitter.emit_line(f"{attr_expr} = {src};")
   509	            if op.error_kind == ERR_FALSE:
   510	                self.emitter.emit_line(f"{dest} = 1;")

   549	    def visit_call(self, op: Call) -> None:
   550	        """Call native function."""
   551	        dest = self.get_dest_assign(op)
   552	        args = ", ".join(self.reg(arg) for arg in op.args)
   553	        lib = self.emitter.get_group_prefix(op.fn)
   554	        cname = op.fn.cname(self.names)
   555	        self.emit_line(f"{dest}{lib}{NATIVE_PREFIX}{cname}({args});")

   557	    def visit_method_call(self, op: MethodCall) -> None:
   558	        """Call native method."""
   559	        dest = self.get_dest_assign(op)
   560	        self.emit_method_call(dest, op.obj, op.method, op.args)

   562	    def emit_method_call(self, dest: str, op_obj: Value, name: str, op_args: list[Value]) -> None:
   563	        obj = self.reg(op_obj)
   564	        rtype = op_obj.type
   565	        assert isinstance(rtype, RInstance)
   566	        class_ir = rtype.class_ir
   567	        method = rtype.class_ir.get_method(name)
   568	        assert method is not None
   569
   570	        # Can we call the method directly, bypassing vtable?
   571	        is_direct = class_ir.is_method_final(name)
   572
   573	        # The first argument gets omitted for static methods and
   574	        # turned into the class for class methods
   575	        obj_args = (
   576	            []
   577	            if method.decl.kind == FUNC_STATICMETHOD
   578	            else [f"(PyObject *)Py_TYPE({obj})"] if method.decl.kind == FUNC_CLASSMETHOD else [obj]
   579	        )
   580	        args = ", ".join(obj_args + [self.reg(arg) for arg in op_args])
   581	        mtype = native_function_type(method, self.emitter)
   582	        version = "_TRAIT" if rtype.class_ir.is_trait else ""
   583	        if is_direct:
   584	            # Directly call method, without going through the vtable.
   585	            lib = self.emitter.get_group_prefix(method.decl)
   586	            self.emit_line(f"{dest}{lib}{NATIVE_PREFIX}{method.cname(self.names)}({args});")
   587	        else:
   588	            # Call using vtable.
   589	            method_idx = rtype.method_index(name)
   590	            self.emit_line(
   591	                "{}CPY_GET_METHOD{}({}, {}, {}, {}, {})({}); /* {} */".format(
   592	                    dest,
   593	                    version,
   594	                    obj,
   595	                    self.emitter.type_struct_name(rtype.class_ir),
   596	                    method_idx,
   597	                    rtype.struct_name(self.names),
   598	                    mtype,
   599	                    args,
   600	                    name,
   601	                )
   602	            )

   604	    def visit_inc_ref(self, op: IncRef) -> None:
   605	        if (
   606	            isinstance(op.src, Box)
   607	            and (is_none_rprimitive(op.src.src.type) or is_bool_rprimitive(op.src.src.type))
   608	            and HAVE_IMMORTAL
   609	        ):
   610	            # On Python 3.12+, None/True/False are immortal, and we can skip inc ref
   611	            return
   612
   613	        if isinstance(op.src, LoadLiteral) and HAVE_IMMORTAL:
   614	            value = op.src.value
   615	            # We can skip inc ref for immortal literals on Python 3.12+
   616	            if type(value) is int and -5 <= value <= 256:
   617	                # Small integers are immortal
   618	                return
   619
   620	        src = self.reg(op.src)
   621	        self.emit_inc_ref(src, op.src.type)

   630	    def visit_cast(self, op: Cast) -> None:
   631	        branch = self.next_branch()
   632	        handler = None
   633	        if branch is not None:
   634	            if (
   635	                branch.value is op
   636	                and branch.op == Branch.IS_ERROR
   637	                and branch.traceback_entry is not None
   638	                and not branch.negated
   639	                and branch.false is self.next_block
   640	            ):
   641	                # Generate code also for the following branch here to avoid
   642	                # redundant branches in the generated code.
   643	                handler = TracebackAndGotoHandler(
   644	                    self.label(branch.true),
   645	                    self.source_path,
   646	                    self.module_name,
   647	                    branch.traceback_entry,
   648	                )
   649	                self.op_index += 1
   650
   651	        self.emitter.emit_cast(
   652	            self.reg(op.src), self.reg(op), op.type, src_type=op.src.type, error=handler
   653	        )

   662	        # TODO: Better escaping of backspaces and such

   664	            if isinstance(op.value, str):

   677	        self.emitter.emit_line(f"{self.reg(op)} = 0;")
