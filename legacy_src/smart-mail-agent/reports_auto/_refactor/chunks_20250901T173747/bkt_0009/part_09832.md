    34	    internal to the archive ('/'-separated), and a filesystem path where it
    35	    will be extracted.  The callback must return the desired extract path

   426	                 mode='w'):

   428	    zip file will be named 'base_dir' + ".zip".  Uses either the "zipfile"
   429	    Python module (if available) or the InfoZIP "zip" utility (if installed

   431	    raises DistutilsExecError.  Returns the name of the output zip file.

   435	    mkpath(os.path.dirname(zip_filename), dry_run=dry_run)

   438	    def visit(z, dirname, names):
   439	        for name in names:
   440	            path = os.path.normpath(os.path.join(dirname, name))
   441	            if os.path.isfile(path):
   442	                p = path[len(base_dir) + 1:]

   449	        z = zipfile.ZipFile(zip_filename, mode, compression=compression)
   450	        for dirname, dirs, files in sorted_walk(base_dir):
   451	            visit(z, dirname, files)
   452	        z.close()

   454	        for dirname, dirs, files in sorted_walk(base_dir):
   455	            visit(None, dirname, files)
