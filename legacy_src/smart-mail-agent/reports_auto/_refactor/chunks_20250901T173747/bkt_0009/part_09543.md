   310	    def start(self) -> None:
   311	        """Start collecting trace information."""
   312	        # We may be a new collector in a forked process.  The old process'
   313	        # collectors will be in self._collectors, but they won't be usable.
   314	        # Find them and discard them.

   319	            else:
   320	                c.post_fork()

   324	            self._collectors[-1].pause()

   328	        try:
   329	            # Install the tracer on this thread.
   330	            self._start_tracer()

   333	                self._collectors[-1].resume()
   334	            raise

   337	        # stack of collectors.
   338	        self._collectors.append(self)

   341	        # threads.

   434	            try:

   436	            except RuntimeError as ex:          # pragma: cant happen
   437	                runtime_err = ex

   290	    def from_file(self, filename: str, warn: Callable[[str], None], our_file: bool) -> bool:
   291	        """Read configuration from a .rc file.
   292
   293	        `filename` is a file name to read.
   294
   295	        `our_file` is True if this config file is specifically for coverage,
   296	        False if we are examining another config file (tox.ini, setup.cfg)
   297	        for possible settings.
   298
   299	        Returns True or False, whether the file could be read, and it had some
   300	        coverage.py settings in it.
   301
   302	        """
   303	        _, ext = os.path.splitext(filename)
   304	        cp: TConfigParser
   305	        if ext == ".toml":
   306	            cp = TomlConfigParser(our_file)
   307	        else:
   308	            cp = HandyConfigParser(our_file)
   309
   310	        self.config_files_attempted.append(os.path.abspath(filename))
   311
   312	        try:
   313	            files_read = cp.read(filename)
   314	        except (configparser.Error, TomlDecodeError) as err:
   315	            raise ConfigError(f"Couldn't read config file {filename}: {err}") from err
   316	        if not files_read:
   317	            return False
   318
   319	        self.config_files_read.extend(map(os.path.abspath, files_read))
   320
   321	        any_set = False
   322	        try:
   323	            for option_spec in self.CONFIG_FILE_OPTIONS:
   324	                was_set = self._set_attr_from_config_option(cp, *option_spec)
   325	                if was_set:
   326	                    any_set = True
   327	        except ValueError as err:
   328	            raise ConfigError(f"Couldn't read config file {filename}: {err}") from err
   329
   330	        # Check that there are no unrecognized options.
   331	        all_options = collections.defaultdict(set)
   332	        for option_spec in self.CONFIG_FILE_OPTIONS:
   333	            section, option = option_spec[1].split(":")
   334	            all_options[section].add(option)
   335
   336	        for section, options in all_options.items():
   337	            real_section = cp.real_section(section)
   338	            if real_section:
   339	                for unknown in set(cp.options(section)) - options:
   340	                    warn(
   341	                        "Unrecognized option '[{}] {}=' in config file {}".format(
   342	                            real_section, unknown, filename,
   343	                        ),
   344	                    )
   345
   346	        # [paths] is special
   347	        if cp.has_section("paths"):
   348	            for option in cp.options("paths"):
   349	                self.paths[option] = cp.getlist("paths", option)
   350	                any_set = True
   351
   352	        # plugins can have options
   353	        for plugin in self.plugins:
   354	            if cp.has_section(plugin):
   355	                self.plugin_options[plugin] = cp.get_section(plugin)
   356	                any_set = True
   357
   358	        # Was this file used as a config file? If it's specifically our file,
   359	        # then it was used.  If we're piggybacking on someone else's file,
   360	        # then it was only used if we found some settings in it.
   361	        if our_file:
   362	            used = True
   363	        else:
   364	            used = any_set
   365
   366	        if used:
   367	            self.config_file = os.path.abspath(filename)
   368	            with open(filename, "rb") as f:
   369	                self._config_contents = f.read()
   370
   371	        return used

   600	    """What config files should we try to read?
   601
   602	    Returns a list of tuples:
   603	        (filename, is_our_file, was_file_specified)

   606	    # Some API users were specifying ".coveragerc" to mean the same as
   607	    # True, so make it so.
   608	    if config_file == ".coveragerc":
   609	        config_file = True
   610	    specified_file = (config_file is not True)
   611	    if not specified_file:
   612	        # No file was specified. Check COVERAGE_RCFILE.
   613	        rcfile = os.getenv("COVERAGE_RCFILE")
   614	        if rcfile:
   615	            config_file = rcfile
   616	            specified_file = True
   617	    if not specified_file:
   618	        # Still no file specified. Default to .coveragerc
   619	        config_file = ".coveragerc"
   620	    assert isinstance(config_file, str)
   621	    files_to_try = [
   622	        (config_file, True, specified_file),
   623	        ("setup.cfg", False, False),
   624	        ("tox.ini", False, False),
   625	        ("pyproject.toml", False, False),
   626	    ]

   630	def read_coverage_config(
   631	    config_file: bool | str,
   632	    warn: Callable[[str], None],
   633	    **kwargs: TConfigValueIn,
   634	) -> CoverageConfig:
   635	    """Read the coverage.py configuration.
   636
   637	    Arguments:
   638	        config_file: a boolean or string, see the `Coverage` class for the
   639	            tricky details.
   640	        warn: a function to issue warnings.
   641	        all others: keyword arguments from the `Coverage` class, used for
   642	            setting values in the configuration.
   643
   644	    Returns:
   645	        config:
   646	            config is a CoverageConfig object read from the appropriate
   647	            configuration file.
   648
   649	    """
   650	    # Build the configuration from a number of sources:
   651	    # 1) defaults:
   652	    config = CoverageConfig()
   653
   654	    # 2) from a file:
   655	    if config_file:
   656	        files_to_try = config_files_to_try(config_file)
   657
   658	        for fname, our_file, specified_file in files_to_try:
   659	            config_read = config.from_file(fname, warn, our_file=our_file)
   660	            if config_read:
   661	                break
   662	            if specified_file:
   663	                raise ConfigError(f"Couldn't read {fname!r} as a config file")
   664
   665	    # 3) from environment variables:
   666	    env_data_file = os.getenv("COVERAGE_FILE")
   667	    if env_data_file:
   668	        config.data_file = env_data_file
   669
   670	    # $set_env.py: COVERAGE_DEBUG - Debug options: https://coverage.rtfd.io/cmd.html#debug
   671	    debugs = os.getenv("COVERAGE_DEBUG")
   672	    if debugs:
   673	        config.debug.extend(d.strip() for d in debugs.split(","))
   674
   675	    # Read the COVERAGE_CORE environment variable for backward compatibility,
   676	    # and because we use it in the test suite to pick a specific core.
   677	    env_core = os.getenv("COVERAGE_CORE")
   678	    if env_core:
   679	        config.core = env_core
   680
   681	    # 4) from constructor arguments:
   682	    config.from_args(**kwargs)
   683
   684	    # 5) for our benchmark, force settings using a secret environment variable:
   685	    force_file = os.getenv("COVERAGE_FORCE_CONFIG")
   686	    if force_file:
   687	        config.from_file(force_file, warn, our_file=True)
   688
   689	    # Once all the config has been collected, there's a little post-processing
   690	    # to do.
   691	    config.post_process()
   692
   693	    return config

   557	    def _init_for_start(self) -> None:
   558	        """Initialization for start()"""
   559	        # Construct the collector.
   560	        concurrency: list[str] = self.config.concurrency or []
   561	        if "multiprocessing" in concurrency:
   562	            if self.config.config_file is None:
   563	                raise ConfigError("multiprocessing requires a configuration file")
   564	            patch_multiprocessing(rcfile=self.config.config_file)
   565
   566	        dycon = self.config.dynamic_context
   567	        if not dycon or dycon == "none":
   568	            context_switchers = []
   569	        elif dycon == "test_function":
   570	            context_switchers = [should_start_context_test_function]
   571	        else:
   572	            raise ConfigError(f"Don't understand dynamic_context setting: {dycon!r}")
   573
   574	        context_switchers.extend(
   575	            plugin.dynamic_context for plugin in self._plugins.context_switchers
   576	        )
   577
   578	        should_start_context = combine_context_switchers(context_switchers)
   579
   580	        self._core = Core(
   581	            warn=self._warn,
   582	            config=self.config,
   583	            dynamic_contexts=(should_start_context is not None),
   584	            metacov=self._metacov,
   585	        )
   586	        self._collector = Collector(
   587	            core=self._core,
   588	            should_trace=self._should_trace,
   589	            check_include=self._check_include_omit_etc,
   590	            should_start_context=should_start_context,
   591	            file_mapper=self._file_mapper,
   592	            branch=self.config.branch,
   593	            warn=self._warn,
   594	            concurrency=concurrency,
   595	        )
   596
   597	        suffix = self._data_suffix_specified
   598	        if suffix:
   599	            if not isinstance(suffix, str):
   600	                # if data_suffix=True, use .machinename.pid.random
   601	                suffix = True
   602	        elif self.config.parallel:
   603	            if suffix is None:
   604	                suffix = True
   605	            elif not isinstance(suffix, str):
   606	                suffix = bool(suffix)
   607	        else:
   608	            suffix = None
   609
   610	        self._init_data(suffix)
   611
   612	        assert self._data is not None
   613	        self._collector.use_data(self._data, self.config.context)
   614
   615	        # Early warning if we aren't going to be able to support plugins.
   616	        if self._plugins.file_tracers and not self._core.supports_plugins:
   617	            self._warn(
   618	                "Plugin file tracers ({}) aren't supported with {}".format(
   619	                    ", ".join(
   620	                        plugin._coverage_plugin_name
   621	                            for plugin in self._plugins.file_tracers
   622	                    ),
   623	                    self._collector.tracer_name(),
   624	                ),
   625	            )
   626	            for plugin in self._plugins.file_tracers:
   627	                plugin._coverage_enabled = False
   628
   629	        # Create the file classifying substructure.
   630	        self._inorout = InOrOut(
   631	            config=self.config,
   632	            warn=self._warn,
   633	            debug=(self._debug if self._debug.should("trace") else None),
   634	            include_namespace_packages=self.config.include_namespace_packages,
   635	        )
   636	        self._inorout.plugins = self._plugins
   637	        self._inorout.disp_class = self._core.file_disposition_class
   638
   639	        # It's useful to write debug info after initing for start.
   640	        self._should_write_debug = True
   641
   642	        # Register our clean-up handlers.
   643	        atexit.register(self._atexit)
   644	        if self.config.sigterm:
   645	            is_main = (threading.current_thread() == threading.main_thread())
   646	            if is_main and not env.WINDOWS:
   647	                # The Python docs seem to imply that SIGTERM works uniformly even
   648	                # on Windows, but that's not my experience, and this agrees:
   649	                # https://stackoverflow.com/questions/35772001/x/35792192#35792192
   650	                self._old_sigterm = signal.signal(      # type: ignore[assignment]
   651	                    signal.SIGTERM, self._on_sigterm,
   652	                )

   670	    def start(self) -> None:
   671	        """Start measuring code coverage.
   672
   673	        Coverage measurement is only collected in functions called after
   674	        :meth:`start` is invoked.  Statements in the same scope as
   675	        :meth:`start` won't be measured.
   676
   677	        Once you invoke :meth:`start`, you must also call :meth:`stop`
   678	        eventually, or your process might not shut down cleanly.
   679
   680	        The :meth:`collect` method is a context manager to handle both
   681	        starting and stopping collection.
   682
   683	        """
   684	        self._init()
   685	        if not self._inited_for_start:
   686	            self._inited_for_start = True
   687	            self._init_for_start()
   688	        self._post_init()
   689
   690	        assert self._collector is not None
   691	        assert self._inorout is not None
   692
   693	        # Issue warnings for possible problems.
   694	        self._inorout.warn_conflicting_settings()
   695
   696	        # See if we think some code that would eventually be measured has
   697	        # already been imported.
   698	        if self._warn_preimported_source:
   699	            self._inorout.warn_already_imported_files()
   700
   701	        if self._auto_load:
   702	            self.load()
   703
   704	        apply_patches(self, self.config, self._debug, make_pth_file=self._make_pth_file)
   705
   706	        self._collector.start()
   707	        self._started = True
   708	        self._instances.append(self)
   709
   710	    def stop(self) -> None:
   711	        """Stop measuring code coverage."""
   712	        if self._instances:
   713	            if self._instances[-1] is self:
   714	                self._instances.pop()
   715	        if self._started:
   716	            assert self._collector is not None
   717	            self._collector.stop()
   718	        self._started = False

   720	    @contextlib.contextmanager
   721	    def collect(self) -> Iterator[None]:
   722	        """A context manager to start/stop coverage measurement collection.
   723
   724	        .. versionadded:: 7.3
   725
   726	        """
   727	        self.start()
   728	        try:
   729	            yield
   730	        finally:
   731	            self.stop()     # pragma: nested

   733	    def _atexit(self, event: str = "atexit") -> None:
   734	        """Clean up on process shutdown."""
   735	        if self._debug.should("process"):
   736	            self._debug.write(f"{event}: pid: {os.getpid()}, instance: {self!r}")
   737	        if self._started:
   738	            self.stop()
   739	        if self._auto_save or event == "sigterm":
   740	            self.save()
   741	        for d in self._data_to_close:
   742	            d.close(force=True)

   744	    def _on_sigterm(self, signum_unused: int, frame_unused: FrameType | None) -> None:
   745	        """A handler for signal.SIGTERM."""
   746	        self._atexit("sigterm")
   747	        # Statements after here won't be seen by metacov because we just wrote
   748	        # the data, and are about to kill the process.
   749	        signal.signal(signal.SIGTERM, self._old_sigterm)    # pragma: not covered
   750	        os.kill(os.getpid(), signal.SIGTERM)                # pragma: not covered

   901	    def get_data(self) -> CoverageData:
   902	        """Get the collected data.
   903
   904	        Also warn about various problems collecting data.
   905
   906	        Returns a :class:`coverage.CoverageData`, the collected coverage data.
   907
   908	        .. versionadded:: 4.0
   909
   910	        """
   911	        self._init()
   912	        self._init_data(suffix=None)
   913	        self._post_init()
   914
   915	        if self._collector is not None:
   916	            for plugin in self._plugins:
   917	                if not plugin._coverage_enabled:
   918	                    self._collector.plugin_was_disabled(plugin)
   919
   920	            if self._collector.flush_data():
   921	                self._post_save_work()
   922
   923	        assert self._data is not None
   924	        return self._data

   926	    def _post_save_work(self) -> None:
   927	        """After saving data, look for warnings, post-work, etc.
   928
   929	        Warn about things that should have happened but didn't.
   930	        Look for un-executed files.
   931
   932	        """
   933	        assert self._data is not None
   934	        assert self._inorout is not None
   935
   936	        # If there are still entries in the source_pkgs_unmatched list,
   937	        # then we never encountered those packages.
   938	        if self._warn_unimported_source:
   939	            self._inorout.warn_unimported_source()
   940
   941	        # Find out if we got any data.
   942	        if not self._data and self._warn_no_data:
   943	            self._warn("No data was collected.", slug="no-data-collected")
   944
   945	        # Touch all the files that could have executed, so that we can
   946	        # mark completely un-executed files as 0% covered.
   947	        file_paths = collections.defaultdict(list)
   948	        for file_path, plugin_name in self._inorout.find_possibly_unexecuted_files():
   949	            file_path = self._file_mapper(file_path)
   950	            file_paths[plugin_name].append(file_path)
   951	        for plugin_name, paths in file_paths.items():
   952	            self._data.touch_files(paths, plugin_name)

  1016	        """Get a FileReporter for a module or file name."""
  1017	        assert self._data is not None
  1018	        plugin = None
  1019	        file_reporter: str | FileReporter = "python"
  1020
  1021	        if isinstance(morf, str):

  1023	            plugin_name = self._data.file_tracer(mapped_morf)
  1024	            if plugin_name:
  1025	                plugin = self._plugins.get(plugin_name)
  1026
  1027	                if plugin:
  1028	                    file_reporter = plugin.file_reporter(mapped_morf)
  1029	                    if file_reporter is None:
  1030	                        raise PluginError(
  1031	                            "Plugin {!r} did not provide a file reporter for {!r}.".format(
  1032	                                plugin._coverage_plugin_name, morf,
  1033	                            ),
  1034	                        )
  1035
  1036	        if file_reporter == "python":
  1037	            file_reporter = PythonFileReporter(morf, self)

  1045	    ) -> list[tuple[FileReporter, TMorf]]:

  1049	        a list pairing FileReporters with the morfs.
