   146	    def _getBoxAnchor(self):
   147	        '''hook for allowing special box anchor effects'''
   148	        ba = self.boxAnchor
   149	        if ba in ('autox', 'autoy'):
   150	            angle = self.angle
   151	            na = (int((angle%360)/45.)*45)%360
   152	            if not (na % 90): # we have a right angle case
   153	                da = (angle - na) % 360
   154	                if abs(da)>5:
   155	                    na = na + (da>0 and 45 or -45)
   156	            ba = _A2BA[ba[-1]][na]
   157	        return ba

   159	    def _getBaseLineRatio(self):
   160	        if self.useAscentDescent:
   161	            self._ascent, self._descent = getAscentDescent(self.fontName,self.fontSize)
   162	            self._baselineRatio = self._ascent/(self._ascent-self._descent)
   163	        else:
   164	            self._baselineRatio = 1/1.2

   166	    def _computeSizeEnd(self,objH):
   167	        self._height = self.height or (objH + self.topPadding + self.bottomPadding)
   168	        self._ewidth = (self._width-self.leftPadding-self.rightPadding)
   169	        self._eheight = (self._height-self.topPadding-self.bottomPadding)
   170	        boxAnchor = self._getBoxAnchor()
   171	        if boxAnchor in ['n','ne','nw']:
   172	            self._top = -self.topPadding
   173	        elif boxAnchor in ['s','sw','se']:
   174	            self._top = self._height-self.topPadding
   175	        else:
   176	            self._top = 0.5*self._eheight
   177	        self._bottom = self._top - self._eheight
   178
   179	        if boxAnchor in ['ne','e','se']:
   180	            self._left = self.leftPadding - self._width
   181	        elif boxAnchor in ['nw','w','sw']:
   182	            self._left = self.leftPadding
   183	        else:
   184	            self._left = -self._ewidth*0.5
   185	        self._right = self._left+self._ewidth

   187	    def computeSize(self):
   188	        # the thing will draw in its own coordinate system
   189	        ddfKlass = getattr(self,'ddfKlass',None)
   190	        if not ddfKlass:
   191	            self._lineWidths = []
   192	            self._lines = simpleSplit(self._text,self.fontName,self.fontSize,self.maxWidth)
   193	            if not self.width:
   194	                self._width = self.leftPadding+self.rightPadding
   195	                if self._lines:
   196	                    self._lineWidths = [stringWidth(line,self.fontName,self.fontSize) for line in self._lines]
   197	                    self._width += max(self._lineWidths)
   198	            else:
   199	                self._width = self.width
   200	            self._getBaseLineRatio()
   201	            if self.leading:
   202	                self._leading = self.leading
   203	            elif self.useAscentDescent:
   204	                self._leading = self._ascent - self._descent
   205	            else:
   206	                self._leading = self.fontSize*1.2
   207	            objH = self._leading*len(self._lines)
   208	        else:
   209	            if self.ddf is None:
   210	                raise RuntimeError('DirectDrawFlowable class is not available you need the rlextra package as well as reportlab')
   211	            sty = dict(
   212	                    name='xlabel-generated',
   213	                    fontName=self.fontName,
   214	                    fontSize=self.fontSize,
   215	                    fillColor=self.fillColor,
   216	                    strokeColor=self.strokeColor,
   217	                    )
   218
   219	            if not self.ddfStyle:
   220	                sty = ParagraphStyle(**sty)
   221	            elif issubclass(self.ddfStyle,PropertySet):
   222	                sty = self.ddfStyle(**sty)
   223	            else:
   224	                sty = self.ddfStyle.clone()
   225
   226	            self._style = sty
   227	            self._getBaseLineRatio()
   228	            if self.useAscentDescent:
   229	                sty.autoLeading = True
   230	                sty.leading = self._ascent - self._descent
   231	            else:
   232	                sty.leading = self.leading if self.leading else self.fontSize*1.2
   233	            self._leading = sty.leading
   234	            ta = self._getTextAnchor()
   235	            aW = self.maxWidth or 0x7fffffff
   236	            if ta!='start':
   237	                sty.alignment = TA_LEFT
   238	                obj = ddfKlass(self._text,style=sty)
   239	                _, objH = obj.wrap(aW,0x7fffffff)
   240	                aW = self.maxWidth or obj._width_max
   241	            sty.alignment = _ta2al[ta]
   242	            self._ddfObj = obj = ddfKlass(self._text,style=sty)
   243	            _, objH = obj.wrap(aW,0x7fffffff)
   244
   245	            if not self.width:
   246	                self._width = self.leftPadding+self.rightPadding
   247	                self._width += obj._width_max
   248	            else:
   249	                self._width = self.width
   250	        self._computeSizeEnd(objH)

   252	    def _getTextAnchor(self):
   253	        '''This can be overridden to allow special effects'''
   254	        ta = self.textAnchor
   255	        if ta=='boxauto': ta = _BA2TA[self._getBoxAnchor()]
   256	        return ta

   258	    def _rawDraw(self):
   259	        _text = self._text
   260	        self._text = _text or ''
   261	        self.computeSize()
   262	        self._text = _text
   263	        g = Group()
   264	        g.translate(self.x + self.dx, self.y + self.dy)
   265	        g.rotate(self.angle)
   266
   267	        ddfKlass = getattr(self,'ddfKlass',None)
   268	        if ddfKlass:
   269	            x = self._left
   270	        else:
   271	            y = self._top - self._leading*self._baselineRatio
   272	            textAnchor = self._getTextAnchor()
   273	            if textAnchor == 'start':
   274	                x = self._left
   275	            elif textAnchor == 'middle':
   276	                x = self._left + self._ewidth*0.5
   277	            else:
   278	                x = self._right
   279
   280	        # paint box behind text just in case they
   281	        # fill it
   282	        if self.boxFillColor or (self.boxStrokeColor and self.boxStrokeWidth):
   283	            g.add(Rect( self._left-self.leftPadding,
   284	                        self._bottom-self.bottomPadding,
   285	                        self._width,
   286	                        self._height,
   287	                        strokeColor=self.boxStrokeColor,
   288	                        strokeWidth=self.boxStrokeWidth,
   289	                        fillColor=self.boxFillColor)
   290	                        )
   291
   292	        if ddfKlass:
   293	            g1 = Group()
   294	            g1.translate(x,self._top-self._eheight)
   295	            g1.add(self.ddf(self._ddfObj))
   296	            g.add(g1)
   297	        else:
   298	            fillColor, fontName, fontSize = self.fillColor, self.fontName, self.fontSize
   299	            strokeColor, strokeWidth, leading = self.strokeColor, self.strokeWidth, self._leading
   300	            svgAttrs=getattr(self,'_svgAttrs',{})
   301	            if strokeColor:
   302	                for line in self._lines:
   303	                    s = _text2Path(line, x, y, fontName, fontSize, textAnchor)
   304	                    s.fillColor = fillColor
   305	                    s.strokeColor = strokeColor
   306	                    s.strokeWidth = strokeWidth
   307	                    g.add(s)
   308	                    y -= leading
   309	            else:
   310	                for line in self._lines:
   311	                    s = String(x, y, line, _svgAttrs=svgAttrs)
   312	                    s.textAnchor = textAnchor
   313	                    s.fontName = fontName
   314	                    s.fontSize = fontSize
   315	                    s.fillColor = fillColor
   316	                    g.add(s)
   317	                    y -= leading
   318
   319	        return g

   321	    def draw(self):
   322	        customDrawChanger = getattr(self,'customDrawChanger',None)
   323	        if customDrawChanger:
   324	            customDrawChanger(True,self)
   325	            try:
   326	                return self._rawDraw()
   327	            finally:
   328	                customDrawChanger(False,self)
   329	        else:
   330	            return self._rawDraw()

   386	    def _getValue(self, chart, val):
   387	        flipXY = chart._flipXY
   388	        A = chart.categoryAxis
   389	        jA = A.joinAxis
   390	        if val>=0:
   391	            mode = self.posMode
   392	            delta = self.pos
   393	        else:
   394	            mode = self.negMode
   395	            delta = self.neg
   396	        if flipXY:
   397	            v = A._x
   398	        else:
   399	            v = A._y
   400	        if jA:
   401	            if flipXY:
   402	                _v = jA._x
   403	            else:
   404	                _v = jA._y
   405	            if mode=='high':
   406	                v = _v + jA._length
   407	            elif mode=='low':
   408	                v = _v
   409	            elif mode=='bar':
   410	                v = _v+val
   411	        return v+delta

    35	def nextRoundNumber(x):
    36	    """Return the first 'nice round number' greater than or equal to x
    37
    38	    Used in selecting apropriate tick mark intervals; we say we want
    39	    an interval which places ticks at least 10 points apart, work out
    40	    what that is in chart space, and ask for the nextRoundNumber().
    41	    Tries the series 1,2,5,10,20,50,100.., going up or down as needed.
    42	    """
    43
    44	    #guess to nearest order of magnitude
    45	    if x in (0, 1):
    46	        return x
    47
    48	    if x < 0:
    49	        return -1.0 * nextRoundNumber(-x)
    50	    else:
    51	        lg = int(log10(x))
    52
    53	        if lg == 0:
    54	            if x < 1:
    55	                base = 0.1
    56	            else:
    57	                base = 1.0
    58	        elif lg < 0:
    59	            base = 10.0 ** (lg - 1)
    60	        else:
    61	            base = 10.0 ** lg    # e.g. base(153) = 100
    62	        # base will always be lower than x
    63
    64	        if base >= x:
    65	            return base * 1.0
    66	        elif (base * 2) >= x:
    67	            return base * 2.0
    68	        elif (base * 5) >= x:
    69	            return base * 5.0
    70	        else:
    71	            return base * 10.0

    75	def find_interval(lo,hi,I=5):
    76	    'determine tick parameters for range [lo, hi] using I intervals'
    77
    78	    if lo >= hi:
    79	        if lo==hi:
    80	            if lo==0:
    81	                lo = -.1
    82	                hi =  .1
    83	            else:
    84	                lo = 0.9*lo
    85	                hi = 1.1*hi
    86	        else:
    87	            raise ValueError("lo>hi")
    88	    x=(hi - lo)/float(I)
    89	    b= (x>0 and (x<1 or x>10)) and 10**floor(log10(x)) or 1
    90	    b = b
    91	    while 1:
    92	        a = x/b
    93	        if a<=_intervals[-1]: break
    94	        b = b*10
    95
    96	    j = 0
    97	    while a>_intervals[j]: j = j + 1
    98
    99	    while 1:
   100	        ss = _intervals[j]*b
   101	        n = lo/ss
   102	        l = int(n)-(n<0)
   103	        n = ss*l
   104	        x = ss*(l+I)
   105	        a = I*ss
   106	        if n>0:
   107	            if a>=hi:
   108	                n = 0.0
   109	                x = a
   110	        elif hi<0:
   111	            a = -a
   112	            if lo>a:
   113	                n = a
   114	                x = 0
   115	        if hi<=x and n<=lo: break
   116	        j = j + 1
   117	        if j>_j_max:
   118	            j = 0
   119	            b = b*10
   120	    return n, x, ss, lo - n + x - hi

   180	def findNones(data):
   181	    m = len(data)
   182	    if None in data:
   183	        b = 0
   184	        while b<m and data[b] is None:
   185	            b += 1
   186	        if b==m: return data
   187	        l = m-1
   188	        while data[l] is None:
   189	            l -= 1
   190	        l+=1

   192	        I = [i for i in range(len(data)) if data[i] is None]
   193	        for i in I:
   194	            data[i] = 0.5*(data[i-1]+data[i+1])
   195	        return b, l, data
   196	    return 0,m,data
