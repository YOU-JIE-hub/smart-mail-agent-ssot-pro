  1069	    """Find the most appropriate :class:`ParamType` for the given Python
  1070	    type. If the type isn't provided, it can be inferred from a default
  1071	    value.
  1072	    """
  1073	    guessed_type = False
  1074
  1075	    if ty is None and default is not None:
  1076	        if isinstance(default, (tuple, list)):
  1077	            # If the default is empty, ty will remain None and will
  1078	            # return STRING.
  1079	            if default:
  1080	                item = default[0]
  1081
  1082	                # A tuple of tuples needs to detect the inner types.
  1083	                # Can't call convert recursively because that would
  1084	                # incorrectly unwind the tuple to a single type.
  1085	                if isinstance(item, (tuple, list)):
  1086	                    ty = tuple(map(type, item))
  1087	                else:
  1088	                    ty = type(item)
  1089	        else:
  1090	            ty = type(default)
  1091
  1092	        guessed_type = True
  1093
  1094	    if isinstance(ty, tuple):
  1095	        return Tuple(ty)
  1096
  1097	    if isinstance(ty, ParamType):
  1098	        return ty
  1099
  1100	    if ty is str or ty is None:
  1101	        return STRING
  1102
  1103	    if ty is int:
  1104	        return INT
  1105
  1106	    if ty is float:
  1107	        return FLOAT
  1108
  1109	    if ty is bool:
  1110	        return BOOL
  1111
  1112	    if guessed_type:
  1113	        return STRING
  1114
  1115	    if __debug__:
  1116	        try:
  1117	            if issubclass(ty, ParamType):
  1118	                raise AssertionError(
  1119	                    f"Attempted to use an uninstantiated parameter type ({ty})."
  1120	                )
  1121	        except TypeError:
  1122	            # ty is an instance (correct), so issubclass fails.
  1123	            pass

   137	                # reading so that we can catch at least some errors in

   222	def echo(
   223	    message: t.Any | None = None,
   224	    file: t.IO[t.Any] | None = None,
   225	    nl: bool = True,
   226	    err: bool = False,
   227	    color: bool | None = None,
   228	) -> None:
   229	    """Print a message and newline to stdout or a file. This should be
   230	    used instead of :func:`print` because it provides better support
   231	    for different data, files, and environments.
   232
   233	    Compared to :func:`print`, this does the following:
   234
   235	    -   Ensures that the output encoding is not misconfigured on Linux.
   236	    -   Supports Unicode in the Windows console.
   237	    -   Supports writing to binary outputs, and supports writing bytes
   238	        to text outputs.
   239	    -   Supports colors and styles on Windows.
   240	    -   Removes ANSI color and style codes if the output does not look
   241	        like an interactive terminal.
   242	    -   Always flushes the output.
   243
   244	    :param message: The string or bytes to output. Other objects are
   245	        converted to strings.
   246	    :param file: The file to write to. Defaults to ``stdout``.
   247	    :param err: Write to ``stderr`` instead of ``stdout``.
   248	    :param nl: Print a newline after the message. Enabled by default.
   249	    :param color: Force showing or hiding colors and other styles. By
   250	        default Click will remove color if the output does not look like
   251	        an interactive terminal.
   252
   253	    .. versionchanged:: 6.0
   254	        Support Unicode output on the Windows console. Click does not
   255	        modify ``sys.stdout``, so ``sys.stdout.write()`` and ``print()``
   256	        will still not support Unicode.
   257
   258	    .. versionchanged:: 4.0
   259	        Added the ``color`` parameter.
   260
   261	    .. versionadded:: 3.0
   262	        Added the ``err`` parameter.
   263
   264	    .. versionchanged:: 2.0
   265	        Support colors on Windows if colorama is installed.
   266	    """
   267	    if file is None:
   268	        if err:
   269	            file = _default_text_stderr()
   270	        else:
   271	            file = _default_text_stdout()
   272
   273	        # There are no standard streams attached to write to. For example,
   274	        # pythonw on Windows.
   275	        if file is None:
   276	            return
   277
   278	    # Convert non bytes/text into the native string type.
   279	    if message is not None and not isinstance(message, (str, bytes, bytearray)):
   280	        out: str | bytes | None = str(message)
   281	    else:
   282	        out = message
   283
   284	    if nl:
   285	        out = out or ""
   286	        if isinstance(out, str):
   287	            out += "\n"
   288	        else:
   289	            out += b"\n"
   290
   291	    if not out:
   292	        file.flush()
   293	        return
   294
   295	    # If there is a message and the value looks like bytes, we manually
   296	    # need to find the binary stream and write the message in there.
   297	    # This is done separately so that most stream types will work as you
   298	    # would expect. Eg: you can write to StringIO for other cases.
   299	    if isinstance(out, (bytes, bytearray)):
   300	        binary_file = _find_binary_writer(file)
   301
   302	        if binary_file is not None:
   303	            file.flush()
   304	            binary_file.write(out)
   305	            binary_file.flush()
   306	            return
   307
   308	    # ANSI style code support. For no message or bytes, nothing happens.
   309	    # When outputting to a file instead of a terminal, strip codes.
   310	    else:
   311	        color = resolve_color_default(color)
   312
   313	        if should_strip_ansi(file, color):
   314	            out = strip_ansi(out)
   315	        elif WIN:
   316	            if auto_wrap_for_ansi is not None:
   317	                file = auto_wrap_for_ansi(file, color)  # type: ignore
   318	            elif not color:
   319	                out = strip_ansi(out)
   320
   321	    file.write(out)  # type: ignore
   322	    file.flush()

   358	def open_file(
   359	    filename: str | os.PathLike[str],
   360	    mode: str = "r",
   361	    encoding: str | None = None,
   362	    errors: str | None = "strict",
   363	    lazy: bool = False,
   364	    atomic: bool = False,
   365	) -> t.IO[t.Any]:
   366	    """Open a file, with extra behavior to handle ``'-'`` to indicate
   367	    a standard stream, lazy open on write, and atomic write. Similar to
   368	    the behavior of the :class:`~click.File` param type.
   369
   370	    If ``'-'`` is given to open ``stdout`` or ``stdin``, the stream is
   371	    wrapped so that using it in a context manager will not close it.
   372	    This makes it possible to use the function without accidentally
   373	    closing a standard stream:
   374
   375	    .. code-block:: python
   376
   377	        with open_file(filename) as f:
   378	            ...
   379
   380	    :param filename: The name or Path of the file to open, or ``'-'`` for
   381	        ``stdin``/``stdout``.
   382	    :param mode: The mode in which to open the file.
   383	    :param encoding: The encoding to decode or encode a file opened in
   384	        text mode.
   385	    :param errors: The error handling mode.
   386	    :param lazy: Wait to open the file until it is accessed. For read
   387	        mode, the file is temporarily opened to raise access errors
   388	        early, then closed until it is read again.
   389	    :param atomic: Write to a temporary file and replace the given file
   390	        on close.
   391
   392	    .. versionadded:: 3.0
   393	    """
   394	    if lazy:
   395	        return t.cast(
   396	            "t.IO[t.Any]", LazyFile(filename, mode, encoding, errors, atomic=atomic)
   397	        )
   398
   399	    f, should_close = open_stream(filename, mode, encoding, errors, atomic=atomic)
   400
   401	    if not should_close:
   402	        f = t.cast("t.IO[t.Any]", KeepOpenFile(f))
   403
   404	    return f

   407	def format_filename(
   408	    filename: str | bytes | os.PathLike[str] | os.PathLike[bytes],
   409	    shorten: bool = False,
   410	) -> str:
   411	    """Format a filename as a string for display. Ensures the filename can be
   412	    displayed by replacing any invalid bytes or surrogate escapes in the name
   413	    with the replacement character ``�``.
   414
   415	    Invalid bytes or surrogate escapes will raise an error when written to a
   416	    stream with ``errors="strict"``. This will typically happen with ``stdout``
   417	    when the locale is something like ``en_GB.UTF-8``.
   418
   419	    Many scenarios *are* safe to write surrogates though, due to PEP 538 and
   420	    PEP 540, including:
   421
   422	    -   Writing to ``stderr``, which uses ``errors="backslashreplace"``.
   423	    -   The system has ``LANG=C.UTF-8``, ``C``, or ``POSIX``. Python opens
   424	        stdout and stderr with ``errors="surrogateescape"``.
   425	    -   None of ``LANG/LC_*`` are set. Python assumes ``LANG=C.UTF-8``.
   426	    -   Python is started in UTF-8 mode  with  ``PYTHONUTF8=1`` or ``-X utf8``.
   427	        Python opens stdout and stderr with ``errors="surrogateescape"``.
   428
   429	    :param filename: formats a filename for UI display.  This will also convert
   430	                     the filename into unicode without failing.
   431	    :param shorten: this optionally shortens the filename to strip of the
   432	                    path that leads up to it.
   433	    """
   434	    if shorten:
   435	        filename = os.path.basename(filename)
   436	    else:
   437	        filename = os.fspath(filename)
   438
   439	    if isinstance(filename, bytes):
   440	        filename = filename.decode(sys.getfilesystemencoding(), "replace")
   441	    else:
   442	        filename = filename.encode("utf-8", "surrogateescape").decode(
   443	            "utf-8", "replace"
   444	        )
   445
   446	    return filename

   498	class PacifyFlushWrapper:
   499	    """This wrapper is used to catch and suppress BrokenPipeErrors resulting
   500	    from ``.flush()`` being called on broken pipe during the shutdown/final-GC
   501	    of the Python interpreter. Notably ``.flush()`` is always called on
   502	    ``sys.stdout`` and ``sys.stderr``. So as to have minimal impact on any
   503	    other cleanup code, and the case where the underlying file is not a broken
   504	    pipe, all calls and attributes are proxied.
   505	    """
   506
   507	    def __init__(self, wrapped: t.IO[t.Any]) -> None:
   508	        self.wrapped = wrapped
   509
   510	    def flush(self) -> None:
   511	        try:
   512	            self.wrapped.flush()
   513	        except OSError as e:
   514	            import errno
   515
   516	            if e.errno != errno.EPIPE:
   517	                raise
   518
   519	    def __getattr__(self, attr: str) -> t.Any:
   520	        return getattr(self.wrapped, attr)

   523	def _detect_program_name(
   524	    path: str | None = None, _main: ModuleType | None = None
   525	) -> str:
   526	    """Determine the command used to run the program, for use in help
   527	    text. If a file or entry point was executed, the file name is
   528	    returned. If ``python -m`` was used to execute a module or package,
   529	    ``python -m name`` is returned.
   530
   531	    This doesn't try to be too precise, the goal is to give a concise
   532	    name for help text. Files are only shown as their name without the
   533	    path. ``python`` is only shown for modules, and the full path to
   534	    ``sys.executable`` is not shown.
   535
   536	    :param path: The Python file being executed. Python puts this in
   537	        ``sys.argv[0]``, which is used by default.
   538	    :param _main: The ``__main__`` module. This should only be passed
   539	        during internal testing.
   540
   541	    .. versionadded:: 8.0
   542	        Based on command args detection in the Werkzeug reloader.
   543
   544	    :meta private:
   545	    """
   546	    if _main is None:
   547	        _main = sys.modules["__main__"]
   548
   549	    if not path:
   550	        path = sys.argv[0]
   551
   552	    # The value of __package__ indicates how Python was called. It may
   553	    # not exist if a setuptools script is installed as an egg. It may be
   554	    # set incorrectly for entry points created with pip on Windows.
   555	    # It is set to "" inside a Shiv or PEX zipapp.
   556	    if getattr(_main, "__package__", None) in {None, ""} or (
   557	        os.name == "nt"
   558	        and _main.__package__ == ""
   559	        and not os.path.exists(path)
   560	        and os.path.exists(f"{path}.exe")
   561	    ):
   562	        # Executed a file, like "python app.py".
   563	        return os.path.basename(path)
   564
   565	    # Executed a module, like "python -m example".
   566	    # Rewritten by Python from "-m script" to "/path/to/script.py".
   567	    # Need to look at main module to determine how it was executed.
   568	    py_module = t.cast(str, _main.__package__)
   569	    name = os.path.splitext(os.path.basename(path))[0]
   570
   571	    # A submodule like "example.cli".
   572	    if name != "__main__":
   573	        py_module = f"{py_module}.{name}"
   574
   575	    return f"python -m {py_module.lstrip('.')}"

   578	def _expand_args(
   579	    args: cabc.Iterable[str],
   580	    *,
   581	    user: bool = True,
   582	    env: bool = True,
   583	    glob_recursive: bool = True,
   584	) -> list[str]:
   585	    """Simulate Unix shell expansion with Python functions.
   586
   587	    See :func:`glob.glob`, :func:`os.path.expanduser`, and
   588	    :func:`os.path.expandvars`.
   589
   590	    This is intended for use on Windows, where the shell does not do any
   591	    expansion. It may not exactly match what a Unix shell would do.
   592
   593	    :param args: List of command line arguments to expand.
   594	    :param user: Expand user home directory.
   595	    :param env: Expand environment variables.
   596	    :param glob_recursive: ``**`` matches directories recursively.
   597
   598	    .. versionchanged:: 8.1
   599	        Invalid glob patterns are treated as empty expansions rather
   600	        than raising an error.
   601
   602	    .. versionadded:: 8.0
   603
   604	    :meta private:
   605	    """
   606	    from glob import glob
   607
   608	    out = []
   609
   610	    for arg in args:
   611	        if user:
   612	            arg = os.path.expanduser(arg)
   613
   614	        if env:
   615	            arg = os.path.expandvars(arg)
   616
   617	        try:
   618	            matches = glob(arg, recursive=glob_recursive)
   619	        except re.error:
   620	            matches = []
   621
   622	        if not matches:
   623	            out.append(arg)
   624	        else:
   625	            out.extend(matches)
   626
   627	    return out

    58	class InstructionWalker:
    59	    """Utility to step through trails of instructions.
    60
    61	    We have two reasons to need sequences of instructions from a code object:
    62	    First, in strict sequence to visit all the instructions in the object.
    63	    This is `walk(follow_jumps=False)`.  Second, we want to follow jumps to
    64	    understand how execution will flow: `walk(follow_jumps=True)`.
    65	    """
    66
    67	    def __init__(self, code: CodeType) -> None:
    68	        self.code = code
    69	        self.insts: dict[TOffset, dis.Instruction] = {}
    70
    71	        inst = None
    72	        for inst in dis.get_instructions(code):
    73	            self.insts[inst.offset] = inst
    74
    75	        assert inst is not None
    76	        self.max_offset = inst.offset
    77
    78	    def walk(
    79	        self, *, start_at: TOffset = 0, follow_jumps: bool = True
    80	    ) -> Iterable[dis.Instruction]:
    81	        """
    82	        Yield instructions starting from `start_at`.  Follow unconditional
    83	        jumps if `follow_jumps` is true.
    84	        """
    85	        seen = set()
    86	        offset = start_at
    87	        while offset < self.max_offset + 1:
    88	            if offset in seen:
    89	                break
    90	            seen.add(offset)
    91	            if inst := self.insts.get(offset):
    92	                yield inst
    93	                if follow_jumps and inst.opcode in ALWAYS_JUMPS:
    94	                    offset = inst.jump_target
    95	                    continue
    96	            offset += 2

   103	def branch_trails(code: CodeType) -> TBranchTrails:
   104	    """
   105	    Calculate branch trails for `code`.
   106
   107	    Instructions can have a jump_target, where they might jump to next.  Some
   108	    instructions with a jump_target are unconditional jumps (ALWAYS_JUMPS), so
   109	    they aren't interesting to us, since they aren't the start of a branch
   110	    possibility.
   111
   112	    Instructions that might or might not jump somewhere else are branch
   113	    possibilities.  For each of those, we track a trail of instructions.  These
   114	    are lists of instruction offsets, the next instructions that can execute.
   115	    We follow the trail until we get to a new source line.  That gives us the
   116	    arc from the original instruction's line to the new source line.
   117
   118	    """
   119	    the_trails: TBranchTrails = collections.defaultdict(
   120	        lambda: collections.defaultdict(set)
   121	    )
   122	    iwalker = InstructionWalker(code)
   123	    for inst in iwalker.walk(follow_jumps=False):
   124	        if not inst.jump_target:
   125	            # We only care about instructions with jump targets.
   126	            continue
   127	        if inst.opcode in ALWAYS_JUMPS:
   128	            # We don't care about unconditional jumps.
   129	            continue
   130
   131	        from_line = inst.line_number
   132	        if from_line is None:
   133	            continue
   134
   135	        def add_one_branch_trail(
   136	            trails: TBranchTrailsOneSource,
   137	            start_at: TOffset,
   138	        ) -> None:
   139	            # pylint: disable=cell-var-from-loop
   140	            inst_offsets: set[TOffset] = set()
   141	            to_line = None
   142	            for inst2 in iwalker.walk(start_at=start_at, follow_jumps=True):
   143	                inst_offsets.add(inst2.offset)
   144	                if inst2.line_number and inst2.line_number != from_line:
   145	                    to_line = inst2.line_number
   146	                    break
   147	                elif inst2.jump_target and (inst2.opcode not in ALWAYS_JUMPS):
   148	                    break
   149	                elif inst2.opcode in RETURNS:
   150	                    to_line = -code.co_firstlineno
   151	                    break
   152	            if to_line is not None:
   153	                trails[(from_line, to_line)].update(inst_offsets)
   154	            else:
   155	                trails[None] = set()
   156
   157	        # Calculate two trails: one from the next instruction, and one from the
   158	        # jump_target instruction.
   159	        trails: TBranchTrailsOneSource = collections.defaultdict(set)
   160	        add_one_branch_trail(trails, start_at=inst.offset + 2)
   161	        add_one_branch_trail(trails, start_at=inst.jump_target)
   162	        the_trails[inst.offset] = trails
   163
   164	        # Sometimes we get BRANCH_RIGHT or BRANCH_LEFT events from instructions
   165	        # other than the original jump possibility instruction.  Register each
   166	        # trail under all of their offsets so we can pick up in the middle of a
   167	        # trail if need be.
   168	        for arc, offsets in trails.items():
   169	            for offset in offsets:
   170	                the_trails[offset][arc].update(offsets)
   171
   172	    return the_trails

   175	def always_jumps(code: CodeType) -> dict[TOffset, TOffset]:
   176	    """Make a map of unconditional bytecodes jumping to others.
   177
   178	    Only include bytecodes that do no work and go to another bytecode.
   179	    """
   180	    jumps = {}
   181	    iwalker = InstructionWalker(code)
   182	    for inst in iwalker.walk(follow_jumps=False):

   184	            jumps[inst.offset] = inst.jump_target
   185	        elif inst.opcode in NOPS:

   559	def show_help(
   560	    error: str | None = None,
   561	    topic: str | None = None,
   562	    parser: optparse.OptionParser | None = None,
   563	) -> None:
   564	    """Display an error message, or the named topic."""
   565	    assert error or topic or parser
   566
   567	    program_path = sys.argv[0]
   568	    if program_path.endswith(os.path.sep + "__main__.py"):
   569	        # The path is the main module of a package; get that path instead.
   570	        program_path = os.path.dirname(program_path)
   571	    program_name = os.path.basename(program_path)
   572	    if env.WINDOWS:
   573	        # entry_points={"console_scripts":...} on Windows makes files
   574	        # called coverage.exe, coverage3.exe, and coverage-3.5.exe. These
   575	        # invoke coverage-script.py, coverage3-script.py, and
   576	        # coverage-3.5-script.py.  argv[0] is the .py file, but we want to
   577	        # get back to the original form.
   578	        auto_suffix = "-script.py"
   579	        if program_name.endswith(auto_suffix):
   580	            program_name = program_name[:-len(auto_suffix)]
   581
   582	    help_params = dict(coverage.__dict__)
   583	    help_params["__url__"] = __url__
   584	    help_params["program_name"] = program_name
   585	    if CTRACER_FILE:
   586	        help_params["extension_modifier"] = "with C extension"
   587	    else:
   588	        help_params["extension_modifier"] = "without C extension"
   589
   590	    if error:
   591	        print(error, file=sys.stderr)
   592	        print(f"Use '{program_name} help' for help.", file=sys.stderr)
   593	    elif parser:
   594	        print(parser.format_help().strip())
   595	        print()
   596	    else:
   597	        assert topic is not None
   598	        help_msg = textwrap.dedent(HELP_TOPICS.get(topic, "")).strip()
   599	        if help_msg:
   600	            print(help_msg.format(**help_params))
   601	        else:
   602	            print(f"Don't know topic {topic!r}")
   603	    print("Full documentation is at {__url__}".format(**help_params))

   609	class CoverageScript:
   610	    """The command-line interface to coverage.py."""
   611
   612	    def __init__(self) -> None:
   613	        self.global_option = False
   614	        self.coverage: Coverage
   615
   616	    def command_line(self, argv: list[str]) -> int:
   617	        """The bulk of the command line interface to coverage.py.
   618
   619	        `argv` is the argument list to process.
   620
   621	        Returns 0 if all is well, 1 if something went wrong.
   622
   623	        """
   624	        # Collect the command-line options.
   625	        if not argv:
   626	            show_help(topic="minimum_help")
   627	            return OK
   628
   629	        # The command syntax we parse depends on the first argument.  Global
   630	        # switch syntax always starts with an option.
   631	        parser: optparse.OptionParser | None
   632	        self.global_option = argv[0].startswith("-")
   633	        if self.global_option:
   634	            parser = GlobalOptionParser()
   635	        else:
   636	            parser = COMMANDS.get(argv[0])
   637	            if not parser:
   638	                show_help(f"Unknown command: {argv[0]!r}")
   639	                return ERR
   640	            argv = argv[1:]
   641
   642	        ok, options, args = parser.parse_args_ok(argv)
   643	        if not ok:
   644	            return ERR
   645	        assert options is not None
   646
   647	        # Handle help and version.
   648	        if self.do_help(options, args, parser):
   649	            return OK
   650
   651	        # Listify the list options.
   652	        source = unshell_list(options.source)
   653	        omit = unshell_list(options.omit)
   654	        include = unshell_list(options.include)
   655	        debug = unshell_list(options.debug)
   656	        contexts = unshell_list(options.contexts)
   657
   658	        if options.concurrency is not None:
   659	            concurrency = options.concurrency.split(",")
   660	        else:
   661	            concurrency = None
   662
   663	        # Do something.
   664	        self.coverage = Coverage(
   665	            data_file=options.data_file or DEFAULT_DATAFILE,
   666	            data_suffix=options.parallel_mode,
   667	            cover_pylib=options.pylib,
   668	            timid=options.timid,
   669	            branch=options.branch,
   670	            config_file=options.rcfile,
   671	            source=source,
   672	            omit=omit,
   673	            include=include,
   674	            debug=debug,
   675	            concurrency=concurrency,
   676	            check_preimported=True,
   677	            context=options.context,
   678	            messages=not options.quiet,
   679	        )
   680
   681	        if options.action == "debug":
   682	            return self.do_debug(args)
   683
   684	        elif options.action == "erase":
   685	            self.coverage.erase()
   686	            return OK
   687
   688	        elif options.action == "run":
   689	            return self.do_run(options, args)
   690
   691	        elif options.action == "combine":
   692	            if options.append:
   693	                self.coverage.load()
   694	            data_paths = args or None
   695	            self.coverage.combine(data_paths, strict=True, keep=bool(options.keep))
   696	            self.coverage.save()
   697	            return OK
   698
   699	        # Remaining actions are reporting, with some common options.
   700	        report_args = dict(
   701	            morfs=unglob_args(args),
   702	            ignore_errors=options.ignore_errors,
   703	            omit=omit,
   704	            include=include,
   705	            contexts=contexts,
   706	        )
   707
   708	        # We need to be able to import from the current directory, because
   709	        # plugins may try to, for example, to read Django settings.
   710	        sys.path.insert(0, "")
   711
   712	        self.coverage.load()
   713
   714	        total = None
   715	        if options.action == "report":
   716	            total = self.coverage.report(
   717	                precision=options.precision,
   718	                show_missing=options.show_missing,
   719	                skip_covered=options.skip_covered,
   720	                skip_empty=options.skip_empty,
   721	                sort=options.sort,
   722	                output_format=options.format,
   723	                **report_args,
   724	            )
   725	        elif options.action == "annotate":
   726	            self.coverage.annotate(directory=options.directory, **report_args)
   727	        elif options.action == "html":
   728	            total = self.coverage.html_report(
   729	                directory=options.directory,
   730	                precision=options.precision,
   731	                skip_covered=options.skip_covered,
   732	                skip_empty=options.skip_empty,
   733	                show_contexts=options.show_contexts,
   734	                title=options.title,
   735	                **report_args,
   736	            )
   737	        elif options.action == "xml":
   738	            total = self.coverage.xml_report(
   739	                outfile=options.outfile,
   740	                skip_empty=options.skip_empty,
   741	                **report_args,
   742	            )
   743	        elif options.action == "json":
   744	            total = self.coverage.json_report(
   745	                outfile=options.outfile,
   746	                pretty_print=options.pretty_print,
   747	                show_contexts=options.show_contexts,
   748	                **report_args,
   749	            )
   750	        elif options.action == "lcov":
   751	            total = self.coverage.lcov_report(
   752	                outfile=options.outfile,
   753	                **report_args,
   754	            )
   755	        else:
   756	            # There are no other possible actions.
   757	            raise AssertionError
   758
   759	        if total is not None:
   760	            # Apply the command line fail-under options, and then use the config
   761	            # value, so we can get fail_under from the config file.
   762	            if options.fail_under is not None:
   763	                self.coverage.set_option("report:fail_under", options.fail_under)
   764	            if options.precision is not None:
   765	                self.coverage.set_option("report:precision", options.precision)
   766
   767	            fail_under = cast(float, self.coverage.get_option("report:fail_under"))
   768	            precision = cast(int, self.coverage.get_option("report:precision"))
   769	            if should_fail_under(total, fail_under, precision):
   770	                msg = "total of {total} is less than fail-under={fail_under:.{p}f}".format(
   771	                    total=display_covered(total, precision),
   772	                    fail_under=fail_under,
   773	                    p=precision,
   774	                )
   775	                print("Coverage failure:", msg)
   776	                return FAIL_UNDER
   777
   778	        return OK

   780	    def do_help(
   781	        self,
   782	        options: optparse.Values,
   783	        args: list[str],
   784	        parser: optparse.OptionParser,
   785	    ) -> bool:
   786	        """Deal with help requests.
   787
   788	        Return True if it handled the request, False if not.
   789
   790	        """
   791	        # Handle help.
   792	        if options.help:
   793	            if self.global_option:
   794	                show_help(topic="help")
   795	            else:
   796	                show_help(parser=parser)
   797	            return True
   798
   799	        if options.action == "help":
   800	            if args:
   801	                for a in args:
   802	                    parser_maybe = COMMANDS.get(a)
   803	                    if parser_maybe is not None:
   804	                        show_help(parser=parser_maybe)
   805	                    else:
   806	                        show_help(topic=a)
   807	            else:
   808	                show_help(topic="help")
   809	            return True
   810
   811	        # Handle version.
   812	        if options.version:
   813	            show_help(topic="version")
   814	            return True
   815
   816	        return False

   823	    def do_run(self, options: optparse.Values, args: list[str]) -> int:
   824	        """Implementation of 'coverage run'."""
   825
   826	        if not args:
   827	            if options.module:
   828	                # Specified -m with nothing else.
   829	                show_help("No module specified for -m")
   830	                return ERR
   831	            command_line = cast(str, self.coverage.get_option("run:command_line"))
   832	            if command_line is not None:
   833	                args = shlex.split(command_line)
   834	                if args and args[0] in {"-m", "--module"}:
   835	                    options.module = True
   836	                    args = args[1:]
   837	        if not args:
   838	            show_help("Nothing to do.")
   839	            return ERR
   840
   841	        if options.append and self.coverage.get_option("run:parallel"):
   842	            show_help("Can't append to data files in parallel mode.")
   843	            return ERR
   844
   845	        if options.concurrency == "multiprocessing":
   846	            # Can't set other run-affecting command line options with
   847	            # multiprocessing.
   848	            for opt_name in ["branch", "include", "omit", "pylib", "source", "timid"]:
   849	                # As it happens, all of these options have no default, meaning
   850	                # they will be None if they have not been specified.
   851	                if getattr(options, opt_name) is not None:
   852	                    show_help(
   853	                        "Options affecting multiprocessing must only be specified " +
   854	                        "in a configuration file.\n" +
   855	                        f"Remove --{opt_name} from the command line.",
   856	                    )
   857	                    return ERR
   858
   859	        os.environ["COVERAGE_RUN"] = "true"
   860
   861	        runner = PyRunner(args, as_module=bool(options.module))
   862	        runner.prepare()
   863
   864	        if options.append:
   865	            self.coverage.load()
   866
   867	        if options.save_signal:
   868	            if env.WINDOWS:
   869	                show_help("--save-signal is not supported on Windows.")
   870	                return ERR
   871	            sig = getattr(signal, f"SIG{options.save_signal}")
   872	            signal.signal(sig, self.do_signal_save)
   873
   874	        # Run the script.
   875	        self.coverage.start()
   876	        code_ran = True
   877	        try:
   878	            runner.run()
   879	        except NoSource:
   880	            code_ran = False
   881	            raise
   882	        finally:
   883	            self.coverage.stop()
   884	            if code_ran:
   885	                self.coverage.save()
   886
   887	        return OK

   889	    def do_debug(self, args: list[str]) -> int:
   890	        """Implementation of 'coverage debug'."""
   891
   892	        if not args:
   893	            show_help("What information would you like: config, data, sys, premain, pybehave?")
   894	            return ERR
   895	        if args[1:]:
   896	            show_help("Only one topic at a time, please")
   897	            return ERR

   899	        if args[0] == "sys":
   900	            write_formatted_info(print, "sys", self.coverage.sys_info())
   901	        elif args[0] == "data":
   902	            print(info_header("data"))
   903	            data_file = self.coverage.config.data_file
   904	            debug_data_file(data_file)
   905	            for filename in combinable_files(data_file):
   906	                print("-----")
   907	                debug_data_file(filename)
   908	        elif args[0] == "config":
   909	            write_formatted_info(print, "config", self.coverage.config.debug_info())
   910	        elif args[0] == "premain":
   911	            print(info_header("premain"))
   912	            print(short_stack(full=True))
   913	        elif args[0] == "pybehave":
   914	            write_formatted_info(print, "pybehave", env.debug_info())
   915	        else:

   919	        return OK

   922	def unshell_list(s: str) -> list[str] | None:
   923	    """Turn a command-line argument into a list."""
   924	    if not s:
   925	        return None
   926	    if env.WINDOWS:
   927	        # When running coverage.py as coverage.exe, some of the behavior
   928	        # of the shell is emulated: wildcards are expanded into a list of
   929	        # file names.  So you have to single-quote patterns on the command
   930	        # line, but (not) helpfully, the single quotes are included in the
   931	        # argument, so we have to strip them off here.
   932	        s = s.strip("'")
   933	    return s.split(",")

   936	def unglob_args(args: list[str]) -> list[str]:
   937	    """Interpret shell wildcards for platforms that need it."""
   938	    if env.WINDOWS:
   939	        globbed = []
   940	        for arg in args:
   941	            if "?" in arg or "*" in arg:
   942	                globbed.extend(glob.glob(arg))
   943	            else:
   944	                globbed.append(arg)
   945	        args = globbed
   946	    return args

   981	def main(argv: list[str] | None = None) -> int | None:
   982	    """The main entry point to coverage.py.
   983
   984	    This is installed as the script entry point.
   985
   986	    """
   987	    if argv is None:
   988	        argv = sys.argv[1:]
   989	    try:
   990	        status = CoverageScript().command_line(argv)
   991	    except _ExceptionDuringRun as err:
   992	        # An exception was caught while running the product code.  The
   993	        # sys.exc_info() return tuple is packed into an _ExceptionDuringRun
   994	        # exception.
   995	        traceback.print_exception(*err.args)    # pylint: disable=no-value-for-parameter
   996	        status = ERR
   997	    except _BaseCoverageException as err:
   998	        # A controlled error inside coverage.py: print the message to the user.
   999	        msg = err.args[0]
  1000	        print(msg)
  1001	        status = ERR
  1002	    except SystemExit as err:
  1003	        # The user called `sys.exit()`.  Exit with their argument, if any.
  1004	        if err.args:
  1005	            status = err.args[0]
  1006	        else:
  1007	            status = None
  1008	    return status
