    44	def parseAFMFile(afmFileName):
    45	    """Quick and dirty - gives back a top-level dictionary
    46	    with top-level items, and a 'widths' key containing
    47	    a dictionary of glyph names and widths.  Just enough
    48	    needed for embedding.  A better parser would accept
    49	    options for what data you wwanted, and preserve the
    50	    order."""
    51
    52	    lines = open_and_readlines(afmFileName, 'r')
    53	    if len(lines)<=1:
    54	        #likely to be a MAC file
    55	        if lines: lines = lines[0].split('\r')
    56	        if len(lines)<=1:
    57	            raise ValueError('AFM file %s hasn\'t enough data' % afmFileName)
    58	    topLevel = {}
    59	    glyphLevel = []
    60
    61	    lines = [l.strip() for l in lines]
    62	    lines = [l for l in lines if not l.lower().startswith('comment')]
    63	    #pass 1 - get the widths
    64	    inMetrics = 0  # os 'TOP', or 'CHARMETRICS'
    65	    for line in lines:
    66	        if line[0:16] == 'StartCharMetrics':
    67	            inMetrics = 1
    68	        elif line[0:14] == 'EndCharMetrics':
    69	            inMetrics = 0
    70	        elif inMetrics:
    71	            chunks = line.split(';')
    72	            chunks = [chunk.strip() for chunk in chunks]
    73	            cidChunk, widthChunk, nameChunk = chunks[0:3]
    74
    75	            # character ID
    76	            l, r = cidChunk.split()
    77	            assert l == 'C', 'bad line in font file %s' % line
    78	            cid = int(r)
    79
    80	            # width
    81	            l, r = widthChunk.split()
    82	            assert l == 'WX', 'bad line in font file %s' % line
    83	            try:
    84	                width = int(r)
    85	            except ValueError:
    86	                width = float(r)
    87
    88	            # name
    89	            l, r = nameChunk.split()
    90	            assert l == 'N', 'bad line in font file %s' % line
    91	            name = r
    92
    93	            glyphLevel.append((cid, width, name))
    94
    95	    # pass 2 font info
    96	    inHeader = 0
    97	    for line in lines:
    98	        if line[0:16] == 'StartFontMetrics':
    99	            inHeader = 1
   100	        if line[0:16] == 'StartCharMetrics':
   101	            inHeader = 0
   102	        elif inHeader:
   103	            if line[0:7] == 'Comment': pass
   104	            try:
   105	                left, right = line.split(' ',1)
   106	            except:
   107	                raise ValueError("Header information error in afm %s: line='%s'" % (afmFileName, line))
   108	            try:
   109	                right = int(right)
   110	            except:
   111	                pass
   112	            topLevel[left] = right
   113
   114
   115	    return (topLevel, glyphLevel)
