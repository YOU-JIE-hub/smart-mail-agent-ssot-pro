     1	"""Always defined attribute analysis.
     2
     3	An always defined attribute has some statements in __init__ or the
     4	class body that cause the attribute to be always initialized when an
     5	instance is constructed. It must also not be possible to read the
     6	attribute before initialization, and it can't be deletable.
     7
     8	We can assume that the value is always defined when reading an always
     9	defined attribute. Otherwise we'll need to raise AttributeError if the
    10	value is undefined (i.e. has the error value).
    11
    12	We use data flow analysis to figure out attributes that are always
    13	defined. Example:
    14
    15	  class C:
    16	      def __init__(self) -> None:
    17	          self.x = 0
    18	          if func():
    19	              self.y = 1
    20	          else:
    21	              self.y = 2
    22	              self.z = 3
    23
    24	In this example, the attributes 'x' and 'y' are always defined, but 'z'
    25	is not. The analysis assumes that we know that there won't be any subclasses.
    26
    27	The analysis also works if there is a known, closed set of subclasses.
    28	An attribute defined in a base class can only be always defined if it's
    29	also always defined in all subclasses.
    30
    31	As soon as __init__ contains an op that can 'leak' self to another
    32	function, we will stop inferring always defined attributes, since the
    33	analysis is mostly intra-procedural and only looks at __init__ methods.
    34	The called code could read an uninitialized attribute. Example:
    35
    36	  class C:
    37	      def __init__(self) -> None:
    38	          self.x = self.foo()
    39
    40	      def foo(self) -> int:
    41	          ...
    42
    43	Now we won't infer 'x' as always defined, since 'foo' might read 'x'
    44	before initialization.
    45
    46	As an exception to the above limitation, we perform inter-procedural
    47	analysis of super().__init__ calls, since these are very common.
    48
    49	Our analysis is somewhat optimistic. We assume that nobody calls a
    50	method of a partially uninitialized object through gc.get_objects(), in
    51	particular. Code like this could potentially cause a segfault with a null
    52	pointer dereference. This seems very unlikely to be an issue in practice,
    53	however.

    56	and thus works if the object is partially uninitialized. This can be used
    57	as a workaround if somebody ever needs to inspect partially uninitialized

   191	def find_always_defined_attributes(
   192	    blocks: list[BasicBlock],
   193	    self_reg: Register,
   194	    all_attrs: set[str],
   195	    maybe_defined: AnalysisResult[str],
   196	    maybe_undefined: AnalysisResult[str],
   197	    dirty: AnalysisResult[None],
   198	) -> set[str]:
   199	    """Find attributes that are always initialized in some basic blocks.
   200
   201	    The analysis results are expected to be up-to-date for the blocks.
   202
   203	    Return a set of always defined attributes.
   204	    """
   205	    attrs = all_attrs.copy()
   206	    for block in blocks:
   207	        for i, op in enumerate(block.ops):
   208	            # If an attribute we *read* may be undefined, it isn't always defined.
   209	            if isinstance(op, GetAttr) and op.obj is self_reg:
   210	                if op.attr in maybe_undefined.before[block, i]:
   211	                    attrs.discard(op.attr)
   212	            # If an attribute we *set* may be sometimes undefined and
   213	            # sometimes defined, don't consider it always defined. Unlike
   214	            # the get case, it's fine for the attribute to be undefined.
   215	            # The set operation will then be treated as initialization.
   216	            if isinstance(op, SetAttr) and op.obj is self_reg:
   217	                if (
   218	                    op.attr in maybe_undefined.before[block, i]
   219	                    and op.attr in maybe_defined.before[block, i]
   220	                ):
   221	                    attrs.discard(op.attr)
   222	            # Treat an op that might run arbitrary code as an "exit"
   223	            # in terms of the analysis -- we can't do any inference
   224	            # afterwards reliably.
   225	            if dirty.after[block, i]:
   226	                if not dirty.before[block, i]:
   227	                    attrs = attrs & (
   228	                        maybe_defined.after[block, i] - maybe_undefined.after[block, i]
   229	                    )
   230	                break

   232	                for target in op.targets():
   233	                    # Gotos/branches can also be "exits".
   234	                    if not dirty.after[block, i] and dirty.before[target, 0]:

   236	                            maybe_defined.after[target, 0] - maybe_undefined.after[target, 0]

    82	def get_cfg(blocks: list[BasicBlock], *, use_yields: bool = False) -> CFG:
    83	    """Calculate basic block control-flow graph.
    84
    85	    If use_yields is set, then we treat returns inserted by yields as gotos
    86	    instead of exits.
    87	    """
    88	    succ_map = {}
    89	    pred_map: dict[BasicBlock, list[BasicBlock]] = {}
    90	    exits = set()
    91	    for block in blocks:
    92	        assert not any(
    93	            isinstance(op, ControlOp) for op in block.ops[:-1]
    94	        ), "Control-flow ops must be at the end of blocks"
    95
    96	        if use_yields and isinstance(block.terminator, Return) and block.terminator.yield_target:
    97	            succ = [block.terminator.yield_target]
    98	        else:
    99	            succ = list(block.terminator.targets())
   100	        if not succ:
   101	            exits.add(block)
   102
   103	        # Errors can occur anywhere inside a block, which means that
   104	        # we can't assume that the entire block has executed before
   105	        # jumping to the error handler. In our CFG construction, we
   106	        # model this as saying that a block can jump to its error
   107	        # handler or the error handlers of any of its normal
   108	        # successors (to represent an error before that next block
   109	        # completes). This works well for analyses like "must
   110	        # defined", where it implies that registers assigned in a
   111	        # block may be undefined in its error handler, but is in
   112	        # general not a precise representation of reality; any
   113	        # analyses that require more fidelity must wait until after
   114	        # exception insertion.
   115	        for error_point in [block] + succ:
   116	            if error_point.error_handler:
   117	                succ.append(error_point.error_handler)
   118
   119	        succ_map[block] = succ
   120	        pred_map[block] = []
   121	    for prev, nxt in succ_map.items():
   122	        for label in nxt:
   123	            pred_map[label].append(prev)
   124	    return CFG(succ_map, pred_map, exits)

   134	    """Cleanup the control flow graph.

   137	    nothing but a single jump.

   139	    There is a lot more that could be done.

   509	def run_analysis(
   510	    blocks: list[BasicBlock],
   511	    cfg: CFG,
   512	    gen_and_kill: OpVisitor[GenAndKill[T]],
   513	    initial: set[T],
   514	    kind: int,
   515	    backward: bool,
   516	    universe: set[T] | None = None,
   517	) -> AnalysisResult[T]:
   518	    """Run a general set-based data flow analysis.
   519
   520	    Args:
   521	        blocks: All basic blocks
   522	        cfg: Control-flow graph for the code
   523	        gen_and_kill: Implementation of gen and kill functions for each op
   524	        initial: Value of analysis for the entry points (for a forward analysis) or the
   525	            exit points (for a backward analysis)
   526	        kind: MUST_ANALYSIS or MAYBE_ANALYSIS
   527	        backward: If False, the analysis is a forward analysis; it's backward otherwise
   528	        universe: For a must analysis, the set of all possible values. This is the starting
   529	            value for the work list algorithm, which will narrow this down until reaching a
   530	            fixed point. For a maybe analysis the iteration always starts from an empty set
   531	            and this argument is ignored.
   532
   533	    Return analysis results: (before, after)
   534	    """
   535	    block_gen = {}
   536	    block_kill = {}
   537
   538	    # Calculate kill and gen sets for entire basic blocks.
   539	    for block in blocks:
   540	        gen: set[T] = set()
   541	        kill: set[T] = set()
   542	        ops = block.ops
   543	        if backward:
   544	            ops = list(reversed(ops))
   545	        for op in ops:
   546	            opgen, opkill = op.accept(gen_and_kill)
   547	            gen = (gen - opkill) | opgen
   548	            kill = (kill - opgen) | opkill
   549	        block_gen[block] = gen
   550	        block_kill[block] = kill
   551
   552	    # Set up initial state for worklist algorithm.
   553	    worklist = list(blocks)
   554	    if not backward:
   555	        worklist.reverse()  # Reverse for a small performance improvement
   556	    workset = set(worklist)
   557	    before: dict[BasicBlock, set[T]] = {}
   558	    after: dict[BasicBlock, set[T]] = {}
   559	    for block in blocks:
   560	        if kind == MAYBE_ANALYSIS:
   561	            before[block] = set()
   562	            after[block] = set()
   563	        else:
   564	            assert universe is not None, "Universe must be defined for a must analysis"
   565	            before[block] = set(universe)
   566	            after[block] = set(universe)
   567
   568	    if backward:
   569	        pred_map = cfg.succ
   570	        succ_map = cfg.pred
   571	    else:
   572	        pred_map = cfg.pred
   573	        succ_map = cfg.succ
   574
   575	    # Run work list algorithm to generate in and out sets for each basic block.
   576	    while worklist:
   577	        label = worklist.pop()
   578	        workset.remove(label)
   579	        if pred_map[label]:
   580	            new_before: set[T] | None = None
   581	            for pred in pred_map[label]:
   582	                if new_before is None:
   583	                    new_before = set(after[pred])
   584	                elif kind == MAYBE_ANALYSIS:
   585	                    new_before |= after[pred]
   586	                else:
   587	                    new_before &= after[pred]
   588	            assert new_before is not None
   589	        else:
   590	            new_before = set(initial)
   591	        before[label] = new_before
   592	        new_after = (new_before - block_kill[label]) | block_gen[label]
   593	        if new_after != after[label]:
   594	            for succ in succ_map[label]:
   595	                if succ not in workset:
   596	                    worklist.append(succ)
   597	                    workset.add(succ)
   598	        after[label] = new_after
   599
   600	    # Run algorithm for each basic block to generate opcode-level sets.
   601	    op_before: dict[tuple[BasicBlock, int], set[T]] = {}
   602	    op_after: dict[tuple[BasicBlock, int], set[T]] = {}
   603	    for block in blocks:
   604	        label = block
   605	        cur = before[label]
   606	        ops_enum: Iterator[tuple[int, Op]] = enumerate(block.ops)
   607	        if backward:
   608	            ops_enum = reversed(list(ops_enum))
   609	        for idx, op in ops_enum:
   610	            op_before[label, idx] = cur
   611	            opgen, opkill = op.accept(gen_and_kill)

    90	    errors = []

   132	def check_op_sources_valid(fn: FuncIR) -> list[FnError]:
   133	    errors = []
   134	    valid_ops: set[Op] = set()
   135	    valid_registers: set[Register] = set()
   136
   137	    for block in fn.blocks:
   138	        valid_ops.update(block.ops)
   139
   140	        for op in block.ops:
   141	            if isinstance(op, BaseAssign):
   142	                valid_registers.add(op.dest)
   143	            elif isinstance(op, LoadAddress) and isinstance(op.src, Register):
   144	                valid_registers.add(op.src)

   148	    for block in fn.blocks:

   152	                    pass

   184	def can_coerce_to(src: RType, dest: RType) -> bool:
   185	    """Check if src can be assigned to dest_rtype.
   186
   187	    Currently okay to have false positives.
   188	    """
   189	    if isinstance(dest, RUnion):
   190	        return any(can_coerce_to(src, d) for d in dest.items)
   191
   192	    if isinstance(dest, RPrimitive):
   193	        if isinstance(src, RPrimitive):
   194	            # If either src or dest is a disjoint type, then they must both be.
   195	            if src.name in disjoint_types and dest.name in disjoint_types:
   196	                return src.name == dest.name
   197	            return src.size == dest.size
   198	        if isinstance(src, RInstance):
   199	            return is_object_rprimitive(dest)
   200	        if isinstance(src, RUnion):
   201	            # IR doesn't have the ability to narrow unions based on
   202	            # control flow, so cannot be a strict all() here.
   203	            return any(can_coerce_to(s, dest) for s in src.items)
   204	        return False
   205
   206	    return True
