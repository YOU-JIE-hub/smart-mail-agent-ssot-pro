   827	    def assert_emit(
   828	        self,
   829	        op: Op | list[Op],
   830	        expected: str,
   831	        next_block: BasicBlock | None = None,
   832	        *,
   833	        rare: bool = False,
   834	        next_branch: Branch | None = None,
   835	        skip_next: bool = False,
   836	    ) -> None:
   837	        block = BasicBlock(0)
   838	        if isinstance(op, Op):
   839	            block.ops.append(op)
   840	        else:
   841	            block.ops.extend(op)
   842	            op = op[-1]
   843	        value_names = generate_names_for_ir(self.registers, [block])
   844	        emitter = Emitter(self.context, value_names)
   845	        declarations = Emitter(self.context, value_names)
   846	        emitter.fragments = []
   847	        declarations.fragments = []
   848
   849	        visitor = FunctionEmitterVisitor(emitter, declarations, "prog.py", "prog")
   850	        visitor.next_block = next_block
   851	        visitor.rare = rare
   852	        if next_branch:
   853	            visitor.ops = [op, next_branch]
   854	        else:
   855	            visitor.ops = [op]
   856	        visitor.op_index = 0
   857
   858	        op.accept(visitor)
   859	        frags = declarations.fragments + emitter.fragments
   860	        actual_lines = [line.strip(" ") for line in frags]
   861	        assert all(line.endswith("\n") for line in actual_lines)
   862	        actual_lines = [line.rstrip("\n") for line in actual_lines]
   863	        if not expected.strip():
   864	            expected_lines = []
   865	        else:
   866	            expected_lines = expected.rstrip().split("\n")
   867	        expected_lines = [line.strip(" ") for line in expected_lines]
   868	        assert_string_arrays_equal(
   869	            expected_lines, actual_lines, msg="Generated code unexpected", traceback=True
   870	        )
   871	        if skip_next:
   872	            assert visitor.op_index == 1
   873	        else:
   874	            assert visitor.op_index == 0

   876	    def assert_emit_binary_op(
   877	        self, op: str, dest: Value, left: Value, right: Value, expected: str
   878	    ) -> None:
   879	        if op in binary_ops:
   880	            ops = binary_ops[op]
   881	            for desc in ops:
   882	                if is_subtype(left.type, desc.arg_types[0]) and is_subtype(
   883	                    right.type, desc.arg_types[1]
   884	                ):
   885	                    args = [left, right]
   886	                    if desc.ordering is not None:
   887	                        args = [args[i] for i in desc.ordering]
   888	                    # This only supports primitives that map to C calls
   889	                    assert desc.c_function_name is not None
   890	                    self.assert_emit(
   891	                        CallC(
   892	                            desc.c_function_name,
   893	                            args,
   894	                            desc.return_type,
   895	                            desc.steals,
   896	                            desc.is_borrowed,
   897	                            desc.error_kind,
   898	                            55,
   899	                        ),
   900	                        expected,
   901	                    )
   902	                    return
   903	        else:
   904	            assert False, "Could not find matching op"

    95	    """Run a setup script in a somewhat controlled environment.
    96
    97	    This is adapted from code in distutils and our goal here is that is
    98	    faster to not need to spin up a python interpreter to run it.
    99
   100	    We had to fork it because the real run_setup swallows errors
   101	    and KeyboardInterrupt with no way to recover them (!).
   102	    The real version has some extra features that we removed since
   103	    we weren't using them.
   104
   105	    Returns whether the setup succeeded.
   106	    """
   107	    save_argv = sys.argv.copy()
   108	    g = {"__file__": script_name}
   109	    try:
   110	        try:
   111	            sys.argv[0] = script_name
   112	            sys.argv[1:] = script_args
   113	            with open(script_name, "rb") as f:
   114	                exec(f.read(), g)
   115	        finally:
   116	            sys.argv = save_argv
   117	    except SystemExit as e:
   118	        # distutils converts KeyboardInterrupt into a SystemExit with
   119	        # "interrupted" as the argument. Convert it back so that
   120	        # pytest will exit instead of just failing the test.
   121	        if e.code == "interrupted":
   122	            raise KeyboardInterrupt from e
   123
   124	        return e.code == 0 or e.code is None
   125
   126	    return True

   131	    dir = os.getcwd()
   132	    os.chdir(target)
   133	    try:
   134	        yield
   135	    finally:
   136	        os.chdir(dir)

   139	class TestRun(MypycDataSuite):
   140	    """Test cases that build a C extension and run code."""

   142	    files = files
   143	    base_path = test_temp_dir

   149	    def run_case(self, testcase: DataDrivenTestCase) -> None:
   150	        # setup.py wants to be run from the root directory of the package, which we accommodate
   151	        # by chdiring into tmp/
   152	        with (
   153	            use_custom_builtins(os.path.join(self.data_prefix, ICODE_GEN_BUILTINS), testcase),
   154	            chdir_manager("tmp"),
   155	        ):
   156	            self.run_case_inner(testcase)

   158	    def run_case_inner(self, testcase: DataDrivenTestCase) -> None:
   159	        if not os.path.isdir(WORKDIR):  # (one test puts something in build...)
   160	            os.mkdir(WORKDIR)
   161
   162	        text = "\n".join(testcase.input)
   163
   164	        with open("native.py", "w", encoding="utf-8") as f:
   165	            f.write(text)
   166	        with open("interpreted.py", "w", encoding="utf-8") as f:
   167	            f.write(text)
   168
   169	        shutil.copyfile(TESTUTIL_PATH, "testutil.py")
   170
   171	        step = 1
   172	        self.run_case_step(testcase, step)
   173
   174	        steps = testcase.find_steps()
   175	        if steps == [[]]:
   176	            steps = []
   177
   178	        for operations in steps:
   179	            # To make sure that any new changes get picked up as being
   180	            # new by distutils, shift the mtime of all of the
   181	            # generated artifacts back by a second.
   182	            fudge_dir_mtimes(WORKDIR, -1)
   183	            # On some OS, changing the mtime doesn't work reliably. As
   184	            # a workaround, sleep.
   185	            # TODO: Figure out a better approach, since this slows down tests.
   186	            time.sleep(1.0)
   187
   188	            step += 1
   189	            with chdir_manager(".."):
   190	                perform_file_operations(operations)
   191	            self.run_case_step(testcase, step)

   193	    def run_case_step(self, testcase: DataDrivenTestCase, incremental_step: int) -> None:
   194	        bench = testcase.config.getoption("--bench", False) and "Benchmark" in testcase.name
   195
   196	        options = Options()
   197	        options.use_builtins_fixtures = True
   198	        options.show_traceback = True
   199	        options.strict_optional = True
   200	        options.python_version = sys.version_info[:2]
   201	        options.export_types = True
   202	        options.preserve_asts = True
   203	        options.allow_empty_bodies = True
   204	        options.incremental = self.separate
   205
   206	        # Avoid checking modules/packages named 'unchecked', to provide a way
   207	        # to test interacting with code we don't have types for.
   208	        options.per_module_options["unchecked.*"] = {"follow_imports": "error"}
   209
   210	        source = build.BuildSource("native.py", "native", None)
   211	        sources = [source]
   212	        module_names = ["native"]
   213	        module_paths = ["native.py"]
   214
   215	        # Hard code another module name to compile in the same compilation unit.
   216	        to_delete = []
   217	        for fn, text in testcase.files:
   218	            fn = os.path.relpath(fn, test_temp_dir)
   219
   220	            if os.path.basename(fn).startswith("other") and fn.endswith(".py"):
   221	                name = fn.split(".")[0].replace(os.sep, ".")
   222	                module_names.append(name)
   223	                sources.append(build.BuildSource(fn, name, None))
   224	                to_delete.append(fn)
   225	                module_paths.append(fn)
   226
   227	                shutil.copyfile(fn, os.path.join(os.path.dirname(fn), name + "_interpreted.py"))
   228
   229	        for source in sources:
   230	            options.per_module_options.setdefault(source.module, {})["mypyc"] = True
   231
   232	        separate = (
   233	            self.get_separate("\n".join(testcase.input), incremental_step)
   234	            if self.separate
   235	            else False
   236	        )
   237
   238	        groups = construct_groups(sources, separate, len(module_names) > 1, None)
   239
   240	        try:
   241	            compiler_options = CompilerOptions(
   242	                multi_file=self.multi_file,
   243	                separate=self.separate,
   244	                strict_dunder_typing=self.strict_dunder_typing,
   245	            )
   246	            result = emitmodule.parse_and_typecheck(
   247	                sources=sources,
   248	                options=options,
   249	                compiler_options=compiler_options,
   250	                groups=groups,
   251	                alt_lib_path=".",
   252	            )
   253	            errors = Errors(options)
   254	            ir, cfiles, _ = emitmodule.compile_modules_to_c(
   255	                result, compiler_options=compiler_options, errors=errors, groups=groups
   256	            )
   257	            if errors.num_errors:
   258	                errors.flush_errors()
   259	                assert False, "Compile error"
   260	        except CompileError as e:
   261	            for line in e.messages:
   262	                print(fix_native_line_number(line, testcase.file, testcase.line))
   263	            assert False, "Compile error"
   264
   265	        # Check that serialization works on this IR. (Only on the first
   266	        # step because the returned ir only includes updated code.)
   267	        if incremental_step == 1:
   268	            check_serialization_roundtrip(ir)
   269
   270	        opt_level = int(os.environ.get("MYPYC_OPT_LEVEL", 0))
   271	        debug_level = int(os.environ.get("MYPYC_DEBUG_LEVEL", 0))
   272
   273	        setup_file = os.path.abspath(os.path.join(WORKDIR, "setup.py"))
   274	        # We pass the C file information to the build script via setup.py unfortunately
   275	        with open(setup_file, "w", encoding="utf-8") as f:
   276	            f.write(
   277	                setup_format.format(
   278	                    module_paths, separate, cfiles, self.multi_file, opt_level, debug_level
   279	                )
   280	            )
   281
   282	        if not run_setup(setup_file, ["build_ext", "--inplace"]):
   283	            if testcase.config.getoption("--mypyc-showc"):
   284	                show_c(cfiles)
   285	            copy_output_files(mypyc_output_dir)
   286	            assert False, "Compilation failed"
   287
   288	        # Assert that an output file got created
   289	        suffix = "pyd" if sys.platform == "win32" else "so"
   290	        assert glob.glob(f"native.*.{suffix}") or glob.glob(f"native.{suffix}")
   291
   292	        driver_path = "driver.py"
   293	        if not os.path.isfile(driver_path):
   294	            # No driver.py provided by test case. Use the default one
   295	            # (mypyc/test-data/driver/driver.py) that calls each
   296	            # function named test_*.
   297	            default_driver = os.path.join(test_data_prefix, "driver", "driver.py")
   298	            shutil.copy(default_driver, driver_path)
   299	        env = os.environ.copy()
   300	        env["MYPYC_RUN_BENCH"] = "1" if bench else "0"
   301
   302	        debugger = testcase.config.getoption("debugger")
   303	        if debugger:
   304	            if debugger == "lldb":
   305	                subprocess.check_call(["lldb", "--", sys.executable, driver_path], env=env)
   306	            elif debugger == "gdb":
   307	                subprocess.check_call(["gdb", "--args", sys.executable, driver_path], env=env)
   308	            else:
   309	                assert False, "Unsupported debugger"
   310	            # TODO: find a way to automatically disable capturing
   311	            # stdin/stdout when in debugging mode
   312	            assert False, (
   313	                "Test can't pass in debugging mode. "
   314	                "(Make sure to pass -s to pytest to interact with the debugger)"
   315	            )
   316	        proc = subprocess.Popen(
   317	            [sys.executable, driver_path],
   318	            stdout=subprocess.PIPE,
   319	            stderr=subprocess.STDOUT,
   320	            env=env,
   321	        )
   322	        if sys.version_info >= (3, 12):
   323	            # TODO: testDecorators1 hangs on 3.12, remove this once fixed
   324	            proc.wait(timeout=30)
   325	        output = proc.communicate()[0].decode("utf8")
   326	        output = output.replace(f'  File "{os.getcwd()}{os.sep}', '  File "')
   327	        outlines = output.splitlines()
   328
   329	        if testcase.config.getoption("--mypyc-showc"):
   330	            show_c(cfiles)
   331	        if proc.returncode != 0:
   332	            print()
   333	            signal = proc.returncode == -11
   334	            extra = ""
   335	            if signal:
   336	                extra = " (likely segmentation fault)"
   337	            print(f"*** Exit status: {proc.returncode}{extra}")
   338	            if signal and not sys.platform.startswith("win"):
   339	                print()
   340	                if sys.platform == "darwin":
   341	                    debugger = "lldb"
   342	                else:
   343	                    debugger = "gdb"
   344	                print(
   345	                    f'hint: Use "pytest -n0 -s --mypyc-debug={debugger} -k <name-substring>" to run test in debugger'
   346	                )
   347	                print("hint: You may need to build a debug version of Python first and use it")
   348	                print('hint: See also "Debugging Segfaults" in mypyc/doc/dev-intro.md')
   349	            copy_output_files(mypyc_output_dir)
   350
   351	        # Verify output.
   352	        if bench:
   353	            print("Test output:")
   354	            print(output)
   355	        else:
   356	            if incremental_step == 1:
   357	                msg = "Invalid output"
   358	                expected = testcase.output
   359	            else:
   360	                msg = f"Invalid output (step {incremental_step})"
   361	                expected = testcase.output2.get(incremental_step, [])
   362
   363	            if not expected:
   364	                # Tweak some line numbers, but only if the expected output is empty,
   365	                # as tweaked output might not match expected output.
   366	                outlines = [
   367	                    fix_native_line_number(line, testcase.file, testcase.line) for line in outlines
   368	                ]
   369	            assert_test_output(testcase, outlines, msg, expected)
   370
   371	        if incremental_step > 1 and options.incremental:
   372	            suffix = "" if incremental_step == 2 else str(incremental_step - 1)
   373	            expected_rechecked = testcase.expected_rechecked_modules.get(incremental_step - 1)
   374	            if expected_rechecked is not None:
   375	                assert_module_equivalence(
   376	                    "rechecked" + suffix, expected_rechecked, result.manager.rechecked_modules
   377	                )
   378	            expected_stale = testcase.expected_stale_modules.get(incremental_step - 1)
   379	            if expected_stale is not None:
   380	                assert_module_equivalence(
   381	                    "stale" + suffix, expected_stale, result.manager.stale_modules
   382	                )
   383
   384	        assert proc.returncode == 0

   386	    def get_separate(self, program_text: str, incremental_step: int) -> Any:
   387	        template = r"# separate{}: (\[.*\])$"
   388	        m = re.search(template.format(incremental_step), program_text, flags=re.MULTILINE)
   389	        if not m:
   390	            m = re.search(template.format(""), program_text, flags=re.MULTILINE)
   391	        if m:
   392	            return ast.literal_eval(m.group(1))
   393	        else:
   394	            return True
