   249	    def _calculateMaxBoundaries(self, colorNamePairs):
   250	        "Calculate the maximum width of some given strings."
   251	        fontName = self.fontName
   252	        fontSize = self.fontSize
   253	        subCols = self.subCols
   254
   255	        M = [_getWidths(i, m, fontName, fontSize, subCols) for i,m in enumerate(self._getTexts(colorNamePairs))]
   256	        if not M:
   257	            return [0,0]
   258	        n = max([len(m) for m in M])
   259	        if self.variColumn:
   260	            columnMaximum = self.columnMaximum
   261	            return [_transMax(n,M[r:r+columnMaximum]) for r in range(0,len(M),self.columnMaximum)]
   262	        else:
   263	            return _transMax(n,M)

   265	    def _calcHeight(self):
   266	        dy = self.dy
   267	        yGap = self.yGap
   268	        thisy = upperlefty = self.y - dy
   269	        fontSize = self.fontSize
   270	        fontName = self.fontName
   271	        ascent=getFont(fontName).face.ascent/1000.
   272	        if ascent==0: ascent=0.718 # default (from helvetica)
   273	        ascent *= fontSize
   274	        leading = fontSize*1.2
   275	        deltay = self.deltay
   276	        if not deltay: deltay = max(dy,leading)+self.autoYPadding
   277	        columnCount = 0
   278	        count = 0
   279	        lowy = upperlefty
   280	        lim = self.columnMaximum - 1
   281	        for name in self._getTexts(self.colorNamePairs):
   282	            y0 = thisy+(dy-ascent)*0.5
   283	            y = y0 - _getLineCount(name)*leading
   284	            leadingMove = 2*y0-y-thisy
   285	            newy = thisy-max(deltay,leadingMove)-yGap
   286	            lowy = min(y,newy,lowy)
   287	            if count==lim:
   288	                count = 0
   289	                thisy = upperlefty
   290	                columnCount += 1
   291	            else:
   292	                thisy = newy
   293	                count = count+1
   294	        return upperlefty - lowy

   303	    def draw(self):
   304	        colorNamePairs = self.colorNamePairs
   305	        autoCP = isAuto(colorNamePairs)
   306	        if autoCP:
   307	            chart = getattr(colorNamePairs,'chart',getattr(colorNamePairs,'obj',None))
   308	            swatchMarker = None
   309	            autoCP = Auto(obj=chart)
   310	            n = chart._seriesCount
   311	            chartTexts = self._getTexts(colorNamePairs)
   312	        else:
   313	            swatchMarker = getattr(self,'swatchMarker',None)
   314	            if isAuto(swatchMarker):
   315	                chart = getattr(swatchMarker,'chart',getattr(swatchMarker,'obj',None))
   316	                swatchMarker = Auto(obj=chart)
   317	            n = len(colorNamePairs)
   318	        dx = self.dx
   319	        dy = self.dy
   320	        alignment = self.alignment
   321	        columnMaximum = self.columnMaximum
   322	        deltax = self.deltax
   323	        deltay = self.deltay
   324	        dxTextSpace = self.dxTextSpace
   325	        fontName = self.fontName
   326	        fontSize = self.fontSize
   327	        fillColor = self.fillColor
   328	        strokeWidth = self.strokeWidth
   329	        strokeColor = self.strokeColor
   330	        subCols = self.subCols
   331	        leading = fontSize*1.2
   332	        yGap = self.yGap
   333	        if not deltay:
   334	            deltay = max(dy,leading)+self.autoYPadding
   335	        ba = self.boxAnchor
   336	        maxWidth = self._calculateMaxBoundaries(colorNamePairs)
   337	        nCols = int((n+columnMaximum-1)/(columnMaximum*1.0))
   338	        xW = dx+dxTextSpace+self.autoXPadding
   339	        variColumn = self.variColumn
   340	        if variColumn:
   341	            width = sum([m[-1] for m in maxWidth])+xW*nCols
   342	        else:
   343	            deltax = max(maxWidth[-1]+xW,deltax)
   344	            width = nCols*deltax
   345	            maxWidth = nCols*[maxWidth]
   346
   347	        thisx = self.x
   348	        thisy = self.y - self.dy
   349	        if ba not in ('ne','n','nw','autoy'):
   350	            height = self._calcHeight()
   351	            if ba in ('e','c','w'):
   352	                thisy += height/2.
   353	            else:
   354	                thisy += height
   355	        if ba not in ('nw','w','sw','autox'):
   356	            if ba in ('n','c','s'):
   357	                thisx -= width/2
   358	            else:
   359	                thisx -= width
   360	        upperlefty = thisy
   361
   362	        g = Group()
   363
   364	        ascent=getFont(fontName).face.ascent/1000.
   365	        if ascent==0: ascent=0.718 # default (from helvetica)
   366	        ascent *= fontSize # normalize
   367
   368	        lim = columnMaximum - 1
   369	        callout = getattr(self,'callout',None)
   370	        scallout = getattr(self,'swatchCallout',None)
   371	        dividerLines = self.dividerLines
   372	        if dividerLines:
   373	            dividerWidth = self.dividerWidth
   374	            dividerColor = self.dividerColor
   375	            dividerDashArray = self.dividerDashArray
   376	            dividerOffsX = self.dividerOffsX
   377	            dividerOffsY = self.dividerOffsY
   378
   379	        for i in range(n):
   380	            if autoCP:
   381	                col = autoCP
   382	                col.index = i
   383	                name = chartTexts[i]
   384	            else:
   385	                col, name = colorNamePairs[i]
   386	                if isAuto(swatchMarker):
   387	                    col = swatchMarker
   388	                    col.index = i
   389	                if isAuto(name):
   390	                    name = getattr(swatchMarker,'chart',getattr(swatchMarker,'obj',None)).getSeriesName(i,'series %d' % i)
   391	            T = _getLines(name)
   392	            S = []
   393	            aS = S.append
   394	            j = int(i/(columnMaximum*1.0))
   395	            jOffs = maxWidth[j]
   396
   397	            # thisy+dy/2 = y+leading/2
   398	            y = y0 = thisy+(dy-ascent)*0.5
   399
   400	            if callout: callout(self,g,thisx,y,(col,name))
   401	            if alignment == "left":
   402	                x = thisx
   403	                xn = thisx+jOffs[-1]+dxTextSpace
   404	            elif alignment == "right":
   405	                x = thisx+dx+dxTextSpace
   406	                xn = thisx
   407	            else:
   408	                raise ValueError("bad alignment")
   409	            if not isSeq(name):
   410	                T = [T]
   411	            lineCount = _getLineCount(name)
   412	            yd = y
   413	            for k,lines in enumerate(T):
   414	                y = y0
   415	                kk = k*2
   416	                x1 = x+jOffs[kk]
   417	                x2 = x+jOffs[kk+1]
   418	                sc = subCols[k,i]
   419	                anchor = sc.align
   420	                scdx = sc.dx
   421	                scdy = sc.dy
   422	                fN = getattr(sc,'fontName',fontName)
   423	                fS = getattr(sc,'fontSize',fontSize)
   424	                fC = getattr(sc,'fillColor',fillColor)
   425	                fL = sc.leading or 1.2*fontSize
   426	                if fN==fontName:
   427	                    fA = (ascent*fS)/fontSize
   428	                else:
   429	                    fA = getFont(fontName).face.ascent/1000.
   430	                    if fA==0: fA=0.718
   431	                    fA *= fS
   432
   433	                vA = sc.vAlign
   434	                if vA=='top':
   435	                    vAdy = 0
   436	                else:
   437	                    vAdy = -fL * (lineCount - len(lines))
   438	                    if vA=='middle': vAdy *= 0.5
   439
   440	                if anchor=='left':
   441	                    anchor = 'start'
   442	                    xoffs = x1
   443	                elif anchor=='right':
   444	                    anchor = 'end'
   445	                    xoffs = x2
   446	                elif anchor=='numeric':
   447	                    xoffs = x2
   448	                else:
   449	                    anchor = 'middle'
   450	                    xoffs = 0.5*(x1+x2)
   451	                for t in lines:
   452	                    aS(String(xoffs+scdx,y+scdy+vAdy,t,fontName=fN,fontSize=fS,fillColor=fC, textAnchor = anchor))
   453	                    y -= fL
   454	                yd = min(yd,y)
   455	                y += fL
   456	                for iy, a in ((y-max(fL-fA,0),'underlines'),(y+fA,'overlines')):
   457	                    il = getattr(sc,a,None)
   458	                    if il:
   459	                        if not isinstance(il,(tuple,list)): il = (il,)
   460	                        for l in il:
   461	                            l = copy.copy(l)
   462	                            l.y1 += iy
   463	                            l.y2 += iy
   464	                            l.x1 += x1
   465	                            l.x2 += x2
   466	                            aS(l)
   467	            x = xn
   468	            y = yd
   469	            leadingMove = 2*y0-y-thisy
   470
   471	            if dividerLines:
   472	                xd = thisx+dx+dxTextSpace+jOffs[-1]+dividerOffsX[1]
   473	                yd = thisy+dy*0.5+dividerOffsY
   474	                if ((dividerLines&1) and i%columnMaximum) or ((dividerLines&2) and not i%columnMaximum):
   475	                    g.add(Line(thisx+dividerOffsX[0],yd,xd,yd,
   476	                        strokeColor=dividerColor, strokeWidth=dividerWidth, strokeDashArray=dividerDashArray))
   477
   478	                if (dividerLines&4) and (i%columnMaximum==lim or i==(n-1)):
   479	                    yd -= max(deltay,leadingMove)+yGap
   480	                    g.add(Line(thisx+dividerOffsX[0],yd,xd,yd,
   481	                        strokeColor=dividerColor, strokeWidth=dividerWidth, strokeDashArray=dividerDashArray))
   482
   483	            # Make a 'normal' color swatch...
   484	            swatchX = x + getattr(self,'swdx',0)
   485	            swatchY = thisy + getattr(self,'swdy',0)
   486
   487	            if isAuto(col):
   488	                chart = getattr(col,'chart',getattr(col,'obj',None))
   489	                c = chart.makeSwatchSample(getattr(col,'index',i),swatchX,swatchY,dx,dy)
   490	            elif isinstance(col, colors.Color):
   491	                if isSymbol(swatchMarker):
   492	                    c = uSymbol2Symbol(swatchMarker,swatchX+dx/2.,swatchY+dy/2.,col)
   493	                else:
   494	                    c = self._defaultSwatch(swatchX,swatchY,dx,dy,fillColor=col,strokeWidth=strokeWidth,strokeColor=strokeColor)
   495	            elif col is not None:
   496	                try:
   497	                    c = copy.deepcopy(col)
   498	                    c.x = swatchX
   499	                    c.y = swatchY
   500	                    c.width = dx
   501	                    c.height = dy
   502	                except:
   503	                    c = None
   504	            else:
   505	                c = None
   506
   507	            if c:
   508	                g.add(c)
   509	                if scallout: scallout(self,g,thisx,y0,i,(col,name),c)
   510
   511	            for s in S: g.add(s)
   512	            if self.colEndCallout and (i%columnMaximum==lim or i==(n-1)):
   513	                if alignment == "left":
   514	                    xt = thisx
   515	                else:
   516	                    xt = thisx+dx+dxTextSpace
   517	                yd = thisy+dy*0.5+dividerOffsY - (max(deltay,leadingMove)+yGap)
   518	                self.colEndCallout(self, g, thisx, xt, yd, jOffs[-1], jOffs[-1]+dx+dxTextSpace)
   519
   520	            if i%columnMaximum==lim:
   521	                if variColumn:
   522	                    thisx += jOffs[-1]+xW
   523	                else:
   524	                    thisx = thisx+deltax
   525	                thisy = upperlefty
   526	            else:
   527	                thisy = thisy-max(deltay,leadingMove)-yGap
   528
   529	        return g

   212
   213	        Sets an attribute _positions which is a list of

   220	        if self.useAbsolute:

   224	            # Dimensions are normalized to fit.
   225	            normWidth = self.groupSpacing

   227	            normFactor = availWidth / normWidth
   228	        self._normFactor = normFactor

   241	                if 0<=other<n:

   305	        g = Group()
   306
   307	        labelFmt = self.lineLabelFormat
   308	        P = self._positions
