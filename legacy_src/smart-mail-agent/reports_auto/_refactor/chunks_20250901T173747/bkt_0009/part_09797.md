   141	            # Init vectro for AES cipher (should be 16 bytes null array)
   142	            iv  = b'\x00' * 16

   144	            # Random User salts
   145	            uvs = os_urandom(8)
   146	            uks = os_urandom(8)

   148	            # the main encryption key
   149	            self.key = asBytes(os_urandom(32))

   151	            if DEBUG:
   152	                print("uvs      (hex)  = %s" % hexText(uvs))
   153	                print("uks      (hex)  = %s" % hexText(uks))
   154	                print("self.key (hex)  = %s" % hexText(self.key))

   167	            self.UE = encrypter.feed(self.key)
   168	            self.UE += encrypter.feed()

   171	                print("self.UE (hex)  = %s" % hexText(self.UE))

   175	            oks = os_urandom(8)

   187	            encrypter = pyaes.Encrypter(pyaes.AESModeOfOperationCBC(md.digest(), iv=iv))

   189	            self.OE += encrypter.feed()

   192	                print("self.OE (hex)  = %s" % hexText(self.OE))

   195	            permsarr = [
   196	                self.P       & 0xFF, # store the permission value in the first 32-bits
   197	                self.P >> 8  & 0xFF,
   198	                self.P >> 16 & 0xFF,
   199	                self.P >> 24 & 0xFF,
   200	                0xFF,
   201	                0xFF,
   202	                0xFF,
   203	                0xFF,
   204	                ord('T'),             # 'T' if EncryptMetaData is True (default), 'F' otherwise
   205	                ord('a'),             # a, d, b are magic values
   206	                ord('d'),
   207	                ord('b'),
   208	                0x01,                   # trailing zeros will be ignored
   209	                0x01,
   210	                0x01,
   211	                0x01
   212	            ]

   214	            # the permission array should be enrypted in the Perms field
   215	            encrypter = pyaes.Encrypter(pyaes.AESModeOfOperationCBC(self.key, iv=iv))
   216	            self.Perms = encrypter.feed(bytes(permsarr))
   217	            self.Perms += encrypter.feed()

   219	            if DEBUG:
   220	                print("self.Perms (hex)  = %s" % hexText(self.Perms))

   232	            self.U = computeU(self.key, revision=self.revision, documentId=internalID)
   233	            if DEBUG:
   234	                print("self.U (as hex) = %s" % hexText(self.U))

   237	        self.prepared = 1

   327	def encryptionkey(password, OwnerKey, Permissions, FileId1, revision=None):
   328	    revision = checkRevision(revision)
   329	    # FileId1 is first string of the fileid array
   330	    # add padding string
   331	    #AR force same as iText example
   332	    #Permissions =  -1836   #int(Permissions - 2**31)
   333	    password = asBytes(password) + PadString
   334	    # truncate to 32 bytes
   335	    password = password[:32]
   336	    # translate permissions to string, low order byte first
   337	    p = Permissions# + 2**32L
   338	    permissionsString = b""
   339	    for i in range(4):
   340	        byte = (p & 0xff)    # seems to match what iText does
   341	        p = p>>8
   342	        permissionsString += int2Byte(byte % 256)
   343
   344	    hash = md5(asBytes(password))
   345	    hash.update(asBytes(OwnerKey))
   346	    hash.update(asBytes(permissionsString))
   347	    hash.update(asBytes(FileId1))
   348
   349	    md5output = hash.digest()
   350
   351	    if revision==2:
   352	        key = md5output[:5]
   353	    elif revision==3:  #revision 3 algorithm - loop 50 times
   354	        for x in range(50):
   355	            md5output = md5(md5output).digest()
   356	        key = md5output[:16]
   357	    if DEBUG: print('encryptionkey(%s,%s,%s,%s,%s)==>%s' % tuple([hexText(str(x)) for x in (password, OwnerKey, Permissions, FileId1, revision, key)]))

   360	def computeO(userPassword, ownerPassword, revision):
   361	    from reportlab.lib.arciv import ArcIV
   362	    #print 'digest of hello is %s' % md5('hello').digest()
   363	    assert revision in (2,3), 'Unknown algorithm revision %s' % revision
   364	    if not ownerPassword:
   365	        ownerPassword = userPassword
   366
   367	    ownerPad = asBytes(ownerPassword) + PadString
   368	    ownerPad = ownerPad[0:32]
   369
   370	    password = asBytes(userPassword) + PadString
   371	    userPad = password[:32]
   372
   373	    digest = md5(ownerPad).digest()
   374	    if DEBUG: print('PadString=%s\nownerPad=%s\npassword=%s\nuserPad=%s\ndigest=%s\nrevision=%s' % (ascii(PadString),ascii(ownerPad),ascii(password),ascii(userPad),ascii(digest),revision))
   375	    if revision == 2:
   376	        O = ArcIV(digest[:5]).encode(userPad)
   377	    elif revision == 3:
   378	        for i in range(50):
   379	            digest = md5(digest).digest()
   380	        digest = digest[:16]
   381	        O = userPad
   382	        for i in range(20):
   383	            thisKey = xorKey(i, digest)
   384	            O = ArcIV(thisKey).encode(O)
   385	    if DEBUG: print('computeO(%s,%s,%s)==>%s' % tuple([hexText(str(x)) for x in (userPassword, ownerPassword, revision,O)]))
   386	    return O

   388	def computeU(encryptionkey, encodestring=PadString,revision=None,documentId=None):
   389	    revision = checkRevision(revision)
   390	    from reportlab.lib.arciv import ArcIV
   391	    if revision == 2:
   392	        result = ArcIV(encryptionkey).encode(encodestring)
   393	    elif revision == 3:
   394	        assert documentId is not None, "Revision 3 algorithm needs the document ID!"
   395	        h = md5(PadString)
   396	        h.update(rawBytes(documentId))
   397	        tmp = h.digest()
   398	        tmp = ArcIV(encryptionkey).encode(tmp)
   399	        for n in range(1,20):
   400	            thisKey = xorKey(n, encryptionkey)
   401	            tmp = ArcIV(thisKey).encode(tmp)
   402	        while len(tmp) < 32:
   403	            tmp += b'\0'
   404	        result = tmp
   405	    if DEBUG: print('computeU(%s,%s,%s,%s)==>%s' % tuple([hexText(str(x)) for x in (encryptionkey, encodestring,revision,documentId,result)]))
   406	    return result

   416	def encodePDF(key, objectNumber, generationNumber, string, revision=None):
   417	    "Encodes a string or stream"
   418	    revision = checkRevision(revision)
   419	    #print 'encodePDF (%s, %d, %d, %s)' % (hexText(key), objectNumber, generationNumber, string)
   420	    # extend 3 bytes of the object Number, low byte first
   421	    if revision in (2,3):
   422	        newkey = key
   423	        n = objectNumber
   424	        for i in range(3):
   425	            newkey += int2Byte(n & 0xff)
   426	            n = n>>8
   427	        # extend 2 bytes of the generationNumber
   428	        n = generationNumber
   429	        for i in range(2):
   430	            newkey += int2Byte(n & 0xff)
   431	            n = n>>8
   432	        md5output = md5(newkey).digest()
   433	        if revision == 2:
   434	            key = md5output[:10]
   435	        elif revision == 3:
   436	            key = md5output #all 16 bytes
   437	        from reportlab.lib.arciv import ArcIV
   438	        encrypted = ArcIV(key).encode(string)
   439	        #print 'encrypted=', hexText(encrypted)
   440	    elif revision == 5:
   441	        iv = os_urandom(16)
   442	        encrypter = pyaes.Encrypter(pyaes.AESModeOfOperationCBC(key, iv=iv))
   443
   444	        # pkcs7 style padding so that the size of the encrypted block is multiple of 16
   445	        string_len = len(string)
   446	        padding = ""
   447	        padding_len = (16 - (string_len % 16)) if string_len > 16 else (16 - string_len)
   448	        if padding_len > 0:
   449	            padding = chr(padding_len) * padding_len
   450
   451	        if isinstance(string, str):
   452	            string = (string + padding).encode("utf-8")
   453	        else:
   454	            string += asBytes(padding)
   455
   456	        encrypted = iv + encrypter.feed(string)
   457	        encrypted += encrypter.feed()
   458
   459	    if DEBUG: print('encodePDF(%s,%s,%s,%s,%s)==>%s' % tuple([hexText(str(x)) for x in (key, objectNumber, generationNumber, string, revision,encrypted)]))
   460	    return encrypted

    15	def _2xpre(s,styles):
    16	    "Helper to transform Pygments HTML output to ReportLab markup"
    17	    s = s.replace('<div class="highlight">','')
    18	    s = s.replace('</div>','')
    19	    s = s.replace('<pre>','')
    20	    s = s.replace('</pre>','')

   126			are sequences again. These 'child' sequences have to be protected

   135				- The element at index 2 is a sequence again with len 2
   136				  - The element at index 1 in this subsequence is a sequence again
   137					with len 2
   138
   139			With this spec '__rl_unpack_sequence__' does something like this for
   140			protection (len checks are omitted):
   141
   142				t = list(__rl_getiter__(g))
   143				t[1] = list(__rl_getiter__(t[1]))
   144				t[2] = list(__rl_getiter__(t[2]))
   145				t[2][1] = list(__rl_getiter__(t[2][1]))
   146				return t

   149				spec = {
   150					'min_len': 3,

   152						(1, {'min_len': 2, 'childs': ()}),
   153						(2, {
   154								'min_len': 2,
   155								'childs': (
   156									(1, {'min_len': 2, 'childs': ()})
   157								)
   158							}
   159						)
   160					)

   164				(a, (b, c), (d, (e, f))) = __rl_unpack_sequence__(g, spec)

   525		def visit_Assign(self, node):
   526			node = self.visit_children(node)
   527
   528			if not any(isinstance(t, ast.Tuple) for t in node.targets):
   529				return node
   530
   531			# Handle sequence unpacking.
   532			# For briefness this example omits cleanup of the temporary variables.
   533			# Check 'transform_tuple_assign' how its done.
   534			#
   535			# - Single target (with nested support)
   536			# (a, (b, (c, d))) = <exp>
   537			# is converted to
   538			# (a, t1) = __rl_getiter__(<exp>)
   539			# (b, t2) = __rl_getiter__(t1)
   540			# (c, d) = __rl_getiter__(t2)
   541			#
   542			# - Multi targets
   543			# (a, b) = (c, d) = <exp>
   544			# is converted to
   545			# (c, d) = __rl_getiter__(<exp>)
   546			# (a, b) = __rl_getiter__(<exp>)
   547			# Why is this valid ? The original bytecode for this multi targets
   548			# behaves the same way.
   549
   550			# ast.NodeTransformer works with list results.
   551			# He injects it at the rightplace of the node's parent statements.
   552			new_nodes = []
   553
   554			# python fills the right most target first.
   555			for target in reversed(node.targets):
   556				if isinstance(target, ast.Tuple):
   557					wrapper = ast.Assign(
   558						targets=[target],
   559						value=self.protect_unpack_sequence(target, node.value))
   560					new_nodes.append(wrapper)
   561				else:
   562					new_node = ast.Assign(targets=[target], value=node.value)
   563					new_nodes.append(new_node)
   564
   565			for new_node in new_nodes:
   566				copy_locations(new_node, node)
   567
   568			return new_nodes

   570		def visit_AugAssign(self, node):
   571			"""Forbid certain kinds of AugAssign
   572
   573			According to the language reference (and ast.c) the following nodes
   574			are are possible:
   575			Name, Attribute, Subscript
   576
   577			Note that although augmented assignment of attributes and
   578			subscripts is disallowed, augmented assignment of names (such
   579			as 'n += 1') is allowed.
   580			'n += 1' becomes 'n = __rl_augAssign__("+=", n, 1)'
   581			"""
   582
   583			node = self.visit_children(node)
   584
   585			if isinstance(node.target, ast.Attribute):
   586				self.error(node, "Augmented assignment of attributes is not allowed.")
   587
   588			elif isinstance(node.target, ast.Subscript):
   589				self.error(node, "Augmented assignment of object items and slices is not allowed.")
   590
   591			elif isinstance(node.target, ast.Name):
   592				new_node = ast.Assign(
   593					targets=[node.target],
   594					value=ast.Call(
   595						func=ast.Name('__rl_augAssign__', ast.Load()),
   596						args=[
   597							ast.Str(augOps[type(node.op)]),
   598							ast.Name(node.target.id, ast.Load()),
   599							node.value
   600							],
   601						keywords=[]))
   602
   603				copy_locations(new_node, node)
   604				return new_node
   605			else:  # pragma: no cover
   606				# Impossible Case - Only Node Types:
   607				# * Name
   608				# * Attribute
   609				# * Subscript
   610				# defined, those are checked before.
   611				raise NotImplementedError("Unknown target type: %s" % type(node.target))

   641			"""Protect tuple unpacking on with statements."""
   642			node = self.visit_children(node)
   643
   644			items = node.items
   645
   646			for item in reversed(items):
   647				if isinstance(item.optional_vars, ast.Tuple):
   648					tmp_target, unpack = self.gen_unpack_wrapper(
   649						node,
   650						item.optional_vars)
   651
   652					item.optional_vars = tmp_target
   653					node.body.insert(0, unpack)
   654
   655			return node
