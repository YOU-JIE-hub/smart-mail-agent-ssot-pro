    21	    dec: Decorator, analyzer: SemanticAnalyzerInterface
    22	) -> None:
    23	    """Try to infer the type of the decorated function.
    24
    25	    This lets us resolve additional references to decorated functions
    26	    during type checking. Otherwise the type might not be available
    27	    when we need it, since module top levels can't be deferred.
    28
    29	    This basically uses a simple special-purpose type inference
    30	    engine just for decorators.
    31	    """
    32	    if dec.var.is_property:
    33	        # Decorators are expected to have a callable type (it's a little odd).
    34	        if dec.func.type is None:
    35	            dec.var.type = CallableType(
    36	                [AnyType(TypeOfAny.special_form)],
    37	                [ARG_POS],
    38	                [None],
    39	                AnyType(TypeOfAny.special_form),
    40	                analyzer.named_type("builtins.function"),
    41	                name=dec.var.name,
    42	            )
    43	        elif isinstance(dec.func.type, CallableType):
    44	            dec.var.type = dec.func.type
    45	        return
    46	    decorator_preserves_type = True
    47	    for expr in dec.decorators:
    48	        preserve_type = False
    49	        if isinstance(expr, RefExpr) and isinstance(expr.node, FuncDef):
    50	            if expr.node.type and is_identity_signature(expr.node.type):
    51	                preserve_type = True
    52	        if not preserve_type:
    53	            decorator_preserves_type = False
    54	            break
    55	    if decorator_preserves_type:
    56	        # No non-identity decorators left. We can trivially infer the type
    57	        # of the function here.
    58	        dec.var.type = function_type(dec.func, analyzer.named_type("builtins.function"))
    59	    if dec.decorators:
    60	        return_type = calculate_return_type(dec.decorators[0])
    61	        if return_type and isinstance(return_type, AnyType):
    62	            # The outermost decorator will return Any so we know the type of the
    63	            # decorated function.
    64	            dec.var.type = AnyType(TypeOfAny.from_another_any, source_any=return_type)
    65	        sig = find_fixed_callable_return(dec.decorators[0])
    66	        if sig:
    67	            # The outermost decorator always returns the same kind of function,
    68	            # so we know that this is the type of the decorated function.
    69	            orig_sig = function_type(dec.func, analyzer.named_type("builtins.function"))
    70	            sig.name = orig_sig.items[0].name
    71	            dec.var.type = sig

   180	    # Process top levels until everything has been bound.

   182	    # Reverse order of the scc so the first modules in the original list will be

   223	                next_id, next_id, state, state.tree, None, final_iteration, patches
   224	            )

   231	        # Reverse to process the targets in the same order on every iteration. This avoids
   232	        # processing the same target twice in a row, which is inefficient.
   233	        worklist = list(reversed(all_deferred))

   237	def order_by_subclassing(targets: list[FullTargetInfo]) -> Iterator[FullTargetInfo]:
   238	    """Make sure that superclass methods are always processed before subclass methods.
   239
   240	    This algorithm is not very optimal, but it is simple and should work well for lists
   241	    that are already almost correctly ordered.
   242	    """
   243
   244	    # First, group the targets by their TypeInfo (since targets are sorted by line,
   245	    # we know that each TypeInfo will appear as group key only once).
   246	    grouped = [(k, list(g)) for k, g in groupby(targets, key=lambda x: x[3])]
   247	    remaining_infos = {info for info, _ in grouped if info is not None}
   248
   249	    next_group = 0
   250	    while grouped:
   251	        if next_group >= len(grouped):
   252	            # This should never happen, if there is an MRO cycle, it should be reported
   253	            # and fixed during top-level processing.
   254	            raise ValueError("Cannot order method targets by MRO")
   255	        next_info, group = grouped[next_group]
   256	        if next_info is None:
   257	            # Trivial case, not methods but functions, process them straight away.
   258	            yield from group
   259	            grouped.pop(next_group)
   260	            continue
   261	        if any(parent in remaining_infos for parent in next_info.mro[1:]):
   262	            # We cannot process this method group yet, try a next one.
   263	            next_group += 1
   264	            continue
   265	        yield from group
   266	        grouped.pop(next_group)
   267	        remaining_infos.discard(next_info)
   268	        # Each time after processing a method group we should retry from start,
   269	        # since there may be some groups that are not blocked on parents anymore.
   270	        next_group = 0

   369	    target: str,
   370	    module: str,
   371	    state: State,
   372	    node: MypyFile | FuncDef | OverloadedFuncDef | Decorator,
   373	    active_type: TypeInfo | None,

   375	    patches: Patches,

   382	    - were any new names defined (or placeholders replaced)

   384	    state.manager.processed_targets.append((module, target))
   385	    tree = state.tree
   386	    assert tree is not None
   387	    analyzer = state.manager.semantic_analyzer
   388	    # TODO: Move initialization to somewhere else
   389	    analyzer.global_decls = [set()]

   406	        )
   407	        if isinstance(node, Decorator):
   408	            infer_decorator_signature_if_simple(node, analyzer)
   409	    for dep in analyzer.imports:
   410	        state.add_dependency(dep)
   411	        priority = mypy.build.PRI_LOW
   412	        if priority <= state.priorities.get(dep, priority):
   413	            state.priorities[dep] = priority
   414
   415	    # Clear out some stale data to avoid memory leaks and astmerge
   416	    # validity check confusion
   417	    analyzer.statement = None
   418	    del analyzer.cur_mod_node
   419
   420	    if analyzer.deferred:
   421	        return [target], analyzer.incomplete, analyzer.progress
   422	    else:
   423	        return [], analyzer.incomplete, analyzer.progress

   175	                self.fail(NAMEDTUP_CLASS_ERROR, stmt)
   176	            elif len(stmt.lvalues) > 1 or not isinstance(stmt.lvalues[0], NameExpr):
   177	                # An assignment, but an invalid one.
   178	                statements.pop()
   179	                defn.removed_statements.append(stmt)
   180	                self.fail(NAMEDTUP_CLASS_ERROR, stmt)
   181	            else:
   182	                # Append name and type in this case...
   183	                name = stmt.lvalues[0].name
   184	                items.append(name)
   185	                if stmt.type is None:
   186	                    types.append(AnyType(TypeOfAny.unannotated))
   187	                else:
   188	                    # We never allow recursive types at function scope. Although it is
   189	                    # possible to support this for named tuples, it is still tricky, and
   190	                    # it would be inconsistent with type aliases.
   191	                    analyzed = self.api.anal_type(
   192	                        stmt.type,
   193	                        allow_placeholder=not self.api.is_func_scope(),
   194	                        prohibit_self_type="NamedTuple item type",
   195	                        prohibit_special_class_field_types="NamedTuple",
   196	                    )
   197	                    if analyzed is None:
   198	                        # Something is incomplete. We need to defer this named tuple.
   199	                        return None
   200	                    types.append(analyzed)
   201	                # ...despite possible minor failures that allow further analysis.
   202	                if name.startswith("_"):
   203	                    self.fail(
   204	                        f"NamedTuple field name cannot start with an underscore: {name}", stmt
   205	                    )
   206	                if stmt.type is None or hasattr(stmt, "new_syntax") and not stmt.new_syntax:
   207	                    self.fail(NAMEDTUP_CLASS_ERROR, stmt)
   208	                elif isinstance(stmt.rvalue, TempNode):
   209	                    # x: int assigns rvalue to TempNode(AnyType())
   210	                    if default_items:
   211	                        self.fail(
   212	                            "Non-default NamedTuple fields cannot follow default fields", stmt
   213	                        )
   214	                else:
   215	                    default_items[name] = stmt.rvalue
   216	        if defn.keywords:
   217	            for_function = ' for "__init_subclass__" of "NamedTuple"'
   218	            for key in defn.keywords:
   219	                self.msg.unexpected_keyword_argument_for_function(for_function, key, defn)
   220	        return items, types, default_items, statements

   222	    def check_namedtuple(
   223	        self, node: Expression, var_name: str | None, is_func_scope: bool
   224	    ) -> tuple[str | None, TypeInfo | None, list[TypeVarLikeType]]:
   225	        """Check if a call defines a namedtuple.
   226
   227	        The optional var_name argument is the name of the variable to
   228	        which this is assigned, if any.
   229
   230	        Return a tuple of two items:
   231	          * Internal name of the named tuple (e.g. the name passed as an argument to namedtuple)
   232	            or None if it is not a valid named tuple
   233	          * Corresponding TypeInfo, or None if not ready.
   234
   235	        If the definition is invalid but looks like a namedtuple,
   236	        report errors but return (some) TypeInfo.
   237	        """
   238	        if not isinstance(node, CallExpr):
   239	            return None, None, []
   240	        call = node
   241	        callee = call.callee
   242	        if not isinstance(callee, RefExpr):
   243	            return None, None, []
   244	        fullname = callee.fullname
   245	        if fullname == "collections.namedtuple":
   246	            is_typed = False
   247	        elif fullname in TYPED_NAMEDTUPLE_NAMES:
   248	            is_typed = True
   249	        else:
   250	            return None, None, []
   251	        result = self.parse_namedtuple_args(call, fullname)
   252	        if result:
   253	            items, types, defaults, typename, tvar_defs, ok = result
   254	        else:
   255	            # Error. Construct dummy return value.
   256	            if var_name:
   257	                name = var_name
   258	                if is_func_scope:
   259	                    name += "@" + str(call.line)
   260	            else:
   261	                name = var_name = "namedtuple@" + str(call.line)
   262	            info = self.build_namedtuple_typeinfo(name, [], [], {}, node.line, None)
   263	            self.store_namedtuple_info(info, var_name, call, is_typed)
   264	            if name != var_name or is_func_scope:
   265	                # NOTE: we skip local namespaces since they are not serialized.
   266	                self.api.add_symbol_skip_local(name, info)
   267	            return var_name, info, []
   268	        if not ok:
   269	            # This is a valid named tuple but some types are not ready.
   270	            return typename, None, []
   271
   272	        # We use the variable name as the class name if it exists. If
   273	        # it doesn't, we use the name passed as an argument. We prefer
   274	        # the variable name because it should be unique inside a
   275	        # module, and so we don't need to disambiguate it with a line
   276	        # number.
   277	        if var_name:
   278	            name = var_name
   279	        else:
   280	            name = typename
   281
   282	        if var_name is None or is_func_scope:
   283	            # There are two special cases where need to give it a unique name derived
   284	            # from the line number:
   285	            #   * This is a base class expression, since it often matches the class name:
   286	            #         class NT(NamedTuple('NT', [...])):
   287	            #             ...
   288	            #   * This is a local (function or method level) named tuple, since
   289	            #     two methods of a class can define a named tuple with the same name,
   290	            #     and they will be stored in the same namespace (see below).
   291	            name += "@" + str(call.line)
   292	        if defaults:
   293	            default_items = {
   294	                arg_name: default for arg_name, default in zip(items[-len(defaults) :], defaults)
   295	            }
   296	        else:
   297	            default_items = {}
   298
   299	        existing_info = None
   300	        if isinstance(node.analyzed, NamedTupleExpr):
   301	            existing_info = node.analyzed.info
   302	        info = self.build_namedtuple_typeinfo(
   303	            name, items, types, default_items, node.line, existing_info
   304	        )
   305
   306	        # If var_name is not None (i.e. this is not a base class expression), we always
   307	        # store the generated TypeInfo under var_name in the current scope, so that
   308	        # other definitions can use it.
   309	        if var_name:
   310	            self.store_namedtuple_info(info, var_name, call, is_typed)
   311	        else:
   312	            call.analyzed = NamedTupleExpr(info, is_typed=is_typed)
   313	            call.analyzed.set_line(call)
   314	        # There are three cases where we need to store the generated TypeInfo
   315	        # second time (for the purpose of serialization):
   316	        #   * If there is a name mismatch like One = NamedTuple('Other', [...])
   317	        #     we also store the info under name 'Other@lineno', this is needed
   318	        #     because classes are (de)serialized using their actual fullname, not
   319	        #     the name of l.h.s.
   320	        #   * If this is a method level named tuple. It can leak from the method
   321	        #     via assignment to self attribute and therefore needs to be serialized
   322	        #     (local namespaces are not serialized).
   323	        #   * If it is a base class expression. It was not stored above, since
   324	        #     there is no var_name (but it still needs to be serialized
   325	        #     since it is in MRO of some class).
   326	        if name != var_name or is_func_scope:
   327	            # NOTE: we skip local namespaces since they are not serialized.
   328	            self.api.add_symbol_skip_local(name, info)
   329	        return typename, info, tvar_defs
   330
   331	    def store_namedtuple_info(
   332	        self, info: TypeInfo, name: str, call: CallExpr, is_typed: bool
   333	    ) -> None:
   334	        self.api.add_symbol(name, info, call)
   335	        call.analyzed = NamedTupleExpr(info, is_typed=is_typed)
   336	        call.analyzed.set_line(call)

   338	    def parse_namedtuple_args(
   339	        self, call: CallExpr, fullname: str
   340	    ) -> None | (tuple[list[str], list[Type], list[Expression], str, list[TypeVarLikeType], bool]):
   341	        """Parse a namedtuple() call into data needed to construct a type.
   342
   343	        Returns a 6-tuple:
   344	        - List of argument names
   345	        - List of argument types
   346	        - List of default values
   347	        - First argument of namedtuple
   348	        - All typevars found in the field definition
   349	        - Whether all types are ready.
   350
   351	        Return None if the definition didn't typecheck.
   352	        """
   353	        type_name = "NamedTuple" if fullname in TYPED_NAMEDTUPLE_NAMES else "namedtuple"
   354	        # TODO: Share code with check_argument_count in checkexpr.py?
   355	        args = call.args
   356	        if len(args) < 2:
   357	            self.fail(f'Too few arguments for "{type_name}()"', call)
   358	            return None
   359	        defaults: list[Expression] = []
   360	        rename = False
   361	        if len(args) > 2:
   362	            # Typed namedtuple doesn't support additional arguments.
   363	            if fullname in TYPED_NAMEDTUPLE_NAMES:
   364	                self.fail('Too many arguments for "NamedTuple()"', call)
   365	                return None
   366	            for i, arg_name in enumerate(call.arg_names[2:], 2):
   367	                if arg_name == "defaults":
   368	                    arg = args[i]
   369	                    # We don't care what the values are, as long as the argument is an iterable
   370	                    # and we can count how many defaults there are.
   371	                    if isinstance(arg, (ListExpr, TupleExpr)):
   372	                        defaults = list(arg.items)
   373	                    else:
   374	                        self.fail(
   375	                            "List or tuple literal expected as the defaults argument to "
   376	                            "{}()".format(type_name),
   377	                            arg,
   378	                        )
   379	                elif arg_name == "rename":
   380	                    arg = args[i]
   381	                    if isinstance(arg, NameExpr) and arg.name in ("True", "False"):
   382	                        rename = arg.name == "True"
   383	                    else:
   384	                        self.fail(
   385	                            f'Boolean literal expected as the "rename" argument to {type_name}()',
   386	                            arg,
   387	                            code=ARG_TYPE,
   388	                        )
   389	        if call.arg_kinds[:2] != [ARG_POS, ARG_POS]:
   390	            self.fail(f'Unexpected arguments to "{type_name}()"', call)
   391	            return None
   392	        if not isinstance(args[0], StrExpr):
   393	            self.fail(f'"{type_name}()" expects a string literal as the first argument', call)
   394	            return None
   395	        typename = args[0].value
   396	        types: list[Type] = []
   397	        tvar_defs = []
   398	        if not isinstance(args[1], (ListExpr, TupleExpr)):
   399	            if fullname == "collections.namedtuple" and isinstance(args[1], StrExpr):
   400	                str_expr = args[1]
   401	                items = str_expr.value.replace(",", " ").split()
   402	            else:
   403	                self.fail(
   404	                    'List or tuple literal expected as the second argument to "{}()"'.format(
   405	                        type_name
   406	                    ),
   407	                    call,
   408	                )
   409	                return None
   410	        else:
   411	            listexpr = args[1]
   412	            if fullname == "collections.namedtuple":
   413	                # The fields argument contains just names, with implicit Any types.
   414	                if not is_StrExpr_list(listexpr.items):
   415	                    self.fail('String literal expected as "namedtuple()" item', call)
   416	                    return None
   417	                items = [item.value for item in listexpr.items]
   418	            else:
   419	                type_exprs = [
   420	                    t.items[1]
   421	                    for t in listexpr.items
   422	                    if isinstance(t, TupleExpr) and len(t.items) == 2
   423	                ]
   424	                tvar_defs = self.api.get_and_bind_all_tvars(type_exprs)
   425	                # The fields argument contains (name, type) tuples.
   426	                result = self.parse_namedtuple_fields_with_types(listexpr.items, call)
   427	                if result is None:
   428	                    # One of the types is not ready, defer.
   429	                    return None
   430	                items, types, _, ok = result
   431	                if not ok:
   432	                    return [], [], [], typename, [], False
   433	        if not types:
   434	            types = [AnyType(TypeOfAny.unannotated) for _ in items]
   435	        processed_items = []
   436	        seen_names: set[str] = set()
   437	        for i, item in enumerate(items):
   438	            problem = self.check_namedtuple_field_name(item, seen_names)
   439	            if problem is None:
   440	                processed_items.append(item)
   441	                seen_names.add(item)
   442	            else:
   443	                if not rename:
   444	                    self.fail(f'"{type_name}()" {problem}', call)
   445	                # Even if rename=False, we pretend that it is True.
   446	                # At runtime namedtuple creation would throw an error;
   447	                # applying the rename logic means we create a more sensible
   448	                # namedtuple.
   449	                new_name = f"_{i}"
   450	                processed_items.append(new_name)
   451	                seen_names.add(new_name)
   452
   453	        if len(defaults) > len(items):
   454	            self.fail(f'Too many defaults given in call to "{type_name}()"', call)
   455	            defaults = defaults[: len(items)]
   456	        return processed_items, types, defaults, typename, tvar_defs, True
   457
   458	    def parse_namedtuple_fields_with_types(
   459	        self, nodes: list[Expression], context: Context
   460	    ) -> tuple[list[str], list[Type], list[Expression], bool] | None:
   461	        """Parse typed named tuple fields.
   462
   463	        Return (names, types, defaults, whether types are all ready), or None if error occurred.
   464	        """
   465	        items: list[str] = []
   466	        types: list[Type] = []
   467	        for item in nodes:
   468	            if isinstance(item, TupleExpr):
   469	                if len(item.items) != 2:
   470	                    self.fail('Invalid "NamedTuple()" field definition', item)
   471	                    return None
   472	                name, type_node = item.items
   473	                if isinstance(name, StrExpr):
   474	                    items.append(name.value)
   475	                else:
   476	                    self.fail('Invalid "NamedTuple()" field name', item)
   477	                    return None
   478	                try:
   479	                    type = expr_to_unanalyzed_type(type_node, self.options, self.api.is_stub_file)
   480	                except TypeTranslationError:
   481	                    self.fail("Invalid field type", type_node)
   482	                    return None
   483	                # We never allow recursive types at function scope.
   484	                analyzed = self.api.anal_type(
   485	                    type,
   486	                    allow_placeholder=not self.api.is_func_scope(),
   487	                    prohibit_self_type="NamedTuple item type",
   488	                    prohibit_special_class_field_types="NamedTuple",
   489	                )
   490	                # Workaround #4987 and avoid introducing a bogus UnboundType
   491	                if isinstance(analyzed, UnboundType):
   492	                    analyzed = AnyType(TypeOfAny.from_error)
   493	                # These should be all known, otherwise we would defer in visit_assignment_stmt().
   494	                if analyzed is None:
   495	                    return [], [], [], False
   496	                types.append(analyzed)
   497	            else:
   498	                self.fail('Tuple expected as "NamedTuple()" field', item)
   499	                return None
   500	        return items, types, [], True

   502	    def build_namedtuple_typeinfo(
   503	        self,
   504	        name: str,
   505	        items: list[str],
   506	        types: list[Type],
   507	        default_items: Mapping[str, Expression],
   508	        line: int,
   509	        existing_info: TypeInfo | None,
   510	    ) -> TypeInfo:
   511	        strtype = self.api.named_type("builtins.str")
   512	        implicit_any = AnyType(TypeOfAny.special_form)
   513	        basetuple_type = self.api.named_type("builtins.tuple", [implicit_any])
   514	        dictype = self.api.named_type("builtins.dict", [strtype, implicit_any])
   515	        # Actual signature should return OrderedDict[str, Union[types]]
   516	        ordereddictype = self.api.named_type("builtins.dict", [strtype, implicit_any])
   517	        fallback = self.api.named_type("builtins.tuple", [implicit_any])
   518	        # Note: actual signature should accept an invariant version of Iterable[UnionType[types]].
   519	        # but it can't be expressed. 'new' and 'len' should be callable types.
   520	        iterable_type = self.api.named_type_or_none("typing.Iterable", [implicit_any])
   521	        function_type = self.api.named_type("builtins.function")
   522
   523	        literals: list[Type] = [LiteralType(item, strtype) for item in items]
   524	        match_args_type = TupleType(literals, basetuple_type)
   525
   526	        info = existing_info or self.api.basic_new_typeinfo(name, fallback, line)
   527	        info.is_named_tuple = True
   528	        tuple_base = TupleType(types, fallback)
   529	        if info.special_alias and has_placeholder(info.special_alias.target):
   530	            self.api.process_placeholder(
   531	                None, "NamedTuple item", info, force_progress=tuple_base != info.tuple_type
   532	            )
   533	        info.update_tuple_type(tuple_base)
   534	        info.line = line
   535	        # For use by mypyc.
   536	        info.metadata["namedtuple"] = {"fields": items.copy()}
   537
   538	        # We can't calculate the complete fallback type until after semantic
   539	        # analysis, since otherwise base classes might be incomplete. Postpone a
   540	        # callback function that patches the fallback.
   541	        if not has_placeholder(tuple_base) and not has_type_vars(tuple_base):
   542	            self.api.schedule_patch(
   543	                PRIORITY_FALLBACKS, lambda: calculate_tuple_fallback(tuple_base)
   544	            )
   545
   546	        def add_field(
   547	            var: Var, is_initialized_in_class: bool = False, is_property: bool = False
   548	        ) -> None:
   549	            var.info = info
   550	            var.is_initialized_in_class = is_initialized_in_class
   551	            var.is_property = is_property
   552	            var._fullname = f"{info.fullname}.{var.name}"
   553	            info.names[var.name] = SymbolTableNode(MDEF, var)
   554
   555	        fields = [Var(item, typ) for item, typ in zip(items, types)]
   556	        for var in fields:
   557	            add_field(var, is_property=True)
   558	        # We can't share Vars between fields and method arguments, since they
   559	        # have different full names (the latter are normally used as local variables
   560	        # in functions, so their full names are set to short names when generated methods
   561	        # are analyzed).
   562	        vars = [Var(item, typ) for item, typ in zip(items, types)]
   563
   564	        tuple_of_strings = TupleType([strtype for _ in items], basetuple_type)
   565	        add_field(Var("_fields", tuple_of_strings), is_initialized_in_class=True)
   566	        add_field(Var("_field_types", dictype), is_initialized_in_class=True)
   567	        add_field(Var("_field_defaults", dictype), is_initialized_in_class=True)
   568	        add_field(Var("_source", strtype), is_initialized_in_class=True)
   569	        add_field(Var("__annotations__", ordereddictype), is_initialized_in_class=True)
   570	        add_field(Var("__doc__", strtype), is_initialized_in_class=True)
   571	        if self.options.python_version >= (3, 10):
   572	            add_field(Var("__match_args__", match_args_type), is_initialized_in_class=True)
   573
   574	        assert info.tuple_type is not None  # Set by update_tuple_type() above.
   575	        shared_self_type = TypeVarType(
   576	            name=SELF_TVAR_NAME,
   577	            fullname=f"{info.fullname}.{SELF_TVAR_NAME}",
   578	            # Namespace is patched per-method below.
   579	            id=self.api.tvar_scope.new_unique_func_id(),
   580	            values=[],
   581	            upper_bound=info.tuple_type,
   582	            default=AnyType(TypeOfAny.from_omitted_generics),
   583	        )
   584
   585	        def add_method(
   586	            funcname: str,
   587	            ret: Type | None,  # None means use (patched) self-type
   588	            args: list[Argument],
   589	            is_classmethod: bool = False,
   590	            is_new: bool = False,
   591	        ) -> None:
   592	            fullname = f"{info.fullname}.{funcname}"
   593	            self_type = shared_self_type.copy_modified(
   594	                id=TypeVarId(shared_self_type.id.raw_id, namespace=fullname)
   595	            )
   596	            if ret is None:
   597	                ret = self_type
   598	            if is_classmethod or is_new:
   599	                first = [Argument(Var("_cls"), TypeType.make_normalized(self_type), None, ARG_POS)]
   600	            else:
   601	                first = [Argument(Var("_self"), self_type, None, ARG_POS)]
   602	            args = first + args
   603
   604	            types = [arg.type_annotation for arg in args]
   605	            items = [arg.variable.name for arg in args]
   606	            arg_kinds = [arg.kind for arg in args]
   607	            assert None not in types
   608	            signature = CallableType(cast(list[Type], types), arg_kinds, items, ret, function_type)
   609	            signature.variables = [self_type]
   610	            func = FuncDef(funcname, args, Block([]))
   611	            func.info = info
   612	            func.is_class = is_classmethod
   613	            func.type = set_callable_name(signature, func)
   614	            func._fullname = fullname
   615	            func.line = line
   616	            if is_classmethod:
   617	                v = Var(funcname, func.type)
   618	                v.is_classmethod = True
   619	                v.info = info
   620	                v._fullname = func._fullname
   621	                func.is_decorated = True
   622	                dec = Decorator(func, [NameExpr("classmethod")], v)
   623	                dec.line = line
   624	                sym = SymbolTableNode(MDEF, dec)
   625	            else:
   626	                sym = SymbolTableNode(MDEF, func)
   627	            sym.plugin_generated = True
   628	            info.names[funcname] = sym
   629
   630	        add_method(
   631	            "_replace",
   632	            ret=None,
   633	            args=[Argument(var, var.type, EllipsisExpr(), ARG_NAMED_OPT) for var in vars],
   634	        )
   635	        if self.options.python_version >= (3, 13):
   636	            add_method(
   637	                "__replace__",
   638	                ret=None,
   639	                args=[Argument(var, var.type, EllipsisExpr(), ARG_NAMED_OPT) for var in vars],
   640	            )
   641
   642	        def make_init_arg(var: Var) -> Argument:
   643	            default = default_items.get(var.name, None)
   644	            kind = ARG_POS if default is None else ARG_OPT
   645	            return Argument(var, var.type, default, kind)
   646
   647	        add_method("__new__", ret=None, args=[make_init_arg(var) for var in vars], is_new=True)
   648	        add_method("_asdict", args=[], ret=ordereddictype)
   649	        add_method(
   650	            "_make",
   651	            ret=None,
   652	            is_classmethod=True,
   653	            args=[Argument(Var("iterable", iterable_type), iterable_type, None, ARG_POS)],
   654	        )
   655
   656	        self_tvar_expr = TypeVarExpr(
   657	            SELF_TVAR_NAME,
   658	            info.fullname + "." + SELF_TVAR_NAME,
   659	            [],
   660	            info.tuple_type,
   661	            AnyType(TypeOfAny.from_omitted_generics),
   662	        )
   663	        info.names[SELF_TVAR_NAME] = SymbolTableNode(MDEF, self_tvar_expr)
   664	        return info

   666	    @contextmanager
   667	    def save_namedtuple_body(self, named_tuple_info: TypeInfo) -> Iterator[None]:
   668	        """Preserve the generated body of class-based named tuple and then restore it.
   669
   670	        Temporarily clear the names dict so we don't get errors about duplicate names
   671	        that were already set in build_namedtuple_typeinfo (we already added the tuple
   672	        field names while generating the TypeInfo, and actual duplicates are
   673	        already reported).
   674	        """
   675	        nt_names = named_tuple_info.names
   676	        named_tuple_info.names = SymbolTable()
   677
   678	        yield
   679
   680	        # Make sure we didn't use illegal names, then reset the names in the typeinfo.
