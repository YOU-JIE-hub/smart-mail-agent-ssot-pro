   233	        dataclass so as to ensure it is fully type checked according

   235	        """
   236	        info = self._cls.info
   237	        attributes = self.collect_attributes()
   238	        if attributes is None:
   239	            # Some definitions are not ready. We need another pass.
   240	            return False

   243	                return False

   245	            "init": self._get_bool_arg("init", True),

   247	            "order": self._get_bool_arg("order", self._spec.order_default),

   249	            "slots": self._get_bool_arg("slots", False),
   250	            "match_args": self._get_bool_arg("match_args", True),
   251	        }
   252	        py_version = self._api.options.python_version

   255	        # processed them yet. In order to work around this, we can simply skip generating
   256	        # __init__ if there are no attributes, because if the user truly did not define any,
   257	        # then the object default __init__ with an empty signature will be present anyway.
   258	        if (
   259	            decorator_arguments["init"]
   260	            and ("__init__" not in info.names or info.names["__init__"].plugin_generated)
   261	            and attributes
   262	        ):
   263	            args = [
   264	                attr.to_argument(info, of="__init__")
   265	                for attr in attributes
   266	                if attr.is_in_init and not self._is_kw_only_type(attr.type)
   267	            ]
   268
   269	            if info.fallback_to_any:
   270	                # Make positional args optional since we don't know their order.
   271	                # This will at least allow us to typecheck them if they are called
   272	                # as kwargs
   273	                for arg in args:
   274	                    if arg.kind == ARG_POS:
   275	                        arg.kind = ARG_OPT
   276
   277	                existing_args_names = {arg.variable.name for arg in args}
   278	                gen_args_name = "generated_args"
   279	                while gen_args_name in existing_args_names:
   280	                    gen_args_name += "_"
   281	                gen_kwargs_name = "generated_kwargs"
   282	                while gen_kwargs_name in existing_args_names:
   283	                    gen_kwargs_name += "_"
   284	                args = [
   285	                    Argument(Var(gen_args_name), AnyType(TypeOfAny.explicit), None, ARG_STAR),
   286	                    *args,
   287	                    Argument(Var(gen_kwargs_name), AnyType(TypeOfAny.explicit), None, ARG_STAR2),
   288	                ]
   289
   290	            add_method_to_class(
   291	                self._api, self._cls, "__init__", args=args, return_type=NoneType()
   292	            )
   293
   294	        if (
   295	            decorator_arguments["eq"]
   296	            and info.get("__eq__") is None
   297	            or decorator_arguments["order"]
   298	        ):
   299	            # Type variable for self types in generated methods.
   300	            obj_type = self._api.named_type("builtins.object")
   301	            self_tvar_expr = TypeVarExpr(
   302	                SELF_TVAR_NAME,
   303	                info.fullname + "." + SELF_TVAR_NAME,
   304	                [],
   305	                obj_type,
   306	                AnyType(TypeOfAny.from_omitted_generics),
   307	            )
   308	            info.names[SELF_TVAR_NAME] = SymbolTableNode(MDEF, self_tvar_expr)
   309
   310	        # Add <, >, <=, >=, but only if the class has an eq method.
   311	        if decorator_arguments["order"]:
   312	            if not decorator_arguments["eq"]:
   313	                self._api.fail('"eq" must be True if "order" is True', self._reason)
   314
   315	            for method_name in ["__lt__", "__gt__", "__le__", "__ge__"]:
   316	                # Like for __eq__ and __ne__, we want "other" to match
   317	                # the self type.
   318	                obj_type = self._api.named_type("builtins.object")
   319	                order_tvar_def = TypeVarType(
   320	                    SELF_TVAR_NAME,
   321	                    f"{info.fullname}.{SELF_TVAR_NAME}",
   322	                    id=TypeVarId(-1, namespace=f"{info.fullname}.{method_name}"),
   323	                    values=[],
   324	                    upper_bound=obj_type,
   325	                    default=AnyType(TypeOfAny.from_omitted_generics),
   326	                )
   327	                order_return_type = self._api.named_type("builtins.bool")
   328	                order_args = [
   329	                    Argument(Var("other", order_tvar_def), order_tvar_def, None, ARG_POS)
   330	                ]
   331
   332	                existing_method = info.get(method_name)
   333	                if existing_method is not None and not existing_method.plugin_generated:
   334	                    assert existing_method.node
   335	                    self._api.fail(
   336	                        f'You may not have a custom "{method_name}" method when "order" is True',
   337	                        existing_method.node,
   338	                    )
   339
   340	                add_method_to_class(
   341	                    self._api,
   342	                    self._cls,
   343	                    method_name,
   344	                    args=order_args,
   345	                    return_type=order_return_type,
   346	                    self_type=order_tvar_def,
   347	                    tvar_def=order_tvar_def,
   348	                )
   349
   350	        parent_decorator_arguments = []
   351	        for parent in info.mro[1:-1]:
   352	            parent_args = parent.metadata.get("dataclass")
   353
   354	            # Ignore parent classes that directly specify a dataclass transform-decorated metaclass
   355	            # when searching for usage of the frozen parameter. PEP 681 states that a class that
   356	            # directly specifies such a metaclass must be treated as neither frozen nor non-frozen.
   357	            if parent_args and not _has_direct_dataclass_transform_metaclass(parent):
   358	                parent_decorator_arguments.append(parent_args)
   359
   360	        if decorator_arguments["frozen"]:
   361	            if any(not parent["frozen"] for parent in parent_decorator_arguments):
   362	                self._api.fail("Frozen dataclass cannot inherit from a non-frozen dataclass", info)
   363	            self._propertize_callables(attributes, settable=False)
   364	            self._freeze(attributes)
   365	        else:
   366	            if any(parent["frozen"] for parent in parent_decorator_arguments):
   367	                self._api.fail("Non-frozen dataclass cannot inherit from a frozen dataclass", info)
   368	            self._propertize_callables(attributes)
   369
   370	        if decorator_arguments["slots"]:
   371	            self.add_slots(info, attributes, correct_version=py_version >= (3, 10))
   372
   373	        self.reset_init_only_vars(info, attributes)
   374
   375	        if (
   376	            decorator_arguments["match_args"]
   377	            and (
   378	                "__match_args__" not in info.names or info.names["__match_args__"].plugin_generated
   379	            )
   380	            and py_version >= (3, 10)
   381	        ):
   382	            str_type = self._api.named_type("builtins.str")
   383	            literals: list[Type] = [
   384	                LiteralType(attr.name, str_type)
   385	                for attr in attributes
   386	                if attr.is_in_init and not attr.kw_only
   387	            ]
   388	            match_args_type = TupleType(literals, self._api.named_type("builtins.tuple"))
   389	            add_attribute_to_class(self._api, self._cls, "__match_args__", match_args_type)
   390
   391	        self._add_dataclass_fields_magic_attribute()
   392	        self._add_internal_replace_method(attributes)
   393	        if self._api.options.python_version >= (3, 13):
   394	            self._add_dunder_replace(attributes)
   395
   396	        if "__post_init__" in info.names:
   397	            self._add_internal_post_init_method(attributes)
   398
   399	        info.metadata["dataclass"] = {
   400	            "attributes": [attr.serialize() for attr in attributes],
   401	            "frozen": decorator_arguments["frozen"],
   402	        }
   403
   404	        return True

   449	    def add_slots(
   450	        self, info: TypeInfo, attributes: list[DataclassAttribute], *, correct_version: bool
   451	    ) -> None:
   452	        if not correct_version:
   453	            # This means that version is lower than `3.10`,
   454	            # it is just a non-existent argument for `dataclass` function.
   455	            self._api.fail(
   456	                'Keyword argument "slots" for "dataclass" is only valid in Python 3.10 and higher',
   457	                self._reason,
   458	            )
   459	            return
   460
   461	        generated_slots = {attr.name for attr in attributes}
   462	        if (info.slots is not None and info.slots != generated_slots) or info.names.get(
   463	            "__slots__"
   464	        ):
   465	            # This means we have a slots conflict.
   466	            # Class explicitly specifies a different `__slots__` field.
   467	            # And `@dataclass(slots=True)` is used.
   468	            # In runtime this raises a type error.
   469	            self._api.fail(
   470	                '"{}" both defines "__slots__" and is used with "slots=True"'.format(
   471	                    self._cls.name
   472	                ),
   473	                self._cls,
   474	            )
   475	            return
   476
   477	        if any(p.slots is None for p in info.mro[1:-1]):
   478	            # At least one type in mro (excluding `self` and `object`)
   479	            # does not have concrete `__slots__` defined. Ignoring.
   480	            return
   481
   482	        info.slots = generated_slots
   483
   484	        # Now, insert `.__slots__` attribute to class namespace:
   485	        slots_type = TupleType(
   486	            [self._api.named_type("builtins.str") for _ in generated_slots],
   487	            self._api.named_type("builtins.tuple"),
   488	        )
   489	        add_attribute_to_class(self._api, self._cls, "__slots__", slots_type)

   491	    def reset_init_only_vars(self, info: TypeInfo, attributes: list[DataclassAttribute]) -> None:
   492	        """Remove init-only vars from the class and reset init var declarations."""
   493	        for attr in attributes:
   494	            if attr.is_init_var:
   495	                if attr.name in info.names:
   496	                    del info.names[attr.name]
   497	                else:
   498	                    # Nodes of superclass InitVars not used in __init__ cannot be reached.
   499	                    assert attr.is_init_var
   500	                for stmt in info.defn.defs.body:
   501	                    if isinstance(stmt, AssignmentStmt) and stmt.unanalyzed_type:
   502	                        lvalue = stmt.lvalues[0]
   503	                        if isinstance(lvalue, NameExpr) and lvalue.name == attr.name:
   504	                            # Reset node so that another semantic analysis pass will
   505	                            # recreate a symbol node for this attribute.
   506	                            lvalue.node = None

   508	    def _get_assignment_statements_from_if_statement(
   509	        self, stmt: IfStmt
   510	    ) -> Iterator[AssignmentStmt]:
   511	        for body in stmt.body:
   512	            if not body.is_unreachable:
   513	                yield from self._get_assignment_statements_from_block(body)
   514	        if stmt.else_body is not None and not stmt.else_body.is_unreachable:
   515	            yield from self._get_assignment_statements_from_block(stmt.else_body)
   516
   517	    def _get_assignment_statements_from_block(self, block: Block) -> Iterator[AssignmentStmt]:
   518	        for stmt in block.body:
   519	            if isinstance(stmt, AssignmentStmt):
   520	                yield stmt
   521	            elif isinstance(stmt, IfStmt):
   522	                yield from self._get_assignment_statements_from_if_statement(stmt)

   524	    def collect_attributes(self) -> list[DataclassAttribute] | None:
   525	        """Collect all attributes declared in the dataclass and its parents.
   526
   527	        All assignments of the form
   528
   529	          a: SomeType
   530	          b: SomeOtherType = ...
   531
   532	        are collected.
   533
   534	        Return None if some dataclass base class hasn't been processed
   535	        yet and thus we'll need to ask for another pass.
   536	        """
   537	        cls = self._cls
   538
   539	        # First, collect attributes belonging to any class in the MRO, ignoring duplicates.
   540	        #
   541	        # We iterate through the MRO in reverse because attrs defined in the parent must appear
   542	        # earlier in the attributes list than attrs defined in the child. See:
   543	        # https://docs.python.org/3/library/dataclasses.html#inheritance
   544	        #
   545	        # However, we also want attributes defined in the subtype to override ones defined
   546	        # in the parent. We can implement this via a dict without disrupting the attr order
   547	        # because dicts preserve insertion order in Python 3.7+.
   548	        found_attrs: dict[str, DataclassAttribute] = {}
   549	        for info in reversed(cls.info.mro[1:-1]):
   550	            if "dataclass_tag" in info.metadata and "dataclass" not in info.metadata:
   551	                # We haven't processed the base class yet. Need another pass.
   552	                return None
   553	            if "dataclass" not in info.metadata:
   554	                continue
   555
   556	            # Each class depends on the set of attributes in its dataclass ancestors.
   557	            self._api.add_plugin_dependency(make_wildcard_trigger(info.fullname))
   558
   559	            for data in info.metadata["dataclass"]["attributes"]:
   560	                name: str = data["name"]
   561
   562	                attr = DataclassAttribute.deserialize(info, data, self._api)
   563	                # TODO: We shouldn't be performing type operations during the main
   564	                #       semantic analysis pass, since some TypeInfo attributes might
   565	                #       still be in flux. This should be performed in a later phase.
   566	                attr.expand_typevar_from_subtype(cls.info)
   567	                found_attrs[name] = attr
   568
   569	                sym_node = cls.info.names.get(name)
   570	                if sym_node and sym_node.node and not isinstance(sym_node.node, Var):
   571	                    self._api.fail(
   572	                        "Dataclass attribute may only be overridden by another attribute",
   573	                        sym_node.node,
   574	                    )
   575
   576	        # Second, collect attributes belonging to the current class.
   577	        current_attr_names: set[str] = set()
   578	        kw_only = self._get_bool_arg("kw_only", self._spec.kw_only_default)
   579	        for stmt in self._get_assignment_statements_from_block(cls.defs):
   580	            # Any assignment that doesn't use the new type declaration
   581	            # syntax can be ignored out of hand.
   582	            if not stmt.new_syntax:
   583	                continue
   584
   585	            # a: int, b: str = 1, 'foo' is not supported syntax so we
   586	            # don't have to worry about it.
   587	            lhs = stmt.lvalues[0]
   588	            if not isinstance(lhs, NameExpr):
   589	                continue
   590
   591	            sym = cls.info.names.get(lhs.name)
   592	            if sym is None:
   593	                # There was probably a semantic analysis error.
   594	                continue
   595
   596	            node = sym.node
   597	            assert not isinstance(node, PlaceholderNode)
   598
   599	            if isinstance(node, TypeAlias):
   600	                self._api.fail(
   601	                    ("Type aliases inside dataclass definitions are not supported at runtime"),
   602	                    node,
   603	                )
   604	                # Skip processing this node. This doesn't match the runtime behaviour,
   605	                # but the only alternative would be to modify the SymbolTable,
   606	                # and it's a little hairy to do that in a plugin.
   607	                continue
   608	            if isinstance(node, Decorator):
   609	                # This might be a property / field name clash.
   610	                # We will issue an error later.
   611	                continue
   612
   613	            assert isinstance(node, Var)
   614
   615	            # x: ClassVar[int] is ignored by dataclasses.
   616	            if node.is_classvar:
   617	                continue
   618
   619	            # x: InitVar[int] is turned into x: int and is removed from the class.
   620	            is_init_var = False
   621	            node_type = get_proper_type(node.type)
   622	            if (
   623	                isinstance(node_type, Instance)
   624	                and node_type.type.fullname == "dataclasses.InitVar"
   625	            ):
   626	                is_init_var = True
   627	                node.type = node_type.args[0]
   628
   629	            if self._is_kw_only_type(node_type):
   630	                kw_only = True
   631
   632	            has_field_call, field_args = self._collect_field_args(stmt.rvalue)
   633
   634	            is_in_init_param = field_args.get("init")
   635	            if is_in_init_param is None:
   636	                is_in_init = self._get_default_init_value_for_field_specifier(stmt.rvalue)
   637	            else:
   638	                is_in_init = bool(self._api.parse_bool(is_in_init_param))
   639
   640	            has_default = False
   641	            # Ensure that something like x: int = field() is rejected
   642	            # after an attribute with a default.
   643	            if has_field_call:
   644	                has_default = (
   645	                    "default" in field_args
   646	                    or "default_factory" in field_args
   647	                    # alias for default_factory defined in PEP 681
   648	                    or "factory" in field_args
   649	                )
   650
   651	            # All other assignments are already type checked.
   652	            elif not isinstance(stmt.rvalue, TempNode):
   653	                has_default = True
   654
   655	            if not has_default and self._spec is _TRANSFORM_SPEC_FOR_DATACLASSES:
   656	                # Make all non-default dataclass attributes implicit because they are de-facto
   657	                # set on self in the generated __init__(), not in the class body. On the other
   658	                # hand, we don't know how custom dataclass transforms initialize attributes,
   659	                # so we don't treat them as implicit. This is required to support descriptors
   660	                # (https://github.com/python/mypy/issues/14868).
   661	                sym.implicit = True
   662
   663	            is_kw_only = kw_only
   664	            # Use the kw_only field arg if it is provided. Otherwise use the
   665	            # kw_only value from the decorator parameter.
   666	            field_kw_only_param = field_args.get("kw_only")
   667	            if field_kw_only_param is not None:
   668	                value = self._api.parse_bool(field_kw_only_param)
   669	                if value is not None:
   670	                    is_kw_only = value
   671	                else:
   672	                    self._api.fail('"kw_only" argument must be a boolean literal', stmt.rvalue)
   673
   674	            if sym.type is None and node.is_final and node.is_inferred:
   675	                # This is a special case, assignment like x: Final = 42 is classified
   676	                # annotated above, but mypy strips the `Final` turning it into x = 42.
   677	                # We do not support inferred types in dataclasses, so we can try inferring
   678	                # type for simple literals, and otherwise require an explicit type
   679	                # argument for Final[...].
   680	                typ = self._api.analyze_simple_literal_type(stmt.rvalue, is_final=True)
   681	                if typ:
   682	                    node.type = typ
   683	                else:
   684	                    self._api.fail(
   685	                        "Need type argument for Final[...] with non-literal default in dataclass",
   686	                        stmt,
   687	                    )
   688	                    node.type = AnyType(TypeOfAny.from_error)
   689
   690	            alias = None
   691	            if "alias" in field_args:
   692	                alias = self._api.parse_str_literal(field_args["alias"])
   693	                if alias is None:
   694	                    self._api.fail(
   695	                        message_registry.DATACLASS_FIELD_ALIAS_MUST_BE_LITERAL,
   696	                        stmt.rvalue,
   697	                        code=errorcodes.LITERAL_REQ,
   698	                    )
   699
   700	            current_attr_names.add(lhs.name)
   701	            with state.strict_optional_set(self._api.options.strict_optional):
   702	                init_type = self._infer_dataclass_attr_init_type(sym, lhs.name, stmt)
   703	            found_attrs[lhs.name] = DataclassAttribute(
   704	                name=lhs.name,
   705	                alias=alias,
   706	                is_in_init=is_in_init,
   707	                is_init_var=is_init_var,
   708	                has_default=has_default,
   709	                line=stmt.line,
   710	                column=stmt.column,
   711	                type=init_type,
   712	                info=cls.info,
   713	                kw_only=is_kw_only,
   714	                is_neither_frozen_nor_nonfrozen=_has_direct_dataclass_transform_metaclass(
   715	                    cls.info
   716	                ),
   717	                api=self._api,
   718	            )
   719
   720	        all_attrs = list(found_attrs.values())
   721	        all_attrs.sort(key=lambda a: a.kw_only)
   722
   723	        # Third, ensure that arguments without a default don't follow
   724	        # arguments that have a default and that the KW_ONLY sentinel
   725	        # is only provided once.
   726	        found_default = False
   727	        found_kw_sentinel = False
   728	        for attr in all_attrs:
   729	            # If we find any attribute that is_in_init, not kw_only, and that
   730	            # doesn't have a default after one that does have one,
   731	            # then that's an error.
   732	            if found_default and attr.is_in_init and not attr.has_default and not attr.kw_only:
   733	                # If the issue comes from merging different classes, report it
   734	                # at the class definition point.
   735	                context: Context = cls
   736	                if attr.name in current_attr_names:
   737	                    context = Context(line=attr.line, column=attr.column)
   738	                self._api.fail(
   739	                    "Attributes without a default cannot follow attributes with one", context
   740	                )
   741
   742	            found_default = found_default or (attr.has_default and attr.is_in_init)
   743	            if found_kw_sentinel and self._is_kw_only_type(attr.type):
   744	                context = cls
   745	                if attr.name in current_attr_names:
   746	                    context = Context(line=attr.line, column=attr.column)
   747	                self._api.fail(
   748	                    "There may not be more than one field with the KW_ONLY type", context
   749	                )
   750	            found_kw_sentinel = found_kw_sentinel or self._is_kw_only_type(attr.type)
   751	        return all_attrs

   753	    def _freeze(self, attributes: list[DataclassAttribute]) -> None:
   754	        """Converts all attributes to @property methods in order to
   755	        emulate frozen classes.
   756	        """
   757	        info = self._cls.info
   758	        for attr in attributes:
   759	            # Classes that directly specify a dataclass_transform metaclass must be neither frozen
   760	            # non non-frozen per PEP681. Though it is surprising, this means that attributes from
   761	            # such a class must be writable even if the rest of the class hierarchy is frozen. This
   762	            # matches the behavior of Pyright (the reference implementation).
   763	            if attr.is_neither_frozen_nor_nonfrozen:
   764	                continue
   765
   766	            sym_node = info.names.get(attr.name)
   767	            if sym_node is not None:
   768	                var = sym_node.node
   769	                if isinstance(var, Var):
   770	                    if var.is_final:
   771	                        continue  # do not turn `Final` attrs to `@property`
   772	                    var.is_property = True
   773	            else:
   774	                var = attr.to_var(info)
   775	                var.info = info
   776	                var.is_property = True
   777	                var._fullname = info.fullname + "." + var.name
   778	                info.names[var.name] = SymbolTableNode(MDEF, var)

   105	    stnode = type_with_gnv.names["_generate_next_value_"]
   106
   107	    # This should be a `CallableType`
   108	    node_type = get_proper_type(stnode.type)
   109	    if isinstance(node_type, CallableType):
   110	        if type_with_gnv.fullname == "enum.Enum":
   111	            int_type = ctx.api.named_generic_type("builtins.int", [])
   112	            return int_type
   113	        return get_proper_type(node_type.ret_type)
   114	    return ctx.default_attr_type

   117	def _implements_new(info: TypeInfo) -> bool:
   118	    """Check whether __new__ comes from enum.Enum or was implemented in a
   119	    subclass. In the latter case, we must infer Any as long as mypy can't infer
   120	    the type of _value_ from assignments in __new__.
   121	    """
   122	    type_with_new = _first(
   123	        ti
   124	        for ti in info.mro
   125	        if ti.names.get("__new__") and not ti.fullname.startswith("builtins.")
   126	    )
   127	    if type_with_new is None:
   128	        return False
   129	    return type_with_new.fullname not in ("enum.Enum", "enum.IntEnum", "enum.StrEnum")

   132	def enum_member_callback(ctx: mypy.plugin.FunctionContext) -> Type:
   133	    """By default `member(1)` will be inferred as `member[int]`,
   134	    we want to improve the inference to be `Literal[1]` here."""
   135	    if ctx.arg_types or ctx.arg_types[0]:
   136	        arg = get_proper_type(ctx.arg_types[0][0])
   137	        proper_return = get_proper_type(ctx.default_return_type)
   138	        if (
   139	            isinstance(arg, Instance)
   140	            and arg.last_known_value
   141	            and isinstance(proper_return, Instance)
   142	            and len(proper_return.args) == 1
   143	        ):
   144	            return proper_return.copy_modified(args=[arg])
   145	    return ctx.default_return_type
