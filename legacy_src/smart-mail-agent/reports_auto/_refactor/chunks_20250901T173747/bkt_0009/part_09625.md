   328	def write_file(path: str, contents: str) -> None:
   329	    """Write data into a file.
   330
   331	    If the file already exists and has the same contents we
   332	    want to write, skip writing so as to preserve the mtime
   333	    and avoid triggering recompilation.
   334	    """
   335	    # We encode it ourselves and open the files as binary to avoid windows
   336	    # newline translation
   337	    encoded_contents = contents.encode("utf-8")
   338	    try:
   339	        with open(path, "rb") as f:
   340	            old_contents: bytes | None = f.read()
   341	    except OSError:
   342	        old_contents = None
   343	    if old_contents != encoded_contents:
   344	        os.makedirs(os.path.dirname(path), exist_ok=True)
   345	        with open(path, "wb") as g:
   346	            g.write(encoded_contents)
   347
   348	        # Fudge the mtime forward because otherwise when two builds happen close
   349	        # together (like in a test) setuptools might not realize the source is newer
   350	        # than the new artifact.
   351	        # XXX: This is bad though.
   352	        new_mtime = os.stat(path).st_mtime + 1
   353	        os.utime(path, times=(new_mtime, new_mtime))

   356	def construct_groups(
   357	    sources: list[BuildSource],
   358	    separate: bool | list[tuple[list[str], str | None]],
   359	    use_shared_lib: bool,
   360	    group_name_override: str | None,
   361	) -> emitmodule.Groups:
   362	    """Compute Groups given the input source list and separate configs.
   363
   364	    separate is the user-specified configuration for how to assign
   365	    modules to compilation groups (see mypycify docstring for details).
   366
   367	    This takes that and expands it into our internal representation of
   368	    group configuration, documented in mypyc.emitmodule's definition
   369	    of Group.
   370	    """
   371
   372	    if separate is True:
   373	        groups: emitmodule.Groups = [([source], None) for source in sources]
   374	    elif isinstance(separate, list):
   375	        groups = []
   376	        used_sources = set()
   377	        for files, name in separate:
   378	            group_sources = [src for src in sources if src.path in files]
   379	            groups.append((group_sources, name))
   380	            used_sources.update(group_sources)
   381	        unused_sources = [src for src in sources if src not in used_sources]
   382	        if unused_sources:
   383	            groups.extend([([source], None) for source in unused_sources])
   384	    else:
   385	        groups = [(sources, None)]
   386
   387	    # Generate missing names
   388	    for i, (group, name) in enumerate(groups):
   389	        if use_shared_lib and not name:
   390	            if group_name_override is not None:
   391	                name = group_name_override
   392	            else:
   393	                name = group_name([source.module for source in group])
   394	        groups[i] = (group, name)
   395
   396	    return groups

   399	def get_header_deps(cfiles: list[tuple[str, str]]) -> list[str]:
   400	    """Find all the headers used by a group of cfiles.

   402	    We do this by just regexping the source, which is a bit simpler than
   403	    properly plumbing the data through.

   416	    paths: list[str],
   417	    compiler_options: CompilerOptions,
   418	    *,
   419	    separate: bool | list[tuple[list[str], str | None]] = False,
   420	    only_compile_paths: Iterable[str] | None = None,

   431	    # of the modules are in package. (Because I didn't want to fuss

   448	            all_sources, options, groups, fscache, compiler_options=compiler_options

     1	"""Encode valid C string literals from Python strings.
     2
     3	If a character is not allowed in C string literals, it is either emitted
     4	as a simple escape sequence (e.g. '\\n'), or an octal escape sequence
     5	with exactly three digits ('\\oXXX'). Question marks are escaped to
     6	prevent trigraphs in the string literal from being interpreted. Note
     7	that '\\?' is an invalid escape sequence in Python.
     8
     9	Consider the string literal "AB\\xCDEF". As one would expect, Python
    10	parses it as ['A', 'B', 0xCD, 'E', 'F']. However, the C standard
    11	specifies that all hexadecimal digits immediately following '\\x' will
    12	be interpreted as part of the escape sequence. Therefore, it is
    13	unexpectedly parsed as ['A', 'B', 0xCDEF].
    14
    15	Emitting ("AB\\xCD" "EF") would avoid this behaviour. However, we opt
    16	for simplicity and use octal escape sequences instead. They do not
    17	suffer from the same issue as they are defined to parse at most three
    18	octal digits.
    19	"""
    20
    21	from __future__ import annotations
    22
    23	import string
    24	from typing import Final
    25
    26	CHAR_MAP: Final = [f"\\{i:03o}" for i in range(256)]
    27
    28	# It is safe to use string.printable as it always uses the C locale.
    29	for c in string.printable:
    30	    CHAR_MAP[ord(c)] = c
    31
    32	# These assignments must come last because we prioritize simple escape
    33	# sequences over any other representation.
    34	for c in ("'", '"', "\\", "a", "b", "f", "n", "r", "t", "v"):
    35	    escaped = f"\\{c}"
    36	    decoded = escaped.encode("ascii").decode("unicode_escape")
    37	    CHAR_MAP[ord(decoded)] = escaped
    38
    39	# This escape sequence is invalid in Python.
    40	CHAR_MAP[ord("?")] = r"\?"

   198	    def object_annotation(self, obj: object, line: str) -> str:
   199	        """Build a C comment with an object's string representation.
   200
   201	        If the comment exceeds the line length limit, it's wrapped into a
   202	        multiline string (with the extra lines indented to be aligned with
   203	        the first line's comment).
   204
   205	        If it contains illegal characters, an empty string is returned."""
   206	        line_width = self._indent + len(line)
   207	        formatted = pprint.pformat(obj, compact=True, width=max(90 - line_width, 20))
   208	        if any(x in formatted for x in ("/*", "*/", "\0")):
   209	            return ""
   210
   211	        if "\n" in formatted:
   212	            first_line, rest = formatted.split("\n", maxsplit=1)
   213	            comment_continued = textwrap.indent(rest, (line_width + 3) * " ")
   214	            return f" /* {first_line}\n{comment_continued} */"
   215	        else:
   216	            return f" /* {formatted} */"

   218	    def emit_line(self, line: str = "", *, ann: object = None) -> None:
   219	        if line.startswith("}"):
   220	            self.dedent()
   221	        comment = self.object_annotation(ann, line) if ann is not None else ""
   222	        self.fragments.append(self._indent * " " + line + comment + "\n")
   223	        if line.endswith("{"):
   224	            self.indent()

   257	    def get_module_group_prefix(self, module_name: str) -> str:
   258	        """Get the group prefix for a module (relative to the current group).
   259
   260	        The prefix should be prepended to the object name whenever
   261	        accessing an object from this module.
   262
   263	        If the module lives is in the current compilation group, there is
   264	        no prefix.  But if it lives in a different group (and hence a separate
   265	        extension module), we need to access objects from it indirectly via an
   266	        export table.
   267
   268	        For example, for code in group `a` to call a function `bar` in group `b`,
   269	        it would need to do `exports_b.CPyDef_bar(...)`, while code that is
   270	        also in group `b` can simply do `CPyDef_bar(...)`.
   271
   272	        Thus the prefix for a module in group `b` is 'exports_b.' if the current
   273	        group is *not* b and just '' if it is.
   274	        """
   275	        groups = self.context.group_map
   276	        target_group_name = groups.get(module_name)
   277	        if target_group_name and target_group_name != self.context.group_name:
   278	            self.context.group_deps.add(target_group_name)
   279	            return f"exports_{exported_name(target_group_name)}."
   280	        else:
   281	            return ""

   385	    def _emit_attr_bitmap_update(
   386	        self, value: str, obj: str, rtype: RType, cl: ClassIR, attr: str, clear: bool
   387	    ) -> None:
   388	        if value:
   389	            check = self.error_value_check(rtype, value, "==")
   390	            self.emit_line(f"if (unlikely({check})) {{")
   391	        index = cl.bitmap_attrs.index(attr)
   392	        mask = 1 << (index & (BITMAP_BITS - 1))
   393	        bitmap = self.attr_bitmap_expr(obj, cl, index)
   394	        if clear:
   395	            self.emit_line(f"{bitmap} &= ~{mask};")
   396	        else:
   397	            self.emit_line(f"{bitmap} |= {mask};")
   398	        if value:
   399	            self.emit_line("}")

   401	    def emit_undefined_attr_check(
   402	        self,
   403	        rtype: RType,
   404	        attr_expr: str,
   405	        compare: str,
   406	        obj: str,
   407	        attr: str,
   408	        cl: ClassIR,
   409	        *,
   410	        unlikely: bool = False,
   411	    ) -> None:

   426	                rtype, value, self.c_error_value, compare, check_exception=False

   431	    def tuple_undefined_check_cond(
   432	        self,
   433	        rtuple: RTuple,
   434	        tuple_expr_in_c: str,
   435	        c_type_compare_val: Callable[[RType], str],
   436	        compare: str,
   437	        *,
   438	        check_exception: bool = True,
   439	    ) -> str:
   440	        if len(rtuple.types) == 0:
   441	            # empty tuple
   442	            return "{}.empty_struct_error_flag {} {}".format(
   443	                tuple_expr_in_c, compare, c_type_compare_val(int_rprimitive)
   444	            )
   445	        if rtuple.error_overlap:
   446	            i = 0
   447	            item_type = rtuple.types[0]
   448	        else:
   449	            for i, typ in enumerate(rtuple.types):
   450	                if not typ.error_overlap:
   451	                    item_type = rtuple.types[i]
   452	                    break
   453	            else:
   454	                assert False, "not expecting tuple with error overlap"
   455	        if isinstance(item_type, RTuple):
   456	            return self.tuple_undefined_check_cond(
   457	                item_type, tuple_expr_in_c + f".f{i}", c_type_compare_val, compare
   458	            )
   459	        else:
   460	            check = f"{tuple_expr_in_c}.f{i} {compare} {c_type_compare_val(item_type)}"
   461	            if rtuple.error_overlap and check_exception:
   462	                check += " && PyErr_Occurred()"
   463	            return check

   495	    def emit_inc_ref(self, dest: str, rtype: RType, *, rare: bool = False) -> None:
   496	        """Increment reference count of C expression `dest`.
   497
   498	        For composite unboxed structures (e.g. tuples) recursively
   499	        increment reference counts for each component.
   500
   501	        If rare is True, optimize for code size and compilation speed.
   502	        """
   503	        if is_int_rprimitive(rtype):
   504	            if rare:
   505	                self.emit_line("CPyTagged_IncRef(%s);" % dest)
   506	            else:
   507	                self.emit_line("CPyTagged_INCREF(%s);" % dest)
   508	        elif isinstance(rtype, RTuple):
   509	            for i, item_type in enumerate(rtype.types):
   510	                self.emit_inc_ref(f"{dest}.f{i}", item_type)
   511	        elif not rtype.is_unboxed:
   512	            # Always inline, since this is a simple but very hot op
   513	            if rtype.may_be_immortal or not HAVE_IMMORTAL:
   514	                self.emit_line("CPy_INCREF(%s);" % dest)
   515	            else:
   516	                self.emit_line("CPy_INCREF_NO_IMM(%s);" % dest)
   517	        # Otherwise assume it's an unboxed, pointerless value and do nothing.
   518

   524	        For composite unboxed structures (e.g. tuples) recursively

   543	                # Inlined
   544	                if rtype.may_be_immortal or not HAVE_IMMORTAL:

   546	                else:

   556	    def emit_cast(
   557	        self,
   558	        src: str,
   559	        dest: str,
   560	        typ: RType,
   561	        *,
   562	        declare_dest: bool = False,
   563	        error: ErrorHandler | None = None,
   564	        raise_exception: bool = True,
   565	        optional: bool = False,
   566	        src_type: RType | None = None,
   567	        likely: bool = True,
   568	    ) -> None:
   569	        """Emit code for casting a value of given type.
   570
   571	        Somewhat strangely, this supports unboxed types but only
   572	        operates on boxed versions.  This is necessary to properly
   573	        handle types such as Optional[int] in compatibility glue.
   574
   575	        By default, assign NULL (error value) to dest if the value has
   576	        an incompatible type and raise TypeError. These can be customized
   577	        using 'error' and 'raise_exception'.
   578
   579	        Always copy/steal the reference in 'src'.
   580
   581	        Args:
   582	            src: Name of source C variable
   583	            dest: Name of target C variable
   584	            typ: Type of value
   585	            declare_dest: If True, also declare the variable 'dest'
   586	            error: What happens on error
   587	            raise_exception: If True, also raise TypeError on failure
   588	            likely: If the cast is likely to succeed (can be False for unions)
   589	        """
   590	        error = error or AssignHandler()
   591
   592	        # Special case casting *from* optional
   593	        if src_type and is_optional_type(src_type) and not is_object_rprimitive(typ):
   594	            value_type = optional_value_type(src_type)
   595	            assert value_type is not None
   596	            if is_same_type(value_type, typ):
   597	                if declare_dest:
   598	                    self.emit_line(f"PyObject *{dest};")
   599	                check = "({} != Py_None)"
   600	                if likely:
   601	                    check = f"(likely{check})"
   602	                self.emit_arg_check(src, dest, typ, check.format(src), optional)
   603	                self.emit_lines(f"    {dest} = {src};", "else {")
   604	                self.emit_cast_error_handler(error, src, dest, typ, raise_exception)
   605	                self.emit_line("}")
   606	                return
   607
   608	        # TODO: Verify refcount handling.
   609	        if (
   610	            is_list_rprimitive(typ)
   611	            or is_dict_rprimitive(typ)
   612	            or is_set_rprimitive(typ)
   613	            or is_frozenset_rprimitive(typ)
   614	            or is_str_rprimitive(typ)
   615	            or is_range_rprimitive(typ)
   616	            or is_float_rprimitive(typ)
   617	            or is_int_rprimitive(typ)
   618	            or is_bool_rprimitive(typ)
   619	            or is_bit_rprimitive(typ)
   620	            or is_fixed_width_rtype(typ)
   621	        ):
   622	            if declare_dest:
   623	                self.emit_line(f"PyObject *{dest};")
   624	            if is_list_rprimitive(typ):
   625	                prefix = "PyList"
   626	            elif is_dict_rprimitive(typ):
   627	                prefix = "PyDict"
   628	            elif is_set_rprimitive(typ):
   629	                prefix = "PySet"
   630	            elif is_frozenset_rprimitive(typ):
   631	                prefix = "PyFrozenSet"
   632	            elif is_str_rprimitive(typ):
   633	                prefix = "PyUnicode"
   634	            elif is_range_rprimitive(typ):
   635	                prefix = "PyRange"
   636	            elif is_float_rprimitive(typ):
   637	                prefix = "CPyFloat"
   638	            elif is_int_rprimitive(typ) or is_fixed_width_rtype(typ):
   639	                # TODO: Range check for fixed-width types?
   640	                prefix = "PyLong"
   641	            elif is_bool_rprimitive(typ) or is_bit_rprimitive(typ):
   642	                prefix = "PyBool"
   643	            else:
   644	                assert False, f"unexpected primitive type: {typ}"
   645	            check = "({}_Check({}))"
   646	            if likely:
   647	                check = f"(likely{check})"
   648	            self.emit_arg_check(src, dest, typ, check.format(prefix, src), optional)
   649	            self.emit_lines(f"    {dest} = {src};", "else {")
   650	            self.emit_cast_error_handler(error, src, dest, typ, raise_exception)
   651	            self.emit_line("}")
   652	        elif is_bytes_rprimitive(typ):
   653	            if declare_dest:
   654	                self.emit_line(f"PyObject *{dest};")
   655	            check = "(PyBytes_Check({}) || PyByteArray_Check({}))"
   656	            if likely:
   657	                check = f"(likely{check})"
   658	            self.emit_arg_check(src, dest, typ, check.format(src, src), optional)
   659	            self.emit_lines(f"    {dest} = {src};", "else {")
   660	            self.emit_cast_error_handler(error, src, dest, typ, raise_exception)
   661	            self.emit_line("}")
   662	        elif is_tuple_rprimitive(typ):
   663	            if declare_dest:
   664	                self.emit_line(f"{self.ctype(typ)} {dest};")
   665	            check = "(PyTuple_Check({}))"
   666	            if likely:
   667	                check = f"(likely{check})"
   668	            self.emit_arg_check(src, dest, typ, check.format(src), optional)
   669	            self.emit_lines(f"    {dest} = {src};", "else {")
   670	            self.emit_cast_error_handler(error, src, dest, typ, raise_exception)
   671	            self.emit_line("}")
   672	        elif isinstance(typ, RInstance):
   673	            if declare_dest:
   674	                self.emit_line(f"PyObject *{dest};")
   675	            concrete = all_concrete_classes(typ.class_ir)
   676	            # If there are too many concrete subclasses or we can't find any
   677	            # (meaning the code ought to be dead or we aren't doing global opts),
   678	            # fall back to a normal typecheck.
   679	            # Otherwise check all the subclasses.
   680	            if not concrete or len(concrete) > FAST_ISINSTANCE_MAX_SUBCLASSES + 1:
   681	                check = "(PyObject_TypeCheck({}, {}))".format(
   682	                    src, self.type_struct_name(typ.class_ir)
   683	                )
   684	            else:
   685	                full_str = "(Py_TYPE({src}) == {targets[0]})"
   686	                for i in range(1, len(concrete)):
   687	                    full_str += " || (Py_TYPE({src}) == {targets[%d]})" % i
   688	                if len(concrete) > 1:
   689	                    full_str = "(%s)" % full_str
   690	                check = full_str.format(
   691	                    src=src, targets=[self.type_struct_name(ir) for ir in concrete]
   692	                )
   693	            if likely:
   694	                check = f"(likely{check})"
   695	            self.emit_arg_check(src, dest, typ, check, optional)
   696	            self.emit_lines(f"    {dest} = {src};", "else {")
   697	            self.emit_cast_error_handler(error, src, dest, typ, raise_exception)
   698	            self.emit_line("}")
   699	        elif is_none_rprimitive(typ):
   700	            if declare_dest:
   701	                self.emit_line(f"PyObject *{dest};")
   702	            check = "({} == Py_None)"
   703	            if likely:
   704	                check = f"(likely{check})"
   705	            self.emit_arg_check(src, dest, typ, check.format(src), optional)
   706	            self.emit_lines(f"    {dest} = {src};", "else {")
   707	            self.emit_cast_error_handler(error, src, dest, typ, raise_exception)
   708	            self.emit_line("}")
   709	        elif is_object_rprimitive(typ):
   710	            if declare_dest:
   711	                self.emit_line(f"PyObject *{dest};")
   712	            self.emit_arg_check(src, dest, typ, "", optional)
   713	            self.emit_line(f"{dest} = {src};")
   714	            if optional:
   715	                self.emit_line("}")
   716	        elif isinstance(typ, RUnion):
   717	            self.emit_union_cast(
   718	                src, dest, typ, declare_dest, error, optional, src_type, raise_exception
   719	            )
   720	        elif isinstance(typ, RTuple):
   721	            assert not optional
   722	            self.emit_tuple_cast(src, dest, typ, declare_dest, error, src_type)
   723	        else:
   724	            assert False, "Cast not implemented: %s" % typ
   725
   726	    def emit_cast_error_handler(
   727	        self, error: ErrorHandler, src: str, dest: str, typ: RType, raise_exception: bool
   728	    ) -> None:
   729	        if raise_exception:
   730	            if isinstance(error, TracebackAndGotoHandler):
   731	                # Merge raising and emitting traceback entry into a single call.
   732	                self.emit_type_error_traceback(
   733	                    error.source_path, error.module_name, error.traceback_entry, typ=typ, src=src
   734	                )
   735	                self.emit_line("goto %s;" % error.label)
   736	                return
   737	            self.emit_line(f'CPy_TypeError("{self.pretty_name(typ)}", {src}); ')
   738	        if isinstance(error, AssignHandler):
   739	            self.emit_line("%s = NULL;" % dest)
   740	        elif isinstance(error, GotoHandler):
   741	            self.emit_line("goto %s;" % error.label)
   742	        elif isinstance(error, TracebackAndGotoHandler):
   743	            self.emit_line("%s = NULL;" % dest)
   744	            self.emit_traceback(error.source_path, error.module_name, error.traceback_entry)
   745	            self.emit_line("goto %s;" % error.label)
   746	        else:
   747	            assert isinstance(error, ReturnHandler)
   748	            self.emit_line("return %s;" % error.value)
   749
   750	    def emit_union_cast(
   751	        self,
   752	        src: str,
   753	        dest: str,
   754	        typ: RUnion,
   755	        declare_dest: bool,
   756	        error: ErrorHandler,
   757	        optional: bool,
   758	        src_type: RType | None,
   759	        raise_exception: bool,
   760	    ) -> None:
   761	        """Emit cast to a union type.
   762
   763	        The arguments are similar to emit_cast.
   764	        """
   765	        if declare_dest:
   766	            self.emit_line(f"PyObject *{dest};")
   767	        good_label = self.new_label()
   768	        if optional:
   769	            self.emit_line(f"if ({src} == NULL) {{")
   770	            self.emit_line(f"{dest} = {self.c_error_value(typ)};")
   771	            self.emit_line(f"goto {good_label};")
   772	            self.emit_line("}")
   773	        for item in typ.items:
   774	            self.emit_cast(
   775	                src,
   776	                dest,
   777	                item,
   778	                declare_dest=False,
   779	                raise_exception=False,
   780	                optional=False,
   781	                likely=False,
   782	            )
   783	            self.emit_line(f"if ({dest} != NULL) goto {good_label};")
   784	        # Handle cast failure.
   785	        self.emit_cast_error_handler(error, src, dest, typ, raise_exception)
   786	        self.emit_label(good_label)
