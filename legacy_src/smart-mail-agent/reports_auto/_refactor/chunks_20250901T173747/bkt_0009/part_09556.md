    19	def aggregate_options(
    20	    manager: OptionManager,
    21	    cfg: configparser.RawConfigParser,
    22	    cfg_dir: str,
    23	    argv: Sequence[str] | None,
    24	) -> argparse.Namespace:
    25	    """Aggregate and merge CLI and config file options."""
    26	    # Get defaults from the option parser
    27	    default_values = manager.parse_args([])
    28
    29	    # Get the parsed config
    30	    parsed_config = config.parse_config(manager, cfg, cfg_dir)
    31
    32	    # store the plugin-set extended default ignore / select
    33	    default_values.extended_default_ignore = manager.extended_default_ignore
    34	    default_values.extended_default_select = manager.extended_default_select
    35
    36	    # Merge values parsed from config onto the default values returned
    37	    for config_name, value in parsed_config.items():
    38	        dest_name = config_name
    39	        # If the config name is somehow different from the destination name,
    40	        # fetch the destination name from our Option
    41	        if not hasattr(default_values, config_name):
    42	            dest_val = manager.config_options_dict[config_name].dest
    43	            assert isinstance(dest_val, str)
    44	            dest_name = dest_val
    45
    46	        LOG.debug(
    47	            'Overriding default value of (%s) for "%s" with (%s)',
    48	            getattr(default_values, dest_name, None),
    49	            dest_name,
    50	            value,
    51	        )
    52	        # Override the default values with the config values
    53	        setattr(default_values, dest_name, value)
    54
    55	    # Finally parse the command-line options
    56	    return manager.parse_args(argv, default_values)

   322	    def next_line(self) -> str:
   323	        """Get the next line from the list."""
   324	        if self.line_number >= self.total_lines:
   325	            return ""
   326	        line = self.lines[self.line_number]
   327	        self.line_number += 1
   328	        if self.indent_char is None and line[:1] in defaults.WHITESPACE:
   329	            self.indent_char = line[0]
   330	        return line

   333	        """Read the lines for this file checker."""
   334	        if self.filename == "-":

   336	            lines = self.read_lines_from_stdin()

   341	    def read_lines_from_filename(self) -> list[str]:
   342	        """Read the lines for a file."""
   343	        try:
   344	            with tokenize.open(self.filename) as fd:
   345	                return fd.readlines()
   346	        except (SyntaxError, UnicodeError):
   347	            # If we can't detect the codec with tokenize.detect_encoding, or
   348	            # the detected encoding is incorrect, just fallback to latin-1.
   349	            with open(self.filename, encoding="latin-1") as fd:
   350	                return fd.readlines()

   356	    def should_ignore_file(self) -> bool:
   357	        """Check if ``flake8: noqa`` is in the file to be ignored.
   358
   359	        :returns:
   360	            True if a line matches :attr:`defaults.NOQA_FILE`,
   361	            otherwise False
   362	        """
   363	        if not self.options.disable_noqa and any(
   364	            defaults.NOQA_FILE.match(line) for line in self.lines
   365	        ):
   366	            return True
   367	        elif any(defaults.NOQA_FILE.search(line) for line in self.lines):
   368	            LOG.warning(
   369	                "Detected `flake8: noqa` on line with code. To ignore an "
   370	                "error on a line use `noqa` instead."
   371	            )
   372	            return False
   373	        else:
   374	            return False

   377	        """Strip the UTF bom from the lines of the file."""
   378	        if not self.lines:
   379	            # If we have nothing to analyze quit early
   380	            return

   382	        # If the first byte of the file is a UTF-8 BOM, strip it
   383	        if self.lines[0][:1] == "\uFEFF":
   384	            self.lines[0] = self.lines[0][1:]
   385	        elif self.lines[0][:3] == "\xEF\xBB\xBF":
   386	            self.lines[0] = self.lines[0][3:]
