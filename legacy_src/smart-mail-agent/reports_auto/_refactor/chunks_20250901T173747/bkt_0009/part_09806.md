count):
   757	                unichar = self.read_ushort()
   758	                charToGlyph[unichar] = glyph
   759	                glyphToChar.setdefault(glyph,[]).append(unichar)
   760	        elif fmt==12:
   761	            segCount = self.read_ulong()
   762	            for n in range(segCount):
   763	                start = self.read_ulong()
   764	                end = self.read_ulong()
   765	                inc = self.read_ulong() - start
   766	                for unichar in range(start,end+1):
   767	                    glyph = unichar + inc
   768	                    charToGlyph[unichar] = glyph
   769	                    glyphToChar.setdefault(glyph,[]).append(unichar)
   770	        elif fmt==13:
   771	            segCount = self.read_ulong()
   772	            for n in range(segCount):
   773	                start = self.read_ulong()
   774	                end = self.read_ulong()
   775	                gid = self.read_ulong()
   776	                for unichar in range(start,end+1):
   777	                    charToGlyph[unichar] = gid
   778	                    glyphToChar.setdefault(gid,[]).append(unichar)
   779	        elif fmt==2:
   780	            T = [self.read_ushort() for i in range(256)]    #subheader keys
   781	            maxSHK = max(T)
   782	            SH = []
   783	            for i in range(maxSHK+1):
   784	                firstCode = self.read_ushort()
   785	                entryCount = self.read_ushort()
   786	                idDelta = self.read_ushort()
   787	                idRangeOffset = (self.read_ushort()-(maxSHK-i)*8-2)>>1
   788	                SH.append(CMapFmt2SubHeader(firstCode,entryCount,idDelta,idRangeOffset))
   789	            #number of glyph indexes to read. it is the length of the entire subtable minus that bit we've read so far
   790	            entryCount = (length-(self._pos-(cmap_offset+encoffs)))>>1
   791	            glyphs = [self.read_short() for i in range(entryCount)]
   792	            last = -1
   793	            for unichar in range(256):
   794	                if T[unichar]==0:
   795	                    #Special case, single byte encoding entry, look unichar up in subhead
   796	                    if last!=-1:
   797	                        glyph = 0
   798	                    elif (unichar<SH[0].firstCode or unichar>=SH[0].firstCode+SH[0].entryCount or
   799	                            SH[0].idRangeOffset+(unichar-SH[0].firstCode)>=entryCount):
   800	                        glyph = 0
   801	                    else:
   802	                        glyph = glyphs[SH[0].idRangeOffset+(unichar-SH[0].firstCode)]
   803	                        if glyph!=0:
   804	                            glyph += SH[0].idDelta
   805	                    #assume the single byte codes are ascii
   806	                    if glyph!=0 and glyph<self.numGlyphs:
   807	                        charToGlyph[unichar] = glyph
   808	                        glyphToChar.setdefault(glyph,[]).append(unichar)
   809	                else:
   810	                    k = T[unichar]
   811	                    for j in range(SH[k].entryCount):
   812	                        if SH[k].idRangeOffset+j>=entryCount:
   813	                            glyph = 0
   814	                        else:
   815	                            glyph = glyphs[SH[k].idRangeOffset+j]
   816	                            if glyph!= 0:
   817	                                glyph += SH[k].idDelta
   818	                        if glyph!=0 and glyph<self.numGlyphs:
   819	                            enc = (unichar<<8)|(j+SH[k].firstCode)
   820	                            charToGlyph[enc] = glyph
   821	                            glyphToChar.setdefault(glyph,[]).append(enc)
   822	                    if last==-1:
   823	                        last = unichar
   824	        else:
   825	            raise ValueError('Unsupported cmap encoding format %d' % fmt)
   826
   827	        # hmtx - Horizontal metrics table
   828	        # (needs data from hhea, maxp, and cmap tables)
   829	        self.seek_table("hmtx")
   830	        aw = None
   831	        self.charWidths = charWidths = {}
   832	        self.hmetrics = []
   833	        for glyph in range(numberOfHMetrics):
   834	            # advance width and left side bearing.  lsb is actually signed
   835	            # short, but we don't need it anyway (except for subsetting)
   836	            aw, lsb = self.read_ushort(), self.read_ushort()
   837	            self.hmetrics.append((aw, lsb))
   838	            aw = scale(aw)
   839	            if glyph == 0:
   840	                self.defaultWidth = aw
   841	            if glyph in glyphToChar:
   842	                for char in glyphToChar[glyph]:
   843	                    charWidths[char] = aw
   844	        for glyph in range(numberOfHMetrics, numGlyphs):
   845	            # the rest of the table only lists advance left side bearings.
   846	            # so we reuse aw set by the last iteration of the previous loop
   847	            lsb = self.read_ushort()
   848	            self.hmetrics.append((aw, lsb))
   849	            if glyph in glyphToChar:
   850	                for char in glyphToChar[glyph]:
   851	                    charWidths[char] = aw
   852
   853	        # loca - Index to location
   854	        if 'loca' not in self.table: raise TTFError('missing location table')
   855	        self.seek_table('loca')
   856	        self.glyphPos = []
   857	        if indexToLocFormat == 0:
   858	            for n in range(numGlyphs + 1):
   859	                self.glyphPos.append(self.read_ushort() << 1)
   860	        elif indexToLocFormat == 1:
   861	            for n in range(numGlyphs + 1):
   862	                self.glyphPos.append(self.read_ulong())
   863	        else:
   864	            raise TTFError('Unknown location table format (%d)' % indexToLocFormat)
   865	        if 0x20 in charToGlyph:
   866	            charToGlyph[0xa0] = charToGlyph[0x20]
   867	            charWidths[0xa0] = charWidths[0x20]
   868	        elif 0xa0 in charToGlyph:
   869	            charToGlyph[0x20] = charToGlyph[0xa0]
   870	            charWidths[0x20] = charWidths[0xa0]
   871

   874	    def makeSubset(self, subset):
   875	        """Create a subset of a TrueType font"""
   876	        output = TTFontMaker()
   877
   878	        # Build a mapping of glyphs in the subset to glyph numbers in
   879	        # the original font.  Also build a mapping of UCS codes to
   880	        # glyph values in the new font.
   881
   882	        # Start with 0 -> 0: "missing character"
   883	        glyphMap = [0]                  # new glyph index -> old glyph index
   884	        glyphSet = {0:0}                # old glyph index -> new glyph index
   885	        codeToGlyph = {}                # unicode -> new glyph index
   886	        for code in subset:
   887	            if code in self.charToGlyph:
   888	                originalGlyphIdx = self.charToGlyph[code]
   889	            else:
   890	                originalGlyphIdx = 0
   891	            if originalGlyphIdx not in glyphSet:
   892	                glyphSet[originalGlyphIdx] = len(glyphMap)
   893	                glyphMap.append(originalGlyphIdx)
   894	            codeToGlyph[code] = glyphSet[originalGlyphIdx]
   895
   896	        # Also include glyphs that are parts of composite glyphs
   897	        start = self.get_table_pos('glyf')[0]
   898	        n = 0
   899	        while n < len(glyphMap):
   900	            originalGlyphIdx = glyphMap[n]
   901	            glyphPos = self.glyphPos[originalGlyphIdx]
   902	            glyphLen = self.glyphPos[originalGlyphIdx + 1] - glyphPos
   903	            n += 1
   904	            if not glyphLen: continue
   905	            self.seek(start + glyphPos)
   906	            numberOfContours = self.read_short()
   907	            if numberOfContours < 0:
   908	                # composite glyph
   909	                self.skip(8)
   910	                flags = GF_MORE_COMPONENTS
   911	                while flags & GF_MORE_COMPONENTS:
   912	                    flags = self.read_ushort()
   913	                    glyphIdx = self.read_ushort()
   914	                    if glyphIdx not in glyphSet:
   915	                        glyphSet[glyphIdx] = len(glyphMap)
   916	                        glyphMap.append(glyphIdx)
   917	                    if flags & GF_ARG_1_AND_2_ARE_WORDS:
   918	                        self.skip(4)
   919	                    else:
   920	                        self.skip(2)
   921	                    if flags & GF_WE_HAVE_A_SCALE:
   922	                        self.skip(2)
   923	                    elif flags & GF_WE_HAVE_AN_X_AND_Y_SCALE:
   924	                        self.skip(4)
   925	                    elif flags & GF_WE_HAVE_A_TWO_BY_TWO:
   926	                        self.skip(8)
   927
   928
   929	        # The following tables are simply copied from the original
   930	        for tag in ('name', 'OS/2', 'cvt ', 'fpgm', 'prep'):
   931	            try:
   932	                output.add(tag, self.get_table(tag))
   933	            except KeyError:
   934	                # Apparently some of the tables are optional (cvt, fpgm, prep).
   935	                # The lack of the required ones (name, OS/2) would have already
   936	                # been caught before.
   937	                pass
   938
   939	        # post - PostScript
   940	        post = b"\x00\x03\x00\x00" + self.get_table('post')[4:16] + b"\x00" * 16
   941	        output.add('post', post)
   942
   943	        numGlyphs = len(glyphMap)
   944
   945	        # hmtx - Horizontal Metrics
   946	        hmtx = []
   947	        for n in range(numGlyphs):
   948	            aw, lsb = self.hmetrics[glyphMap[n]]
   949	            hmtx.append(int(aw))
   950	            hmtx.append(int(lsb))
   951
   952	        #work out n as 0 or first aw that's the start of a run
   953	        n = len(hmtx)-2
   954	        while n and hmtx[n]==hmtx[n-2]:
   955	            n -= 2
   956	        if not n: n = 2                 #need at least one pair
   957	        numberOfHMetrics = n>>1         #number of full H Metric pairs
   958	        hmtx = hmtx[:n] + hmtx[n+1::2]  #full pairs + all the trailing lsb's
   959
   960	        hmtx = pack(*([">%dH" % len(hmtx)] + hmtx))
   961	        output.add('hmtx', hmtx)
   962
   963	        # hhea - Horizontal Header
   964	        hhea = self.get_table('hhea')
   965	        hhea = _set_ushort(hhea, 34, numberOfHMetrics)
   966	        output.add('hhea', hhea)
   967
   968	        # maxp - Maximum Profile
   969	        maxp = self.get_table('maxp')
   970	        maxp = _set_ushort(maxp, 4, numGlyphs)
   971	        output.add('maxp', maxp)
   972
   973	        # cmap - Character to glyph mapping
   974	        # XXX maybe use format 0 if possible, not 6?
   975	        entryCount = len(subset)
   976	        length = 10 + entryCount * 2
   977	        cmap = [0, 1,           # version, number of tables
   978	                1, 0, 0,12,     # platform, encoding, offset (hi,lo)
   979	                6, length, 0,   # format, length, language
   980	                0,
   981	                entryCount] + \
   982	               list(map(codeToGlyph.get, subset))
   983	        cmap = pack(*([">%dH" % len(cmap)] + cmap))
   984	        output.add('cmap', cmap)
   985
   986	        # glyf - Glyph data
   987	        glyphData = self.get_table('glyf')
   988	        offsets = []
   989	        glyf = []
   990	        pos = 0
   991	        for n in range(numGlyphs):
   992	            offsets.append(pos)
   993	            originalGlyphIdx = glyphMap[n]
   994	            glyphPos = self.glyphPos[originalGlyphIdx]
   995	            glyphLen = self.glyphPos[originalGlyphIdx + 1] - glyphPos
   996	            data = glyphData[glyphPos:glyphPos+glyphLen]
   997	            # Fix references in composite glyphs
   998	            if glyphLen > 2 and unpack(">h", data[:2])[0] < 0:
   999	                # composite glyph
  1000	                pos_in_glyph = 10
  1001	                flags = GF_MORE_COMPONENTS
  1002	                while flags & GF_MORE_COMPONENTS:
  1003	                    flags = unpack(">H", data[pos_in_glyph:pos_in_glyph+2])[0]
  1004	                    glyphIdx = unpack(">H", data[pos_in_glyph+2:pos_in_glyph+4])[0]
  1005	                    data = _set_ushort(data, pos_in_glyph + 2, glyphSet[glyphIdx])
  1006	                    pos_in_glyph = pos_in_glyph + 4
  1007	                    if flags & GF_ARG_1_AND_2_ARE_WORDS:
  1008	                        pos_in_glyph = pos_in_glyph + 4
  1009	                    else:
  1010	                        pos_in_glyph = pos_in_glyph + 2
  1011	                    if flags & GF_WE_HAVE_A_SCALE:
  1012	                        pos_in_glyph = pos_in_glyph + 2
  1013	                    elif flags & GF_WE_HAVE_AN_X_AND_Y_SCALE:
  1014	                        pos_in_glyph = pos_in_glyph + 4
  1015	                    elif flags & GF_WE_HAVE_A_TWO_BY_TWO:
  1016	                        pos_in_glyph = pos_in_glyph + 8
  1017	            glyf.append(data)
  1018	            pos = pos + glyphLen
  1019	            if pos % 4 != 0:
  1020	                padding = 4 - pos % 4
  1021	                glyf.append(b'\0' * padding)
  1022	                pos = pos + padding
  1023	        offsets.append(pos)
  1024	        output.add('glyf', b''.join(glyf))
  1025
  1026	        # loca - Index to location
  1027	        loca = []
  1028	        if (pos + 1) >> 1 > 0xFFFF:
  1029	            indexToLocFormat = 1        # long format
  1030	            for offset in offsets:
  1031	                loca.append(offset)
  1032	            loca = pack(*([">%dL" % len(loca)] + loca))
  1033	        else:
  1034	            indexToLocFormat = 0        # short format
  1035	            for offset in offsets:
  1036	                loca.append(offset >> 1)
  1037	            loca = pack(*([">%dH" % len(loca)] + loca))
  1038	        output.add('loca', loca)
  1039
  1040	        # head - Font header
  1041	        head = self.get_table('head')
  1042	        head = _set_ushort(head, 50, indexToLocFormat)
  1043	        output.add('head', head)
  1044
  1045	        return output.makeStream()

  1201	    def splitString(self, text, doc, encoding='utf-8'):
  1202	        """Splits text into a number of chunks, each of which belongs to a
  1203	        single subset.  Returns a list of tuples (subset, string).  Use subset
  1204	        numbers with getSubsetInternalName.  Doc is needed for distinguishing
  1205	        subsets when building different documents at the same time."""
  1206	        asciiReadable = self._asciiReadable
  1207	        try: state = self.state[doc]
  1208	        except KeyError: state = self.state[doc] = TTFont.State(asciiReadable,self)
  1209	        curSet = -1
  1210	        cur = []
  1211	        results = []
  1212	        if not isUnicode(text):
  1213	            text = text.decode('utf-8')     # encoding defaults to utf-8
  1214	        assignments = state.assignments
  1215	        subsets = state.subsets
  1216	        reserveTTFNotdef = rl_config.reserveTTFNotdef
  1217	        for code in map(ord,text):
  1218	            if code==0xa0: code = 32    #map nbsp into space
  1219	            if code in assignments:
  1220	                n = assignments[code]
  1221	            else:
  1222	                if state.frozen:
  1223	                    raise pdfdoc.PDFError("Font %s is already frozen, cannot add new character U+%04X" % (self.fontName, code))
  1224	                n = state.nextCode
  1225	                if n&0xFF==32:
  1226	                    # make code 32 always be a space character
  1227	                    if n!=32: subsets[n >> 8].append(32)
  1228	                    state.nextCode += 1
  1229	                    n = state.nextCode
  1230	                if n>32:
  1231	                    if not(n&0xFF):
  1232	                        if reserveTTFNotdef:
  1233	                            subsets.append([0]) #force code 0 in as notdef
  1234	                            state.nextCode += 1
  1235	                            n = state.nextCode
  1236	                        else:
  1237	                            subsets.append([])
  1238	                    subsets[n >> 8].append(code)
  1239	                else:
  1240	                    subsets[0][n] = code
  1241	                state.nextCode += 1
  1242	                assignments[code] = n
  1243	                #subsets[n>>8].append(code)
  1244	            if (n >> 8) != curSet:
  1245	                if cur:
  1246	                    results.append((curSet,bytes(cur)))
  1247	                curSet = (n >> 8)
  1248	                cur = []
  1249	            cur.append(n & 0xFF)
  1250	        if cur:
  1251	            results.append((curSet,bytes(cur)))
  1252	        return results

    95	def _buildColorFunction(colors, positions):
    96	    from reportlab.pdfbase.pdfdoc import PDFExponentialFunction, PDFStitchingFunction
    97	    if positions is not None and len(positions) != len(colors):
    98	        raise ValueError("need to have the same number of colors and positions")
    99	    # simplified functions for edge cases
   100	    if len(colors) == 1:
   101	        # for completeness
   102	        return PDFExponentialFunction(N=1, C0=colors[0], C1=colors[0])
   103	    if len(colors) == 2:
   104	        if positions is None or (positions[0] == 0 and positions[1] == 1):
   105	            return PDFExponentialFunction(N=1, C0=colors[0], C1=colors[1])
   106	    # equally distribute if positions not specified
   107	    if positions is None:
   108	        nc = len(colors)
   109	        positions = [float(x)/(nc-1) for x in range(nc)]
   110	    else:
   111	        # sort positions and colors in increasing order
   112	        poscolors = list(zip(positions, colors))
   113	        poscolors.sort(key=lambda x: x[0])
   114	        # add endpoint positions if not already present
   115	        if poscolors[0][0] != 0:
   116	            poscolors.insert(0, (0.0, poscolors[0][1]))
   117	        if poscolors[-1][0] != 1:
   118	            poscolors.append((1.0, poscolors[-1][1]))
   119	        positions, colors = list(zip(*poscolors)) # unzip
   120	    # build stitching function
   121	    functions = []
   122	    bounds = [pos for pos in positions[1:-1]]
   123	    encode = []
   124	    lastcolor = colors[0]
   125	    for color in colors[1:]:
   126	        functions.append(PDFExponentialFunction(N=1, C0=lastcolor, C1=color))
   127	        lastcolor = color
   128	        encode.append(0.0)
   129	        encode.append(1.0)
   130	    return PDFStitchingFunction(functions, bounds, encode, Domain="[0.0 1.0]")

   626	    def showPage(self):
   627	        """Close the current page and possibly start on a new page."""
   628	        # ensure a space at the end of the stream - Acrobat does
   629	        # not mind, but Ghostscript dislikes 'Qendstream' even if
   630	        # the length marker finishes after 'Q'
   631
   632	        pageWidth = self._pagesize[0]
   633	        pageHeight = self._pagesize[1]
   634	        cM = self._cropMarks
   635	        code = self._code
   636	        if cM:
   637	            bw = max(0,getattr(cM,'borderWidth',36))
   638	            if bw:
   639	                markLast = getattr(cM,'markLast',1)
   640	                ml = min(bw,max(0,getattr(cM,'markLength',18)))
   641	                mw = getattr(cM,'markWidth',0.5)
   642	                mc = getattr(cM,'markColor',black)
   643	                mg = 2*bw-ml
   644	                cx0 = len(code)
   645	                if ml and mc:
   646	                    self.saveState()
   647	                    self.setStrokeColor(mc)
   648	                    self.setLineWidth(mw)
   649	                    self.lines([
   650	                        (bw,0,bw,ml),
   651	                        (pageWidth+bw,0,pageWidth+bw,ml),
   652	                        (bw,pageHeight+mg,bw,pageHeight+2*bw),
   653	                        (pageWidth+bw,pageHeight+mg,pageWidth+bw,pageHeight+2*bw),
   654	                        (0,bw,ml,bw),
   655	                        (pageWidth+mg,bw,pageWidth+2*bw,bw),
   656	                        (0,pageHeight+bw,ml,pageHeight+bw),
   657	                        (pageWidth+mg,pageHeight+bw,pageWidth+2*bw,pageHeight+bw),
   658	                        ])
   659	                    self.restoreState()
   660	                    if markLast:
   661	                        #if the marks are to be drawn after the content
   662	                        #save the code we just drew for later use
   663	                        L = code[cx0:]
   664	                        del code[cx0:]
   665	                        cx0 = len(code)
   666
   667	                bleedW = max(0,getattr(cM,'bleedWidth',0))
   668	                self.saveState()
   669	                self.translate(bw-bleedW,bw-bleedW)
   670	                if bleedW:
   671	                    #scale everything
   672	                    self.scale(1+(2.0*bleedW)/pageWidth,1+(2.0*bleedW)/pageHeight)

   674	                #move our translation/expansion code to the beginning
   675	                C = code[cx0:]
   676	                del code[cx0:]
   677	                code[0:0] = C
   678	                self.restoreState()
   679	                if markLast:
   680	                    code.extend(L)

   682	                pageHeight = 2*bw + pageHeight

   687	        page.pageheight = pageHeight

   689	        page.hasImages = self._currentPageHasImages
   690	        page.setPageTransition(self._pageTransition)
   691	        page.setCompression(self._pageCompression)

   702	        self._setExtGState(page)

   705	        self._setAnnotations(page)

   756	                      fit="Fit",
   757	                      left=None,
   758	                      top=None,
   759	                      bottom=None,
   760	                      right=None,
   761	                      zoom=None
   762	                      ):
   763	        """
   764	        This creates a bookmark to the current page which can
   765	        be referred to with the given key elsewhere.
   766
   767	        PDF offers very fine grained control over how Acrobat
   768	        reader is zoomed when people link to this. The default
   769	        is to keep the user's current zoom settings. the last
   770	        arguments may or may not be needed depending on the
   771	        choice of 'fitType'.
   772
   773	        Fit types and the other arguments they use are:
   774
   775	        - XYZ left top zoom - fine grained control.  null
   776	          or zero for any of the parameters means 'leave
   777	          as is', so "0,0,0" will keep the reader's settings.
   778	          NB. Adobe Reader appears to prefer "null" to 0's.
   779
   780	        - Fit - entire page fits in window
   781
   782	        - FitH top - top coord at top of window, width scaled
   783	          to fit.
   784
   785	        - FitV left - left coord at left of window, height
   786	          scaled to fit
   787
   788	        - FitR left bottom right top - scale window to fit
   789	          the specified rectangle
   790
   791	        (question: do we support /FitB, FitBH and /FitBV
   792	        which are hangovers from version 1.1 / Acrobat 3.0?)"""
   793	        dest = self._bookmarkReference(key)
   794	        self._doc.inPage() # try to enable page-only features
   795	        pageref = self._doc.thisPageRef()
   796
   797	        #None = "null" for PDF
   798	        if left is None:
   799	            left = "null"
   800	        if top is None:
   801	            top = "null"
   802	        if bottom is None:
   803	            bottom = "null"
   804	        if right is None:
   805	            right = "null"
   806	        if zoom is None:
   807	            zoom = "null"
   808
   809	        if fit == "XYZ":
   810	            dest.xyz(left,top,zoom)
   811	        elif fit == "Fit":
   812	            dest.fit()
   813	        elif fit == "FitH":
   814	            dest.fith(top)
   815	        elif fit == "FitV":
   816	            dest.fitv(left)
   817	        elif fit == "FitR":
   818	            dest.fitr(left,bottom,right,top)
   819	        #Do we need these (version 1.1 / Acrobat 3 versions)?
   820	        elif fit == "FitB":
   821	            dest.fitb()
   822	        elif fit == "FitBH":
   823	            dest.fitbh(top)
   824	        elif fit == "FitBV":
   825	            dest.fitbv(left)

   899	    def drawImage(self, image, x, y, width=None, height=None, mask=None,
   900	            preserveAspectRatio=False, anchor='c', anchorAtXY=False, showBoundary=False):
   901	        """Draws the image (ImageReader object or filename) as specified.
   902
   903	        "image" may be an image filename or an ImageReader object.
   904
   905	        x and y define the lower left corner of the image you wish to
   906	        draw (or of its bounding box, if using preserveAspectRation below).
   907
   908	        If width and height are not given, the width and height of the
   909	        image in pixels is used at a scale of 1 point to 1 pixel.
   910
   911	        If width and height are given, the image will be stretched to fill
   912	        the given rectangle bounded by (x, y, x+width, y-height).
   913
   914	        If you supply negative widths and/or heights, it inverts them and adjusts
   915	        x and y accordingly.
   916
   917	        The method returns the width and height of the underlying image, since
   918	        this is often useful for layout algorithms and saves you work if you have
   919	        not specified them yourself.
   920
   921	        The mask parameter supports transparent backgrounds. It takes 6 numbers
   922	        and defines the range of RGB values which will be masked out or treated
   923	        as transparent.  For example with [0,2,40,42,136,139], it will mask out
   924	        any pixels with a Red value from 0-2, Green from 40-42 and
   925	        Blue from 136-139  (on a scale of 0-255).
   926
   927	        New post version 2.0:  drawImage can center an image in a box you
   928	        provide, while preserving its aspect ratio.  For example, you might
   929	        have a fixed square box in your design, and a collection of photos
   930	        which might be landscape or portrait that you want to appear within
   931	        the box.  If preserveAspectRatio is true, your image will appear within
   932	        the box specified.
   933
   934
   935	        If preserveAspectRatio is True, the anchor property can be used to
   936	        specify how images should fit into the given box.  It should
   937	        be set to one of the following values, taken from the points of
   938	        the compass (plus 'c' for 'centre'):
   939
   940	                nw   n   ne
   941	                w    c    e
   942	                sw   s   se
   943
   944	        The default value is 'c' for 'centre'.  Thus, if you want your
   945	        bitmaps to always be centred and appear at the top of the given box,
   946	        set anchor='n'.      There are good examples of this in the output
   947	        of test_pdfgen_general.py
   948
   949	        Unlike drawInlineImage, this creates 'external images' which
   950	        are only stored once in the PDF file but can be drawn many times.
   951	        If you give it the same filename twice, even at different locations
   952	        and sizes, it will reuse the first occurrence, resulting in a saving
   953	        in file size and generation time.  If you use ImageReader objects,
   954	        it tests whether the image content has changed before deciding
   955	        whether to reuse it.

   958	        unless you have read the PDF Spec and understand the tradeoffs."""

   982	        # say it is an XObject.  Does it exist yet?
   983	        regName = self._doc.getXObjectName(name)
   984	        imgObj = self._doc.idToObject.get(regName, None)
   985	        if not imgObj:

   987	            imgObj = pdfdoc.PDFImageXObject(name, image, mask=mask)
   988	            imgObj.name = name

   991	            self._doc.addForm(name, imgObj)

  1333	    def resetTransforms(self):
  1334	        """I want to draw something (eg, string underlines) w.r.t. the default user space.
  1335	           Reset the matrix! This should be used usually as follows::
  1336
  1337	              canv.saveState()
  1338	              canv.resetTransforms()
  1339	              #...draw some stuff in default space coords...
  1340	              canv.restoreState() # go back!
  1341	        """
  1342	        # we have to adjoin the inverse, since reset is not a basic operation (without save/restore)
  1343	        (selfa, selfb, selfc, selfd, selfe, selff) = self._currentMatrix
  1344	        det = selfa*selfd - selfc*selfb
  1345	        resulta = selfd/det
  1346	        resultc = -selfc/det
  1347	        resulte = (selfc*selff - selfd*selfe)/det
  1348	        resultd = selfa/det
  1349	        resultb = -selfb/det
  1350	        resultf = (selfe*selfb - selff*selfa)/det
  1351	        self.transform(resulta, resultb, resultc, resultd, resulte, resultf)

  1355	           Not recommended for beginners."""

  1357	        if ENABLE_TRACKING:
  1358	            a0,b0,c0,d0,e0,f0 = self._currentMatrix
  1359	            self._currentMatrix = (a0*a+c0*b,    b0*a+d0*b,
  1360	                                   a0*c+c0*d,    b0*c+d0*d,
