  2527	        assert self.ancestors is not None

  2529	            all_deps = self.suppressed

  2594	def find_module_and_diagnose(
  2595	    manager: BuildManager,
  2596	    id: str,
  2597	    options: Options,
  2598	    caller_state: State | None = None,
  2599	    caller_line: int = 0,
  2600	    ancestor_for: State | None = None,
  2601	    root_source: bool = False,
  2602	    skip_diagnose: bool = False,
  2603	) -> tuple[str, str]:
  2604	    """Find a module by name, respecting follow_imports and producing diagnostics.
  2605
  2606	    If the module is not found, then the ModuleNotFound exception is raised.
  2607
  2608	    Args:
  2609	      id: module to find
  2610	      options: the options for the module being loaded
  2611	      caller_state: the state of the importing module, if applicable
  2612	      caller_line: the line number of the import
  2613	      ancestor_for: the child module this is an ancestor of, if applicable
  2614	      root_source: whether this source was specified on the command line
  2615	      skip_diagnose: skip any error diagnosis and reporting (but ModuleNotFound is
  2616	          still raised if the module is missing)
  2617
  2618	    The specified value of follow_imports for a module can be overridden
  2619	    if the module is specified on the command line or if it is a stub,
  2620	    so we compute and return the "effective" follow_imports of the module.
  2621
  2622	    Returns a tuple containing (file path, target's effective follow_imports setting)
  2623	    """
  2624	    result = find_module_with_reason(id, manager)
  2625	    if isinstance(result, str):
  2626	        # For non-stubs, look at options.follow_imports:
  2627	        # - normal (default) -> fully analyze
  2628	        # - silent -> analyze but silence errors
  2629	        # - skip -> don't analyze, make the type Any
  2630	        follow_imports = options.follow_imports
  2631	        if (
  2632	            root_source  # Honor top-level modules
  2633	            or (
  2634	                result.endswith(".pyi")  # Stubs are always normal
  2635	                and not options.follow_imports_for_stubs  # except when they aren't
  2636	            )
  2637	            or id in CORE_BUILTIN_MODULES  # core is always normal
  2638	        ):
  2639	            follow_imports = "normal"
  2640	        if skip_diagnose:
  2641	            pass
  2642	        elif follow_imports == "silent":
  2643	            # Still import it, but silence non-blocker errors.
  2644	            manager.log(f"Silencing {result} ({id})")
  2645	        elif follow_imports == "skip" or follow_imports == "error":
  2646	            # In 'error' mode, produce special error messages.
  2647	            if id not in manager.missing_modules:
  2648	                manager.log(f"Skipping {result} ({id})")
  2649	            if follow_imports == "error":
  2650	                if ancestor_for:
  2651	                    skipping_ancestor(manager, id, result, ancestor_for)
  2652	                else:
  2653	                    skipping_module(manager, caller_line, caller_state, id, result)
  2654	            raise ModuleNotFound
  2655	        if is_silent_import_module(manager, result) and not root_source:
  2656	            follow_imports = "silent"
  2657	        return (result, follow_imports)
  2658	    else:
  2659	        # Could not find a module.  Typically the reason is a
  2660	        # misspelled module name, missing stub, module not in
  2661	        # search path or the module has not been installed.
  2662
  2663	        ignore_missing_imports = options.ignore_missing_imports
  2664
  2665	        # Don't honor a global (not per-module) ignore_missing_imports
  2666	        # setting for modules that used to have bundled stubs, as
  2667	        # otherwise updating mypy can silently result in new false
  2668	        # negatives. (Unless there are stubs but they are incomplete.)
  2669	        global_ignore_missing_imports = manager.options.ignore_missing_imports
  2670	        if (
  2671	            is_module_from_legacy_bundled_package(id)
  2672	            and global_ignore_missing_imports
  2673	            and not options.ignore_missing_imports_per_module
  2674	            and result is ModuleNotFoundReason.APPROVED_STUBS_NOT_INSTALLED
  2675	        ):
  2676	            ignore_missing_imports = False
  2677
  2678	        if skip_diagnose:
  2679	            raise ModuleNotFound
  2680	        if caller_state:
  2681	            if not (ignore_missing_imports or in_partial_package(id, manager)):
  2682	                module_not_found(manager, caller_line, caller_state, id, result)
  2683	            raise ModuleNotFound
  2684	        elif root_source:
  2685	            # If we can't find a root source it's always fatal.
  2686	            # TODO: This might hide non-fatal errors from
  2687	            # root sources processed earlier.
  2688	            raise CompileError([f"mypy: can't find module '{id}'"])
  2689	        else:
  2690	            raise ModuleNotFound

  2735	def in_partial_package(id: str, manager: BuildManager) -> bool:
  2736	    """Check if a missing module can potentially be a part of a package.
  2737
  2738	    This checks if there is any existing parent __init__.pyi stub that
  2739	    defines a module-level __getattr__ (a.k.a. partial stub package).
  2740	    """
  2741	    while "." in id:
  2742	        parent, _ = id.rsplit(".", 1)
  2743	        if parent in manager.modules:
  2744	            parent_mod: MypyFile | None = manager.modules[parent]
  2745	        else:
  2746	            # Parent is not in build, try quickly if we can find it.
  2747	            try:
  2748	                parent_st = State(
  2749	                    id=parent, path=None, source=None, manager=manager, temporary=True
  2750	                )
  2751	            except (ModuleNotFound, CompileError):
  2752	                parent_mod = None
  2753	            else:
  2754	                parent_mod = parent_st.tree
  2755	        if parent_mod is not None:
  2756	            # Bail out soon, complete subpackage found
  2757	            return parent_mod.is_partial_stub_package
  2758	        id = parent
  2759	    return False

  2763	    manager: BuildManager,
  2764	    line: int,

  2882	def dispatch(sources: list[BuildSource], manager: BuildManager, stdout: TextIO) -> Graph:
  2883	    log_configuration(manager, sources)
  2884
  2885	    t0 = time.time()
  2886	    graph = load_graph(sources, manager)
  2887
  2888	    # This is a kind of unfortunate hack to work around some of fine-grained's
  2889	    # fragility: if we have loaded less than 50% of the specified files from
  2890	    # cache in fine-grained cache mode, load the graph again honestly.
  2891	    # In this case, we just turn the cache off entirely, so we don't need
  2892	    # to worry about some files being loaded and some from cache and so
  2893	    # that fine-grained mode never *writes* to the cache.
  2894	    if manager.use_fine_grained_cache() and len(graph) < 0.50 * len(sources):
  2895	        manager.log("Redoing load_graph without cache because too much was missing")
  2896	        manager.cache_enabled = False
  2897	        graph = load_graph(sources, manager)
  2898
  2899	    t1 = time.time()
  2900	    manager.add_stats(
  2901	        graph_size=len(graph),
  2902	        stubs_found=sum(g.path is not None and g.path.endswith(".pyi") for g in graph.values()),
  2903	        graph_load_time=(t1 - t0),
  2904	        fm_cache_size=len(manager.find_module_cache.results),
  2905	    )
  2906	    if not graph:
  2907	        print("Nothing to do?!", file=stdout)
  2908	        return graph
  2909	    manager.log(f"Loaded graph with {len(graph)} nodes ({t1 - t0:.3f} sec)")
  2910	    if manager.options.dump_graph:
  2911	        dump_graph(graph, stdout)
  2912	        return graph
  2913
  2914	    # Fine grained dependencies that didn't have an associated module in the build
  2915	    # are serialized separately, so we read them after we load the graph.
  2916	    # We need to read them both for running in daemon mode and if we are generating
  2917	    # a fine-grained cache (so that we can properly update them incrementally).
  2918	    # The `read_deps_cache` will also validate
  2919	    # the deps cache against the loaded individual cache files.
  2920	    if manager.options.cache_fine_grained or manager.use_fine_grained_cache():
  2921	        t2 = time.time()
  2922	        fg_deps_meta = read_deps_cache(manager, graph)
  2923	        manager.add_stats(load_fg_deps_time=time.time() - t2)
  2924	        if fg_deps_meta is not None:
  2925	            manager.fg_deps_meta = fg_deps_meta
  2926	        elif manager.stats.get("fresh_metas", 0) > 0:
  2927	            # Clear the stats so we don't infinite loop because of positive fresh_metas
  2928	            manager.stats.clear()
  2929	            # There were some cache files read, but no fine-grained dependencies loaded.
  2930	            manager.log("Error reading fine-grained dependencies cache -- aborting cache load")
  2931	            manager.cache_enabled = False
  2932	            manager.log("Falling back to full run -- reloading graph...")
  2933	            return dispatch(sources, manager, stdout)
  2934
  2935	    # If we are loading a fine-grained incremental mode cache, we
  2936	    # don't want to do a real incremental reprocess of the
  2937	    # graph---we'll handle it all later.
  2938	    if not manager.use_fine_grained_cache():
  2939	        process_graph(graph, manager)
  2940	        # Update plugins snapshot.
  2941	        write_plugins_snapshot(manager)
  2942	        manager.old_plugins_snapshot = manager.plugins_snapshot
  2943	        if manager.options.cache_fine_grained or manager.options.fine_grained_incremental:
  2944	            # If we are running a daemon or are going to write cache for further fine grained use,
  2945	            # then we need to collect fine grained protocol dependencies.
  2946	            # Since these are a global property of the program, they are calculated after we
  2947	            # processed the whole graph.
  2948	            type_state.add_all_protocol_deps(manager.fg_deps)
  2949	            if not manager.options.fine_grained_incremental:
  2950	                rdeps = generate_deps_for_cache(manager, graph)
  2951	                write_deps_cache(rdeps, manager, graph)
  2952
  2953	    if manager.options.dump_deps:
  2954	        # This speeds up startup a little when not using the daemon mode.
  2955	        from mypy.server.deps import dump_all_dependencies

  3205	def process_graph(graph: Graph, manager: BuildManager) -> None:
  3206	    """Process everything in dependency order."""
  3207	    sccs = sorted_components(graph)
  3208	    manager.log("Found %d SCCs; largest has %d nodes" % (len(sccs), max(len(scc) for scc in sccs)))
  3209
  3210	    fresh_scc_queue: list[list[str]] = []
  3211
  3212	    # We're processing SCCs from leaves (those without further
  3213	    # dependencies) to roots (those from which everything else can be
  3214	    # reached).
  3215	    for ascc in sccs:
  3216	        # Order the SCC's nodes using a heuristic.
  3217	        # Note that ascc is a set, and scc is a list.
  3218	        scc = order_ascc(graph, ascc)
  3219	        # Make the order of the SCC that includes 'builtins' and 'typing',
  3220	        # among other things, predictable. Various things may  break if
  3221	        # the order changes.
  3222	        if "builtins" in ascc:
  3223	            scc = sorted(scc, reverse=True)
  3224	            # If builtins is in the list, move it last.  (This is a bit of
  3225	            # a hack, but it's necessary because the builtins module is
  3226	            # part of a small cycle involving at least {builtins, abc,
  3227	            # typing}.  Of these, builtins must be processed last or else
  3228	            # some builtin objects will be incompletely processed.)
  3229	            scc.remove("builtins")
  3230	            scc.append("builtins")
  3231	        if manager.options.verbosity >= 2:
  3232	            for id in scc:
  3233	                manager.trace(
  3234	                    f"Priorities for {id}:",
  3235	                    " ".join(
  3236	                        "%s:%d" % (x, graph[id].priorities[x])
  3237	                        for x in graph[id].dependencies
  3238	                        if x in ascc and x in graph[id].priorities
  3239	                    ),
  3240	                )
  3241	        # Because the SCCs are presented in topological sort order, we
  3242	        # don't need to look at dependencies recursively for staleness
  3243	        # -- the immediate dependencies are sufficient.
  3244	        stale_scc = {id for id in scc if not graph[id].is_fresh()}
  3245	        fresh = not stale_scc
  3246	        deps = set()
  3247	        for id in scc:
  3248	            deps.update(graph[id].dependencies)
  3249	        deps -= ascc
  3250	        stale_deps = {id for id in deps if id in graph and not graph[id].is_interface_fresh()}
  3251	        fresh = fresh and not stale_deps
  3252	        undeps = set()
  3253	        if fresh:
  3254	            # Check if any dependencies that were suppressed according
  3255	            # to the cache have been added back in this run.
  3256	            # NOTE: Newly suppressed dependencies are handled by is_fresh().
  3257	            for id in scc:
  3258	                undeps.update(graph[id].suppressed)
  3259	            undeps &= graph.keys()
  3260	            if undeps:
  3261	                fresh = False
  3262	        if fresh:
  3263	            # All cache files are fresh.  Check that no dependency's
  3264	            # cache file is newer than any scc node's cache file.
  3265	            oldest_in_scc = min(graph[id].xmeta.data_mtime for id in scc)
  3266	            viable = {id for id in stale_deps if graph[id].meta is not None}
  3267	            newest_in_deps = (
  3268	                0 if not viable else max(graph[dep].xmeta.data_mtime for dep in viable)
  3269	            )
  3270	            if manager.options.verbosity >= 3:  # Dump all mtimes for extreme debugging.
  3271	                all_ids = sorted(ascc | viable, key=lambda id: graph[id].xmeta.data_mtime)
  3272	                for id in all_ids:
  3273	                    if id in scc:
  3274	                        if graph[id].xmeta.data_mtime < newest_in_deps:
  3275	                            key = "*id:"
  3276	                        else:
  3277	                            key = "id:"
  3278	                    else:
  3279	                        if graph[id].xmeta.data_mtime > oldest_in_scc:
  3280	                            key = "+dep:"
  3281	                        else:
  3282	                            key = "dep:"
  3283	                    manager.trace(" %5s %.0f %s" % (key, graph[id].xmeta.data_mtime, id))
  3284	            # If equal, give the benefit of the doubt, due to 1-sec time granularity
  3285	            # (on some platforms).
  3286	            if oldest_in_scc < newest_in_deps:
  3287	                fresh = False
  3288	                fresh_msg = f"out of date by {newest_in_deps - oldest_in_scc:.0f} seconds"
  3289	            else:
  3290	                fresh_msg = "fresh"
  3291	        elif undeps:
  3292	            fresh_msg = f"stale due to changed suppression ({' '.join(sorted(undeps))})"
  3293	        elif stale_scc:
  3294	            fresh_msg = "inherently stale"
  3295	            if stale_scc != ascc:
  3296	                fresh_msg += f" ({' '.join(sorted(stale_scc))})"
  3297	            if stale_deps:
  3298	                fresh_msg += f" with stale deps ({' '.join(sorted(stale_deps))})"
  3299	        else:
  3300	            fresh_msg = f"stale due to deps ({' '.join(sorted(stale_deps))})"
  3301
  3302	        # Initialize transitive_error for all SCC members from union
  3303	        # of transitive_error of dependencies.
  3304	        if any(graph[dep].transitive_error for dep in deps if dep in graph):
  3305	            for id in scc:
  3306	                graph[id].transitive_error = True
  3307
  3308	        scc_str = " ".join(scc)
  3309	        if fresh:
  3310	            manager.trace(f"Queuing {fresh_msg} SCC ({scc_str})")
  3311	            fresh_scc_queue.append(scc)
  3312	        else:
  3313	            if fresh_scc_queue:
  3314	                manager.log(f"Processing {len(fresh_scc_queue)} queued fresh SCCs")
  3315	                # Defer processing fresh SCCs until we actually run into a stale SCC
  3316	                # and need the earlier modules to be loaded.
  3317	                #
  3318	                # Note that `process_graph` may end with us not having processed every
  3319	                # single fresh SCC. This is intentional -- we don't need those modules
  3320	                # loaded if there are no more stale SCCs to be rechecked.
  3321	                #
  3322	                # Also note we shouldn't have to worry about transitive_error here,
  3323	                # since modules with transitive errors aren't written to the cache,
  3324	                # and if any dependencies were changed, this SCC would be stale.
  3325	                # (Also, in quick_and_dirty mode we don't care about transitive errors.)
  3326	                #
  3327	                # TODO: see if it's possible to determine if we need to process only a
  3328	                # _subset_ of the past SCCs instead of having to process them all.
  3329	                for prev_scc in fresh_scc_queue:
  3330	                    process_fresh_modules(graph, prev_scc, manager)
  3331	                fresh_scc_queue = []
  3332	            size = len(scc)
  3333	            if size == 1:
  3334	                manager.log(f"Processing SCC singleton ({scc_str}) as {fresh_msg}")
  3335	            else:
  3336	                manager.log("Processing SCC of size %d (%s) as %s" % (size, scc_str, fresh_msg))
  3337	            process_stale_scc(graph, scc, manager)
  3338
  3339	    sccs_left = len(fresh_scc_queue)

  3341	    manager.add_stats(sccs_left=sccs_left, nodes_left=nodes_left)
  3342	    if sccs_left:
  3343	        manager.log(
  3344	            "{} fresh SCCs ({} nodes) left in queue (and will remain unprocessed)".format(
  3345	                sccs_left, nodes_left
  3346	            )
  3347	        )
  3348	        manager.trace(str(fresh_scc_queue))
  3349	    else:
  3350	        manager.log("No fresh SCCs left in queue")

  3353	def order_ascc(graph: Graph, ascc: AbstractSet[str], pri_max: int = PRI_ALL) -> list[str]:
  3354	    """Come up with the ideal processing order within an SCC.
  3355
  3356	    Using the priorities assigned by all_imported_modules_in_file(),
  3357	    try to reduce the cycle to a DAG, by omitting arcs representing
  3358	    dependencies of lower priority.
  3359
  3360	    In the simplest case, if we have A <--> B where A has a top-level
  3361	    "import B" (medium priority) but B only has the reverse "import A"
  3362	    inside a function (low priority), we turn the cycle into a DAG by
  3363	    dropping the B --> A arc, which leaves only A --> B.
  3364
  3365	    If all arcs have the same priority, we fall back to sorting by
  3366	    reverse global order (the order in which modules were first
  3367	    encountered).
  3368
  3369	    The algorithm is recursive, as follows: when as arcs of different
  3370	    priorities are present, drop all arcs of the lowest priority,
  3371	    identify SCCs in the resulting graph, and apply the algorithm to
  3372	    each SCC thus found.  The recursion is bounded because at each
  3373	    recursion the spread in priorities is (at least) one less.
  3374
  3375	    In practice there are only a few priority levels (less than a
  3376	    dozen) and in the worst case we just carry out the same algorithm
  3377	    for finding SCCs N times.  Thus the complexity is no worse than
  3378	    the complexity of the original SCC-finding algorithm -- see
  3379	    strongly_connected_components() below for a reference.
  3380	    """
  3381	    if len(ascc) == 1:
  3382	        return list(ascc)
  3383	    pri_spread = set()
  3384	    for id in ascc:
  3385	        state = graph[id]
  3386	        for dep in state.dependencies:
  3387	            if dep in ascc:
  3388	                pri = state.priorities.get(dep, PRI_HIGH)
  3389	                if pri < pri_max:
  3390	                    pri_spread.add(pri)
  3391	    if len(pri_spread) == 1:
  3392	        # Filtered dependencies are uniform -- order by global order.
  3393	        return sorted(ascc, key=lambda id: -graph[id].order)
  3394	    pri_max = max(pri_spread)
  3395	    sccs = sorted_components(graph, ascc, pri_max)
  3396	    # The recursion is bounded by the len(pri_spread) check above.
  3397	    return [s for ss in sccs for s in order_ascc(graph, ss, pri_max)]

  3468	def sorted_components(
  3469	    graph: Graph, vertices: AbstractSet[str] | None = None, pri_max: int = PRI_ALL
  3470	) -> list[AbstractSet[str]]:
  3471	    """Return the graph's SCCs, topologically sorted by dependencies.
  3472
  3473	    The sort order is from leaves (nodes without dependencies) to
  3474	    roots (nodes on which no other nodes depend).
  3475
  3476	    This works for a subset of the full dependency graph too;
  3477	    dependencies that aren't present in graph.keys() are ignored.
  3478	    """
  3479	    # Compute SCCs.
  3480	    if vertices is None:
  3481	        vertices = set(graph)
  3482	    edges = {id: deps_filtered(graph, vertices, id, pri_max) for id in vertices}
  3483	    sccs = list(strongly_connected_components(vertices, edges))
  3484	    # Topsort.
  3485	    res = []
  3486	    for ready in topsort(prepare_sccs(sccs, edges)):
  3487	        # Sort the sets in ready by reversed smallest State.order.  Examples:
  3488	        #
  3489	        # - If ready is [{x}, {y}], x.order == 1, y.order == 2, we get
  3490	        #   [{y}, {x}].
  3491	        #
  3492	        # - If ready is [{a, b}, {c, d}], a.order == 1, b.order == 3,
  3493	        #   c.order == 2, d.order == 4, the sort keys become [1, 2]
  3494	        #   and the result is [{c, d}, {a, b}].
  3495	        res.extend(sorted(ready, key=lambda scc: -min(graph[id].order for id in scc)))
  3496	    return res

   460	        Deferred functions will be processed by check_second_pass().
   461	        """
   462	        self.recurse_into_functions = True
   463	        with state.strict_optional_set(self.options.strict_optional), checker_state.set(self):
   464	            self.errors.set_file(
   465	                self.path, self.tree.fullname, scope=self.tscope, options=self.options
   466	            )
   467	            with self.tscope.module_scope(self.tree.fullname):
   468	                with self.enter_partial_types(), self.binder.top_frame_context():
   469	                    for d in self.tree.defs:
   470	                        if self.binder.is_unreachable():
   471	                            if not self.should_report_unreachable_issues():
   472	                                break
   473	                            if not self.is_noop_for_reachability(d):
   474	                                self.msg.unreachable_statement(d)
   475	                                break
   476	                        else:
   477	                            self.accept(d)
   478
   479	                assert not self.current_node_deferred
   480
   481	                all_ = self.globals.get("__all__")
   482	                if all_ is not None and all_.type is not None:
   483	                    all_node = all_.node
   484	                    assert all_node is not None
   485	                    seq_str = self.named_generic_type(
   486	                        "typing.Sequence", [self.named_type("builtins.str")]
   487	                    )
   488	                    if not is_subtype(all_.type, seq_str):
   489	                        str_seq_s, all_s = format_type_distinctly(
   490	                            seq_str, all_.type, options=self.options
   491	                        )
   492	                        self.fail(
   493	                            message_registry.ALL_MUST_BE_SEQ_STR.format(str_seq_s, all_s), all_node
   494	                        )
   495
   496	    def check_second_pass(
   497	        self, todo: Sequence[DeferredNode | FineGrainedDeferredNode] | None = None
   498	    ) -> bool:
   499	        """Run second or following pass of type checking.
   500
   501	        This goes through deferred nodes, returning True if there were any.
   502	        """
   503	        self.recurse_into_functions = True
   504	        with state.strict_optional_set(self.options.strict_optional), checker_state.set(self):
   505	            if not todo and not self.deferred_nodes:
   506	                return False
   507	            self.errors.set_file(
   508	                self.path, self.tree.fullname, scope=self.tscope, options=self.options
   509	            )
   510	            with self.tscope.module_scope(self.tree.fullname):
   511	                self.pass_num += 1
   512	                if not todo:
   513	                    todo = self.deferred_nodes
   514	                else:
   515	                    assert not self.deferred_nodes
   516	                self.deferred_nodes = []
   517	                done: set[DeferredNodeType | FineGrainedDeferredNodeType] = set()
   518	                for node, active_typeinfo in todo:
   519	                    if node in done:
   520	                        continue
   521	                    # This is useful for debugging:
   522	                    # print("XXX in pass %d, class %s, function %s" %
   523	                    #       (self.pass_num, type_name, node.fullname or node.name))
   524	                    done.add(node)
   525	                    with ExitStack() as stack:
   526	                        if active_typeinfo:
   527	                            stack.enter_context(self.tscope.class_scope(active_typeinfo))
   528	                            stack.enter_context(self.scope.push_class(active_typeinfo))
   529	                        self.check_partial(node)
   530	            return True
   531
   532	    def check_partial(self, node: DeferredNodeType | FineGrainedDeferredNodeType) -> None:
   533	        self.widened_vars = []
   534	        if isinstance(node, MypyFile):
   535	            self.check_top_level(node)
   536	        else:
   537	            self.recurse_into_functions = True
