  8254	def expand_callable_variants(c: CallableType) -> list[CallableType]:
  8255	    """Expand a generic callable using all combinations of type variables' values/bounds."""
  8256	    for tv in c.variables:
  8257	        # We need to expand self-type before other variables, because this is the only
  8258	        # type variable that can have other type variables in the upper bound.
  8259	        if tv.id.is_self():
  8260	            c = expand_type(c, {tv.id: tv.upper_bound}).copy_modified(
  8261	                variables=[v for v in c.variables if not v.id.is_self()]
  8262	            )
  8263	            break
  8264
  8265	    if not c.is_generic():
  8266	        # Fast path.
  8267	        return [c]
  8268
  8269	    tvar_values = []
  8270	    for tvar in c.variables:
  8271	        if isinstance(tvar, TypeVarType) and tvar.values:
  8272	            tvar_values.append(tvar.values)
  8273	        else:
  8274	            tvar_values.append([tvar.upper_bound])
  8275
  8276	    variants = []
  8277	    for combination in itertools.product(*tvar_values):
  8278	        tvar_map = {tv.id: subst for (tv, subst) in zip(c.variables, combination)}
  8279	        variants.append(expand_type(c, tvar_map).copy_modified(variables=[]))
  8280	    return variants

  8283	def is_unsafe_overlapping_overload_signatures(
  8284	    signature: CallableType,
  8285	    other: CallableType,
  8286	    class_type_vars: list[TypeVarLikeType],
  8287	    partial_only: bool = True,
  8288	) -> bool:
  8289	    """Check if two overloaded signatures are unsafely overlapping or partially overlapping.
  8290
  8291	    We consider two functions 's' and 't' to be unsafely overlapping if three
  8292	    conditions hold:
  8293
  8294	    1.  s's parameters are partially overlapping with t's. i.e. there are calls that are
  8295	        valid for both signatures.
  8296	    2.  for these common calls, some of t's parameters types are wider that s's.
  8297	    3.  s's return type is NOT a subset of t's.
  8298
  8299	    Note that we use subset rather than subtype relationship in these checks because:
  8300	    * Overload selection happens at runtime, not statically.
  8301	    * This results in more lenient behavior.
  8302	    This can cause false negatives (e.g. if overloaded function returns an externally
  8303	    visible attribute with invariant type), but such situations are rare. In general,
  8304	    overloads in Python are generally unsafe, so we intentionally try to avoid giving
  8305	    non-actionable errors (see more details in comments below).
  8306
  8307	    Assumes that 'signature' appears earlier in the list of overload
  8308	    alternatives then 'other' and that their argument counts are overlapping.
  8309	    """
  8310	    # Try detaching callables from the containing class so that all TypeVars
  8311	    # are treated as being free, i.e. the signature is as seen from inside the class,
  8312	    # where "self" is not yet bound to anything.
  8313	    signature = detach_callable(signature, class_type_vars)
  8314	    other = detach_callable(other, class_type_vars)
  8315
  8316	    # Note: We repeat this check twice in both directions compensate for slight
  8317	    # asymmetries in 'is_callable_compatible'.
  8318
  8319	    for sig_variant in expand_callable_variants(signature):
  8320	        for other_variant in expand_callable_variants(other):
  8321	            # Using only expanded callables may cause false negatives, we can add
  8322	            # more variants (e.g. using inference between callables) in the future.
  8323	            if is_subset_no_promote(sig_variant.ret_type, other_variant.ret_type):
  8324	                continue
  8325	            if not (
  8326	                is_callable_compatible(
  8327	                    sig_variant,
  8328	                    other_variant,
  8329	                    is_compat=is_overlapping_types_for_overload,
  8330	                    check_args_covariantly=False,
  8331	                    is_proper_subtype=False,
  8332	                    is_compat_return=lambda l, r: not is_subset_no_promote(l, r),
  8333	                    allow_partial_overlap=True,
  8334	                )
  8335	                or is_callable_compatible(
  8336	                    other_variant,
  8337	                    sig_variant,
  8338	                    is_compat=is_overlapping_types_for_overload,
  8339	                    check_args_covariantly=True,
  8340	                    is_proper_subtype=False,
  8341	                    is_compat_return=lambda l, r: not is_subset_no_promote(r, l),
  8342	                    allow_partial_overlap=True,
  8343	                )
  8344	            ):
  8345	                continue
  8346	            # Using the same `allow_partial_overlap` flag as before, can cause false
  8347	            # negatives in case where star argument is used in a catch-all fallback overload.
  8348	            # But again, practicality beats purity here.
  8349	            if not partial_only or not is_callable_compatible(
  8350	                other_variant,
  8351	                sig_variant,
  8352	                is_compat=is_subset_no_promote,
  8353	                check_args_covariantly=True,
  8354	                is_proper_subtype=False,
  8355	                ignore_return=True,
  8356	                allow_partial_overlap=True,
  8357	            ):
  8358	                return True
  8359	    return False

  8539	class DisjointDict(Generic[TKey, TValue]):
  8540	    """An variation of the union-find algorithm/data structure where instead of keeping
  8541	    track of just disjoint sets, we keep track of disjoint dicts -- keep track of multiple
  8542	    Set[Key] -> Set[Value] mappings, where each mapping's keys are guaranteed to be disjoint.
  8543
  8544	    This data structure is currently used exclusively by 'group_comparison_operands' below
  8545	    to merge chains of '==' and 'is' comparisons when two or more chains use the same expression
  8546	    in best-case O(n), where n is the number of operands.
  8547
  8548	    Specifically, the `add_mapping()` function and `items()` functions will take on average
  8549	    O(k + v) and O(n) respectively, where k and v are the number of keys and values we're adding
  8550	    for a given chain. Note that k <= n and v <= n.
  8551
  8552	    We hit these average/best-case scenarios for most user code: e.g. when the user has just
  8553	    a single chain like 'a == b == c == d == ...' or multiple disjoint chains like
  8554	    'a==b < c==d < e==f < ...'. (Note that a naive iterative merging would be O(n^2) for
  8555	    the latter case).
  8556
  8557	    In comparison, this data structure will make 'group_comparison_operands' have a worst-case
  8558	    runtime of O(n*log(n)): 'add_mapping()' and 'items()' are worst-case O(k*log(n) + v) and
  8559	    O(k*log(n)) respectively. This happens only in the rare case where the user keeps repeatedly
  8560	    making disjoint mappings before merging them in a way that persistently dodges the path
  8561	    compression optimization in '_lookup_root_id', which would end up constructing a single
  8562	    tree of height log_2(n). This makes root lookups no longer amoritized constant time when we
  8563	    finally call 'items()'.
  8564	    """
  8565
  8566	    def __init__(self) -> None:
  8567	        # Each key maps to a unique ID
  8568	        self._key_to_id: dict[TKey, int] = {}
  8569
  8570	        # Each id points to the parent id, forming a forest of upwards-pointing trees. If the
  8571	        # current id already is the root, it points to itself. We gradually flatten these trees
  8572	        # as we perform root lookups: eventually all nodes point directly to its root.
  8573	        self._id_to_parent_id: dict[int, int] = {}
  8574
  8575	        # Each root id in turn maps to the set of values.
  8576	        self._root_id_to_values: dict[int, set[TValue]] = {}
  8577
  8578	    def add_mapping(self, keys: set[TKey], values: set[TValue]) -> None:
  8579	        """Adds a 'Set[TKey] -> Set[TValue]' mapping. If there already exists a mapping
  8580	        containing one or more of the given keys, we merge the input mapping with the old one.
  8581
  8582	        Note that the given set of keys must be non-empty -- otherwise, nothing happens.
  8583	        """
  8584	        if not keys:
  8585	            return
  8586
  8587	        subtree_roots = [self._lookup_or_make_root_id(key) for key in keys]
  8588	        new_root = subtree_roots[0]
  8589
  8590	        root_values = self._root_id_to_values[new_root]
  8591	        root_values.update(values)
  8592	        for subtree_root in subtree_roots[1:]:
  8593	            if subtree_root == new_root or subtree_root not in self._root_id_to_values:
  8594	                continue
  8595	            self._id_to_parent_id[subtree_root] = new_root
  8596	            root_values.update(self._root_id_to_values.pop(subtree_root))
  8597
  8598	    def items(self) -> list[tuple[set[TKey], set[TValue]]]:
  8599	        """Returns all disjoint mappings in key-value pairs."""
  8600	        root_id_to_keys: dict[int, set[TKey]] = {}
  8601	        for key in self._key_to_id:
  8602	            root_id = self._lookup_root_id(key)
  8603	            if root_id not in root_id_to_keys:
  8604	                root_id_to_keys[root_id] = set()
  8605	            root_id_to_keys[root_id].add(key)
  8606
  8607	        output = []
  8608	        for root_id, keys in root_id_to_keys.items():
  8609	            output.append((keys, self._root_id_to_values[root_id]))
  8610
  8611	        return output
  8612
  8613	    def _lookup_or_make_root_id(self, key: TKey) -> int:
  8614	        if key in self._key_to_id:
  8615	            return self._lookup_root_id(key)
  8616	        else:
  8617	            new_id = len(self._key_to_id)
  8618	            self._key_to_id[key] = new_id
  8619	            self._id_to_parent_id[new_id] = new_id
  8620	            self._root_id_to_values[new_id] = set()
  8621	            return new_id
  8622
  8623	    def _lookup_root_id(self, key: TKey) -> int:
  8624	        i = self._key_to_id[key]
  8625	        while i != self._id_to_parent_id[i]:
  8626	            # Optimization: make keys directly point to their grandparents to speed up
  8627	            # future traversals. This prevents degenerate trees of height n from forming.
  8628	            new_parent = self._id_to_parent_id[self._id_to_parent_id[i]]
  8629	            self._id_to_parent_id[i] = new_parent
  8630	            i = new_parent
  8631	        return i

  8634	def group_comparison_operands(
  8635	    pairwise_comparisons: Iterable[tuple[str, Expression, Expression]],
  8636	    operand_to_literal_hash: Mapping[int, Key],
  8637	    operators_to_group: set[str],
  8638	) -> list[tuple[str, list[int]]]:
  8639	    """Group a series of comparison operands together chained by any operand
  8640	    in the 'operators_to_group' set. All other pairwise operands are kept in
  8641	    groups of size 2.
  8642
  8643	    For example, suppose we have the input comparison expression:
  8644
  8645	        x0 == x1 == x2 < x3 < x4 is x5 is x6 is not x7 is not x8
  8646
  8647	    If we get these expressions in a pairwise way (e.g. by calling ComparisonExpr's
  8648	    'pairwise()' method), we get the following as input:
  8649
  8650	        [('==', x0, x1), ('==', x1, x2), ('<', x2, x3), ('<', x3, x4),
  8651	         ('is', x4, x5), ('is', x5, x6), ('is not', x6, x7), ('is not', x7, x8)]
  8652
  8653	    If `operators_to_group` is the set {'==', 'is'}, this function will produce
  8654	    the following "simplified operator list":
  8655
  8656	       [("==", [0, 1, 2]), ("<", [2, 3]), ("<", [3, 4]),
  8657	        ("is", [4, 5, 6]), ("is not", [6, 7]), ("is not", [7, 8])]
  8658
  8659	    Note that (a) we yield *indices* to the operands rather then the operand
  8660	    expressions themselves and that (b) operands used in a consecutive chain
  8661	    of '==' or 'is' are grouped together.
  8662
  8663	    If two of these chains happen to contain operands with the same underlying
  8664	    literal hash (e.g. are assignable and correspond to the same expression),
  8665	    we combine those chains together. For example, if we had:
  8666
  8667	        same == x < y == same
  8668
  8669	    ...and if 'operand_to_literal_hash' contained the same values for the indices
  8670	    0 and 3, we'd produce the following output:
  8671
  8672	        [("==", [0, 1, 2, 3]), ("<", [1, 2])]
  8673
  8674	    But if the 'operand_to_literal_hash' did *not* contain an entry, we'd instead
  8675	    default to returning:
  8676
  8677	        [("==", [0, 1]), ("<", [1, 2]), ("==", [2, 3])]
  8678
  8679	    This function is currently only used to assist with type-narrowing refinements
  8680	    and is extracted out to a helper function so we can unit test it.
  8681	    """
  8682	    groups: dict[str, DisjointDict[Key, int]] = {op: DisjointDict() for op in operators_to_group}
  8683
  8684	    simplified_operator_list: list[tuple[str, list[int]]] = []
  8685	    last_operator: str | None = None
  8686	    current_indices: set[int] = set()
  8687	    current_hashes: set[Key] = set()
  8688	    for i, (operator, left_expr, right_expr) in enumerate(pairwise_comparisons):
  8689	        if last_operator is None:
  8690	            last_operator = operator
  8691
  8692	        if current_indices and (operator != last_operator or operator not in operators_to_group):
  8693	            # If some of the operands in the chain are assignable, defer adding it: we might
  8694	            # end up needing to merge it with other chains that appear later.
  8695	            if not current_hashes:
  8696	                simplified_operator_list.append((last_operator, sorted(current_indices)))
  8697	            else:
  8698	                groups[last_operator].add_mapping(current_hashes, current_indices)
  8699	            last_operator = operator
  8700	            current_indices = set()
  8701	            current_hashes = set()
  8702
  8703	        # Note: 'i' corresponds to the left operand index, so 'i + 1' is the
  8704	        # right operand.
  8705	        current_indices.add(i)
  8706	        current_indices.add(i + 1)
  8707

  8954	    """Do types have IntEnum/StrEnum types that are potentially overlapping with other types?
  8955
  8956	    If True, we shouldn't attempt type narrowing based on enum values, as it gets
  8957	    too ambiguous.
  8958
  8959	    For example, return True if there's an 'int' type together with an IntEnum literal.
  8960	    However, IntEnum together with a literal of the same IntEnum type is not ambiguous.
  8961	    """

  8963	    #  (1) an IntEnum or StrEnum type
  8964	    #  (2) either a different IntEnum/StrEnum type or a non-enum type ("<other>")
  8965	    #
  8966	    # It would be slightly more correct to calculate this separately for IntEnum and
  8967	    # StrEnum related types, as an IntEnum can't be confused with a StrEnum.
  8968	    return len(_ambiguous_enum_variants(types)) > 1

   252	def extract_refexpr_names(expr: RefExpr) -> set[str]:
   253	    """Recursively extracts all module references from a reference expression.
   254
   255	    Note that currently, the only two subclasses of RefExpr are NameExpr and
   256	    MemberExpr."""
   257	    output: set[str] = set()
   258	    while isinstance(expr.node, MypyFile) or expr.fullname:
   259	        if isinstance(expr.node, MypyFile) and expr.fullname:
   260	            # If it's None, something's wrong (perhaps due to an
   261	            # import cycle or a suppressed error).  For now we just
   262	            # skip it.
   263	            output.add(expr.fullname)
   264
   265	        if isinstance(expr, NameExpr):
   266	            is_suppressed_import = isinstance(expr.node, Var) and expr.node.is_suppressed_import
   267	            if isinstance(expr.node, TypeInfo):
   268	                # Reference to a class or a nested class
   269	                output.update(split_module_names(expr.node.module_name))
   270	            elif "." in expr.fullname and not is_suppressed_import:
   271	                # Everything else (that is not a silenced import within a class)
   272	                output.add(expr.fullname.rsplit(".", 1)[0])
   273	            break
   274	        elif isinstance(expr, MemberExpr):
   275	            if isinstance(expr.expr, RefExpr):
   276	                expr = expr.expr
   277	            else:
   278	                break
   279	        else:
   280	            raise AssertionError(f"Unknown RefExpr subclass: {type(expr)}")
   281	    return output

   494	                immutable.add(name)

   496	                module_attrs[name] = AnyType(TypeOfAny.from_error)
   497	            else:
   498	                # TODO: what to do about nested module references?
   499	                # They are non-trivial because there may be import cycles.
   500	                module_attrs[name] = self.analyze_static_reference(
   501	                    n.node, n.node, False, include_modules=False, suppress_errors=True
   502	                )
   503	        result.extra_attrs = ExtraAttrs(module_attrs, immutable, node.fullname)
   504	        return result

   506	    def visit_call_expr(self, e: CallExpr, allow_none_return: bool = False) -> Type:

   511	                # on the typeshed stubs to type check the arguments.
   512	                self.visit_call_expr_inner(e)

   514	            return self.accept(e.analyzed, self.type_context[-1])
   515	        return self.visit_call_expr_inner(e, allow_none_return=allow_none_return)

   521	            # Direct reference.

   524	            get_proper_type(base.node.target), TypedDictType

   527	    def visit_call_expr_inner(self, e: CallExpr, allow_none_return: bool = False) -> Type:
   528	        if (
   529	            self.refers_to_typeddict(e.callee)
   530	            or isinstance(e.callee, IndexExpr)
   531	            and self.refers_to_typeddict(e.callee.base)
   532	        ):
   533	            typeddict_callable = get_proper_type(self.accept(e.callee, is_callee=True))
   534	            if isinstance(typeddict_callable, CallableType):
   535	                typeddict_type = get_proper_type(typeddict_callable.ret_type)
   536	                assert isinstance(typeddict_type, TypedDictType)
   537	                return self.check_typeddict_call(
   538	                    typeddict_type, e.arg_kinds, e.arg_names, e.args, e, typeddict_callable
   539	                )
   540	        if (
   541	            isinstance(e.callee, NameExpr)
   542	            and e.callee.name in ("isinstance", "issubclass")
   543	            and len(e.args) == 2
   544	        ):
   545	            for typ in mypy.checker.flatten(e.args[1]):
   546	                node = None
   547	                if isinstance(typ, NameExpr):
   548	                    try:
   549	                        node = self.chk.lookup_qualified(typ.name)
   550	                    except KeyError:
   551	                        # Undefined names should already be reported in semantic analysis.
   552	                        pass
   553	                if is_expr_literal_type(typ):
   554	                    self.msg.cannot_use_function_with_type(e.callee.name, "Literal", e)
   555	                    continue
   556	                if node and isinstance(node.node, TypeAlias):
   557	                    target = get_proper_type(node.node.target)
   558	                    if isinstance(target, AnyType):
   559	                        self.msg.cannot_use_function_with_type(e.callee.name, "Any", e)
   560	                        continue
   561	                    if isinstance(target, NoneType):
   562	                        continue
   563	                if (
   564	                    isinstance(typ, IndexExpr)
   565	                    and isinstance(typ.analyzed, (TypeApplication, TypeAliasExpr))
   566	                ) or (
   567	                    isinstance(typ, NameExpr)
   568	                    and node
   569	                    and isinstance(node.node, TypeAlias)
   570	                    and not node.node.no_args
   571	                    and not (
   572	                        isinstance(union_target := get_proper_type(node.node.target), UnionType)
   573	                        and union_target.uses_pep604_syntax
   574	                    )
   575	                ):
   576	                    self.msg.type_arguments_not_allowed(e)
   577	                if isinstance(typ, RefExpr) and isinstance(typ.node, TypeInfo):
   578	                    if typ.node.typeddict_type:
   579	                        self.msg.cannot_use_function_with_type(e.callee.name, "TypedDict", e)
   580	                    elif typ.node.is_newtype:
   581	                        self.msg.cannot_use_function_with_type(e.callee.name, "NewType", e)
   582	        self.try_infer_partial_type(e)
   583	        type_context = None
   584	        if isinstance(e.callee, LambdaExpr):
   585	            formal_to_actual = map_actuals_to_formals(
   586	                e.arg_kinds,
   587	                e.arg_names,
   588	                e.callee.arg_kinds,
   589	                e.callee.arg_names,
   590	                lambda i: self.accept(e.args[i]),
   591	            )
   592
   593	            arg_types = [
   594	                join.join_type_list([self.accept(e.args[j]) for j in formal_to_actual[i]])
   595	                for i in range(len(e.callee.arg_kinds))
   596	            ]
   597	            type_context = CallableType(
   598	                arg_types,
   599	                e.callee.arg_kinds,
   600	                e.callee.arg_names,
   601	                ret_type=self.object_type(),
   602	                fallback=self.named_type("builtins.function"),
   603	            )
   604	        callee_type = get_proper_type(
   605	            self.accept(e.callee, type_context, always_allow_any=True, is_callee=True)
   606	        )
   607
   608	        # Figure out the full name of the callee for plugin lookup.
   609	        object_type = None
   610	        member = None
   611	        fullname = None
   612	        if isinstance(e.callee, RefExpr):
   613	            # There are two special cases where plugins might act:
   614	            # * A "static" reference/alias to a class or function;
   615	            #   get_function_hook() will be invoked for these.
   616	            fullname = e.callee.fullname or None
   617	            if isinstance(e.callee.node, TypeAlias):
   618	                target = get_proper_type(e.callee.node.target)
   619	                if isinstance(target, Instance):
   620	                    fullname = target.type.fullname
   621	            # * Call to a method on object that has a full name (see
   622	            #   method_fullname() for details on supported objects);
   623	            #   get_method_hook() and get_method_signature_hook() will
   624	            #   be invoked for these.
   625	            if (
   626	                not fullname
   627	                and isinstance(e.callee, MemberExpr)
   628	                and self.chk.has_type(e.callee.expr)
   629	            ):
   630	                member = e.callee.name
   631	                object_type = self.chk.lookup_type(e.callee.expr)
   632
   633	        if (
   634	            self.chk.options.disallow_untyped_calls
   635	            and self.chk.in_checked_function()
   636	            and isinstance(callee_type, CallableType)
   637	            and callee_type.implicit
   638	            and callee_type.name != LAMBDA_NAME
   639	        ):
   640	            if fullname is None and member is not None:
   641	                assert object_type is not None
   642	                fullname = self.method_fullname(object_type, member)
   643	            if not fullname or not any(
   644	                fullname == p or fullname.startswith(f"{p}.")
   645	                for p in self.chk.options.untyped_calls_exclude
   646	            ):
   647	                self.msg.untyped_function_call(callee_type, e)
   648
   649	        ret_type = self.check_call_expr_with_callee_type(
   650	            callee_type, e, fullname, object_type, member
   651	        )
   652	        if isinstance(e.callee, RefExpr) and len(e.args) == 2:
   653	            if e.callee.fullname in ("builtins.isinstance", "builtins.issubclass"):
   654	                self.check_runtime_protocol_test(e)
   655	            if e.callee.fullname == "builtins.issubclass":
   656	                self.check_protocol_issubclass(e)
   657	        if isinstance(e.callee, MemberExpr) and e.callee.name == "format":
   658	            self.check_str_format_call(e)
   659	        ret_type = get_proper_type(ret_type)
   660	        if isinstance(ret_type, UnionType):
   661	            ret_type = make_simplified_union(ret_type.items)
   662	        if isinstance(ret_type, UninhabitedType) and not ret_type.ambiguous:
   663	            self.chk.binder.unreachable()
   664	        # Warn on calls to functions that always return None. The check
   665	        # of ret_type is both a common-case optimization and prevents reporting
   666	        # the error in dynamic functions (where it will be Any).
   667	        if (
   668	            not allow_none_return
   669	            and isinstance(ret_type, NoneType)
   670	            and self.always_returns_none(e.callee)
   671	        ):
   672	            self.chk.msg.does_not_return_value(callee_type, e)
   673	            return AnyType(TypeOfAny.from_error)
   674	        return ret_type

   699	    def method_fullname(self, object_type: Type, method_name: str) -> str | None:
   700	        """Convert a method name to a fully qualified name, based on the type of the object that
   701	        it is invoked on. Return `None` if the name of `object_type` cannot be determined.
   702	        """
   703	        object_type = get_proper_type(object_type)
   704
   705	        if isinstance(object_type, CallableType) and object_type.is_type_obj():
   706	            # For class method calls, object_type is a callable representing the class object.
   707	            # We "unwrap" it to a regular type, as the class/instance method difference doesn't
   708	            # affect the fully qualified name.
   709	            object_type = get_proper_type(object_type.ret_type)
   710	        elif isinstance(object_type, TypeType):
   711	            object_type = object_type.item
   712
   713	        type_name = None
   714	        if isinstance(object_type, Instance):
   715	            type_name = object_type.type.fullname
   716	        elif isinstance(object_type, (TypedDictType, LiteralType)):
   717	            info = object_type.fallback.type.get_containing_type_info(method_name)
   718	            type_name = info.fullname if info is not None else None
   719	        elif isinstance(object_type, TupleType):
   720	            type_name = tuple_fallback(object_type).type.fullname
   721
   722	        if type_name:
   723	            return f"{type_name}.{method_name}"
   724	        else:
   725	            return None

   727	    def always_returns_none(self, node: Expression) -> bool:
   728	        """Check if `node` refers to something explicitly annotated as only returning None."""
   729	        if isinstance(node, RefExpr):
   730	            if self.defn_returns_none(node.node):
   731	                return True
   732	        if isinstance(node, MemberExpr) and node.node is None:  # instance or class attribute
   733	            typ = get_proper_type(self.chk.lookup_type(node.expr))
   734	            if isinstance(typ, Instance):
   735	                info = typ.type
   736	            elif isinstance(typ, CallableType) and typ.is_type_obj():
   737	                ret_type = get_proper_type(typ.ret_type)
   738	                if isinstance(ret_type, Instance):
   739	                    info = ret_type.type
   740	                else:
   741	                    return False
   742	            else:
   743	                return False
   744	            sym = info.get(node.name)
   745	            if sym and self.defn_returns_none(sym.node):
   746	                return True
   747	        return False

   749	    def defn_returns_none(self, defn: SymbolNode | None) -> bool:
   750	        """Check if `defn` can _only_ return None."""
   751	        if isinstance(defn, FuncDef):
   752	            return isinstance(defn.type, CallableType) and isinstance(
   753	                get_proper_type(defn.type.ret_type), NoneType
   754	            )
   755	        if isinstance(defn, OverloadedFuncDef):
   756	            return all(self.defn_returns_none(item) for item in defn.items)
   757	        if isinstance(defn, Var):
   758	            typ = get_proper_type(defn.type)
   759	            if (
   760	                not defn.is_inferred
   761	                and isinstance(typ, CallableType)
   762	                and isinstance(get_proper_type(typ.ret_type), NoneType)
   763	            ):
   764	                return True
   765	            if isinstance(typ, Instance):
   766	                sym = typ.type.get("__call__")
   767	                if sym and self.defn_returns_none(sym.node):
   768	                    return True
   769	        return False

   790	    def check_typeddict_call(
   791	        self,
   792	        callee: TypedDictType,
   793	        arg_kinds: list[ArgKind],
   794	        arg_names: Sequence[str | None],
   795	        args: list[Expression],
   796	        context: Context,
   797	        orig_callee: Type | None,
   798	    ) -> Type:
   799	        if args and all(ak in (ARG_NAMED, ARG_STAR2) for ak in arg_kinds):
   800	            # ex: Point(x=42, y=1337, **extras)
   801	            # This is a bit ugly, but this is a price for supporting all possible syntax
   802	            # variants for TypedDict constructors.
   803	            kwargs = zip([StrExpr(n) if n is not None else None for n in arg_names], args)
   804	            result = self.validate_typeddict_kwargs(kwargs=kwargs, callee=callee)
   805	            if result is not None:
   806	                validated_kwargs, always_present_keys = result
   807	                return self.check_typeddict_call_with_kwargs(
   808	                    callee, validated_kwargs, context, orig_callee, always_present_keys
   809	                )
   810	            return AnyType(TypeOfAny.from_error)
   811
   812	        if len(args) == 1 and arg_kinds[0] == ARG_POS:
   813	            unique_arg = args[0]
   814	            if isinstance(unique_arg, DictExpr):
   815	                # ex: Point({'x': 42, 'y': 1337, **extras})
   816	                return self.check_typeddict_call_with_dict(
   817	                    callee, unique_arg.items, context, orig_callee
   818	                )
   819	            if isinstance(unique_arg, CallExpr) and isinstance(unique_arg.analyzed, DictExpr):
   820	                # ex: Point(dict(x=42, y=1337, **extras))
   821	                return self.check_typeddict_call_with_dict(
   822	                    callee, unique_arg.analyzed.items, context, orig_callee
   823	                )
   824
   825	        if not args:
   826	            # ex: EmptyDict()
   827	            return self.check_typeddict_call_with_kwargs(callee, {}, context, orig_callee, set())
   828
   829	        self.chk.fail(message_registry.INVALID_TYPEDDICT_ARGS, context)
   830	        return AnyType(TypeOfAny.from_error)

   832	    def validate_typeddict_kwargs(
   833	        self, kwargs: Iterable[tuple[Expression | None, Expression]], callee: TypedDictType
   834	    ) -> tuple[dict[str, list[Expression]], set[str]] | None:
   835	        # All (actual or mapped from ** unpacks) expressions that can match given key.
   836	        result = defaultdict(list)
   837	        # Keys that are guaranteed to be present no matter what (e.g. for all items of a union)
   838	        always_present_keys = set()
   839	        # Indicates latest encountered ** unpack among items.
   840	        last_star_found = None
   841
   842	        for item_name_expr, item_arg in kwargs:
   843	            if item_name_expr:
   844	                key_type = self.accept(item_name_expr)
   845	                values = try_getting_str_literals(item_name_expr, key_type)
   846	                literal_value = None
   847	                if values and len(values) == 1:
   848	                    literal_value = values[0]
   849	                if literal_value is None:
   850	                    key_context = item_name_expr or item_arg
   851	                    self.chk.fail(
   852	                        message_registry.TYPEDDICT_KEY_MUST_BE_STRING_LITERAL,
   853	                        key_context,
   854	                        code=codes.LITERAL_REQ,
   855	                    )
   856	                    return None
   857	                else:
   858	                    # A directly present key unconditionally shadows all previously found
   859	                    # values from ** items.
   860	                    # TODO: for duplicate keys, type-check all values.
   861	                    result[literal_value] = [item_arg]
   862	                    always_present_keys.add(literal_value)
   863	            else:
   864	                last_star_found = item_arg
   865	                if not self.validate_star_typeddict_item(
   866	                    item_arg, callee, result, always_present_keys
   867	                ):
   868	                    return None
   869	        if self.chk.options.extra_checks and last_star_found is not None:
   870	            absent_keys = []
   871	            for key in callee.items:
   872	                if key not in callee.required_keys and key not in result:
   873	                    absent_keys.append(key)
   874	            if absent_keys:
   875	                # Having an optional key not explicitly declared by a ** unpacked
   876	                # TypedDict is unsafe, it may be an (incompatible) subtype at runtime.
   877	                # TODO: catch the cases where a declared key is overridden by a subsequent
   878	                # ** item without it (and not again overridden with complete ** item).
   879	                self.msg.non_required_keys_absent_with_star(absent_keys, last_star_found)
   880	        return result, always_present_keys
   881
   882	    def validate_star_typeddict_item(
   883	        self,
   884	        item_arg: Expression,
   885	        callee: TypedDictType,
   886	        result: dict[str, list[Expression]],
   887	        always_present_keys: set[str],
   888	    ) -> bool:
   889	        """Update keys/expressions from a ** expression in TypedDict constructor.
   890
   891	        Note `result` and `always_present_keys` are updated in place. Return true if the
   892	        expression `item_arg` may valid in `callee` TypedDict context.
   893	        """
   894	        inferred = get_proper_type(self.accept(item_arg, type_context=callee))
   895	        possible_tds = []
   896	        if isinstance(inferred, TypedDictType):
   897	            possible_tds = [inferred]
   898	        elif isinstance(inferred, UnionType):
   899	            for item in get_proper_types(inferred.relevant_items()):
   900	                if isinstance(item, TypedDictType):
   901	                    possible_tds.append(item)
   902	                elif not self.valid_unpack_fallback_item(item):
   903	                    self.msg.unsupported_target_for_star_typeddict(item, item_arg)
   904	                    return False
   905	        elif not self.valid_unpack_fallback_item(inferred):
   906	            self.msg.unsupported_target_for_star_typeddict(inferred, item_arg)
   907	            return False
   908	        all_keys: set[str] = set()
   909	        for td in possible_tds:
   910	            all_keys |= td.items.keys()
   911	        for key in all_keys:
   912	            arg = TempNode(
   913	                UnionType.make_union([td.items[key] for td in possible_tds if key in td.items])
   914	            )
   915	            arg.set_line(item_arg)
   916	            if all(key in td.required_keys for td in possible_tds):
   917	                always_present_keys.add(key)
   918	                # Always present keys override previously found values. This is done
   919	                # to support use cases like `Config({**defaults, **overrides})`, where
   920	                # some `overrides` types are narrower that types in `defaults`, and
   921	                # former are too wide for `Config`.
   922	                if result[key]:
   923	                    first = result[key][0]
   924	                    if not isinstance(first, TempNode):
   925	                        # We must always preserve any non-synthetic values, so that
   926	                        # we will accept them even if they are shadowed.
   927	                        result[key] = [first, arg]
   928	                    else:
   929	                        result[key] = [arg]
   930	                else:
   931	                    result[key] = [arg]
   932	            else:
   933	                # If this key is not required at least in some item of a union
   934	                # it may not shadow previous item, so we need to type check both.
   935	                result[key].append(arg)
   936	        return True

   938	    def valid_unpack_fallback_item(self, typ: ProperType) -> bool:
   939	        if isinstance(typ, AnyType):
   940	            return True
   941	        if not isinstance(typ, Instance) or not typ.type.has_base("typing.Mapping"):
   942	            return False
   943	        mapped = map_instance_to_supertype(typ, self.chk.lookup_typeinfo("typing.Mapping"))
   944	        return all(isinstance(a, AnyType) for a in get_proper_types(mapped.args))
