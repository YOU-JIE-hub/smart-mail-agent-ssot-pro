  1105	    def _preRangeAdjust(self,valueMin,valueMax):
  1106	        rr = self.requiredRange
  1107	        if rr>0:
  1108	            r = valueMax - valueMin
  1109	            if r<rr:
  1110	                m = 0.5*(valueMax+valueMin)
  1111	                rr *= 0.5
  1112	                y1 = min(m-rr,valueMin)
  1113	                y2 = max(m+rr,valueMax)
  1114	                if valueMin>=100 and y1<100:
