   516	    def createData(version, errorCorrectLevel, dataList):
   517	        rsBlocks = QRRSBlock.getRSBlocks(version, errorCorrectLevel)
   518	        buffer = QRBitBuffer();
   519	        for data in dataList:
   520	            data.write(buffer, version)
   521	        # calc num max data.
   522	        totalDataCount = 0;
   523	        for block in rsBlocks:
   524	            totalDataCount += block.dataCount
   525	        if (buffer.getLengthInBits() > totalDataCount * 8):
   526	            raise Exception("code length overflow. (%d > %d)" %
   527	                            (buffer.getLengthInBits(), totalDataCount * 8))
   528	        # end code
   529	        if (buffer.getLengthInBits() + 4 <= totalDataCount * 8):
   530	            buffer.put(0, 4)
   531	        # padding
   532	        while (buffer.getLengthInBits() % 8 != 0):
   533	            buffer.putBit(False)
   534	        # padding
   535	        while (True):
   536	            if (buffer.getLengthInBits() >= totalDataCount * 8):
   537	                break
   538	            buffer.put(QRCode.PAD0, 8)
   539	            if (buffer.getLengthInBits() >= totalDataCount * 8):
   540	                break

   569	                zip_longest(*dcdata), zip_longest(*ecdata))

   790	class QRPolynomial:
   791	    def __init__(self, num, shift):
   792	        if (len(num) == 0):
   793	            raise Exception(len(num) + "/" + shift)
   794	        offset = 0
   795	        while offset < len(num) and num[offset] == 0:
   796	            offset += 1
   797	        self.num = num[offset:] + [0]*shift
   798
   799	    def get(self, index):
   800	        return self.num[index]
   801
   802	    def getLength(self):
   803	        return len(self.num)
   804
   805	    def multiply(self, e):
   806	        num = [0] * (self.getLength() + e.getLength() - 1);
   807	        for i in range(self.getLength()):
   808	            for j in range(e.getLength()):
   809	                num[i + j] ^= QRMath.gexp(QRMath.glog(self.get(i) ) +
   810	                                          QRMath.glog(e.get(j) ) )
   811	        return QRPolynomial(num, 0);
   812
   813	    def mod(self, e):
   814	        if (self.getLength() < e.getLength()):
   815	            return self;
   816	        ratio = QRMath.glog(self.num[0] ) - QRMath.glog(e.num[0] )
   817	        num = [nn ^ QRMath.gexp(QRMath.glog(en) + ratio)
   818	               for nn,en in zip(self.num, e.num)]
   819	        num += self.num[e.getLength():]
   820	        # recursive call
   821	        return QRPolynomial(num, 0).mod(e);

   381	def _crc11(value):
   382	    '''
   383	    >>> usps = [USPS_4State('01234567094987654321',x).binary for x in ('','01234','012345678','01234567891')]
   384	    >>> print(' '.join(nhex(x) for x in usps))
   385	    0x1122103b5c2004b1 0xd138a87bab5cf3804b1 0x202bdc097711204d21804b1 0x16907b2a24abc16a2e5c004b1
   386	    >>> print(' '.join(nhex(_crc11(x)) for x in usps))
   387	    0x51 0x65 0x606 0x751
   388	    '''
   389	    hexbytes = nhex(int(value))[2:]
   390	    hexbytes = '0'*(26-len(hexbytes))+hexbytes
   391	    gp = 0x0F35
   392	    fcs = 0x07FF
   393	    data = int(hexbytes[:2],16)<<5
   394	    for b in range(2,8):
   395	        if (fcs ^ data)&0x400:
   396	            fcs = (fcs<<1)^gp
   397	        else:
   398	            fcs = fcs<<1
   399	        fcs &= 0x7ff
   400	        data <<= 1
   401
   402	    for x in range(2,2*13,2):
   403	        data = int(hexbytes[x:x+2],16)<<3
   404	        for b in range(8):
   405	            if (fcs ^ data)&0x400:
   406	                fcs = (fcs<<1)^gp
   407	            else:
   408	                fcs = fcs<<1
   409	            fcs &= 0x7ff
   410	            data <<= 1
   411	    return fcs

   414	    '''reverse unsigned 13 bit number
   415	    >>> print(_ru13(7936), _ru13(31), _ru13(47), _ru13(7808))
   416	    31 7936 7808 47
   417	    '''
   418	    r = 0
   419	    for x in range(13):
   420	        r <<= 1
   421	        r |= i & 1
   422	        i >>= 1

   426	    '''create and return table of 13 bit values with N bits on
   427	    >>> T = _initNof13Table(5,1287)
   428	    >>> print(' '.join('T[%d]=%d' % (i, T[i]) for i in (0,1,2,3,4,1271,1272,1284,1285,1286)))

   430	    '''

   445	            T[l] = c
   446	            l += 1
   447	            T[l] = r
   448	            l += 1
   449	    assert l==(u+1), 'u+1(%d)!=l(%d) for %d of 13 table' % (u+1,l,N)
