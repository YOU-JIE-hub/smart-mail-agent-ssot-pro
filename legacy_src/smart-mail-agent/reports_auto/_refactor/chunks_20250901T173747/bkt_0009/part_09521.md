   495	    def _force_enable_logging(
   496	        self, level: int | str, logger_obj: logging.Logger
   497	    ) -> int:
   498	        """Enable the desired logging level if the global level was disabled via ``logging.disabled``.
   499
   500	        Only enables logging levels greater than or equal to the requested ``level``.
   501
   502	        Does nothing if the desired ``level`` wasn't disabled.
   503
   504	        :param level:
   505	            The logger level caplog should capture.
   506	            All logging is enabled if a non-standard logging level string is supplied.
   507	            Valid level strings are in :data:`logging._nameToLevel`.
   508	        :param logger_obj: The logger object to check.
   509
   510	        :return: The original disabled logging level.
   511	        """
   512	        original_disable_level: int = logger_obj.manager.disable
   513
   514	        if isinstance(level, str):
   515	            # Try to translate the level string to an int for `logging.disable()`
   516	            level = logging.getLevelName(level)
   517
   518	        if not isinstance(level, int):
   519	            # The level provided was not valid, so just un-disable all logging.
   520	            logging.disable(logging.NOTSET)
   521	        elif not logger_obj.isEnabledFor(level):
   522	            # Each level is `10` away from other levels.
   523	            # https://docs.python.org/3/library/logging.html#logging-levels
   524	            disable_level = max(level - 10, logging.NOTSET)
   525	            logging.disable(disable_level)
   526
   527	        return original_disable_level

   529	    def set_level(self, level: int | str, logger: str | None = None) -> None:
   530	        """Set the threshold level of a logger for the duration of a test.
   531
   532	        Logging messages which are less severe than this level will not be captured.
   533
   534	        .. versionchanged:: 3.4
   535	            The levels of the loggers changed by this function will be
   536	            restored to their initial values at the end of the test.
   537
   538	        Will enable the requested logging level if it was disabled via :func:`logging.disable`.
   539
   540	        :param level: The level.
   541	        :param logger: The logger to update. If not given, the root logger.
   542	        """
   543	        logger_obj = logging.getLogger(logger)
   544	        # Save the original log-level to restore it during teardown.
   545	        self._initial_logger_levels.setdefault(logger, logger_obj.level)
   546	        logger_obj.setLevel(level)
   547	        if self._initial_handler_level is None:
   548	            self._initial_handler_level = self.handler.level
   549	        self.handler.setLevel(level)
   550	        initial_disabled_logging_level = self._force_enable_logging(level, logger_obj)
   551	        if self._initial_disabled_logging_level is None:
   552	            self._initial_disabled_logging_level = initial_disabled_logging_level

   555	    def at_level(self, level: int | str, logger: str | None = None) -> Generator[None]:
   556	        """Context manager that sets the level for capturing of logs. After
   557	        the end of the 'with' statement the level is restored to its original
   558	        value.
   559
   560	        Will enable the requested logging level if it was disabled via :func:`logging.disable`.
   561
   562	        :param level: The level.
   563	        :param logger: The logger to update. If not given, the root logger.
   564	        """
   565	        logger_obj = logging.getLogger(logger)
   566	        orig_level = logger_obj.level
   567	        logger_obj.setLevel(level)
   568	        handler_orig_level = self.handler.level
   569	        self.handler.setLevel(level)
   570	        original_disable_level = self._force_enable_logging(level, logger_obj)
   571	        try:
   572	            yield
   573	        finally:
   574	            logger_obj.setLevel(orig_level)
   575	            self.handler.setLevel(handler_orig_level)
   576	            logging.disable(original_disable_level)

   613	    for setting_name in setting_names:

   311	                if exc.returncode is not None:

   313	                sys.stderr.write(f"{type(exc).__name__}: {exc}\n")

   315	                if isinstance(excinfo.value, SystemExit):
   316	                    sys.stderr.write("mainloop: caught unexpected SystemExit!\n")

   319	        # Explicitly break reference cycle.
   320	        excinfo = None  # type: ignore

   325	                    session=session, exitstatus=session.exitstatus

   396	def pytest_ignore_collect(collection_path: Path, config: Config) -> bool | None:
   397	    if collection_path.name == "__pycache__":
   398	        return True
   399
   400	    ignore_paths = config._getconftest_pathlist(
   401	        "collect_ignore", path=collection_path.parent
   402	    )
   403	    ignore_paths = ignore_paths or []
   404	    excludeopt = config.getoption("ignore")
   405	    if excludeopt:
   406	        ignore_paths.extend(absolutepath(x) for x in excludeopt)
   407
   408	    if collection_path in ignore_paths:
   409	        return True
   410
   411	    ignore_globs = config._getconftest_pathlist(
   412	        "collect_ignore_glob", path=collection_path.parent
   413	    )
   414	    ignore_globs = ignore_globs or []
   415	    excludeglobopt = config.getoption("ignore_glob")
   416	    if excludeglobopt:
   417	        ignore_globs.extend(absolutepath(x) for x in excludeglobopt)
   418
   419	    if any(fnmatch.fnmatch(str(collection_path), str(glob)) for glob in ignore_globs):
   420	        return True
   421
   422	    allow_in_venv = config.getoption("collect_in_virtualenv")
   423	    if not allow_in_venv and _in_venv(collection_path):
   424	        return True
   425
   426	    if collection_path.is_dir():
   427	        norecursepatterns = config.getini("norecursedirs")
   428	        if any(fnmatch_ex(pat, collection_path) for pat in norecursepatterns):
   429	            return True
   430
   431	    return None

   749	        self, args: Sequence[str] | None = None, genitems: bool = True

   752

   754	        implementation; see the documentation of this hook for more details.

   791	            self._initialpaths_with_parents = frozenset(initialpaths_with_parents)

   843	    def collect(self) -> Iterator[nodes.Item | nodes.Collector]:
   844	        # This is a cache for the root directories of the initial paths.
   845	        # We can't use collection_cache for Session because of its special
   846	        # role as the bootstrapping collector.
   847	        path_cache: dict[Path, Sequence[nodes.Collector]] = {}
   848
   849	        pm = self.config.pluginmanager
   850
   851	        for collection_argument in self._initial_parts:
   852	            self.trace("processing argument", collection_argument)
   853	            self.trace.root.indent += 1
   854
   855	            argpath = collection_argument.path
   856	            names = collection_argument.parts
   857	            module_name = collection_argument.module_name
   858
   859	            # resolve_collection_argument() ensures this.
   860	            if argpath.is_dir():
   861	                assert not names, f"invalid arg {(argpath, names)!r}"
   862
   863	            paths = [argpath]
   864	            # Add relevant parents of the path, from the root, e.g.
   865	            #   /a/b/c.py -> [/, /a, /a/b, /a/b/c.py]
   866	            if module_name is None:
   867	                # Paths outside of the confcutdir should not be considered.
   868	                for path in argpath.parents:
   869	                    if not pm._is_in_confcutdir(path):
   870	                        break
   871	                    paths.insert(0, path)
   872	            else:
   873	                # For --pyargs arguments, only consider paths matching the module
   874	                # name. Paths beyond the package hierarchy are not included.
   875	                module_name_parts = module_name.split(".")
   876	                for i, path in enumerate(argpath.parents, 2):
   877	                    if i > len(module_name_parts) or path.stem != module_name_parts[-i]:
   878	                        break
   879	                    paths.insert(0, path)
   880
   881	            # Start going over the parts from the root, collecting each level
   882	            # and discarding all nodes which don't match the level's part.
   883	            any_matched_in_initial_part = False
   884	            notfound_collectors = []
   885	            work: list[tuple[nodes.Collector | nodes.Item, list[Path | str]]] = [
   886	                (self, [*paths, *names])
   887	            ]
   888	            while work:
   889	                matchnode, matchparts = work.pop()
   890
   891	                # Pop'd all of the parts, this is a match.
   892	                if not matchparts:
   893	                    yield matchnode
   894	                    any_matched_in_initial_part = True
   895	                    continue
   896
   897	                # Should have been matched by now, discard.
   898	                if not isinstance(matchnode, nodes.Collector):
   899	                    continue
   900
   901	                # Collect this level of matching.
   902	                # Collecting Session (self) is done directly to avoid endless
   903	                # recursion to this function.
   904	                subnodes: Sequence[nodes.Collector | nodes.Item]
   905	                if isinstance(matchnode, Session):
   906	                    assert isinstance(matchparts[0], Path)
   907	                    subnodes = matchnode._collect_path(matchparts[0], path_cache)
   908	                else:
   909	                    # For backward compat, files given directly multiple
   910	                    # times on the command line should not be deduplicated.
   911	                    handle_dupes = not (
   912	                        len(matchparts) == 1
   913	                        and isinstance(matchparts[0], Path)
   914	                        and matchparts[0].is_file()
   915	                    )
   916	                    rep, duplicate = self._collect_one_node(matchnode, handle_dupes)
   917	                    if not duplicate and not rep.passed:
   918	                        # Report collection failures here to avoid failing to
   919	                        # run some test specified in the command line because
   920	                        # the module could not be imported (#134).
   921	                        matchnode.ihook.pytest_collectreport(report=rep)
   922	                    if not rep.passed:
   923	                        continue
   924	                    subnodes = rep.result
   925
   926	                # Prune this level.
   927	                any_matched_in_collector = False
   928	                for node in reversed(subnodes):
   929	                    # Path part e.g. `/a/b/` in `/a/b/test_file.py::TestIt::test_it`.
   930	                    if isinstance(matchparts[0], Path):
   931	                        is_match = node.path == matchparts[0]
   932	                        if sys.platform == "win32" and not is_match:
   933	                            # In case the file paths do not match, fallback to samefile() to
   934	                            # account for short-paths on Windows (#11895).
   935	                            same_file = os.path.samefile(node.path, matchparts[0])
   936	                            # We don't want to match links to the current node,
   937	                            # otherwise we would match the same file more than once (#12039).
   938	                            is_match = same_file and (
   939	                                os.path.islink(node.path)
   940	                                == os.path.islink(matchparts[0])
   941	                            )
   942
   943	                    # Name part e.g. `TestIt` in `/a/b/test_file.py::TestIt::test_it`.
   944	                    else:
   945	                        # TODO: Remove parametrized workaround once collection structure contains
   946	                        # parametrization.
   947	                        is_match = (
   948	                            node.name == matchparts[0]
   949	                            or node.name.split("[")[0] == matchparts[0]
   950	                        )
   951	                    if is_match:
   952	                        work.append((node, matchparts[1:]))
   953	                        any_matched_in_collector = True
   954
   955	                if not any_matched_in_collector:
   956	                    notfound_collectors.append(matchnode)
   957
   958	            if not any_matched_in_initial_part:
   959	                report_arg = "::".join((str(argpath), *names))
   960	                self._notfound.append((report_arg, notfound_collectors))
   961
   962	            self.trace.root.indent -= 1

  1017	    parts for specific tests selection, for example:
  1018
  1019	        "pkg/tests/test_foo.py::TestClass::test_foo"

  1021	    This function ensures the path exists, and returns a resolved `CollectionArgument`:
  1022
  1023	        CollectionArgument(
  1024	            path=Path("/full/path/to/pkg/tests/test_foo.py"),

  1030	    module paths instead of file-system paths:
  1031
  1032	        "pkg.tests.test_foo::TestClass::test_foo"
  1033
  1034	    In which case we search sys.path for a matching module, and then return the *path* to the
  1035	    found module, which may look like this:
  1036
  1037	        CollectionArgument(
  1038	            path=Path("/home/u/myvenv/lib/site-packages/pkg/tests/test_foo.py"),
  1039	            parts=["TestClass", "test_foo"],
  1040	            module_name="pkg.tests.test_foo",
  1041	        )
  1042
  1043	    If the path doesn't exist, raise UsageError.
  1044	    If the path is a directory and selection parts are present, raise UsageError.
  1045	    """
  1046	    base, squacket, rest = str(arg).partition("[")
  1047	    strpath, *parts = base.split("::")
  1048	    if parts:
  1049	        parts[-1] = f"{parts[-1]}{squacket}{rest}"
  1050	    module_name = None
  1051	    if as_pypath:
  1052	        pyarg_strpath = search_pypath(strpath)
  1053	        if pyarg_strpath is not None:
  1054	            module_name = strpath
  1055	            strpath = pyarg_strpath
  1056	    fspath = invocation_path / strpath
  1057	    fspath = absolutepath(fspath)
  1058	    if not safe_exists(fspath):
  1059	        msg = (
  1060	            "module or package not found: {arg} (missing __init__.py?)"
  1061	            if as_pypath
  1062	            else "file or directory not found: {arg}"
  1063	        )
  1064	        raise UsageError(msg.format(arg=arg))
  1065	    if parts and fspath.is_dir():
  1066	        msg = (
  1067	            "package argument cannot contain :: selection parts: {arg}"
  1068	            if as_pypath
  1069	            else "directory argument cannot contain :: selection parts: {arg}"
  1070	        )
  1071	        raise UsageError(msg.format(arg=arg))
  1072	    return CollectionArgument(
  1073	        path=fspath,
  1074	        parts=parts,
  1075	        module_name=module_name,
  1076	    )
