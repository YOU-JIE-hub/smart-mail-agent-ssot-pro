    93	    body_block = BasicBlock()

    97	    else_block = BasicBlock()
    98	    # Block executed after the loop
    99	    exit_block = BasicBlock()

   102	    normal_loop_exit = else_block if else_insts is not None else exit_block

   104	    for_gen = make_for_loop_generator(
   105	        builder, index, expr, body_block, normal_loop_exit, line, is_async=is_async

   295	def comprehension_helper(
   296	    builder: IRBuilder,
   297	    loop_params: list[tuple[Lvalue, Expression, list[Expression], bool]],
   298	    gen_inner_stmts: Callable[[], None],
   299	    line: int,
   300	) -> None:
   301	    """Helper function for list comprehensions.
   302
   303	    Args:
   304	        loop_params: a list of (index, expr, [conditions]) tuples defining nested loops:
   305	            - "index" is the Lvalue indexing that loop;
   306	            - "expr" is the expression for the object to be iterated over;
   307	            - "conditions" is a list of conditions, evaluated in order with short-circuiting,
   308	                that must all be true for the loop body to be executed
   309	        gen_inner_stmts: function to generate the IR for the body of the innermost loop
   310	    """
   311
   312	    def handle_loop(loop_params: list[tuple[Lvalue, Expression, list[Expression], bool]]) -> None:
   313	        """Generate IR for a loop.
   314
   315	        Given a list of (index, expression, [conditions]) tuples, generate IR
   316	        for the nested loops the list defines.
   317	        """
   318	        index, expr, conds, is_async = loop_params[0]
   319	        for_loop_helper(
   320	            builder,
   321	            index,
   322	            expr,
   323	            lambda: loop_contents(conds, loop_params[1:]),
   324	            None,
   325	            is_async=is_async,
   326	            line=line,
   327	        )
   328
   329	    def loop_contents(
   330	        conds: list[Expression],
   331	        remaining_loop_params: list[tuple[Lvalue, Expression, list[Expression], bool]],
   332	    ) -> None:
   333	        """Generate the body of the loop.
   334
   335	        Args:
   336	            conds: a list of conditions to be evaluated (in order, with short circuiting)
   337	                to gate the body of the loop
   338	            remaining_loop_params: the parameters for any further nested loops; if it's empty
   339	                we'll instead evaluate the "gen_inner_stmts" function
   340	        """
   341	        # Check conditions, in order, short circuiting them.
   342	        for cond in conds:
   343	            cond_val = builder.accept(cond)
   344	            cont_block, rest_block = BasicBlock(), BasicBlock()
   345	            # If the condition is true we'll skip the continue.
   346	            builder.add_bool_branch(cond_val, rest_block, cont_block)
   347	            builder.activate_block(cont_block)
   348	            builder.nonlocal_control[-1].gen_continue(builder, cond.line)
   349	            builder.goto_and_activate(rest_block)
   350
   351	        if remaining_loop_params:
   352	            # There's another nested level, so the body of this loop is another loop.
   353	            return handle_loop(remaining_loop_params)
   354	        else:
   355	            # We finally reached the actual body of the generator.
   356	            # Generate the IR for the inner loop body.
   357	            gen_inner_stmts()
   358
   359	    handle_loop(loop_params)

   371	    builder: IRBuilder,
   372	    index: Lvalue,
   373	    expr: Expression,
   374	    body_block: BasicBlock,
   375	    loop_exit: BasicBlock,
   376	    line: int,
   377	    is_async: bool = False,
   378	    nested: bool = False,
   379	) -> ForGenerator:
   380	    """Return helper object for generating a for loop over an iterable.
   381
   382	    If "nested" is True, this is a nested iterator such as "e" in "enumerate(e)".
   383	    """
   384
   385	    # Do an async loop if needed. async is always generic
   386	    if is_async:
   387	        expr_reg = builder.accept(expr)
   388	        async_obj = ForAsyncIterable(builder, index, body_block, loop_exit, line, nested)
   389	        item_type = builder._analyze_iterable_item_type(expr)
   390	        item_rtype = builder.type_to_rtype(item_type)
   391	        async_obj.init(expr_reg, item_rtype)
   392	        return async_obj
   393
   394	    rtyp = builder.node_type(expr)
   395	    if is_sequence_rprimitive(rtyp):
   396	        # Special case "for x in <list>".
   397	        expr_reg = builder.accept(expr)
   398	        target_type = builder.get_sequence_type(expr)
   399
   400	        for_list = ForSequence(builder, index, body_block, loop_exit, line, nested)
   401	        for_list.init(expr_reg, target_type, reverse=False)
   402	        return for_list
   403
   404	    if is_dict_rprimitive(rtyp):
   405	        # Special case "for k in <dict>".
   406	        expr_reg = builder.accept(expr)
   407	        target_type = builder.get_dict_key_type(expr)
   408
   409	        for_dict = ForDictionaryKeys(builder, index, body_block, loop_exit, line, nested)
   410	        for_dict.init(expr_reg, target_type)
   411	        return for_dict
   412
   413	    if isinstance(expr, CallExpr) and isinstance(expr.callee, RefExpr):
   414	        if (
   415	            is_range_ref(expr.callee)
   416	            and (
   417	                len(expr.args) <= 2
   418	                or (len(expr.args) == 3 and builder.extract_int(expr.args[2]) is not None)
   419	            )
   420	            and set(expr.arg_kinds) == {ARG_POS}
   421	        ):
   422	            # Special case "for x in range(...)".
   423	            # We support the 3 arg form but only for int literals, since it doesn't
   424	            # seem worth the hassle of supporting dynamically determining which
   425	            # direction of comparison to do.
   426	            if len(expr.args) == 1:
   427	                start_reg: Value = Integer(0)
   428	                end_reg = builder.accept(expr.args[0])
   429	            else:
   430	                start_reg = builder.accept(expr.args[0])
   431	                end_reg = builder.accept(expr.args[1])
   432	            if len(expr.args) == 3:
   433	                step = builder.extract_int(expr.args[2])
   434	                assert step is not None
   435	                if step == 0:
   436	                    builder.error("range() step can't be zero", expr.args[2].line)
   437	            else:
   438	                step = 1
   439
   440	            for_range = ForRange(builder, index, body_block, loop_exit, line, nested)
   441	            for_range.init(start_reg, end_reg, step)
   442	            return for_range
   443
   444	        elif (
   445	            expr.callee.fullname == "builtins.enumerate"
   446	            and len(expr.args) == 1
   447	            and expr.arg_kinds == [ARG_POS]
   448	            and isinstance(index, TupleExpr)
   449	            and len(index.items) == 2
   450	        ):
   451	            # Special case "for i, x in enumerate(y)".
   452	            lvalue1 = index.items[0]
   453	            lvalue2 = index.items[1]
   454	            for_enumerate = ForEnumerate(builder, index, body_block, loop_exit, line, nested)
   455	            for_enumerate.init(lvalue1, lvalue2, expr.args[0])
   456	            return for_enumerate
   457
   458	        elif (
   459	            expr.callee.fullname == "builtins.zip"
   460	            and len(expr.args) >= 2
   461	            and set(expr.arg_kinds) == {ARG_POS}
   462	            and isinstance(index, TupleExpr)
   463	            and len(index.items) == len(expr.args)
   464	        ):
   465	            # Special case "for x, y in zip(a, b)".
   466	            for_zip = ForZip(builder, index, body_block, loop_exit, line, nested)
   467	            for_zip.init(index.items, expr.args)
   468	            return for_zip
   469
   470	        if (
   471	            expr.callee.fullname == "builtins.reversed"
   472	            and len(expr.args) == 1
   473	            and expr.arg_kinds == [ARG_POS]
   474	            and is_sequence_rprimitive(builder.node_type(expr.args[0]))
   475	        ):
   476	            # Special case "for x in reversed(<list>)".
   477	            expr_reg = builder.accept(expr.args[0])
   478	            target_type = builder.get_sequence_type(expr)
   479
   480	            for_list = ForSequence(builder, index, body_block, loop_exit, line, nested)
   481	            for_list.init(expr_reg, target_type, reverse=True)
   482	            return for_list
   483	    if isinstance(expr, CallExpr) and isinstance(expr.callee, MemberExpr) and not expr.args:
   484	        # Special cases for dictionary iterator methods, like dict.items().
   485	        rtype = builder.node_type(expr.callee.expr)
   486	        if is_dict_rprimitive(rtype) and expr.callee.name in ("keys", "values", "items"):
   487	            expr_reg = builder.accept(expr.callee.expr)
   488	            for_dict_type: type[ForGenerator] | None = None
   489	            if expr.callee.name == "keys":
   490	                target_type = builder.get_dict_key_type(expr.callee.expr)
   491	                for_dict_type = ForDictionaryKeys
   492	            elif expr.callee.name == "values":
   493	                target_type = builder.get_dict_value_type(expr.callee.expr)
   494	                for_dict_type = ForDictionaryValues
   495	            else:
   496	                target_type = builder.get_dict_item_type(expr.callee.expr)
   497	                for_dict_type = ForDictionaryItems
   498	            for_dict_gen = for_dict_type(builder, index, body_block, loop_exit, line, nested)
   499	            for_dict_gen.init(expr_reg, target_type)
   500	            return for_dict_gen
   501
   502	    iterable_expr_reg: Value | None = None
   503	    if isinstance(expr, SetExpr):
   504	        # Special case "for x in <set literal>".
   505	        from mypyc.irbuild.expression import precompute_set_literal
   506
   507	        set_literal = precompute_set_literal(builder, expr)
   508	        if set_literal is not None:
   509	            iterable_expr_reg = set_literal
   510
   511	    # Default to a generic for loop.
   512	    if iterable_expr_reg is None:
   513	        iterable_expr_reg = builder.accept(expr)
   514	    for_obj = ForIterable(builder, index, body_block, loop_exit, line, nested)
   515	    item_type = builder._analyze_iterable_item_type(expr)
   516	    item_rtype = builder.type_to_rtype(item_type)
   517	    for_obj.init(iterable_expr_reg, item_rtype)
   518	    return for_obj

   521	class ForGenerator:
   522	    """Abstract base class for generating for loops."""
   523
   524	    def __init__(
   525	        self,
   526	        builder: IRBuilder,
   527	        index: Lvalue,
   528	        body_block: BasicBlock,
   529	        loop_exit: BasicBlock,
   530	        line: int,
   531	        nested: bool,
   532	    ) -> None:
   533	        self.builder = builder
   534	        self.index = index
   535	        self.body_block = body_block
   536	        self.line = line
   537	        # Some for loops need a cleanup block that we execute at exit. We
   538	        # create a cleanup block if needed. However, if we are generating a for
   539	        # loop for a nested iterator, such as "e" in "enumerate(e)", the
   540	        # outermost generator should generate the cleanup block -- we don't
   541	        # need to do it here.
   542	        if self.need_cleanup() and not nested:
   543	            # Create a new block to handle cleanup after loop exit.
   544	            self.loop_exit = BasicBlock()
   545	        else:
   546	            # Just use the existing loop exit block.
   547	            self.loop_exit = loop_exit
   548
   549	    def need_cleanup(self) -> bool:
   550	        """If this returns true, we need post-loop cleanup."""
   551	        return False
   552
   553	    def add_cleanup(self, exit_block: BasicBlock) -> None:
   554	        """Add post-loop cleanup, if needed."""
   555	        if self.need_cleanup():
   556	            self.builder.activate_block(self.loop_exit)
   557	            self.gen_cleanup()
   558	            self.builder.goto(exit_block)
   559
   560	    def gen_condition(self) -> None:
   561	        """Generate check for loop exit (e.g. exhaustion of iteration)."""
   562
   563	    def begin_body(self) -> None:
   564	        """Generate ops at the beginning of the body (if needed)."""
   565
   566	    def gen_step(self) -> None:
   567	        """Generate stepping to the next item (if needed)."""
   568
   569	    def gen_cleanup(self) -> None:
   570	        """Generate post-loop cleanup (if needed)."""
   571
   572	    def load_len(self, expr: Value | AssignmentTarget) -> Value:
   573	        """A helper to get collection length, used by several subclasses."""
   574	        return self.builder.builder.builtin_len(self.builder.read(expr, self.line), self.line)

   577	class ForIterable(ForGenerator):
   578	    """Generate IR for a for loop over an arbitrary iterable (the general case)."""
   579
   580	    def need_cleanup(self) -> bool:
   581	        # Create a new cleanup block for when the loop is finished.
   582	        return True
   583
   584	    def init(self, expr_reg: Value, target_type: RType) -> None:
   585	        # Define targets to contain the expression, along with the iterator that will be used
   586	        # for the for-loop. If we are inside of a generator function, spill these into the
   587	        # environment class.
   588	        builder = self.builder
   589	        iter_reg = builder.primitive_op(iter_op, [expr_reg], self.line)
   590	        builder.maybe_spill(expr_reg)
   591	        self.iter_target = builder.maybe_spill(iter_reg)
   592	        self.target_type = target_type
   593
   594	    def gen_condition(self) -> None:
   595	        # We call __next__ on the iterator and check to see if the return value
   596	        # is NULL, which signals either the end of the Iterable being traversed
   597	        # or an exception being raised. Note that Branch.IS_ERROR checks only
   598	        # for NULL (an exception does not necessarily have to be raised).
   599	        builder = self.builder
   600	        line = self.line
   601	        self.next_reg = builder.call_c(next_op, [builder.read(self.iter_target, line)], line)
   602	        builder.add(Branch(self.next_reg, self.loop_exit, self.body_block, Branch.IS_ERROR))
   603
   604	    def begin_body(self) -> None:
   605	        # Assign the value obtained from __next__ to the
   606	        # lvalue so that it can be referenced by code in the body of the loop.
   607	        builder = self.builder
   608	        line = self.line
   609	        # We unbox here so that iterating with tuple unpacking generates a tuple based
   610	        # unpack instead of an iterator based one.
   611	        next_reg = builder.coerce(self.next_reg, self.target_type, line)
   612	        builder.assign(builder.get_assignment_target(self.index), next_reg, line)
   613
   614	    def gen_step(self) -> None:
   615	        # Nothing to do here, since we get the next item as part of gen_condition().
   616	        pass
   617
   618	    def gen_cleanup(self) -> None:
   619	        # We set the branch to go here if the conditional evaluates to true. If
   620	        # an exception was raised during the loop, then err_reg will be set to
   621	        # True. If no_err_occurred_op returns False, then the exception will be
   622	        # propagated using the ERR_FALSE flag.
   623	        self.builder.call_c(no_err_occurred_op, [], self.line)

   626	class ForAsyncIterable(ForGenerator):
   627	    """Generate IR for an async for loop."""
   628
   629	    def init(self, expr_reg: Value, target_type: RType) -> None:
   630	        # Define targets to contain the expression, along with the
   631	        # iterator that will be used for the for-loop. We are inside
   632	        # of a generator function, so we will spill these into
   633	        # environment class.
   634	        builder = self.builder
   635	        iter_reg = builder.call_c(aiter_op, [expr_reg], self.line)
   636	        builder.maybe_spill(expr_reg)
   637	        self.iter_target = builder.maybe_spill(iter_reg)
   638	        self.target_type = target_type
   639	        self.stop_reg = Register(bool_rprimitive)
   640
   641	    def gen_condition(self) -> None:
   642	        # This does the test and fetches the next value
   643	        # try:
   644	        #     TARGET = await type(iter).__anext__(iter)
   645	        #     stop = False
   646	        # except StopAsyncIteration:
   647	        #     stop = True
   648	        #
   649	        # What a pain.
   650	        # There are optimizations available here if we punch through some abstractions.
   651
   652	        from mypyc.irbuild.statement import emit_await, transform_try_except
   653
   654	        builder = self.builder
   655	        line = self.line
   656
   657	        def except_match() -> Value:
   658	            addr = builder.add(LoadAddress(pointer_rprimitive, stop_async_iteration_op.src, line))
   659	            return builder.add(LoadMem(stop_async_iteration_op.type, addr))
   660
   661	        def try_body() -> None:
   662	            awaitable = builder.call_c(anext_op, [builder.read(self.iter_target)], line)
   663	            self.next_reg = emit_await(builder, awaitable, line)
   664	            builder.assign(self.stop_reg, builder.false(), -1)
   665
   666	        def except_body() -> None:
   667	            builder.assign(self.stop_reg, builder.true(), line)
   668
   669	        transform_try_except(
   670	            builder, try_body, [((except_match, line), None, except_body)], None, line
   671	        )
   672
   673	        builder.add(Branch(self.stop_reg, self.loop_exit, self.body_block, Branch.BOOL))
   674
   675	    def begin_body(self) -> None:
   676	        # Assign the value obtained from await __anext__ to the
   677	        # lvalue so that it can be referenced by code in the body of the loop.
   678	        builder = self.builder
   679	        line = self.line
   680	        # We unbox here so that iterating with tuple unpacking generates a tuple based
   681	        # unpack instead of an iterator based one.
   682	        next_reg = builder.coerce(self.next_reg, self.target_type, line)
   683	        builder.assign(builder.get_assignment_target(self.index), next_reg, line)
   684
   685	    def gen_step(self) -> None:
   686	        # Nothing to do here, since we get the next item as part of gen_condition().
   687	        pass

   690	def unsafe_index(builder: IRBuilder, target: Value, index: Value, line: int) -> Value:
   691	    """Emit a potentially unsafe index into a target."""
   692	    # This doesn't really fit nicely into any of our data-driven frameworks
   693	    # since we want to use __getitem__ if we don't have an unsafe version,
   694	    # so we just check manually.
   695	    if is_list_rprimitive(target.type):
   696	        return builder.primitive_op(list_get_item_unsafe_op, [target, index], line)
   697	    else:
   698	        return builder.gen_method_call(target, "__getitem__", [index], None, line)

   701	class ForSequence(ForGenerator):
   702	    """Generate optimized IR for a for loop over a sequence.
   703
   704	    Supports iterating in both forward and reverse.
   705	    """
   706
   707	    def init(self, expr_reg: Value, target_type: RType, reverse: bool) -> None:
   708	        builder = self.builder
   709	        self.reverse = reverse
   710	        # Define target to contain the expression, along with the index that will be used
   711	        # for the for-loop. If we are inside of a generator function, spill these into the
   712	        # environment class.
   713	        self.expr_target = builder.maybe_spill(expr_reg)
   714	        if not reverse:
   715	            index_reg: Value = Integer(0)
   716	        else:
   717	            index_reg = builder.binary_op(
   718	                self.load_len(self.expr_target), Integer(1), "-", self.line
   719	            )
   720	        self.index_target = builder.maybe_spill_assignable(index_reg)
   721	        self.target_type = target_type
   722
   723	    def gen_condition(self) -> None:
   724	        builder = self.builder
   725	        line = self.line
   726	        # TODO: Don't reload the length each time when iterating an immutable sequence?
   727	        if self.reverse:
   728	            # If we are iterating in reverse order, we obviously need
   729	            # to check that the index is still positive. Somewhat less
   730	            # obviously we still need to check against the length,
   731	            # since it could shrink out from under us.
   732	            comparison = builder.binary_op(
   733	                builder.read(self.index_target, line), Integer(0), ">=", line
   734	            )
   735	            second_check = BasicBlock()
   736	            builder.add_bool_branch(comparison, second_check, self.loop_exit)
   737	            builder.activate_block(second_check)
   738	        # For compatibility with python semantics we recalculate the length
   739	        # at every iteration.
   740	        len_reg = self.load_len(self.expr_target)
   741	        comparison = builder.binary_op(builder.read(self.index_target, line), len_reg, "<", line)
   742	        builder.add_bool_branch(comparison, self.body_block, self.loop_exit)
   743
   744	    def begin_body(self) -> None:
   745	        builder = self.builder
   746	        line = self.line
   747	        # Read the next list item.
   748	        value_box = unsafe_index(
   749	            builder,
   750	            builder.read(self.expr_target, line),
   751	            builder.read(self.index_target, line),
   752	            line,
   753	        )
   754	        assert value_box
   755	        # We coerce to the type of list elements here so that
   756	        # iterating with tuple unpacking generates a tuple based
   757	        # unpack instead of an iterator based one.
   758	        builder.assign(
   759	            builder.get_assignment_target(self.index),
   760	            builder.coerce(value_box, self.target_type, line),
   761	            line,
   762	        )
   763
   764	    def gen_step(self) -> None:
   765	        # Step to the next item.
   766	        builder = self.builder
   767	        line = self.line
   768	        step = 1 if not self.reverse else -1
   769	        add = builder.int_op(
   770	            short_int_rprimitive,
   771	            builder.read(self.index_target, line),
   772	            Integer(step),
   773	            IntOp.ADD,
   774	            line,
   775	        )
   776	        builder.assign(self.index_target, add, line)

   779	class ForDictionaryCommon(ForGenerator):
   780	    """Generate optimized IR for a for loop over dictionary keys/values.
   781
   782	    The logic is pretty straightforward, we use PyDict_Next() API wrapped in
   783	    a tuple, so that we can modify only a single register. The layout of the tuple:
   784	      * f0: are there more items (bool)
   785	      * f1: current offset (int)
   786	      * f2: next key (object)
   787	      * f3: next value (object)
   788	    For more info see https://docs.python.org/3/c-api/dict.html#c.PyDict_Next.
   789
   790	    Note that for subclasses we fall back to generic PyObject_GetIter() logic,
   791	    since they may override some iteration methods in subtly incompatible manner.
   792	    The fallback logic is implemented in CPy.h via dynamic type check.
   793	    """
   794
   795	    dict_next_op: ClassVar[CFunctionDescription]
   796	    dict_iter_op: ClassVar[CFunctionDescription]
   797
   798	    def need_cleanup(self) -> bool:
   799	        # Technically, a dict subclass can raise an unrelated exception
   800	        # in __next__(), so we need this.
   801	        return True
   802
   803	    def init(self, expr_reg: Value, target_type: RType) -> None:
   804	        builder = self.builder
   805	        self.target_type = target_type
   806
   807	        # We add some variables to environment class, so they can be read across yield.
   808	        self.expr_target = builder.maybe_spill(expr_reg)
   809	        offset = Integer(0)
   810	        self.offset_target = builder.maybe_spill_assignable(offset)
   811	        self.size = builder.maybe_spill(self.load_len(self.expr_target))
   812
   813	        # For dict class (not a subclass) this is the dictionary itself.
   814	        iter_reg = builder.call_c(self.dict_iter_op, [expr_reg], self.line)
   815	        self.iter_target = builder.maybe_spill(iter_reg)
   816
   817	    def gen_condition(self) -> None:
   818	        """Get next key/value pair, set new offset, and check if we should continue."""
   819	        builder = self.builder
   820	        line = self.line
   821	        self.next_tuple = self.builder.call_c(
   822	            self.dict_next_op,
   823	            [builder.read(self.iter_target, line), builder.read(self.offset_target, line)],
   824	            line,
   825	        )
   826
   827	        # Do this here instead of in gen_step() to minimize variables in environment.
   828	        new_offset = builder.add(TupleGet(self.next_tuple, 1, line))
   829	        builder.assign(self.offset_target, new_offset, line)
   830
   831	        should_continue = builder.add(TupleGet(self.next_tuple, 0, line))
   832	        builder.add(Branch(should_continue, self.body_block, self.loop_exit, Branch.BOOL))
   833
   834	    def gen_step(self) -> None:
   835	        """Check that dictionary didn't change size during iteration.
   836
   837	        Raise RuntimeError if it is not the case to match CPython behavior.
   838	        """
   839	        builder = self.builder
   840	        line = self.line
   841	        # Technically, we don't need a new primitive for this, but it is simpler.
   842	        builder.call_c(
   843	            dict_check_size_op,
   844	            [builder.read(self.expr_target, line), builder.read(self.size, line)],
   845	            line,
   846	        )
   847
   848	    def gen_cleanup(self) -> None:
   849	        # Same as for generic ForIterable.
   850	        self.builder.call_c(no_err_occurred_op, [], self.line)

   921	class ForRange(ForGenerator):
   922	    """Generate optimized IR for a for loop over an integer range."""
   923
   924	    def init(self, start_reg: Value, end_reg: Value, step: int) -> None:
   925	        builder = self.builder
   926	        self.start_reg = start_reg
   927	        self.end_reg = end_reg
   928	        self.step = step
   929	        self.end_target = builder.maybe_spill(end_reg)
   930	        if is_short_int_rprimitive(start_reg.type) and is_short_int_rprimitive(end_reg.type):
   931	            index_type: RType = short_int_rprimitive
   932	        elif is_fixed_width_rtype(end_reg.type):
   933	            index_type = end_reg.type
   934	        else:
   935	            index_type = int_rprimitive
   936	        index_reg = Register(index_type)
   937	        builder.assign(index_reg, start_reg, -1)
   938	        self.index_reg = builder.maybe_spill_assignable(index_reg)
   939	        # Initialize loop index to 0. Assert that the index target is assignable.
   940	        self.index_target: Register | AssignmentTarget = builder.get_assignment_target(self.index)
   941	        builder.assign(self.index_target, builder.read(self.index_reg, self.line), self.line)
   942
   943	    def gen_condition(self) -> None:
   944	        builder = self.builder
   945	        line = self.line
   946	        # Add loop condition check.
   947	        cmp = "<" if self.step > 0 else ">"
   948	        comparison = builder.binary_op(
   949	            builder.read(self.index_reg, line), builder.read(self.end_target, line), cmp, line
   950	        )
   951	        builder.add_bool_branch(comparison, self.body_block, self.loop_exit)
   952
   953	    def gen_step(self) -> None:
   954	        builder = self.builder
   955	        line = self.line
   956
   957	        # Increment index register. If the range is known to fit in short ints, use
   958	        # short ints.
   959	        if is_short_int_rprimitive(self.start_reg.type) and is_short_int_rprimitive(
   960	            self.end_reg.type
   961	        ):
   962	            new_val = builder.int_op(
   963	                short_int_rprimitive,
   964	                builder.read(self.index_reg, line),
   965	                Integer(self.step),
   966	                IntOp.ADD,
   967	                line,
   968	            )
   969
   970	        else:
   971	            new_val = builder.binary_op(
   972	                builder.read(self.index_reg, line), Integer(self.step), "+", line
   973	            )
   974	        builder.assign(self.index_reg, new_val, line)
   975	        builder.assign(self.index_target, new_val, line)

   238	    )
   239	    is_generator = fn_info.is_generator
   240	    builder.enter(fn_info, ret_type=sig.ret_type)
   241
   242	    # Functions that contain nested functions need an environment class to store variables that
   243	    # are free in their nested functions. Generator functions need an environment class to
   244	    # store a variable denoting the next instruction to be executed when the __next__ function
   245	    # is called, along with all the variables inside the function itself.
   246	    if contains_nested or (
   247	        is_generator and not builder.fn_info.can_merge_generator_and_env_classes()
   248	    ):
   249	        setup_env_class(builder)
   250
   251	    if is_nested or in_non_ext:
   252	        setup_callable_class(builder)
   253
   254	    if is_generator:
   255	        # First generate a function that just constructs and returns a generator object.
   256	        func_ir, func_reg = gen_generator_func(
   257	            builder,
   258	            lambda args, blocks, fn_info: gen_func_ir(
   259	                builder, args, blocks, sig, fn_info, cdef, is_singledispatch
   260	            ),
   261	        )
   262
   263	        # Re-enter the FuncItem and visit the body of the function this time.
   264	        gen_generator_func_body(builder, fn_info, sig, func_reg)
   265	    else:
   266	        func_ir, func_reg = gen_func_body(builder, sig, cdef, is_singledispatch)
   267
   268	    if is_singledispatch:
   269	        # add the generated main singledispatch function
   270	        builder.functions.append(func_ir)
   271	        # create the dispatch function
   272	        assert isinstance(fitem, FuncDef)
   273	        return gen_dispatch_func_ir(builder, fitem, fn_info.name, name, sig)
   274
   275	    return func_ir, func_reg

   278	def gen_func_body(
   279	    builder: IRBuilder, sig: FuncSignature, cdef: ClassDef | None, is_singledispatch: bool
   280	) -> tuple[FuncIR, Value | None]:
   281	    load_env_registers(builder)
   282	    gen_arg_defaults(builder)
   283	    if builder.fn_info.contains_nested:
   284	        finalize_env_class(builder)
   285	    add_vars_to_env(builder)
   286	    builder.accept(builder.fn_info.fitem.body)
   287	    builder.maybe_add_implicit_return()
   288
   289	    # Hang on to the local symbol table for a while, since we use it
   290	    # to calculate argument defaults below.
   291	    symtable = builder.symtables[-1]
   292
   293	    args, _, blocks, ret_type, fn_info = builder.leave()
   294
   295	    func_ir, func_reg = gen_func_ir(builder, args, blocks, sig, fn_info, cdef, is_singledispatch)
   296
   297	    # Evaluate argument defaults in the surrounding scope, since we
   298	    # calculate them *once* when the function definition is evaluated.
   299	    calculate_arg_defaults(builder, fn_info, func_reg, symtable)
   300	    return func_ir, func_reg

   306	    If a nested function only has references in the surrounding function,
   307	    we don't need to add it to the environment.

   317	def gen_func_ir(
   318	    builder: IRBuilder,
   319	    args: list[Register],
   320	    blocks: list[BasicBlock],
   321	    sig: FuncSignature,
   322	    fn_info: FuncInfo,
   323	    cdef: ClassDef | None,
   324	    is_singledispatch_main_func: bool = False,
   325	) -> tuple[FuncIR, Value | None]:
   326	    """Generate the FuncIR for a function.
   327
   328	    This takes the basic blocks and function info of a particular
   329	    function and returns the IR. If the function is nested,
   330	    also returns the register containing the instance of the
   331	    corresponding callable class.
   332	    """
   333	    func_reg: Value | None = None
   334	    if fn_info.is_nested or fn_info.in_non_ext:
   335	        func_ir = add_call_to_callable_class(builder, args, blocks, sig, fn_info)
   336	        add_get_to_callable_class(builder, fn_info)
   337	        func_reg = instantiate_callable_class(builder, fn_info)
   338	    else:
   339	        assert isinstance(fn_info.fitem, FuncDef)
   340	        func_decl = builder.mapper.func_to_decl[fn_info.fitem]
   341	        if fn_info.is_decorated or is_singledispatch_main_func:
   342	            class_name = None if cdef is None else cdef.name
   343	            func_decl = FuncDecl(
   344	                fn_info.name,
   345	                class_name,
   346	                builder.module_name,
   347	                sig,
   348	                func_decl.kind,
   349	                func_decl.is_prop_getter,
   350	                func_decl.is_prop_setter,
   351	            )
   352	            func_ir = FuncIR(
   353	                func_decl, args, blocks, fn_info.fitem.line, traceback_name=fn_info.fitem.name
   354	            )
   355	        else:
   356	            func_ir = FuncIR(
   357	                func_decl, args, blocks, fn_info.fitem.line, traceback_name=fn_info.fitem.name
   358	            )
   359	    return (func_ir, func_reg)

   362	def handle_ext_method(builder: IRBuilder, cdef: ClassDef, fdef: FuncDef) -> None:
   363	    # Perform the function of visit_method for methods inside extension classes.
   364	    name = fdef.name
   365	    class_ir = builder.mapper.type_to_ir[cdef.info]
   366	    sig = builder.mapper.fdef_to_sig(fdef, builder.options.strict_dunders_typing)
   367	    func_ir, func_reg = gen_func_item(builder, fdef, name, sig, cdef)
   368	    builder.functions.append(func_ir)
   369
   370	    if is_decorated(builder, fdef):
   371	        # Obtain the function name in order to construct the name of the helper function.
   372	        _, _, name = fdef.fullname.rpartition(".")
   373	        # Read the PyTypeObject representing the class, get the callable object
   374	        # representing the non-decorated method
   375	        typ = builder.load_native_type_object(cdef.fullname)
   376	        orig_func = builder.py_get_attr(typ, name, fdef.line)
   377
   378	        # Decorate the non-decorated method
   379	        decorated_func = load_decorated_func(builder, fdef, orig_func)
   380
   381	        # Set the callable object representing the decorated method as an attribute of the
   382	        # extension class.
   383	        builder.primitive_op(
   384	            py_setattr_op, [typ, builder.load_str(name), decorated_func], fdef.line
   385	        )
   386
   387	    if fdef.is_property:
   388	        # If there is a property setter, it will be processed after the getter,
   389	        # We populate the optional setter field with none for now.
   390	        assert name not in class_ir.properties
   391	        class_ir.properties[name] = (func_ir, None)
   392
   393	    elif fdef in builder.prop_setters:
   394	        # The respective property getter must have been processed already
   395	        assert name in class_ir.properties
   396	        getter_ir, _ = class_ir.properties[name]
   397	        class_ir.properties[name] = (getter_ir, func_ir)
   398
   399	    class_ir.methods[func_ir.decl.name] = func_ir
   400
   401	    # If this overrides a parent class method with a different type, we need
   402	    # to generate a glue method to mediate between them.
   403	    for base in class_ir.mro[1:]:
   404	        if (
   405	            name in base.method_decls
   406	            and name != "__init__"
   407	            and not is_same_method_signature(
   408	                class_ir.method_decls[name].sig, base.method_decls[name].sig
   409	            )
   410	        ):
   411	            # TODO: Support contravariant subtyping in the input argument for
   412	            # property setters. Need to make a special glue method for handling this,
   413	            # similar to gen_glue_property.
   414
   415	            f = gen_glue(builder, base.method_decls[name].sig, func_ir, class_ir, base, fdef)
   416	            class_ir.glue_methods[(base, name)] = f
   417	            builder.functions.append(f)
   418
   419	    # If the class allows interpreted children, create glue
   420	    # methods that dispatch via the Python API. These will go in a
   421	    # "shadow vtable" that will be assigned to interpreted
   422	    # children.
   423	    if class_ir.allow_interpreted_subclasses:
   424	        f = gen_glue(builder, func_ir.sig, func_ir, class_ir, class_ir, fdef, do_py_ops=True)
   425	        class_ir.glue_methods[(class_ir, name)] = f
   426	        builder.functions.append(f)

   429	def handle_non_ext_method(
   430	    builder: IRBuilder, non_ext: NonExtClassInfo, cdef: ClassDef, fdef: FuncDef
   431	) -> None:
   432	    # Perform the function of visit_method for methods inside non-extension classes.
   433	    name = fdef.name
   434	    sig = builder.mapper.fdef_to_sig(fdef, builder.options.strict_dunders_typing)
   435	    func_ir, func_reg = gen_func_item(builder, fdef, name, sig, cdef)
   436	    assert func_reg is not None
   437	    builder.functions.append(func_ir)
   438
   439	    if is_decorated(builder, fdef):
   440	        # The undecorated method is a generated callable class
   441	        orig_func = func_reg
   442	        func_reg = load_decorated_func(builder, fdef, orig_func)
   443
   444	    # TODO: Support property setters in non-extension classes
   445	    if fdef.is_property:
   446	        prop = builder.load_module_attr_by_fullname("builtins.property", fdef.line)
   447	        func_reg = builder.py_call(prop, [func_reg], fdef.line)
   448
   449	    elif builder.mapper.func_to_decl[fdef].kind == FUNC_CLASSMETHOD:
   450	        cls_meth = builder.load_module_attr_by_fullname("builtins.classmethod", fdef.line)
   451	        func_reg = builder.py_call(cls_meth, [func_reg], fdef.line)
   452
   453	    elif builder.mapper.func_to_decl[fdef].kind == FUNC_STATICMETHOD:
   454	        stat_meth = builder.load_module_attr_by_fullname("builtins.staticmethod", fdef.line)
   455	        func_reg = builder.py_call(stat_meth, [func_reg], fdef.line)

   457	    builder.add_to_non_ext_dict(non_ext, name, func_reg, fdef.line)

   469	def load_decorated_func(builder: IRBuilder, fdef: FuncDef, orig_func_reg: Value) -> Value:
   470	    """Apply decorators to a function.
   471
   472	    Given a decorated FuncDef and an instance of the callable class
   473	    representing that FuncDef, apply the corresponding decorator
   474	    functions on that decorated FuncDef and return the decorated
   475	    function.
   476	    """
   477	    if not is_decorated(builder, fdef):
   478	        # If there are no decorators associated with the function, then just return the
   479	        # original function.
   480	        return orig_func_reg
   481
   482	    decorators = builder.fdefs_to_decorators[fdef]
   483	    func_reg = orig_func_reg
   484	    for d in reversed(decorators):
   485	        decorator = d.accept(builder.visitor)
   486	        assert isinstance(decorator, Value)
   487	        func_reg = builder.py_call(decorator, [func_reg], func_reg.line)
   488	    return func_reg

   541	def gen_glue_method(
   542	    builder: IRBuilder,
   543	    base_sig: FuncSignature,
   544	    target: FuncIR,
   545	    cls: ClassIR,
   546	    base: ClassIR,
   547	    line: int,
   548	    do_pycall: bool,
   549	) -> FuncIR:
   550	    """Generate glue methods that mediate between different method types in subclasses.
   551
   552	    For example, if we have:
   553
   554	    class A:
   555	        def f(builder: IRBuilder, x: int) -> object: ...
   556
   557	    then it is totally permissible to have a subclass
   558
   559	    class B(A):
   560	        def f(builder: IRBuilder, x: object) -> int: ...
   561
   562	    since '(object) -> int' is a subtype of '(int) -> object' by the usual
   563	    contra/co-variant function subtyping rules.
   564
   565	    The trickiness here is that int and object have different
   566	    runtime representations in mypyc, so A.f and B.f have
   567	    different signatures at the native C level. To deal with this,
   568	    we need to generate glue methods that mediate between the
   569	    different versions by coercing the arguments and return
   570	    values.
   571
   572	    If do_pycall is True, then make the call using the C API
   573	    instead of a native call.
   574	    """
   575	    check_native_override(builder, base_sig, target.decl.sig, line)
   576
   577	    builder.enter()
   578	    builder.ret_types[-1] = base_sig.ret_type
   579
   580	    rt_args = list(base_sig.args)
   581	    if target.decl.kind == FUNC_NORMAL:
   582	        rt_args[0] = RuntimeArg(base_sig.args[0].name, RInstance(cls))
   583
   584	    arg_info = get_args(builder, rt_args, line)
   585	    args, arg_kinds, arg_names = arg_info.args, arg_info.arg_kinds, arg_info.arg_names
   586
   587	    bitmap_args = None
   588	    if base_sig.num_bitmap_args:
   589	        args = args[: -base_sig.num_bitmap_args]
   590	        arg_kinds = arg_kinds[: -base_sig.num_bitmap_args]
   591	        arg_names = arg_names[: -base_sig.num_bitmap_args]
   592	        bitmap_args = list(builder.builder.args[-base_sig.num_bitmap_args :])
   593
   594	    # We can do a passthrough *args/**kwargs with a native call, but if the
   595	    # args need to get distributed out to arguments, we just let python handle it
   596	    if any(kind.is_star() for kind in arg_kinds) and any(
   597	        not arg.kind.is_star() for arg in target.decl.sig.args
   598	    ):
   599	        do_pycall = True
   600
   601	    if do_pycall:
   602	        if target.decl.kind == FUNC_STATICMETHOD:
   603	            # FIXME: this won't work if we can do interpreted subclasses
   604	            first = builder.builder.get_native_type(cls)
   605	            st = 0
   606	        else:
   607	            first = args[0]
   608	            st = 1
   609	        retval = builder.builder.py_method_call(
   610	            first, target.name, args[st:], line, arg_kinds[st:], arg_names[st:]
   611	        )
   612	    else:
   613	        retval = builder.builder.call(
   614	            target.decl, args, arg_kinds, arg_names, line, bitmap_args=bitmap_args
   615	        )
   616	    retval = builder.coerce(retval, base_sig.ret_type, line)
   617	    builder.add(Return(retval))
   618
   619	    arg_regs, _, blocks, ret_type, _ = builder.leave()
   620	    if base_sig.num_bitmap_args:
   621	        rt_args = rt_args[: -base_sig.num_bitmap_args]
   622	    return FuncIR(
   623	        FuncDecl(
   624	            target.name + "__" + base.name + "_glue",
   625	            cls.name,
   626	            builder.module_name,
   627	            FuncSignature(rt_args, ret_type),
   628	            target.decl.kind,
   629	        ),
   630	        arg_regs,
   631	        blocks,
   632	    )

   635	def check_native_override(
   636	    builder: IRBuilder, base_sig: FuncSignature, sub_sig: FuncSignature, line: int
   637	) -> None:
   638	    """Report an error if an override changes signature in unsupported ways.
   639
   640	    Glue methods can work around many signature changes but not all of them.
   641	    """
   642	    for base_arg, sub_arg in zip(base_sig.real_args(), sub_sig.real_args()):
   643	        if base_arg.type.error_overlap:
   644	            if not base_arg.optional and sub_arg.optional and base_sig.num_bitmap_args:
   645	                # This would change the meanings of bits in the argument defaults
   646	                # bitmap, which we don't support. We'd need to do tricky bit
   647	                # manipulations to support this generally.
   648	                builder.error(
   649	                    "An argument with type "
   650	                    + f'"{base_arg.type}" cannot be given a default value in a method override',
   651	                    line,
   652	                )
   653	        if base_arg.type.error_overlap or sub_arg.type.error_overlap:
   654	            if not is_same_type(base_arg.type, sub_arg.type):
   655	                # This would change from signaling a default via an error value to
   656	                # signaling a default via bitmap, which we don't support.
   657	                builder.error(
   658	                    "Incompatible argument type "
   659	                    + f'"{sub_arg.type}" (base class has type "{base_arg.type}")',
   660	                    line,
   661	                )

   664	def gen_glue_property(
   665	    builder: IRBuilder,
   666	    sig: FuncSignature,
   667	    target: FuncIR,
   668	    cls: ClassIR,
   669	    base: ClassIR,
   670	    line: int,
   671	    do_pygetattr: bool,
   672	) -> FuncIR:
   673	    """Generate glue methods for properties that mediate between different subclass types.
   674
   675	    Similarly to methods, properties of derived types can be covariantly subtyped. Thus,
   676	    properties also require glue. However, this only requires the return type to change.
   677	    Further, instead of a method call, an attribute get is performed.
   678
   679	    If do_pygetattr is True, then get the attribute using the Python C
   680	    API instead of a native call.
   681	    """
   682	    builder.enter()
   683
   684	    rt_arg = RuntimeArg(SELF_NAME, RInstance(cls))
   685	    self_target = builder.add_self_to_env(cls)
   686	    arg = builder.read(self_target, line)
   687	    builder.ret_types[-1] = sig.ret_type
   688	    if do_pygetattr:
   689	        retval = builder.py_get_attr(arg, target.name, line)
   690	    else:
   691	        retval = builder.add(GetAttr(arg, target.name, line))
   692	    retbox = builder.coerce(retval, sig.ret_type, line)
   693	    builder.add(Return(retbox))
   694
   695	    args, _, blocks, return_type, _ = builder.leave()
   696	    return FuncIR(
   697	        FuncDecl(
   698	            target.name + "__" + base.name + "_glue",
   699	            cls.name,
   700	            builder.module_name,
   701	            FuncSignature([rt_arg], return_type),
   702	        ),
   703	        args,
   704	        blocks,
   705	    )

   724	# This function still does not support the following imports.
   725	# import json as _json
   726	# from json import decoder
   727	# Using either _json.JSONDecoder or decoder.JSONDecoder as a type hint for a dataclass field will fail.
   728	# See issue mypyc/mypyc#1099.
   729	def load_type(builder: IRBuilder, typ: TypeInfo, unbounded_type: Type | None, line: int) -> Value:
   730	    # typ.fullname contains the module where the class object was defined. However, it is possible
   731	    # that the class object's module was not imported in the file currently being compiled. So, we
   732	    # use unbounded_type.name (if provided by caller) to load the class object through one of the
   733	    # imported modules.
   734	    # Example: for `json.JSONDecoder`, typ.fullname is `json.decoder.JSONDecoder` but the Python
   735	    # file may import `json` not `json.decoder`.
   736	    # Another corner case: The Python file being compiled imports mod1 and has a type hint
   737	    # `mod1.OuterClass.InnerClass`. But, mod1/__init__.py might import OuterClass like this:
   738	    # `from mod2.mod3 import OuterClass`. In this case, typ.fullname is
   739	    # `mod2.mod3.OuterClass.InnerClass` and `unbounded_type.name` is `mod1.OuterClass.InnerClass`.
   740	    # So, we must use unbounded_type.name to load the class object.
   741	    # See issue mypyc/mypyc#1087.
   742	    load_attr_path = (
   743	        unbounded_type.name if isinstance(unbounded_type, UnboundType) else typ.fullname
   744	    ).removesuffix(f".{typ.name}")
   745	    if typ in builder.mapper.type_to_ir:
   746	        class_ir = builder.mapper.type_to_ir[typ]
   747	        class_obj = builder.builder.get_native_type(class_ir)
   748	    elif typ.fullname in builtin_names:
   749	        builtin_addr_type, src = builtin_names[typ.fullname]
   750	        class_obj = builder.add(LoadAddress(builtin_addr_type, src, line))
   751	    # This elif-condition finds the longest import that matches the load_attr_path.
   752	    elif module_name := max(
   753	        (i for i in builder.imports if load_attr_path == i or load_attr_path.startswith(f"{i}.")),
   754	        default="",
   755	        key=len,
   756	    ):
   757	        # Load the imported module.
   758	        loaded_module = builder.load_module(module_name)
   759	        # Recursively load attributes of the imported module. These may be submodules, classes or
   760	        # any other object.
   761	        for attr in (
   762	            load_attr_path.removeprefix(f"{module_name}.").split(".")
   763	            if load_attr_path != module_name
   764	            else []
   765	        ):
   766	            loaded_module = builder.py_get_attr(loaded_module, attr, line)
   767	        class_obj = builder.builder.get_attr(
   768	            loaded_module, typ.name, object_rprimitive, line, borrow=False
   769	        )
   770	    else:
   771	        class_obj = builder.load_global_str(typ.name, line)
   772
   773	    return class_obj

   776	def load_func(builder: IRBuilder, func_name: str, fullname: str | None, line: int) -> Value:
   777	    if fullname and not fullname.startswith(builder.current_module):
   778	        # we're calling a function in a different module
   779
   780	        # We can't use load_module_attr_by_fullname here because we need to load the function using
   781	        # func_name, not the name specified by fullname (which can be different for underscore
   782	        # function)
   783	        module = fullname.rsplit(".")[0]
   784	        loaded_module = builder.load_module(module)
   785
   786	        func = builder.py_get_attr(loaded_module, func_name, line)
   787	    else:
   788	        func = builder.load_global_str(func_name, line)
   789	    return func

   792	def generate_singledispatch_dispatch_function(
   793	    builder: IRBuilder, main_singledispatch_function_name: str, fitem: FuncDef
   794	) -> None:
   795	    line = fitem.line
   796	    current_func_decl = builder.mapper.func_to_decl[fitem]
   797	    arg_info = get_args(builder, current_func_decl.sig.args, line)
   798
   799	    dispatch_func_obj = builder.self()
   800
   801	    arg_type = builder.builder.get_type_of_obj(arg_info.args[0], line)
   802	    dispatch_cache = builder.builder.get_attr(
   803	        dispatch_func_obj, "dispatch_cache", dict_rprimitive, line
   804	    )
   805	    call_find_impl, use_cache, call_func = BasicBlock(), BasicBlock(), BasicBlock()
   806	    get_result = builder.primitive_op(dict_get_method_with_none, [dispatch_cache, arg_type], line)
   807	    is_not_none = builder.translate_is_op(get_result, builder.none_object(), "is not", line)
   808	    impl_to_use = Register(object_rprimitive)
   809	    builder.add_bool_branch(is_not_none, use_cache, call_find_impl)
   810
   811	    builder.activate_block(use_cache)
   812	    builder.assign(impl_to_use, get_result, line)
   813	    builder.goto(call_func)
   814
   815	    builder.activate_block(call_find_impl)
   816	    find_impl = builder.load_module_attr_by_fullname("functools._find_impl", line)
   817	    registry = load_singledispatch_registry(builder, dispatch_func_obj, line)
   818	    uncached_impl = builder.py_call(find_impl, [arg_type, registry], line)
   819	    builder.primitive_op(dict_set_item_op, [dispatch_cache, arg_type, uncached_impl], line)
   820	    builder.assign(impl_to_use, uncached_impl, line)
   821	    builder.goto(call_func)
   822
   823	    builder.activate_block(call_func)
   824	    gen_calls_to_correct_impl(builder, impl_to_use, arg_info, fitem, line)

   827	def gen_calls_to_correct_impl(
   828	    builder: IRBuilder, impl_to_use: Value, arg_info: ArgInfo, fitem: FuncDef, line: int
   829	) -> None:
   830	    current_func_decl = builder.mapper.func_to_decl[fitem]
   831
   832	    def gen_native_func_call_and_return(fdef: FuncDef) -> None:
   833	        func_decl = builder.mapper.func_to_decl[fdef]
   834	        ret_val = builder.builder.call(
   835	            func_decl, arg_info.args, arg_info.arg_kinds, arg_info.arg_names, line
   836	        )
   837	        coerced = builder.coerce(ret_val, current_func_decl.sig.ret_type, line)
   838	        builder.add(Return(coerced))
   839
   840	    typ, src = builtin_names["builtins.int"]
   841	    int_type_obj = builder.add(LoadAddress(typ, src, line))
   842	    is_int = builder.builder.type_is_op(impl_to_use, int_type_obj, line)
   843
   844	    native_call, non_native_call = BasicBlock(), BasicBlock()
   845	    builder.add_bool_branch(is_int, native_call, non_native_call)
   846	    builder.activate_block(native_call)
   847
   848	    passed_id = builder.add(Unbox(impl_to_use, int_rprimitive, line))
   849
   850	    native_ids = get_native_impl_ids(builder, fitem)
   851	    for impl, i in native_ids.items():
   852	        call_impl, next_impl = BasicBlock(), BasicBlock()
   853
   854	        current_id = builder.load_int(i)
   855	        cond = builder.binary_op(passed_id, current_id, "==", line)
   856	        builder.add_bool_branch(cond, call_impl, next_impl)
   857
   858	        # Call the registered implementation
   859	        builder.activate_block(call_impl)
   860
   861	        gen_native_func_call_and_return(impl)
   862	        builder.activate_block(next_impl)
   863
   864	    # We've already handled all the possible integer IDs, so we should never get here
   865	    builder.add(Unreachable())
   866
   867	    builder.activate_block(non_native_call)
   868	    ret_val = builder.py_call(
   869	        impl_to_use, arg_info.args, line, arg_info.arg_kinds, arg_info.arg_names
   870	    )
   871	    coerced = builder.coerce(ret_val, current_func_decl.sig.ret_type, line)
   872	    builder.add(Return(coerced))
