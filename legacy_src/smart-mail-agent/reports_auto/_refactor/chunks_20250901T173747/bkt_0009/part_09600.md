   315	class LineCoverageVisitor(TraverserVisitor):
   316	    def __init__(self, source: list[str]) -> None:
   317	        self.source = source
   318
   319	        # For each line of source, we maintain a pair of
   320	        #  * the indentation level of the surrounding function
   321	        #    (-1 if not inside a function), and
   322	        #  * whether the surrounding function is typed.
   323	        # Initially, everything is covered at indentation level -1.
   324	        self.lines_covered = [(-1, True) for l in source]
   325
   326	    # The Python AST has position information for the starts of
   327	    # elements, but not for their ends. Fortunately the
   328	    # indentation-based syntax makes it pretty easy to find where a
   329	    # block ends without doing any real parsing.
   330
   331	    # TODO: Handle line continuations (explicit and implicit) and
   332	    # multi-line string literals. (But at least line continuations
   333	    # are normally more indented than their surrounding block anyways,
   334	    # by PEP 8.)
   335
   336	    def indentation_level(self, line_number: int) -> int | None:
   337	        """Return the indentation of a line of the source (specified by
   338	        zero-indexed line number). Returns None for blank lines or comments."""
   339	        line = self.source[line_number]
   340	        indent = 0
   341	        for char in list(line):
   342	            if char == " ":
   343	                indent += 1
   344	            elif char == "\t":
   345	                indent = 8 * ((indent + 8) // 8)
   346	            elif char == "#":
   347	                # Line is a comment; ignore it
   348	                return None
   349	            elif char == "\n":
   350	                # Line is entirely whitespace; ignore it
   351	                return None
   352	            # TODO line continuation (\)
   353	            else:
   354	                # Found a non-whitespace character
   355	                return indent
   356	        # Line is entirely whitespace, and at end of file
   357	        # with no trailing newline; ignore it
   358	        return None
   359
   360	    def visit_func_def(self, defn: FuncDef) -> None:
   361	        start_line = defn.line - 1
   362	        start_indent = None
   363	        # When a function is decorated, sometimes the start line will point to
   364	        # whitespace or comments between the decorator and the function, so
   365	        # we have to look for the start.
   366	        while start_line < len(self.source):
   367	            start_indent = self.indentation_level(start_line)
   368	            if start_indent is not None:
   369	                break
   370	            start_line += 1
   371	        # If we can't find the function give up and don't annotate anything.
   372	        # Our line numbers are not reliable enough to be asserting on.
   373	        if start_indent is None:
   374	            return
   375
   376	        cur_line = start_line + 1
   377	        end_line = cur_line
   378	        # After this loop, function body will be lines [start_line, end_line)
   379	        while cur_line < len(self.source):
   380	            cur_indent = self.indentation_level(cur_line)
   381	            if cur_indent is None:
   382	                # Consume the line, but don't mark it as belonging to the function yet.
   383	                cur_line += 1
   384	            elif cur_indent > start_indent:
   385	                # A non-blank line that belongs to the function.
   386	                cur_line += 1
   387	                end_line = cur_line
   388	            else:
   389	                # We reached a line outside the function definition.
   390	                break
   391
   392	        is_typed = defn.type is not None
   393	        for line in range(start_line, end_line):
   394	            old_indent, _ = self.lines_covered[line]
   395	            # If there was an old indent level for this line, and the new
   396	            # level isn't increasing the indentation, ignore it.
   397	            # This is to be defensive against funniness in our line numbers,
   398	            # which are not always reliable.
   399	            if old_indent <= start_indent:
   400	                self.lines_covered[line] = (start_indent, is_typed)
   401
   402	        # Visit the body, in case there are nested functions
   403	        super().visit_func_def(defn)

   406	class LineCoverageReporter(AbstractReporter):
   407	    """Exact line coverage reporter.
   408
   409	    This reporter writes a JSON dictionary with one field 'lines' to
   410	    the file 'coverage.json' in the specified report directory. The
   411	    value of that field is a dictionary which associates to each
   412	    source file's absolute pathname the list of line numbers that
   413	    belong to typed functions in that file.
   414	    """
   415
   416	    def __init__(self, reports: Reports, output_dir: str) -> None:
   417	        super().__init__(reports, output_dir)
   418	        self.lines_covered: dict[str, list[int]] = {}
   419
   420	    def on_file(
   421	        self,
   422	        tree: MypyFile,
   423	        modules: dict[str, MypyFile],
   424	        type_map: dict[Expression, Type],
   425	        options: Options,
   426	    ) -> None:
   427	        with open(tree.path) as f:
   428	            tree_source = f.readlines()

   430	        coverage_visitor = LineCoverageVisitor(tree_source)

   435	            if typed:
   436	                covered_lines.append(line_number + 1)

   461	class MemoryXmlReporter(AbstractReporter):
   462	    """Internal reporter that generates XML in memory.
   463
   464	    This is used by all other XML-based reporters to avoid duplication.
   465	    """
   466
   467	    def __init__(self, reports: Reports, output_dir: str) -> None:
   468	        super().__init__(reports, output_dir)
   469
   470	        self.xslt_html_path = os.path.join(reports.data_dir, "xml", "mypy-html.xslt")
   471	        self.xslt_txt_path = os.path.join(reports.data_dir, "xml", "mypy-txt.xslt")
   472	        self.css_html_path = os.path.join(reports.data_dir, "xml", "mypy-html.css")
   473	        xsd_path = os.path.join(reports.data_dir, "xml", "mypy.xsd")
   474	        self.schema = etree.XMLSchema(etree.parse(xsd_path))
   475	        self.last_xml: Any | None = None
   476	        self.files: list[FileInfo] = []
   477
   478	    # XML doesn't like control characters, but they are sometimes
   479	    # legal in source code (e.g. comments, string literals).
   480	    # Tabs (#x09) are allowed in XML content.
   481	    control_fixer: Final = str.maketrans("".join(chr(i) for i in range(32) if i != 9), "?" * 31)
   482
   483	    def on_file(
   484	        self,
   485	        tree: MypyFile,
   486	        modules: dict[str, MypyFile],
   487	        type_map: dict[Expression, Type],
   488	        options: Options,
   489	    ) -> None:
   490	        self.last_xml = None
   491
   492	        try:
   493	            path = os.path.relpath(tree.path)
   494	        except ValueError:
   495	            return
   496
   497	        if should_skip_path(path) or os.path.isdir(path):
   498	            return  # `path` can sometimes be a directory, see #11334
   499
   500	        visitor = stats.StatisticsVisitor(
   501	            inferred=True,
   502	            filename=tree.fullname,
   503	            modules=modules,
   504	            typemap=type_map,
   505	            all_nodes=True,
   506	        )
   507	        tree.accept(visitor)
   508
   509	        root = etree.Element("mypy-report-file", name=path, module=tree._fullname)
   510	        doc = etree.ElementTree(root)
   511	        file_info = FileInfo(path, tree._fullname)
   512
   513	        for lineno, line_text in iterate_python_lines(path):
   514	            status = visitor.line_map.get(lineno, stats.TYPE_EMPTY)
   515	            file_info.counts[status] += 1
   516	            etree.SubElement(
   517	                root,
   518	                "line",
   519	                any_info=self._get_any_info_for_line(visitor, lineno),
   520	                content=line_text.rstrip("\n").translate(self.control_fixer),
   521	                number=str(lineno),
   522	                precision=stats.precision_names[status],
   523	            )
   524	        # Assumes a layout similar to what XmlReporter uses.
   525	        xslt_path = os.path.relpath("mypy-html.xslt", path)
   526	        transform_pi = etree.ProcessingInstruction(
   527	            "xml-stylesheet", f'type="text/xsl" href="{pathname2url(xslt_path)}"'
   528	        )
   529	        root.addprevious(transform_pi)
   530	        self.schema.assertValid(doc)
   531
   532	        self.last_xml = doc
   533	        self.files.append(file_info)
   534
   535	    @staticmethod
   536	    def _get_any_info_for_line(visitor: stats.StatisticsVisitor, lineno: int) -> str:
   537	        if lineno in visitor.any_line_map:
   538	            result = "Any Types on this line: "
   539	            counter: collections.Counter[int] = collections.Counter()
   540	            for typ in visitor.any_line_map[lineno]:
   541	                counter[typ.type_of_any] += 1
   542	            for any_type, occurrences in counter.items():
   543	                result += f"\n{type_of_any_name_map[any_type]} (x{occurrences})"
   544	            return result
   545	        else:
   546	            return "No Anys on this line!"
   547
   548	    def on_finish(self) -> None:
   549	        self.last_xml = None
   550	        # index_path = os.path.join(self.output_dir, 'index.xml')
   551	        output_files = sorted(self.files, key=lambda x: x.module)
   552
   553	        root = etree.Element("mypy-report-index", name="index")
   554	        doc = etree.ElementTree(root)
   555
   556	        for file_info in output_files:
   557	            etree.SubElement(
   558	                root,
   559	                "file",
   560	                file_info.attrib(),
   561	                module=file_info.module,
   562	                name=pathname2url(file_info.name),
   563	                total=str(file_info.total()),
   564	            )
   565	        xslt_path = os.path.relpath("mypy-html.xslt", ".")
   566	        transform_pi = etree.ProcessingInstruction(
   567	            "xml-stylesheet", f'type="text/xsl" href="{pathname2url(xslt_path)}"'
   568	        )
   569	        root.addprevious(transform_pi)
   570	        self.schema.assertValid(doc)
   571
   572	        self.last_xml = doc

   612	class CoberturaXmlReporter(AbstractReporter):
   613	    """Reporter for generating Cobertura compliant XML."""
   614
   615	    def __init__(self, reports: Reports, output_dir: str) -> None:
   616	        super().__init__(reports, output_dir)
   617
   618	        self.root = etree.Element("coverage", timestamp=str(int(time.time())), version=__version__)
   619	        self.doc = etree.ElementTree(self.root)
   620	        self.root_package = CoberturaPackage(".")
   621
   622	    def on_file(
   623	        self,
   624	        tree: MypyFile,
   625	        modules: dict[str, MypyFile],
   626	        type_map: dict[Expression, Type],
   627	        options: Options,
   628	    ) -> None:
   629	        path = os.path.relpath(tree.path)
   630	        visitor = stats.StatisticsVisitor(
   631	            inferred=True,
   632	            filename=tree.fullname,
   633	            modules=modules,
   634	            typemap=type_map,
   635	            all_nodes=True,
   636	        )
   637	        tree.accept(visitor)
   638
   639	        class_name = os.path.basename(path)
   640	        file_info = FileInfo(path, tree._fullname)
   641	        class_element = etree.Element("class", complexity="1.0", filename=path, name=class_name)
   642	        etree.SubElement(class_element, "methods")
   643	        lines_element = etree.SubElement(class_element, "lines")
   644
   645	        class_lines_covered = 0
   646	        class_total_lines = 0
   647	        for lineno, _ in iterate_python_lines(path):
   648	            status = visitor.line_map.get(lineno, stats.TYPE_EMPTY)
   649	            hits = 0
   650	            branch = False
   651	            if status == stats.TYPE_EMPTY:
   652	                continue
   653	            class_total_lines += 1
   654	            if status != stats.TYPE_ANY:
   655	                class_lines_covered += 1
   656	                hits = 1
   657	            if status == stats.TYPE_IMPRECISE:
   658	                branch = True
   659	            file_info.counts[status] += 1
   660	            line_element = etree.SubElement(
   661	                lines_element,
   662	                "line",
   663	                branch=str(branch).lower(),
   664	                hits=str(hits),
   665	                number=str(lineno),
   666	                precision=stats.precision_names[status],
   667	            )
   668	            if branch:
   669	                line_element.attrib["condition-coverage"] = "50% (1/2)"
   670	        class_element.attrib["branch-rate"] = "0"
   671	        class_element.attrib["line-rate"] = get_line_rate(class_lines_covered, class_total_lines)
   672	        # parent_module is set to whichever module contains this file.  For most files, we want
   673	        # to simply strip the last element off of the module.  But for __init__.py files,
   674	        # the module == the parent module.
   675	        parent_module = file_info.module.rsplit(".", 1)[0]
   676	        if file_info.name.endswith("__init__.py"):
   677	            parent_module = file_info.module
   678
   679	        if parent_module not in self.root_package.packages:
   680	            self.root_package.packages[parent_module] = CoberturaPackage(parent_module)
   681	        current_package = self.root_package.packages[parent_module]
   682	        packages_to_update = [self.root_package, current_package]
   683	        for package in packages_to_update:
   684	            package.total_lines += class_total_lines
   685	            package.covered_lines += class_lines_covered
   686	        current_package.classes[class_name] = class_element
   687
   688	    def on_finish(self) -> None:
   689	        self.root.attrib["line-rate"] = get_line_rate(
   690	            self.root_package.covered_lines, self.root_package.total_lines
   691	        )
   692	        self.root.attrib["branch-rate"] = "0"
   693	        self.root.attrib["lines-covered"] = str(self.root_package.covered_lines)
   694	        self.root.attrib["lines-valid"] = str(self.root_package.total_lines)
   695	        sources = etree.SubElement(self.root, "sources")
   696	        source_element = etree.SubElement(sources, "source")
   697	        source_element.text = os.getcwd()
   698	        self.root_package.add_packages(self.root)
   699	        out_path = os.path.join(self.output_dir, "cobertura.xml")
   700	        self.doc.write(out_path, encoding="utf-8", pretty_print=True)
   701	        print("Generated Cobertura report:", os.path.abspath(out_path))

   550	    def prepare_file(self, file_node: MypyFile) -> None:
   551	        """Prepare a freshly parsed file for semantic analysis."""
   552	        if "builtins" in self.modules:
   553	            file_node.names["__builtins__"] = SymbolTableNode(GDEF, self.modules["builtins"])
   554	        if file_node.fullname == "builtins":
   555	            self.prepare_builtins_namespace(file_node)
   556	        if file_node.fullname == "typing":
   557	            self.prepare_typing_namespace(file_node, type_aliases)

   561	    def prepare_typing_namespace(self, file_node: MypyFile, aliases: dict[str, str]) -> None:
   562	        """Remove dummy alias definitions such as List = TypeAlias(object) from typing.
   563
   564	        They will be replaced with real aliases when corresponding targets are ready.
   565	        """
   566
   567	        # This is all pretty unfortunate. typeshed now has a
   568	        # sys.version_info check for OrderedDict, and we shouldn't
   569	        # take it out, because it is correct and a typechecker should
   570	        # use that as a source of truth. But instead we rummage
   571	        # through IfStmts to remove the info first.  (I tried to
   572	        # remove this whole machinery and ran into issues with the
   573	        # builtins/typing import cycle.)
   574	        def helper(defs: list[Statement]) -> None:
   575	            for stmt in defs.copy():
   576	                if isinstance(stmt, IfStmt):
   577	                    for body in stmt.body:
   578	                        helper(body.body)
   579	                    if stmt.else_body:
   580	                        helper(stmt.else_body.body)
   581	                if (
   582	                    isinstance(stmt, AssignmentStmt)
   583	                    and len(stmt.lvalues) == 1
   584	                    and isinstance(stmt.lvalues[0], NameExpr)
   585	                ):
   586	                    # Assignment to a simple name, remove it if it is a dummy alias.
   587	                    if f"{file_node.fullname}.{stmt.lvalues[0].name}" in aliases:
   588	                        defs.remove(stmt)
   589
   590	        helper(file_node.defs)

   592	    def prepare_builtins_namespace(self, file_node: MypyFile) -> None:
   593	        """Add certain special-cased definitions to the builtins module.
   594
   595	        Some definitions are too special or fundamental to be processed
   596	        normally from the AST.
   597	        """
   598	        names = file_node.names
   599
   600	        # Add empty definition for core built-in classes, since they are required for basic
   601	        # operation. These will be completed later on.
   602	        for name in CORE_BUILTIN_CLASSES:
   603	            cdef = ClassDef(name, Block([]))  # Dummy ClassDef, will be replaced later
   604	            info = TypeInfo(SymbolTable(), cdef, "builtins")
   605	            info._fullname = f"builtins.{name}"
   606	            names[name] = SymbolTableNode(GDEF, info)
   607
   608	        bool_info = names["bool"].node
   609	        assert isinstance(bool_info, TypeInfo)
   610	        bool_type = Instance(bool_info, [])
   611
   612	        special_var_types: list[tuple[str, Type]] = [
   613	            ("None", NoneType()),
   614	            # reveal_type is a mypy-only function that gives an error with
   615	            # the type of its arg.
   616	            ("reveal_type", AnyType(TypeOfAny.special_form)),
   617	            # reveal_locals is a mypy-only function that gives an error with the types of
   618	            # locals
   619	            ("reveal_locals", AnyType(TypeOfAny.special_form)),
   620	            ("True", bool_type),
   621	            ("False", bool_type),
   622	            ("__debug__", bool_type),
   623	        ]
   624
   625	        for name, typ in special_var_types:
   626	            v = Var(name, typ)
   627	            v._fullname = f"builtins.{name}"
   628	            file_node.names[name] = SymbolTableNode(GDEF, v)

   634	    def refresh_partial(
   635	        self,
   636	        node: MypyFile | FuncDef | OverloadedFuncDef,
   637	        patches: list[tuple[int, Callable[[], None]]],
   638	        final_iteration: bool,
   639	        file_node: MypyFile,
   640	        options: Options,
   641	        active_type: TypeInfo | None = None,
   642	    ) -> None:
   643	        """Refresh a stale target in fine-grained incremental mode."""
   644	        self.patches = patches
   645	        self.deferred = False
   646	        self.incomplete = False
   647	        self._final_iteration = final_iteration
   648	        self.missing_names[-1] = set()
   649
   650	        with self.file_context(file_node, options, active_type):
   651	            if isinstance(node, MypyFile):
   652	                self.refresh_top_level(node)
   653	            else:

   655	                self.accept(node)

   658	    def refresh_top_level(self, file_node: MypyFile) -> None:
   659	        """Reanalyze a stale module top-level in fine-grained incremental mode."""
   660	        if self.options.allow_redefinition_new and not self.options.local_partial_types:
   661	            n = TempNode(AnyType(TypeOfAny.special_form))
   662	            n.line = 1
   663	            n.column = 0
   664	            n.end_line = 1
   665	            n.end_column = 0
   666	            self.fail("--local-partial-types must be enabled if using --allow-redefinition-new", n)
   667	        self.recurse_into_functions = False
   668	        self.add_implicit_module_attrs(file_node)
   669	        for d in file_node.defs:
   670	            self.accept(d)
   671	        if file_node.fullname == "typing":
   672	            self.add_builtin_aliases(file_node)
   673	        if file_node.fullname == "typing_extensions":
   674	            self.add_typing_extension_aliases(file_node)
   675	        self.adjust_public_exports()
   676	        self.export_map[self.cur_mod_id] = self.all_exports
   677	        self.all_exports = []

   679	    def add_implicit_module_attrs(self, file_node: MypyFile) -> None:
   680	        """Manually add implicit definitions of module '__name__' etc."""
   681	        str_type: Type | None = self.named_type_or_none("builtins.str")
   682	        if str_type is None:
   683	            str_type = UnboundType("builtins.str")
   684	        inst: Type | None
   685	        for name, t in implicit_module_attrs.items():
   686	            if name == "__doc__":
   687	                typ: Type = str_type
   688	            elif name == "__path__":
   689	                if not file_node.is_package_init_file():
   690	                    continue
   691	                # Need to construct the type ourselves, to avoid issues with __builtins__.list
   692	                # not being subscriptable or typing.List not getting bound
   693	                inst = self.named_type_or_none("builtins.list", [str_type])
   694	                if inst is None:
   695	                    assert not self.final_iteration, "Cannot find builtins.list to add __path__"
   696	                    self.defer()
   697	                    return
   698	                typ = inst
   699	            elif name == "__annotations__":
   700	                inst = self.named_type_or_none(
   701	                    "builtins.dict", [str_type, AnyType(TypeOfAny.special_form)]
   702	                )
   703	                if inst is None:
   704	                    assert (
   705	                        not self.final_iteration
   706	                    ), "Cannot find builtins.dict to add __annotations__"
   707	                    self.defer()
   708	                    return
   709	                typ = inst
   710	            elif name == "__spec__":
   711	                if self.options.use_builtins_fixtures:
   712	                    inst = self.named_type_or_none("builtins.object")
   713	                else:
   714	                    inst = self.named_type_or_none("importlib.machinery.ModuleSpec")
   715	                if inst is None:
   716	                    if self.final_iteration:
   717	                        inst = self.named_type_or_none("builtins.object")
   718	                        assert inst is not None, "Cannot find builtins.object"
   719	                    else:
   720	                        self.defer()
   721	                        return
   722	                if file_node.name == "__main__":
   723	                    # https://docs.python.org/3/reference/import.html#main-spec
   724	                    inst = UnionType.make_union([inst, NoneType()])
   725	                typ = inst
   726	            else:
   727	                assert t is not None, f"type should be specified for {name}"
   728	                typ = UnboundType(t)
   729
   730	            existing = file_node.names.get(name)
   731	            if existing is not None and not isinstance(existing.node, PlaceholderNode):
   732	                # Already exists.
   733	                continue
   734
   735	            an_type = self.anal_type(typ)
   736	            if an_type:
   737	                var = Var(name, an_type)
   738	                var._fullname = self.qualified_name(name)
   739	                var.is_ready = True
   740	                self.add_symbol(name, var, dummy_context())
   741	            else:
   742	                self.add_symbol(
   743	                    name,
   744	                    PlaceholderNode(self.qualified_name(name), file_node, -1),
   745	                    dummy_context(),
   746	                )

   748	    def add_builtin_aliases(self, tree: MypyFile) -> None:
   749	        """Add builtin type aliases to typing module.
   750
   751	        For historical reasons, the aliases like `List = list` are not defined
   752	        in typeshed stubs for typing module. Instead we need to manually add the
   753	        corresponding nodes on the fly. We explicitly mark these aliases as normalized,
   754	        so that a user can write `typing.List[int]`.
   755	        """
   756	        assert tree.fullname == "typing"
   757	        for alias, target_name in type_aliases.items():
   758	            if (
   759	                alias in type_aliases_source_versions
   760	                and type_aliases_source_versions[alias] > self.options.python_version
   761	            ):
   762	                # This alias is not available on this Python version.
   763	                continue
   764	            name = alias.split(".")[-1]
   765	            if name in tree.names and not isinstance(tree.names[name].node, PlaceholderNode):
   766	                continue
   767	            self.create_alias(tree, target_name, alias, name)

   769	    def add_typing_extension_aliases(self, tree: MypyFile) -> None:
   770	        """Typing extensions module does contain some type aliases.
   771
   772	        We need to analyze them as such, because in typeshed
   773	        they are just defined as `_Alias()` call.
   774	        Which is not supported natively.
   775	        """
   776	        assert tree.fullname == "typing_extensions"
   777
   778	        for alias, target_name in typing_extensions_aliases.items():
   779	            name = alias.split(".")[-1]
   780	            if name in tree.names and isinstance(tree.names[name].node, TypeAlias):
   781	                continue  # Do not reset TypeAliases on the second pass.
   782
   783	            # We need to remove any node that is there at the moment. It is invalid.
   784	            tree.names.pop(name, None)
   785
   786	            # Now, create a new alias.
   787	            self.create_alias(tree, target_name, alias, name)

   789	    def create_alias(self, tree: MypyFile, target_name: str, alias: str, name: str) -> None:
   790	        tag = self.track_incomplete_refs()
   791	        n = self.lookup_fully_qualified_or_none(target_name)
   792	        if n:
   793	            if isinstance(n.node, PlaceholderNode):
   794	                self.mark_incomplete(name, tree)
   795	            else:
   796	                # Found built-in class target. Create alias.
   797	                target = self.named_type_or_none(target_name, [])
   798	                assert target is not None
   799	                # Transform List to List[Any], etc.
   800	                fix_instance(
   801	                    target, self.fail, self.note, disallow_any=False, options=self.options
   802	                )
   803	                alias_node = TypeAlias(
   804	                    target,
   805	                    alias,
   806	                    line=-1,
   807	                    column=-1,  # there is no context
   808	                    no_args=True,
   809	                    normalized=True,
   810	                )
   811	                self.add_symbol(name, alias_node, tree)
   812	        elif self.found_incomplete_ref(tag):
   813	            # Built-in class target may not ready yet -- defer.
   814	            self.mark_incomplete(name, tree)
   815	        else:
   816	            # Test fixtures may be missing some builtin classes, which is okay.
   817	            # Kill the placeholder if there is one.
   818	            if name in tree.names:
   819	                assert isinstance(tree.names[name].node, PlaceholderNode)
   820	                del tree.names[name]

   823	        """Adjust the module visibility of globals due to __all__."""
   824	        if "__all__" in self.globals:

   835	    @contextmanager
   836	    def file_context(
   837	        self, file_node: MypyFile, options: Options, active_type: TypeInfo | None = None
   838	    ) -> Iterator[None]:
   839	        """Configure analyzer for analyzing targets within a file/class.
   840
   841	        Args:
   842	            file_node: target file
   843	            options: options specific to the file
   844	            active_type: must be the surrounding class to analyze method targets
   845	        """
   846	        scope = self.scope
   847	        self.options = options
   848	        self.errors.set_file(file_node.path, file_node.fullname, scope=scope, options=options)
   849	        self.cur_mod_node = file_node
   850	        self.cur_mod_id = file_node.fullname
   851	        with scope.module_scope(self.cur_mod_id):
   852	            self._is_stub_file = file_node.path.lower().endswith(".pyi")
   853	            self._is_typeshed_stub_file = file_node.is_typeshed_file(options)
   854	            self.globals = file_node.names
   855	            self.tvar_scope = TypeVarLikeScope()
   856
   857	            self.named_tuple_analyzer = NamedTupleAnalyzer(options, self, self.msg)
   858	            self.typed_dict_analyzer = TypedDictAnalyzer(options, self, self.msg)
   859	            self.enum_call_analyzer = EnumCallAnalyzer(options, self)
   860	            self.newtype_analyzer = NewTypeAnalyzer(options, self, self.msg)
   861
   862	            # Counter that keeps track of references to undefined things potentially caused by
   863	            # incomplete namespaces.
   864	            self.num_incomplete_refs = 0
   865
   866	            if active_type:
   867	                enclosing_fullname = active_type.fullname.rsplit(".", 1)[0]
   868	                if "." in enclosing_fullname:
   869	                    enclosing_node = self.lookup_fully_qualified_or_none(enclosing_fullname)
   870	                    if enclosing_node and isinstance(enclosing_node.node, TypeInfo):
   871	                        self._type = enclosing_node.node
   872	                self.push_type_args(active_type.defn.type_args, active_type.defn)
   873	                self.incomplete_type_stack.append(False)
   874	                scope.enter_class(active_type)
   875	                self.enter_class(active_type.defn.info)
   876	                for tvar in active_type.defn.type_vars:
   877	                    self.tvar_scope.bind_existing(tvar)
   878
   879	            yield
   880
   881	            if active_type:
   882	                scope.leave_class()
   883	                self.leave_class()
   884	                self._type = None
   885	                self.incomplete_type_stack.pop()
   886	                self.pop_type_args(active_type.defn.type_args)
   887	        del self.options

   893	    def visit_func_def(self, defn: FuncDef) -> None:
   894	        self.statement = defn
   895
   896	        # Visit default values because they may contain assignment expressions.
   897	        for arg in defn.arguments:
   898	            if arg.initializer:
   899	                arg.initializer.accept(self)
   900
   901	        defn.is_conditional = self.block_depth[-1] > 0
   902
   903	        # Set full names even for those definitions that aren't added
   904	        # to a symbol table. For example, for overload items.
   905	        defn._fullname = self.qualified_name(defn.name)
   906
   907	        # We don't add module top-level functions to symbol tables
   908	        # when we analyze their bodies in the second phase on analysis,
   909	        # since they were added in the first phase. Nested functions
   910	        # get always added, since they aren't separate targets.
   911	        if not self.recurse_into_functions or len(self.function_stack) > 0:
   912	            if not defn.is_decorated and not defn.is_overload:
   913	                self.add_function_to_symbol_table(defn)
   914
   915	        if not self.recurse_into_functions:
   916	            return
   917
   918	        with self.scope.function_scope(defn):
   919	            with self.inside_except_star_block_set(value=False):
   920	                self.analyze_func_def(defn)

   923	        if self.current_overload_item is None:
   924	            return fullname
   925	        return f"{fullname}#{self.current_overload_item}"

   927	    def analyze_func_def(self, defn: FuncDef) -> None:
   928	        if self.push_type_args(defn.type_args, defn) is None:
   929	            self.defer(defn)
   930	            return
   931
   932	        self.function_stack.append(defn)
   933
   934	        if defn.type:
   935	            assert isinstance(defn.type, CallableType)
   936	            has_self_type = self.update_function_type_variables(defn.type, defn)
   937	        else:
   938	            has_self_type = False
   939
   940	        self.function_stack.pop()
   941
   942	        if self.is_class_scope():
   943	            # Method definition
   944	            assert self.type is not None
   945	            defn.info = self.type
   946	            if defn.type is not None and defn.name in ("__init__", "__init_subclass__"):
   947	                assert isinstance(defn.type, CallableType)
   948	                if isinstance(get_proper_type(defn.type.ret_type), AnyType):
   949	                    defn.type = defn.type.copy_modified(ret_type=NoneType())
   950	            self.prepare_method_signature(defn, self.type, has_self_type)
   951
   952	        # Analyze function signature
   953	        fullname = self.function_fullname(defn.fullname)
   954	        with self.tvar_scope_frame(self.tvar_scope.method_frame(fullname)):
   955	            if defn.type:
   956	                self.check_classvar_in_signature(defn.type)
   957	                assert isinstance(defn.type, CallableType)
   958	                # Signature must be analyzed in the surrounding scope so that
   959	                # class-level imported names and type variables are in scope.
   960	                analyzer = self.type_analyzer()
   961	                tag = self.track_incomplete_refs()
   962	                result = analyzer.visit_callable_type(defn.type, nested=False, namespace=fullname)
   963	                # Don't store not ready types (including placeholders).
   964	                if self.found_incomplete_ref(tag) or has_placeholder(result):
   965	                    self.defer(defn)
   966	                    self.pop_type_args(defn.type_args)
   967	                    return
   968	                assert isinstance(result, ProperType)
   969	                if isinstance(result, CallableType):
   970	                    # type guards need to have a positional argument, to spec
   971	                    skip_self = self.is_class_scope() and not defn.is_static
   972	                    if result.type_guard and ARG_POS not in result.arg_kinds[skip_self:]:
   973	                        self.fail(
   974	                            "TypeGuard functions must have a positional argument",
   975	                            result,
   976	                            code=codes.VALID_TYPE,
   977	                        )
   978	                        # in this case, we just kind of just ... remove the type guard.
   979	                        result = result.copy_modified(type_guard=None)
   980	                    if result.type_is and ARG_POS not in result.arg_kinds[skip_self:]:
   981	                        self.fail(
   982	                            '"TypeIs" functions must have a positional argument',
   983	                            result,
   984	                            code=codes.VALID_TYPE,
   985	                        )
   986	                        result = result.copy_modified(type_is=None)
   987
   988	                    result = self.remove_unpack_kwargs(defn, result)
   989	                    if has_self_type and self.type is not None:
   990	                        info = self.type
   991	                        if info.self_type is not None:
   992	                            result.variables = [info.self_type] + list(result.variables)
   993	                defn.type = result
   994	                self.add_type_alias_deps(analyzer.aliases_used)
   995	                self.check_function_signature(defn)
   996	                if isinstance(defn, FuncDef):
   997	                    assert isinstance(defn.type, CallableType)
   998	                    defn.type = set_callable_name(defn.type, defn)
   999
  1000	        self.analyze_arg_initializers(defn)
  1001	        self.analyze_function_body(defn)
  1002
  1003	        if self.is_class_scope():
  1004	            assert self.type is not None
  1005	            # Mark protocol methods with empty bodies as implicitly abstract.
  1006	            # This makes explicit protocol subclassing type-safe.
  1007	            if (
  1008	                self.type.is_protocol
  1009	                and not self.is_stub_file  # Bodies in stub files are always empty.
  1010	                and (not isinstance(self.scope.function, OverloadedFuncDef) or defn.is_property)
  1011	                and defn.abstract_status != IS_ABSTRACT
  1012	                and is_trivial_body(defn.body)
  1013	            ):
  1014	                defn.abstract_status = IMPLICITLY_ABSTRACT
  1015	            if (
  1016	                is_trivial_body(defn.body)
  1017	                and not self.is_stub_file
  1018	                and defn.abstract_status != NOT_ABSTRACT
  1019	            ):
  1020	                defn.is_trivial_body = True
  1021
  1022	        if (
  1023	            defn.is_coroutine
  1024	            and isinstance(defn.type, CallableType)
  1025	            and self.wrapped_coro_return_types.get(defn) != defn.type
  1026	        ):
  1027	            if defn.is_async_generator:
  1028	                # Async generator types are handled elsewhere
  1029	                pass
  1030	            else:
  1031	                # A coroutine defined as `async def foo(...) -> T: ...`
  1032	                # has external return type `Coroutine[Any, Any, T]`.
  1033	                any_type = AnyType(TypeOfAny.special_form)
  1034	                ret_type = self.named_type_or_none(
  1035	                    "typing.Coroutine", [any_type, any_type, defn.type.ret_type]
  1036	                )
  1037	                assert ret_type is not None, "Internal error: typing.Coroutine not found"
  1038	                defn.type = defn.type.copy_modified(ret_type=ret_type)
  1039	                self.wrapped_coro_return_types[defn] = defn.type
  1040
  1041	        self.pop_type_args(defn.type_args)

  1043	    def remove_unpack_kwargs(self, defn: FuncDef, typ: CallableType) -> CallableType:
  1044	        if not typ.arg_kinds or typ.arg_kinds[-1] is not ArgKind.ARG_STAR2:
  1045	            return typ
  1046	        last_type = typ.arg_types[-1]
  1047	        if not isinstance(last_type, UnpackType):
  1048	            return typ
  1049	        last_type = get_proper_type(last_type.type)
  1050	        if not isinstance(last_type, TypedDictType):
  1051	            self.fail("Unpack item in ** argument must be a TypedDict", last_type)
  1052	            new_arg_types = typ.arg_types[:-1] + [AnyType(TypeOfAny.from_error)]
  1053	            return typ.copy_modified(arg_types=new_arg_types)
  1054	        overlap = set(typ.arg_names) & set(last_type.items)
  1055	        # It is OK for TypedDict to have a key named 'kwargs'.
  1056	        overlap.discard(typ.arg_names[-1])
  1057	        if overlap:
  1058	            overlapped = ", ".join([f'"{name}"' for name in overlap])
  1059	            self.fail(f"Overlap between argument names and ** TypedDict items: {overlapped}", defn)
  1060	            new_arg_types = typ.arg_types[:-1] + [AnyType(TypeOfAny.from_error)]
  1061	            return typ.copy_modified(arg_types=new_arg_types)
  1062	        # OK, everything looks right now, mark the callable type as using unpack.
  1063	        new_arg_types = typ.arg_types[:-1] + [last_type]
  1064	        return typ.copy_modified(arg_types=new_arg_types, unpack_kwargs=True)

  1066	    def prepare_method_signature(self, func: FuncDef, info: TypeInfo, has_self_type: bool) -> None:
  1067	        """Check basic signature validity and tweak annotation of self/cls argument."""
  1068	        # Only non-static methods are special, as well as __new__.
  1069	        functype = func.type
  1070	        if func.name == "__new__":
  1071	            func.is_static = True
  1072	        if not func.is_static or func.name == "__new__":
  1073	            if func.name in ["__init_subclass__", "__class_getitem__"]:
  1074	                func.is_class = True
  1075	            if not func.arguments:
  1076	                self.fail(
  1077	                    'Method must have at least one argument. Did you forget the "self" argument?',
  1078	                    func,
  1079	                )
  1080	            elif isinstance(functype, CallableType):
  1081	                self_type = get_proper_type(functype.arg_types[0])
  1082	                if isinstance(self_type, AnyType):
  1083	                    if has_self_type:
  1084	                        assert self.type is not None and self.type.self_type is not None
  1085	                        leading_type: Type = self.type.self_type
  1086	                    else:
  1087	                        func.is_trivial_self = True
  1088	                        leading_type = fill_typevars(info)
  1089	                    if func.is_class or func.name == "__new__":
  1090	                        leading_type = self.class_type(leading_type)
  1091	                    func.type = replace_implicit_first_type(functype, leading_type)
  1092	                elif has_self_type and isinstance(func.unanalyzed_type, CallableType):
  1093	                    if not isinstance(get_proper_type(func.unanalyzed_type.arg_types[0]), AnyType):
  1094	                        if self.is_expected_self_type(
  1095	                            self_type, func.is_class or func.name == "__new__"
  1096	                        ):
  1097	                            # This error is off by default, since it is explicitly allowed
  1098	                            # by the PEP 673.
  1099	                            self.fail(
  1100	                                'Redundant "Self" annotation for the first method argument',
  1101	                                func,
  1102	                                code=codes.REDUNDANT_SELF_TYPE,
  1103	                            )
  1104	                        else:
  1105	                            self.fail(
  1106	                                "Method cannot have explicit self annotation and Self type", func
  1107	                            )
  1108	        elif has_self_type:
  1109	            self.fail("Static methods cannot use Self type", func)

  1111	    def is_expected_self_type(self, typ: Type, is_classmethod: bool) -> bool:
  1112	        """Does this (analyzed or not) type represent the expected Self type for a method?"""
  1113	        assert self.type is not None
  1114	        typ = get_proper_type(typ)
  1115	        if is_classmethod:
  1116	            if isinstance(typ, TypeType):
  1117	                return self.is_expected_self_type(typ.item, is_classmethod=False)
  1118	            if isinstance(typ, UnboundType):
  1119	                sym = self.lookup_qualified(typ.name, typ, suppress_errors=True)
  1120	                if sym is not None and sym.fullname in TYPE_NAMES and typ.args:
  1121	                    return self.is_expected_self_type(typ.args[0], is_classmethod=False)
  1122	            return False
  1123	        if isinstance(typ, TypeVarType):
  1124	            return typ == self.type.self_type
  1125	        if isinstance(typ, UnboundType):
  1126	            sym = self.lookup_qualified(typ.name, typ, suppress_errors=True)
  1127	            return sym is not None and sym.fullname in SELF_TYPE_NAMES

  1130	    def set_original_def(self, previous: Node | None, new: FuncDef | Decorator) -> bool:
  1131	        """If 'new' conditionally redefine 'previous', set 'previous' as original
  1132
  1133	        We reject straight redefinitions of functions, as they are usually
  1134	        a programming error. For example:
  1135
  1136	          def f(): ...
  1137	          def f(): ...  # Error: 'f' redefined
  1138	        """
  1139	        if isinstance(new, Decorator):
  1140	            new = new.func
  1141	        if (
  1142	            isinstance(previous, (FuncDef, Decorator))
  1143	            and unnamed_function(new.name)
  1144	            and unnamed_function(previous.name)
  1145	        ):
  1146	            return True
  1147	        if isinstance(previous, (FuncDef, Var, Decorator)) and new.is_conditional:
  1148	            new.original_def = previous
  1149	            return True
  1150	        else:
  1151	            return False

  1227	    def analyze_overloaded_func_def(self, defn: OverloadedFuncDef) -> None:
  1228	        # OverloadedFuncDef refers to any legitimate situation where you have
  1229	        # more than one declaration for the same function in a row.  This occurs
  1230	        # with a @property with a setter or a deleter, and for a classic
  1231	        # @overload.
  1232
  1233	        defn._fullname = self.qualified_name(defn.name)
  1234	        # TODO: avoid modifying items.
  1235	        defn.items = defn.unanalyzed_items.copy()
  1236
  1237	        first_item = defn.items[0]
  1238	        first_item.is_overload = True
  1239	        with self.overload_item_set(0):
  1240	            first_item.accept(self)
  1241
  1242	        bare_setter_type = None
  1243	        is_property = False
  1244	        if isinstance(first_item, Decorator) and first_item.func.is_property:
  1245	            is_property = True
  1246	            # This is a property.
  1247	            first_item.func.is_overload = True
  1248	            bare_setter_type = self.analyze_property_with_multi_part_definition(defn)
  1249	            typ = function_type(first_item.func, self.named_type("builtins.function"))
  1250	            assert isinstance(typ, CallableType)
  1251	            types = [typ]
  1252	        else:
  1253	            # This is a normal overload. Find the item signatures, the
  1254	            # implementation (if outside a stub), and any missing @overload
  1255	            # decorators.
  1256	            types, impl, non_overload_indexes = self.analyze_overload_sigs_and_impl(defn)
  1257	            defn.impl = impl
  1258	            if non_overload_indexes:
  1259	                self.handle_missing_overload_decorators(
  1260	                    defn, non_overload_indexes, some_overload_decorators=len(types) > 0
  1261	                )
  1262	            # If we found an implementation, remove it from the overload item list,
  1263	            # as it's special.
  1264	            if impl is not None:
  1265	                assert impl is defn.items[-1]
  1266	                defn.items = defn.items[:-1]
  1267	            elif not non_overload_indexes:
  1268	                self.handle_missing_overload_implementation(defn)
  1269
  1270	        if types and not any(
  1271	            # If some overload items are decorated with other decorators, then
  1272	            # the overload type will be determined during type checking.
  1273	            # Note: bare @property is removed in visit_decorator().
  1274	            isinstance(it, Decorator)
  1275	            and len(it.decorators) > (1 if i > 0 or not is_property else 0)
  1276	            for i, it in enumerate(defn.items)
  1277	        ):
  1278	            # TODO: should we enforce decorated overloads consistency somehow?
  1279	            # Some existing code uses both styles:
  1280	            #   * Put decorator only on implementation, use "effective" types in overloads
  1281	            #   * Put decorator everywhere, use "bare" types in overloads.
  1282	            defn.type = Overloaded(types)
  1283	            defn.type.line = defn.line
  1284	            # In addition, we can set the getter/setter type for valid properties as some
  1285	            # code paths may either use the above type, or var.type etc. of the first item.
  1286	            if isinstance(first_item, Decorator) and bare_setter_type:
  1287	                first_item.var.type = types[0]
  1288	                first_item.var.setter_type = bare_setter_type
  1289
  1290	        if not defn.items:
  1291	            # It was not a real overload after all, but function redefinition. We've
  1292	            # visited the redefinition(s) already.
  1293	            if not defn.impl:
  1294	                # For really broken overloads with no items and no implementation we need to keep
  1295	                # at least one item to hold basic information like function name.
  1296	                defn.impl = defn.unanalyzed_items[-1]
  1297	            return
  1298
  1299	        # We know this is an overload def. Infer properties and perform some checks.
  1300	        self.process_deprecated_overload(defn)
  1301	        self.process_final_in_overload(defn)
  1302	        self.process_static_or_class_method_in_overload(defn)
  1303	        self.process_overload_impl(defn)

  1305	    def process_deprecated_overload(self, defn: OverloadedFuncDef) -> None:
  1306	        if defn.is_property:

  1361	    def analyze_overload_sigs_and_impl(
  1362	        self, defn: OverloadedFuncDef
  1363	    ) -> tuple[list[CallableType], OverloadPart | None, list[int]]:
  1364	        """Find overload signatures, the implementation, and items with missing @overload.
  1365
  1366	        Assume that the first was already analyzed. As a side effect:
  1367	        analyzes remaining items and updates 'is_overload' flags.
  1368	        """
  1369	        types = []
  1370	        non_overload_indexes = []
  1371	        impl: OverloadPart | None = None
  1372	        for i, item in enumerate(defn.items):
  1373	            if i != 0:
  1374	                # Assume that the first item was already visited
  1375	                item.is_overload = True
  1376	                with self.overload_item_set(i if i < len(defn.items) - 1 else None):
  1377	                    item.accept(self)
