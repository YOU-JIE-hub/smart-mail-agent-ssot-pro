  4930	    def visit_try_stmt(self, s: TryStmt) -> None:
  4931	        """Type check a try statement."""
  4932
  4933	        iter_errors = None
  4934
  4935	        # Our enclosing frame will get the result if the try/except falls through.
  4936	        # This one gets all possible states after the try block exited abnormally
  4937	        # (by exception, return, break, etc.)
  4938	        with self.binder.frame_context(can_skip=False, fall_through=0):
  4939	            # Not only might the body of the try statement exit
  4940	            # abnormally, but so might an exception handler or else
  4941	            # clause. The finally clause runs in *all* cases, so we
  4942	            # need an outer try frame to catch all intermediate states
  4943	            # in case an exception is raised during an except or else
  4944	            # clause. As an optimization, only create the outer try
  4945	            # frame when there actually is a finally clause.
  4946	            self.visit_try_without_finally(s, try_frame=bool(s.finally_body))
  4947	            if s.finally_body:
  4948	                # First we check finally_body is type safe on all abnormal exit paths
  4949	                iter_errors = IterationDependentErrors()
  4950	                with IterationErrorWatcher(self.msg.errors, iter_errors):
  4951	                    self.accept(s.finally_body)
  4952
  4953	        if s.finally_body:
  4954	            # Then we try again for the more restricted set of options
  4955	            # that can fall through. (Why do we need to check the
  4956	            # finally clause twice? Depending on whether the finally
  4957	            # clause was reached by the try clause falling off the end
  4958	            # or exiting abnormally, after completing the finally clause
  4959	            # either flow will continue to after the entire try statement
  4960	            # or the exception/return/etc. will be processed and control
  4961	            # flow will escape. We need to check that the finally clause
  4962	            # type checks in both contexts, but only the resulting types
  4963	            # from the latter context affect the type state in the code
  4964	            # that follows the try statement.)
  4965	            assert iter_errors is not None
  4966	            if not self.binder.is_unreachable():
  4967	                with IterationErrorWatcher(self.msg.errors, iter_errors):
  4968	                    self.accept(s.finally_body)
  4969	            self.msg.iteration_dependent_errors(iter_errors)

  4971	    def visit_try_without_finally(self, s: TryStmt, try_frame: bool) -> None:
  4972	        """Type check a try statement, ignoring the finally block.
  4973
  4974	        On entry, the top frame should receive all flow that exits the
  4975	        try block abnormally (i.e., such that the else block does not
  4976	        execute), and its parent should receive all flow that exits
  4977	        the try block normally.
  4978	        """
  4979	        # This frame will run the else block if the try fell through.
  4980	        # In that case, control flow continues to the parent of what
  4981	        # was the top frame on entry.
  4982	        with self.binder.frame_context(can_skip=False, fall_through=2, try_frame=try_frame):
  4983	            # This frame receives exit via exception, and runs exception handlers
  4984	            with self.binder.frame_context(can_skip=False, conditional_frame=True, fall_through=2):
  4985	                # Finally, the body of the try statement
  4986	                with self.binder.frame_context(can_skip=False, fall_through=2, try_frame=True):
  4987	                    self.accept(s.body)
  4988	                for i in range(len(s.handlers)):
  4989	                    with self.binder.frame_context(can_skip=True, fall_through=4):
  4990	                        typ = s.types[i]
  4991	                        if typ:
  4992	                            t = self.check_except_handler_test(typ, s.is_star)
  4993	                            var = s.vars[i]
  4994	                            if var:
  4995	                                # To support local variables, we make this a definition line,
  4996	                                # causing assignment to set the variable's type.
  4997	                                var.is_inferred_def = True
  4998	                                self.check_assignment(var, self.temp_node(t, var))
  4999	                        self.accept(s.handlers[i])
  5000	                        var = s.vars[i]
  5001	                        if var:
  5002	                            # Exception variables are deleted.
  5003	                            # Unfortunately, this doesn't let us detect usage before the
  5004	                            # try/except block.
  5005	                            source = var.name
  5006	                            if isinstance(var.node, Var):
  5007	                                new_type = DeletedType(source=source)
  5008	                                var.node.type = new_type
  5009	                                if self.options.allow_redefinition_new:
  5010	                                    # TODO: Should we use put() here?
  5011	                                    self.binder.assign_type(var, new_type, new_type)
  5012	                            if not self.options.allow_redefinition_new:
  5013	                                self.binder.cleanse(var)
  5014	            if s.else_body:
  5015	                self.accept(s.else_body)

  5410	    def visit_match_stmt(self, s: MatchStmt) -> None:
  5411	        named_subject: Expression
  5412	        if isinstance(s.subject, CallExpr):
  5413	            # Create a dummy subject expression to handle cases where a match statement's subject
  5414	            # is not a literal value. This lets us correctly narrow types and check exhaustivity
  5415	            # This is hack!
  5416	            if s.subject_dummy is None:
  5417	                id = s.subject.callee.fullname if isinstance(s.subject.callee, RefExpr) else ""
  5418	                name = "dummy-match-" + id

  5420	                s.subject_dummy = NameExpr(name)
  5421	                s.subject_dummy.node = v
  5422	            named_subject = s.subject_dummy
  5423	        else:
  5424	            named_subject = s.subject
  5425
  5426	        with self.binder.frame_context(can_skip=False, fall_through=0):
  5427	            subject_type = get_proper_type(self.expr_checker.accept(s.subject))
  5428
  5429	            if isinstance(subject_type, DeletedType):
  5430	                self.msg.deleted_as_rvalue(subject_type, s)
  5431
  5432	            # We infer types of patterns twice. The first pass is used
  5433	            # to infer the types of capture variables. The type of a
  5434	            # capture variable may depend on multiple patterns (it
  5435	            # will be a union of all capture types). This pass ignores
  5436	            # guard expressions.
  5437	            pattern_types = [self.pattern_checker.accept(p, subject_type) for p in s.patterns]

  5439	            inferred_types = self.infer_variable_types_from_type_maps(type_maps)
  5440

  5442	            unmatched_types: TypeMap = None
  5443	            for p, g, b in zip(s.patterns, s.guards, s.bodies):
  5444	                current_subject_type = self.expr_checker.narrow_type_from_binder(
  5445	                    named_subject, subject_type
  5446	                )
  5447	                pattern_type = self.pattern_checker.accept(p, current_subject_type)
  5448	                with self.binder.frame_context(can_skip=True, fall_through=2):
  5449	                    if b.is_unreachable or isinstance(
  5450	                        get_proper_type(pattern_type.type), UninhabitedType
  5451	                    ):

  5456	                            named_subject, pattern_type.type, pattern_type.rest_type
  5457	                        )
  5458	                        pattern_map = self.propagate_up_typemap_info(pattern_map)
  5459	                        else_map = self.propagate_up_typemap_info(else_map)

  5463	                            for expr, typ in pattern_map.items():
  5464	                                self.push_type_map(
  5465	                                    self._get_recursive_sub_patterns_map(expr, typ),
  5466	                                    from_assignment=False,
  5467	                                )
  5468	                        self.push_type_map(pattern_type.captures, from_assignment=False)
  5469	                    if g is not None:
  5470	                        with self.binder.frame_context(can_skip=False, fall_through=3):
  5471	                            gt = get_proper_type(self.expr_checker.accept(g))
  5472
  5473	                            if isinstance(gt, DeletedType):
  5474	                                self.msg.deleted_as_rvalue(gt, s)
  5475
  5476	                            guard_map, guard_else_map = self.find_isinstance_check(g)
  5477	                            else_map = or_conditional_maps(else_map, guard_else_map)
  5478
  5479	                            # If the guard narrowed the subject, copy the narrowed types over
  5480	                            if isinstance(p, AsPattern):
  5481	                                case_target = p.pattern or p.name
  5482	                                if isinstance(case_target, NameExpr):
  5483	                                    for type_map in (guard_map, else_map):

  5485	                                            continue
  5486	                                        for expr in list(type_map):
  5487	                                            if not (
  5488	                                                isinstance(expr, NameExpr)
  5489	                                                and expr.fullname == case_target.fullname
  5490	                                            ):
  5491	                                                continue
  5492	                                            type_map[named_subject] = type_map[expr]
  5493
  5494	                            self.push_type_map(guard_map, from_assignment=False)
  5495	                            self.accept(b)

  5499	                unmatched_types = else_map
  5500
  5501	            if unmatched_types is not None:
  5502	                for typ in list(unmatched_types.values()):
  5503	                    self.msg.match_statement_inexhaustive_match(typ, s)

  5505	            # This is needed due to a quirk in frame_context. Without it types will stay narrowed
  5506	            # after the match.
  5507	            with self.binder.frame_context(can_skip=False, fall_through=2):

  5528	        # necessary `Var`s (e.g. a function defined earlier with the same name).

  5607	    def intersect_instances(
  5608	        self, instances: tuple[Instance, Instance], errors: list[tuple[str, str]]
  5609	    ) -> Instance | None:
  5610	        """Try creating an ad-hoc intersection of the given instances.
  5611
  5612	        Note that this function does *not* try and create a full-fledged
  5613	        intersection type. Instead, it returns an instance of a new ad-hoc
  5614	        subclass of the given instances.
  5615
  5616	        This is mainly useful when you need a way of representing some
  5617	        theoretical subclass of the instances the user may be trying to use
  5618	        the generated intersection can serve as a placeholder.
  5619
  5620	        This function will create a fresh subclass the first time you call it.
  5621	        So this means calling `self.intersect_intersection([inst_1, inst_2], ctx)`
  5622	        twice will return the same subclass of inst_1 and inst_2.
  5623
  5624	        Returns None if creating the subclass is impossible (e.g. due to
  5625	        MRO errors or incompatible signatures). If we do successfully create
  5626	        a subclass, its TypeInfo will automatically be added to the global scope.
  5627	        """
  5628	        curr_module = self.scope.stack[0]
  5629	        assert isinstance(curr_module, MypyFile)
  5630
  5631	        # First, retry narrowing while allowing promotions (they are disabled by default
  5632	        # for isinstance() checks, etc). This way we will still type-check branches like
  5633	        # x: complex = 1
  5634	        # if isinstance(x, int):
  5635	        #     ...
  5636	        left, right = instances
  5637	        if is_proper_subtype(left, right, ignore_promotions=False):
  5638	            return left
  5639	        if is_proper_subtype(right, left, ignore_promotions=False):
  5640	            return right
  5641
  5642	        def _get_base_classes(instances_: tuple[Instance, Instance]) -> list[Instance]:
  5643	            base_classes_ = []
  5644	            for inst in instances_:
  5645	                if inst.type.is_intersection:
  5646	                    expanded = inst.type.bases
  5647	                else:
  5648	                    expanded = [inst]
  5649
  5650	                for expanded_inst in expanded:
  5651	                    base_classes_.append(expanded_inst)
  5652	            return base_classes_

  5660	                assert isinstance(symbol.node, TypeInfo)
  5661	                return symbol.node, name
  5662	            cdef, info_ = self.make_fake_typeinfo(curr_module_.fullname, name, name, base_classes_)
  5663	            return info_, name
  5664
  5665	        base_classes = _get_base_classes(instances)

  5667	        # into the symbol table because it is not specific enough.

  5685	                self.check_multiple_inheritance(info)
  5686	            if local_errors.has_new_errors():
  5687	                # "class A(B, C)" unsafe, now check "class A(C, B)":
  5688	                base_classes = _get_base_classes(instances[::-1])

  5695	        except MroError:

  5697	            return None
  5698	        if local_errors.has_new_errors():
  5699	            errors.append((pretty_names_list, "would have incompatible method signatures"))
  5700	            return None
  5701
  5702	        curr_module.names[full_name] = SymbolTableNode(GDEF, info)

  5748	    def partition_by_callable(
  5749	        self, typ: Type, unsound_partition: bool
  5750	    ) -> tuple[list[Type], list[Type]]:
  5751	        """Partitions a type into callable subtypes and uncallable subtypes.
  5752
  5753	        Thus, given:
  5754	        `callables, uncallables = partition_by_callable(type)`
  5755
  5756	        If we assert `callable(type)` then `type` has type Union[*callables], and
  5757	        If we assert `not callable(type)` then `type` has type Union[*uncallables]
  5758
  5759	        If unsound_partition is set, assume that anything that is not
  5760	        clearly callable is in fact not callable. Otherwise we generate a
  5761	        new subtype that *is* callable.
  5762
  5763	        Guaranteed to not return [], [].
  5764	        """
  5765	        typ = get_proper_type(typ)
  5766
  5767	        if isinstance(typ, (FunctionLike, TypeType)):
  5768	            return [typ], []
  5769
  5770	        if isinstance(typ, AnyType):
  5771	            return [typ], [typ]
  5772
  5773	        if isinstance(typ, NoneType):
  5774	            return [], [typ]
  5775
  5776	        if isinstance(typ, UnionType):
  5777	            callables = []
  5778	            uncallables = []
  5779	            for subtype in typ.items:
  5780	                # Use unsound_partition when handling unions in order to
  5781	                # allow the expected type discrimination.
  5782	                subcallables, subuncallables = self.partition_by_callable(
  5783	                    subtype, unsound_partition=True
  5784	                )
  5785	                callables.extend(subcallables)
  5786	                uncallables.extend(subuncallables)
  5787	            return callables, uncallables
  5788
  5789	        if isinstance(typ, TypeVarType):
  5790	            # We could do better probably?
  5791	            # Refine the type variable's bound as our type in the case that
  5792	            # callable() is true. This unfortunately loses the information that
  5793	            # the type is a type variable in that branch.
  5794	            # This matches what is done for isinstance, but it may be possible to
  5795	            # do better.
  5796	            # If it is possible for the false branch to execute, return the original
  5797	            # type to avoid losing type information.
  5798	            callables, uncallables = self.partition_by_callable(
  5799	                erase_to_union_or_bound(typ), unsound_partition
  5800	            )
  5801	            uncallables = [typ] if uncallables else []
  5802	            return callables, uncallables
  5803
  5804	        # A TupleType is callable if its fallback is, but needs special handling
  5805	        # when we dummy up a new type.
  5806	        ityp = typ
  5807	        if isinstance(typ, TupleType):
  5808	            ityp = tuple_fallback(typ)
  5809
  5810	        if isinstance(ityp, Instance):
  5811	            method = ityp.type.get_method("__call__")
  5812	            if method and method.type:
  5813	                callables, uncallables = self.partition_by_callable(
  5814	                    method.type, unsound_partition=False
  5815	                )
  5816	                if callables and not uncallables:
  5817	                    # Only consider the type callable if its __call__ method is
  5818	                    # definitely callable.
  5819	                    return [typ], []
  5820
  5821	            if not unsound_partition:
  5822	                fake = self.make_fake_callable(ityp)
  5823	                if isinstance(typ, TupleType):
  5824	                    fake.type.tuple_type = TupleType(typ.items, fake)
  5825	                    return [fake.type.tuple_type], [typ]
  5826	                return [fake], [typ]
  5827
  5828	        if unsound_partition:
  5829	            return [], [typ]
  5830	        else:
  5831	            # We don't know how properly make the type callable.
  5832	            return [typ], [typ]

  5834	    def conditional_callable_type_map(
  5835	        self, expr: Expression, current_type: Type | None
  5836	    ) -> tuple[TypeMap, TypeMap]:
  5837	        """Takes in an expression and the current type of the expression.
  5838
  5839	        Returns a 2-tuple: The first element is a map from the expression to
  5840	        the restricted type if it were callable. The second element is a
  5841	        map from the expression to the type it would hold if it weren't
  5842	        callable.
  5843	        """
  5844	        if not current_type:
  5845	            return {}, {}
  5846
  5847	        if isinstance(get_proper_type(current_type), AnyType):
  5848	            return {}, {}
  5849
  5850	        callables, uncallables = self.partition_by_callable(current_type, unsound_partition=False)
  5851
  5852	        if callables and uncallables:
  5853	            callable_map = {expr: UnionType.make_union(callables)} if callables else None
  5854	            uncallable_map = {expr: UnionType.make_union(uncallables)} if uncallables else None
  5855	            return callable_map, uncallable_map
  5856
  5857	        elif callables:
  5858	            return {}, None
  5859
  5860	        return None, {}

  5862	    def conditional_types_for_iterable(
  5863	        self, item_type: Type, iterable_type: Type
  5864	    ) -> tuple[Type | None, Type | None]:
  5865	        """
  5866	        Narrows the type of `iterable_type` based on the type of `item_type`.
  5867	        For now, we only support narrowing unions of TypedDicts based on left operand being literal string(s).
  5868	        """
  5869	        if_types: list[Type] = []
  5870	        else_types: list[Type] = []
  5871
  5872	        iterable_type = get_proper_type(iterable_type)
  5873	        if isinstance(iterable_type, UnionType):
  5874	            possible_iterable_types = get_proper_types(iterable_type.relevant_items())
  5875	        else:
  5876	            possible_iterable_types = [iterable_type]
  5877
  5878	        item_str_literals = try_getting_str_literals_from_type(item_type)
  5879
  5880	        for possible_iterable_type in possible_iterable_types:
  5881	            if item_str_literals and isinstance(possible_iterable_type, TypedDictType):
  5882	                for key in item_str_literals:
  5883	                    if key in possible_iterable_type.required_keys:
  5884	                        if_types.append(possible_iterable_type)
  5885	                    elif (
  5886	                        key in possible_iterable_type.items or not possible_iterable_type.is_final
  5887	                    ):
  5888	                        if_types.append(possible_iterable_type)
  5889	                        else_types.append(possible_iterable_type)
  5890	                    else:
  5891	                        else_types.append(possible_iterable_type)
  5892	            else:
  5893	                if_types.append(possible_iterable_type)
  5894	                else_types.append(possible_iterable_type)
  5895
  5896	        return (
  5897	            UnionType.make_union(if_types) if if_types else None,
  5898	            UnionType.make_union(else_types) if else_types else None,
  5899	        )

  5917	    def check_for_truthy_type(self, t: Type, expr: Expression) -> None:
  5918	        """
  5919	        Check if a type can have a truthy value.
  5920
  5921	        Used in checks like::
  5922
  5923	            if x: # <---
  5924
  5925	            not x  # <---
  5926	        """
  5927	        if not state.strict_optional:
  5928	            return  # if everything can be None, all bets are off
  5929
  5930	        t = get_proper_type(t)
  5931	        if not self._is_truthy_type(t):
  5932	            return
  5933
  5934	        def format_expr_type() -> str:
  5935	            typ = format_type(t, self.options)
  5936	            if isinstance(expr, MemberExpr):
  5937	                return f'Member "{expr.name}" has type {typ}'
  5938	            elif isinstance(expr, RefExpr) and expr.fullname:
  5939	                return f'"{expr.fullname}" has type {typ}'
  5940	            elif isinstance(expr, CallExpr):
  5941	                if isinstance(expr.callee, MemberExpr):
  5942	                    return f'"{expr.callee.name}" returns {typ}'
  5943	                elif isinstance(expr.callee, RefExpr) and expr.callee.fullname:
  5944	                    return f'"{expr.callee.fullname}" returns {typ}'
  5945	                return f"Call returns {typ}"
  5946	            else:
  5947	                return f"Expression has type {typ}"
  5948
  5949	        def get_expr_name() -> str:
  5950	            if isinstance(expr, (NameExpr, MemberExpr)):
  5951	                return f'"{expr.name}"'
  5952	            else:
  5953	                # return type if expr has no name
  5954	                return format_type(t, self.options)
  5955
  5956	        if isinstance(t, FunctionLike):
  5957	            self.fail(message_registry.FUNCTION_ALWAYS_TRUE.format(get_expr_name()), expr)
  5958	        elif isinstance(t, UnionType):
  5959	            self.fail(message_registry.TYPE_ALWAYS_TRUE_UNIONTYPE.format(format_expr_type()), expr)
  5960	        elif isinstance(t, Instance) and t.type.fullname == "typing.Iterable":
  5961	            _, info = self.make_fake_typeinfo("typing", "Collection", "Collection", [])
  5962	            self.fail(
  5963	                message_registry.ITERABLE_ALWAYS_TRUE.format(
  5964	                    format_expr_type(), format_type(Instance(info, t.args), self.options)
  5965	                ),
  5966	                expr,
  5967	            )
  5968	        else:
  5969	            self.fail(message_registry.TYPE_ALWAYS_TRUE.format(format_expr_type()), expr)

  5971	    def find_type_equals_check(
  5972	        self, node: ComparisonExpr, expr_indices: list[int]
  5973	    ) -> tuple[TypeMap, TypeMap]:
  5974	        """Narrow types based on any checks of the type ``type(x) == T``
  5975
  5976	        Args:
  5977	            node: The node that might contain the comparison
  5978	            expr_indices: The list of indices of expressions in ``node`` that are being
  5979	                compared
  5980	        """
  5981
  5982	        def is_type_call(expr: CallExpr) -> bool:
  5983	            """Is expr a call to type with one argument?"""
  5984	            return refers_to_fullname(expr.callee, "builtins.type") and len(expr.args) == 1
  5985
  5986	        # exprs that are being passed into type
  5987	        exprs_in_type_calls: list[Expression] = []
  5988	        # type that is being compared to type(expr)
  5989	        type_being_compared: list[TypeRange] | None = None
  5990	        # whether the type being compared to is final
  5991	        is_final = False
  5992
  5993	        for index in expr_indices:
  5994	            expr = node.operands[index]
  5995
  5996	            if isinstance(expr, CallExpr) and is_type_call(expr):
  5997	                exprs_in_type_calls.append(expr.args[0])
  5998	            else:
  5999	                current_type = self.get_isinstance_type(expr)
  6000	                if current_type is None:
  6001	                    continue
  6002	                if type_being_compared is not None:
  6003	                    # It doesn't really make sense to have several types being
  6004	                    # compared to the output of type (like type(x) == int == str)
  6005	                    # because whether that's true is solely dependent on what the
  6006	                    # types being compared are, so we don't try to narrow types any
  6007	                    # further because we can't really get any information about the
  6008	                    # type of x from that check
  6009	                    return {}, {}
  6010	                else:
  6011	                    if isinstance(expr, RefExpr) and isinstance(expr.node, TypeInfo):
  6012	                        is_final = expr.node.is_final
  6013	                    type_being_compared = current_type
  6014
  6015	        if not exprs_in_type_calls:
  6016	            return {}, {}
  6017
  6018	        if_maps: list[TypeMap] = []
  6019	        else_maps: list[TypeMap] = []
  6020	        for expr in exprs_in_type_calls:
  6021	            current_if_type, current_else_type = self.conditional_types_with_intersection(
  6022	                self.lookup_type(expr), type_being_compared, expr
  6023	            )
  6024	            current_if_map, current_else_map = conditional_types_to_typemaps(
  6025	                expr, current_if_type, current_else_type
  6026	            )
  6027	            if_maps.append(current_if_map)
  6028	            else_maps.append(current_else_map)
  6029
  6030	        def combine_maps(list_maps: list[TypeMap]) -> TypeMap:
  6031	            """Combine all typemaps in list_maps into one typemap"""
  6032	            if all(m is None for m in list_maps):
  6033	                return None
  6034	            result_map = {}
  6035	            for d in list_maps:
  6036	                if d is not None:
  6037	                    result_map.update(d)
  6038	            return result_map
  6039
  6040	        if_map = combine_maps(if_maps)
  6041	        # type(x) == T is only true when x has the same type as T, meaning
  6042	        # that it can be false if x is an instance of a subclass of T. That means
  6043	        # we can't do any narrowing in the else case unless T is final, in which
  6044	        # case T can't be subclassed
  6045	        if is_final:
  6046	            else_map = combine_maps(else_maps)
  6047	        else:
  6048	            else_map = {}
  6049	        return if_map, else_map

  6051	    def find_isinstance_check(
  6052	        self, node: Expression, *, in_boolean_context: bool = True
  6053	    ) -> tuple[TypeMap, TypeMap]:
  6054	        """Find any isinstance checks (within a chain of ands).  Includes
  6055	        implicit and explicit checks for None and calls to callable.
  6056	        Also includes TypeGuard and TypeIs functions.
  6057
  6058	        Return value is a map of variables to their types if the condition
  6059	        is true and a map of variables to their types if the condition is false.
  6060
  6061	        If either of the values in the tuple is None, then that particular
  6062	        branch can never occur.
  6063
  6064	        If `in_boolean_context=True` is passed, it means that we handle
  6065	        a walrus expression. We treat rhs values
  6066	        in expressions like `(a := A())` specially:
  6067	        for example, some errors are suppressed.
  6068
  6069	        May return {}, {}.
  6070	        Can return None, None in situations involving NoReturn.
  6071	        """
  6072	        if_map, else_map = self.find_isinstance_check_helper(
  6073	            node, in_boolean_context=in_boolean_context
  6074	        )
  6075	        new_if_map = self.propagate_up_typemap_info(if_map)
  6076	        new_else_map = self.propagate_up_typemap_info(else_map)
  6077	        return new_if_map, new_else_map
  6078
  6079	    def find_isinstance_check_helper(
  6080	        self, node: Expression, *, in_boolean_context: bool = True
  6081	    ) -> tuple[TypeMap, TypeMap]:
  6082	        if is_true_literal(node):
  6083	            return {}, None
  6084	        if is_false_literal(node):
  6085	            return None, {}
  6086
  6087	        if isinstance(node, CallExpr) and len(node.args) != 0:
  6088	            expr = collapse_walrus(node.args[0])
  6089	            if refers_to_fullname(node.callee, "builtins.isinstance"):
  6090	                if len(node.args) != 2:  # the error will be reported elsewhere
  6091	                    return {}, {}
  6092	                if literal(expr) == LITERAL_TYPE:
  6093	                    return conditional_types_to_typemaps(
  6094	                        expr,
  6095	                        *self.conditional_types_with_intersection(
  6096	                            self.lookup_type(expr), self.get_isinstance_type(node.args[1]), expr
  6097	                        ),
  6098	                    )
  6099	            elif refers_to_fullname(node.callee, "builtins.issubclass"):
  6100	                if len(node.args) != 2:  # the error will be reported elsewhere
  6101	                    return {}, {}
  6102	                if literal(expr) == LITERAL_TYPE:
  6103	                    return self.infer_issubclass_maps(node, expr)
  6104	            elif refers_to_fullname(node.callee, "builtins.callable"):
  6105	                if len(node.args) != 1:  # the error will be reported elsewhere
  6106	                    return {}, {}
  6107	                if literal(expr) == LITERAL_TYPE:
  6108	                    vartype = self.lookup_type(expr)
  6109	                    return self.conditional_callable_type_map(expr, vartype)
  6110	            elif refers_to_fullname(node.callee, "builtins.hasattr"):
  6111	                if len(node.args) != 2:  # the error will be reported elsewhere
  6112	                    return {}, {}
  6113	                attr = try_getting_str_literals(node.args[1], self.lookup_type(node.args[1]))
  6114	                if literal(expr) == LITERAL_TYPE and attr and len(attr) == 1:
  6115	                    return self.hasattr_type_maps(expr, self.lookup_type(expr), attr[0])
  6116	            elif isinstance(node.callee, RefExpr):
  6117	                if node.callee.type_guard is not None or node.callee.type_is is not None:
  6118	                    # TODO: Follow *args, **kwargs
  6119	                    if node.arg_kinds[0] != nodes.ARG_POS:
  6120	                        # the first argument might be used as a kwarg
  6121	                        called_type = get_proper_type(self.lookup_type(node.callee))
  6122
  6123	                        # TODO: there are some more cases in check_call() to handle.
  6124	                        if isinstance(called_type, Instance):
  6125	                            call = find_member(
  6126	                                "__call__", called_type, called_type, is_operator=True
  6127	                            )
  6128	                            if call is not None:
  6129	                                called_type = get_proper_type(call)
  6130
  6131	                        # *assuming* the overloaded function is correct, there's a couple cases:
  6132	                        #  1) The first argument has different names, but is pos-only. We don't
  6133	                        #     care about this case, the argument must be passed positionally.
  6134	                        #  2) The first argument allows keyword reference, therefore must be the
  6135	                        #     same between overloads.
  6136	                        if isinstance(called_type, (CallableType, Overloaded)):
  6137	                            name = called_type.items[0].arg_names[0]
  6138	                            if name in node.arg_names:
  6139	                                idx = node.arg_names.index(name)
  6140	                                # we want the idx-th variable to be narrowed
  6141	                                expr = collapse_walrus(node.args[idx])
  6142	                            else:
  6143	                                kind = (
  6144	                                    "guard" if node.callee.type_guard is not None else "narrower"
  6145	                                )
  6146	                                self.fail(
  6147	                                    message_registry.TYPE_GUARD_POS_ARG_REQUIRED.format(kind), node
  6148	                                )
  6149	                                return {}, {}
  6150	                    if literal(expr) == LITERAL_TYPE:
  6151	                        # Note: we wrap the target type, so that we can special case later.
  6152	                        # Namely, for isinstance() we use a normal meet, while TypeGuard is
  6153	                        # considered "always right" (i.e. even if the types are not overlapping).
  6154	                        # Also note that a care must be taken to unwrap this back at read places
  6155	                        # where we use this to narrow down declared type.
  6156	                        if node.callee.type_guard is not None:
  6157	                            return {expr: TypeGuardedType(node.callee.type_guard)}, {}
  6158	                        else:
  6159	                            assert node.callee.type_is is not None
  6160	                            return conditional_types_to_typemaps(
  6161	                                expr,
  6162	                                *self.conditional_types_with_intersection(
  6163	                                    self.lookup_type(expr),
  6164	                                    [TypeRange(node.callee.type_is, is_upper_bound=False)],
  6165	                                    expr,
  6166	                                ),
  6167	                            )
  6168	        elif isinstance(node, ComparisonExpr):
  6169	            return self.comparison_type_narrowing_helper(node)
  6170	        elif isinstance(node, AssignmentExpr):
  6171	            if_map: dict[Expression, Type] | None
  6172	            else_map: dict[Expression, Type] | None
  6173	            if_map = {}
  6174	            else_map = {}
  6175
  6176	            if_assignment_map, else_assignment_map = self.find_isinstance_check(node.target)
  6177
  6178	            if if_assignment_map is not None:
  6179	                if_map.update(if_assignment_map)
  6180	            if else_assignment_map is not None:
  6181	                else_map.update(else_assignment_map)
  6182
  6183	            if_condition_map, else_condition_map = self.find_isinstance_check(
  6184	                node.value, in_boolean_context=False
  6185	            )
  6186
  6187	            if if_condition_map is not None:
  6188	                if_map.update(if_condition_map)
  6189	            if else_condition_map is not None:
  6190	                else_map.update(else_condition_map)
  6191
  6192	            return (
  6193	                (None if if_assignment_map is None or if_condition_map is None else if_map),
  6194	                (None if else_assignment_map is None or else_condition_map is None else else_map),
  6195	            )
  6196	        elif isinstance(node, OpExpr) and node.op == "and":
  6197	            left_if_vars, left_else_vars = self.find_isinstance_check(node.left)
  6198	            right_if_vars, right_else_vars = self.find_isinstance_check(node.right)
  6199
  6200	            # (e1 and e2) is true if both e1 and e2 are true,
  6201	            # and false if at least one of e1 and e2 is false.
  6202	            return (
  6203	                and_conditional_maps(left_if_vars, right_if_vars),
  6204	                # Note that if left else type is Any, we can't add any additional
  6205	                # types to it, since the right maps were computed assuming
  6206	                # the left is True, which may be not the case in the else branch.
  6207	                or_conditional_maps(left_else_vars, right_else_vars, coalesce_any=True),
  6208	            )
  6209	        elif isinstance(node, OpExpr) and node.op == "or":
  6210	            left_if_vars, left_else_vars = self.find_isinstance_check(node.left)
  6211	            right_if_vars, right_else_vars = self.find_isinstance_check(node.right)
  6212
  6213	            # (e1 or e2) is true if at least one of e1 or e2 is true,
  6214	            # and false if both e1 and e2 are false.
