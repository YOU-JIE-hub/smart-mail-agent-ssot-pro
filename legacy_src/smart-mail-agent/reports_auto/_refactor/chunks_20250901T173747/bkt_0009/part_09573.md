   373	        if declared_type is None:
   374	            # Not sure why this happens.  It seems to mainly happen in
   375	            # member initialization.
   376	            return
   377	        if not is_subtype(type, declared_type):
   378	            # Pretty sure this is only happens when there's a type error.
   379
   380	            # Ideally this function wouldn't be called if the
   381	            # expression has a type error, though -- do other kinds of
   382	            # errors cause this function to get called at invalid
   383	            # times?
   384	            return

   386	        p_declared = get_proper_type(declared_type)
   387	        p_type = get_proper_type(type)
   388	        if isinstance(p_type, AnyType):
   389	            # Any type requires some special casing, for both historical reasons,
   390	            # and to optimise user experience without sacrificing correctness too much.
   391	            if isinstance(expr, RefExpr) and isinstance(expr.node, Var) and expr.node.is_inferred:
   392	                # First case: a local/global variable without explicit annotation,
   393	                # in this case we just assign Any (essentially following the SSA logic).
   394	                self.put(expr, type)
   395	            elif isinstance(p_declared, UnionType) and any(
   396	                isinstance(get_proper_type(item), NoneType) for item in p_declared.items
   397	            ):

   399	                # pattern when an untyped value used as a fallback replacing None.
   400	                new_items = [
   401	                    type if isinstance(get_proper_type(item), NoneType) else item
   402	                    for item in p_declared.items
   403	                ]
   404	                self.put(expr, UnionType(new_items))
   405	            elif isinstance(p_declared, UnionType) and any(
   406	                isinstance(get_proper_type(item), AnyType) for item in p_declared.items
   407	            ):
   408	                # Third case: a union already containing Any (most likely from an un-imported
   409	                # name), in this case we allow assigning Any as well.
   410	                self.put(expr, type)
   411	            else:
   412	                # In all other cases we don't narrow to Any to minimize false negatives.
   413	                self.put(expr, declared_type)
   414	        elif isinstance(p_declared, AnyType):
   415	            # Mirroring the first case above, we don't narrow to a precise type if the variable
   416	            # has an explicit `Any` type annotation.
   417	            if isinstance(expr, RefExpr) and isinstance(expr.node, Var) and expr.node.is_inferred:
   418	                self.put(expr, type)
   419	            else:
   420	                self.put(expr, declared_type)
   421	        else:
   422	            self.put(expr, type)

   424	        for i in self.try_frames:

   426	            # just copy this variable into a single stored frame.
   427	            self.allow_jump(i)

   429	    def invalidate_dependencies(self, expr: BindableExpression) -> None:

   444	        # so make sure the index is positive

   462	    @contextmanager
   463	    def frame_context(
   464	        self,
   465	        *,
   466	        can_skip: bool,
   467	        fall_through: int = 1,
   468	        break_frame: int = 0,
   469	        continue_frame: int = 0,
   470	        conditional_frame: bool = False,
   471	        try_frame: bool = False,
   472	    ) -> Iterator[Frame]:
   473	        """Return a context manager that pushes/pops frames on enter/exit.
   474
   475	        If can_skip is True, control flow is allowed to bypass the
   476	        newly-created frame.
   477
   478	        If fall_through > 0, then it will allow control flow that
   479	        falls off the end of the frame to escape to its ancestor
   480	        `fall_through` levels higher. Otherwise control flow ends
   481	        at the end of the frame.
   482
   483	        If break_frame > 0, then 'break' statements within this frame
   484	        will jump out to the frame break_frame levels higher than the
   485	        frame created by this call to frame_context. Similarly for
   486	        continue_frame and 'continue' statements.
   487
   488	        If try_frame is true, then execution is allowed to jump at any
   489	        point within the newly created frame (or its descendants) to
   490	        its parent (i.e., to the frame that was on top before this
   491	        call to frame_context).
   492
   493	        After the context manager exits, self.last_pop_changed indicates
   494	        whether any types changed in the newly-topmost frame as a result
   495	        of popping this frame.
   496	        """
   497	        assert len(self.frames) > 1
   498
   499	        if break_frame:
   500	            self.break_frames.append(len(self.frames) - break_frame)
   501	        if continue_frame:
   502	            self.continue_frames.append(len(self.frames) - continue_frame)
   503	        if try_frame:
   504	            self.try_frames.add(len(self.frames) - 1)
   505
   506	        new_frame = self.push_frame(conditional_frame)
   507	        if try_frame:
   508	            # An exception may occur immediately
   509	            self.allow_jump(-1)
   510	        yield new_frame
   511	        self.pop_frame(can_skip, fall_through)
   512
   513	        if break_frame:
   514	            self.break_frames.pop()
   515	        if continue_frame:
   516	            self.continue_frames.pop()
   517	        if try_frame:
   518	            self.try_frames.remove(len(self.frames) - 1)

   520	    @contextmanager
   521	    def top_frame_context(self) -> Iterator[Frame]:
   522	        """A variant of frame_context for use at the top level of
   523	        a namespace (module, function, or class).
   524	        """
   525	        assert len(self.frames) == 1
   526	        yield self.push_frame()
   527	        self.pop_frame(True, 0)
   528	        assert len(self.frames) == 1

   547	def collapse_variadic_union(typ: UnionType) -> Type:
   548	    """Simplify a union involving variadic tuple if possible.
   549
   550	    This will collapse a type like e.g.
   551	        tuple[X, Z] | tuple[X, Y, Z] | tuple[X, Y, Y, *tuple[Y, ...], Z]
   552	    back to
   553	        tuple[X, *tuple[Y, ...], Z]
   554	    which is equivalent, but much simpler form of the same type.
   555	    """
   556	    tuple_items = []
   557	    other_items = []
   558	    for t in typ.items:
   559	        p_t = get_proper_type(t)
   560	        if isinstance(p_t, TupleType):
   561	            tuple_items.append(p_t)
   562	        else:
   563	            other_items.append(t)
   564	    if len(tuple_items) <= 1:
   565	        # This type cannot be simplified further.
   566	        return typ
   567	    tuple_items = sorted(tuple_items, key=lambda t: len(t.items))
   568	    first = tuple_items[0]
   569	    last = tuple_items[-1]
   570	    unpack_index = find_unpack_in_list(last.items)
   571	    if unpack_index is None:
   572	        return typ
   573	    unpack = last.items[unpack_index]
   574	    assert isinstance(unpack, UnpackType)
   575	    unpacked = get_proper_type(unpack.type)
   576	    if not isinstance(unpacked, Instance):
   577	        return typ
   578	    assert unpacked.type.fullname == "builtins.tuple"
   579	    suffix = last.items[unpack_index + 1 :]
   580
   581	    # Check that first item matches the expected pattern and infer prefix.
   582	    if len(first.items) < len(suffix):
   583	        return typ
   584	    if suffix and first.items[-len(suffix) :] != suffix:
   585	        return typ
   586	    if suffix:
   587	        prefix = first.items[: -len(suffix)]
   588	    else:
   589	        prefix = first.items
   590
   591	    # Check that all middle types match the expected pattern as well.
   592	    arg = unpacked.args[0]
   593	    for i, it in enumerate(tuple_items[1:-1]):
   594	        if it.items != prefix + [arg] * (i + 1) + suffix:
   595	            return typ
   596
   597	    # Check the last item (the one with unpack), and choose an appropriate simplified type.
   598	    if last.items != prefix + [arg] * (len(typ.items) - 1) + [unpack] + suffix:
   599	        return typ
   600	    if len(first.items) == 0:
   601	        simplified: Type = unpacked.copy_modified()
   602	    else:
   603	        simplified = TupleType(prefix + [unpack] + suffix, fallback=last.partial_fallback)
   604	    return UnionType.make_union([simplified] + other_items)

   377	# Priorities used for imports.  (Here, top-level includes inside a class.)
   378	# These are used to determine a more predictable order in which the
   379	# nodes in an import cycle are processed.
   380	PRI_HIGH: Final = 5  # top-level "from X import blah"
   381	PRI_MED: Final = 10  # top-level "import X"
   382	PRI_LOW: Final = 20  # either form inside a function
   383	PRI_MYPY: Final = 25  # inside "if MYPY" or "if TYPE_CHECKING"
   384	PRI_INDIRECT: Final = 30  # an indirect dependency
   385	PRI_ALL: Final = 99  # include all priorities
   386
   387
   388	def import_priority(imp: ImportBase, toplevel_priority: int) -> int:
   389	    """Compute import priority from an import node."""
   390	    if not imp.is_top_level:
   391	        # Inside a function
   392	        return PRI_LOW
   393	    if imp.is_mypy_only:
   394	        # Inside "if MYPY" or "if typing.TYPE_CHECKING"
   395	        return max(PRI_MYPY, toplevel_priority)
   396	    # A regular import; priority determined by argument.
   397	    return toplevel_priority

   400	def load_plugins_from_config(
   401	    options: Options, errors: Errors, stdout: TextIO
   402	) -> tuple[list[Plugin], dict[str, str]]:
   403	    """Load all configured plugins.
   404
   405	    Return a list of all the loaded plugins from the config file.
   406	    The second return value is a snapshot of versions/hashes of loaded user
   407	    plugins (for cache validation).
   408	    """
   409	    import importlib
   410
   411	    snapshot: dict[str, str] = {}
   412
   413	    if not options.config_file:
   414	        return [], snapshot
   415
   416	    line = find_config_file_line_number(options.config_file, "mypy", "plugins")
   417	    if line == -1:
   418	        line = 1  # We need to pick some line number that doesn't look too confusing
   419
   420	    def plugin_error(message: str) -> NoReturn:
   421	        errors.report(line, 0, message)
   422	        errors.raise_error(use_stdout=False)
   423
   424	    custom_plugins: list[Plugin] = []
   425	    errors.set_file(options.config_file, None, options)
   426	    for plugin_path in options.plugins:
   427	        func_name = "plugin"
   428	        plugin_dir: str | None = None
   429	        if ":" in os.path.basename(plugin_path):
   430	            plugin_path, func_name = plugin_path.rsplit(":", 1)
   431	        if plugin_path.endswith(".py"):
   432	            # Plugin paths can be relative to the config file location.
   433	            plugin_path = os.path.join(os.path.dirname(options.config_file), plugin_path)
   434	            if not os.path.isfile(plugin_path):
   435	                plugin_error(f'Can\'t find plugin "{plugin_path}"')
   436	            # Use an absolute path to avoid populating the cache entry
   437	            # for 'tmp' during tests, since it will be different in
   438	            # different tests.
   439	            plugin_dir = os.path.abspath(os.path.dirname(plugin_path))
   440	            fnam = os.path.basename(plugin_path)
   441	            module_name = fnam[:-3]
   442	            sys.path.insert(0, plugin_dir)
   443	        elif re.search(r"[\\/]", plugin_path):
   444	            fnam = os.path.basename(plugin_path)
   445	            plugin_error(f'Plugin "{fnam}" does not have a .py extension')
   446	        else:
   447	            module_name = plugin_path
   448
   449	        try:
   450	            module = importlib.import_module(module_name)
   451	        except Exception as exc:
   452	            plugin_error(f'Error importing plugin "{plugin_path}": {exc}')
   453	        finally:
   454	            if plugin_dir is not None:
   455	                assert sys.path[0] == plugin_dir
   456	                del sys.path[0]
   457
   458	        if not hasattr(module, func_name):
   459	            plugin_error(
   460	                'Plugin "{}" does not define entry point function "{}"'.format(
   461	                    plugin_path, func_name
   462	                )
   463	            )
   464
   465	        try:
   466	            plugin_type = getattr(module, func_name)(__version__)
   467	        except Exception:
   468	            print(f"Error calling the plugin(version) entry point of {plugin_path}\n", file=stdout)
   469	            raise  # Propagate to display traceback
   470
   471	        if not isinstance(plugin_type, type):
   472	            plugin_error(
   473	                'Type object expected as the return value of "plugin"; got {!r} (in {})'.format(
   474	                    plugin_type, plugin_path
   475	                )
   476	            )
   477	        if not issubclass(plugin_type, Plugin):
   478	            plugin_error(
   479	                'Return value of "plugin" must be a subclass of "mypy.plugin.Plugin" '
   480	                "(in {})".format(plugin_path)
   481	            )
   482	        try:
   483	            custom_plugins.append(plugin_type(options))
   484	            snapshot[module_name] = take_module_snapshot(module)
   485	        except Exception:
   486	            print(f"Error constructing plugin instance of {plugin_type.__name__}\n", file=stdout)
   487	            raise  # Propagate to display traceback
   488
   489	    return custom_plugins, snapshot

   492	def load_plugins(
   493	    options: Options, errors: Errors, stdout: TextIO, extra_plugins: Sequence[Plugin]
   494	) -> tuple[Plugin, dict[str, str]]:
   495	    """Load all configured plugins.
   496
   497	    Return a plugin that encapsulates all plugins chained together. Always
   498	    at least include the default plugin (it's last in the chain).
   499	    The second return value is a snapshot of versions/hashes of loaded user
   500	    plugins (for cache validation).
   501	    """
   502	    custom_plugins, snapshot = load_plugins_from_config(options, errors, stdout)
   503
   504	    custom_plugins += extra_plugins
   505
   506	    default_plugin: Plugin = DefaultPlugin(options)
   507	    if not custom_plugins:
   508	        return default_plugin, snapshot
   509
   510	    # Custom plugins take precedence over the default plugin.
   511	    return ChainedPlugin(options, custom_plugins + [default_plugin]), snapshot

   514	def take_module_snapshot(module: types.ModuleType) -> str:
   515	    """Take plugin module snapshot by recording its version and hash.
   516
   517	    We record _both_ hash and the version to detect more possible changes
   518	    (e.g. if there is a change in modules imported by a plugin).
   519	    """
   520	    if hasattr(module, "__file__"):
   521	        assert module.__file__ is not None
   522	        with open(module.__file__, "rb") as f:
   523	            digest = hash_digest(f.read())
   524	    else:
   525	        digest = "unknown"
   526	    ver = getattr(module, "__version__", "none")
   527	    return f"{ver}:{digest}"

   530	def find_config_file_line_number(path: str, section: str, setting_name: str) -> int:
   531	    """Return the approximate location of setting_name within mypy config file.
   532
   533	    Return -1 if can't determine the line unambiguously.
   534	    """
   535	    in_desired_section = False
   536	    try:
   537	        results = []
   538	        with open(path, encoding="UTF-8") as f:
   539	            for i, line in enumerate(f):
   540	                line = line.strip()
   541	                if line.startswith("[") and line.endswith("]"):
   542	                    current_section = line[1:-1].strip()
   543	                    in_desired_section = current_section == section
   544	                elif in_desired_section and re.match(rf"{setting_name}\s*=", line):
   545	                    results.append(i + 1)
   546	        if len(results) == 1:
   547	            return results[0]
   548	    except OSError:
   549	        pass
   550	    return -1

   553	class BuildManager:
   554	    """This class holds shared state for building a mypy program.
   555
   556	    It is used to coordinate parsing, import processing, semantic
   557	    analysis and type checking.  The actual build steps are carried
   558	    out by dispatch().
   559
   560	    Attributes:
   561	      data_dir:        Mypy data directory (contains stubs)
   562	      search_paths:    SearchPaths instance indicating where to look for modules
   563	      modules:         Mapping of module ID to MypyFile (shared by the passes)
   564	      semantic_analyzer:
   565	                       Semantic analyzer, pass 2
   566	      all_types:       Map {Expression: Type} from all modules (enabled by export_types)
   567	      options:         Build options
   568	      missing_modules: Set of modules that could not be imported encountered so far
   569	      stale_modules:   Set of modules that needed to be rechecked (only used by tests)
   570	      fg_deps_meta:    Metadata for fine-grained dependencies caches associated with modules
   571	      fg_deps:         A fine-grained dependency map
   572	      version_id:      The current mypy version (based on commit id when possible)
   573	      plugin:          Active mypy plugin(s)
   574	      plugins_snapshot:
   575	                       Snapshot of currently active user plugins (versions and hashes)
   576	      old_plugins_snapshot:
   577	                       Plugins snapshot from previous incremental run (or None in
   578	                       non-incremental mode and if cache was not found)
   579	      errors:          Used for reporting all errors
   580	      flush_errors:    A function for processing errors after each SCC
   581	      cache_enabled:   Whether cache is being read. This is set based on options,
   582	                       but is disabled if fine-grained cache loading fails
   583	                       and after an initial fine-grained load. This doesn't
   584	                       determine whether we write cache files or not.
   585	      quickstart_state:
   586	                       A cache of filename -> mtime/size/hash info used to avoid
   587	                       needing to hash source files when using a cache with mismatching mtimes
   588	      stats:           Dict with various instrumentation numbers, it is used
   589	                       not only for debugging, but also required for correctness,

   752	    def all_imported_modules_in_file(self, file: MypyFile) -> list[tuple[int, str, int]]:
   753	        """Find all reachable import statements in a file.
   754
   755	        Return list of tuples (priority, module id, import line number)
   756	        for all modules imported in file; lower numbers == higher priority.
   757
   758	        Can generate blocking errors on bogus relative imports.
   759	        """
   760
   761	        def correct_rel_imp(imp: ImportFrom | ImportAll) -> str:
   762	            """Function to correct for relative imports."""
   763	            file_id = file.fullname
   764	            rel = imp.relative
   765	            if rel == 0:
   766	                return imp.id
   767	            if os.path.basename(file.path).startswith("__init__."):
   768	                rel -= 1
   769	            if rel != 0:
   770	                file_id = ".".join(file_id.split(".")[:-rel])
   771	            new_id = file_id + "." + imp.id if imp.id else file_id
   772
   773	            if not new_id:
   774	                self.errors.set_file(file.path, file.name, self.options)
   775	                self.errors.report(
   776	                    imp.line, 0, "No parent module -- cannot perform relative import", blocker=True
   777	                )
   778
   779	            return new_id
   780
   781	        res: list[tuple[int, str, int]] = []
   782	        for imp in file.imports:
   783	            if not imp.is_unreachable:
   784	                if isinstance(imp, Import):
   785	                    pri = import_priority(imp, PRI_MED)
   786	                    ancestor_pri = import_priority(imp, PRI_LOW)
   787	                    for id, _ in imp.ids:
   788	                        res.append((pri, id, imp.line))
   789	                        ancestor_parts = id.split(".")[:-1]
   790	                        ancestors = []
   791	                        for part in ancestor_parts:
   792	                            ancestors.append(part)
   793	                            res.append((ancestor_pri, ".".join(ancestors), imp.line))
   794	                elif isinstance(imp, ImportFrom):
   795	                    cur_id = correct_rel_imp(imp)
   796	                    all_are_submodules = True
   797	                    # Also add any imported names that are submodules.
   798	                    pri = import_priority(imp, PRI_MED)
   799	                    for name, __ in imp.names:
   800	                        sub_id = cur_id + "." + name
   801	                        if self.is_module(sub_id):
   802	                            res.append((pri, sub_id, imp.line))
   803	                        else:
   804	                            all_are_submodules = False
   805	                    # Add cur_id as a dependency, even if all of the
   806	                    # imports are submodules. Processing import from will try
   807	                    # to look through cur_id, so we should depend on it.
   808	                    # As a workaround for for some bugs in cycle handling (#4498),
   809	                    # if all of the imports are submodules, do the import at a lower
   810	                    # priority.
   811	                    pri = import_priority(imp, PRI_HIGH if not all_are_submodules else PRI_LOW)
   812	                    res.append((pri, cur_id, imp.line))
   813	                elif isinstance(imp, ImportAll):
   814	                    pri = import_priority(imp, PRI_HIGH)
   815	                    res.append((pri, correct_rel_imp(imp), imp.line))
   816
   817	        # Sort such that module (e.g. foo.bar.baz) comes before its ancestors (e.g. foo
   818	        # and foo.bar) so that, if FindModuleCache finds the target module in a
   819	        # package marked with py.typed underneath a namespace package installed in
   820	        # site-packages, (gasp), that cache's knowledge of the ancestors
   821	        # (aka FindModuleCache.ns_ancestors) can be primed when it is asked to find
   822	        # the parent.
   823	        res.sort(key=lambda x: -x[1].count("."))
   824	        return res

   830	    def parse_file(
   831	        self, id: str, path: str, source: str, ignore_errors: bool, options: Options
   832	    ) -> MypyFile:
   833	        """Parse the source of a file with the given name.
   834
   835	        Raise CompileError if there is a parse error.
   836	        """
   837	        t0 = time.time()
   838	        if ignore_errors:
   839	            self.errors.ignored_files.add(path)
   840	        tree = parse(source, path, id, self.errors, options=options)
   841	        tree._fullname = id
   842	        self.add_stats(
   843	            files_parsed=1,
   844	            modules_parsed=int(not tree.is_stub),
   845	            stubs_parsed=int(tree.is_stub),
   846	            parse_time=time.time() - t0,
   847	        )
   848
   849	        if self.errors.is_blockers():

   927	def write_deps_cache(
   928	    rdeps: dict[str, dict[str, set[str]]], manager: BuildManager, graph: Graph
   929	) -> None:
   930	    """Write cache files for fine-grained dependencies.
   931
   932	    Serialize fine-grained dependencies map for fine grained mode.
   933
   934	    Dependencies on some module 'm' is stored in the dependency cache
   935	    file m.deps.json.  This entails some spooky action at a distance:
   936	    if module 'n' depends on 'm', that produces entries in m.deps.json.
   937	    When there is a dependency on a module that does not exist in the
   938	    build, it is stored with its first existing parent module. If no
   939	    such module exists, it is stored with the fake module FAKE_ROOT_MODULE.
   940
   941	    This means that the validity of the fine-grained dependency caches
   942	    are a global property, so we store validity checking information for
   943	    fine-grained dependencies in a global cache file:
   944	     * We take a snapshot of current sources to later check consistency
   945	       between the fine-grained dependency cache and module cache metadata
   946	     * We store the mtime of all of the dependency files to verify they
   947	       haven't changed
   948	    """
   949	    metastore = manager.metastore
   950
   951	    error = False
   952
   953	    fg_deps_meta = manager.fg_deps_meta.copy()
   954
   955	    for id in rdeps:
   956	        if id != FAKE_ROOT_MODULE:
   957	            _, _, deps_json = get_cache_names(id, graph[id].xpath, manager.options)
   958	        else:
   959	            deps_json = DEPS_ROOT_FILE
   960	        assert deps_json
   961	        manager.log("Writing deps cache", deps_json)
   962	        if not manager.metastore.write(deps_json, deps_to_json(rdeps[id])):
   963	            manager.log(f"Error writing fine-grained deps JSON file {deps_json}")
   964	            error = True
   965	        else:
   966	            fg_deps_meta[id] = {"path": deps_json, "mtime": manager.getmtime(deps_json)}
   967
   968	    meta_snapshot: dict[str, str] = {}
   969	    for id, st in graph.items():
   970	        # If we didn't parse a file (so it doesn't have a
   971	        # source_hash), then it must be a module with a fresh cache,
   972	        # so use the hash from that.
   973	        if st.source_hash:
   974	            hash = st.source_hash
   975	        else:
   976	            if st.meta:
   977	                hash = st.meta.hash
   978	            else:
   979	                hash = ""
   980	        meta_snapshot[id] = hash
   981
   982	    meta = {"snapshot": meta_snapshot, "deps_meta": fg_deps_meta}
   983
   984	    if not metastore.write(DEPS_META_FILE, json_dumps(meta)):
   985	        manager.log(f"Error writing fine-grained deps meta JSON file {DEPS_META_FILE}")
   986	        error = True
   987
   988	    if error:
   989	        manager.errors.set_file(_cache_dir_prefix(manager.options), None, manager.options)
   990	        manager.errors.report(0, 0, "Error writing fine-grained dependencies cache", blocker=True)

   993	def invert_deps(deps: dict[str, set[str]], graph: Graph) -> dict[str, dict[str, set[str]]]:
   994	    """Splits fine-grained dependencies based on the module of the trigger.
   995
   996	    Returns a dictionary from module ids to all dependencies on that
   997	    module. Dependencies not associated with a module in the build will be
   998	    associated with the nearest parent module that is in the build, or the
   999	    fake module FAKE_ROOT_MODULE if none are.
  1000	    """
  1001	    # Lazy import to speed up startup
  1002	    from mypy.server.target import trigger_to_target
  1003
  1004	    # Prepopulate the map for all the modules that have been processed,
  1005	    # so that we always generate files for processed modules (even if
  1006	    # there aren't any dependencies to them.)
  1007	    rdeps: dict[str, dict[str, set[str]]] = {id: {} for id, st in graph.items() if st.tree}
  1008	    for trigger, targets in deps.items():
  1009	        module = module_prefix(graph, trigger_to_target(trigger))
  1010	        if not module or not graph[module].tree:
  1011	            module = FAKE_ROOT_MODULE
  1012
  1013	        mod_rdeps = rdeps.setdefault(module, {})
  1014	        mod_rdeps.setdefault(trigger, set()).update(targets)
  1015
  1016	    return rdeps

  1019	def generate_deps_for_cache(manager: BuildManager, graph: Graph) -> dict[str, dict[str, set[str]]]:
  1020	    """Generate fine-grained dependencies into a form suitable for serializing.
  1021
  1022	    This does a couple things:
  1023	    1. Splits fine-grained deps based on the module of the trigger
  1024	    2. For each module we generated fine-grained deps for, load any previous
  1025	       deps and merge them in.
  1026
  1027	    Returns a dictionary from module ids to all dependencies on that
  1028	    module. Dependencies not associated with a module in the build will be
  1029	    associated with the nearest parent module that is in the build, or the
  1030	    fake module FAKE_ROOT_MODULE if none are.
  1031	    """
  1032	    from mypy.server.deps import merge_dependencies  # Lazy import to speed up startup
  1033
  1034	    # Split the dependencies out into based on the module that is depended on.
  1035	    rdeps = invert_deps(manager.fg_deps, graph)
  1036
  1037	    # We can't just clobber existing dependency information, so we
  1038	    # load the deps for every module we've generated new dependencies
  1039	    # to and merge the new deps into them.
  1040	    for module, mdeps in rdeps.items():
  1041	        old_deps = manager.load_fine_grained_deps(module)
  1042	        merge_dependencies(old_deps, mdeps)
  1043
  1044	    return rdeps

  1097	def read_deps_cache(manager: BuildManager, graph: Graph) -> dict[str, FgDepMeta] | None:
  1098	    """Read and validate the fine-grained dependencies cache.
  1099
  1100	    See the write_deps_cache documentation for more information on
  1101	    the details of the cache.
  1102
  1103	    Returns None if the cache was invalid in some way.
  1104	    """
  1105	    deps_meta = _load_json_file(
  1106	        DEPS_META_FILE,
  1107	        manager,
  1108	        log_success="Deps meta ",
  1109	        log_error="Could not load fine-grained dependency metadata: ",
  1110	    )
  1111	    if deps_meta is None:
  1112	        return None
  1113	    meta_snapshot = deps_meta["snapshot"]
  1114	    # Take a snapshot of the source hashes from all of the metas we found.
  1115	    # (Including the ones we rejected because they were out of date.)
  1116	    # We use this to verify that they match up with the proto_deps.
  1117	    current_meta_snapshot = {
  1118	        id: st.meta_source_hash for id, st in graph.items() if st.meta_source_hash is not None
  1119	    }
  1120
  1121	    common = set(meta_snapshot.keys()) & set(current_meta_snapshot.keys())
  1122	    if any(meta_snapshot[id] != current_meta_snapshot[id] for id in common):
  1123	        # TODO: invalidate also if options changed (like --strict-optional)?
  1124	        manager.log("Fine-grained dependencies cache inconsistent, ignoring")
  1125	        return None
  1126
  1127	    module_deps_metas = deps_meta["deps_meta"]
  1128	    assert isinstance(module_deps_metas, dict)
  1129	    if not manager.options.skip_cache_mtime_checks:
  1130	        for meta in module_deps_metas.values():
  1131	            try:
  1132	                matched = manager.getmtime(meta["path"]) == meta["mtime"]
  1133	            except FileNotFoundError:
  1134	                matched = False
  1135	            if not matched:
  1136	                manager.log(f"Invalid or missing fine-grained deps cache: {meta['path']}")
  1137	                return None
  1138
  1139	    return module_deps_metas

  1229	def get_cache_names(id: str, path: str, options: Options) -> tuple[str, str, str | None]:
  1230	    """Return the file names for the cache files.
  1231
  1232	    Args:
  1233	      id: module ID
  1234	      path: module path
  1235	      cache_dir: cache directory
  1236	      pyversion: Python version (major, minor)
  1237
  1238	    Returns:
  1239	      A tuple with the file names to be used for the meta JSON, the
  1240	      data JSON, and the fine-grained deps JSON, respectively.
  1241	    """
  1242	    if options.cache_map:
  1243	        pair = options.cache_map.get(normpath(path, options))
  1244	    else:
  1245	        pair = None
  1246	    if pair is not None:
  1247	        # The cache map paths were specified relative to the base directory,
  1248	        # but the filesystem metastore APIs operates relative to the cache
  1249	        # prefix directory.
  1250	        # Solve this by rewriting the paths as relative to the root dir.
  1251	        # This only makes sense when using the filesystem backed cache.
  1252	        root = _cache_dir_prefix(options)
  1253	        return (os.path.relpath(pair[0], root), os.path.relpath(pair[1], root), None)
  1254	    prefix = os.path.join(*id.split("."))
  1255	    is_package = os.path.basename(path).startswith("__init__.py")
  1256	    if is_package:
  1257	        prefix = os.path.join(prefix, "__init__")
  1258
  1259	    deps_json = None
  1260	    if options.cache_fine_grained:
  1261	        deps_json = prefix + ".deps.json"
  1262	    return (prefix + ".meta.json", prefix + ".data.json", deps_json)

  1265	def find_cache_meta(id: str, path: str, manager: BuildManager) -> CacheMeta | None:
  1266	    """Find cache data for a module.
  1267
  1268	    Args:
  1269	      id: module ID
  1270	      path: module path
  1271	      manager: the build manager (for pyversion, log/trace, and build options)
  1272
  1273	    Returns:
  1274	      A CacheMeta instance if the cache data was found and appears
  1275	      valid; otherwise None.
  1276	    """
  1277	    # TODO: May need to take more build options into account
  1278	    meta_json, data_json, _ = get_cache_names(id, path, manager.options)
  1279	    manager.trace(f"Looking for {id} at {meta_json}")
  1280	    t0 = time.time()
  1281	    meta = _load_json_file(
  1282	        meta_json, manager, log_success=f"Meta {id} ", log_error=f"Could not load cache for {id}: "
  1283	    )
  1284	    t1 = time.time()
  1285	    if meta is None:
  1286	        return None
  1287	    if not isinstance(meta, dict):
  1288	        manager.log(f"Could not load cache for {id}: meta cache is not a dict: {repr(meta)}")  # type: ignore[unreachable]
  1289	        return None
  1290	    m = cache_meta_from_dict(meta, data_json)
  1291	    t2 = time.time()
  1292	    manager.add_stats(
  1293	        load_meta_time=t2 - t0, load_meta_load_time=t1 - t0, load_meta_from_dict_time=t2 - t1
  1294	    )
  1295
  1296	    # Don't check for path match, that is dealt with in validate_meta().
  1297	    #
  1298	    # TODO: these `type: ignore`s wouldn't be necessary
  1299	    # if the type annotations for CacheMeta were more accurate
  1300	    # (all of these attributes can be `None`)
  1301	    if (
  1302	        m.id != id
  1303	        or m.mtime is None  # type: ignore[redundant-expr]
  1304	        or m.size is None  # type: ignore[redundant-expr]
  1305	        or m.dependencies is None  # type: ignore[redundant-expr]
  1306	        or m.data_mtime is None
  1307	    ):
  1308	        manager.log(f"Metadata abandoned for {id}: attributes are missing")
  1309	        return None
  1310
  1311	    # Ignore cache if generated by an older mypy version.
  1312	    if (
  1313	        (m.version_id != manager.version_id and not manager.options.skip_version_check)
  1314	        or m.options is None
  1315	        or len(m.dependencies) + len(m.suppressed) != len(m.dep_prios)
  1316	        or len(m.dependencies) + len(m.suppressed) != len(m.dep_lines)
  1317	    ):
  1318	        manager.log(f"Metadata abandoned for {id}: new attributes are missing")
  1319	        return None
  1320
  1321	    # Ignore cache if (relevant) options aren't the same.
  1322	    # Note that it's fine to mutilate cached_options since it's only used here.
  1323	    cached_options = m.options
  1324	    current_options = manager.options.clone_for_module(id).select_options_affecting_cache()
  1325	    if manager.options.skip_version_check:
  1326	        # When we're lax about version we're also lax about platform.
  1327	        cached_options["platform"] = current_options["platform"]
  1328	    if "debug_cache" in cached_options:
  1329	        # Older versions included debug_cache, but it's silly to compare it.
  1330	        del cached_options["debug_cache"]
  1331	    if cached_options != current_options:
  1332	        manager.log(f"Metadata abandoned for {id}: options differ")
  1333	        if manager.options.verbosity >= 2:
  1334	            for key in sorted(set(cached_options) | set(current_options)):
  1335	                if cached_options.get(key) != current_options.get(key):
  1336	                    manager.trace(
  1337	                        "    {}: {} != {}".format(
  1338	                            key, cached_options.get(key), current_options.get(key)
  1339	                        )
  1340	                    )
  1341	        return None
  1342	    if manager.old_plugins_snapshot and manager.plugins_snapshot:
  1343	        # Check if plugins are still the same.
  1344	        if manager.plugins_snapshot != manager.old_plugins_snapshot:
  1345	            manager.log(f"Metadata abandoned for {id}: plugins differ")
  1346	            return None
  1347	    # So that plugins can return data with tuples in it without
  1348	    # things silently always invalidating modules, we round-trip
  1349	    # the config data. This isn't beautiful.
  1350	    plugin_data = json_loads(
  1351	        json_dumps(manager.plugin.report_config_data(ReportConfigContext(id, path, is_check=True)))
  1352	    )
  1353	    if m.plugin_data != plugin_data:
  1354	        manager.log(f"Metadata abandoned for {id}: plugin configuration differs")
  1355	        return None
  1356
  1357	    manager.add_stats(fresh_metas=1)
  1358	    return m

  1361	def validate_meta(
  1362	    meta: CacheMeta | None, id: str, path: str | None, ignore_all: bool, manager: BuildManager
  1363	) -> CacheMeta | None:
  1364	    """Checks whether the cached AST of this module can be used.
  1365
  1366	    Returns:
  1367	      None, if the cached AST is unusable.
  1368	      Original meta, if mtime/size matched.
  1369	      Meta with mtime updated to match source file, if hash/size matched but mtime/path didn't.
  1370	    """
  1371	    # This requires two steps. The first one is obvious: we check that the module source file
  1372	    # contents is the same as it was when the cache data file was created. The second one is not
  1373	    # too obvious: we check that the cache data file mtime has not changed; it is needed because
  1374	    # we use cache data file mtime to propagate information about changes in the dependencies.
  1375
  1376	    if meta is None:
  1377	        manager.log(f"Metadata not found for {id}")
  1378	        return None
  1379
  1380	    if meta.ignore_all and not ignore_all:
  1381	        manager.log(f"Metadata abandoned for {id}: errors were previously ignored")
  1382	        return None
  1383
  1384	    t0 = time.time()
  1385	    bazel = manager.options.bazel
  1386	    assert path is not None, "Internal error: meta was provided without a path"
  1387	    if not manager.options.skip_cache_mtime_checks:
  1388	        # Check data_json; assume if its mtime matches it's good.
  1389	        try:
  1390	            data_mtime = manager.getmtime(meta.data_json)
  1391	        except OSError:
  1392	            manager.log(f"Metadata abandoned for {id}: failed to stat data_json")
  1393	            return None
  1394	        if data_mtime != meta.data_mtime:
  1395	            manager.log(f"Metadata abandoned for {id}: data cache is modified")
  1396	            return None
  1397
  1398	    if bazel:
  1399	        # Normalize path under bazel to make sure it isn't absolute
  1400	        path = normpath(path, manager.options)
  1401
  1402	    st = manager.get_stat(path)
  1403	    if st is None:
  1404	        return None
  1405	    if not stat.S_ISDIR(st.st_mode) and not stat.S_ISREG(st.st_mode):
  1406	        manager.log(f"Metadata abandoned for {id}: file or directory {path} does not exist")
  1407	        return None
  1408
  1409	    manager.add_stats(validate_stat_time=time.time() - t0)
  1410
  1411	    # When we are using a fine-grained cache, we want our initial
  1412	    # build() to load all of the cache information and then do a
  1413	    # fine-grained incremental update to catch anything that has
  1414	    # changed since the cache was generated. We *don't* want to do a
  1415	    # coarse-grained incremental rebuild, so we accept the cache
  1416	    # metadata even if it doesn't match the source file.
  1417	    #
  1418	    # We still *do* the mtime/hash checks, however, to enable
  1419	    # fine-grained mode to take advantage of the mtime-updating
  1420	    # optimization when mtimes differ but hashes match.  There is
  1421	    # essentially no extra time cost to computing the hash here, since
  1422	    # it will be cached and will be needed for finding changed files
  1423	    # later anyways.
  1424	    fine_grained_cache = manager.use_fine_grained_cache()
  1425
  1426	    size = st.st_size
  1427	    # Bazel ensures the cache is valid.
  1428	    if size != meta.size and not bazel and not fine_grained_cache:
  1429	        manager.log(f"Metadata abandoned for {id}: file {path} has different size")
  1430	        return None
  1431
  1432	    # Bazel ensures the cache is valid.
  1433	    mtime = 0 if bazel else int(st.st_mtime)
  1434	    if not bazel and (mtime != meta.mtime or path != meta.path):
  1435	        if manager.quickstart_state and path in manager.quickstart_state:
  1436	            # If the mtime and the size of the file recorded in the quickstart dump matches
  1437	            # what we see on disk, we know (assume) that the hash matches the quickstart
  1438	            # data as well. If that hash matches the hash in the metadata, then we know
  1439	            # the file is up to date even though the mtime is wrong, without needing to hash it.
  1440	            qmtime, qsize, qhash = manager.quickstart_state[path]
  1441	            if int(qmtime) == mtime and qsize == size and qhash == meta.hash:
  1442	                manager.log(f"Metadata fresh (by quickstart) for {id}: file {path}")
  1443	                meta = meta._replace(mtime=mtime, path=path)
  1444	                return meta
  1445
  1446	        t0 = time.time()
  1447	        try:
  1448	            # dir means it is a namespace package
  1449	            if stat.S_ISDIR(st.st_mode):
  1450	                source_hash = ""
  1451	            else:
  1452	                source_hash = manager.fscache.hash_digest(path)
  1453	        except (OSError, UnicodeDecodeError, DecodeError):
  1454	            return None
  1455	        manager.add_stats(validate_hash_time=time.time() - t0)
  1456	        if source_hash != meta.hash:
  1457	            if fine_grained_cache:
  1458	                manager.log(f"Using stale metadata for {id}: file {path}")
  1459	                return meta
  1460	            else:
  1461	                manager.log(f"Metadata abandoned for {id}: file {path} has different hash")
  1462	                return None
  1463	        else:
  1464	            t0 = time.time()
  1465	            # Optimization: update mtime and path (otherwise, this mismatch will reappear).
  1466	            meta = meta._replace(mtime=mtime, path=path)
  1467	            # Construct a dict we can pass to json.dumps() (compare to write_cache()).
  1468	            meta_dict = {
  1469	                "id": id,
  1470	                "path": path,
  1471	                "mtime": mtime,
  1472	                "size": size,
  1473	                "hash": source_hash,
  1474	                "data_mtime": meta.data_mtime,
  1475	                "dependencies": meta.dependencies,
  1476	                "suppressed": meta.suppressed,
  1477	                "options": (manager.options.clone_for_module(id).select_options_affecting_cache()),
  1478	                "dep_prios": meta.dep_prios,
  1479	                "dep_lines": meta.dep_lines,
  1480	                "interface_hash": meta.interface_hash,
  1481	                "version_id": manager.version_id,
  1482	                "ignore_all": meta.ignore_all,
  1483	                "plugin_data": meta.plugin_data,
  1484	            }
  1485	            meta_bytes = json_dumps(meta_dict, manager.options.debug_cache)
  1486	            meta_json, _, _ = get_cache_names(id, path, manager.options)
  1487	            manager.log(
  1488	                "Updating mtime for {}: file {}, meta {}, mtime {}".format(
  1489	                    id, path, meta_json, meta.mtime
  1490	                )
  1491	            )
  1492	            t1 = time.time()
  1493	            manager.metastore.write(meta_json, meta_bytes)  # Ignore errors, just an optimization.
  1494	            manager.add_stats(validate_update_time=time.time() - t1, validate_munging_time=t1 - t0)
  1495	            return meta
  1496
  1497	    # It's a match on (id, path, size, hash, mtime).
  1498	    manager.log(f"Metadata fresh for {id}: file {path}")
  1499	    return meta

  1502	def compute_hash(text: str) -> str:
  1503	    # We use a crypto hash instead of the builtin hash(...) function
  1504	    # because the output of hash(...)  can differ between runs due to
  1505	    # hash randomization (enabled by default in Python 3.3).  See the
  1506	    # note in
  1507	    # https://docs.python.org/3/reference/datamodel.html#object.__hash__.
  1508	    return hash_digest(text.encode("utf-8"))

  1511	def write_cache(
  1512	    id: str,
  1513	    path: str,
  1514	    tree: MypyFile,
  1515	    dependencies: list[str],
  1516	    suppressed: list[str],
  1517	    dep_prios: list[int],
  1518	    dep_lines: list[int],
  1519	    old_interface_hash: str,
  1520	    source_hash: str,
  1521	    ignore_all: bool,
  1522	    manager: BuildManager,
  1523	) -> tuple[str, CacheMeta | None]:
  1524	    """Write cache files for a module.
  1525
  1526	    Note that this mypy's behavior is still correct when any given
  1527	    write_cache() call is replaced with a no-op, so error handling
  1528	    code that bails without writing anything is okay.
  1529
  1530	    Args:
  1531	      id: module ID
  1532	      path: module path
  1533	      tree: the fully checked module data
  1534	      dependencies: module IDs on which this module depends
  1535	      suppressed: module IDs which were suppressed as dependencies
  1536	      dep_prios: priorities (parallel array to dependencies)
  1537	      dep_lines: import line locations (parallel array to dependencies)
  1538	      old_interface_hash: the hash from the previous version of the data cache file
  1539	      source_hash: the hash of the source code
  1540	      ignore_all: the ignore_all flag for this module
  1541	      manager: the build manager (for pyversion, log/trace)
  1542
  1543	    Returns:
  1544	      A tuple containing the interface hash and CacheMeta
  1545	      corresponding to the metadata that was written (the latter may
  1546	      be None if the cache could not be written).
  1547	    """
  1548	    metastore = manager.metastore
  1549	    # For Bazel we use relative paths and zero mtimes.
  1550	    bazel = manager.options.bazel
  1551
  1552	    # Obtain file paths.
  1553	    meta_json, data_json, _ = get_cache_names(id, path, manager.options)
  1554	    manager.log(f"Writing {id} {path} {meta_json} {data_json}")
  1555
  1556	    # Update tree.path so that in bazel mode it's made relative (since
  1557	    # sometimes paths leak out).
  1558	    if bazel:
  1559	        tree.path = path
  1560
  1561	    # Serialize data and analyze interface
  1562	    data = tree.serialize()
  1563	    data_bytes = json_dumps(data, manager.options.debug_cache)
  1564	    interface_hash = hash_digest(data_bytes)
  1565
  1566	    plugin_data = manager.plugin.report_config_data(ReportConfigContext(id, path, is_check=False))
  1567
  1568	    # Obtain and set up metadata
  1569	    st = manager.get_stat(path)
  1570	    if st is None:
  1571	        manager.log(f"Cannot get stat for {path}")
  1572	        # Remove apparently-invalid cache files.
  1573	        # (This is purely an optimization.)
  1574	        for filename in [data_json, meta_json]:
  1575	            try:
  1576	                os.remove(filename)
  1577	            except OSError:
  1578	                pass
  1579	        # Still return the interface hash we computed.
  1580	        return interface_hash, None
  1581
  1582	    # Write data cache file, if applicable
  1583	    # Note that for Bazel we don't record the data file's mtime.
  1584	    if old_interface_hash == interface_hash:
  1585	        manager.trace(f"Interface for {id} is unchanged")
  1586	    else:
  1587	        manager.trace(f"Interface for {id} has changed")
  1588	        if not metastore.write(data_json, data_bytes):
  1589	            # Most likely the error is the replace() call
  1590	            # (see https://github.com/python/mypy/issues/3215).
  1591	            manager.log(f"Error writing data JSON file {data_json}")
  1592	            # Let's continue without writing the meta file.  Analysis:
  1593	            # If the replace failed, we've changed nothing except left
  1594	            # behind an extraneous temporary file; if the replace
  1595	            # worked but the getmtime() call failed, the meta file
  1596	            # will be considered invalid on the next run because the
  1597	            # data_mtime field won't match the data file's mtime.
  1598	            # Both have the effect of slowing down the next run a
  1599	            # little bit due to an out-of-date cache file.
  1600	            return interface_hash, None
  1601
  1602	    try:
  1603	        data_mtime = manager.getmtime(data_json)
  1604	    except OSError:
  1605	        manager.log(f"Error in os.stat({data_json!r}), skipping cache write")
  1606	        return interface_hash, None
  1607
  1608	    mtime = 0 if bazel else int(st.st_mtime)
  1609	    size = st.st_size
  1610	    # Note that the options we store in the cache are the options as
  1611	    # specified by the command line/config file and *don't* reflect
  1612	    # updates made by inline config directives in the file. This is
  1613	    # important, or otherwise the options would never match when
  1614	    # verifying the cache.
  1615	    options = manager.options.clone_for_module(id)
  1616	    assert source_hash is not None
  1617	    meta = {
  1618	        "id": id,
  1619	        "path": path,
  1620	        "mtime": mtime,
  1621	        "size": size,
  1622	        "hash": source_hash,
  1623	        "data_mtime": data_mtime,
  1624	        "dependencies": dependencies,
  1625	        "suppressed": suppressed,
  1626	        "options": options.select_options_affecting_cache(),
  1627	        "dep_prios": dep_prios,
  1628	        "dep_lines": dep_lines,
  1629	        "interface_hash": interface_hash,
  1630	        "version_id": manager.version_id,
  1631	        "ignore_all": ignore_all,
  1632	        "plugin_data": plugin_data,
  1633	    }
  1634
  1635	    # Write meta cache file
  1636	    meta_str = json_dumps(meta, manager.options.debug_cache)
  1637	    if not metastore.write(meta_json, meta_str):
  1638	        # Most likely the error is the replace() call
  1639	        # (see https://github.com/python/mypy/issues/3215).
  1640	        # The next run will simply find the cache entry out of date.
  1641	        manager.log(f"Error writing meta JSON file {meta_json}")
  1642
  1643	    return interface_hash, cache_meta_from_dict(meta, data_json)

  1646	def delete_cache(id: str, path: str, manager: BuildManager) -> None:
  1647	    """Delete cache files for a module.
  1648
  1649	    The cache files for a module are deleted when mypy finds errors there.
  1650	    This avoids inconsistent states with cache files from different mypy runs,
  1651	    see #4043 for an example.
  1652	    """
  1653	    # We don't delete .deps files on errors, since the dependencies
  1654	    # are mostly generated from other files and the metadata is
  1655	    # tracked separately.
  1656	    meta_path, data_path, _ = get_cache_names(id, path, manager.options)
  1657	    cache_paths = [meta_path, data_path]
  1658	    manager.log(f"Deleting {id} {path} {' '.join(x for x in cache_paths if x)}")
  1659
  1660	    for filename in cache_paths:
  1661	        try:
  1662	            manager.metastore.remove(filename)
  1663	        except OSError as e:
  1664	            if e.errno != errno.ENOENT:
  1665	                manager.log(f"Error deleting cache file {filename}: {e.strerror}")

  1667
  1668	"""Dependency manager.
  1669
  1670	Design
  1671	======
  1672
  1673	Ideally
  1674	-------
  1675
  1676	A. Collapse cycles (each SCC -- strongly connected component --
  1677	   becomes one "supernode").
  1678
  1679	B. Topologically sort nodes based on dependencies.
  1680
  1681	C. Process from leaves towards roots.
  1682
  1683	Wrinkles
  1684	--------
  1685
  1686	a. Need to parse source modules to determine dependencies.
  1687
  1688	b. Processing order for modules within an SCC.
  1689
  1690	c. Must order mtimes of files to decide whether to re-process; depends
  1691	   on clock never resetting.
  1692
  1693	d. from P import M; checks filesystem whether module P.M exists in
  1694	   filesystem.
  1695
  1696	e. Race conditions, where somebody modifies a file while we're
  1697	   processing. Solved by using a FileSystemCache.
  1698
  1699
  1700	Steps
  1701	-----
  1702
  1703	1. For each explicitly given module find the source file location.
  1704
  1705	2. For each such module load and check the cache metadata, and decide
  1706	   whether it's valid.
  1707
  1708	3. Now recursively (or iteratively) find dependencies and add those to
  1709	   the graph:
  1710
  1711	   - for cached nodes use the list of dependencies from the cache
  1712	     metadata (this will be valid even if we later end up re-parsing
  1713	     the same source);
  1714
  1715	   - for uncached nodes parse the file and process all imports found,
  1716	     taking care of (a) above.
  1717
  1718	Step 3 should also address (d) above.
  1719
  1720	Once step 3 terminates we have the entire dependency graph, and for
  1721	each module we've either loaded the cache metadata or parsed the
  1722	source code.  (However, we may still need to parse those modules for
  1723	which we have cache metadata but that depend, directly or indirectly,
  1724	on at least one module for which the cache metadata is stale.)
  1725
  1726	Now we can execute steps A-C from the first section.  Finding SCCs for
  1727	step A shouldn't be hard; there's a recipe here:
  1728	https://code.activestate.com/recipes/578507/.  There's also a plethora
  1729	of topsort recipes, e.g. https://code.activestate.com/recipes/577413/.
  1730
  1731	For single nodes, processing is simple.  If the node was cached, we
  1732	deserialize the cache data and fix up cross-references.  Otherwise, we
  1733	do semantic analysis followed by type checking.  We also handle (c)
  1734	above; if a module has valid cache data *but* any of its
  1735	dependencies was processed from source, then the module should be
  1736	processed from source.
  1737
  1738	A relatively simple optimization (outside SCCs) we might do in the
  1739	future is as follows: if a node's cache data is valid, but one or more
  1740	of its dependencies are out of date so we have to re-parse the node
  1741	from source, once we have fully type-checked the node, we can decide
  1742	whether its symbol table actually changed compared to the cache data
  1743	(by reading the cache data and comparing it to the data we would be
  1744	writing).  If there is no change we can declare the node up to date,
  1745	and any node that depends (and for which we have cached data, and
  1746	whose other dependencies are up to date) on it won't need to be
  1747	re-parsed from source.
  1748
  1749	Import cycles
  1750	-------------
  1751
  1752	Finally we have to decide how to handle (c), import cycles.  Here
  1753	we'll need a modified version of the original state machine
  1754	(build.py), but we only need to do this per SCC, and we won't have to
  1755	deal with changes to the list of nodes while we're processing it.
  1756
  1757	If all nodes in the SCC have valid cache metadata and all dependencies
  1758	outside the SCC are still valid, we can proceed as follows:
  1759
  1760	  1. Load cache data for all nodes in the SCC.
  1761
  1762	  2. Fix up cross-references for all nodes in the SCC.
  1763
  1764	Otherwise, the simplest (but potentially slow) way to proceed is to
  1765	invalidate all cache data in the SCC and re-parse all nodes in the SCC
  1766	from source.  We can do this as follows:
  1767
  1768	  1. Parse source for all nodes in the SCC.
  1769
  1770	  2. Semantic analysis for all nodes in the SCC.
  1771
  1772	  3. Type check all nodes in the SCC.
  1773
  1774	(If there are more passes the process is the same -- each pass should
  1775	be done for all nodes before starting the next pass for any nodes in
  1776	the SCC.)
  1777
  1778	We could process the nodes in the SCC in any order.  For sentimental
  1779	reasons, I've decided to process them in the reverse order in which we
  1780	encountered them when originally constructing the graph.  That's how
  1781	the old build.py deals with cycles, and at least this reproduces the
  1782	previous implementation more accurately.
  1783
  1784	Can we do better than re-parsing all nodes in the SCC when any of its
  1785	dependencies are out of date?  It's doubtful.  The optimization
  1786	mentioned at the end of the previous section would require re-parsing
  1787	and type-checking a node and then comparing its symbol table to the
  1788	cached data; but because the node is part of a cycle we can't
  1789	technically type-check it until the semantic analysis of all other
  1790	nodes in the cycle has completed.  (This is an important issue because
  1791	Dropbox has a very large cycle in production code.  But I'd like to
  1792	deal with it later.)
  1793
  1794	Additional wrinkles
  1795	-------------------
  1796
  1797	During implementation more wrinkles were found.
  1798
  1799	- When a submodule of a package (e.g. x.y) is encountered, the parent
  1800	  package (e.g. x) must also be loaded, but it is not strictly a
  1801	  dependency.  See State.add_ancestors() below.
  1802	"""
