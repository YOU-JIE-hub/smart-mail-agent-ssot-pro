  5051	        module-level constant definitions.

  5053	        Limitations:
  5054	         - no active type context
  5055	         - no star expressions
  5056	         - the joined type of all entries must be an Instance or Tuple type

  5058	        ctx = self.type_context[-1]
  5059	        if ctx:
  5060	            return None
  5061	        rt = self.resolved_type.get(e, None)
  5062	        if rt is not None:
  5063	            return rt if isinstance(rt, Instance) else None
  5064	        values: list[Type] = []
  5065	        for item in e.items:
  5066	            if isinstance(item, StarExpr):
  5067	                # fallback to slow path
  5068	                self.resolved_type[e] = NoneType()
  5069	                return None
  5070	            values.append(self.accept(item))
  5071	        vt = join.join_type_list(values)
  5072	        if not allow_fast_container_literal(vt):
  5073	            self.resolved_type[e] = NoneType()
  5074	            return None
  5075	        ct = self.chk.named_generic_type(container_fullname, [vt])
  5076	        self.resolved_type[e] = ct
  5077	        return ct

  5114	    def tuple_context_matches(self, expr: TupleExpr, ctx: TupleType) -> bool:
  5115	        ctx_unpack_index = find_unpack_in_list(ctx.items)
  5116	        if ctx_unpack_index is None:
  5117	            # For fixed tuples accept everything that can possibly match, even if this
  5118	            # requires all star items to be empty.
  5119	            return len([e for e in expr.items if not isinstance(e, StarExpr)]) <= len(ctx.items)
  5120	        # For variadic context, the only easy case is when structure matches exactly.
  5121	        # TODO: try using tuple type context in more cases.
  5122	        if len([e for e in expr.items if isinstance(e, StarExpr)]) != 1:
  5123	            return False
  5124	        expr_star_index = next(i for i, lv in enumerate(expr.items) if isinstance(lv, StarExpr))
  5125	        return len(expr.items) == len(ctx.items) and ctx_unpack_index == expr_star_index

  5127	    def visit_tuple_expr(self, e: TupleExpr) -> Type:
  5128	        """Type check a tuple expression."""
  5129	        # Try to determine type context for type inference.
  5130	        type_context = get_proper_type(self.type_context[-1])
  5131	        type_context_items = None
  5132	        if isinstance(type_context, UnionType):
  5133	            tuples_in_context = [
  5134	                t
  5135	                for t in get_proper_types(type_context.items)
  5136	                if (isinstance(t, TupleType) and self.tuple_context_matches(e, t))
  5137	                or is_named_instance(t, TUPLE_LIKE_INSTANCE_NAMES)
  5138	            ]
  5139	            if len(tuples_in_context) == 1:
  5140	                type_context = tuples_in_context[0]
  5141	            else:
  5142	                # There are either no relevant tuples in the Union, or there is
  5143	                # more than one.  Either way, we can't decide on a context.
  5144	                pass
  5145
  5146	        if isinstance(type_context, TupleType) and self.tuple_context_matches(e, type_context):
  5147	            type_context_items = type_context.items
  5148	        elif type_context and is_named_instance(type_context, TUPLE_LIKE_INSTANCE_NAMES):
  5149	            assert isinstance(type_context, Instance)
  5150	            if type_context.args:
  5151	                type_context_items = [type_context.args[0]] * len(e.items)
  5152	        # NOTE: it's possible for the context to have a different
  5153	        # number of items than e.  In that case we use those context
  5154	        # items that match a position in e, and we'll worry about type
  5155	        # mismatches later.
  5156
  5157	        unpack_in_context = False
  5158	        if type_context_items is not None:
  5159	            unpack_in_context = find_unpack_in_list(type_context_items) is not None
  5160	        seen_unpack_in_items = False
  5161	        allow_precise_tuples = (
  5162	            unpack_in_context or PRECISE_TUPLE_TYPES in self.chk.options.enable_incomplete_feature
  5163	        )
  5164
  5165	        # Infer item types.  Give up if there's a star expression
  5166	        # that's not a Tuple.
  5167	        items: list[Type] = []
  5168	        j = 0  # Index into type_context_items; irrelevant if type_context_items is none
  5169	        for i in range(len(e.items)):
  5170	            item = e.items[i]
  5171	            if isinstance(item, StarExpr):
  5172	                # Special handling for star expressions.
  5173	                # TODO: If there's a context, and item.expr is a
  5174	                # TupleExpr, flatten it, so we can benefit from the
  5175	                # context?  Counterargument: Why would anyone write
  5176	                # (1, *(2, 3)) instead of (1, 2, 3) except in a test?
  5177	                if unpack_in_context:
  5178	                    # Note: this logic depends on full structure match in tuple_context_matches().
  5179	                    assert type_context_items
  5180	                    ctx_item = type_context_items[j]
  5181	                    assert isinstance(ctx_item, UnpackType)
  5182	                    ctx = ctx_item.type
  5183	                else:
  5184	                    ctx = None
  5185	                tt = self.accept(item.expr, ctx)
  5186	                tt = get_proper_type(tt)
  5187	                if isinstance(tt, TupleType):
  5188	                    if find_unpack_in_list(tt.items) is not None:
  5189	                        if seen_unpack_in_items:
  5190	                            # Multiple unpack items are not allowed in tuples,
  5191	                            # fall back to instance type.
  5192	                            return self.check_lst_expr(e, "builtins.tuple", "<tuple>")
  5193	                        else:
  5194	                            seen_unpack_in_items = True
  5195	                    items.extend(tt.items)
  5196	                    # Note: this logic depends on full structure match in tuple_context_matches().
  5197	                    if unpack_in_context:
  5198	                        j += 1
  5199	                    else:
  5200	                        # If there is an unpack in expressions, but not in context, this will
  5201	                        # result in an error later, just do something predictable here.
  5202	                        j += len(tt.items)
  5203	                else:
  5204	                    if allow_precise_tuples and not seen_unpack_in_items:
  5205	                        # Handle (x, *y, z), where y is e.g. tuple[Y, ...].
  5206	                        if isinstance(tt, Instance) and self.chk.type_is_iterable(tt):
  5207	                            item_type = self.chk.iterable_item_type(tt, e)
  5208	                            mapped = self.chk.named_generic_type("builtins.tuple", [item_type])
  5209	                            items.append(UnpackType(mapped))
  5210	                            seen_unpack_in_items = True
  5211	                            continue
  5212	                    # A star expression that's not a Tuple.
  5213	                    # Treat the whole thing as a variable-length tuple.
  5214	                    return self.check_lst_expr(e, "builtins.tuple", "<tuple>")
  5215	            else:
  5216	                if not type_context_items or j >= len(type_context_items):
  5217	                    tt = self.accept(item)
  5218	                else:
  5219	                    tt = self.accept(item, type_context_items[j])
  5220	                    j += 1
  5221	                items.append(tt)
  5222	        # This is a partial fallback item type. A precise type will be calculated on demand.
  5223	        fallback_item = AnyType(TypeOfAny.special_form)
  5224	        result: ProperType = TupleType(
  5225	            items, self.chk.named_generic_type("builtins.tuple", [fallback_item])
  5226	        )
  5227	        if seen_unpack_in_items:
  5228	            # Return already normalized tuple type just in case.
  5229	            result = expand_type(result, {})
  5230	        return result

  5232	    def fast_dict_type(self, e: DictExpr) -> Type | None:
  5233	        """
  5234	        Fast path to determine the type of a dict literal,
  5235	        based on the list of entries. This mostly impacts large
  5236	        module-level constant definitions.
  5237
  5238	        Limitations:
  5239	         - no active type context
  5240	         - only supported star expressions are other dict instances
  5241	         - the joined types of all keys and values must be Instance or Tuple types
  5242	        """
  5243	        ctx = self.type_context[-1]
  5244	        if ctx:
  5245	            return None
  5246	        rt = self.resolved_type.get(e, None)
  5247	        if rt is not None:
  5248	            return rt if isinstance(rt, Instance) else None
  5249	        keys: list[Type] = []
  5250	        values: list[Type] = []
  5251	        stargs: tuple[Type, Type] | None = None
  5252	        for key, value in e.items:
  5253	            if key is None:
  5254	                st = get_proper_type(self.accept(value))
  5255	                if (
  5256	                    isinstance(st, Instance)
  5257	                    and st.type.fullname == "builtins.dict"
  5258	                    and len(st.args) == 2
  5259	                ):
  5260	                    stargs = (st.args[0], st.args[1])
  5261	                else:
  5262	                    self.resolved_type[e] = NoneType()
  5263	                    return None
  5264	            else:
  5265	                keys.append(self.accept(key))
  5266	                values.append(self.accept(value))
  5267	        kt = join.join_type_list(keys)
  5268	        vt = join.join_type_list(values)
  5269	        if not (allow_fast_container_literal(kt) and allow_fast_container_literal(vt)):
  5270	            self.resolved_type[e] = NoneType()
  5271	            return None
  5272	        if stargs and (stargs[0] != kt or stargs[1] != vt):
  5273	            self.resolved_type[e] = NoneType()
  5274	            return None
  5275	        dt = self.chk.named_generic_type("builtins.dict", [kt, vt])
  5276	        self.resolved_type[e] = dt
  5277	        return dt

  5290	    def visit_dict_expr(self, e: DictExpr) -> Type:
  5291	        """Type check a dict expression.
  5292
  5293	        Translate it into a call to dict(), with provisions for **expr.
  5294	        """
  5295	        # if the dict literal doesn't match TypedDict, check_typeddict_call_with_dict reports
  5296	        # an error, but returns the TypedDict type that matches the literal it found
  5297	        # that would cause a second error when that TypedDict type is returned upstream
  5298	        # to avoid the second error, we always return TypedDict type that was requested
  5299	        typeddict_contexts = self.find_typeddict_context(self.type_context[-1], e)
  5300	        if typeddict_contexts:
  5301	            if len(typeddict_contexts) == 1:
  5302	                return self.check_typeddict_literal_in_context(e, typeddict_contexts[0])
  5303	            # Multiple items union, check if at least one of them matches cleanly.
  5304	            for typeddict_context in typeddict_contexts:
  5305	                with self.msg.filter_errors() as err, self.chk.local_type_map() as tmap:
  5306	                    ret_type = self.check_typeddict_literal_in_context(e, typeddict_context)
  5307	                if err.has_new_errors():
  5308	                    continue
  5309	                self.chk.store_types(tmap)
  5310	                return ret_type
  5311	            # No item matched without an error, so we can't unambiguously choose the item.
  5312	            self.msg.typeddict_context_ambiguous(typeddict_contexts, e)
  5313
  5314	        # fast path attempt
  5315	        dt = self.fast_dict_type(e)
  5316	        if dt:
  5317	            return dt
  5318
  5319	        # Define type variables (used in constructors below).
  5320	        kt = TypeVarType(
  5321	            "KT",
  5322	            "KT",
  5323	            id=TypeVarId(-1, namespace="<dict>"),
  5324	            values=[],
  5325	            upper_bound=self.object_type(),
  5326	            default=AnyType(TypeOfAny.from_omitted_generics),
  5327	        )
  5328	        vt = TypeVarType(
  5329	            "VT",
  5330	            "VT",
  5331	            id=TypeVarId(-2, namespace="<dict>"),
  5332	            values=[],
  5333	            upper_bound=self.object_type(),
  5334	            default=AnyType(TypeOfAny.from_omitted_generics),
  5335	        )
  5336
  5337	        # Collect function arguments, watching out for **expr.
  5338	        args: list[Expression] = []
  5339	        expected_types: list[Type] = []
  5340	        for key, value in e.items:
  5341	            if key is None:
  5342	                args.append(value)
  5343	                expected_types.append(
  5344	                    self.chk.named_generic_type("_typeshed.SupportsKeysAndGetItem", [kt, vt])
  5345	                )
  5346	            else:
  5347	                tup = TupleExpr([key, value])
  5348	                if key.line >= 0:
  5349	                    tup.line = key.line
  5350	                    tup.column = key.column
  5351	                else:
  5352	                    tup.line = value.line
  5353	                    tup.column = value.column
  5354	                tup.end_line = value.end_line
  5355	                tup.end_column = value.end_column
  5356	                args.append(tup)
  5357	                expected_types.append(TupleType([kt, vt], self.named_type("builtins.tuple")))
  5358
  5359	        # The callable type represents a function like this (except we adjust for **expr):
  5360	        #   def <dict>(*v: Tuple[kt, vt]) -> Dict[kt, vt]: ...
  5361	        constructor = CallableType(
  5362	            expected_types,
  5363	            [nodes.ARG_POS] * len(expected_types),
  5364	            [None] * len(expected_types),
  5365	            self.chk.named_generic_type("builtins.dict", [kt, vt]),
  5366	            self.named_type("builtins.function"),
  5367	            name="<dict>",
  5368	            variables=[kt, vt],
  5369	        )
  5370	        return self.check_call(constructor, args, [nodes.ARG_POS] * len(args), e)[0]

  5391	    def visit_lambda_expr(self, e: LambdaExpr) -> Type:
  5392	        """Type check lambda expression."""
  5393	        self.chk.check_default_args(e, body_is_trivial=False)
  5394	        inferred_type, type_override = self.infer_lambda_type_using_context(e)
  5395	        if not inferred_type:
  5396	            self.chk.return_types.append(AnyType(TypeOfAny.special_form))
  5397	            # Type check everything in the body except for the final return
  5398	            # statement (it can contain tuple unpacking before return).
  5399	            with (
  5400	                self.chk.binder.frame_context(can_skip=True, fall_through=0),
  5401	                self.chk.scope.push_function(e),
  5402	            ):
  5403	                # Lambdas can have more than one element in body,
  5404	                # when we add "fictional" AssignmentStatement nodes, like in:
  5405	                # `lambda (a, b): a`
  5406	                for stmt in e.body.body[:-1]:
  5407	                    stmt.accept(self.chk)
  5408	                # Only type check the return expression, not the return statement.
  5409	                # There's no useful type context.
  5410	                ret_type = self.accept(e.expr(), allow_none_return=True)
  5411	            fallback = self.named_type("builtins.function")
  5412	            self.chk.return_types.pop()
  5413	            return callable_type(e, fallback, ret_type)
  5414	        else:
  5415	            # Type context available.
  5416	            self.chk.return_types.append(inferred_type.ret_type)
  5417	            with self.chk.tscope.function_scope(e):
  5418	                self.chk.check_func_item(e, type_override=type_override)
  5419	            if not self.chk.has_type(e.expr()):
  5420	                # TODO: return expression must be accepted before exiting function scope.
  5421	                with self.chk.binder.frame_context(can_skip=True, fall_through=0):
  5422	                    self.accept(e.expr(), allow_none_return=True)
  5423	            ret_type = self.chk.lookup_type(e.expr())
  5424	            self.chk.return_types.pop()
  5425	            return replace_callable_return_type(inferred_type, ret_type)

  5427	    def infer_lambda_type_using_context(
  5428	        self, e: LambdaExpr
  5429	    ) -> tuple[CallableType | None, CallableType | None]:
  5430	        """Try to infer lambda expression type using context.
  5431
  5432	        Return None if could not infer type.
  5433	        The second item in the return type is the type_override parameter for check_func_item.
  5434	        """
  5435	        # TODO also accept 'Any' context
  5436	        ctx = get_proper_type(self.type_context[-1])
  5437
  5438	        if isinstance(ctx, UnionType):
  5439	            callables = [
  5440	                t for t in get_proper_types(ctx.relevant_items()) if isinstance(t, CallableType)
  5441	            ]
  5442	            if len(callables) == 1:
  5443	                ctx = callables[0]
  5444
  5445	        if not ctx or not isinstance(ctx, CallableType):
  5446	            return None, None
  5447
  5448	        # The context may have function type variables in it. We replace them
  5449	        # since these are the type variables we are ultimately trying to infer;
  5450	        # they must be considered as indeterminate. We use ErasedType since it
  5451	        # does not affect type inference results (it is for purposes like this
  5452	        # only).
  5453	        if not self.chk.options.old_type_inference:
  5454	            # With new type inference we can preserve argument types even if they
  5455	            # are generic, since new inference algorithm can handle constraints
  5456	            # like S <: T (we still erase return type since it's ultimately unknown).
  5457	            extra_vars = []
  5458	            for arg in ctx.arg_types:
  5459	                meta_vars = [tv for tv in get_all_type_vars(arg) if tv.id.is_meta_var()]
  5460	                extra_vars.extend([tv for tv in meta_vars if tv not in extra_vars])
  5461	            callable_ctx = ctx.copy_modified(
  5462	                ret_type=replace_meta_vars(ctx.ret_type, ErasedType()),
  5463	                variables=list(ctx.variables) + extra_vars,
  5464	            )
  5465	        else:
  5466	            erased_ctx = replace_meta_vars(ctx, ErasedType())
  5467	            assert isinstance(erased_ctx, ProperType) and isinstance(erased_ctx, CallableType)
  5468	            callable_ctx = erased_ctx
  5469
  5470	        # The callable_ctx may have a fallback of builtins.type if the context
  5471	        # is a constructor -- but this fallback doesn't make sense for lambdas.
  5472	        callable_ctx = callable_ctx.copy_modified(fallback=self.named_type("builtins.function"))
  5473
  5474	        if callable_ctx.type_guard is not None or callable_ctx.type_is is not None:
  5475	            # Lambda's return type cannot be treated as a `TypeGuard`,
  5476	            # because it is implicit. And `TypeGuard`s must be explicit.
  5477	            # See https://github.com/python/mypy/issues/9927
  5478	            return None, None
  5479
  5480	        arg_kinds = [arg.kind for arg in e.arguments]
  5481
  5482	        if callable_ctx.is_ellipsis_args or ctx.param_spec() is not None:
  5483	            # Fill in Any arguments to match the arguments of the lambda.
  5484	            callable_ctx = callable_ctx.copy_modified(
  5485	                is_ellipsis_args=False,
  5486	                arg_types=[AnyType(TypeOfAny.special_form)] * len(arg_kinds),
  5487	                arg_kinds=arg_kinds,
  5488	                arg_names=e.arg_names.copy(),
  5489	            )
  5490
  5491	        if ARG_STAR in arg_kinds or ARG_STAR2 in arg_kinds:
  5492	            # TODO treat this case appropriately
  5493	            return callable_ctx, None
  5494
  5495	        if callable_ctx.arg_kinds != arg_kinds:
  5496	            # Incompatible context; cannot use it to infer types.
  5497	            self.chk.fail(message_registry.CANNOT_INFER_LAMBDA_TYPE, e)
  5498	            return None, None
  5499
  5500	        # Type of lambda must have correct argument names, to prevent false
  5501	        # negatives when lambdas appear in `ParamSpec` context.
  5502	        return callable_ctx.copy_modified(arg_names=e.arg_names), callable_ctx

  5504	    def visit_super_expr(self, e: SuperExpr) -> Type:
  5505	        """Type check a super expression (non-lvalue)."""
  5506
  5507	        # We have an expression like super(T, var).member
  5508
  5509	        # First compute the types of T and var
  5510	        types = self._super_arg_types(e)
  5511	        if isinstance(types, tuple):
  5512	            type_type, instance_type = types
  5513	        else:
  5514	            return types
  5515
  5516	        # Now get the MRO
  5517	        type_info = type_info_from_type(type_type)
  5518	        if type_info is None:
  5519	            self.chk.fail(message_registry.UNSUPPORTED_ARG_1_FOR_SUPER, e)
  5520	            return AnyType(TypeOfAny.from_error)
  5521
  5522	        instance_info = type_info_from_type(instance_type)

  5524	            self.chk.fail(message_registry.UNSUPPORTED_ARG_2_FOR_SUPER, e)
  5525	            return AnyType(TypeOfAny.from_error)
  5526
  5527	        mro = instance_info.mro
  5528
  5529	        # The base is the first MRO entry *after* type_info that has a member
  5530	        # with the right name
  5531	        index = None
  5532	        if type_info in mro:
  5533	            index = mro.index(type_info)
  5534	        else:
  5535	            method = self.chk.scope.current_function()
  5536	            # Mypy explicitly allows supertype upper bounds (and no upper bound at all)
  5537	            # for annotating self-types. However, if such an annotation is used for
  5538	            # checking super() we will still get an error. So to be consistent, we also
  5539	            # allow such imprecise annotations for use with super(), where we fall back
  5540	            # to the current class MRO instead. This works only from inside a method.
  5541	            if method is not None and is_self_type_like(
  5542	                instance_type, is_classmethod=method.is_class
  5543	            ):

  5545	                    mro = e.info.mro
  5546	                    index = mro.index(type_info)

  5549	                instance_info.is_protocol
  5550	                and instance_info != type_info

  5552	            ):

  5554	                # directly to the host protocol, this is not safe, since the real MRO

  5556	                index = -1

  5563	            return AnyType(TypeOfAny.from_error)

  5568	                    # There's an undefined base class, and we're at the end of the
  5569	                    # chain.  That's not an error.
  5570	                    return AnyType(TypeOfAny.special_form)

  5572	                return analyze_member_access(
  5573	                    name=e.name,
  5574	                    typ=instance_type,
  5575	                    is_lvalue=False,
  5576	                    is_super=True,
  5577	                    is_operator=False,
  5578	                    original_type=instance_type,
  5579	                    override_info=base,
  5580	                    context=e,
  5581	                    chk=self.chk,
  5582	                    in_literal_context=self.is_literal_context(),

  5585	        assert False, "unreachable"

  5587	    def _super_arg_types(self, e: SuperExpr) -> Type | tuple[Type, Type]:
  5588	        """
  5589	        Computes the types of the type and instance expressions in super(T, instance), or the
  5590	        implicit ones for zero-argument super() expressions.  Returns a single type for the whole
  5591	        super expression when possible (for errors, anys), otherwise the pair of computed types.
  5592	        """
  5593
  5594	        if not self.chk.in_checked_function():
  5595	            return AnyType(TypeOfAny.unannotated)
  5596	        elif len(e.call.args) == 0:
  5597	            if not e.info:
  5598	                # This has already been reported by the semantic analyzer.
  5599	                return AnyType(TypeOfAny.from_error)
  5600	            elif self.chk.scope.active_class():
  5601	                self.chk.fail(message_registry.SUPER_OUTSIDE_OF_METHOD_NOT_SUPPORTED, e)
  5602	                return AnyType(TypeOfAny.from_error)
  5603
  5604	            # Zero-argument super() is like super(<current class>, <self>)
  5605	            current_type = fill_typevars(e.info)
  5606	            type_type: ProperType = TypeType(current_type)
  5607
  5608	            # Use the type of the self argument, in case it was annotated
  5609	            method = self.chk.scope.current_function()
  5610	            assert method is not None
  5611	            if method.arguments:
  5612	                instance_type: Type = method.arguments[0].variable.type or current_type
  5613	            else:
  5614	                self.chk.fail(message_registry.SUPER_ENCLOSING_POSITIONAL_ARGS_REQUIRED, e)
  5615	                return AnyType(TypeOfAny.from_error)
  5616	        elif ARG_STAR in e.call.arg_kinds:
  5617	            self.chk.fail(message_registry.SUPER_VARARGS_NOT_SUPPORTED, e)
  5618	            return AnyType(TypeOfAny.from_error)
  5619	        elif set(e.call.arg_kinds) != {ARG_POS}:
  5620	            self.chk.fail(message_registry.SUPER_POSITIONAL_ARGS_REQUIRED, e)
  5621	            return AnyType(TypeOfAny.from_error)
  5622	        elif len(e.call.args) == 1:
  5623	            self.chk.fail(message_registry.SUPER_WITH_SINGLE_ARG_NOT_SUPPORTED, e)
  5624	            return AnyType(TypeOfAny.from_error)
  5625	        elif len(e.call.args) == 2:
  5626	            type_type = get_proper_type(self.accept(e.call.args[0]))
  5627	            instance_type = self.accept(e.call.args[1])
  5628	        else:
  5629	            self.chk.fail(message_registry.TOO_MANY_ARGS_FOR_SUPER, e)
  5630	            return AnyType(TypeOfAny.from_error)
  5631
  5632	        # Imprecisely assume that the type is the current class
  5633	        if isinstance(type_type, AnyType):
  5634	            if e.info:
  5635	                type_type = TypeType(fill_typevars(e.info))
  5636	            else:
  5637	                return AnyType(TypeOfAny.from_another_any, source_any=type_type)
  5638	        elif isinstance(type_type, TypeType):
  5639	            type_item = type_type.item
  5640	            if isinstance(type_item, AnyType):
  5641	                if e.info:
  5642	                    type_type = TypeType(fill_typevars(e.info))
  5643	                else:
  5644	                    return AnyType(TypeOfAny.from_another_any, source_any=type_item)
  5645
  5646	        if not isinstance(type_type, TypeType) and not (
  5647	            isinstance(type_type, FunctionLike) and type_type.is_type_obj()
  5648	        ):
  5649	            self.msg.first_argument_for_super_must_be_type(type_type, e)
  5650	            return AnyType(TypeOfAny.from_error)
  5651
  5652	        # Imprecisely assume that the instance is of the current class
  5653	        instance_type = get_proper_type(instance_type)
  5654	        if isinstance(instance_type, AnyType):
  5655	            if e.info:
  5656	                instance_type = fill_typevars(e.info)
  5657	            else:
  5658	                return AnyType(TypeOfAny.from_another_any, source_any=instance_type)
  5659	        elif isinstance(instance_type, TypeType):
  5660	            instance_item = instance_type.item
  5661	            if isinstance(instance_item, AnyType):
  5662	                if e.info:
  5663	                    instance_type = TypeType(fill_typevars(e.info))
  5664	                else:
  5665	                    return AnyType(TypeOfAny.from_another_any, source_any=instance_item)
  5666
  5667	        return type_type, instance_type

  5822	    def visit_conditional_expr(self, e: ConditionalExpr, allow_none_return: bool = False) -> Type:
  5823	        self.accept(e.cond)
  5824	        ctx = self.type_context[-1]
  5825
  5826	        # Gain type information from isinstance if it is there
  5827	        # but only for the current expression
  5828	        if_map, else_map = self.chk.find_isinstance_check(e.cond)
  5829	        if codes.REDUNDANT_EXPR in self.chk.options.enabled_error_codes:
  5830	            if if_map is None:
  5831	                self.msg.redundant_condition_in_if(False, e.cond)
  5832	            elif else_map is None:
  5833	                self.msg.redundant_condition_in_if(True, e.cond)
  5834
  5835	        if_type = self.analyze_cond_branch(
  5836	            if_map, e.if_expr, context=ctx, allow_none_return=allow_none_return
  5837	        )
  5838
  5839	        # we want to keep the narrowest value of if_type for union'ing the branches
  5840	        # however, it would be silly to pass a literal as a type context. Pass the
  5841	        # underlying fallback type instead.
  5842	        if_type_fallback = simple_literal_type(get_proper_type(if_type)) or if_type
  5843
  5844	        # Analyze the right branch using full type context and store the type
  5845	        full_context_else_type = self.analyze_cond_branch(
  5846	            else_map, e.else_expr, context=ctx, allow_none_return=allow_none_return
  5847	        )
  5848
  5849	        if not mypy.checker.is_valid_inferred_type(if_type, self.chk.options):
  5850	            # Analyze the right branch disregarding the left branch.
  5851	            else_type = full_context_else_type
  5852	            # we want to keep the narrowest value of else_type for union'ing the branches
  5853	            # however, it would be silly to pass a literal as a type context. Pass the
  5854	            # underlying fallback type instead.
  5855	            else_type_fallback = simple_literal_type(get_proper_type(else_type)) or else_type
  5856
  5857	            # If it would make a difference, re-analyze the left
  5858	            # branch using the right branch's type as context.
  5859	            if ctx is None or not is_equivalent(else_type_fallback, ctx):
  5860	                # TODO: If it's possible that the previous analysis of
  5861	                # the left branch produced errors that are avoided
  5862	                # using this context, suppress those errors.
  5863	                if_type = self.analyze_cond_branch(
  5864	                    if_map,
  5865	                    e.if_expr,
  5866	                    context=else_type_fallback,
  5867	                    allow_none_return=allow_none_return,
  5868	                )
  5869
  5870	        elif if_type_fallback == ctx:
  5871	            # There is no point re-running the analysis if if_type is equal to ctx.
  5872	            # That would  be an exact duplicate of the work we just did.
  5873	            # This optimization is particularly important to avoid exponential blowup with nested
  5874	            # if/else expressions: https://github.com/python/mypy/issues/9591
  5875	            # TODO: would checking for is_proper_subtype also work and cover more cases?
  5876	            else_type = full_context_else_type
  5877	        else:
  5878	            # Analyze the right branch in the context of the left
  5879	            # branch's type.
  5880	            else_type = self.analyze_cond_branch(
  5881	                else_map,
  5882	                e.else_expr,
  5883	                context=if_type_fallback,
  5884	                allow_none_return=allow_none_return,
  5885	            )
  5886
  5887	        # In most cases using if_type as a context for right branch gives better inferred types.
  5888	        # This is however not the case for literal types, so use the full context instead.
  5889	        if is_literal_type_like(full_context_else_type) and not is_literal_type_like(else_type):
  5890	            else_type = full_context_else_type
  5891
  5892	        res: Type = make_simplified_union([if_type, else_type])
  5893	        if has_uninhabited_component(res) and not isinstance(
  5894	            get_proper_type(self.type_context[-1]), UnionType
  5895	        ):
  5896	            # In rare cases with empty collections join may give a better result.
  5897	            alternative = join.join_types(if_type, else_type)
  5898	            p_alt = get_proper_type(alternative)
  5899	            if not isinstance(p_alt, Instance) or p_alt.type.fullname != "builtins.object":
  5900	                res = alternative
  5901	        return res

  5904	        self,
  5905	        map: dict[Expression, Type] | None,
  5906	        node: Expression,
  5907	        context: Type | None,

  5926	    def accept(
  5927	        self,
  5928	        node: Expression,
  5929	        type_context: Type | None = None,
  5930	        allow_none_return: bool = False,
  5931	        always_allow_any: bool = False,
  5932	        is_callee: bool = False,
  5933	    ) -> Type:
  5934	        """Type check a node in the given type context.  If allow_none_return
  5935	        is True and this expression is a call, allow it to return None.  This
  5936	        applies only to this expression and not any subexpressions.
  5937	        """
  5938	        if node in self.type_overrides:
  5939	            # This branch is very fast, there is no point timing it.
  5940	            return self.type_overrides[node]
  5941	        # We don't use context manager here to get most precise data (and avoid overhead).
  5942	        record_time = False
  5943	        if self.collect_line_checking_stats and not self.in_expression:
  5944	            t0 = time.perf_counter_ns()
  5945	            self.in_expression = True
  5946	            record_time = True
  5947	        self.type_context.append(type_context)
  5948	        old_is_callee = self.is_callee
  5949	        self.is_callee = is_callee
  5950	        try:
  5951	            if allow_none_return and isinstance(node, CallExpr):
  5952	                typ = self.visit_call_expr(node, allow_none_return=True)
  5953	            elif allow_none_return and isinstance(node, YieldFromExpr):
  5954	                typ = self.visit_yield_from_expr(node, allow_none_return=True)
  5955	            elif allow_none_return and isinstance(node, ConditionalExpr):
  5956	                typ = self.visit_conditional_expr(node, allow_none_return=True)
  5957	            elif allow_none_return and isinstance(node, AwaitExpr):
  5958	                typ = self.visit_await_expr(node, allow_none_return=True)
  5959	            else:
  5960	                typ = node.accept(self)
  5961	        except Exception as err:
  5962	            report_internal_error(
  5963	                err, self.chk.errors.file, node.line, self.chk.errors, self.chk.options
  5964	            )
  5965	        self.is_callee = old_is_callee
  5966	        self.type_context.pop()
  5967	        assert typ is not None
  5968	        self.chk.store_type(node, typ)
  5969
  5970	        if (
  5971	            self.chk.options.disallow_any_expr
  5972	            and not always_allow_any
  5973	            and not self.chk.is_stub
  5974	            and self.chk.in_checked_function()
  5975	            and has_any_type(typ)
  5976	            and not self.chk.current_node_deferred
  5977	        ):
  5978	            self.msg.disallowed_any_type(typ, node)
  5979
  5980	        if not self.chk.in_checked_function() or self.chk.current_node_deferred:
  5981	            result: Type = AnyType(TypeOfAny.unannotated)
  5982	        else:
  5983	            result = typ
  5984	        if record_time:
  5985	            self.per_line_checking_time_ns[node.line] += time.perf_counter_ns() - t0
  5986	            self.in_expression = False
  5987	        return result

  6421	def arg_approximate_similarity(actual: Type, formal: Type) -> bool:
  6422	    """Return if caller argument (actual) is roughly compatible with signature arg (formal).
  6423
  6424	    This function is deliberately loose and will report two types are similar
  6425	    as long as their "shapes" are plausibly the same.
  6426
  6427	    This is useful when we're doing error reporting: for example, if we're trying
  6428	    to select an overload alternative and there's no exact match, we can use
  6429	    this function to help us identify which alternative the user might have
  6430	    *meant* to match.
  6431	    """
  6432	    actual = get_proper_type(actual)
  6433	    formal = get_proper_type(formal)
  6434
  6435	    # Erase typevars: we'll consider them all to have the same "shape".
  6436	    if isinstance(actual, TypeVarType):
  6437	        actual = erase_to_union_or_bound(actual)
  6438	    if isinstance(formal, TypeVarType):
  6439	        formal = erase_to_union_or_bound(formal)
  6440
  6441	    # Callable or Type[...]-ish types
  6442	    def is_typetype_like(typ: ProperType) -> bool:
  6443	        return (
  6444	            isinstance(typ, TypeType)
  6445	            or (isinstance(typ, FunctionLike) and typ.is_type_obj())
  6446	            or (isinstance(typ, Instance) and typ.type.fullname == "builtins.type")
  6447	        )
  6448
  6449	    if isinstance(formal, CallableType):
  6450	        if isinstance(actual, (CallableType, Overloaded, TypeType)):
  6451	            return True
  6452	    if is_typetype_like(actual) and is_typetype_like(formal):
  6453	        return True
  6454
  6455	    # Unions
  6456	    if isinstance(actual, UnionType):

   341	            validate_super_call(method, mx)
   342
   343	        if method.is_property:
   344	            assert isinstance(method, OverloadedFuncDef)
   345	            getter = method.items[0]
   346	            assert isinstance(getter, Decorator)
   347	            if mx.is_lvalue and getter.var.is_settable_property:
   348	                mx.chk.warn_deprecated(method.setter, mx.context)
   349	            return analyze_var(name, getter.var, typ, mx)
   350
   351	        if mx.is_lvalue and not mx.suppress_errors:
   352	            mx.msg.cant_assign_to_method(mx.context)
   353	        if not isinstance(method, OverloadedFuncDef):
   354	            signature = function_type(method, mx.named_type("builtins.function"))
   355	        else:
   356	            if method.type is None:
   357	                # Overloads may be not ready if they are decorated. Handle this in same
   358	                # manner as we would handle a regular decorated function: defer if possible.
   359	                if not mx.no_deferral and method.items:
   360	                    mx.not_ready_callback(method.name, mx.context)
   361	                return AnyType(TypeOfAny.special_form)
   362	            assert isinstance(method.type, Overloaded)
   363	            signature = method.type
   364	        if not mx.preserve_type_var_ids:
   365	            signature = freshen_all_functions_type_vars(signature)
   366	        if not method.is_static:
   367	            if isinstance(method, (FuncDef, OverloadedFuncDef)) and method.is_trivial_self:
   368	                signature = bind_self_fast(signature, mx.self_type)
   369	            else:
   370	                signature = check_self_arg(
   371	                    signature, mx.self_type, method.is_class, mx.context, name, mx.msg
   372	                )
   373	                signature = bind_self(signature, mx.self_type, is_classmethod=method.is_class)
   374	        # TODO: should we skip these steps for static methods as well?
   375	        # Since generic static methods should not be allowed.
   376	        typ = map_instance_to_supertype(typ, method.info)
   377	        member_type = expand_type_by_instance(signature, typ)
   378	        freeze_all_type_vars(member_type)
   379	        return member_type
   380	    else:
   381	        # Not a method.
   382	        return analyze_member_var_access(name, typ, info, mx)

   385	def validate_super_call(node: FuncBase, mx: MemberContext) -> None:
   386	    unsafe_super = False
   387	    if isinstance(node, FuncDef) and node.is_trivial_body:
   388	        unsafe_super = True
   389	    elif isinstance(node, OverloadedFuncDef):
   390	        if node.impl:
   391	            impl = node.impl if isinstance(node.impl, FuncDef) else node.impl.func
   392	            unsafe_super = impl.is_trivial_body
   393	        elif not node.is_property and node.items:
   394	            assert isinstance(node.items[0], Decorator)
   395	            unsafe_super = node.items[0].func.is_trivial_body
   396	    if unsafe_super:
   397	        mx.msg.unsafe_super(node.name, node.info.name, mx.context)

   400	def analyze_type_callable_member_access(name: str, typ: FunctionLike, mx: MemberContext) -> Type:
   401	    # Class attribute.
   402	    # TODO super?
   403	    ret_type = typ.items[0].ret_type
   404	    assert isinstance(ret_type, ProperType)
   405	    if isinstance(ret_type, TupleType):
   406	        ret_type = tuple_fallback(ret_type)
   407	    if isinstance(ret_type, TypedDictType):
   408	        ret_type = ret_type.fallback
   409	    if isinstance(ret_type, Instance):
   410	        if not mx.is_operator:
   411	            # When Python sees an operator (eg `3 == 4`), it automatically translates that
   412	            # into something like `int.__eq__(3, 4)` instead of `(3).__eq__(4)` as an
   413	            # optimization.
   414	            #
   415	            # While it normally it doesn't matter which of the two versions are used, it
   416	            # does cause inconsistencies when working with classes. For example, translating
   417	            # `int == int` to `int.__eq__(int)` would not work since `int.__eq__` is meant to
   418	            # compare two int _instances_. What we really want is `type(int).__eq__`, which
   419	            # is meant to compare two types or classes.
   420	            #
   421	            # This check makes sure that when we encounter an operator, we skip looking up
   422	            # the corresponding method in the current instance to avoid this edge case.
   423	            # See https://github.com/python/mypy/pull/1787 for more info.
   424	            # TODO: do not rely on same type variables being present in all constructor overloads.
   425	            result = analyze_class_attribute_access(
   426	                ret_type, name, mx, original_vars=typ.items[0].variables, mcs_fallback=typ.fallback
   427	            )
   428	            if result:
   429	                return result
   430	        # Look up from the 'type' type.
   431	        return _analyze_member_access(name, typ.fallback, mx)
   432	    else:
   433	        assert False, f"Unexpected type {ret_type!r}"
