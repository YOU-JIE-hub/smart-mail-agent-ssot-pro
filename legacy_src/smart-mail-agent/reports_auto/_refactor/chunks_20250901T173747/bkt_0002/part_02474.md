   552	        rootpath: pathlib.Path,
   553	        confcutdir: pathlib.Path | None,
   554	        invocation_dir: pathlib.Path,
   555	        importmode: ImportMode | str,
   556	        *,
   557	        consider_namespace_packages: bool,
   558	    ) -> None:
   559	        """Load initial conftest files given a preparsed "namespace".
   560
   561	        As conftest files may add their own command line options which have
   562	        arguments ('--my-opt somepath') we might get some false positives.
   563	        All builtin and 3rd party plugins will have been loaded, however, so
   564	        common options will not confuse our logic here.
   565	        """
   566	        self._confcutdir = (
   567	            absolutepath(invocation_dir / confcutdir) if confcutdir else None
   568	        )
   569	        self._noconftest = noconftest
   570	        self._using_pyargs = pyargs
   571	        foundanchor = False
   572	        for initial_path in args:
   573	            path = str(initial_path)
   574	            # remove node-id syntax
   575	            i = path.find("::")
   576	            if i != -1:
   577	                path = path[:i]
   578	            anchor = absolutepath(invocation_dir / path)
   579
   580	            # Ensure we do not break if what appears to be an anchor
   581	            # is in fact a very long option (#10169, #11394).
   582	            if safe_exists(anchor):
   583	                self._try_load_conftest(
   584	                    anchor,
   585	                    importmode,
   586	                    rootpath,
   587	                    consider_namespace_packages=consider_namespace_packages,
   588	                )
   589	                foundanchor = True
   590	        if not foundanchor:
   591	            self._try_load_conftest(
   592	                invocation_dir,
   593	                importmode,
   594	                rootpath,
   595	                consider_namespace_packages=consider_namespace_packages,
   596	            )
   597
   598	    def _is_in_confcutdir(self, path: pathlib.Path) -> bool:
   599	        """Whether to consider the given path to load conftests from."""
   600	        if self._confcutdir is None:
   601	            return True
   602	        # The semantics here are literally:
   603	        #   Do not load a conftest if it is found upwards from confcut dir.
   604	        # But this is *not* the same as:
   605	        #   Load only conftests from confcutdir or below.
   606	        # At first glance they might seem the same thing, however we do support use cases where
   607	        # we want to load conftests that are not found in confcutdir or below, but are found
   608	        # in completely different directory hierarchies like packages installed
   609	        # in out-of-source trees.
   610	        # (see #9767 for a regression where the logic was inverted).
   611	        return path not in self._confcutdir.parents

   613	    def _try_load_conftest(
   614	        self,
   615	        anchor: pathlib.Path,
   616	        importmode: str | ImportMode,
   617	        rootpath: pathlib.Path,
   618	        *,
   619	        consider_namespace_packages: bool,
   620	    ) -> None:
   621	        self._loadconftestmodules(
   622	            anchor,
   623	            importmode,
   624	            rootpath,
   625	            consider_namespace_packages=consider_namespace_packages,
   626	        )
   627	        # let's also consider test* subdirs
   628	        if anchor.is_dir():
   629	            for x in anchor.glob("test*"):
   630	                if x.is_dir():
   631	                    self._loadconftestmodules(
   632	                        x,
   633	                        importmode,
   634	                        rootpath,
   635	                        consider_namespace_packages=consider_namespace_packages,
   636	                    )

   638	    def _loadconftestmodules(
   639	        self,
   640	        path: pathlib.Path,
   641	        importmode: str | ImportMode,
   642	        rootpath: pathlib.Path,
   643	        *,
   644	        consider_namespace_packages: bool,
   645	    ) -> None:
   646	        if self._noconftest:
   647	            return
   648
   649	        directory = self._get_directory(path)
   650
   651	        # Optimization: avoid repeated searches in the same directory.
   652	        # Assumes always called with same importmode and rootpath.
   653	        if directory in self._dirpath2confmods:
   654	            return
   655
   656	        clist = []
   657	        for parent in reversed((directory, *directory.parents)):
   658	            if self._is_in_confcutdir(parent):
   659	                conftestpath = parent / "conftest.py"
   660	                if conftestpath.is_file():
   661	                    mod = self._importconftest(
   662	                        conftestpath,
   663	                        importmode,
   664	                        rootpath,
   665	                        consider_namespace_packages=consider_namespace_packages,
   666	                    )
   667	                    clist.append(mod)
   668	        self._dirpath2confmods[directory] = clist
   669
   670	    def _getconftestmodules(self, path: pathlib.Path) -> Sequence[types.ModuleType]:
   671	        directory = self._get_directory(path)
   672	        return self._dirpath2confmods.get(directory, ())

   674	    def _rget_with_confmod(
   675	        self,
   676	        name: str,
   677	        path: pathlib.Path,
   678	    ) -> tuple[types.ModuleType, Any]:
   679	        modules = self._getconftestmodules(path)
   680	        for mod in reversed(modules):
   681	            try:
   682	                return mod, getattr(mod, name)
   683	            except AttributeError:
   684	                continue
   685	        raise KeyError(name)

   687	    def _importconftest(
   688	        self,
   689	        conftestpath: pathlib.Path,
   690	        importmode: str | ImportMode,
   691	        rootpath: pathlib.Path,
   692	        *,
   693	        consider_namespace_packages: bool,
   694	    ) -> types.ModuleType:
   695	        conftestpath_plugin_name = str(conftestpath)
   696	        existing = self.get_plugin(conftestpath_plugin_name)
   697	        if existing is not None:
   698	            return cast(types.ModuleType, existing)
   699
   700	        # conftest.py files there are not in a Python package all have module
   701	        # name "conftest", and thus conflict with each other. Clear the existing
   702	        # before loading the new one, otherwise the existing one will be
   703	        # returned from the module cache.
   704	        pkgpath = resolve_package_path(conftestpath)
   705	        if pkgpath is None:
   706	            try:
   707	                del sys.modules[conftestpath.stem]
   708	            except KeyError:
   709	                pass
   710
   711	        try:
   712	            mod = import_path(
   713	                conftestpath,
   714	                mode=importmode,
   715	                root=rootpath,
   716	                consider_namespace_packages=consider_namespace_packages,
   717	            )
   718	        except Exception as e:
   719	            assert e.__traceback__ is not None
   720	            raise ConftestImportFailure(conftestpath, cause=e) from e
   721
   722	        self._check_non_top_pytest_plugins(mod, conftestpath)
   723
   724	        self._conftest_plugins.add(mod)
   725	        dirpath = conftestpath.parent
   726	        if dirpath in self._dirpath2confmods:
   727	            for path, mods in self._dirpath2confmods.items():
   728	                if dirpath in path.parents or path == dirpath:
   729	                    if mod in mods:
   730	                        raise AssertionError(
   731	                            f"While trying to load conftest path {conftestpath!s}, "
   732	                            f"found that the module {mod} is already loaded with path {mod.__file__}. "
   733	                            "This is not supposed to happen. Please report this issue to pytest."
   734	                        )
   735	                    mods.append(mod)
   736	        self.trace(f"loading conftestmodule {mod!r}")
   737	        self.consider_conftest(mod, registration_name=conftestpath_plugin_name)
   738	        return mod
   739
   740	    def _check_non_top_pytest_plugins(
   741	        self,
   742	        mod: types.ModuleType,
   743	        conftestpath: pathlib.Path,
   744	    ) -> None:
   745	        if (
   746	            hasattr(mod, "pytest_plugins")
   747	            and self._configured
   748	            and not self._using_pyargs
   749	        ):
   750	            msg = (
   751	                "Defining 'pytest_plugins' in a non-top-level conftest is no longer supported:\n"
   752	                "It affects the entire test suite instead of just below the conftest as expected.\n"
   753	                "  {}\n"
   754	                "Please move it to a top level conftest file at the rootdir:\n"
   755	                "  {}\n"
   756	                "For more information, visit:\n"
   757	                "  https://docs.pytest.org/en/stable/deprecations.html#pytest-plugins-in-non-top-level-conftest-files"
   758	            )
   759	            fail(msg.format(conftestpath, self._confcutdir), pytrace=False)

   766	    def consider_preparse(
   767	        self, args: Sequence[str], *, exclude_only: bool = False
   768	    ) -> None:
   769	        """:meta private:"""
   770	        i = 0
   771	        n = len(args)
   772	        while i < n:
   773	            opt = args[i]
   774	            i += 1
   775	            if isinstance(opt, str):
   776	                if opt == "-p":
   777	                    try:
   778	                        parg = args[i]
   779	                    except IndexError:
   780	                        return
   781	                    i += 1
   782	                elif opt.startswith("-p"):
   783	                    parg = opt[2:]
   784	                else:
   785	                    continue
   786	                parg = parg.strip()
   787	                if exclude_only and not parg.startswith("no:"):
   788	                    continue
   789	                self.consider_pluginarg(parg)

   791	    def consider_pluginarg(self, arg: str) -> None:
   792	        """:meta private:"""
   793	        if arg.startswith("no:"):
   794	            name = arg[3:]
   795	            if name in essential_plugins:
   796	                raise UsageError(f"plugin {name} cannot be disabled")
   797
   798	            # PR #4304: remove stepwise if cacheprovider is blocked.
   799	            if name == "cacheprovider":
   800	                self.set_blocked("stepwise")
   801	                self.set_blocked("pytest_stepwise")
   802
   803	            self.set_blocked(name)
   804	            if not name.startswith("pytest_"):
   805	                self.set_blocked("pytest_" + name)
   806	        else:
   807	            name = arg
   808	            # Unblock the plugin.
   809	            self.unblock(name)
   810	            if not name.startswith("pytest_"):
   811	                self.unblock("pytest_" + name)
   812	            self.import_plugin(arg, consider_entry_points=True)
   813

   815	        self, conftestmodule: types.ModuleType, registration_name: str
   816	    ) -> None:

   818	        self.register(conftestmodule, name=registration_name)

   835	    def import_plugin(self, modname: str, consider_entry_points: bool = False) -> None:
   836	        """Import a plugin with ``modname``.
   837
   838	        If ``consider_entry_points`` is True, entry point names are also
   839	        considered to find a plugin.
   840	        """
   841	        # Most often modname refers to builtin modules, e.g. "pytester",
   842	        # "terminal" or "capture".  Those plugins are registered under their
   843	        # basename for historic purposes but must be imported with the
   844	        # _pytest prefix.
   845	        assert isinstance(modname, str), (
   846	            f"module name as text required, got {modname!r}"
   847	        )
   848	        if self.is_blocked(modname) or self.get_plugin(modname) is not None:
   849	            return
   850
   851	        importspec = "_pytest." + modname if modname in builtin_plugins else modname
   852	        self.rewrite_hook.mark_rewrite(importspec)
   853
   854	        if consider_entry_points:
   855	            loaded = self.load_setuptools_entrypoints("pytest11", name=modname)
   856	            if loaded:
   857	                return
   858
   859	        try:
   860	            __import__(importspec)
   861	        except ImportError as e:
   862	            raise ImportError(
   863	                f'Error importing plugin "{modname}": {e.args[0]}'
   864	            ).with_traceback(e.__traceback__) from e
   865
   866	        except Skipped as e:
   867	            self.skipped_plugins.append((modname, e.msg or ""))
   868	        else:
   869	            mod = sys.modules[importspec]
   870	            self.register(mod, modname)

   873	def _get_plugin_specs_as_list(
   874	    specs: None | types.ModuleType | str | Sequence[str],
   875	) -> list[str]:
   876	    """Parse a plugins specification into a list of plugin names."""
   877	    # None means empty.
   878	    if specs is None:
   879	        return []
   880	    # Workaround for #3899 - a submodule which happens to be called "pytest_plugins".
   881	    if isinstance(specs, types.ModuleType):
   882	        return []
   883	    # Comma-separated list.
   884	    if isinstance(specs, str):
   885	        return specs.split(",") if specs else []
   886	    # Direct specification.
   887	    if isinstance(specs, collections.abc.Sequence):
   888	        return list(specs)
   889	    raise UsageError(
   890	        f"Plugins may be specified as a sequence or a ','-separated string of plugin names. Got: {specs!r}"
   891	    )

   902	def _iter_rewritable_modules(package_files: Iterable[str]) -> Iterator[str]:
   903	    """Given an iterable of file names in a source distribution, return the "names" that should
   904	    be marked for assertion rewrite.
   905
   906	    For example the package "pytest_mock/__init__.py" should be added as "pytest_mock" in
   907	    the assertion rewrite mechanism.
   908
   909	    This function has to deal with dist-info based distributions and egg based distributions
   910	    (which are still very much in use for "editable" installs).
   911
   912	    Here are the file names as seen in a dist-info based distribution:
   913
   914	        pytest_mock/__init__.py
   915	        pytest_mock/_version.py
   916	        pytest_mock/plugin.py
   917	        pytest_mock.egg-info/PKG-INFO
   918
   919	    Here are the file names as seen in an egg based distribution:
   920
   921	        src/pytest_mock/__init__.py
   922	        src/pytest_mock/_version.py
   923	        src/pytest_mock/plugin.py
   924	        src/pytest_mock.egg-info/PKG-INFO
   925	        LICENSE
   926	        setup.py
   927
   928	    We have to take in account those two distribution flavors in order to determine which
   929	    names should be considered for assertion rewriting.
   930
   931	    More information:
   932	        https://github.com/pytest-dev/pytest-mock/issues/167
   933	    """
   934	    package_files = list(package_files)
   935	    seen_some = False
   936	    for fn in package_files:
   937	        is_simple_module = "/" not in fn and fn.endswith(".py")
   938	        is_package = fn.count("/") == 1 and fn.endswith("__init__.py")
   939	        if is_simple_module:
   940	            module_name, _ = os.path.splitext(fn)
   941	            # we ignore "setup.py" at the root of the distribution
   942	            # as well as editable installation finder modules made by setuptools
   943	            if module_name != "setup" and not module_name.startswith("__editable__"):
   944	                seen_some = True
   945	                yield module_name
   946	        elif is_package:
   947	            package_name = os.path.dirname(fn)
   948	            seen_some = True
   949	            yield package_name
   950
   951	    if not seen_some:
   952	        # At this point we did not find any packages or modules suitable for assertion
   953	        # rewriting, so we try again by stripping the first path component (to account for
   954	        # "src" based source trees for example).
   955	        # This approach lets us have the common case continue to be fast, as egg-distributions
   956	        # are rarer.
   957	        new_package_files = []
   958	        for fn in package_files:
   959	            parts = fn.split("/")
   960	            new_fn = "/".join(parts[1:])
   961	            if new_fn:
   962	                new_package_files.append(new_fn)
   963	        if new_package_files:
   964	            yield from _iter_rewritable_modules(new_package_files)

  1546

  1551	        ``pytest_configure`` (or similar stages).

  1631	    def _getini(self, name: str):
  1632	        try:
  1633	            description, type, default = self._parser._inidict[name]
  1634	        except KeyError as e:

  1636	        override_value = self._get_override_ini_value(name)
  1637	        if override_value is None:

  1639	                value = self.inicfg[name]

  1642	        else:
  1643	            value = override_value
  1644	        # Coerce the values based on types.
  1645	        #
  1646	        # Note: some coercions are only required if we are reading from .ini files, because
  1647	        # the file format doesn't contain type information, but when reading from toml we will
  1648	        # get either str or list of str values (see _parse_ini_config_from_pyproject_toml).
  1649	        # For example:
  1650	        #
  1651	        #   ini:
  1652	        #     a_line_list = "tests acceptance"

  1654	        #
  1655	        #   toml:
  1656	        #     a_line_list = ["tests", "acceptance"]

   275	class Argument:
   276	    """Class that mimics the necessary behaviour of optparse.Option.
   277
   278	    It's currently a least effort implementation and ignoring choices
   279	    and integer prefixes.
   280
   281	    https://docs.python.org/3/library/optparse.html#optparse-standard-option-types
   282	    """
   283
   284	    def __init__(self, *names: str, **attrs: Any) -> None:
   285	        """Store params in private vars for use in add_argument."""
   286	        self._attrs = attrs
   287	        self._short_opts: list[str] = []
   288	        self._long_opts: list[str] = []
   289	        try:
   290	            self.type = attrs["type"]
   291	        except KeyError:
   292	            pass
   293	        try:
   294	            # Attribute existence is tested in Config._processopt.
   295	            self.default = attrs["default"]
   296	        except KeyError:
   297	            pass
   298	        self._set_opt_strings(names)
   299	        dest: str | None = attrs.get("dest")
   300	        if dest:
   301	            self.dest = dest
   302	        elif self._long_opts:
   303	            self.dest = self._long_opts[0][2:].replace("-", "_")
   304	        else:
   305	            try:
   306	                self.dest = self._short_opts[0][1:]
   307	            except IndexError as e:
   308	                self.dest = "???"  # Needed for the error repr.
   309	                raise ArgumentError("need a long or short option", self) from e
   310
   311	    def names(self) -> list[str]:
   312	        return self._short_opts + self._long_opts
   313
   314	    def attrs(self) -> Mapping[str, Any]:
   315	        # Update any attributes set by processopt.
   316	        attrs = "default dest help".split()
   317	        attrs.append(self.dest)
   318	        for attr in attrs:
   319	            try:
   320	                self._attrs[attr] = getattr(self, attr)
   321	            except AttributeError:
   322	                pass
   323	        return self._attrs

   325	    def _set_opt_strings(self, opts: Sequence[str]) -> None:
   326	        """Directly from optparse.
   327
   328	        Might not be necessary as this is passed to argparse later on.
   329	        """
   330	        for opt in opts:
   331	            if len(opt) < 2:
   332	                raise ArgumentError(
   333	                    f"invalid option string {opt!r}: "
   334	                    "must be at least two characters long",
   335	                    self,
   336	                )
   337	            elif len(opt) == 2:
   338	                if not (opt[0] == "-" and opt[1] != "-"):
   339	                    raise ArgumentError(
   340	                        f"invalid short option string {opt!r}: "
   341	                        "must be of the form -x, (x any non-dash char)",
   342	                        self,
   343	                    )
   344	                self._short_opts.append(opt)
   345	            else:
   346	                if not (opt[0:2] == "--" and opt[2] != "-"):
   347	                    raise ArgumentError(
   348	                        f"invalid long option string {opt!r}: "
   349	                        "must start with --, followed by non-dash",
   350	                        self,
   351	                    )
   352	                self._long_opts.append(opt)

   471	class DropShorterLongHelpFormatter(argparse.HelpFormatter):
   472	    """Shorten help for long options that differ only in extra hyphens.
   473
   474	    - Collapse **long** options that are the same except for extra hyphens.
   475	    - Shortcut if there are only two options and one of them is a short one.
   476	    - Cache result on the action object as this is called at least 2 times.
   477	    """
   478
   479	    def __init__(self, *args: Any, **kwargs: Any) -> None:
   480	        # Use more accurate terminal width.
   481	        if "width" not in kwargs:
   482	            kwargs["width"] = _pytest._io.get_terminal_width()
   483	        super().__init__(*args, **kwargs)
   484
   485	    def _format_action_invocation(self, action: argparse.Action) -> str:
   486	        orgstr = super()._format_action_invocation(action)
   487	        if orgstr and orgstr[0] != "-":  # only optional arguments
   488	            return orgstr
   489	        res: str | None = getattr(action, "_formatted_action_invocation", None)
   490	        if res:
   491	            return res
   492	        options = orgstr.split(", ")
   493	        if len(options) == 2 and (len(options[0]) == 2 or len(options[1]) == 2):
   494	            # a shortcut for '-h, --help' or '--abc', '-a'
   495	            action._formatted_action_invocation = orgstr  # type: ignore
   496	            return orgstr
   497	        return_list = []
   498	        short_long: dict[str, str] = {}
   499	        for option in options:
   500	            if len(option) == 2 or option[2] == " ":
   501	                continue
   502	            if not option.startswith("--"):
   503	                raise ArgumentError(
   504	                    f'long optional argument without "--": [{option}]', option
   505	                )
   506	            xxoption = option[2:]
   507	            shortened = xxoption.replace("-", "")
   508	            if shortened not in short_long or len(short_long[shortened]) < len(
   509	                xxoption
   510	            ):
   511	                short_long[shortened] = xxoption
   512	        # now short_long has been filled out to the longest with dashes
   513	        # **and** we keep the right option ordering from add_argument
   514	        for option in options:
   515	            if len(option) == 2 or option[2] == " ":
   516	                return_list.append(option)
   517	            if option[2:] == short_long.get(option.replace("-", "")):
   518	                return_list.append(option.replace(" ", "=", 1))
   519	        formatted_action_invocation = ", ".join(return_list)
   520	        action._formatted_action_invocation = formatted_action_invocation  # type: ignore
   521	        return formatted_action_invocation
   522
   523	    def _split_lines(self, text, width):
   524	        """Wrap lines after splitting on original newlines.
   525
   526	        This allows to have explicit line breaks in the help text.
   527	        """
   528	        import textwrap
   529
   530	        lines = []
   531	        for line in text.splitlines():
   532	            lines.extend(textwrap.wrap(line.strip(), width))
   533	        return lines

   180	def determine_setup(
   181	    *,
   182	    inifile: str | None,
   183	    args: Sequence[str],
   184	    rootdir_cmd_arg: str | None,
   185	    invocation_dir: Path,
   186	) -> tuple[Path, Path | None, ConfigDict]:
   187	    """Determine the rootdir, inifile and ini configuration values from the
   188	    command line arguments.
   189
   190	    :param inifile:
   191	        The `--inifile` command line argument, if given.
   192	    :param args:
   193	        The free command line arguments.
   194	    :param rootdir_cmd_arg:
   195	        The `--rootdir` command line argument, if given.
   196	    :param invocation_dir:
   197	        The working directory when pytest was invoked.
   198	    """
   199	    rootdir = None
   200	    dirs = get_dirs_from_args(args)
   201	    if inifile:
   202	        inipath_ = absolutepath(inifile)
   203	        inipath: Path | None = inipath_
   204	        inicfg = load_config_dict_from_file(inipath_) or {}
   205	        if rootdir_cmd_arg is None:
   206	            rootdir = inipath_.parent
   207	    else:
   208	        ancestor = get_common_ancestor(invocation_dir, dirs)
   209	        rootdir, inipath, inicfg = locate_config(invocation_dir, [ancestor])
   210	        if rootdir is None and rootdir_cmd_arg is None:
   211	            for possible_rootdir in (ancestor, *ancestor.parents):
   212	                if (possible_rootdir / "setup.py").is_file():
   213	                    rootdir = possible_rootdir
   214	                    break
   215	            else:
   216	                if dirs != [ancestor]:
   217	                    rootdir, inipath, inicfg = locate_config(invocation_dir, dirs)
   218	                if rootdir is None:
   219	                    rootdir = get_common_ancestor(
   220	                        invocation_dir, [invocation_dir, ancestor]
   221	                    )
   222	                    if is_fs_root(rootdir):
   223	                        rootdir = ancestor
   224	    if rootdir_cmd_arg:
   225	        rootdir = absolutepath(os.path.expandvars(rootdir_cmd_arg))
   226	        if not rootdir.is_dir():
   227	            raise UsageError(
   228	                f"Directory '{rootdir}' not found. Check your '--rootdir' option."
   229	            )
   230	    assert rootdir is not None
   231	    return rootdir, inipath, inicfg or {}
