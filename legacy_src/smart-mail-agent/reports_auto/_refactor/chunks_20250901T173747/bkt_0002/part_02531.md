  3353	def order_ascc(graph: Graph, ascc: AbstractSet[str], pri_max: int = PRI_ALL) -> list[str]:
  3354	    """Come up with the ideal processing order within an SCC.
  3355
  3356	    Using the priorities assigned by all_imported_modules_in_file(),
  3357	    try to reduce the cycle to a DAG, by omitting arcs representing
  3358	    dependencies of lower priority.
  3359
  3360	    In the simplest case, if we have A <--> B where A has a top-level
  3361	    "import B" (medium priority) but B only has the reverse "import A"
  3362	    inside a function (low priority), we turn the cycle into a DAG by
  3363	    dropping the B --> A arc, which leaves only A --> B.
  3364
  3365	    If all arcs have the same priority, we fall back to sorting by
  3366	    reverse global order (the order in which modules were first
  3367	    encountered).
  3368
  3369	    The algorithm is recursive, as follows: when as arcs of different
  3370	    priorities are present, drop all arcs of the lowest priority,
  3371	    identify SCCs in the resulting graph, and apply the algorithm to
  3372	    each SCC thus found.  The recursion is bounded because at each
  3373	    recursion the spread in priorities is (at least) one less.
  3374
  3375	    In practice there are only a few priority levels (less than a
  3376	    dozen) and in the worst case we just carry out the same algorithm
  3377	    for finding SCCs N times.  Thus the complexity is no worse than
  3378	    the complexity of the original SCC-finding algorithm -- see
  3379	    strongly_connected_components() below for a reference.
  3380	    """
  3381	    if len(ascc) == 1:
  3382	        return list(ascc)
  3383	    pri_spread = set()
  3384	    for id in ascc:
  3385	        state = graph[id]
  3386	        for dep in state.dependencies:
  3387	            if dep in ascc:
  3388	                pri = state.priorities.get(dep, PRI_HIGH)
  3389	                if pri < pri_max:
  3390	                    pri_spread.add(pri)
  3391	    if len(pri_spread) == 1:
  3392	        # Filtered dependencies are uniform -- order by global order.
  3393	        return sorted(ascc, key=lambda id: -graph[id].order)
  3394	    pri_max = max(pri_spread)
  3395	    sccs = sorted_components(graph, ascc, pri_max)
  3396	    # The recursion is bounded by the len(pri_spread) check above.
  3397	    return [s for ss in sccs for s in order_ascc(graph, ss, pri_max)]

  3468	def sorted_components(
  3469	    graph: Graph, vertices: AbstractSet[str] | None = None, pri_max: int = PRI_ALL
  3470	) -> list[AbstractSet[str]]:
  3471	    """Return the graph's SCCs, topologically sorted by dependencies.
  3472
  3473	    The sort order is from leaves (nodes without dependencies) to
  3474	    roots (nodes on which no other nodes depend).
  3475
  3476	    This works for a subset of the full dependency graph too;
  3477	    dependencies that aren't present in graph.keys() are ignored.
  3478	    """
  3479	    # Compute SCCs.
  3480	    if vertices is None:
  3481	        vertices = set(graph)
  3482	    edges = {id: deps_filtered(graph, vertices, id, pri_max) for id in vertices}
  3483	    sccs = list(strongly_connected_components(vertices, edges))
  3484	    # Topsort.
  3485	    res = []
  3486	    for ready in topsort(prepare_sccs(sccs, edges)):
  3487	        # Sort the sets in ready by reversed smallest State.order.  Examples:
  3488	        #
  3489	        # - If ready is [{x}, {y}], x.order == 1, y.order == 2, we get
  3490	        #   [{y}, {x}].
  3491	        #
  3492	        # - If ready is [{a, b}, {c, d}], a.order == 1, b.order == 3,
  3493	        #   c.order == 2, d.order == 4, the sort keys become [1, 2]
  3494	        #   and the result is [{c, d}, {a, b}].
  3495	        res.extend(sorted(ready, key=lambda scc: -min(graph[id].order for id in scc)))
  3496	    return res

  3505	        dep
  3506	        for dep in state.dependencies
  3507	        if dep in vertices and state.priorities.get(dep, PRI_HIGH) < pri_max

   537	            self.recurse_into_functions = True
   538	            with self.binder.top_frame_context():
   539	                self.accept(node)

   542	        """Check only the top-level of a module, skipping function definitions."""

   552	    def defer_node(self, node: DeferredNodeType, enclosing_class: TypeInfo | None) -> None:
   553	        """Defer a node for processing during next type-checking pass.
   554
   555	        Args:
   556	            node: function/method being deferred
   557	            enclosing_class: for methods, the class where the method is defined
   558	        NOTE: this can't handle nested functions/methods.
   559	        """
   560	        # We don't freeze the entire scope since only top-level functions and methods
   561	        # can be deferred. Only module/class level scope information is needed.
   562	        # Module-level scope information is preserved in the TypeChecker instance.
   563	        self.deferred_nodes.append(DeferredNode(node, enclosing_class))

   565	    def handle_cannot_determine_type(self, name: str, context: Context) -> None:
   566	        node = self.scope.top_level_function()
   567	        if self.pass_num < self.last_pass and isinstance(node, FuncDef):
   568	            # Don't report an error yet. Just defer. Note that we don't defer
   569	            # lambdas because they are coupled to the surrounding function
   570	            # through the binder and the inferred type of the lambda, so it
   571	            # would get messy.
   572	            enclosing_class = self.scope.enclosing_class(node)
   573	            self.defer_node(node, enclosing_class)
   574	            # Set a marker so that we won't infer additional types in this
   575	            # function. Any inferred types could be bogus, because there's at
   576	            # least one type that we don't know.
   577	            self.current_node_deferred = True
   578	        else:
   579	            self.msg.cannot_determine_type(name, context)

   588	    def accept_loop(
   589	        self,
   590	        body: Statement,
   591	        else_body: Statement | None = None,
   592	        *,
   593	        exit_condition: Expression | None = None,
   594	        on_enter_body: Callable[[], None] | None = None,
   595	    ) -> None:
   596	        """Repeatedly type check a loop body until the frame doesn't change."""
   597
   598	        # The outer frame accumulates the results of all iterations:
   599	        with self.binder.frame_context(can_skip=False, conditional_frame=True):
   600	            # Check for potential decreases in the number of partial types so as not to stop the
   601	            # iteration too early:
   602	            partials_old = sum(len(pts.map) for pts in self.partial_types)
   603	            # Check if assignment widened the inferred type of a variable; in this case we
   604	            # need to iterate again (we only do one extra iteration, since this could go
   605	            # on without bound otherwise)
   606	            widened_old = len(self.widened_vars)
   607
   608	            iter_errors = IterationDependentErrors()
   609	            iter = 1
   610	            while True:
   611	                with self.binder.frame_context(can_skip=True, break_frame=2, continue_frame=1):
   612	                    if on_enter_body is not None:
   613	                        on_enter_body()
   614
   615	                    with IterationErrorWatcher(self.msg.errors, iter_errors):
   616	                        self.accept(body)
   617
   618	                partials_new = sum(len(pts.map) for pts in self.partial_types)
   619	                widened_new = len(self.widened_vars)
   620	                # Perform multiple iterations if something changed that might affect
   621	                # inferred types. Also limit the number of iterations. The limits are
   622	                # somewhat arbitrary, but they were chosen to 1) avoid slowdown from
   623	                # multiple iterations in common cases and 2) support common, valid use
   624	                # cases. Limits are needed since otherwise we could infer infinitely
   625	                # complex types.
   626	                if (
   627	                    (partials_new == partials_old)
   628	                    and (not self.binder.last_pop_changed or iter > 3)
   629	                    and (widened_new == widened_old or iter > 1)
   630	                ):
   631	                    break
   632	                partials_old = partials_new
   633	                widened_old = widened_new
   634	                iter += 1
   635	                if iter == 20:
   636	                    raise RuntimeError("Too many iterations when checking a loop")
   637
   638	            self.msg.iteration_dependent_errors(iter_errors)
   639
   640	            # If exit_condition is set, assume it must be False on exit from the loop:
   641	            if exit_condition:
   642	                _, else_map = self.find_isinstance_check(exit_condition)
   643	                self.push_type_map(else_map)
   644
   645	            # Check the else body:
   646	            if else_body:
   647	                self.accept(else_body)

   659	    def _visit_overloaded_func_def(self, defn: OverloadedFuncDef) -> None:
   660	        num_abstract = 0
   661	        if not defn.items:
   662	            # In this case we have already complained about none of these being
   663	            # valid overloads.
   664	            return
   665	        if len(defn.items) == 1:
   666	            self.fail(message_registry.MULTIPLE_OVERLOADS_REQUIRED, defn)
   667
   668	        if defn.is_property:
   669	            # HACK: Infer the type of the property.
   670	            assert isinstance(defn.items[0], Decorator)
   671	            self.visit_decorator(defn.items[0])
   672	            if defn.items[0].var.is_settable_property:
   673	                # Perform a reduced visit just to infer the actual setter type.
   674	                self.visit_decorator_inner(defn.setter, skip_first_item=True)
   675	                setter_type = defn.setter.var.type
   676	                # Check if the setter can accept two positional arguments.
   677	                any_type = AnyType(TypeOfAny.special_form)
   678	                fallback_setter_type = CallableType(
   679	                    arg_types=[any_type, any_type],
   680	                    arg_kinds=[ARG_POS, ARG_POS],
   681	                    arg_names=[None, None],
   682	                    ret_type=any_type,
   683	                    fallback=self.named_type("builtins.function"),
   684	                )
   685	                if setter_type and not is_subtype(setter_type, fallback_setter_type):
   686	                    self.fail("Invalid property setter signature", defn.setter.func)
   687	                setter_type = self.extract_callable_type(setter_type, defn)
   688	                if not isinstance(setter_type, CallableType) or len(setter_type.arg_types) != 2:
   689	                    # TODO: keep precise type for callables with tricky but valid signatures.
   690	                    setter_type = fallback_setter_type
   691	                defn.items[0].var.setter_type = setter_type
   692	        for i, fdef in enumerate(defn.items):
   693	            assert isinstance(fdef, Decorator)
   694	            if defn.is_property:
   695	                assert isinstance(defn.items[0], Decorator)
   696	                settable = defn.items[0].var.is_settable_property
   697	                # Do not visit the second time the items we checked above.
   698	                if (settable and i > 1) or (not settable and i > 0):
   699	                    self.check_func_item(fdef.func, name=fdef.func.name, allow_empty=True)
   700	            else:
   701	                # Perform full check for real overloads to infer type of all decorated
   702	                # overload variants.
   703	                self.visit_decorator_inner(fdef, allow_empty=True)
   704	            if fdef.func.abstract_status in (IS_ABSTRACT, IMPLICITLY_ABSTRACT):
   705	                num_abstract += 1
   706	        if num_abstract not in (0, len(defn.items)):
   707	            self.fail(message_registry.INCONSISTENT_ABSTRACT_OVERLOAD, defn)
   708	        if defn.impl:
   709	            defn.impl.accept(self)
   710	        if not defn.is_property:
   711	            self.check_overlapping_overloads(defn)
   712	            if defn.type is None:
   713	                item_types = []
   714	                for item in defn.items:
   715	                    assert isinstance(item, Decorator)
   716	                    item_type = self.extract_callable_type(item.var.type, item)
   717	                    if item_type is not None:
   718	                        item_types.append(item_type)
   719	                if item_types:
   720	                    defn.type = Overloaded(item_types)
   721	        elif defn.type is None:
   722	            # We store the getter type as an overall overload type, as some
   723	            # code paths are getting property type this way.
   724	            assert isinstance(defn.items[0], Decorator)
   725	            var_type = self.extract_callable_type(defn.items[0].var.type, defn)
   726	            if not isinstance(var_type, CallableType):
   727	                # Construct a fallback type, invalid types should be already reported.
   728	                any_type = AnyType(TypeOfAny.special_form)
   729	                var_type = CallableType(
   730	                    arg_types=[any_type],
   731	                    arg_kinds=[ARG_POS],
   732	                    arg_names=[None],
   733	                    ret_type=any_type,
   734	                    fallback=self.named_type("builtins.function"),
   735	                )
   736	            defn.type = Overloaded([var_type])
   737	        # Check override validity after we analyzed current definition.
   738	        if defn.info:
   739	            found_method_base_classes = self.check_method_override(defn)
   740	            if (
   741	                defn.is_explicit_override
   742	                and not found_method_base_classes
   743	                and found_method_base_classes is not None
   744	                # If the class has Any fallback, we can't be certain that a method
   745	                # is really missing - it might come from unfollowed import.
   746	                and not defn.info.fallback_to_any
   747	            ):
   748	                self.msg.no_overridable_method(defn.name, defn)
   749	            self.check_explicit_override_decorator(defn, found_method_base_classes, defn.impl)
   750	            self.check_inplace_operator_method(defn)

   752	    def extract_callable_type(self, inner_type: Type | None, ctx: Context) -> CallableType | None:
   753	        """Get type as seen by an overload item caller."""
   754	        inner_type = get_proper_type(inner_type)
   755	        outer_type: FunctionLike | None = None
   756	        if inner_type is None or isinstance(inner_type, AnyType):
   757	            return None
   758	        if isinstance(inner_type, TypeVarLikeType):
   759	            inner_type = get_proper_type(inner_type.upper_bound)
   760	        if isinstance(inner_type, TypeType):
   761	            inner_type = get_proper_type(
   762	                self.expr_checker.analyze_type_type_callee(inner_type.item, ctx)
   763	            )
   764
   765	        if isinstance(inner_type, FunctionLike):
   766	            outer_type = inner_type
   767	        elif isinstance(inner_type, Instance):
   768	            inner_call = get_proper_type(
   769	                analyze_member_access(
   770	                    name="__call__",
   771	                    typ=inner_type,
   772	                    context=ctx,
   773	                    is_lvalue=False,
   774	                    is_super=False,
   775	                    is_operator=True,
   776	                    original_type=inner_type,
   777	                    chk=self,
   778	                )
   779	            )
   780	            if isinstance(inner_call, FunctionLike):
   781	                outer_type = inner_call
   782	        elif isinstance(inner_type, UnionType):
   783	            union_type = make_simplified_union(inner_type.items)
   784	            if isinstance(union_type, UnionType):
   785	                items = []
   786	                for item in union_type.items:
   787	                    callable_item = self.extract_callable_type(item, ctx)
   788	                    if callable_item is None:
   789	                        break
   790	                    items.append(callable_item)
   791	                else:
   792	                    joined_type = get_proper_type(join.join_type_list(items))
   793	                    if isinstance(joined_type, FunctionLike):
   794	                        outer_type = joined_type
   795	            else:
   796	                return self.extract_callable_type(union_type, ctx)
   797
   798	        if outer_type is None:
   799	            self.msg.not_callable(inner_type, ctx)
   800	            return None
   801	        if isinstance(outer_type, Overloaded):
   802	            return None
   803
   804	        assert isinstance(outer_type, CallableType)
   805	        return outer_type

   807	    def check_overlapping_overloads(self, defn: OverloadedFuncDef) -> None:
   808	        # At this point we should have set the impl already, and all remaining
   809	        # items are decorators
   810
   811	        if self.msg.errors.file in self.msg.errors.ignored_files or (
   812	            self.is_typeshed_stub and self.options.test_env
   813	        ):
   814	            # This is a little hacky, however, the quadratic check here is really expensive, this
   815	            # method has no side effects, so we should skip it if we aren't going to report
   816	            # anything. In some other places we swallow errors in stubs, but this error is very
   817	            # useful for stubs!
   818	            return
   819
   820	        # Compute some info about the implementation (if it exists) for use below
   821	        impl_type: CallableType | None = None
   822	        if defn.impl:
   823	            if isinstance(defn.impl, FuncDef):
   824	                inner_type: Type | None = defn.impl.type
   825	            elif isinstance(defn.impl, Decorator):
   826	                inner_type = defn.impl.var.type
   827	            else:
   828	                assert False, "Impl isn't the right type"
   829
   830	            # This can happen if we've got an overload with a different
   831	            # decorator or if the implementation is untyped -- we gave up on the types.
   832	            impl_type = self.extract_callable_type(inner_type, defn.impl)
   833
   834	        is_descriptor_get = defn.info and defn.name == "__get__"
   835	        for i, item in enumerate(defn.items):
   836	            assert isinstance(item, Decorator)
   837	            sig1 = self.extract_callable_type(item.var.type, item)
   838	            if sig1 is None:
   839	                continue
   840
   841	            for j, item2 in enumerate(defn.items[i + 1 :]):
   842	                assert isinstance(item2, Decorator)
   843	                sig2 = self.extract_callable_type(item2.var.type, item2)
   844	                if sig2 is None:
   845	                    continue
   846
   847	                if not are_argument_counts_overlapping(sig1, sig2):
   848	                    continue
   849
   850	                if overload_can_never_match(sig1, sig2):
   851	                    self.msg.overloaded_signature_will_never_match(i + 1, i + j + 2, item2.func)
   852	                elif not is_descriptor_get:
   853	                    # Note: we force mypy to check overload signatures in strict-optional mode
   854	                    # so we don't incorrectly report errors when a user tries typing an overload
   855	                    # that happens to have a 'if the argument is None' fallback.
   856	                    #
   857	                    # For example, the following is fine in strict-optional mode but would throw
   858	                    # the unsafe overlap error when strict-optional is disabled:
   859	                    #
   860	                    #     @overload
   861	                    #     def foo(x: None) -> int: ...
   862	                    #     @overload
   863	                    #     def foo(x: str) -> str: ...
   864	                    #
   865	                    # See Python 2's map function for a concrete example of this kind of overload.
   866	                    current_class = self.scope.active_class()
   867	                    type_vars = current_class.defn.type_vars if current_class else []
   868	                    with state.strict_optional_set(True):
   869	                        if is_unsafe_overlapping_overload_signatures(sig1, sig2, type_vars):
   870	                            flip_note = (
   871	                                j == 0
   872	                                and not is_unsafe_overlapping_overload_signatures(
   873	                                    sig2, sig1, type_vars
   874	                                )
   875	                                and not overload_can_never_match(sig2, sig1)
   876	                            )
   877	                            self.msg.overloaded_signatures_overlap(
   878	                                i + 1, i + j + 2, flip_note, item.func
   879	                            )
   880
   881	            if impl_type is not None:
   882	                assert defn.impl is not None
   883
   884	                # This is what we want from implementation, it should accept all arguments
   885	                # of an overload, but the return types should go the opposite way.
   886	                if is_callable_compatible(
   887	                    impl_type,
   888	                    sig1,
   889	                    is_compat=is_subtype,
   890	                    is_proper_subtype=False,
   891	                    is_compat_return=lambda l, r: is_subtype(r, l),
   892	                ):
   893	                    continue
   894	                # If the above check didn't work, we repeat some key steps in
   895	                # is_callable_compatible() to give a better error message.
   896
   897	                # We perform a unification step that's very similar to what
   898	                # 'is_callable_compatible' does -- the only difference is that
   899	                # we check and see if the impl_type's return value is a
   900	                # *supertype* of the overload alternative, not a *subtype*.
   901	                #
   902	                # This is to match the direction the implementation's return
   903	                # needs to be compatible in.
   904	                if impl_type.variables:
   905	                    impl: CallableType | None = unify_generic_callable(
   906	                        # Normalize both before unifying
   907	                        impl_type.with_unpacked_kwargs(),
   908	                        sig1.with_unpacked_kwargs(),
   909	                        ignore_return=False,
   910	                        return_constraint_direction=SUPERTYPE_OF,
   911	                    )
   912	                    if impl is None:
   913	                        self.msg.overloaded_signatures_typevar_specific(i + 1, defn.impl)
   914	                        continue
   915	                else:
   916	                    impl = impl_type
   917
   918	                # Prevent extra noise from inconsistent use of @classmethod by copying
   919	                # the first arg from the method being checked against.
   920	                if sig1.arg_types and defn.info:
   921	                    impl = impl.copy_modified(arg_types=[sig1.arg_types[0]] + impl.arg_types[1:])
   922
   923	                # Is the overload alternative's arguments subtypes of the implementation's?
   924	                if not is_callable_compatible(
   925	                    impl, sig1, is_compat=is_subtype, is_proper_subtype=False, ignore_return=True
   926	                ):
   927	                    self.msg.overloaded_signatures_arg_specific(i + 1, defn.impl)
   928
   929	                # Is the overload alternative's return type a subtype of the implementation's?
   930	                if not (
   931	                    is_subtype(sig1.ret_type, impl.ret_type)
   932	                    or is_subtype(impl.ret_type, sig1.ret_type)
   933	                ):
   934	                    self.msg.overloaded_signatures_ret_specific(i + 1, defn.impl)

   936	    # Here's the scoop about generators and coroutines.
   937	    #
   938	    # There are two kinds of generators: classic generators (functions
   939	    # with `yield` or `yield from` in the body) and coroutines
   940	    # (functions declared with `async def`).  The latter are specified
   941	    # in PEP 492 and only available in Python >= 3.5.
   942	    #
   943	    # Classic generators can be parameterized with three types:
   944	    # - ty is the Yield type (the type of y in `yield y`)
   945	    # - tc is the type reCeived by yield (the type of c in `c = yield`).
   946	    # - tr is the Return type (the type of r in `return r`)
   947	    #
   948	    # A classic generator must define a return type that's either
   949	    # `Generator[ty, tc, tr]`, Iterator[ty], or Iterable[ty] (or
   950	    # object or Any).  If tc/tr are not given, both are None.
   951	    #
   952	    # A coroutine must define a return type corresponding to tr; the
   953	    # other two are unconstrained.  The "external" return type (seen
   954	    # by the caller) is Awaitable[tr].
   955	    #
   956	    # In addition, there's the synthetic type AwaitableGenerator: it
   957	    # inherits from both Awaitable and Generator and can be used both
   958	    # in `yield from` and in `await`.  This type is set automatically
   959	    # for functions decorated with `@types.coroutine` or
   960	    # `@asyncio.coroutine`.  Its single parameter corresponds to tr.
   961	    #
   962	    # PEP 525 adds a new type, the asynchronous generator, which was
   963	    # first released in Python 3.6. Async generators are `async def`
   964	    # functions that can also `yield` values. They can be parameterized
   965	    # with two types, ty and tc, because they cannot return a value.
   966	    #
   967	    # There are several useful methods, each taking a type t and a
   968	    # flag c indicating whether it's for a generator or coroutine:
   969	    #
   970	    # - is_generator_return_type(t, c) returns whether t is a Generator,
   971	    #   Iterator, Iterable (if not c), or Awaitable (if c), or
   972	    #   AwaitableGenerator (regardless of c).
   973	    # - is_async_generator_return_type(t) returns whether t is an
   974	    #   AsyncGenerator.
   975	    # - get_generator_yield_type(t, c) returns ty.
   976	    # - get_generator_receive_type(t, c) returns tc.
   977	    # - get_generator_return_type(t, c) returns tr.
   978
   979	    def is_generator_return_type(self, typ: Type, is_coroutine: bool) -> bool:
   980	        """Is `typ` a valid type for a generator/coroutine?
   981
   982	        True if `typ` is a *supertype* of Generator or Awaitable.
   983	        Also true it it's *exactly* AwaitableGenerator (modulo type parameters).
   984	        """
   985	        typ = get_proper_type(typ)
   986	        if is_coroutine:
   987	            # This means we're in Python 3.5 or later.
   988	            at = self.named_generic_type("typing.Awaitable", [AnyType(TypeOfAny.special_form)])
   989	            if is_subtype(at, typ):
   990	                return True
   991	        else:
   992	            any_type = AnyType(TypeOfAny.special_form)
   993	            gt = self.named_generic_type("typing.Generator", [any_type, any_type, any_type])
   994	            if is_subtype(gt, typ):
   995	                return True
   996	        return isinstance(typ, Instance) and typ.type.fullname == "typing.AwaitableGenerator"

   998	    def is_async_generator_return_type(self, typ: Type) -> bool:
   999	        """Is `typ` a valid type for an async generator?
  1000
  1001	        True if `typ` is a supertype of AsyncGenerator.
  1002	        """
  1003	        try:
  1004	            any_type = AnyType(TypeOfAny.special_form)
  1005	            agt = self.named_generic_type("typing.AsyncGenerator", [any_type, any_type])
  1006	        except KeyError:
  1007	            # we're running on a version of typing that doesn't have AsyncGenerator yet
  1008	            return False
  1009	        return is_subtype(agt, typ)

  1011	    def get_generator_yield_type(self, return_type: Type, is_coroutine: bool) -> Type:
  1012	        """Given the declared return type of a generator (t), return the type it yields (ty)."""
  1013	        return_type = get_proper_type(return_type)
  1014
  1015	        if isinstance(return_type, AnyType):
  1016	            return AnyType(TypeOfAny.from_another_any, source_any=return_type)
  1017	        elif isinstance(return_type, UnionType):
  1018	            return make_simplified_union(
  1019	                [self.get_generator_yield_type(item, is_coroutine) for item in return_type.items]
  1020	            )
  1021	        elif not self.is_generator_return_type(
  1022	            return_type, is_coroutine
  1023	        ) and not self.is_async_generator_return_type(return_type):
  1024	            # If the function doesn't have a proper Generator (or
  1025	            # Awaitable) return type, anything is permissible.
  1026	            return AnyType(TypeOfAny.from_error)
  1027	        elif not isinstance(return_type, Instance):
  1028	            # Same as above, but written as a separate branch so the typechecker can understand.
  1029	            return AnyType(TypeOfAny.from_error)
  1030	        elif return_type.type.fullname == "typing.Awaitable":
  1031	            # Awaitable: ty is Any.
  1032	            return AnyType(TypeOfAny.special_form)
  1033	        elif return_type.args:
  1034	            # AwaitableGenerator, Generator, AsyncGenerator, Iterator, or Iterable; ty is args[0].
  1035	            ret_type = return_type.args[0]
  1036	            # TODO not best fix, better have dedicated yield token
  1037	            return ret_type
  1038	        else:
  1039	            # If the function's declared supertype of Generator has no type
  1040	            # parameters (i.e. is `object`), then the yielded values can't
  1041	            # be accessed so any type is acceptable.  IOW, ty is Any.
  1042	            # (However, see https://github.com/python/mypy/issues/1933)
  1043	            return AnyType(TypeOfAny.special_form)

  1045	    def get_generator_receive_type(self, return_type: Type, is_coroutine: bool) -> Type:
  1046	        """Given a declared generator return type (t), return the type its yield receives (tc)."""
  1047	        return_type = get_proper_type(return_type)
  1048
  1049	        if isinstance(return_type, AnyType):
  1050	            return AnyType(TypeOfAny.from_another_any, source_any=return_type)
  1051	        elif isinstance(return_type, UnionType):
  1052	            return make_simplified_union(
  1053	                [self.get_generator_receive_type(item, is_coroutine) for item in return_type.items]
  1054	            )
  1055	        elif not self.is_generator_return_type(
  1056	            return_type, is_coroutine
  1057	        ) and not self.is_async_generator_return_type(return_type):
  1058	            # If the function doesn't have a proper Generator (or
  1059	            # Awaitable) return type, anything is permissible.
  1060	            return AnyType(TypeOfAny.from_error)
  1061	        elif not isinstance(return_type, Instance):
  1062	            # Same as above, but written as a separate branch so the typechecker can understand.
  1063	            return AnyType(TypeOfAny.from_error)
  1064	        elif return_type.type.fullname == "typing.Awaitable":
  1065	            # Awaitable, AwaitableGenerator: tc is Any.
  1066	            return AnyType(TypeOfAny.special_form)
  1067	        elif (
  1068	            return_type.type.fullname in ("typing.Generator", "typing.AwaitableGenerator")
  1069	            and len(return_type.args) >= 3
  1070	        ):
  1071	            # Generator: tc is args[1].
  1072	            return return_type.args[1]
  1073	        elif return_type.type.fullname == "typing.AsyncGenerator" and len(return_type.args) >= 2:
  1074	            return return_type.args[1]
  1075	        else:
  1076	            # `return_type` is a supertype of Generator, so callers won't be able to send it
  1077	            # values.  IOW, tc is None.
  1078	            return NoneType()

  1080	    def get_coroutine_return_type(self, return_type: Type) -> Type:
  1081	        return_type = get_proper_type(return_type)
  1082	        if isinstance(return_type, AnyType):
  1083	            return AnyType(TypeOfAny.from_another_any, source_any=return_type)
  1084	        assert isinstance(return_type, Instance), "Should only be called on coroutine functions."
  1085	        # Note: return type is the 3rd type parameter of Coroutine.

  1088	    def get_generator_return_type(self, return_type: Type, is_coroutine: bool) -> Type:
  1089	        """Given the declared return type of a generator (t), return the type it returns (tr)."""
  1090	        return_type = get_proper_type(return_type)
  1091
  1092	        if isinstance(return_type, AnyType):
  1093	            return AnyType(TypeOfAny.from_another_any, source_any=return_type)
  1094	        elif isinstance(return_type, UnionType):
  1095	            return make_simplified_union(
  1096	                [self.get_generator_return_type(item, is_coroutine) for item in return_type.items]
  1097	            )
  1098	        elif not self.is_generator_return_type(return_type, is_coroutine):
  1099	            # If the function doesn't have a proper Generator (or
  1100	            # Awaitable) return type, anything is permissible.
  1101	            return AnyType(TypeOfAny.from_error)
  1102	        elif not isinstance(return_type, Instance):
  1103	            # Same as above, but written as a separate branch so the typechecker can understand.
  1104	            return AnyType(TypeOfAny.from_error)
  1105	        elif return_type.type.fullname == "typing.Awaitable" and len(return_type.args) == 1:
  1106	            # Awaitable: tr is args[0].
  1107	            return return_type.args[0]
  1108	        elif (
  1109	            return_type.type.fullname in ("typing.Generator", "typing.AwaitableGenerator")
  1110	            and len(return_type.args) >= 3
  1111	        ):
  1112	            # AwaitableGenerator, Generator: tr is args[2].
  1113	            return return_type.args[2]
  1114	        else:
  1115	            # We have a supertype of Generator (Iterator, Iterable, object)
  1116	            # Treat `Iterator[X]` as a shorthand for `Generator[X, Any, None]`.
  1117	            return NoneType()

  1125	    def _visit_func_def(self, defn: FuncDef) -> None:
  1126	        """Type check a function definition."""
  1127	        self.check_func_item(defn, name=defn.name)
  1128	        if defn.info:
  1129	            if not defn.is_overload and not defn.is_decorated:
  1130	                # If the definition is the implementation for an
  1131	                # overload, the legality of the override has already
  1132	                # been typechecked, and decorated methods will be
  1133	                # checked when the decorator is.
  1134	                found_method_base_classes = self.check_method_override(defn)
  1135	                self.check_explicit_override_decorator(defn, found_method_base_classes)
  1136	            self.check_inplace_operator_method(defn)
  1137	        if defn.original_def:
  1138	            # Override previous definition.
  1139	            new_type = self.function_type(defn)
  1140	            self.check_func_def_override(defn, new_type)

  1142	    def check_func_item(
  1143	        self,
  1144	        defn: FuncItem,
  1145	        type_override: CallableType | None = None,
  1146	        name: str | None = None,
  1147	        allow_empty: bool = False,
  1148	    ) -> None:
  1149	        """Type check a function.
  1150
  1151	        If type_override is provided, use it as the function type.
  1152	        """
  1153	        self.dynamic_funcs.append(defn.is_dynamic() and not type_override)
  1154
  1155	        enclosing_node_deferred = self.current_node_deferred
  1156	        with self.enter_partial_types(is_function=True):
  1157	            typ = self.function_type(defn)
  1158	            if type_override:
  1159	                typ = type_override.copy_modified(line=typ.line, column=typ.column)
  1160	            if isinstance(typ, CallableType):
  1161	                with self.enter_attribute_inference_context():
  1162	                    self.check_func_def(defn, typ, name, allow_empty)
  1163	            else:
  1164	                raise RuntimeError("Not supported")
  1165
  1166	        self.dynamic_funcs.pop()
  1167	        self.current_node_deferred = enclosing_node_deferred
  1168
  1169	        if name == "__exit__":
  1170	            self.check__exit__return_type(defn)
  1171	        # TODO: the following logic should move to the dataclasses plugin
  1172	        #  https://github.com/python/mypy/issues/15515
  1173	        if name == "__post_init__":
  1174	            if dataclasses_plugin.is_processed_dataclass(defn.info):
  1175	                dataclasses_plugin.check_post_init(self, defn, defn.info)

  1177	    def check_func_def_override(self, defn: FuncDef, new_type: FunctionLike) -> None:
  1178	        assert defn.original_def is not None
  1179	        if isinstance(defn.original_def, FuncDef):
  1180	            # Function definition overrides function definition.
  1181	            old_type = self.function_type(defn.original_def)
  1182	            if not is_same_type(new_type, old_type):
  1183	                self.msg.incompatible_conditional_function_def(defn, old_type, new_type)
  1184	        else:
  1185	            # Function definition overrides a variable initialized via assignment or a
  1186	            # decorated function.
  1187	            orig_type = defn.original_def.type
  1188	            if orig_type is None:
  1189	                # If other branch is unreachable, we don't type check it and so we might
  1190	                # not have a type for the original definition
  1191	                return
  1192	            if isinstance(orig_type, PartialType):
  1193	                if orig_type.type is None:
  1194	                    # Ah this is a partial type. Give it the type of the function.
  1195	                    orig_def = defn.original_def
  1196	                    if isinstance(orig_def, Decorator):
  1197	                        var = orig_def.var
  1198	                    else:
  1199	                        var = orig_def
  1200	                    partial_types = self.find_partial_types(var)
  1201	                    if partial_types is not None:
  1202	                        var.type = new_type
  1203	                        del partial_types[var]
  1204	                else:
  1205	                    # Trying to redefine something like partial empty list as function.
  1206	                    self.fail(message_registry.INCOMPATIBLE_REDEFINITION, defn)
  1207	            else:
  1208	                name_expr = NameExpr(defn.name)
  1209	                name_expr.node = defn.original_def
  1210	                self.binder.assign_type(name_expr, new_type, orig_type)
  1211	                self.check_subtype(
  1212	                    new_type,
  1213	                    orig_type,
  1214	                    defn,
  1215	                    message_registry.INCOMPATIBLE_REDEFINITION,
  1216	                    "redefinition with type",
  1217	                    "original type",
  1218	                )

  1220	    @contextmanager
  1221	    def enter_attribute_inference_context(self) -> Iterator[None]:
  1222	        old_types = self.inferred_attribute_types
  1223	        self.inferred_attribute_types = {}
  1224	        yield None
  1225	        self.inferred_attribute_types = old_types

  1227	    def check_func_def(
  1228	        self, defn: FuncItem, typ: CallableType, name: str | None, allow_empty: bool = False
  1229	    ) -> None:
  1230	        """Type check a function definition."""
  1231	        # Expand type variables with value restrictions to ordinary types.
  1232	        self.check_typevar_defaults(typ.variables)
  1233	        expanded = self.expand_typevars(defn, typ)
  1234	        original_typ = typ
  1235	        for item, typ in expanded:
  1236	            old_binder = self.binder
  1237	            self.binder = ConditionalTypeBinder(self.options)
  1238	            with self.binder.top_frame_context():
  1239	                defn.expanded.append(item)
  1240
  1241	                # We may be checking a function definition or an anonymous
  1242	                # function. In the first case, set up another reference with the
  1243	                # precise type.
  1244	                if isinstance(item, FuncDef):
  1245	                    fdef = item
  1246	                    # Check if __init__ has an invalid return type.
  1247	                    if (
  1248	                        fdef.info
  1249	                        and fdef.name in ("__init__", "__init_subclass__")
  1250	                        and not isinstance(
  1251	                            get_proper_type(typ.ret_type), (NoneType, UninhabitedType)
  1252	                        )
  1253	                        and not self.dynamic_funcs[-1]
  1254	                    ):
  1255	                        self.fail(
  1256	                            message_registry.MUST_HAVE_NONE_RETURN_TYPE.format(fdef.name), item
  1257	                        )
  1258
  1259	                    # Check validity of __new__ signature
  1260	                    if fdef.info and fdef.name == "__new__":
  1261	                        self.check___new___signature(fdef, typ)
  1262
  1263	                    self.check_for_missing_annotations(fdef)
  1264	                    if self.options.disallow_any_unimported:
  1265	                        if fdef.type and isinstance(fdef.type, CallableType):
  1266	                            ret_type = fdef.type.ret_type
  1267	                            if has_any_from_unimported_type(ret_type):
  1268	                                self.msg.unimported_type_becomes_any("Return type", ret_type, fdef)
  1269	                            for idx, arg_type in enumerate(fdef.type.arg_types):
  1270	                                if has_any_from_unimported_type(arg_type):
  1271	                                    prefix = f'Argument {idx + 1} to "{fdef.name}"'
  1272	                                    self.msg.unimported_type_becomes_any(prefix, arg_type, fdef)
  1273	                    check_for_explicit_any(
  1274	                        fdef.type, self.options, self.is_typeshed_stub, self.msg, context=fdef
  1275	                    )
  1276
  1277	                if name:  # Special method names
  1278	                    if defn.info and self.is_reverse_op_method(name):
  1279	                        self.check_reverse_op_method(item, typ, name, defn)
  1280	                    elif name in ("__getattr__", "__getattribute__"):
  1281	                        self.check_getattr_method(typ, defn, name)
  1282	                    elif name == "__setattr__":
  1283	                        self.check_setattr_method(typ, defn)
  1284
  1285	                # Refuse contravariant return type variable
  1286	                if isinstance(typ.ret_type, TypeVarType):

  1288	                        self.fail(
  1289	                            message_registry.RETURN_TYPE_CANNOT_BE_CONTRAVARIANT, typ.ret_type
  1290	                        )

  1292	                elif (
  1293	                    isinstance(original_typ.ret_type, TypeVarType) and original_typ.ret_type.values
  1294	                ):

  1296	                    # to a raw value. This is a hack to get it back.
  1297	                    self.check_unbound_return_typevar(original_typ)
  1298

  1300	                if defn.is_generator:
  1301	                    if defn.is_async_generator:
  1302	                        if not self.is_async_generator_return_type(typ.ret_type):
  1303	                            self.fail(
  1304	                                message_registry.INVALID_RETURN_TYPE_FOR_ASYNC_GENERATOR, typ
  1305	                            )
  1306	                    else:
  1307	                        if not self.is_generator_return_type(typ.ret_type, defn.is_coroutine):
  1308	                            self.fail(message_registry.INVALID_RETURN_TYPE_FOR_GENERATOR, typ)
  1309
  1310	                # Fix the type if decorated with `@types.coroutine` or `@asyncio.coroutine`.
  1311	                if defn.is_awaitable_coroutine:
  1312	                    # Update the return type to AwaitableGenerator.
  1313	                    # (This doesn't exist in typing.py, only in typing.pyi.)
  1314	                    t = typ.ret_type
  1315	                    c = defn.is_coroutine

  1317	                    tc = self.get_generator_receive_type(t, c)
  1318	                    if c:
  1319	                        tr = self.get_coroutine_return_type(t)
  1320	                    else:
  1321	                        tr = self.get_generator_return_type(t, c)
  1322	                    ret_type = self.named_generic_type(
  1323	                        "typing.AwaitableGenerator", [ty, tc, tr, t]
  1324	                    )
  1325	                    typ = typ.copy_modified(ret_type=ret_type)
  1326	                    defn.type = typ
  1327
  1328	                # Push return type.
  1329	                self.return_types.append(typ.ret_type)
  1330
  1331	                with self.scope.push_function(defn):
  1332	                    # We temporary push the definition to get the self type as
  1333	                    # visible from *inside* of this function/method.
  1334	                    ref_type: Type | None = self.scope.active_self_type()
  1335
  1336	                if typ.type_is:
  1337	                    arg_index = 0
  1338	                    # For methods and classmethods, we want the second parameter
  1339	                    if ref_type is not None and (not defn.is_static or defn.name == "__new__"):
  1340	                        arg_index = 1
  1341	                    if arg_index < len(typ.arg_types) and not is_subtype(

  1343	                    ):
  1344	                        self.fail(
  1345	                            message_registry.NARROWED_TYPE_NOT_SUBTYPE.format(
  1346	                                format_type(typ.type_is, self.options),
  1347	                                format_type(typ.arg_types[arg_index], self.options),
  1348	                            ),
  1349	                            item,
  1350	                        )
  1351

  1354	                    arg_type = typ.arg_types[i]
  1355	                    if (
  1356	                        isinstance(defn, FuncDef)
  1357	                        and ref_type is not None
  1358	                        and i == 0
  1359	                        and (not defn.is_static or defn.name == "__new__")

  1369	                                if (
  1370	                                    isinstance(erased, Instance)
  1371	                                    and erased.type.is_protocol

  1374	                                    and erased.item.type.is_protocol

  1379	                                    msg = None

  1385	                                else:
  1386	                                    msg = message_registry.MISSING_OR_INVALID_SELF_TYPE

  1388	                                    self.fail(msg, defn)
  1389	                    elif isinstance(arg_type, TypeVarType):
  1390	                        # Refuse covariant parameter type variables
  1391	                        # TODO: check recursively for inner type variables
  1392	                        if (
  1393	                            arg_type.variance == COVARIANT
  1394	                            and defn.name not in ("__init__", "__new__", "__post_init__")

  1399	                                ctx = typ

  1402	                    store_argument_type(item, i, typ, self.named_generic_type)

  1408	            # Type check body in a new scope.
  1409	            with self.binder.top_frame_context():

  1411	                # (i.e. we can't find an assignment that might change the type of the
  1412	                # variable afterwards).
  1413	                new_frame: Frame | None = None
  1414	                for frame in old_binder.frames:

  1431	                        if v.type is not None:

  1433	                            n.node = v

  1452	                    self.binder.pop_frame(True, 0)
  1453
  1454	            if not unreachable:
  1455	                if defn.is_generator or is_named_instance(
  1456	                    self.return_types[-1], "typing.AwaitableGenerator"
  1457	                ):
  1458	                    return_type = self.get_generator_return_type(
  1459	                        self.return_types[-1], defn.is_coroutine

  1464	                    return_type = self.return_types[-1]
  1465	                return_type = get
