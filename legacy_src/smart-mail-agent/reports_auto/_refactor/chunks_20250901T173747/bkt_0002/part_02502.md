   306	    def should_trace(self, filename: str, frame: FrameType | None = None) -> TFileDisposition:
   307	        """Decide whether to trace execution in `filename`, with a reason.
   308
   309	        This function is called from the trace function.  As each new file name
   310	        is encountered, this function determines whether it is traced or not.
   311
   312	        Returns a FileDisposition object.
   313
   314	        """
   315	        original_filename = filename
   316	        disp = disposition_init(self.disp_class, filename)
   317
   318	        def nope(disp: TFileDisposition, reason: str) -> TFileDisposition:
   319	            """Simple helper to make it easy to return NO."""
   320	            disp.trace = False
   321	            disp.reason = reason
   322	            return disp
   323
   324	        if original_filename.startswith("<"):
   325	            return nope(disp, "original file name is not real")
   326
   327	        if frame is not None:
   328	            # Compiled Python files have two file names: frame.f_code.co_filename is
   329	            # the file name at the time the .pyc was compiled.  The second name is
   330	            # __file__, which is where the .pyc was actually loaded from.  Since
   331	            # .pyc files can be moved after compilation (for example, by being
   332	            # installed), we look for __file__ in the frame and prefer it to the
   333	            # co_filename value.
   334	            dunder_file = frame.f_globals and frame.f_globals.get("__file__")
   335	            if dunder_file:
   336	                # Danger: __file__ can (rarely?) be of type Path.
   337	                filename = source_for_file(str(dunder_file))
   338	                if original_filename and not original_filename.startswith("<"):
   339	                    orig = os.path.basename(original_filename)
   340	                    if orig != os.path.basename(filename):
   341	                        # Files shouldn't be renamed when moved. This happens when
   342	                        # exec'ing code.  If it seems like something is wrong with
   343	                        # the frame's file name, then just use the original.
   344	                        filename = original_filename
   345
   346	        if not filename:
   347	            # Empty string is pretty useless.
   348	            return nope(disp, "empty string isn't a file name")
   349
   350	        if filename.startswith("memory:"):
   351	            return nope(disp, "memory isn't traceable")
   352
   353	        if filename.startswith("<"):
   354	            # Lots of non-file execution is represented with artificial
   355	            # file names like "<string>", "<doctest readme.txt[0]>", or
   356	            # "<exec_function>".  Don't ever trace these executions, since we
   357	            # can't do anything with the data later anyway.
   358	            return nope(disp, "file name is not real")
   359
   360	        canonical = canonical_filename(filename)
   361	        disp.canonical_filename = canonical
   362
   363	        # Try the plugins, see if they have an opinion about the file.
   364	        plugin = None
   365	        for plugin in self.plugins.file_tracers:
   366	            if not plugin._coverage_enabled:
   367	                continue

   369	            try:
   370	                file_tracer = plugin.file_tracer(canonical)
   371	                if file_tracer is not None:

   374	                    disp.file_tracer = file_tracer
   375	                    if file_tracer.has_dynamic_source_filename():
   376	                        disp.has_dynamic_filename = True
   377	                    else:
   378	                        disp.source_filename = canonical_filename(
   379	                            file_tracer.source_filename(),
   380	                        )
   381	                    break

   383	                plugin_name = plugin._coverage_plugin_name
   384	                tb = traceback.format_exc()
   385	                self.warn(f"Disabling plug-in {plugin_name!r} due to an exception:\n{tb}")
   386	                plugin._coverage_enabled = False
   387	                continue
   388	        else:
   389	            # No plugin wanted it: it's Python.
   390	            disp.trace = True
   391	            disp.source_filename = canonical
   392
   393	        if not disp.has_dynamic_filename:
   394	            if not disp.source_filename:
   395	                raise PluginError(
   396	                    f"Plugin {plugin!r} didn't set source_filename for '{disp.original_filename}'",
   397	                )

   404	    def check_include_omit_etc(self, filename: str, frame: FrameType | None) -> str | None:
   405	        """Check a file name against the include, omit, etc, rules.
   406
   407	        Returns a string or None.  String means, don't trace, and is the reason
   408	        why.  None means no reason found to not trace.

   410	        """

   413	        # If the user specified source or include, then that's authoritative

   433	                return "inside --source, but is third-party"
   434	        elif self.include_match:

   437	        else:

   439	            # will be measured otherwise.

   441	                return "is part of coverage.py"

   443	            # If we aren't supposed to trace installed code, then check if this
   444	            # is near the Python standard library and skip it if so.
   445	            if self.pylib_match and self.pylib_match.match(filename):
   446	                return "is in the stdlib"

   453	        if self.omit_match and self.omit_match.match(filename):
   454	            return "is inside an --omit pattern"

   457	        try:
   458	            filename.encode("utf-8")

    23	def line_hash(line: str) -> str:
    24	    """Produce a hash of a source line for use in the LCOV file."""
    25	    # The LCOV file format optionally allows each line to be MD5ed as a
    26	    # fingerprint of the file.  This is not a security use.  Some security
    27	    # scanners raise alarms about the use of MD5 here, but it is a false
    28	    # positive.  This is not a security concern.
    29	    # The unusual encoding of the MD5 hash, as a base64 sequence with the
    30	    # trailing = signs stripped, is specified by the LCOV file format.
    31	    hashed = hashlib.md5(line.encode("utf-8"), usedforsecurity=False).digest()
    32	    return base64.b64encode(hashed).decode("ascii").rstrip("=")

    98	def lcov_arcs(
    99	    fr: FileReporter,
   100	    analysis: Analysis,
   101	    lines: list[int],
   102	    outfile: IO[str],
   103	) -> None:
   104	    """Emit branch coverage records for an analyzed file."""
   105	    branch_stats = analysis.branch_stats()
   106	    executed_arcs = analysis.executed_branch_arcs()
   107	    missing_arcs = analysis.missing_branch_arcs()
   108
   109	    for line in lines:
   110	        if line not in branch_stats:
   111	            continue
   112
   113	        # This is only one of several possible ways to map our sets of executed
   114	        # and not-executed arcs to BRDA codes.  It seems to produce reasonable
   115	        # results when fed through genhtml.
   116	        _, taken = branch_stats[line]
   117
   118	        if taken == 0:
   119	            # When _none_ of the out arcs from 'line' were executed,
   120	            # it can mean the line always raised an exception.
   121	            assert len(executed_arcs[line]) == 0
   122	            destinations = [
   123	                (dst, "-") for dst in missing_arcs[line]
   124	            ]
   125	        else:
   126	            # Q: can we get counts of the number of times each arc was executed?
   127	            # branch_stats has "total" and "taken" counts for each branch,
   128	            # but it doesn't have "taken" broken down by destination.
   129	            destinations = [
   130	                (dst, "1") for dst in executed_arcs[line]
   131	            ]
   132	            destinations.extend(
   133	                (dst, "0") for dst in missing_arcs[line]
   134	            )
   135
   136	        # Sort exit arcs after normal arcs.  Exit arcs typically come from
   137	        # an if statement, at the end of a function, with no else clause.
   138	        # This structure reads like you're jumping to the end of the function
   139	        # when the conditional expression is false, so it should be presented
   140	        # as the second alternative for the branch, after the alternative that
   141	        # enters the if clause.
   142	        destinations.sort(key=lambda d: (d[0] < 0, d))
   143
   144	        for dst, hit in destinations:
   145	            branch = fr.arc_description(line, dst)
   146	            outfile.write(f"BRDA:{line},0,{branch},{hit}\n")
   147
   148	    # Summary of the branch coverage.
   149	    brf = sum(t for t, k in branch_stats.values())
   150	    brh = brf - sum(t - k for t, k in branch_stats.values())
   151	    if brf > 0:
   152	        outfile.write(f"BRF:{brf}\n")
   153	        outfile.write(f"BRH:{brh}\n")

   159	class Hasher:
   160	    """Hashes Python data for fingerprinting."""
   161	    def __init__(self) -> None:
   162	        self.hash = hashlib.new("sha3_256", usedforsecurity=False)
   163
   164	    def update(self, v: Any) -> None:
   165	        """Add `v` to the hash, recursively if needed."""
   166	        self.hash.update(str(type(v)).encode("utf-8"))
   167	        if isinstance(v, str):
   168	            self.hash.update(v.encode("utf-8"))
   169	        elif isinstance(v, bytes):
   170	            self.hash.update(v)
   171	        elif v is None:
   172	            pass
   173	        elif isinstance(v, (int, float)):
   174	            self.hash.update(str(v).encode("utf-8"))
   175	        elif isinstance(v, (tuple, list)):
   176	            for e in v:
   177	                self.update(e)
   178	        elif isinstance(v, dict):
   179	            keys = v.keys()
   180	            for k in sorted(keys):
   181	                self.update(k)
   182	                self.update(v[k])
   183	        else:
   184	            for k in dir(v):
   185	                if k.startswith("__"):
   186	                    continue
   187	                a = getattr(v, k)
   188	                if inspect.isroutine(a):
   189	                    continue
   190	                self.update(k)
   191	                self.update(a)
   192	        self.hash.update(b".")

   228	def substitute_variables(text: str, variables: Mapping[str, str]) -> str:
   229	    """Substitute ``${VAR}`` variables in `text` with their values.
   230
   231	    Variables in the text can take a number of shell-inspired forms::
   232
   233	        $VAR
   234	        ${VAR}
   235	        ${VAR?}             strict: an error if VAR isn't defined.
   236	        ${VAR-missing}      defaulted: "missing" if VAR isn't defined.
   237	        $$                  just a dollar sign.
   238
   239	    `variables` is a dictionary of variable values.
   240
   241	    Returns the resulting text with values substituted.
   242
   243	    """
   244	    dollar_pattern = r"""(?x)   # Use extended regex syntax
   245	        \$                      # A dollar sign,
   246	        (?:                     # then
   247	            (?P<dollar> \$ ) |      # a dollar sign, or
   248	            (?P<word1> \w+ ) |      # a plain word, or
   249	            \{                      # a {-wrapped
   250	                (?P<word2> \w+ )        # word,
   251	                (?:                         # either
   252	                    (?P<strict> \? ) |      # with a strict marker
   253	                    -(?P<defval> [^}]* )    # or a default value
   254	                )?                      # maybe.
   255	            }
   256	        )
   257	        """
   258
   259	    dollar_groups = ("dollar", "word1", "word2")
   260
   261	    def dollar_replace(match: re.Match[str]) -> str:
   262	        """Called for each $replacement."""
   263	        # Only one of the dollar_groups will have matched, just get its text.
   264	        word = next(g for g in match.group(*dollar_groups) if g)    # pragma: always breaks
   265	        if word == "$":
   266	            return "$"
   267	        elif word in variables:
   268	            return variables[word]
   269	        elif match["strict"]:
   270	            msg = f"Variable {word} is undefined: {text!r}"
   271	            raise CoverageException(msg)
   272	        else:
   273	            return match["defval"]
   274
   275	    text = re.sub(dollar_pattern, dollar_replace, text)
   276	    return text

    25	def nums_to_numbits(nums: Iterable[int]) -> bytes:
    26	    """Convert `nums` into a numbits.
    27
    28	    Arguments:
    29	        nums: a reusable iterable of integers, the line numbers to store.
    30
    31	    Returns:
    32	        A binary blob.
    33	    """
    34	    try:
    35	        nbytes = max(nums) // 8 + 1
    36	    except ValueError:
    37	        # nums was empty.
    38	        return b""
    39	    b = bytearray(nbytes)
    40	    for num in nums:
    41	        b[num//8] |= 1 << num % 8

    45	def numbits_to_nums(numbits: bytes) -> list[int]:
    46	    """Convert a numbits into a list of numbers.
    47
    48	    Arguments:
    49	        numbits: a binary blob, the packed number set.
    50
    51	    Returns:
    52	        A list of ints.
    53
    54	    When registered as a SQLite function by :func:`register_sqlite_functions`,
    55	    this returns a string, a JSON-encoded list of ints.

    57	    """
    58	    nums = []
    59	    for byte_i, byte in enumerate(numbits):
    60	        for bit_i in range(8):
    61	            if (byte & (1 << bit_i)):
    62	                nums.append(byte_i * 8 + bit_i)

   312	        """Adjust arcs to fix jumps leaving `with` statements.

   314	        Consider this code:

   316	            with open("/tmp/test", "w") as f1:
   317	                a = 2
   318	                b = 3

   321	        In 3.10+, we get traces for lines 1, 2, 3, 1, 4.  But we want to present

   326	        is the actual measured arc from the end of the with block back to the
   327	        start of the with-statement.  The values are start_next (the with
   328	        statement to the next statement after the with), and end_next (the end
   329	        of the with-statement to the next statement after the with).

   332	        correct a longer chain of arcs.

   339	                end0 = arc[0]
   340	                to_remove.add(arc)
   341	                start_next, end_next = self._with_jump_fixers[arc]
   342	                while start_next in self._with_jump_fixers:
   343	                    to_remove.add(start_next)
   344	                    start_next, end_next = self._with_jump_fixers[start_next]
   345	                    to_remove.add(end_next)
   346	                to_add.add((end0, end_next[1]))
   347	                to_remove.add(start_next)
   348	        arcs = (set(arcs) | to_add) - to_remove
   349	        return arcs
   350
   351	    @functools.lru_cache
   352	    def exit_counts(self) -> dict[TLineNo, int]:
   353	        """Get a count of exits from that each line.
   354
   355	        Excluded lines are excluded.
   356
   357	        """
   358	        exit_counts: dict[TLineNo, int] = collections.defaultdict(int)
   359	        for l1, l2 in self.arcs():
   360	            assert l1 > 0, f"{l1=} should be greater than zero in {self.filename}"
   361	            if l1 in self.excluded:
   362	                # Don't report excluded lines as line numbers.
   363	                continue
   364	            if l2 in self.excluded:
   365	                # Arcs to excluded lines shouldn't count.
   366	                continue
   367	            exit_counts[l1] += 1
   368
   369	        return exit_counts

   371	    def _finish_action_msg(self, action_msg: str | None, end: TLineNo) -> str:
   372	        """Apply some defaulting and formatting to an arc's description."""
   373	        if action_msg is None:
   374	            if end < 0:
   375	                action_msg = "jump to the function exit"
   376	            else:
   377	                action_msg = "jump to line {lineno}"
   378	        action_msg = action_msg.format(lineno=end)
   379	        return action_msg

   381	    def missing_arc_description(self, start: TLineNo, end: TLineNo) -> str:
   382	        """Provide an English sentence describing a missing arc."""
   383	        if self._missing_arc_fragments is None:
   384	            self._analyze_ast()
   385	            assert self._missing_arc_fragments is not None
   386
   387	        fragment_pairs = self._missing_arc_fragments.get((start, end), [(None, None)])
   388
   389	        msgs = []
   390	        for missing_cause_msg, action_msg in fragment_pairs:
   391	            action_msg = self._finish_action_msg(action_msg, end)
   392	            msg = f"line {start} didn't {action_msg}"
   393	            if missing_cause_msg is not None:
   394	                msg += f" because {missing_cause_msg.format(lineno=start)}"
   395
   396	            msgs.append(msg)
   397
   398	        return " or ".join(msgs)

   401	        """Provide an English description of an arc's effect."""
   402	        if self._missing_arc_fragments is None:
   403	            self._analyze_ast()
   404	            assert self._missing_arc_fragments is not None

   406	        fragment_pairs = self._missing_arc_fragments.get((start, end), [(None, None)])
   407	        action_msg = self._finish_action_msg(fragment_pairs[0][1], end)
   408	        return action_msg

   411	class ByteParser:
   412	    """Parse bytecode to understand the structure of code."""
   413
   414	    def __init__(
   415	        self,
   416	        text: str,
   417	        code: CodeType | None = None,
   418	        filename: str | None = None,
   419	    ) -> None:
   420	        self.text = text
   421	        if code is not None:
   422	            self.code = code
   423	        else:
   424	            assert filename is not None
   425	            # We only get here if earlier ast parsing succeeded, so no need to
   426	            # catch errors.
   427	            self.code = compile(text, filename, "exec", dont_inherit=True)
   428
   429	    def child_parsers(self) -> Iterable[ByteParser]:
   430	        """Iterate over all the code objects nested within this one.
   431
   432	        The iteration includes `self` as its first value.
   433
   434	        We skip code objects named `__annotate__` since they are deferred
   435	        annotations that usually are never run.  If there are errors in the
   436	        annotations, they will be caught by type checkers or other tools that
   437	        use annotations.
   438
   439	        """
   440	        return (
   441	            ByteParser(self.text, code=c)
   442	            for c in code_objects(self.code)
   443	            if c.co_name != "__annotate__"
   444	        )
   445
   446	    def _line_numbers(self) -> Iterable[TLineNo]:
   447	        """Yield the line numbers possible in this code object.
   448
   449	        Uses co_lnotab described in Python/compile.c to find the
   450	        line numbers.  Produces a sequence: l0, l1, ...
   451	        """
   452	        if hasattr(self.code, "co_lines"):
   453	            # PYVERSIONS: new in 3.10
   454	            for _, _, line in self.code.co_lines():
   455	                if line:
   456	                    yield line
   457	        else:
   458	            # Adapted from dis.py in the standard library.
   459	            byte_increments = self.code.co_lnotab[0::2]
   460	            line_increments = self.code.co_lnotab[1::2]
   461
   462	            last_line_num: TLineNo | None = None
   463	            line_num = self.code.co_firstlineno
   464	            byte_num = 0
   465	            for byte_incr, line_incr in zip(byte_increments, line_increments):
   466	                if byte_incr:
   467	                    if line_num != last_line_num:
   468	                        yield line_num
   469	                        last_line_num = line_num
   470	                    byte_num += byte_incr
   471	                if line_incr >= 0x80:
   472	                    line_incr -= 0x100
   473	                line_num += line_incr
   474	            if line_num != last_line_num:
   475	                yield line_num

   477	    def _find_statements(self) -> Iterable[TLineNo]:
   478	        """Find the statements in `self.code`.
   479
   480	        Produce a sequence of line numbers that start statements.  Recurses
   481	        into all code objects reachable from `self.code`.
   482
   483	        """
   484	        for bp in self.child_parsers():
   485	            # Get all of the lineno information from this code.
   486	            yield from bp._line_numbers()

   495	    """The information needed to start an arc.
   496
   497	    `lineno` is the line number the arc starts from.
   498
   499	    `cause` is an English text fragment used as the `missing_cause_msg` for
   500	    AstArcAnalyzer.missing_arc_fragments.  It will be used to describe why an

   514	    - ArcStart(1, "the condition on line 1 was never true")

   516	    The first would be used to create an arc from 1 to 3, creating a message like
   517	    "line 1 didn't jump to line 3 because the condition on line 1 was always true."
   518
   519	    The second would be used for the arc from 1 to 2, creating a message like
   520	    "line 1 didn't jump to line 2 because the condition on line 1 was never true."
   521
   522	    """
   523	    lineno: TLineNo
   524	    cause: str = ""

   650	def is_constant_test_expr(node: ast.AST) -> tuple[bool, bool]:
   651	    """Is this a compile-time constant test expression?
   652
   653	    We don't try to mimic all of CPython's optimizations.  We just have to
   654	    handle the kinds of constant expressions people might actually use.
   655
   656	    """
   657	    if isinstance(node, ast.Constant):
   658	        return True, bool(node.value)
   659	    elif isinstance(node, ast.Name):
   660	        if node.id in ["True", "False", "None", "__debug__"]:
   661	            return True, eval(node.id)  # pylint: disable=eval-used
   662	    elif isinstance(node, ast.UnaryOp) and isinstance(node.op, ast.Not):
   663	        is_constant, val = is_constant_test_expr(node.operand)
   664	        return is_constant, not val
   665	    elif isinstance(node, ast.BoolOp):
   666	        rets = [is_constant_test_expr(v) for v in node.values]
   667	        is_constant = all(is_const for is_const, _ in rets)
   668	        if is_constant:
   669	            op = any if isinstance(node.op, ast.Or) else all
   670	            return True, op(v for _, v in rets)
   671	    return False, False

   674	class AstArcAnalyzer:
   675	    """Analyze source text with an AST to find executable code paths.
   676
   677	    The .analyze() method does the work, and populates these attributes:
   678
   679	    `arcs`: a set of (from, to) pairs of the the arcs possible in the code.
   680
   681	    `missing_arc_fragments`: a dict mapping (from, to) arcs to lists of
   682	    message fragments explaining why the arc is missing from execution::
   683
   684	        { (start, end): [(missing_cause_msg, action_msg), ...], }
   685
   686	    For an arc starting from line 17, they should be usable to form complete
   687	    sentences like: "Line 17 didn't {action_msg} because {missing_cause_msg}".
   688
   689	    NOTE: Starting in July 2024, I've been whittling this down to only report
   690	    arc that are part of true branches.  It's not clear how far this work will
   691	    go.
   692
   693	    """
   694
   695	    def __init__(
   696	        self,
   697	        filename: str,
   698	        root_node: ast.AST,
   699	        statements: set[TLineNo],
   700	        multiline: dict[TLineNo, TLineNo],
   701	    ) -> None:
   702	        self.filename = filename
   703	        self.root_node = root_node
   704	        self.statements = {multiline.get(l, l) for l in statements}
   705	        self.multiline = multiline
   706
   707	        # Turn on AST dumps with an environment variable.
   708	        # $set_env.py: COVERAGE_AST_DUMP - Dump the AST nodes when parsing code.
   709	        dump_ast = bool(int(os.getenv("COVERAGE_AST_DUMP", "0")))
   710
   711	        if dump_ast:                                # pragma: debugging
   712	            # Dump the AST so that failing tests have helpful output.
   713	            print(f"Statements: {self.statements}")
   714	            print(f"Multiline map: {self.multiline}")
   715	            print(ast.dump(self.root_node, include_attributes=True, indent=4))
   716
   717	        self.arcs: set[TArc] = set()
   718	        self.missing_arc_fragments: TArcFragments = collections.defaultdict(list)
   719	        self.block_stack: list[Block] = []
   720
   721	        # If `with` clauses jump to their start on the way out, we need
   722	        # information to be able to skip over that jump.  We record the arcs
   723	        # from `with` into the clause (with_entries), and the arcs from the
   724	        # clause to the `with` (with_exits).
   725	        self.current_with_starts: set[TLineNo] = set()
   726	        self.all_with_starts: set[TLineNo] = set()
   727	        self.with_entries: set[TArc] = set()
   728	        self.with_exits: set[TArc] = set()
   729
   730	        # $set_env.py: COVERAGE_TRACK_ARCS - Trace possible arcs added while parsing code.
   731	        self.debug = bool(int(os.getenv("COVERAGE_TRACK_ARCS", "0")))
   732
   733	    def analyze(self) -> None:
   734	        """Examine the AST tree from `self.root_node` to determine possible arcs."""
   735	        for node in ast.walk(self.root_node):
   736	            node_name = node.__class__.__name__
   737	            code_object_handler = getattr(self, f"_code_object__{node_name}", None)
   738	            if code_object_handler is not None:
   739	                code_object_handler(node)
   740
   741	    def with_jump_fixers(self) -> dict[TArc, tuple[TArc, TArc]]:
   742	        """Get a dict with data for fixing jumps out of with statements.
   743
   744	        Returns a dict.  The keys are arcs leaving a with-statement by jumping
   745	        back to its start.  The values are pairs: first, the arc from the start
   746	        to the next statement, then the arc that exits the with without going
   747	        to the start.
   748
   749	        """
   750	        fixers = {}
   751	        with_nexts = {
   752	            arc
   753	            for arc in self.arcs
   754	            if arc[0] in self.all_with_starts and arc not in self.with_entries
   755	        }
   756	        for start in self.all_with_starts:
   757	            nexts = {arc[1] for arc in with_nexts if arc[0] == start}
   758	            if not nexts:
   759	                continue
   760	            assert len(nexts) == 1, f"Expected one arc, got {nexts} with {start = }"
   761	            nxt = nexts.pop()
   762	            ends = {arc[0] for arc in self.with_exits if arc[1] == start}
   763	            for end in ends:
   764	                fixers[(end, start)] = ((start, nxt), (end, nxt))
   765	        return fixers
