    39	        # Reduce the set of arcs to the ones that could be branches.
    40	        dests = collections.defaultdict(set)
    41	        for fromno, tono in arc_possibilities_set:
    42	            dests[fromno].add(tono)

    49	        for fromno, tono in arcs:
    50	            if fromno != tono:
    51	                new_arcs.add((fromno, tono))
    52	            else:
    53	                if fromno in single_dests:
    54	                    new_arcs.add((fromno, single_dests[fromno]))
    55
    56	        arcs_executed_set = file_reporter.translate_arcs(new_arcs)
    57	        exit_counts = file_reporter.exit_counts()
    58	        no_branch = file_reporter.no_branch_lines()
    59	    else:
    60	        arc_possibilities_set = set()
    61	        arcs_executed_set = set()
    62	        exit_counts = {}
    63	        no_branch = set()
    64
    65	    return Analysis(
    66	        precision=precision,
    67	        filename=filename,
    68	        has_arcs=has_arcs,
    69	        statements=statements,
    70	        excluded=excluded,
    71	        executed=executed,

    73	        arcs_executed_set=arcs_executed_set,

   364	def format_lines(
   365	    statements: Iterable[TLineNo],
   366	    lines: Iterable[TLineNo],
   367	    arcs: Iterable[tuple[TLineNo, list[TLineNo]]] | None = None,
   368	) -> str:
   369	    """Nicely format a list of line numbers.
   370
   371	    Format a list of line numbers for printing by coalescing groups of lines as
   372	    long as the lines represent consecutive statements.  This will coalesce
   373	    even if there are gaps between statements.
   374
   375	    For example, if `statements` is [1,2,3,4,5,10,11,12,13,14] and
   376	    `lines` is [1,2,5,10,11,13,14] then the result will be "1-2, 5-11, 13-14".
   377
   378	    Both `lines` and `statements` can be any iterable. All of the elements of
   379	    `lines` must be in `statements`, and all of the values must be positive
   380	    integers.
   381
   382	    If `arcs` is provided, they are (start,[end,end,end]) pairs that will be
   383	    included in the output as long as start isn't in `lines`.
   384
   385	    """
   386	    line_items = [(pair[0], nice_pair(pair)) for pair in _line_ranges(statements, lines)]
   387	    if arcs is not None:
   388	        line_exits = sorted(arcs)
   389	        for line, exits in line_exits:
   390	            for ex in sorted(exits):
   391	                if line not in lines and ex not in lines:
   392	                    dest = (ex if ex > 0 else "exit")
   393	                    line_items.append((line, f"{line}->{dest}"))
   394
   395	    ret = ", ".join(t[-1] for t in sorted(line_items))
   396	    return ret

   124	class CoverageData:
   125	    """Manages collected coverage data, including file storage.
   126
   127	    This class is the public supported API to the data that coverage.py
   128	    collects during program execution.  It includes information about what code
   129	    was executed. It does not include information from the analysis phase, to
   130	    determine what lines could have been executed, or what lines were not
   131	    executed.
   132
   133	    .. note::
   134
   135	        The data file is currently a SQLite database file, with a
   136	        :ref:`documented schema <dbschema>`. The schema is subject to change
   137	        though, so be careful about querying it directly. Use this API if you
   138	        can to isolate yourself from changes.
   139
   140	    There are a number of kinds of data that can be collected:
   141
   142	    * **lines**: the line numbers of source lines that were executed.
   143	      These are always available.
   144
   145	    * **arcs**: pairs of source and destination line numbers for transitions
   146	      between source lines.  These are only available if branch coverage was
   147	      used.
   148
   149	    * **file tracer names**: the module names of the file tracer plugins that
   150	      handled each file in the data.
   151
   152	    Lines, arcs, and file tracer names are stored for each source file. File
   153	    names in this API are case-sensitive, even on platforms with
   154	    case-insensitive file systems.
   155

   157
   158	    A data file is associated with the data when the :class:`CoverageData`
   159	    is created, using the parameters `basename`, `suffix`, and `no_disk`. The
   160	    base name can be queried with :meth:`base_filename`, and the actual file
   161	    name being used is available from :meth:`data_filename`.
   162
   163	    To read an existing coverage.py data file, use :meth:`read`.  You can then
   164	    access the line, arc, or file tracer data with :meth:`lines`, :meth:`arcs`,
   165	    or :meth:`file_tracer`.
   166
   167	    The :meth:`has_arcs` method indicates whether arc data is available.  You
   168	    can get a set of the files in the data with :meth:`measured_files`.  As
   169	    with most Python containers, you can determine if there is any data at all
   170	    by using this object as a boolean value.
   171
   172	    The contexts for each line in a file can be read with
   173	    :meth:`contexts_by_lineno`.

   175	    To limit querying to certain contexts, use :meth:`set_query_context` or
   176	    :meth:`set_query_contexts`. These will narrow the focus of subsequent
   177	    :meth:`lines`, :meth:`arcs`, and :meth:`contexts_by_lineno` calls. The set
   178	    of all measured context names can be retrieved with
   179	    :meth:`measured_contexts`.
   180
   181	    Most data files will be created by coverage.py itself, but you can use
   182	    methods here to create data files if you like.  The :meth:`add_lines`,
   183	    :meth:`add_arcs`, and :meth:`add_file_tracers` methods add data, in ways
   184	    that are convenient for coverage.py.
   185
   186	    To record data for contexts, use :meth:`set_context` to set a context to
   187	    be used for subsequent :meth:`add_lines` and :meth:`add_arcs` calls.
   188

   195	    files can be removed from the database with :meth:`purge_files`.

   198	    :class:`CoverageData`, passing it the other.
   199
   200	    Data in a :class:`CoverageData` can be serialized and deserialized with
   201	    :meth:`dumps` and :meth:`loads`.
   202
   203	    The methods used during the coverage.py collection phase
   204	    (:meth:`add_lines`, :meth:`add_arcs`, :meth:`set_context`, and
   205	    :meth:`add_file_tracers`) are thread-safe.  Other methods may not be.
   206
   207	    """
   208
   209	    def __init__(
   210	        self,
   211	        basename: FilePath | None = None,
   212	        suffix: str | bool | None = None,
   213	        no_disk: bool = False,
   214	        warn: TWarnFn | None = None,
   215	        debug: TDebugCtl | None = None,
   216	    ) -> None:
   217	        """Create a :class:`CoverageData` object to hold coverage-measured data.
   218
   219	        Arguments:
   220	            basename (str): the base name of the data file, defaulting to
   221	                ".coverage". This can be a path to a file in another directory.
   222	            suffix (str or bool): has the same meaning as the `data_suffix`
   223	                argument to :class:`coverage.Coverage`.
   224	            no_disk (bool): if True, keep all data in memory, and don't
   225	                write any disk file.
   226	            warn: a warning callback function, accepting a warning message
   227	                argument.
   228	            debug: a `DebugControl` object (optional)
   229
   230	        """
   231	        self._no_disk = no_disk
   232	        self._basename = os.path.abspath(basename or ".coverage")
   233	        self._suffix = suffix
   234	        self._warn = warn
   235	        self._debug = debug or NoDebugging()
   236
   237	        self._choose_filename()
   238	        # Maps filenames to row ids.
   239	        self._file_map: dict[str, int] = {}
   240	        # Maps thread ids to SqliteDb objects.
   241	        self._dbs: dict[int, SqliteDb] = {}
   242	        self._pid = os.getpid()
   243	        # Synchronize the operations used during collection.
   244	        self._lock = threading.RLock()
   245
   246	        # Are we in sync with the data file?
   247	        self._have_used = False
   248
   249	        self._has_lines = False
   250	        self._has_arcs = False
   251
   252	        self._current_context: str | None = None
   253	        self._current_context_id: int | None = None
   254	        self._query_context_ids: list[int] | None = None
   255
   256	    __repr__ = auto_repr

   258	    def _debug_dataio(self, msg: str, filename: str) -> None:
   259	        """A helper for debug messages which are all similar."""
   260	        if self._debug.should("dataio"):
   261	            self._debug.write(f"{msg} {filename!r} ({file_summary(filename)})")

   263	    def _choose_filename(self) -> None:
   264	        """Set self._filename based on inited attributes."""
   265	        if self._no_disk:
   266	            self._filename = ":memory:"
   267	        else:
   268	            self._filename = self._basename
   269	            suffix = filename_suffix(self._suffix)
   270	            if suffix:
   271	                self._filename += f".{suffix}"

   273	    def _reset(self) -> None:
   274	        """Reset our attributes."""
   275	        if not self._no_disk:
   276	            self.close()
   277	        self._file_map = {}
   278	        self._have_used = False
   279	        self._current_context_id = None

   281	    def close(self, force: bool = False) -> None:
   282	        """Really close all the database objects."""
   283	        if self._debug.should("dataio"):
   284	            self._debug.write(f"Closing dbs, force={force}: {self._dbs}")
   285	        for db in self._dbs.values():
   286	            db.close(force=force)
   287	        self._dbs = {}

   289	    def _open_db(self) -> None:
   290	        """Open an existing db file, and read its metadata."""
   291	        self._debug_dataio("Opening data file", self._filename)
   292	        self._dbs[threading.get_ident()] = SqliteDb(self._filename, self._debug)
   293	        self._read_db()
   294
   295	    def _read_db(self) -> None:
   296	        """Read the metadata from a database so that we are ready to use it."""
   297	        with self._dbs[threading.get_ident()] as db:
   298	            try:
   299	                row = db.execute_one("select version from coverage_schema")
   300	                assert row is not None
   301	            except Exception as exc:
   302	                if "no such table: coverage_schema" in str(exc):
   303	                    self._init_db(db)
   304	                else:
   305	                    raise DataError(
   306	                        "Data file {!r} doesn't seem to be a coverage data file: {}".format(
   307	                            self._filename, exc,
   308	                        ),
   309	                    ) from exc
   310	            else:
   311	                schema_version = row[0]
   312	                if schema_version != SCHEMA_VERSION:
   313	                    raise DataError(
   314	                        "Couldn't use data file {!r}: wrong schema: {} instead of {}".format(
   315	                            self._filename, schema_version, SCHEMA_VERSION,
   316	                        ),
   317	                    )
   318
   319	            row = db.execute_one("select value from meta where key = 'has_arcs'")
   320	            if row is not None:
   321	                self._has_arcs = bool(int(row[0]))
   322	                self._has_lines = not self._has_arcs
   323
   324	            with db.execute("select id, path from file") as cur:
   325	                for file_id, path in cur:
   326	                    self._file_map[path] = file_id
   327
   328	    def _init_db(self, db: SqliteDb) -> None:
   329	        """Write the initial contents of the database."""
   330	        self._debug_dataio("Initing data file", self._filename)
   331	        db.executescript(SCHEMA)
   332	        db.execute_void("insert into coverage_schema (version) values (?)", (SCHEMA_VERSION,))
   333
   334	        # When writing metadata, avoid information that will needlessly change
   335	        # the hash of the data file, unless we're debugging processes.
   336	        meta_data = [
   337	            ("version", __version__),
   338	        ]
   339	        if self._debug.should("process"):
   340	            meta_data.extend([
   341	                ("sys_argv", str(getattr(sys, "argv", None))),
   342	                ("when", datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")),
   343	            ])
   344	        db.executemany_void("insert or ignore into meta (key, value) values (?, ?)", meta_data)
   345
   346	    def _connect(self) -> SqliteDb:
   347	        """Get the SqliteDb object to use."""
   348	        if threading.get_ident() not in self._dbs:
   349	            self._open_db()
   350	        return self._dbs[threading.get_ident()]
   351
   352	    def __bool__(self) -> bool:
   353	        if (threading.get_ident() not in self._dbs and not os.path.exists(self._filename)):
   354	            return False
   355	        try:
   356	            with self._connect() as con:
   357	                with con.execute("select * from file limit 1") as cur:
   358	                    return bool(list(cur))
   359	        except CoverageException:
   360	            return False

   362	    def dumps(self) -> bytes:
   363	        """Serialize the current data to a byte string.
   364
   365	        The format of the serialized data is not documented. It is only
   366	        suitable for use with :meth:`loads` in the same version of
   367	        coverage.py.
   368
   369	        Note that this serialization is not what gets stored in coverage data
   370	        files.  This method is meant to produce bytes that can be transmitted
   371	        elsewhere and then deserialized with :meth:`loads`.
   372
   373	        Returns:
   374	            A byte string of serialized data.
   375
   376	        .. versionadded:: 5.0
   377
   378	        """
   379	        self._debug_dataio("Dumping data from data file", self._filename)
   380	        with self._connect() as con:
   381	            script = con.dump()
   382	            return b"z" + zlib.compress(script.encode("utf-8"))
   383
   384	    def loads(self, data: bytes) -> None:
   385	        """Deserialize data from :meth:`dumps`.
   386
   387	        Use with a newly-created empty :class:`CoverageData` object.  It's
   388	        undefined what happens if the object already has data in it.
   389
   390	        Note that this is not for reading data from a coverage data file.  It
   391	        is only for use on data you produced with :meth:`dumps`.
   392
   393	        Arguments:
   394	            data: A byte string of serialized data produced by :meth:`dumps`.
   395
   396	        .. versionadded:: 5.0
   397
   398	        """
   399	        self._debug_dataio("Loading data into data file", self._filename)
   400	        if data[:1] != b"z":
   401	            raise DataError(
   402	                f"Unrecognized serialization: {data[:40]!r} (head of {len(data)} bytes)",
   403	            )
   404	        script = zlib.decompress(data[1:]).decode("utf-8")
   405	        self._dbs[threading.get_ident()] = db = SqliteDb(self._filename, self._debug)
   406	        with db:
   407	            db.executescript(script)
   408	        self._read_db()
   409	        self._have_used = True

   411	    def _file_id(self, filename: str, add: bool = False) -> int | None:
   412	        """Get the file id for `filename`.
   413
   414	        If filename is not in the database yet, add it if `add` is True.
   415	        If `add` is not True, return None.
   416	        """
   417	        if filename not in self._file_map:
   418	            if add:
   419	                with self._connect() as con:
   420	                    self._file_map[filename] = con.execute_for_rowid(
   421	                        "insert or replace into file (path) values (?)",
   422	                        (filename,),
   423	                    )
   424	        return self._file_map.get(filename)

   426	    def _context_id(self, context: str) -> int | None:
   427	        """Get the id for a context."""
   428	        assert context is not None
   429	        self._start_using()
   430	        with self._connect() as con:
   431	            row = con.execute_one("select id from context where context = ?", (context,))
   432	            if row is not None:
   433	                return cast(int, row[0])
   434	            else:
   435	                return None

   481	    @_locked
   482	    def add_lines(self, line_data: Mapping[str, Collection[TLineNo]]) -> None:
   483	        """Add measured line data.
   484
   485	        `line_data` is a dictionary mapping file names to iterables of ints::
   486
   487	            { filename: { line1, line2, ... }, ...}
   488
   489	        """
   490	        if self._debug.should("dataop"):
   491	            self._debug.write("Adding lines: %d files, %d lines total" % (
   492	                len(line_data), sum(len(lines) for lines in line_data.values()),
   493	            ))
   494	            if self._debug.should("dataop2"):
   495	                for filename, linenos in sorted(line_data.items()):
   496	                    self._debug.write(f"  {filename}: {linenos}")
   497	        self._start_using()
   498	        self._choose_lines_or_arcs(lines=True)
   499	        if not line_data:
   500	            return
   501	        with self._connect() as con:
   502	            self._set_context_id()
   503	            for filename, linenos in line_data.items():
   504	                line_bits = nums_to_numbits(linenos)
   505	                file_id = self._file_id(filename, add=True)
   506	                query = "select numbits from line_bits where file_id = ? and context_id = ?"
   507	                with con.execute(query, (file_id, self._current_context_id)) as cur:
   508	                    existing = list(cur)
   509	                if existing:
   510	                    line_bits = numbits_union(line_bits, existing[0][0])
   511
   512	                con.execute_void(
   513	                    "insert or replace into line_bits " +
   514	                    " (file_id, context_id, numbits) values (?, ?, ?)",
   515	                    (file_id, self._current_context_id, line_bits),
   516	                )

   518	    @_locked
   519	    def add_arcs(self, arc_data: Mapping[str, Collection[TArc]]) -> None:
   520	        """Add measured arc data.
   521
   522	        `arc_data` is a dictionary mapping file names to iterables of pairs of
   523	        ints::
   524
   525	            { filename: { (l1,l2), (l1,l2), ... }, ...}
   526
   527	        """
   528	        if self._debug.should("dataop"):
   529	            self._debug.write("Adding arcs: %d files, %d arcs total" % (
   530	                len(arc_data), sum(len(arcs) for arcs in arc_data.values()),
   531	            ))
   532	            if self._debug.should("dataop2"):
   533	                for filename, arcs in sorted(arc_data.items()):
   534	                    self._debug.write(f"  {filename}: {arcs}")
   535	        self._start_using()
   536	        self._choose_lines_or_arcs(arcs=True)
   537	        if not arc_data:
   538	            return
   539	        with self._connect() as con:
   540	            self._set_context_id()
   541	            for filename, arcs in arc_data.items():
   542	                if not arcs:
   543	                    continue
   544	                file_id = self._file_id(filename, add=True)
   545	                data = [(file_id, self._current_context_id, fromno, tono) for fromno, tono in arcs]
   546	                con.executemany_void(
   547	                    "insert or ignore into arc " +
   548	                    "(file_id, context_id, fromno, tono) values (?, ?, ?, ?)",
   549	                    data,
   550	                )

   552	    def _choose_lines_or_arcs(self, lines: bool = False, arcs: bool = False) -> None:
   553	        """Force the data file to choose between lines and arcs."""
   554	        assert lines or arcs
   555	        assert not (lines and arcs)
   556	        if lines and self._has_arcs:
   557	            if self._debug.should("dataop"):
   558	                self._debug.write("Error: Can't add line measurements to existing branch data")
   559	            raise DataError("Can't add line measurements to existing branch data")
   560	        if arcs and self._has_lines:
   561	            if self._debug.should("dataop"):
   562	                self._debug.write("Error: Can't add branch measurements to existing line data")
   563	            raise DataError("Can't add branch measurements to existing line data")
   564	        if not self._has_arcs and not self._has_lines:
   565	            self._has_lines = lines
   566	            self._has_arcs = arcs
   567	            with self._connect() as con:
   568	                con.execute_void(
   569	                    "insert or ignore into meta (key, value) values (?, ?)",
   570	                    ("has_arcs", str(int(arcs))),
   571	                )

   573	    @_locked
   574	    def add_file_tracers(self, file_tracers: Mapping[str, str]) -> None:
   575	        """Add per-file plugin information.
   576
   577	        `file_tracers` is { filename: plugin_name, ... }
   578
   579	        """
   580	        if self._debug.should("dataop"):
   581	            self._debug.write("Adding file tracers: %d files" % (len(file_tracers),))
   582	        if not file_tracers:
   583	            return
   584	        self._start_using()
   585	        with self._connect() as con:
   586	            for filename, plugin_name in file_tracers.items():
   587	                file_id = self._file_id(filename, add=True)
   588	                existing_plugin = self.file_tracer(filename)
   589	                if existing_plugin:
   590	                    if existing_plugin != plugin_name:
   591	                        raise DataError(
   592	                            "Conflicting file tracer name for '{}': {!r} vs {!r}".format(
   593	                                filename, existing_plugin, plugin_name,
   594	                            ),
   595	                        )
   596	                elif plugin_name:
   597	                    con.execute_void(
   598	                        "insert into tracer (file_id, tracer) values (?, ?)",
   599	                        (file_id, plugin_name),
   600	                    )

   610	    def touch_files(self, filenames: Collection[str], plugin_name: str | None = None) -> None:
   611	        """Ensure that `filenames` appear in the data, empty if needed.
   612
   613	        `plugin_name` is the name of the plugin responsible for these files.
   614	        It is used to associate the right filereporter, etc.
   615	        """
   616	        if self._debug.should("dataop"):
   617	            self._debug.write(f"Touching {filenames!r}")
   618	        self._start_using()
   619	        with self._connect(): # Use this to get one transaction.
   620	            if not self._has_arcs and not self._has_lines:
   621	                raise DataError("Can't touch files in an empty CoverageData")
   622
   623	            for filename in filenames:
   624	                self._file_id(filename, add=True)
   625	                if plugin_name:
   626	                    # Set the tracer for this file
   627	                    self.add_file_tracers({filename: plugin_name})

   629	    def purge_files(self, filenames: Collection[str]) -> None:
   630	        """Purge any existing coverage data for the given `filenames`.
   631
   632	        .. versionadded:: 7.2
   633
   634	        """
   635	        if self._debug.should("dataop"):
   636	            self._debug.write(f"Purging data for {filenames!r}")
   637	        self._start_using()
   638	        with self._connect() as con:
   639
   640	            if self._has_lines:
   641	                sql = "delete from line_bits where file_id=?"
   642	            elif self._has_arcs:
   643	                sql = "delete from arc where file_id=?"
   644	            else:
   645	                raise DataError("Can't purge files in an empty CoverageData")
   646
   647	            for filename in filenames:
   648	                file_id = self._file_id(filename, add=False)
   649	                if file_id is None:
   650	                    continue
   651	                con.execute_void(sql, (file_id,))

   653	    def update(
   654	        self,
   655	        other_data: CoverageData,
   656	        map_path: Callable[[str], str] | None = None,
   657	    ) -> None:
   658	        """Update this data with data from another :class:`CoverageData`.
   659
   660	        If `map_path` is provided, it's a function that re-map paths to match
   661	        the local machine's.  Note: `map_path` is None only when called
   662	        directly from the test suite.
   663
   664	        """
   665	        if self._debug.should("dataop"):
   666	            self._debug.write("Updating with data from {!r}".format(
   667	                getattr(other_data, "_filename", "???"),
   668	            ))
   669	        if self._has_lines and other_data._has_arcs:
   670	            raise DataError("Can't combine branch coverage data with statement data")
   671	        if self._has_arcs and other_data._has_lines:
   672	            raise DataError("Can't combine statement coverage data with branch data")
   673
   674	        map_path = map_path or (lambda p: p)
   675
   676	        # Force the database we're writing to to exist before we start nesting contexts.
   677	        self._start_using()
   678
   679	        # Collector for all arcs, lines and tracers
   680	        other_data.read()
   681	        with other_data._connect() as con:
   682	            # Get files data.
   683	            with con.execute("select path from file") as cur:
   684	                files = {path: map_path(path) for (path,) in cur}
   685
   686	            # Get contexts data.
   687	            with con.execute("select context from context") as cur:
   688	                contexts = [context for (context,) in cur]
   689
   690	            # Get arc data.
   691	            with con.execute(
   692	                "select file.path, context.context, arc.fromno, arc.tono " +
   693	                "from arc " +
   694	                "inner join file on file.id = arc.file_id " +
   695	                "inner join context on context.id = arc.context_id",
   696	            ) as cur:
   697	                arcs = [
   698	                    (files[path], context, fromno, tono)
   699	                    for (path, context, fromno, tono) in cur
   700	                ]
   701
   702	            # Get line data.
   703	            with con.execute(
   704	                "select file.path, context.context, line_bits.numbits " +
   705	                "from line_bits " +
   706	                "inner join file on file.id = line_bits.file_id " +
   707	                "inner join context on context.id = line_bits.context_id",
   708	            ) as cur:
   709	                lines: dict[tuple[str, str], bytes] = {}
   710	                for path, context, numbits in cur:
   711	                    key = (files[path], context)
   712	                    if key in lines:
   713	                        numbits = numbits_union(lines[key], numbits)
   714	                    lines[key] = numbits
   715
   716	            # Get tracer data.
   717	            with con.execute(
   718	                "select file.path, tracer " +
   719	                "from tracer " +
   720	                "inner join file on file.id = tracer.file_id",
   721	            ) as cur:
   722	                tracers = {files[path]: tracer for (path, tracer) in cur}
   723
   724	        with self._connect() as con:
   725	            assert con.con is not None
   726	            con.con.isolation_level = "IMMEDIATE"
   727
   728	            # Get all tracers in the DB. Files not in the tracers are assumed
   729	            # to have an empty string tracer. Since Sqlite does not support
   730	            # full outer joins, we have to make two queries to fill the
   731	            # dictionary.
   732	            with con.execute("select path from file") as cur:
   733	                this_tracers = {path: "" for path, in cur}
   734	            with con.execute(
   735	                "select file.path, tracer from tracer " +
   736	                "inner join file on file.id = tracer.file_id",
   737	            ) as cur:
   738	                this_tracers.update({
   739	                    map_path(path): tracer
   740	                    for path, tracer in cur
   741	                })
   742
   743	            # Create all file and context rows in the DB.
   744	            con.executemany_void(
   745	                "insert or ignore into file (path) values (?)",
   746	                ((file,) for file in files.values()),
   747	            )
   748	            with con.execute("select id, path from file") as cur:
   749	                file_ids = {path: id for id, path in cur}
   750	            self._file_map.update(file_ids)
   751	            con.executemany_void(
   752	                "insert or ignore into context (context) values (?)",
   753	                ((context,) for context in contexts),
   754	            )
   755	            with con.execute("select id, context from context") as cur:
   756	                context_ids = {context: id for id, context in cur}
