   574	        firstDay = (year * 365) + int((year - 1) / 4) - 693596

   192	                print("self.OE (hex)  = %s" % hexText(self.OE))
   193
   194	            # Compute permissions array
   195	            permsarr = [
   196	                self.P       & 0xFF, # store the permission value in the first 32-bits
   197	                self.P >> 8  & 0xFF,
   198	                self.P >> 16 & 0xFF,
   199	                self.P >> 24 & 0xFF,
   200	                0xFF,
   201	                0xFF,
   202	                0xFF,
   203	                0xFF,
   204	                ord('T'),             # 'T' if EncryptMetaData is True (default), 'F' otherwise
   205	                ord('a'),             # a, d, b are magic values
   206	                ord('d'),
   207	                ord('b'),
   208	                0x01,                   # trailing zeros will be ignored
   209	                0x01,
   210	                0x01,
   211	                0x01
   212	            ]
   213
   214	            # the permission array should be enrypted in the Perms field
   215	            encrypter = pyaes.Encrypter(pyaes.AESModeOfOperationCBC(self.key, iv=iv))
   216	            self.Perms = encrypter.feed(bytes(permsarr))
   217	            self.Perms += encrypter.feed()
   218
   219	            if DEBUG:
   220	                print("self.Perms (hex)  = %s" % hexText(self.Perms))

   226	                print("self.O (as hex) = %s" % hexText(self.O))

   228	            #print "\nself.O", self.O, repr(self.O)

   230	            if DEBUG:
   231	                print("self.key (hex)  = %s" % hexText(self.key))
   232	            self.U = computeU(self.key, revision=self.revision, documentId=internalID)
   233	            if DEBUG:
   234	                print("self.U (as hex) = %s" % hexText(self.U))

   236	        self.objnum = self.version = None
   237	        self.prepared = 1
   238	    def register(self, objnum, version):
   239	        # enter a new direct object
   240	        if not self.prepared:
   241	            raise ValueError("encryption not prepared!")
   242	        self.objnum = objnum
   243	        self.version = version

   246	        if not self.prepared:
   247	            raise ValueError("encryption not prepared!")
   248	        return StandardEncryptionDictionary(O=self.O, OE=self.OE, U=self.U, UE=self.UE, P=self.P, Perms=self.Perms, revision=self.revision)

   327	def encryptionkey(password, OwnerKey, Permissions, FileId1, revision=None):
   328	    revision = checkRevision(revision)
   329	    # FileId1 is first string of the fileid array
   330	    # add padding string
   331	    #AR force same as iText example
   332	    #Permissions =  -1836   #int(Permissions - 2**31)
   333	    password = asBytes(password) + PadString
   334	    # truncate to 32 bytes
   335	    password = password[:32]
   336	    # translate permissions to string, low order byte first
   337	    p = Permissions# + 2**32L
   338	    permissionsString = b""
   339	    for i in range(4):
   340	        byte = (p & 0xff)    # seems to match what iText does
   341	        p = p>>8
   342	        permissionsString += int2Byte(byte % 256)
   343
   344	    hash = md5(asBytes(password))
   345	    hash.update(asBytes(OwnerKey))
   346	    hash.update(asBytes(permissionsString))
   347	    hash.update(asBytes(FileId1))
   348
   349	    md5output = hash.digest()
   350
   351	    if revision==2:
   352	        key = md5output[:5]
   353	    elif revision==3:  #revision 3 algorithm - loop 50 times
   354	        for x in range(50):
   355	            md5output = md5(md5output).digest()
   356	        key = md5output[:16]
   357	    if DEBUG: print('encryptionkey(%s,%s,%s,%s,%s)==>%s' % tuple([hexText(str(x)) for x in (password, OwnerKey, Permissions, FileId1, revision, key)]))
   358	    return key

   360	def computeO(userPassword, ownerPassword, revision):
   361	    from reportlab.lib.arciv import ArcIV
   362	    #print 'digest of hello is %s' % md5('hello').digest()
   363	    assert revision in (2,3), 'Unknown algorithm revision %s' % revision
   364	    if not ownerPassword:
   365	        ownerPassword = userPassword
   366
   367	    ownerPad = asBytes(ownerPassword) + PadString
   368	    ownerPad = ownerPad[0:32]
   369
   370	    password = asBytes(userPassword) + PadString
   371	    userPad = password[:32]
   372
   373	    digest = md5(ownerPad).digest()
   374	    if DEBUG: print('PadString=%s\nownerPad=%s\npassword=%s\nuserPad=%s\ndigest=%s\nrevision=%s' % (ascii(PadString),ascii(ownerPad),ascii(password),ascii(userPad),ascii(digest),revision))
   375	    if revision == 2:
   376	        O = ArcIV(digest[:5]).encode(userPad)
   377	    elif revision == 3:
   378	        for i in range(50):
   379	            digest = md5(digest).digest()
   380	        digest = digest[:16]
   381	        O = userPad
   382	        for i in range(20):
   383	            thisKey = xorKey(i, digest)
   384	            O = ArcIV(thisKey).encode(O)
   385	    if DEBUG: print('computeO(%s,%s,%s)==>%s' % tuple([hexText(str(x)) for x in (userPassword, ownerPassword, revision,O)]))
   386	    return O

   388	def computeU(encryptionkey, encodestring=PadString,revision=None,documentId=None):
   389	    revision = checkRevision(revision)
   390	    from reportlab.lib.arciv import ArcIV
   391	    if revision == 2:
   392	        result = ArcIV(encryptionkey).encode(encodestring)
   393	    elif revision == 3:
   394	        assert documentId is not None, "Revision 3 algorithm needs the document ID!"
   395	        h = md5(PadString)
   396	        h.update(rawBytes(documentId))
   397	        tmp = h.digest()
   398	        tmp = ArcIV(encryptionkey).encode(tmp)
   399	        for n in range(1,20):
   400	            thisKey = xorKey(n, encryptionkey)
   401	            tmp = ArcIV(thisKey).encode(tmp)
   402	        while len(tmp) < 32:
   403	            tmp += b'\0'
   404	        result = tmp
   405	    if DEBUG: print('computeU(%s,%s,%s,%s)==>%s' % tuple([hexText(str(x)) for x in (encryptionkey, encodestring,revision,documentId,result)]))
   406	    return result

   416	def encodePDF(key, objectNumber, generationNumber, string, revision=None):
   417	    "Encodes a string or stream"
   418	    revision = checkRevision(revision)
   419	    #print 'encodePDF (%s, %d, %d, %s)' % (hexText(key), objectNumber, generationNumber, string)
   420	    # extend 3 bytes of the object Number, low byte first
   421	    if revision in (2,3):
   422	        newkey = key
   423	        n = objectNumber
   424	        for i in range(3):
   425	            newkey += int2Byte(n & 0xff)
   426	            n = n>>8
   427	        # extend 2 bytes of the generationNumber
   428	        n = generationNumber
   429	        for i in range(2):
   430	            newkey += int2Byte(n & 0xff)
   431	            n = n>>8
   432	        md5output = md5(newkey).digest()
   433	        if revision == 2:
   434	            key = md5output[:10]
   435	        elif revision == 3:
   436	            key = md5output #all 16 bytes
   437	        from reportlab.lib.arciv import ArcIV
   438	        encrypted = ArcIV(key).encode(string)
   439	        #print 'encrypted=', hexText(encrypted)
   440	    elif revision == 5:
   441	        iv = os_urandom(16)
   442	        encrypter = pyaes.Encrypter(pyaes.AESModeOfOperationCBC(key, iv=iv))
   443
   444	        # pkcs7 style padding so that the size of the encrypted block is multiple of 16
   445	        string_len = len(string)
   446	        padding = ""
   447	        padding_len = (16 - (string_len % 16)) if string_len > 16 else (16 - string_len)
   448	        if padding_len > 0:
   449	            padding = chr(padding_len) * padding_len
   450
   451	        if isinstance(string, str):
   452	            string = (string + padding).encode("utf-8")
   453	        else:
   454	            string += asBytes(padding)
   455
   456	        encrypted = iv + encrypter.feed(string)
   457	        encrypted += encrypter.feed()
   458
   459	    if DEBUG: print('encodePDF(%s,%s,%s,%s,%s)==>%s' % tuple([hexText(str(x)) for x in (key, objectNumber, generationNumber, string, revision,encrypted)]))
   460	    return encrypted

     3	Platypus has an 'XPreformatted' flowable for handling preformatted

    12	import re
    13	from io import StringIO

    15	def _2xpre(s,styles):
    16	    "Helper to transform Pygments HTML output to ReportLab markup"
    17	    s = s.replace('<div class="highlight">','')
    18	    s = s.replace('</div>','')
    19	    s = s.replace('<pre>','')
    20	    s = s.replace('</pre>','')
    21	    for k,c in styles+[('p','#000000'),('n','#000000'),('err','#000000')]:
    22	        s = s.replace('<span class="%s">' % k,'<span color="%s">' % c)
    23	        s = re.sub(r'<span class="%s\s+.*">'% k,'<span color="%s">' % c,s)
    24	    s = re.sub(r'<span class=".*">','<span color="#0f0f0f">',s)
    25	    return s

    27	def pygments2xpre(s, language="python"):
    28	    "Return markup suitable for XPreformatted"
    29	    try:
    30	        from pygments import highlight
    31	        from pygments.formatters import HtmlFormatter
    32	    except ImportError:
    33	        return s
    34
    35	    from pygments.lexers import get_lexer_by_name
    36	    rconv = lambda x: x
    37	    out = StringIO()
    38
    39	    l = get_lexer_by_name(language)
    40
    41	    h = HtmlFormatter()
    42	    highlight(s,l,h,out)
    43	    styles = [(cls, style.split(';')[0].split(':')[1].strip())
    44	                for cls, (style, ttype, level) in h.class2style.items()
    45	                if cls and style and style.startswith('color:')]
    46	    return rconv(_2xpre(out.getvalue(),styles))
