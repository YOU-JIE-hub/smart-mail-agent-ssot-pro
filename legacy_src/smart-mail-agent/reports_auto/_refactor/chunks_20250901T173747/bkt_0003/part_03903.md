    74	def _usConv(s, vMap, default=None):
    75	    '''convert a strike/underline distance to a number'''
    76	    if isStr(s):
    77	        s = s.strip()
    78	        if s:
    79	            m = _re_us_value.match(s)
    80	            if m:
    81	                return float(m.group(1))*vMap[m.group(2)]
    82	            else:
    83	                return _parser_num(s,allowRelative=False)
    84	        elif default:
    85	            return default
    86	    return s

   221	def imgVRange(h,va,fontSize):
   222	    '''return bottom,top offsets relative to baseline(0)'''
   223	    if va=='baseline':
   224	        iyo = 0
   225	    elif va in ('text-top','top'):
   226	        iyo = fontSize-h
   227	    elif va=='middle':
   228	        iyo = fontSize - (1.2*fontSize+h)*0.5
   229	    elif va in ('text-bottom','bottom'):
   230	        iyo = fontSize - 1.2*fontSize
   231	    elif va=='super':
   232	        iyo = 0.5*fontSize
   233	    elif va=='sub':
   234	        iyo = -0.5*fontSize
   235	    elif hasattr(va,'normalizedValue'):
   236	        iyo = va.normalizedValue(fontSize)
   237	    else:
   238	        iyo = va
   239	    return iyo,iyo+h

   267	def _putFragLine(cur_x, tx, line, last, pKind):
   268	    preformatted = tx.preformatted
   269	    xs = tx.XtraState
   270	    cur_y = xs.cur_y
   271	    x0 = tx._x0
   272	    autoLeading = xs.autoLeading
   273	    leading = xs.leading
   274	    cur_x += xs.leftIndent
   275	    dal = autoLeading in ('min','max')
   276	    if dal:
   277	        if autoLeading=='max':
   278	            ascent = max(_56*leading,line.ascent)
   279	            descent = max(_16*leading,-line.descent)
   280	        else:
   281	            ascent = line.ascent
   282	            descent = -line.descent
   283	        leading = ascent+descent
   284	    if tx._leading!=leading:
   285	        tx.setLeading(leading)
   286	    if dal:
   287	        olb = tx._olb
   288	        if olb is not None:
   289	            xcy = olb-ascent
   290	            if tx._oleading!=leading:
   291	                cur_y += leading - tx._oleading
   292	            if abs(xcy-cur_y)>1e-8:
   293	                cur_y = xcy
   294	                tx.setTextOrigin(x0,cur_y)
   295	                xs.cur_y = cur_y
   296	        tx._olb = cur_y - descent
   297	        tx._oleading = leading
   298	    ws = getattr(tx,'_wordSpace',0)
   299	    nSpaces = 0
   300	    words = line.words
   301	    AL = []
   302	    LL = []
   303	    us_lines = xs.us_lines
   304	    links = xs.links
   305	    for i, f in enumerate(words):
   306	        if hasattr(f,'cbDefn'):
   307	            cbDefn = f.cbDefn
   308	            kind = cbDefn.kind
   309	            if kind=='img':
   310	                #draw image cbDefn,cur_y,cur_x
   311	                txfs = tx._fontsize
   312	                if txfs is None:
   313	                    txfs = xs.style.fontSize
   314	                w = imgNormV(cbDefn.width,xs.paraWidth)
   315	                h = imgNormV(cbDefn.height,txfs)
   316	                iy0,iy1 = imgVRange(h,cbDefn.valign,txfs)
   317	                cur_x_s = cur_x + nSpaces*ws
   318	                tx._canvas.drawImage(cbDefn.image,cur_x_s,cur_y+iy0,w,h,mask='auto')
   319	                cur_x += w
   320	                cur_x_s += w
   321	                setXPos(tx,cur_x_s-tx._x0)
   322	            else:
   323	                name = cbDefn.name
   324	                if kind=='anchor':
   325	                    tx._canvas.bookmarkHorizontal(name,cur_x,cur_y+leading)
   326	                else:
   327	                    func = getattr(tx._canvas,name,None)
   328	                    if not func:
   329	                        raise AttributeError("Missing %s callback attribute '%s'" % (kind,name))
   330	                    tx._canvas._curr_tx_info=dict(tx=tx,cur_x=cur_x,cur_y=cur_y,leading=leading,xs=tx.XtraState)
   331	                    try:
   332	                        func(tx._canvas,kind,getattr(cbDefn,'label',None))
   333	                    finally:
   334	                        del tx._canvas._curr_tx_info
   335	            if f is words[-1]:
   336	                if not tx._fontname:
   337	                    tx.setFont(xs.style.fontName,xs.style.fontSize)
   338	                tx._textOut('',1)
   339	        else:
   340	            cur_x_s = cur_x + nSpaces*ws
   341	            end_x = cur_x_s
   342	            fontSize = f.fontSize
   343	            textColor = f.textColor
   344	            rise = f.rise
   345	            if i > 0:
   346	                end_x = cur_x_s - (0 if preformatted else _trailingSpaceLength(words[i-1].text, tx))
   347	            if (tx._fontname,tx._fontsize)!=(f.fontName,fontSize):
   348	                tx._setFont(f.fontName, fontSize)
   349	            if xs.textColor!=textColor:
   350	                xs.textColor = textColor
   351	                tx.setFillColor(textColor)
   352	            if xs.rise!=rise:
   353	                xs.rise=rise
   354	                tx.setRise(rise)
   355	            text = f.text
   356	            tx._textOut(text,f is words[-1])    # cheap textOut
   357	            if LL != f.us_lines:
   358	                S = set(LL)
   359	                NS = set(f.us_lines)
   360	                nL = NS - S #new lines
   361	                eL = S - NS #ending lines
   362	                for l in eL:
   363	                    us_lines[l] = us_lines[l],end_x
   364	                for l in nL:
   365	                    us_lines[l] = (l,fontSize,textColor,cur_x_s),fontSize
   366	                LL = f.us_lines
   367	            if LL:
   368	                for l in LL:
   369	                    l0, fsmax = us_lines[l]
   370	                    if fontSize>fsmax:
   371	                        us_lines[l] = l0, fontSize
   372
   373	            nlo = rise - 0.2*fontSize
   374	            nhi = rise + fontSize
   375	            if AL != f.link:
   376	                S = set(AL)
   377	                NS = set(f.link)
   378	                nL = NS - S #new linkis
   379	                eL = S - NS #ending links
   380	                for l in eL:
   381	                    links[l] = links[l],end_x
   382	                for l in nL:
   383	                    links[l] = (l,cur_x),nlo,nhi
   384	                AL = f.link
   385	            if AL:
   386	                for l in AL:
   387	                    l0, lo, hi = links[l]
   388	                    if nlo<lo or nhi>hi:
   389	                        links[l] = l0,min(nlo,lo),max(nhi,hi)
   390
   391	            bg = getattr(f,'backColor',None)
   392	            if bg and not xs.backColor:
   393	                xs.backColor = bg
   394	                xs.backColor_x = cur_x_s
   395	            elif xs.backColor:
   396	                if not bg:
   397	                    xs.backColors.append( (xs.backColor_x, end_x, xs.backColor) )
   398	                    xs.backColor = None
   399	                elif f.backColor!=xs.backColor or xs.textColor!=xs.backColor:
   400	                    xs.backColors.append( (xs.backColor_x, end_x, xs.backColor) )
   401	                    xs.backColor = bg
   402	                    xs.backColor_x = cur_x_s
   403	            txtlen = tx._canvas.stringWidth(text, tx._fontname, tx._fontsize)
   404	            cur_x += txtlen
   405	            nSpaces += text.count(' ')+_nbspCount(text)
   406
   407	    cur_x_s = cur_x+(nSpaces-1)*ws
   408	    if last and xs.style.endDots:
   409	        if xs.style.wordWrap!='RTL':    #assume dots left --> right
   410	            if pKind!='right':
   411	                _do_dots_frag(cur_x,cur_x_s,line.maxWidth,xs,tx)
   412	        elif pKind!='left':
   413	            start = tx._x_offset
   414	            _do_dots_frag(start, start, x0 - start, xs, tx, left=False)
   415
   416	    if LL:
   417	        for l in LL:
   418	            us_lines[l] = us_lines[l], cur_x_s
   419
   420	    if AL:
   421	        for l in AL:
   422	            links[l] = links[l], cur_x_s
   423
   424	    if xs.backColor:
   425	        xs.backColors.append( (xs.backColor_x, cur_x_s, xs.backColor) )
   426	    if tx._x0!=x0:
   427	        setXPos(tx,x0-tx._x0)

   429	def _do_dots_frag(cur_x, cur_x_s, maxWidth, xs, tx, left=True):
   430	    text,fontName,fontSize,textColor,backColor,dy = _getDotsInfo(xs.style)
   431	    txtlen = tx._canvas.stringWidth(text, fontName, fontSize)
   432	    if cur_x_s+txtlen<=maxWidth:
   433	        if tx._fontname!=fontName or tx._fontsize!=fontSize:
   434	            tx.setFont(fontName,fontSize)
   435	        if left: maxWidth += getattr(tx,'_dotsOffsetX',tx._x0)
   436	        tx.setTextOrigin(0,xs.cur_y+dy)
   437	        setXPos(tx,cur_x_s-cur_x)
   438	        n = int((maxWidth-cur_x_s)/txtlen)
   439	        setXPos(tx,maxWidth - txtlen*n)
   440	        if xs.textColor!=textColor:
   441	            tx.setFillColor(textColor)
   442	        if backColor: xs.backColors.append((cur_x,maxWidth,backColor))
   443	        tx._textOut(n*text,1)
   444	        if dy: tx.setTextOrigin(tx._x0,xs.cur_y-dy)

   591	        ww = self[0]

   968	        pfx = m.group(0)

   970	    else:

   973	        sfx = u'-'
   974	        s = s[:-1]
   975	    else:
   976	        m = _hy_sfx_pat.search(s)
   977	        if m:
   978	            sfx = m.group(0)
   979	            s = s[:-len(sfx)]
   980	        else:
   981	            sfx = u''
   982	    if len(s) < hymwl: return
   983
   984	    w0 = newWidth - ww
   985	    R = _uri_split_pairs(s)

   987	        #a uri match was seen
   988	        if ww>maxWidth or (uriWasteReduce and w0 <= (1-uriWasteReduce)*maxWidth):

  1642	def cjkFragSplit(frags, maxWidths, calcBounds, encoding='utf8'):
  1643	    '''This attempts to be wordSplit for frags using the dumb algorithm'''
  1644	    U = []  #get a list of single glyphs with their widths etc etc
  1645	    for f in frags:
  1646	        text = f.text
  1647	        if isBytes(text):
  1648	            text = text.decode(encoding)
  1649	        if text:
  1650	            U.extend([cjkU(t,f,encoding) for t in text])
  1651	        else:
  1652	            U.append(cjkU(text,f,encoding))
  1653	    lines = []
  1654	    i = widthUsed = lineStartPos = 0
  1655	    maxWidth = maxWidths[0]
  1656	    nU = len(U)
  1657	    while i<nU:
  1658	        u = U[i]
  1659	        i += 1
  1660	        w = u.width
  1661	        if hasattr(w,'normalizedValue'):
  1662	            w._normalizer = maxWidth
  1663	            w = w.normalizedValue(maxWidth)
  1664	        widthUsed += w
  1665	        lineBreak = hasattr(u.frag,'lineBreak')
  1666	        endLine = (widthUsed>maxWidth + _FUZZ and widthUsed>0) or lineBreak
  1667	        if endLine:
  1668	            extraSpace = maxWidth - widthUsed
  1669	            if not lineBreak:
  1670	                if ord(u)<0x3000:
  1671	                    # we appear to be inside a non-Asian script section.
  1672	                    # (this is a very crude test but quick to compute).
  1673	                    # This is likely to be quite rare so the speed of the
  1674	                    # code below is hopefully not a big issue.  The main
  1675	                    # situation requiring this is that a document title
  1676	                    # with an english product name in it got cut.
  1677
  1678
  1679	                    # we count back and look for
  1680	                    #  - a space-like character
  1681	                    #  - reversion to Kanji (which would be a good split point)
  1682	                    #  - in the worst case, roughly half way back along the line
  1683	                    limitCheck = (lineStartPos+i)>>1        #(arbitrary taste issue)
  1684	                    for j in range(i-1,limitCheck,-1):
  1685	                        uj = U[j]
  1686	                        if uj and category(uj)=='Zs' or ord(uj)>=0x3000:
  1687	                            k = j+1
  1688	                            if k<i:
  1689	                                j = k+1
  1690	                                extraSpace += sum(U[ii].width for ii in range(j,i))
  1691	                                w = U[k].width
  1692	                                u = U[k]
  1693	                                i = j
  1694	                                break
  1695
  1696	                #we are pushing this character back, but
  1697	                #the most important of the Japanese typography rules
  1698	                #if this character cannot start a line, wrap it up to this line so it hangs
  1699	                #in the right margin. We won't do two or more though - that's unlikely and
  1700	                #would result in growing ugliness.
  1701	                #and increase the extra space
  1702	                #bug fix contributed by Alexander Vasilenko <alexs.vasilenko@gmail.com>
  1703	                if u not in ALL_CANNOT_START and i>lineStartPos+1:
  1704	                    #otherwise we need to push the character back
  1705	                    #the i>lineStart+1 condition ensures progress
  1706	                    i -= 1
  1707	                    extraSpace += w
  1708	            lines.append(makeCJKParaLine(U[lineStartPos:i],maxWidth,widthUsed,extraSpace,lineBreak,calcBounds))
  1709	            try:
  1710	                maxWidth = maxWidths[len(lines)]
  1711	            except IndexError:
  1712	                maxWidth = maxWidths[-1]  # use the last one
  1713
  1714	            lineStartPos = i
  1715	            widthUsed = 0
  1716
  1717	    #any characters left?
  1718	    if widthUsed > 0:
  1719	        lines.append(makeCJKParaLine(U[lineStartPos:],maxWidth,widthUsed,maxWidth-widthUsed,False,calcBounds))
