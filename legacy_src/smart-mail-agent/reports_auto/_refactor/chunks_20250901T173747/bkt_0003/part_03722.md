   612	class TupleNameVisitor(RTypeVisitor[str]):
   613	    """Produce a tuple name based on the concrete representations of types."""
   614
   615	    def visit_rinstance(self, t: RInstance) -> str:
   616	        return "O"
   617
   618	    def visit_runion(self, t: RUnion) -> str:
   619	        return "O"
   620
   621	    def visit_rprimitive(self, t: RPrimitive) -> str:
   622	        if t._ctype == "CPyTagged":
   623	            return "I"
   624	        elif t._ctype == "char":
   625	            return "C"
   626	        elif t._ctype == "int64_t":
   627	            return "8"  # "8 byte integer"
   628	        elif t._ctype == "int32_t":
   629	            return "4"  # "4 byte integer"
   630	        elif t._ctype == "int16_t":
   631	            return "2"  # "2 byte integer"
   632	        elif t._ctype == "uint8_t":
   633	            return "U1"  # "1 byte unsigned integer"
   634	        elif t._ctype == "double":
   635	            return "F"
   636	        assert not t.is_unboxed, f"{t} unexpected unboxed type"
   637	        return "O"
   638
   639	    def visit_rtuple(self, t: RTuple) -> str:
   640	        parts = [elem.accept(self) for elem in t.types]
   641	        return "T{}{}".format(len(parts), "".join(parts))
   642
   643	    def visit_rstruct(self, t: RStruct) -> str:
   644	        assert False, "RStruct not supported in tuple"
   645
   646	    def visit_rarray(self, t: RArray) -> str:
   647	        assert False, "RArray not supported in tuple"
   648
   649	    def visit_rvoid(self, t: RVoid) -> str:
   650	        assert False, "rvoid in tuple?"

   653	class RTuple(RType):
   654	    """Fixed-length unboxed tuple (represented as a C struct).
   655
   656	    These are used to represent mypy TupleType values (fixed-length
   657	    Python tuples). Since this is unboxed, the identity of a tuple
   658	    object is not preserved within compiled code. If the identity of a
   659	    tuple is important, or there is a need to have multiple references
   660	    to a single tuple object, a variable-length tuple should be used
   661	    (tuple_rprimitive or Tuple[T, ...]  with explicit '...'), as they
   662	    are boxed.
   663
   664	    These aren't immutable. However, user code won't be able to mutate
   665	    individual tuple items.
   666	    """
   667
   668	    is_unboxed = True
   669
   670	    def __init__(self, types: list[RType]) -> None:
   671	        self.name = "tuple"
   672	        self.types = tuple(types)
   673	        self.is_refcounted = any(t.is_refcounted for t in self.types)
   674	        # Generate a unique id which is used in naming corresponding C identifiers.
   675	        # This is necessary since C does not have anonymous structural type equivalence
   676	        # in the same way python can just assign a Tuple[int, bool] to a Tuple[int, bool].
   677	        self.unique_id = self.accept(TupleNameVisitor())
   678	        # Nominally the max c length is 31 chars, but I'm not honestly worried about this.
   679	        self.struct_name = f"tuple_{self.unique_id}"
   680	        self._ctype = f"{self.struct_name}"
   681	        self.error_overlap = all(t.error_overlap for t in self.types) and bool(self.types)
   682
   683	    def accept(self, visitor: RTypeVisitor[T]) -> T:
   684	        return visitor.visit_rtuple(self)
   685
   686	    @property
   687	    def may_be_immortal(self) -> bool:
   688	        return False
   689
   690	    def __str__(self) -> str:
   691	        return "tuple[%s]" % ", ".join(str(typ) for typ in self.types)
   692
   693	    def __repr__(self) -> str:
   694	        return "<RTuple %s>" % ", ".join(repr(typ) for typ in self.types)
   695
   696	    def __eq__(self, other: object) -> bool:
   697	        return isinstance(other, RTuple) and self.types == other.types
   698
   699	    def __hash__(self) -> int:
   700	        return hash((self.name, self.types))
   701
   702	    def serialize(self) -> JsonDict:
   703	        types = [x.serialize() for x in self.types]
   704	        return {".class": "RTuple", "types": types}
   705
   706	    @classmethod
   707	    def deserialize(cls, data: JsonDict, ctx: DeserMaps) -> RTuple:
   708	        types = [deserialize_type(t, ctx) for t in data["types"]]
   709	        return RTuple(types)

   724	def compute_rtype_alignment(typ: RType) -> int:
   725	    """Compute alignment of a given type based on platform alignment rule"""
   726	    platform_alignment = PLATFORM_SIZE
   727	    if isinstance(typ, RPrimitive):
   728	        return typ.size
   729	    elif isinstance(typ, RInstance):
   730	        return platform_alignment
   731	    elif isinstance(typ, RUnion):
   732	        return platform_alignment
   733	    elif isinstance(typ, RArray):
   734	        return compute_rtype_alignment(typ.item_type)
   735	    else:
   736	        if isinstance(typ, RTuple):
   737	            items = list(typ.types)
   738	        elif isinstance(typ, RStruct):
   739	            items = typ.types
   740	        else:
   741	            assert False, "invalid rtype for computing alignment"
   742	        max_alignment = max(compute_rtype_alignment(item) for item in items)
   743	        return max_alignment
   744
   745
   746	def compute_rtype_size(typ: RType) -> int:
   747	    """Compute unaligned size of rtype"""
   748	    if isinstance(typ, RPrimitive):
   749	        return typ.size
   750	    elif isinstance(typ, RTuple):
   751	        return compute_aligned_offsets_and_size(list(typ.types))[1]
   752	    elif isinstance(typ, RUnion):
   753	        return PLATFORM_SIZE
   754	    elif isinstance(typ, RStruct):
   755	        return compute_aligned_offsets_and_size(typ.types)[1]
   756	    elif isinstance(typ, RInstance):
   757	        return PLATFORM_SIZE
   758	    elif isinstance(typ, RArray):
   759	        alignment = compute_rtype_alignment(typ)
   760	        aligned_size = (compute_rtype_size(typ.item_type) + (alignment - 1)) & ~(alignment - 1)
   761	        return aligned_size * typ.length
   762	    else:
   763	        assert False, "invalid rtype for computing size"
   764
   765
   766	def compute_aligned_offsets_and_size(types: list[RType]) -> tuple[list[int], int]:
   767	    """Compute offsets and total size of a list of types after alignment
   768
   769	    Note that the types argument are types of values that are stored
   770	    sequentially with platform default alignment.
   771	    """
   772	    unaligned_sizes = [compute_rtype_size(typ) for typ in types]
   773	    alignments = [compute_rtype_alignment(typ) for typ in types]
   774
   775	    current_offset = 0
   776	    offsets = []
   777	    final_size = 0
   778	    for i in range(len(unaligned_sizes)):
   779	        offsets.append(current_offset)
   780	        if i + 1 < len(unaligned_sizes):
   781	            cur_size = unaligned_sizes[i]
   782	            current_offset += cur_size
   783	            next_alignment = alignments[i + 1]
   784	            # compute aligned offset,
   785	            # check https://en.wikipedia.org/wiki/Data_structure_alignment for more information
   786	            current_offset = (current_offset + (next_alignment - 1)) & -next_alignment
   787	        else:
   788	            struct_alignment = max(alignments)
   789	            final_size = current_offset + unaligned_sizes[i]
   790	            final_size = (final_size + (struct_alignment - 1)) & -struct_alignment
   791	    return offsets, final_size

   907	class RUnion(RType):
   908	    """union[x, ..., y]"""
   909
   910	    is_unboxed = False
   911
   912	    def __init__(self, items: list[RType]) -> None:
   913	        self.name = "union"
   914	        self.items = items
   915	        self.items_set = frozenset(items)
   916	        self._ctype = "PyObject *"
   917
   918	    @staticmethod
   919	    def make_simplified_union(items: list[RType]) -> RType:
   920	        """Return a normalized union that covers the given items.
   921
   922	        Flatten nested unions and remove duplicate items.
   923
   924	        Overlapping items are *not* simplified. For example,
   925	        [object, str] will not be simplified.
   926	        """
   927	        items = flatten_nested_unions(items)
   928	        assert items
   929
   930	        unique_items = dict.fromkeys(items)
   931	        if len(unique_items) > 1:
   932	            return RUnion(list(unique_items))
   933	        else:
   934	            return next(iter(unique_items))
   935
   936	    def accept(self, visitor: RTypeVisitor[T]) -> T:
   937	        return visitor.visit_runion(self)
   938
   939	    @property
   940	    def may_be_immortal(self) -> bool:
   941	        return any(item.may_be_immortal for item in self.items)
   942
   943	    def __repr__(self) -> str:
   944	        return "<RUnion %s>" % ", ".join(str(item) for item in self.items)
   945
   946	    def __str__(self) -> str:
   947	        return "union[%s]" % ", ".join(str(item) for item in self.items)
   948
   949	    # We compare based on the set because order in a union doesn't matter
   950	    def __eq__(self, other: object) -> bool:
   951	        return isinstance(other, RUnion) and self.items_set == other.items_set
   952
   953	    def __hash__(self) -> int:
   954	        return hash(("union", self.items_set))

   956	    def serialize(self) -> JsonDict:
   957	        types = [x.serialize() for x in self.items]
   958	        return {".class": "RUnion", "types": types}
   959
   960	    @classmethod
   961	    def deserialize(cls, data: JsonDict, ctx: DeserMaps) -> RUnion:
   962	        types = [deserialize_type(t, ctx) for t in data["types"]]
   963	        return RUnion(types)

   966	def flatten_nested_unions(types: list[RType]) -> list[RType]:
   967	    if not any(isinstance(t, RUnion) for t in types):
   968	        return types  # Fast path
   969
   970	    flat_items: list[RType] = []
   971	    for t in types:
   972	        if isinstance(t, RUnion):
   973	            flat_items.extend(flatten_nested_unions(t.items))
   974	        else:
   975	            flat_items.append(t)
   976	    return flat_items

    29	def process_conditional(
    30	    self: IRBuilder, e: Expression, true: BasicBlock, false: BasicBlock
    31	) -> None:
    32	    if isinstance(e, OpExpr) and e.op in ["and", "or"]:
    33	        if e.op == "and":
    34	            # Short circuit 'and' in a conditional context.
    35	            new = BasicBlock()
    36	            process_conditional(self, e.left, new, false)
    37	            self.activate_block(new)
    38	            process_conditional(self, e.right, true, false)

    40	            # Short circuit 'or' in a conditional context.
    41	            new = BasicBlock()
    42	            process_conditional(self, e.left, true, new)
    43	            self.activate_block(new)
    44	            process_conditional(self, e.right, true, false)

    46	        process_conditional(self, e.expr, false, true)
    47	    else:
    48	        res = maybe_process_conditional_comparison(self, e, true, false)

    52	        reg = self.accept(e)

    56	def maybe_process_conditional_comparison(
    57	    self: IRBuilder, e: Expression, true: BasicBlock, false: BasicBlock
    58	) -> bool:
    59	    """Transform simple tagged integer comparisons in a conditional context.
    60
    61	    Return True if the operation is supported (and was transformed). Otherwise,
    62	    do nothing and return False.
    63
    64	    Args:
    65	        self: IR form Builder
    66	        e: Arbitrary expression
    67	        true: Branch target if comparison is true
    68	        false: Branch target if comparison is false
    69	    """
    70	    if not isinstance(e, ComparisonExpr) or len(e.operands) != 2:
    71	        return False
    72	    ltype = self.node_type(e.operands[0])
    73	    rtype = self.node_type(e.operands[1])
    74	    if not (
    75	        (is_tagged(ltype) or is_fixed_width_rtype(ltype))
    76	        and (is_tagged(rtype) or is_fixed_width_rtype(rtype))
    77	    ):
    78	        return False
    79	    op = e.operators[0]
    80	    if op not in ("==", "!=", "<", "<=", ">", ">="):
    81	        return False
    82	    left_expr = e.operands[0]
    83	    right_expr = e.operands[1]
    84	    borrow_left = is_borrow_friendly_expr(self, right_expr)
    85	    left = self.accept(left_expr, can_borrow=borrow_left)
    86	    right = self.accept(right_expr, can_borrow=True)
    87	    if is_fixed_width_rtype(ltype) or is_fixed_width_rtype(rtype):
    88	        if not is_fixed_width_rtype(ltype):
    89	            left = self.coerce(left, rtype, e.line)
    90	        elif not is_fixed_width_rtype(rtype):
    91	            right = self.coerce(right, ltype, e.line)
    92	        reg = self.binary_op(left, right, op, e.line)
    93	        self.builder.flush_keep_alives()
    94	        self.add_bool_branch(reg, true, false)
    95	    else:
    96	        # "left op right" for two tagged integers
    97	        reg = self.builder.binary_op(left, right, op, e.line)
    98	        self.flush_keep_alives()
    99	        self.add_bool_branch(reg, true, false)
   100	    return True

   103	def is_borrow_friendly_expr(self: IRBuilder, expr: Expression) -> bool:
   104	    """Can the result of the expression borrowed temporarily?
   105
   106	    Borrowing means keeping a reference without incrementing the reference count.
   107	    """
   108	    if isinstance(expr, (IntExpr, FloatExpr, StrExpr, BytesExpr)):
   109	        # Literals are immortal and can always be borrowed
   110	        return True
   111	    if (
   112	        isinstance(expr, (UnaryExpr, OpExpr, NameExpr, MemberExpr))
   113	        and constant_fold_expr(self, expr) is not None
   114	    ):
   115	        # Literal expressions are similar to literals
   116	        return True
   117	    if isinstance(expr, NameExpr):
   118	        if isinstance(expr.node, Var) and expr.kind == LDEF:
   119	            # Local variable reference can be borrowed
   120	            return True
   121	    if isinstance(expr, MemberExpr) and self.is_native_attr_ref(expr):
   122	        return True
   123	    return False

   615	    def get_assignment_target(
   616	        self, lvalue: Lvalue, line: int = -1, *, for_read: bool = False
   617	    ) -> AssignmentTarget:
   618	        if line == -1:
   619	            line = lvalue.line
   620	        if isinstance(lvalue, NameExpr):
   621	            # If we are visiting a decorator, then the SymbolNode we really want to be looking at
   622	            # is the function that is decorated, not the entire Decorator node itself.
   623	            symbol = lvalue.node
   624	            if isinstance(symbol, Decorator):
   625	                symbol = symbol.func
   626	            if symbol is None:
   627	                # Semantic analyzer doesn't create ad-hoc Vars for special forms.
   628	                assert lvalue.is_special_form
   629	                symbol = Var(lvalue.name)
   630	            if not for_read and isinstance(symbol, Var) and symbol.is_cls:
   631	                self.error("Cannot assign to the first argument of classmethod", line)
   632	            if lvalue.kind == LDEF:
   633	                if symbol not in self.symtables[-1]:
   634	                    if isinstance(symbol, Var) and not isinstance(symbol.type, DeletedType):
   635	                        reg_type = self.type_to_rtype(symbol.type)
   636	                    else:
   637	                        reg_type = self.node_type(lvalue)
   638	                    # If the function is a generator function, then first define a new variable
   639	                    # in the current function's environment class. Next, define a target that
   640	                    # refers to the newly defined variable in that environment class. Add the
   641	                    # target to the table containing class environment variables, as well as the
   642	                    # current environment.
   643	                    if self.fn_info.is_generator:
   644	                        return self.add_var_to_env_class(
   645	                            symbol, reg_type, self.fn_info.generator_class, reassign=False
   646	                        )
   647
   648	                    # Otherwise define a new local variable.
   649	                    return self.add_local_reg(symbol, reg_type)
   650	                else:
   651	                    # Assign to a previously defined variable.
   652	                    return self.lookup(symbol)
   653	            elif lvalue.kind == GDEF:
   654	                globals_dict = self.load_globals_dict()
   655	                name = self.load_str(lvalue.name)
   656	                return AssignmentTargetIndex(globals_dict, name)
   657	            else:
   658	                assert False, lvalue.kind
   659	        elif isinstance(lvalue, IndexExpr):
   660	            # Indexed assignment x[y] = e
   661	            base = self.accept(lvalue.base)
   662	            index = self.accept(lvalue.index)
   663	            return AssignmentTargetIndex(base, index)
   664	        elif isinstance(lvalue, MemberExpr):
   665	            # Attribute assignment x.y = e
   666	            can_borrow = self.is_native_attr_ref(lvalue)
   667	            obj = self.accept(lvalue.expr, can_borrow=can_borrow)
   668	            return AssignmentTargetAttr(obj, lvalue.name, can_borrow=can_borrow)
   669	        elif isinstance(lvalue, TupleExpr):
   670	            # Multiple assignment a, ..., b = e
   671	            star_idx: int | None = None
   672	            lvalues = []
   673	            for idx, item in enumerate(lvalue.items):
   674	                targ = self.get_assignment_target(item)
   675	                lvalues.append(targ)
   676	                if isinstance(item, StarExpr):
   677	                    if star_idx is not None:
   678	                        self.error("Two starred expressions in assignment", line)
   679	                    star_idx = idx
   680
   681	            return AssignmentTargetTuple(lvalues, star_idx)
   682
   683	        elif isinstance(lvalue, StarExpr):
   684	            return self.get_assignment_target(lvalue.expr)
   685
   686	        assert False, "Unsupported lvalue: %r" % lvalue
   687
   688	    def read(
   689	        self, target: Value | AssignmentTarget, line: int = -1, can_borrow: bool = False
   690	    ) -> Value:
   691	        if isinstance(target, Value):
   692	            return target
   693	        if isinstance(target, AssignmentTargetRegister):
   694	            return target.register
   695	        if isinstance(target, AssignmentTargetIndex):
   696	            reg = self.gen_method_call(
   697	                target.base, "__getitem__", [target.index], target.type, line
   698	            )

   700	                return reg
   701	            assert False, target.base.type
   702	        if isinstance(target, AssignmentTargetAttr):
   703	            if isinstance(target.obj.type, RInstance) and target.obj.type.class_ir.is_ext_class:
   704	                borrow = can_borrow and target.can_borrow
   705	                return self.add(GetAttr(target.obj, target.attr, line, borrow=borrow))
   706	            else:
   707	                return self.py_get_attr(target.obj, target.attr, line)

   709	        assert False, "Unsupported lvalue: %r" % target

   720	    def assign(self, target: Register | AssignmentTarget, rvalue_reg: Value, line: int) -> None:
   721	        if isinstance(target, Register):
   722	            self.add(Assign(target, self.coerce_rvalue(rvalue_reg, target.type, line)))
   723	        elif isinstance(target, AssignmentTargetRegister):
   724	            rvalue_reg = self.coerce_rvalue(rvalue_reg, target.type, line)
   725	            self.add(Assign(target.register, rvalue_reg))
   726	        elif isinstance(target, AssignmentTargetAttr):
   727	            if isinstance(target.obj_type, RInstance):
   728	                rvalue_reg = self.coerce_rvalue(rvalue_reg, target.type, line)
   729	                self.add(SetAttr(target.obj, target.attr, rvalue_reg, line))
   730	            else:
   731	                key = self.load_str(target.attr)
   732	                boxed_reg = self.builder.box(rvalue_reg)
   733	                self.primitive_op(py_setattr_op, [target.obj, key, boxed_reg], line)
   734	        elif isinstance(target, AssignmentTargetIndex):
   735	            target_reg2 = self.gen_method_call(
   736	                target.base, "__setitem__", [target.index, rvalue_reg], None, line
   737	            )
   738	            assert target_reg2 is not None, target.base.type
   739	        elif isinstance(target, AssignmentTargetTuple):
   740	            if isinstance(rvalue_reg.type, RTuple) and target.star_idx is None:
   741	                rtypes = rvalue_reg.type.types
   742	                assert len(rtypes) == len(target.items)
   743	                for i in range(len(rtypes)):
   744	                    item_value = self.add(TupleGet(rvalue_reg, i, line))
   745	                    self.assign(target.items[i], item_value, line)
   746	            elif (
   747	                is_list_rprimitive(rvalue_reg.type) or is_tuple_rprimitive(rvalue_reg.type)
   748	            ) and target.star_idx is None:
   749	                self.process_sequence_assignment(target, rvalue_reg, line)
   750	            else:
   751	                self.process_iterator_tuple_assignment(target, rvalue_reg, line)
   752	        else:
   753	            assert False, "Unsupported assignment target"

   755	    def coerce_rvalue(self, rvalue: Value, rtype: RType, line: int) -> Value:
   756	        if is_float_rprimitive(rtype) and is_tagged(rvalue.type):
   757	            typename = rvalue.type.short_name()
   758	            if typename == "short_int":
   759	                typename = "int"
   760	            self.error(
   761	                "Incompatible value representations in assignment "
   762	                + f'(expression has type "{typename}", variable has type "float")',
   763	                line,
   764	            )
   765	        return self.coerce(rvalue, rtype, line)

   767	    def process_sequence_assignment(
   768	        self, target: AssignmentTargetTuple, rvalue: Value, line: int
   769	    ) -> None:
   770	        """Process assignment like 'x, y = s', where s is a variable-length list or tuple."""
   771	        # Check the length of sequence.
   772	        expected_len = Integer(len(target.items), c_pyssize_t_rprimitive)
   773	        self.builder.call_c(check_unpack_count_op, [rvalue, expected_len], line)
   774
   775	        # Read sequence items.
   776	        values = []
   777	        for i in range(len(target.items)):
   778	            item = target.items[i]
   779	            index = self.builder.load_int(i)
   780	            if is_list_rprimitive(rvalue.type):
   781	                item_value = self.primitive_op(list_get_item_unsafe_op, [rvalue, index], line)
   782	            else:
   783	                item_value = self.builder.gen_method_call(
   784	                    rvalue, "__getitem__", [index], item.type, line
   785	                )
   786	            values.append(item_value)
   787
   788	        # Assign sequence items to the target lvalues.
   789	        for lvalue, value in zip(target.items, values):
   790	            self.assign(lvalue, value, line)

   792	    def process_iterator_tuple_assignment_helper(
   793	        self, litem: AssignmentTarget, ritem: Value, line: int
   794	    ) -> None:
   795	        error_block, ok_block = BasicBlock(), BasicBlock()
   796	        self.add(Branch(ritem, error_block, ok_block, Branch.IS_ERROR))
   797
   798	        self.activate_block(error_block)
   799	        self.add(
   800	            RaiseStandardError(RaiseStandardError.VALUE_ERROR, "not enough values to unpack", line)
   801	        )
   802	        self.add(Unreachable())
   803
   804	        self.activate_block(ok_block)
   805	        self.assign(litem, ritem, line)
   806
   807	    def process_iterator_tuple_assignment(
   808	        self, target: AssignmentTargetTuple, rvalue_reg: Value, line: int
   809	    ) -> None:
   810	        iterator = self.primitive_op(iter_op, [rvalue_reg], line)
   811
   812	        # This may be the whole lvalue list if there is no starred value
   813	        split_idx = target.star_idx if target.star_idx is not None else len(target.items)
   814
   815	        # Assign values before the first starred value
   816	        for litem in target.items[:split_idx]:
   817	            ritem = self.call_c(next_op, [iterator], line)
   818	            error_block, ok_block = BasicBlock(), BasicBlock()
   819	            self.add(Branch(ritem, error_block, ok_block, Branch.IS_ERROR))
   820
   821	            self.activate_block(error_block)
   822	            self.add(
   823	                RaiseStandardError(
   824	                    RaiseStandardError.VALUE_ERROR, "not enough values to unpack", line
   825	                )
   826	            )
   827	            self.add(Unreachable())
   828
   829	            self.activate_block(ok_block)
   830
   831	            self.assign(litem, ritem, line)
   832
   833	        # Assign the starred value and all values after it
   834	        if target.star_idx is not None:
   835	            post_star_vals = target.items[split_idx + 1 :]
   836	            iter_list = self.primitive_op(to_list, [iterator], line)
   837	            iter_list_len = self.builtin_len(iter_list, line)
   838	            post_star_len = Integer(len(post_star_vals))
   839	            condition = self.binary_op(post_star_len, iter_list_len, "<=", line)
   840
   841	            error_block, ok_block = BasicBlock(), BasicBlock()
   842	            self.add(Branch(condition, ok_block, error_block, Branch.BOOL))
   843
   844	            self.activate_block(error_block)
   845	            self.add(
   846	                RaiseStandardError(
   847	                    RaiseStandardError.VALUE_ERROR, "not enough values to unpack", line
   848	                )
   849	            )
   850	            self.add(Unreachable())
   851
   852	            self.activate_block(ok_block)
   853
   854	            for litem in reversed(post_star_vals):
   855	                ritem = self.primitive_op(list_pop_last, [iter_list], line)
   856	                self.assign(litem, ritem, line)
   857
   858	            # Assign the starred value
   859	            self.assign(target.items[target.star_idx], iter_list, line)
   860
   861	        # There is no starred value, so check if there are extra values in rhs that
   862	        # have not been assigned.
   863	        else:
   864	            extra = self.call_c(next_op, [iterator], line)
   865	            error_block, ok_block = BasicBlock(), BasicBlock()
   866	            self.add(Branch(extra, ok_block, error_block, Branch.IS_ERROR))
   867
   868	            self.activate_block(error_block)
   869	            self.add(
   870	                RaiseStandardError(
   871	                    RaiseStandardError.VALUE_ERROR, "too many values to unpack", line
   872	                )
   873	            )
   874	            self.add(Unreachable())
   875
   876	            self.activate_block(ok_block)
