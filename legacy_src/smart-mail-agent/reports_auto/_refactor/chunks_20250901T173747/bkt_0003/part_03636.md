ation file to read:
   187
   188	            * If it is ".coveragerc", it is interpreted as if it were True,
   189	              for backward compatibility.
   190
   191	            * If it is a string, it is the name of the file to read.  If the
   192	              file can't be read, it is an error.
   193
   194	            * If it is True, then a few standard files names are tried
   195	              (".coveragerc", "setup.cfg", "tox.ini").  It is not an error for
   196	              these files to not be found.
   197
   198	            * If it is False, then no configuration file is read.
   199
   200	        `source` is a list of file paths or package names.  Only code located
   201	        in the trees indicated by the file paths or package names will be
   202	        measured.
   203
   204	        `source_pkgs` is a list of package names. It works the same as
   205	        `source`, but can be used to name packages where the name can also be
   206	        interpreted as a file path.
   207
   208	        `source_dirs` is a list of file paths. It works the same as
   209	        `source`, but raises an error if the path doesn't exist, rather
   210	        than being treated as a package name.
   211
   212	        `include` and `omit` are lists of file name patterns. Files that match
   213	        `include` will be measured, files that match `omit` will not.  Each
   214	        will also accept a single string argument.
   215
   216	        `debug` is a list of strings indicating what debugging information is
   217	        desired.
   218
   219	        `concurrency` is a string indicating the concurrency library being used
   220	        in the measured code.  Without this, coverage.py will get incorrect
   221	        results if these libraries are in use.  Valid strings are "greenlet",
   222	        "eventlet", "gevent", "multiprocessing", or "thread" (the default).
   223	        This can also be a list of these strings.
   224
   225	        If `check_preimported` is true, then when coverage is started, the
   226	        already-imported files will be checked to see if they should be
   227	        measured by coverage.  Importing measured files before coverage is
   228	        started can mean that code is missed.
   229
   230	        `context` is a string to use as the :ref:`static context
   231	        <static_contexts>` label for collected data.
   232
   233	        If `messages` is true, some messages will be printed to stdout
   234	        indicating what is happening.
   235
   236	        If `plugins` are passed, they are an iterable of function objects
   237	        accepting a `reg` object to register plugins, as described in
   238	        :ref:`api_plugin`.  When they are provided, they will override the
   239	        plugins found in the coverage configuration file.
   240
   241	        .. versionadded:: 4.0
   242	            The `concurrency` parameter.
   243
   244	        .. versionadded:: 4.2
   245	            The `concurrency` parameter can now be a list of strings.
   246
   247	        .. versionadded:: 5.0
   248	            The `check_preimported` and `context` parameters.
   249
   250	        .. versionadded:: 5.3
   251	            The `source_pkgs` parameter.
   252
   253	        .. versionadded:: 6.0
   254	            The `messages` parameter.
   255
   256	        .. versionadded:: 7.7
   257	            The `plugins` parameter.
   258
   259	        .. versionadded:: 7.8
   260	            The `source_dirs` parameter.
   261	        """
   262	        # Start self.config as a usable default configuration. It will soon be
   263	        # replaced with the real configuration.
   264	        self.config = CoverageConfig()
   265
   266	        # data_file=None means no disk file at all. data_file missing means
   267	        # use the value from the config file.
   268	        self._no_disk = data_file is None
   269	        if isinstance(data_file, DefaultValue):
   270	            data_file = None
   271	        if data_file is not None:
   272	            data_file = os.fspath(data_file)
   273
   274	        # This is injectable by tests.
   275	        self._debug_file: IO[str] | None = None
   276
   277	        self._auto_load = self._auto_save = auto_data
   278	        self._data_suffix_specified = data_suffix
   279
   280	        # Is it ok for no data to be collected?
   281	        self._warn_no_data = True
   282	        self._warn_unimported_source = True
   283	        self._warn_preimported_source = check_preimported
   284	        self._no_warn_slugs: set[str] = set()
   285	        self._messages = messages
   286
   287	        # If we're invoked from a .pth file, we shouldn't try to make another one.
   288	        self._make_pth_file = True
   289
   290	        # A record of all the warnings that have been issued.
   291	        self._warnings: list[str] = []
   292
   293	        # Other instance attributes, set with placebos or placeholders.
   294	        # More useful objects will be created later.
   295	        self._debug: DebugControl = NoDebugging()
   296	        self._inorout: InOrOut | None = None
   297	        self._plugins: Plugins = Plugins()
   298	        self._plugin_override = cast(Union[Iterable[TCoverageInit], None], plugins)
   299	        self._data: CoverageData | None = None
   300	        self._data_to_close: list[CoverageData] = []
   301	        self._core: Core | None = None
   302	        self._collector: Collector | None = None
   303	        self._metacov = False
   304
   305	        self._file_mapper: Callable[[str], str] = abs_file
   306	        self._data_suffix = self._run_suffix = None
   307	        self._exclude_re: dict[str, str] = {}
   308	        self._old_sigterm: Callable[[int, FrameType | None], Any] | None = None
   309
   310	        # State machine variables:
   311	        # Have we initialized everything?
   312	        self._inited = False
   313	        self._inited_for_start = False
   314	        # Have we started collecting and not stopped it?
   315	        self._started = False
   316	        # Should we write the debug output?
   317	        self._should_write_debug = True
   318
   319	        # Build our configuration from a number of sources.
   320	        if isinstance(config_file, str) and config_file.startswith(CONFIG_DATA_PREFIX):
   321	            self.config = deserialize_config(config_file[len(CONFIG_DATA_PREFIX):])
   322	        else:
   323	            if not isinstance(config_file, bool):
   324	                config_file = os.fspath(config_file)
   325	            self.config = read_coverage_config(
   326	                config_file=config_file,
   327	                warn=self._warn,
   328	                data_file=data_file,
   329	                cover_pylib=cover_pylib,
   330	                timid=timid,
   331	                branch=branch,
   332	                parallel=bool_or_none(data_suffix),
   333	                source=source,
   334	                source_pkgs=source_pkgs,
   335	                source_dirs=source_dirs,
   336	                run_omit=omit,
   337	                run_include=include,
   338	                debug=debug,
   339	                report_omit=omit,
   340	                report_include=include,
   341	                concurrency=concurrency,
   342	                context=context,
   343	            )
   344
   345	        # If we have subprocess measurement happening automatically, then we
   346	        # want any explicit creation of a Coverage object to mean, this process
   347	        # is already coverage-aware, so don't auto-measure it.  By now, the
   348	        # auto-creation of a Coverage object has already happened.  But we can
   349	        # find it and tell it not to save its data.
   350	        if not env.METACOV:
   351	            _prevent_sub_process_measurement()

   353	    def _init(self) -> None:
   354	        """Set all the initial state.
   355
   356	        This is called by the public methods to initialize state. This lets us
   357	        construct a :class:`Coverage` object, then tweak its state before this
   358	        function is called.
   359
   360	        """
   361	        if self._inited:
   362	            return
   363
   364	        self._inited = True
   365
   366	        # Create and configure the debugging controller.
   367	        self._debug = DebugControl(self.config.debug, self._debug_file, self.config.debug_file)
   368	        if self._debug.should("process"):
   369	            self._debug.write("Coverage._init")
   370
   371	        if "multiprocessing" in (self.config.concurrency or ()):
   372	            # Multi-processing uses parallel for the subprocesses, so also use
   373	            # it for the main process.
   374	            self.config.parallel = True
   375
   376	        # _exclude_re is a dict that maps exclusion list names to compiled regexes.
   377	        self._exclude_re = {}
   378
   379	        set_relative_directory()
   380	        if self.config.relative_files:
   381	            self._file_mapper = relative_filename
   382
   383	        # Load plugins
   384	        self._plugins = Plugins(self._debug)
   385	        if self._plugin_override:
   386	            self._plugins.load_from_callables(self._plugin_override)
   387	        else:
   388	            self._plugins.load_from_config(self.config.plugins, self.config)
   389
   390	        # Run configuring plugins.
   391	        for plugin in self._plugins.configurers:
   392	            # We need an object with set_option and get_option. Either self or
   393	            # self.config will do. Choosing randomly stops people from doing
   394	            # other things with those objects, against the public API.  Yes,
   395	            # this is a bit childish. :)
   396	            plugin.configure([self, self.config][int(time.time()) % 2])

   398	    def _post_init(self) -> None:
   399	        """Stuff to do after everything is initialized."""
   400	        if self._should_write_debug:
   401	            self._should_write_debug = False
   402	            self._write_startup_debug()
   403
   404	        # "[run] _crash" will raise an exception if the value is close by in
   405	        # the call stack, for testing error handling.
   406	        if self.config._crash and self.config._crash in short_stack():
   407	            raise RuntimeError(f"Crashing because called by {self.config._crash}")

   409	    def _write_startup_debug(self) -> None:
   410	        """Write out debug info at startup if needed."""
   411	        wrote_any = False
   412	        with self._debug.without_callers():
   413	            if self._debug.should("config"):
   414	                config_info = self.config.debug_info()
   415	                write_formatted_info(self._debug.write, "config", config_info)
   416	                wrote_any = True
   417
   418	            if self._debug.should("sys"):
   419	                write_formatted_info(self._debug.write, "sys", self.sys_info())
   420	                for plugin in self._plugins:
   421	                    header = "sys: " + plugin._coverage_plugin_name
   422	                    info = plugin.sys_info()
   423	                    write_formatted_info(self._debug.write, header, info)
   424	                wrote_any = True
   425
   426	            if self._debug.should("pybehave"):
   427	                write_formatted_info(self._debug.write, "pybehave", env.debug_info())
   428	                wrote_any = True
   429
   430	        if wrote_any:
   431	            write_formatted_info(self._debug.write, "end", ())

   433	    def _should_trace(self, filename: str, frame: FrameType) -> TFileDisposition:
   434	        """Decide whether to trace execution in `filename`.
   435
   436	        Calls `_should_trace_internal`, and returns the FileDisposition.
   437
   438	        """
   439	        assert self._inorout is not None
   440	        disp = self._inorout.should_trace(filename, frame)
   441	        if self._debug.should("trace"):
   442	            self._debug.write(disposition_debug_msg(disp))
   443	        return disp

   445	    def _check_include_omit_etc(self, filename: str, frame: FrameType) -> bool:
   446	        """Check a file name against the include/omit/etc, rules, verbosely.
   447
   448	        Returns a boolean: True if the file should be traced, False if not.
   449
   450	        """
   451	        assert self._inorout is not None
   452	        reason = self._inorout.check_include_omit_etc(filename, frame)
   453	        if self._debug.should("trace"):
   454	            if not reason:
   455	                msg = f"Including {filename!r}"
   456	            else:
   457	                msg = f"Not including {filename!r}: {reason}"
   458	            self._debug.write(msg)
   459
   460	        return not reason

   462	    def _warn(self, msg: str, slug: str | None = None, once: bool = False) -> None:
   463	        """Use `msg` as a warning.
   464
   465	        For warning suppression, use `slug` as the shorthand.
   466
   467	        If `once` is true, only show this warning once (determined by the
   468	        slug.)
   469
   470	        """
   471	        if not self._no_warn_slugs:
   472	            self._no_warn_slugs = set(self.config.disable_warnings)
   473
   474	        if slug in self._no_warn_slugs:
   475	            # Don't issue the warning
   476	            return
   477
   478	        self._warnings.append(msg)
   479	        if slug:
   480	            msg = f"{msg} ({slug})"
   481	        if self._debug.should("pid"):
   482	            msg = f"[{os.getpid()}] {msg}"
   483	        warnings.warn(msg, category=CoverageWarning, stacklevel=2)
   484
   485	        if once:
   486	            assert slug is not None
   487	            self._no_warn_slugs.add(slug)

   490	        """Write a message to the user, if configured to do so."""
   491	        if self._messages:
   492	            print(msg)

   544	    def load(self) -> None:
   545	        """Load previously-collected coverage data from the data file."""
   546	        self._init()
   547	        if self._collector is not None:
   548	            self._collector.reset()
   549	        should_skip = self.config.parallel and not os.path.exists(self.config.data_file)
   550	        if not should_skip:
   551	            self._init_data(suffix=None)
   552	        self._post_init()
   553	        if not should_skip:
   554	            assert self._data is not None
   555	            self._data.read()

   557	    def _init_for_start(self) -> None:
   558	        """Initialization for start()"""
   559	        # Construct the collector.
   560	        concurrency: list[str] = self.config.concurrency or []
   561	        if "multiprocessing" in concurrency:
   562	            if self.config.config_file is None:
   563	                raise ConfigError("multiprocessing requires a configuration file")
   564	            patch_multiprocessing(rcfile=self.config.config_file)
   565
   566	        dycon = self.config.dynamic_context
   567	        if not dycon or dycon == "none":
   568	            context_switchers = []
   569	        elif dycon == "test_function":
   570	            context_switchers = [should_start_context_test_function]
   571	        else:
   572	            raise ConfigError(f"Don't understand dynamic_context setting: {dycon!r}")
   573
   574	        context_switchers.extend(
   575	            plugin.dynamic_context for plugin in self._plugins.context_switchers
   576	        )
   577
   578	        should_start_context = combine_context_switchers(context_switchers)
   579
   580	        self._core = Core(
   581	            warn=self._warn,
   582	            config=self.config,
   583	            dynamic_contexts=(should_start_context is not None),
   584	            metacov=self._metacov,
   585	        )
   586	        self._collector = Collector(
   587	            core=self._core,
   588	            should_trace=self._should_trace,
   589	            check_include=self._check_include_omit_etc,
   590	            should_start_context=should_start_context,
   591	            file_mapper=self._file_mapper,
   592	            branch=self.config.branch,
   593	            warn=self._warn,
   594	            concurrency=concurrency,
   595	        )
   596
   597	        suffix = self._data_suffix_specified
   598	        if suffix:
   599	            if not isinstance(suffix, str):
   600	                # if data_suffix=True, use .machinename.pid.random
   601	                suffix = True
   602	        elif self.config.parallel:
   603	            if suffix is None:
   604	                suffix = True
   605	            elif not isinstance(suffix, str):
   606	                suffix = bool(suffix)
   607	        else:
   608	            suffix = None
   609
   610	        self._init_data(suffix)
   611
   612	        assert self._data is not None
   613	        self._collector.use_data(self._data, self.config.context)
   614
   615	        # Early warning if we aren't going to be able to support plugins.
   616	        if self._plugins.file_tracers and not self._core.supports_plugins:
   617	            self._warn(
   618	                "Plugin file tracers ({}) aren't supported with {}".format(
   619	                    ", ".join(
   620	                        plugin._coverage_plugin_name
   621	                            for plugin in self._plugins.file_tracers
   622	                    ),
   623	                    self._collector.tracer_name(),
   624	                ),
   625	            )
   626	            for plugin in self._plugins.file_tracers:
   627	                plugin._coverage_enabled = False
   628
   629	        # Create the file classifying substructure.
   630	        self._inorout = InOrOut(
   631	            config=self.config,
   632	            warn=self._warn,
   633	            debug=(self._debug if self._debug.should("trace") else None),
   634	            include_namespace_packages=self.config.include_namespace_packages,
   635	        )
   636	        self._inorout.plugins = self._plugins
   637	        self._inorout.disp_class = self._core.file_disposition_class
   638
   639	        # It's useful to write debug info after initing for start.
   640	        self._should_write_debug = True
   641
   642	        # Register our clean-up handlers.
   643	        atexit.register(self._atexit)
   644	        if self.config.sigterm:
   645	            is_main = (threading.current_thread() == threading.main_thread())
   646	            if is_main and not env.WINDOWS:
   647	                # The Python docs seem to imply that SIGTERM works uniformly even
   648	                # on Windows, but that's not my experience, and this agrees:
   649	                # https://stackoverflow.com/questions/35772001/x/35792192#35792192
   650	                self._old_sigterm = signal.signal(      # type: ignore[assignment]
   651	                    signal.SIGTERM, self._on_sigterm,
   652	                )

   654	    def _init_data(self, suffix: str | bool | None) -> None:
   655	        """Create a data file if we don't have one yet."""
   656	        if self._data is None:
   657	            # Create the data file.  We do this at construction time so that the
   658	            # data file will be written into the directory where the process
   659	            # started rather than wherever the process eventually chdir'd to.
   660	            ensure_dir_for_file(self.config.data_file)
   661	            self._data = CoverageData(
   662	                basename=self.config.data_file,
   663	                suffix=suffix,
   664	                warn=self._warn,
   665	                debug=self._debug,
   666	                no_disk=self._no_disk,
   667	            )
   668	            self._data_to_close.append(self._data)

   670	    def start(self) -> None:
   671	        """Start measuring code coverage.
   672
   673	        Coverage measurement is only collected in functions called after
   674	        :meth:`start` is invoked.  Statements in the same scope as
   675	        :meth:`start` won't be measured.
   676
   677	        Once you invoke :meth:`start`, you must also call :meth:`stop`
   678	        eventually, or your process might not shut down cleanly.
   679
   680	        The :meth:`collect` method is a context manager to handle both
   681	        starting and stopping collection.
   682
   683	        """
   684	        self._init()
   685	        if not self._inited_for_start:
   686	            self._inited_for_start = True
   687	            self._init_for_start()
   688	        self._post_init()
   689
   690	        assert self._collector is not None
   691	        assert self._inorout is not None
   692
   693	        # Issue warnings for possible problems.
   694	        self._inorout.warn_conflicting_settings()
   695
   696	        # See if we think some code that would eventually be measured has
   697	        # already been imported.
   698	        if self._warn_preimported_source:
   699	            self._inorout.warn_already_imported_files()
   700
   701	        if self._auto_load:
   702	            self.load()
   703
   704	        apply_patches(self, self.config, self._debug, make_pth_file=self._make_pth_file)
   705
   706	        self._collector.start()
   707	        self._started = True
   708	        self._instances.append(self)
   709
   710	    def stop(self) -> None:
   711	        """Stop measuring code coverage."""
   712	        if self._instances:
   713	            if self._instances[-1] is self:
   714	                self._instances.pop()
   715	        if self._started:
   716	            assert self._collector is not None
   717	            self._collector.stop()
   718	        self._started = False

   720	    @contextlib.contextmanager
   721	    def collect(self) -> Iterator[None]:
   722	        """A context manager to start/stop coverage measurement collection.
   723
   724	        .. versionadded:: 7.3
   725
   726	        """
   727	        self.start()
   728	        try:
   729	            yield
   730	        finally:
   731	            self.stop()     # pragma: nested
   732
   733	    def _atexit(self, event: str = "atexit") -> None:
   734	        """Clean up on process shutdown."""
   735	        if self._debug.should("process"):
   736	            self._debug.write(f"{event}: pid: {os.getpid()}, instance: {self!r}")
   737	        if self._started:
   738	            self.stop()
   739	        if self._auto_save or event == "sigterm":
   740	            self.save()
   741	        for d in self._data_to_close:
   742	            d.close(force=True)
   743
   744	    def _on_sigterm(self, signum_unused: int, frame_unused: FrameType | None) -> None:
   745	        """A handler for signal.SIGTERM."""
   746	        self._atexit("sigterm")
   747	        # Statements after here won't be seen by metacov because we just wrote
   748	        # the data, and are about to kill the process.
   749	        signal.signal(signal.SIGTERM, self._old_sigterm)    # pragma: not covered
   750	        os.kill(os.getpid(), signal.SIGTERM)                # pragma: not covered

   752	    def erase(self) -> None:
   753	        """Erase previously collected coverage data.
   754
   755	        This removes the in-memory data collected in this session as well as
   756	        discarding the data file.
   757
   758	        """
   759	        self._init()
   760	        self._post_init()
   761	        if self._collector is not None:
   762	            self._collector.reset()
   763	        self._init_data(suffix=None)
   764	        assert self._data is not None
   765	        self._data.erase(parallel=self.config.parallel)
   766	        self._data = None

   902	        """Get the collected data.
   903

   905
   906	        Returns a :class:`coverage.CoverageData`, the collected coverage data.
   907
   908	        .. versionadded:: 4.0

   910	        """

   926	    def _post_save_work(self) -> None:
   927	        """After saving data, look for warnings, post-work, etc.
   928
   929	        Warn about things that should have happened but didn't.
   930	        Look for un-executed files.
   931
   932	        """
   933	        assert self._data is not None
   934	        assert self._inorout is not None
   935
   936	        # If there are still entries in the source_pkgs_unmatched list,
   937	        # then we never encountered those packages.
   938	        if self._warn_unimported_source:
   939	            self._inorout.warn_unimported_source()

   941	        # Find out if we got any data.
   942	        if not self._data and self._warn_no_data:
   943	            self._warn("No data was collected.", slug="no-data-collected")

   945	        # Touch all the files that could have executed, so that we can
   946	        # mark completely un-executed files as 0% covered.
   947	        file_paths = collections.defaultdict(list)
   948	        for file_path, plugin_name in self._inorout.find_possibly_unexecuted_files():
   949	            file_path = self._file_mapper(file_path)
   950	            file_paths[plugin_name].append(file_path)
   951	        for plugin_name, paths in file_paths.items():
   952	            self._data.touch_files(paths, plugin_name)

   140	def info_formatter(info: Iterable[tuple[str, Any]]) -> Iterator[str]:
   141	    """Produce a sequence of formatted lines from info.
   142
   143	    `info` is a sequence of pairs (label, data).  The produced lines are
   144	    nicely formatted, ready to print.
   145
   146	    """
   147	    info = list(info)
   148	    if not info:
   149	        return
   150	    label_len = 30
   151	    assert all(len(l) < label_len for l, _ in info)
   152	    for label, data in info:
   153	        if data == []:
   154	            data = "-none-"
   155	        if isinstance(data, tuple) and len(repr(tuple(data))) < 30:
   156	            # Convert to tuple to scrub namedtuples.
   157	            yield "%*s: %r" % (label_len, label, tuple(data))
   158	        elif isinstance(data, (list, set, tuple)):
   159	            prefix = "%*s:" % (label_len, label)
   160	            for e in data:
   161	                yield "%*s %s" % (label_len+1, prefix, e)
   162	                prefix = ""
   163	        else:
   164	            yield "%*s: %s" % (label_len, label, data)

   234	def short_stack(
   235	    skip: int = 0,
   236	    full: bool = False,
   237	    frame_ids: bool = False,
   238	    short_filenames: bool = False,
   239	) -> str:
   240	    """Return a string summarizing the call stack.
   241
   242	    The string is multi-line, with one line per stack frame. Each line shows
   243	    the function name, the file name, and the line number:
   244
   245	        ...
   246	        start_import_stop : /Users/ned/coverage/trunk/tests/coveragetest.py:95
   247	        import_local_file : /Users/ned/coverage/trunk/tests/coveragetest.py:81
   248	        import_local_file : /Users/ned/coverage/trunk/coverage/backward.py:159
   249	        ...
   250
   251	    `skip` is the number of closest immediate frames to skip, so that debugging
   252	    functions can call this and not be included in the result.
   253
   254	    If `full` is true, then include all frames.  Otherwise, initial "boring"
   255	    frames (ones in site-packages and earlier) are omitted.
   256
   257	    `short_filenames` will shorten filenames using `short_filename`, to reduce
   258	    the amount of repetitive noise in stack traces.
   259
   260	    """
   261	    # Regexes in initial frames that we don't care about.
   262	    BORING_PRELUDE = [
   263	        "<string>",             # pytest-xdist has string execution.
   264	        r"\bigor.py$",          # Our test runner.
   265	        r"\bsite-packages\b",   # pytest etc getting to our tests.
   266	    ]
   267
   268	    stack: Iterable[inspect.FrameInfo] = inspect.stack()[:skip:-1]
   269	    if not full:
   270	        for pat in BORING_PRELUDE:
   271	            stack = itertools.dropwhile(
   272	                (lambda fi, pat=pat: re.search(pat, fi.filename)),  # type: ignore[misc]
   273	                stack,
   274	            )
   275	    lines = []
   276	    for frame_info in stack:
   277	        line = f"{frame_info.function:>30s} : "
   278	        if frame_ids:
   279	            line += f"{id(frame_info.frame):#x} "
   280	        filename = frame_info.filename
   281	        if short_filenames:
   282	            filename = short_filename(filename)
   283	        line += f"{filename}:{frame_info.lineno}"
   284	        lines.append(line)
   285	    return "\n".join(lines)

   351	def filter_text(text: str, filters: Iterable[Callable[[str], str]]) -> str:
   352	    """Run `text` through a series of filters.
   353
   354	    `filters` is a list of functions. Each takes a string and returns a
   355	    string.  Each is run in turn. After each filter, the text is split into
   356	    lines, and each line is passed through the next filter.
   357
   358	    Returns: the final string that results after all of the filters have
   359	    run.
   360
   361	    """
   362	    clean_text = text.rstrip()
   363	    ending = text[len(clean_text):]
   364	    text = clean_text
   365	    for filter_fn in filters:
   366	        lines = []
   367	        for line in text.splitlines():
   368	            lines.extend(filter_fn(line).splitlines())
   369	        text = "\n".join(lines)
   370	    return text + ending

   429	class DebugOutputFile:
   430	    """A file-like object that includes pid and cwd information."""
   431	    def __init__(
   432	        self,
   433	        outfile: IO[str] | None,
   434	        filters: Iterable[Callable[[str], str]],
   435	    ):
   436	        self.outfile = outfile
   437	        self.filters = list(filters)
   438	        self.pid = os.getpid()
   439
   440	    @classmethod
   441	    def get_one(
   442	        cls,
   443	        fileobj: IO[str] | None = None,
   444	        file_name: str | None = None,
   445	        filters: Iterable[Callable[[str], str]] = (),
   446	        interim: bool = False,
   447	    ) -> DebugOutputFile:
   448	        """Get a DebugOutputFile.
   449
   450	        If `fileobj` is provided, then a new DebugOutputFile is made with it.
   451
   452	        If `fileobj` isn't provided, then a file is chosen (`file_name` if
   453	        provided, or COVERAGE_DEBUG_FILE, or stderr), and a process-wide
   454	        singleton DebugOutputFile is made.
   455
   456	        `filters` are the text filters to apply to the stream to annotate with
   457	        pids, etc.
   458
   459	        If `interim` is true, then a future `get_one` can replace this one.
   460
   461	        """
   462	        if fileobj is not None:
   463	            # Make DebugOutputFile around the fileobj passed.
   464	            return cls(fileobj, filters)
   465
   466	        the_one, is_interim = cls._get_singleton_data()
   467	        if the_one is None or is_interim:
   468	            if file_name is not None:
   469	                fileobj = open(file_name, "a", encoding="utf-8")
   470	            else:
   471	                # $set_env.py: COVERAGE_DEBUG_FILE - Where to write debug output
   472	                file_name = os.getenv("COVERAGE_DEBUG_FILE", FORCED_DEBUG_FILE)
   473	                if file_name in ["stdout", "stderr"]:
   474	                    fileobj = getattr(sys, file_name)
   475	                elif file_name:
   476	                    fileobj = open(file_name, "a", encoding="utf-8")
   477	                    atexit.register(fileobj.close)
   478	                else:
   479	                    fileobj = sys.stderr
   480	            the_one = cls(fileobj, filters)
   481	            cls._set_singleton_data(the_one, interim)
   482
   483	        if not(the_one.filters):
   484	            the_one.filters = list(filters)
   485	        return the_one
   486
   487	    # Because of the way igor.py deletes and re-imports modules,
   488	    # this class can be defined more than once. But we really want
   489	    # a process-wide singleton. So stash it in sys.modules instead of
   490	    # on a class attribute. Yes, this is aggressively gross.
   491
   492	    SYS_MOD_NAME: Final[str] = "$coverage.debug.DebugOutputFile.the_one"
   493	    SINGLETON_ATTR: Final[str] = "the_one_and_is_interim"
   494
   495	    @classmethod
   496	    def _set_singleton_data(cls, the_one: DebugOutputFile, interim: bool) -> None:
   497	        """Set the one DebugOutputFile to rule them all."""
   498	        singleton_module = types.ModuleType(cls.SYS_MOD_NAME)
   499	        setattr(singleton_module, cls.SINGLETON_ATTR, (the_one, interim))
   500	        sys.modules[cls.SYS_MOD_NAME] = singleton_module
   501
   502	    @classmethod
   503	    def _get_singleton_data(cls) -> tuple[DebugOutputFile | None, bool]:
   504	        """Get the one DebugOutputFile."""
   505	        singleton_module = sys.modules.get(cls.SYS_MOD_NAME)
   506	        return getattr(singleton_module, cls.SINGLETON_ATTR, (None, True))
   507
   508	    @classmethod
   509	    def _del_singleton_data(cls) -> None:
   510	        """Delete the one DebugOutputFile, just for tests to use."""
   511	        if cls.SYS_MOD_NAME in sys.modules:
   512	            del sys.modules[cls.SYS_MOD_NAME]
   513
   514	    def write(self, text: str) -> None:
   515	        """Just like file.write, but filter through all our filters."""
   516	        assert self.outfile is not None
   517	        if not self.outfile.closed:
   518	            self.outfile.write(filter_text(text, self.filters))
   519	            self.outfile.flush()
   520
   521	    def flush(self) -> None:
   522	        """Flush our file."""
   523	        assert self.outfile is not None
   524	        if not self.outfile.closed:
   525	            self.outfile.flush()
