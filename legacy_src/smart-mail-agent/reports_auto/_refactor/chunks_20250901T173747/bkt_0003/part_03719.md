   108	def generate_native_function(
   109	    fn: FuncIR, emitter: Emitter, source_path: str, module_name: str
   110	) -> None:
   111	    declarations = Emitter(emitter.context)
   112	    names = generate_names_for_ir(fn.arg_regs, fn.blocks)
   113	    body = Emitter(emitter.context, names)
   114	    visitor = FunctionEmitterVisitor(body, declarations, source_path, module_name)
   115
   116	    declarations.emit_line(f"{native_function_header(fn.decl, emitter)} {{")
   117	    body.indent()
   118
   119	    for r in all_values(fn.arg_regs, fn.blocks):
   120	        if isinstance(r.type, RTuple):
   121	            emitter.declare_tuple_struct(r.type)
   122	        if isinstance(r.type, RArray):
   123	            continue  # Special: declared on first assignment
   124
   125	        if r in fn.arg_regs:
   126	            continue  # Skip the arguments
   127
   128	        ctype = emitter.ctype_spaced(r.type)
   129	        init = ""
   130	        declarations.emit_line(
   131	            "{ctype}{prefix}{name}{init};".format(
   132	                ctype=ctype, prefix=REG_PREFIX, name=names[r], init=init
   133	            )
   134	        )
   135
   136	    # Before we emit the blocks, give them all labels
   137	    blocks = fn.blocks
   138	    for i, block in enumerate(blocks):
   139	        block.label = i
   140
   141	    # Find blocks that are never jumped to or are only jumped to from the
   142	    # block directly above it. This allows for more labels and gotos to be
   143	    # eliminated during code generation.
   144	    for block in fn.blocks:
   145	        terminator = block.terminator
   146	        assert isinstance(terminator, ControlOp)
   147
   148	        for target in terminator.targets():
   149	            is_next_block = target.label == block.label + 1
   150
   151	            # Always emit labels for GetAttr error checks since the emit code that
   152	            # generates them will add instructions between the branch and the
   153	            # next label, causing the label to be wrongly removed. A better
   154	            # solution would be to change the IR so that it adds a basic block
   155	            # in between the calls.
   156	            is_problematic_op = isinstance(terminator, Branch) and any(
   157	                isinstance(s, GetAttr) for s in terminator.sources()
   158	            )
   159
   160	            if not is_next_block or is_problematic_op:
   161	                fn.blocks[target.label].referenced = True
   162
   163	    common = frequently_executed_blocks(fn.blocks[0])
   164
   165	    for i in range(len(blocks)):
   166	        block = blocks[i]
   167	        visitor.rare = block not in common
   168	        next_block = None
   169	        if i + 1 < len(blocks):
   170	            next_block = blocks[i + 1]
   171	        body.emit_label(block)
   172	        visitor.next_block = next_block
   173
   174	        ops = block.ops
   175	        visitor.ops = ops
   176	        visitor.op_index = 0
   177	        while visitor.op_index < len(ops):
   178	            ops[visitor.op_index].accept(visitor)
   179	            visitor.op_index += 1

   184	    emitter.emit_from_emitter(body)

   187	class FunctionEmitterVisitor(OpVisitor[None]):
   188	    def __init__(
   189	        self, emitter: Emitter, declarations: Emitter, source_path: str, module_name: str
   190	    ) -> None:
   191	        self.emitter = emitter
   192	        self.names = emitter.names
   193	        self.declarations = declarations
   194	        self.source_path = source_path
   195	        self.module_name = module_name
   196	        self.literals = emitter.context.literals
   197	        self.rare = False
   198	        # Next basic block to be processed after the current one (if any), set by caller
   199	        self.next_block: BasicBlock | None = None
   200	        # Ops in the basic block currently being processed, set by caller
   201	        self.ops: list[Op] = []
   202	        # Current index within ops; visit methods can increment this to skip/merge ops
   203	        self.op_index = 0

   205	    def temp_name(self) -> str:
   206	        return self.emitter.temp_name()

   208	    def visit_goto(self, op: Goto) -> None:
   209	        if op.label is not self.next_block:
   210	            self.emit_line("goto %s;" % self.label(op.label))
   211
   212	    def visit_branch(self, op: Branch) -> None:
   213	        true, false = op.true, op.false
   214	        negated = op.negated
   215	        negated_rare = False
   216	        if true is self.next_block and op.traceback_entry is None:
   217	            # Switch true/false since it avoids an else block.
   218	            true, false = false, true
   219	            negated = not negated
   220	            negated_rare = True
   221
   222	        neg = "!" if negated else ""
   223	        cond = ""
   224	        if op.op == Branch.BOOL:
   225	            expr_result = self.reg(op.value)
   226	            cond = f"{neg}{expr_result}"
   227	        elif op.op == Branch.IS_ERROR:
   228	            typ = op.value.type
   229	            compare = "!=" if negated else "=="
   230	            if isinstance(typ, RTuple):
   231	                # TODO: What about empty tuple?
   232	                cond = self.emitter.tuple_undefined_check_cond(
   233	                    typ, self.reg(op.value), self.c_error_value, compare
   234	                )
   235	            else:
   236	                cond = f"{self.reg(op.value)} {compare} {self.c_error_value(typ)}"
   237	        else:
   238	            assert False, "Invalid branch"
   239
   240	        # For error checks, tell the compiler the branch is unlikely
   241	        if op.traceback_entry is not None or op.rare:
   242	            if not negated_rare:
   243	                cond = f"unlikely({cond})"
   244	            else:
   245	                cond = f"likely({cond})"
   246
   247	        if false is self.next_block:
   248	            if op.traceback_entry is None:
   249	                if true is not self.next_block:
   250	                    self.emit_line(f"if ({cond}) goto {self.label(true)};")
   251	            else:
   252	                self.emit_line(f"if ({cond}) {{")
   253	                self.emit_traceback(op)
   254	                self.emit_lines("goto %s;" % self.label(true), "}")
   255	        else:
   256	            self.emit_line(f"if ({cond}) {{")
   257	            self.emit_traceback(op)
   258
   259	            if true is not self.next_block:
   260	                self.emit_line("goto %s;" % self.label(true))
   261
   262	            self.emit_lines("} else", "    goto %s;" % self.label(false))

   300	                dest,
   301	                len(op.src),
   302	                c_array_initializer([self.reg(s) for s in op.src], indented=True),
   303	            )
   304	        )

   306	    def visit_load_error_value(self, op: LoadErrorValue) -> None:
   307	        if isinstance(op.type, RTuple):

   311	            self.emit_line(f"{self.reg(op)} = {tmp};")
   312	        else:
   313	            self.emit_line(f"{self.reg(op)} = {self.c_error_value(op.type)};")

   319	        else:

   322	            )

   324	    def get_attr_expr(self, obj: str, op: GetAttr | SetAttr, decl_cl: ClassIR) -> str:
   325	        """Generate attribute accessor for normal (non-property) access.
   326
   327	        This either has a form like obj->attr_name for attributes defined in non-trait
   328	        classes, and *(obj + attr_offset) for attributes defined by traits. We also
   329	        insert all necessary C casts here.
   330	        """
   331	        cast = f"({op.class_type.struct_name(self.emitter.names)} *)"
   332	        if decl_cl.is_trait and op.class_type.class_ir.is_trait:
   333	            # For pure trait access find the offset first, offsets
   334	            # are ordered by attribute position in the cl.attributes dict.
   335	            # TODO: pre-calculate the mapping to make this faster.
   336	            trait_attr_index = list(decl_cl.attributes).index(op.attr)
   337	            # TODO: reuse these names somehow?
   338	            offset = self.emitter.temp_name()
   339	            self.declarations.emit_line(f"size_t {offset};")
   340	            self.emitter.emit_line(
   341	                "{} = {};".format(
   342	                    offset,
   343	                    "CPy_FindAttrOffset({}, {}, {})".format(
   344	                        self.emitter.type_struct_name(decl_cl),
   345	                        f"({cast}{obj})->vtable",
   346	                        trait_attr_index,
   347	                    ),
   348	                )
   349	            )
   350	            attr_cast = f"({self.ctype(op.class_type.attr_type(op.attr))} *)"
   351	            return f"*{attr_cast}((char *){obj} + {offset})"
   352	        else:
   353	            # Cast to something non-trait. Note: for this to work, all struct
   354	            # members for non-trait classes must obey monotonic linear growth.
   355	            if op.class_type.class_ir.is_trait:
   356	                assert not decl_cl.is_trait
   357	                cast = f"({decl_cl.struct_name(self.emitter.names)} *)"
   358	            return f"({cast}{obj})->{self.emitter.attr(op.attr)}"
   359
   360	    def visit_get_attr(self, op: GetAttr) -> None:
   361	        if op.allow_null:
   362	            self.get_attr_with_allow_null(op)
   363	            return
   364	        dest = self.reg(op)
   365	        obj = self.reg(op.obj)
   366	        rtype = op.class_type
   367	        cl = rtype.class_ir
   368	        attr_rtype, decl_cl = cl.attr_details(op.attr)
   369	        prefer_method = cl.is_trait and attr_rtype.error_overlap
   370	        if cl.get_method(op.attr, prefer_method=prefer_method):
   371	            # Properties are essentially methods, so use vtable access for them.
   372	            if cl.is_method_final(op.attr):
   373	                self.emit_method_call(f"{dest} = ", op.obj, op.attr, [])
   374	            else:
   375	                version = "_TRAIT" if cl.is_trait else ""
   376	                self.emit_line(
   377	                    "%s = CPY_GET_ATTR%s(%s, %s, %d, %s, %s); /* %s */"
   378	                    % (
   379	                        dest,
   380	                        version,
   381	                        obj,
   382	                        self.emitter.type_struct_name(rtype.class_ir),
   383	                        rtype.getter_index(op.attr),
   384	                        rtype.struct_name(self.names),
   385	                        self.ctype(rtype.attr_type(op.attr)),
   386	                        op.attr,
   387	                    )
   388	                )
   389	        else:
   390	            # Otherwise, use direct or offset struct access.
   391	            attr_expr = self.get_attr_expr(obj, op, decl_cl)
   392	            self.emitter.emit_line(f"{dest} = {attr_expr};")
   393	            always_defined = cl.is_always_defined(op.attr)
   394	            merged_branch = None
   395	            if not always_defined:
   396	                self.emitter.emit_undefined_attr_check(
   397	                    attr_rtype, dest, "==", obj, op.attr, cl, unlikely=True
   398	                )
   399	                branch = self.next_branch()
   400	                if branch is not None:
   401	                    if (
   402	                        branch.value is op
   403	                        and branch.op == Branch.IS_ERROR
   404	                        and branch.traceback_entry is not None
   405	                        and not branch.negated
   406	                    ):
   407	                        # Generate code for the following branch here to avoid
   408	                        # redundant branches in the generated code.
   409	                        self.emit_attribute_error(branch, cl.name, op.attr)
   410	                        self.emit_line("goto %s;" % self.label(branch.true))
   411	                        merged_branch = branch
   412	                        self.emitter.emit_line("}")
   413	                if not merged_branch:
   414	                    exc_class = "PyExc_AttributeError"
   415	                    self.emitter.emit_line(
   416	                        'PyErr_SetString({}, "attribute {} of {} undefined");'.format(
   417	                            exc_class, repr(op.attr), repr(cl.name)
   418	                        )
   419	                    )
   420
   421	            if attr_rtype.is_refcounted and not op.is_borrowed:
   422	                if not merged_branch and not always_defined:
   423	                    self.emitter.emit_line("} else {")
   424	                self.emitter.emit_inc_ref(dest, attr_rtype)
   425	            if merged_branch:
   426	                if merged_branch.false is not self.next_block:
   427	                    self.emit_line("goto %s;" % self.label(merged_branch.false))
   428	                self.op_index += 1
   429	            elif not always_defined:
   430	                self.emitter.emit_line("}")
   431
   432	    def get_attr_with_allow_null(self, op: GetAttr) -> None:
   433	        """Handle GetAttr with allow_null=True which allows NULL without raising AttributeError."""
   434	        dest = self.reg(op)
   435	        obj = self.reg(op.obj)
   436	        rtype = op.class_type
   437	        cl = rtype.class_ir
   438	        attr_rtype, decl_cl = cl.attr_details(op.attr)
   439
   440	        # Direct struct access without NULL check
   441	        attr_expr = self.get_attr_expr(obj, op, decl_cl)
   442	        self.emitter.emit_line(f"{dest} = {attr_expr};")
   443
   444	        # Only emit inc_ref if not NULL
   445	        if attr_rtype.is_refcounted and not op.is_borrowed:
   446	            self.emitter.emit_line(f"if ({dest} != NULL) {{")
   447	            self.emitter.emit_inc_ref(dest, attr_rtype)
   448	            self.emitter.emit_line("}")

   450	    def next_branch(self) -> Branch | None:
   451	        if self.op_index + 1 < len(self.ops):
   452	            next_op = self.ops[self.op_index + 1]
   453	            if isinstance(next_op, Branch):
   454	                return next_op
   455	        return None

   457	    def visit_set_attr(self, op: SetAttr) -> None:
   458	        if op.error_kind == ERR_FALSE:
   459	            dest = self.reg(op)
   460	        obj = self.reg(op.obj)
   461	        src = self.reg(op.src)
   462	        rtype = op.class_type
   463	        cl = rtype.class_ir
   464	        attr_rtype, decl_cl = cl.attr_details(op.attr)
   465	        if cl.get_method(op.attr):
   466	            # Again, use vtable access for properties...
   467	            assert not op.is_init and op.error_kind == ERR_FALSE, "%s %d %d %s" % (
   468	                op.attr,
   469	                op.is_init,
   470	                op.error_kind,
   471	                rtype,
   472	            )
   473	            version = "_TRAIT" if cl.is_trait else ""
   474	            self.emit_line(
   475	                "%s = CPY_SET_ATTR%s(%s, %s, %d, %s, %s, %s); /* %s */"
   476	                % (
   477	                    dest,
   478	                    version,
   479	                    obj,
   480	                    self.emitter.type_struct_name(rtype.class_ir),
   481	                    rtype.setter_index(op.attr),
   482	                    src,
   483	                    rtype.struct_name(self.names),
   484	                    self.ctype(rtype.attr_type(op.attr)),
   485	                    op.attr,
   486	                )
   487	            )
   488	        else:
   489	            # ...and struct access for normal attributes.
   490	            attr_expr = self.get_attr_expr(obj, op, decl_cl)
   491	            if not op.is_init and attr_rtype.is_refcounted:
   492	                # This is not an initialization (where we know that the attribute was
   493	                # previously undefined), so decref the old value.
   494	                always_defined = cl.is_always_defined(op.attr)
   495	                if not always_defined:
   496	                    self.emitter.emit_undefined_attr_check(
   497	                        attr_rtype, attr_expr, "!=", obj, op.attr, cl
   498	                    )
   499	                self.emitter.emit_dec_ref(attr_expr, attr_rtype)
   500	                if not always_defined:
   501	                    self.emitter.emit_line("}")
   502	            elif attr_rtype.error_overlap and not cl.is_always_defined(op.attr):
   503	                # If there is overlap with the error value, update bitmap to mark
   504	                # attribute as defined.
   505	                self.emitter.emit_attr_bitmap_set(src, obj, attr_rtype, cl, op.attr)
   506
   507	            # This steals the reference to src, so we don't need to increment the arg
   508	            self.emitter.emit_line(f"{attr_expr} = {src};")
   509	            if op.error_kind == ERR_FALSE:
   510	                self.emitter.emit_line(f"{dest} = 1;")

   562	    def emit_method_call(self, dest: str, op_obj: Value, name: str, op_args: list[Value]) -> None:
   563	        obj = self.reg(op_obj)
   564	        rtype = op_obj.type
   565	        assert isinstance(rtype, RInstance)
   566	        class_ir = rtype.class_ir
   567	        method = rtype.class_ir.get_method(name)
   568	        assert method is not None
   569
   570	        # Can we call the method directly, bypassing vtable?
   571	        is_direct = class_ir.is_method_final(name)
   572
   573	        # The first argument gets omitted for static methods and
   574	        # turned into the class for class methods
   575	        obj_args = (
   576	            []
   577	            if method.decl.kind == FUNC_STATICMETHOD
   578	            else [f"(PyObject *)Py_TYPE({obj})"] if method.decl.kind == FUNC_CLASSMETHOD else [obj]
   579	        )
   580	        args = ", ".join(obj_args + [self.reg(arg) for arg in op_args])
   581	        mtype = native_function_type(method, self.emitter)
   582	        version = "_TRAIT" if rtype.class_ir.is_trait else ""
   583	        if is_direct:
   584	            # Directly call method, without going through the vtable.
   585	            lib = self.emitter.get_group_prefix(method.decl)
   586	            self.emit_line(f"{dest}{lib}{NATIVE_PREFIX}{method.cname(self.names)}({args});")
   587	        else:
   588	            # Call using vtable.
   589	            method_idx = rtype.method_index(name)
   590	            self.emit_line(
   591	                "{}CPY_GET_METHOD{}({}, {}, {}, {}, {})({}); /* {} */".format(
   592	                    dest,
   593	                    version,
   594	                    obj,
   595	                    self.emitter.type_struct_name(rtype.class_ir),
   596	                    method_idx,
   597	                    rtype.struct_name(self.names),
   598	                    mtype,
   599	                    args,
   600	                    name,
   601	                )
   602	            )

   604	    def visit_inc_ref(self, op: IncRef) -> None:
   605	        if (
   606	            isinstance(op.src, Box)
   607	            and (is_none_rprimitive(op.src.src.type) or is_bool_rprimitive(op.src.src.type))
   608	            and HAVE_IMMORTAL
   609	        ):
   610	            # On Python 3.12+, None/True/False are immortal, and we can skip inc ref
   611	            return
   612
   613	        if isinstance(op.src, LoadLiteral) and HAVE_IMMORTAL:
   614	            value = op.src.value
   615	            # We can skip inc ref for immortal literals on Python 3.12+
   616	            if type(value) is int and -5 <= value <= 256:
   617	                # Small integers are immortal
   618	                return
   619
   620	        src = self.reg(op.src)
   621	        self.emit_inc_ref(src, op.src.type)

   630	    def visit_cast(self, op: Cast) -> None:
   631	        branch = self.next_branch()
   632	        handler = None
   633	        if branch is not None:
   634	            if (
   635	                branch.value is op
   636	                and branch.op == Branch.IS_ERROR
   637	                and branch.traceback_entry is not None
   638	                and not branch.negated
   639	                and branch.false is self.next_block
   640	            ):

   642	                # redundant branches in the generated code.

   644	                    self.label(branch.true),
   645	                    self.source_path,
   646	                    self.module_name,
   647	                    branch.traceback_entry,
   648	                )

   652	            self.reg(op.src), self.reg(op), op.type, src_type=op.src.type, error=handler

   721	    def visit_comparison_op(self, op: ComparisonOp) -> None:
   722	        dest = self.reg(op)
   723	        lhs = self.reg(op.lhs)
   724	        rhs = self.reg(op.rhs)
   725	        lhs_cast = ""
   726	        rhs_cast = ""
   727	        if op.op in (ComparisonOp.SLT, ComparisonOp.SGT, ComparisonOp.SLE, ComparisonOp.SGE):
   728	            # Always signed comparison op
   729	            lhs_cast = self.emit_signed_int_cast(op.lhs.type)
   730	            rhs_cast = self.emit_signed_int_cast(op.rhs.type)
   731	        elif op.op in (ComparisonOp.ULT, ComparisonOp.UGT, ComparisonOp.ULE, ComparisonOp.UGE):
   732	            # Always unsigned comparison op
   733	            lhs_cast = self.emit_unsigned_int_cast(op.lhs.type)
   734	            rhs_cast = self.emit_unsigned_int_cast(op.rhs.type)
   735	        elif isinstance(op.lhs, Integer) and op.lhs.value < 0:
   736	            # Force signed ==/!= with negative operand
   737	            rhs_cast = self.emit_signed_int_cast(op.rhs.type)
   738	        elif isinstance(op.rhs, Integer) and op.rhs.value < 0:
   739	            # Force signed ==/!= with negative operand
   740	            lhs_cast = self.emit_signed_int_cast(op.lhs.type)
   741	        self.emit_line(f"{dest} = {lhs_cast}{lhs} {op.op_str[op.op]} {rhs_cast}{rhs};")

   106
   107	    Basically the point of this plugin is to force mypy to recheck things
   108	    based on the demands of mypyc in a couple situations:

   120	        self.group_map: dict[str, tuple[str | None, list[str]]] = {}
   121	        for sources, name in groups:
   122	            modules = sorted(source.module for source in sources)
   123	            for id in modules:
   124	                self.group_map[id] = (name, modules)

   126	        self.compiler_options = compiler_options
   127	        self.metastore = create_metastore(options)

   129	    def report_config_data(self, ctx: ReportConfigContext) -> tuple[str | None, list[str]] | None:
   130	        # The config data we report is the group map entry for the module.
   131	        # If the data is being used to check validity, we do additional checks
   132	        # that the IR cache exists and matches the metadata cache and all
   133	        # output source files exist and are up to date.
   134
   135	        id, path, is_check = ctx.id, ctx.path, ctx.is_check
   136
   137	        if id not in self.group_map:
   138	            return None
   139
   140	        # If we aren't doing validity checks, just return the cache data
   141	        if not is_check:
   142	            return self.group_map[id]
   143
   144	        # Load the metadata and IR cache
   145	        meta_path, _, _ = get_cache_names(id, path, self.options)
   146	        ir_path = get_ir_cache_name(id, path, self.options)
   147	        try:
   148	            meta_json = self.metastore.read(meta_path)
   149	            ir_json = self.metastore.read(ir_path)
   150	        except FileNotFoundError:
   151	            # This could happen if mypyc failed after mypy succeeded
   152	            # in the previous run or if some cache files got

   154	            return None

   158	        # Check that the IR cache matches the metadata cache
   159	        if hash_digest(meta_json) != ir_data["meta_hash"]:

   162	        # Check that all of the source files are present and as
   163	        # expected. The main situation where this would come up is the
   164	        # user deleting the build directory without deleting
   165	        # .mypy_cache, which we should handle gracefully.
   166	        for path, hash in ir_data["src_hashes"].items():
   167	            try:
   168	                with open(os.path.join(self.compiler_options.target_dir, path), "rb") as f:
   169	                    contents = f.read()

   171	                return None
   172	            real_hash = hash_digest(contents)
   173	            if hash != real_hash:
   174	                return None

   343	def write_cache(
   344	    modules: ModuleIRs,
   345	    result: BuildResult,
   346	    group_map: dict[str, str | None],
   347	    ctext: dict[str | None, list[tuple[str, str]]],
   348	) -> None:
   349	    """Write out the cache information for modules.
   350
   351	    Each module has the following cache information written (which is
   352	    in addition to the cache information written by mypy itself):
   353	      * A serialized version of its mypyc IR, minus the bodies of
   354	        functions. This allows code that depends on it to use
   355	        these serialized data structures when compiling against it
   356	        instead of needing to recompile it. (Compiling against a
   357	        module requires access to both its mypy and mypyc data
   358	        structures.)
   359	      * The hash of the mypy metadata cache file for the module.
   360	        This is used to ensure that the mypyc cache and the mypy
   361	        cache are in sync and refer to the same version of the code.
   362	        This is particularly important if mypyc crashes/errors/is
   363	        stopped after mypy has written its cache but before mypyc has.
   364	      * The hashes of all of the source file outputs for the group
   365	        the module is in. This is so that the module will be
   366	        recompiled if the source outputs are missing.
   367	    """
   368
   369	    hashes = {}
   370	    for name, files in ctext.items():
   371	        hashes[name] = {file: compute_hash(data) for file, data in files}
   372
   373	    # Write out cache data
   374	    for id, module in modules.items():
   375	        st = result.graph[id]
   376
   377	        meta_path, _, _ = get_cache_names(id, st.xpath, result.manager.options)
   378	        # If the metadata isn't there, skip writing the cache.
   379	        try:
   380	            meta_data = result.manager.metastore.read(meta_path)
   381	        except OSError:
   382	            continue
   383
   384	        newpath = get_state_ir_cache_name(st)
   385	        ir_data = {
   386	            "ir": module.serialize(),
   387	            "meta_hash": hash_digest(meta_data),
   388	            "src_hashes": hashes[group_map[id]],
   389	        }
   390
   391	        result.manager.metastore.write(newpath, json_dumps(ir_data))
   392
   393	    result.manager.metastore.commit()

   436	    mapper = Mapper(group_map)
   437
   438	    # Sometimes when we call back into mypy, there might be errors.
   439	    # We don't want to crash when that happens.
   440	    result.manager.errors.set_file(
   441	        "<mypyc>", module=None, scope=None, options=result.manager.options
   442	    )
   443
   444	    modules = compile_modules_to_ir(result, mapper, compiler_options, errors)
   445	    if errors.num_errors > 0:
   446	        return {}, [], Mapper({})
   447
   448	    ctext = compile_ir_to_c(groups, modules, result, mapper, compiler_options)
   449	    write_cache(modules, result, group_map, ctext)
   450
   451	    return modules, [ctext[name] for _, name in groups], mapper

   469	def pointerize(decl: str, name: str) -> str:
   470	    """Given a C decl and its name, modify it to be a declaration to a pointer."""
   471	    # This doesn't work in general but does work for all our types...
   472	    if "(" in decl:
   473	        # Function pointer. Stick an * in front of the name and wrap it in parens.
   474	        return decl.replace(name, f"(*{name})")
   475	    else:
   476	        # Non-function pointer. Just stick an * in front of the name.
   477	        return decl.replace(name, f"*{name}")

   485	class GroupGenerator:
   486	    def __init__(
   487	        self,
   488	        modules: dict[str, ModuleIR],
   489	        source_paths: dict[str, str],
   490	        group_name: str | None,
   491	        group_map: dict[str, str | None],
   492	        names: NameGenerator,
   493	        compiler_options: CompilerOptions,
   494	    ) -> None:
   495	        """Generator for C source for a compilation group.
   496
   497	        The code for a compilation group contains an internal and an
   498	        external .h file, and then one .c if not in multi_file mode or
   499	        one .c file per module if in multi_file mode.)
   500
   501	        Arguments:
   502	            modules: (name, ir) pairs for each module in the group
   503	            source_paths: Map from module names to source file paths
   504	            group_name: The name of the group (or None if this is single-module compilation)
   505	            group_map: A map of modules to their group names
   506	            names: The name generator for the compilation
   507	            multi_file: Whether to put each module in its own source file regardless
   508	                        of group structure.
   509	        """
   510	        self.modules = modules
   511	        self.source_paths = source_paths
   512	        self.context = EmitterContext(names, group_name, group_map)
   513	        self.names = names
   514	        # Initializations of globals to simple values that we can't
   515	        # do statically because the windows loader is bad.
   516	        self.simple_inits: list[tuple[str, str]] = []
   517	        self.group_name = group_name
   518	        self.use_shared_lib = group_name is not None
   519	        self.compiler_options = compiler_options
   520	        self.multi_file = compiler_options.multi_file
   521	        # Multi-phase init is needed to enable free-threading. In the future we'll
   522	        # probably want to enable it always, but we'll wait until it's stable.
   523	        self.multi_phase_init = IS_FREE_THREADED

   533	    def generate_c_for_modules(self) -> list[tuple[str, str]]:
   534	        file_contents = []
   535	        multi_file = self.use_shared_lib and self.multi_file
   536
   537	        # Collect all literal refs in IR.
   538	        for module in self.modules.values():
   539	            for fn in module.functions:
   540	                collect_literals(fn, self.context.literals)
   541
   542	        base_emitter = Emitter(self.context)
   543	        # Optionally just include the runtime library c files to
   544	        # reduce the number of compiler invocations needed
   545	        if self.compiler_options.include_runtime_files:
   546	            for name in RUNTIME_C_FILES:
   547	                base_emitter.emit_line(f'#include "{name}"')
   548	        base_emitter.emit_line(f'#include "__native{self.short_group_suffix}.h"')
   549	        base_emitter.emit_line(f'#include "__native_internal{self.short_group_suffix}.h"')
   550	        emitter = base_emitter
   551
   552	        self.generate_literal_tables()
   553
   554	        for module_name, module in self.modules.items():
   555	            if multi_file:
   556	                emitter = Emitter(self.context)
   557	                emitter.emit_line(f'#include "__native{self.short_group_suffix}.h"')
   558	                emitter.emit_line(f'#include "__native_internal{self.short_group_suffix}.h"')
   559
   560	            self.declare_module(module_name, emitter)
   561	            self.declare_internal_globals(module_name, emitter)
   562	            self.declare_imports(module.imports, emitter)
   563
   564	            for cl in module.classes:
   565	                if cl.is_ext_class:
   566	                    generate_class(cl, module_name, emitter)
   567
   568	            # Generate Python extension module definitions and module initialization functions.
   569	            self.generate_module_def(emitter, module_name, module)
   570
   571	            for fn in module.functions:
   572	                emitter.emit_line()
   573	                generate_native_function(fn, emitter, self.source_paths[module_name], module_name)
   574	                if fn.name != TOP_LEVEL_NAME and not fn.internal:
   575	                    emitter.emit_line()
   576	                    if is_fastcall_supported(fn, emitter.capi_version):
   577	                        generate_wrapper_function(
   578	                            fn, emitter, self.source_paths[module_name], module_name
   579	                        )
   580	                    else:
   581	                        generate_legacy_wrapper_function(
   582	                            fn, emitter, self.source_paths[module_name], module_name
   583	                        )
   584	            if multi_file:
   585	                name = f"__native_{exported_name(module_name)}.c"
   586	                file_contents.append((name, "".join(emitter.fragments)))
   587
   588	        # The external header file contains type declarations while
   589	        # the internal contains declarations of functions and objects
   590	        # (which are shared between shared libraries via dynamic
   591	        # exports tables and not accessed directly.)
   592	        ext_declarations = Emitter(self.context)
   593	        ext_declarations.emit_line(f"#ifndef MYPYC_NATIVE{self.group_suffix}_H")
   594	        ext_declarations.emit_line(f"#define MYPYC_NATIVE{self.group_suffix}_H")
   595	        ext_declarations.emit_line("#include <Python.h>")
   596	        ext_declarations.emit_line("#include <CPy.h>")
   597
   598	        declarations = Emitter(self.context)
   599	        declarations.emit_line(f"#ifndef MYPYC_NATIVE_INTERNAL{self.group_suffix}_H")
   600	        declarations.emit_line(f"#define MYPYC_NATIVE_INTERNAL{self.group_suffix}_H")
   601	        declarations.emit_line("#include <Python.h>")
   602	        declarations.emit_line("#include <CPy.h>")
   603	        declarations.emit_line(f'#include "__native{self.short_group_suffix}.h"')
   604	        declarations.emit_line()
   605	        declarations.emit_line("int CPyGlobalsInit(void);")
   606	        declarations.emit_line()
   607
   608	        for module_name, module in self.modules.items():
   609	            self.declare_finals(module_name, module.final_names, declarations)
   610	            for cl in module.classes:
   611	                generate_class_type_decl(cl, emitter, ext_declarations, declarations)
   612	            self.declare_type_vars(module_name, module.type_var_names, declarations)
   613	            for fn in module.functions:
   614	                generate_function_declaration(fn, declarations)
   615
   616	        for lib in sorted(self.context.group_deps):
   617	            elib = exported_name(lib)
   618	            short_lib = exported_name(lib.split(".")[-1])
   619	            declarations.emit_lines(
   620	                "#include <{}>".format(os.path.join(group_dir(lib), f"__native_{short_lib}.h")),
   621	                f"struct export_table_{elib} exports_{elib};",
   622	            )
   623
   624	        sorted_decls = self.toposort_declarations()
   625
   626	        emitter = base_emitter
   627	        self.generate_globals_init(emitter)
   628
   629	        emitter.emit_line()
   630
   631	        for declaration in sorted_decls:
   632	            decls = ext_declarations if declaration.is_type else declarations
   633	            if not declaration.is_type:
   634	                decls.emit_lines(f"extern {declaration.decl[0]}", *declaration.decl[1:])
   635	                # If there is a definition, emit it. Otherwise repeat the declaration
   636	                # (without an extern).
   637	                if declaration.defn:
   638	                    emitter.emit_lines(*declaration.defn)
   639	                else:
   640	                    emitter.emit_lines(*declaration.decl)
   641	            else:
   642	                decls.emit_lines(*declaration.decl)
   643
   644	        if self.group_name:
   645	            self.generate_export_table(ext_declarations, emitter)
   646
   647	            self.generate_shared_lib_init(emitter)
   648
   649	        ext_declarations.emit_line("#endif")
   650	        declarations.emit_line("#endif")
   651
   652	        output_dir = group_dir(self.group_name) if self.group_name else ""
   653	        return file_contents + [
   654	            (
   655	                os.path.join(output_dir, f"__native{self.short_group_suffix}.c"),
   656	                "".join(emitter.fragments),
   657	            ),
   658	            (
   659	                os.path.join(output_dir, f"__native_internal{self.short_group_suffix}.h"),
   660	                "".join(declarations.fragments),
   661	            ),
   662	            (
   663	                os.path.join(output_dir, f"__native{self.short_group_suffix}.h"),
   664	                "".join(ext_declarations.fragments),
   665	            ),
   666	        ]

   668	    def generate_literal_tables(self) -> None:
   669	        """Generate tables containing descriptions of Python literals to construct.
   670
   671	        We will store the constructed literals in a single array that contains
   672	        literals of all types. This way we can refer to an arbitrary literal by
   673	        its index.
   674	        """
   675	        literals = self.context.literals
   676	        # During module initialization we store all the constructed objects here
   677	        self.declare_global("PyObject *[%d]" % literals.num_literals(), "CPyStatics")
   678	        # Descriptions of str literals
   679	        init_str = c_string_array_initializer(literals.encoded_str_values())
   680	        self.declare_global("const char * const []", "CPyLit_Str", initializer=init_str)
   681	        # Descriptions of bytes literals
   682	        init_bytes = c_string_array_initializer(literals.encoded_bytes_values())
   683	        self.declare_global("const char * const []", "CPyLit_Bytes", initializer=init_bytes)
   684	        # Descriptions of int literals
   685	        init_int = c_string_array_initializer(literals.encoded_int_values())
   686	        self.declare_global("const char * const []", "CPyLit_Int", initializer=init_int)
   687	        # Descriptions of float literals
   688	        init_floats = c_array_initializer(literals.encoded_float_values())
   689	        self.declare_global("const double []", "CPyLit_Float", initializer=init_floats)
   690	        # Descriptions of complex literals
   691	        init_complex = c_array_initializer(literals.encoded_complex_values())
   692	        self.declare_global("const double []", "CPyLit_Complex", initializer=init_complex)
   693	        # Descriptions of tuple literals
   694	        init_tuple = c_array_initializer(literals.encoded_tuple_values())
   695	        self.declare_global("const int []", "CPyLit_Tuple", initializer=init_tuple)
   696	        # Descriptions of frozenset literals
   697	        init_frozenset = c_array_initializer(literals.encoded_frozenset_values())
   698	        self.declare_global("const int []", "CPyLit_FrozenSet", initializer=init_frozenset)

   700	    def generate_export_table(self, decl_emitter: Emitter, code_emitter: Emitter) -> None:
   701	        """Generate the declaration and definition of the group's export struct.
   702
   703	        To avoid needing to deal with deeply platform specific issues
   704	        involving dynamic library linking (and some possibly
   705	        insurmountable issues involving cyclic dependencies), compiled
   706	        code accesses functions and data in other compilation groups
   707	        via an explicit "export struct".
   708
   709	        Each group declares a struct type that contains a pointer to
   710	        every function and static variable it exports. It then
   711	        populates this struct and stores a pointer to it in a capsule
   712	        stored as an attribute named 'exports' on the group's shared
   713	        library's python module.
   714
   715	        On load, a group's init function will import all of its
   716	        dependencies' exports tables using the capsule mechanism and
   717	        copy the contents into a local copy of the table (to eliminate
   718	        the need for a pointer indirection when accessing it).
   719
   720	        Then, all calls to functions in another group and accesses to statics
   721	        from another group are done indirectly via the export table.
   722
   723	        For example, a group containing a module b, where b contains a class B
   724	        and a function bar, would declare an export table like:
   725	            struct export_table_b {
   726	                PyTypeObject **CPyType_B;
   727	                PyObject *(*CPyDef_B)(CPyTagged cpy_r_x);
   728	                CPyTagged (*CPyDef_B___foo)(PyObject *cpy_r_self, CPyTagged cpy_r_y);
   729	                tuple_T2OI (*CPyDef_bar)(PyObject *cpy_r_x);
   730	                char (*CPyDef___top_level__)(void);
   731	            };
   732	        that would be initialized with:
   733	            static struct export_table_b exports = {
   734	                &CPyType_B,
   735	                &CPyDef_B,
   736	                &CPyDef_B___foo,
   737	                &CPyDef_bar,
   738	                &CPyDef___top_level__,
   739	            };
   740	        To call `b.foo`, then, a function in another group would do
   741	        `exports_b.CPyDef_bar(...)`.
   742	        """
   743
   744	        decls = decl_emitter.context.declarations
   745
   746	        decl_emitter.emit_lines("", f"struct export_table{self.group_suffix} {{")
   747	        for name, decl in decls.items():
   748	            if decl.needs_export:
   749	                decl_emitter.emit_line(pointerize("\n".join(decl.decl), name))
   750
   751	        decl_emitter.emit_line("};")
   752
   753	        code_emitter.emit_lines("", f"static struct export_table{self.group_suffix} exports = {{")
   754	        for name, decl in decls.items():
   755	            if decl.needs_export:
   756	                code_emitter.emit_line(f"&{name},")
   757
   758	        code_emitter.emit_line("};")

   760	    def generate_shared_lib_init(self, emitter: Emitter) -> None:
   761	        """Generate the init function for a shared library.
   762
   763	        A shared library contains all of the actual code for a
   764	        compilation group.
   765
   766	        The init function is responsible for creating Capsules that
   767	        wrap pointers to the initialization function of all the real
   768	        init functions for modules in this shared library as well as
   769	        the export table containing all of the exported functions and
   770	        values from all the modules.
   771
   772	        These capsules are stored in attributes of the shared library.
   773	        """
   774	        assert self.group_name is not None
   775
   776	        emitter.emit_line()
   777	        emitter.emit_lines(
   778	            "PyMODINIT_FUNC PyInit_{}(void)".format(
   779	                shared_lib_name(self.group_name).split(".")[-1]
   780	            ),
   781	            "{",
   782	            (
   783	                'static PyModuleDef def = {{ PyModuleDef_HEAD_INIT, "{}", NULL, -1, NULL, NULL }};'.format(
   784	                    shared_lib_name(self.group_name)
   785	                )
   786	            ),
   787	            "int res;",
   788	            "PyObject *capsule;",
   789	            "PyObject *tmp;",
   790	            "static PyObject *module;",
   791	            "if (module) {",
   792	            "Py_INCREF(module);",
   793	            "return module;",
   794	            "}",
   795	            "module = PyModule_Create(&def);",
   796	            "if (!module) {",
   797	            "goto fail;",
   798	            "}",
   799	            "",
   800	        )
   801
   802	        emitter.emit_lines(
   803	            'capsule = PyCapsule_New(&exports, "{}.exports", NULL);'.format(
   804	                shared_lib_name(self.group_name)
   805	            ),
   806	            "if (!capsule) {",
   807	            "goto fail;",
   808	            "}",
   809	            'res = PyObject_SetAttrString(module, "exports", capsule);',
   810	            "Py_DECREF(capsule);",
   811	            "if (res < 0) {",
   812	            "goto fail;",
   813	            "}",
   814	            "",
   815	        )
   816
   817	        for mod in self.modules:
   818	            name = exported_name(mod)
   819	            emitter.emit_lines(
   820	                f"extern PyObject *CPyInit_{name}(void);",
   821	                'capsule = PyCapsule_New((void *)CPyInit_{}, "{}.init_{}", NULL);'.format(
   822	                    name, shared_lib_name(self.group_name), name
   823	                ),
   824	                "if (!capsule) {",
   825	                "goto fail;",
   826	                "}",
   827	                f'res = PyObject_SetAttrString(module, "init_{name}", capsule);',
   828	                "Py_DECREF(capsule);",
   829	                "if (res < 0) {",
   830	                "goto fail;",
   831	                "}",
   832	                "",
   833	            )
   834
   835	        for group in sorted(self.context.group_deps):
   836	            egroup = exported_name(group)
   837	            emitter.emit_lines(
   838	                'tmp = PyImport_ImportModule("{}"); if (!tmp) goto fail; Py_DECREF(tmp);'.format(
   839	                    shared_lib_name(group)
   840	                ),
   841	                'struct export_table_{} *pexports_{} = PyCapsule_Import("{}.exports", 0);'.format(
   842	                    egroup, egroup, shared_lib_name(group)
   843	                ),
   844	                f"if (!pexports_{egroup}) {{",
   845	                "goto fail;",
   846	                "}",
   847	                "memcpy(&exports_{group}, pexports_{group}, sizeof(exports_{group}));".format(
   848	                    group=egroup
   849	                ),
   850	                "",
   851	            )
   852
   853	        emitter.emit_lines("return module;", "fail:", "Py_XDECREF(module);", "return NULL;", "}")
   854

   876	    def generate_module_def(self, emitter: Emitter, module_name: str, module: ModuleIR) -> None:
   877	        """Emit the PyModuleDef struct for a module and the module init function."""
   878	        module_prefix = emitter.names.private_name(module_name)

   880	        if self.multi_phase_init:
   881	            self.emit_module_def_slots(emitter, module_prefix)
   882	        self.emit_module_methods(emitter, module_name, module_prefix, module)
   883	        self.emit_module_def_struct(emitter, module_name, module_prefix)
   884	        self.emit_module_init_func(emitter, module_name, module_prefix)

   886	    def emit_module_def_slots(self, emitter: Emitter, module_prefix: str) -> None:
   887	        name = f"{module_prefix}_slots"
   888	        exec_name = f"{module_prefix}_exec"
   889
   890	        emitter.emit_line(f"static PyModuleDef_Slot {name}[] = {{")
   891	        emitter.emit_line(f"{{Py_mod_exec, {exec_name}}},")
   892	        if sys.version_info >= (3, 12):
   893	            # Multiple interpreter support requires not using any C global state,
   894	            # which we don't support yet.
   895	            emitter.emit_line(
   896	                "{Py_mod_multiple_interpreters, Py_MOD_MULTIPLE_INTERPRETERS_NOT_SUPPORTED},"
   897	            )
   898	        if sys.version_info >= (3, 13):
   899	            # Declare support for free-threading to enable experimentation,
   900	            # even if we don't properly support it.
   901	            emitter.emit_line("{Py_mod_gil, Py_MOD_GIL_NOT_USED},")
   902	        emitter.emit_line("{0, NULL},")
   903	        emitter.emit_line("};")

   905	    def emit_module_methods(
   906	        self, emitter: Emitter, module_name: str, module_prefix: str, module: ModuleIR
   907	    ) -> None:
   908	        """Emit module methods (the static PyMethodDef table)."""
   909	        emitter.emit_line(f"static PyMethodDef {module_prefix}module_methods[] = {{")
   910	        for fn in module.functions:
   911	            if fn.class_name is not None or fn.name == TOP_LEVEL_NAME:
   912	                continue
   913	            name = short_id_from_name(fn.name, fn.decl.shortname, fn.line)
   914	            if is_fastcall_supported(fn, emitter.capi_version):
   915	                flag = "METH_FASTCALL"
   916	            else:
   917	                flag = "METH_VARARGS"
   918	            emitter.emit_line(
   919	                (
   920	                    '{{"{name}", (PyCFunction){prefix}{cname}, {flag} | METH_KEYWORDS, '
   921	                    "NULL /* docstring */}},"
   922	                ).format(name=name, cname=fn.cname(emitter.names), prefix=PREFIX, flag=flag)
   923	            )
   924	        emitter.emit_line("{NULL, NULL, 0, NULL}")
   925	        emitter.emit_line("};")
   926	        emitter.emit_line()

   928	    def emit_module_def_struct(
   929	        self, emitter: Emitter, module_name: str, module_prefix: str
   930	    ) -> None:
   931	        """Emit the static module definition struct (PyModuleDef)."""
   932	        emitter.emit_lines(
   933	            f"static struct PyModuleDef {module_prefix}module = {{",
   934	            "PyModuleDef_HEAD_INIT,",
   935	            f'"{module_name}",',
   936	            "NULL, /* docstring */",
   937	            "0,       /* size of per-interpreter state of the module */",

   948	    def emit_module_exec_func(
   949	        self, emitter: Emitter, module_name: str, module_prefix: str, module: ModuleIR
   950	    ) -> None:
   951	        """Emit the module init function.
   952
   953	        If we are compiling just one module, this will be the C API init
   954	        function. If we are compiling 2+ modules, we generate a shared
   955	        library for the modules and shims that call into the shared
   956	        library, and in this case we use an internal module initialized
   957	        function that will be called by the shim.
   958	        """
   959	        declaration = f"static int {module_prefix}_exec(PyObject *module)"
   960	        module_static = self.module_internal_static_name(module_name, emitter)
   961	        emitter.emit_lines(declaration, "{")
   962	        emitter.emit_line("PyObject* modname = NULL;")
   963	        if self.multi_phase_init:
   964	            emitter.emit_line(f"{module_static} = module;")
   965	        emitter.emit_line(
   966	            f'modname = PyObject_GetAttrString((PyObject *){module_static}, "__name__");'
   967	        )
   968
   969	        module_globals = emitter.static_name("globals", module_name)
   970	        emitter.emit_lines(
   971	            f"{module_globals} = PyModule_GetDict({module_static});",
   972	            f"if (unlikely({module_globals} == NULL))",
   973	            "    goto fail;",
   974	        )
   975
   976	        # HACK: Manually instantiate generated classes here
   977	        type_structs: list[str] = []
   978	        for cl in module.classes:
   979	            type_struct = emitter.type_struct_name(cl)
   980	            type_structs.append(type_struct)
   981	            if cl.is_generated:
   982	                emitter.emit_lines(
   983	                    "{t} = (PyTypeObject *)CPyType_FromTemplate("
   984	                    "(PyObject *){t}_template, NULL, modname);".format(t=type_struct)
   985	                )
   986	                emitter.emit_lines(f"if (unlikely(!{type_struct}))", "    goto fail;")
   987
   988	        emitter.emit_lines("if (CPyGlobalsInit() < 0)", "    goto fail;")
   989
   990	        self.generate_top_level_call(module, emitter)
   991
   992	        emitter.emit_lines("Py_DECREF(modname);")
   993
   994	        emitter.emit_line("return 0;")
   995	        emitter.emit_lines("fail:")
   996	        if self.multi_phase_init:
   997	            emitter.emit_lines(f"{module_static} = NULL;", "Py_CLEAR(modname);")
   998	        else:
   999	            emitter.emit_lines(f"Py_CLEAR({module_static});", "Py_CLEAR(modname);")
  1000	        for name, typ in module.final_names:
  1001	            static_name = emitter.static_name(name, module_name)
  1002	            emitter.emit_dec_ref(static_name, typ, is_xdec=True)
  1003	            undef = emitter.c_undefined_value(typ)
  1004	            emitter.emit_line(f"{static_name} = {undef};")
  1005	        # the type objects returned from CPyType_FromTemplate are all new references
  1006	        # so we have to decref them
  1007	        for t in type_structs:
  1008	            emitter.emit_line(f"Py_CLEAR({t});")
  1009	        emitter.emit_line("return -1;")
  1010	        emitter.emit_line("}")

  1012	    def emit_module_init_func(
  1013	        self, emitter: Emitter, module_name: str, module_prefix: str
  1014	    ) -> None:
  1015	        if not self.use_shared_lib:
  1016	            declaration = f"PyMODINIT_FUNC PyInit_{module_name}(void)"
  1017	        else:
  1018	            declaration = f"PyObject *CPyInit_{exported_name(module_name)}(void)"
  1019	        emitter.emit_lines(declaration, "{")
  1020
  1021	        if self.multi_phase_init:
  1022	            def_name = f"{module_prefix}module"
  1023	            emitter.emit_line(f"return PyModuleDef_Init(&{def_name});")
  1024	            emitter.emit_line("}")
  1025	            return
  1026
  1027	        exec_func = f"{module_prefix}_exec"
  1028
  1029	        # Store the module reference in a static and return it when necessary.
  1030	        # This is separate from the *global* reference to the module that will
  1031	        # be populated when it is imported by a compiled module. We want that
  1032	        # reference to only be populated when the module has been successfully
  1033	        # imported, whereas this we want to have to stop a circular import.
  1034	        module_static = self.module_internal_static_name(module_name, emitter)
  1035
  1036	        emitter.emit_lines(
  1037	            f"if ({module_static}) {{",
  1038	            f"Py_INCREF({module_static});",
  1039	            f"return {module_static};",
  1040	            "}",
  1041	        )
  1042
  1043	        emitter.emit_lines(
  1044	            f"{module_static} = PyModule_Create(&{module_prefix}module);",
  1045	            f"if (unlikely({module_static} == NULL))",
  1046	            "    goto fail;",
  1047	        )
  1048	        emitter.emit_lines(f"if ({exec_func}({module_static}) != 0)", "    goto fail;")
  1049	        emitter.emit_line(f"return {module_static};")
  1050	        emitter.emit_lines("fail:", "return NULL;")
  1051	        emitter.emit_lines("}")
