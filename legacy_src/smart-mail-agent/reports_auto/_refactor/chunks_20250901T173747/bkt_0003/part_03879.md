   810	    def _computeMaxSpace(self,size,required):
   811	        '''helper for madmen who want to put stuff inside their barcharts
   812	        basically after _computebarPositions we slide a line of length size
   813	        down the bar profile on either side of the bars to find the
   814	        maximum space. If the space at any point is >= required then we're
   815	        done. Otherwise we return the largest space location and amount.
   816	        '''
   817	        flipXY = self._flipXY
   818	        self._computeBarPositions()
   819	        lenData = len(self.data)
   820	        BP = self._barPositions
   821	        C = []
   822	        aC = C.append
   823	        if flipXY:
   824	            lo = self.x
   825	            hi = lo + self.width
   826	            end = self.y+self.height
   827	            for bp in BP:
   828	                for x, y, w, h in bp:
   829	                    v = x+w
   830	                    z = y+h
   831	                    aC((min(y,z),max(y,z), min(x,v) - lo, hi - max(x,v)))
   832	        else:
   833	            lo = self.y
   834	            hi = lo + self.height
   835	            end = self.x+self.width
   836	            for bp in BP:
   837	                for x, y, w, h in bp:
   838	                    v = y+h
   839	                    z = x+w
   840	                    aC((min(x,z), max(x,z), min(y,v) - lo, hi - max(y,v)))
   841	        C.sort()
   842	        R = [C[0]]
   843	        for c in C:
   844	            r = R[-1]
   845	            if r[0]<c[1] and c[0]<r[1]: #merge overlapping space
   846	                R[-1] = (min(r[0],c[0]),max(r[1],c[1]),min(r[2],c[2]),min(r[3],c[3]))
   847	            else:
   848	                R.append(c)
   849	        C = R
   850	        maxS = -0x7fffffff
   851	        maxP = None
   852	        nC = len(C)
   853	        for i,ci in enumerate(C):
   854	            v0 = ci[0]
   855	            v1 = v0+size
   856	            if v1>end: break
   857	            j = i
   858	            alo = ahi = 0x7fffffff
   859	            while j<nC and C[j][1]<=v1:
   860	                alo = min(C[j][2],alo)
   861	                ahi = min(C[j][3],ahi)
   862	                j += 1
   863	            if alo>ahi:
   864	                if alo>maxS:
   865	                    maxS = alo
   866	                    maxP = flipXY and (lo,v0,lo+alo,v0+size,0) or (v0,lo,v0+size,lo+alo,0)
   867	                    if maxS >= required: break
   868	            elif ahi>maxS:
   869	                maxS = ahi
   870	                maxP = flipXY and (hi-ahi,v0,hi,v0+size,1) or (v0,hi-ahi,v0+size,hi,1)
   871	                if maxS >= required: break
   872	        return maxS, maxP

   265	    def _calcHeight(self):
   266	        dy = self.dy
   267	        yGap = self.yGap
   268	        thisy = upperlefty = self.y - dy
   269	        fontSize = self.fontSize
   270	        fontName = self.fontName
   271	        ascent=getFont(fontName).face.ascent/1000.
   272	        if ascent==0: ascent=0.718 # default (from helvetica)
   273	        ascent *= fontSize
   274	        leading = fontSize*1.2
   275	        deltay = self.deltay
   276	        if not deltay: deltay = max(dy,leading)+self.autoYPadding
   277	        columnCount = 0
   278	        count = 0
   279	        lowy = upperlefty
   280	        lim = self.columnMaximum - 1
   281	        for name in self._getTexts(self.colorNamePairs):
   282	            y0 = thisy+(dy-ascent)*0.5
   283	            y = y0 - _getLineCount(name)*leading
   284	            leadingMove = 2*y0-y-thisy
   285	            newy = thisy-max(deltay,leadingMove)-yGap
   286	            lowy = min(y,newy,lowy)
   287	            if count==lim:
   288	                count = 0
   289	                thisy = upperlefty
   290	                columnCount += 1
   291	            else:
   292	                thisy = newy
   293	                count = count+1
   294	        return upperlefty - lowy
