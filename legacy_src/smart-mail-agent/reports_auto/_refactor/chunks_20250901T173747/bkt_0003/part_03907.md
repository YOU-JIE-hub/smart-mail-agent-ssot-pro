    96	def _calc_pc(V,avail):
    97	    '''check list V for percentage or * values
    98	    1) absolute values go through unchanged
    99	    2) percentages are used as weights for unconsumed space
   100	    3) if no None values were seen '*' weights are
   101	    set equally with unclaimed space
   102	    otherwise * weights are assigned as None'''
   103	    R = []
   104	    r = R.append
   105	    I = []
   106	    i = I.append
   107	    J = []
   108	    j = J.append
   109	    s = avail
   110	    w = n = 0.
   111	    for v in V:
   112	        if isinstance(v,strTypes):
   113	            v = str(v).strip()
   114	            if not v:
   115	                v = None
   116	                n += 1
   117	            elif v.endswith('%'):
   118	                v = float(v[:-1])
   119	                w += v
   120	                i(len(R))
   121	            elif v=='*':
   122	                j(len(R))
   123	            else:
   124	                v = float(v)
   125	                s -= v
   126	        elif v is None:
   127	            n += 1
   128	        else:
   129	            s -= v
   130	        r(v)
   131	    s = max(0.,s)
   132	    f = s/max(100.,w)
   133	    for i in I:
   134	        R[i] *= f
   135	        s -= R[i]

   232	    for v,(x0,x1) in reversed(sorted(((iv,ik) for ik,iv in spanCons.items()))):
   233	        if x0>=lim: continue
   234	        x1 += 1
   235	        t = sum([V[x]+M.get(x,0) for x in range(x0,x1)])
   236	        if t>=v-FUZZ: continue      #already good enough
   237	        X = [x for x in range(x0,x1) if V0[x] is None] #variable candidates
   238	        if not X: continue          #something wrong here mate
   239	        v -= t
   240	        v /= float(len(X))
   241	        for x in X:
   242	            M[x] = M.get(x,0)+v
   243	    for x,v in M.items():
   244	        V[x] += v

   393	    def identity(self, maxLen=30):
   394	        '''Identify our selves as well as possible'''
   395	        if self.ident: return self.ident
   396	        vx = None
   397	        nr = getattr(self,'_nrows','unknown')
   398	        nc = getattr(self,'_ncols','unknown')
   399	        cv = getattr(self,'_cellvalues',None)
   400	        rh = getattr(self, '_rowHeights', None)
   401	        if cv and 'unknown' not in (nr,nc):
   402	            b = 0
   403	            for i in range(nr):
   404	                for j in range(nc):
   405	                    v = cv[i][j]
   406	                    if isinstance(v,(list,tuple,Flowable)):
   407	                        if not isinstance(v,(tuple,list)): v = (v,)
   408	                        r = ''
   409	                        for vij in v:
   410	                            r = vij.identity(maxLen)
   411	                            if r and r[-4:]!='>...':
   412	                                break

   414	                            ix, jx, vx, b = i, j, r, 1
   415	                    else:
   416	                        v = v is None and '' or str(v)
   417	                        ix, jx, vx = i, j, v
   418	                        b = (vx and isinstance(v,strTypes)) and 1 or 0

   431	        return "<%s@0x%8.8X %s rows x %s cols%s>%s" % (self.__class__.__name__, id(self), nr, nc, tallest, vx)

   586	            except:
   587	                return 0

   592	    def _calc_height(self, availHeight, availWidth, H=None, W=None):
   593	        H = self._argH
   594	        if not W: W = _calc_pc(self._argW,availWidth)   #widths array
   595
   596	        hmax = lim = len(H)
   597	        longTable = self._longTableOptimize
   598
   599	        if None in H:
   600	            minRowHeights = self._minRowHeights
   601	            canv = getattr(self,'canv',None)
   602	            saved = None
   603	            #get a handy list of any cells which span rows. should be ignored for sizing
   604	            if self._spanCmds:
   605	                rowSpanCells = self._rowSpanCells
   606	                colSpanCells = self._colSpanCells
   607	                spanRanges = self._spanRanges
   608	                colpositions = self._colpositions
   609	            else:
   610	                rowSpanCells = colSpanCells = ()
   611	                spanRanges = {}
   612	            if canv: saved = canv._fontname, canv._fontsize, canv._leading
   613	            H0 = H
   614	            H = H[:]    #make a copy as we'll change it
   615	            self._rowHeights = H
   616	            spanCons = {}
   617	            FUZZ = rl_config._FUZZ
   618	            while None in H:
   619	                i = H.index(None)
   620	                V = self._cellvalues[i] # values for row i
   621	                S = self._cellStyles[i] # styles for row i
   622	                h = 0
   623	                j = 0
   624	                for j,(v, s, w) in enumerate(list(zip(V, S, W))): # value, style, width (lengths must match)
   625	                    ji = j,i
   626	                    span = spanRanges.get(ji,None)
   627	                    if ji in rowSpanCells and not span:
   628	                        continue # don't count it, it's either occluded or unreliable
   629	                    else:
   630	                        if isinstance(v,(tuple,list,Flowable)):
   631	                            if isinstance(v,Flowable): v = (v,)
   632	                            else: v = flatten(v)
   633	                            v = V[j] = self._cellListProcess(v,w,None)
   634	                            if w is None and not self._canGetWidth(v):
   635	                                raise ValueError(f'Flowable {v[0].identity()} in cell({i},{j}) can\'t have auto width\n{self.identity(30)}')
   636	                            if canv: canv._fontname, canv._fontsize, canv._leading = s.fontname, s.fontsize, s.leading or 1.2*s.fontsize
   637	                            if ji in colSpanCells:
   638	                                if not span: continue
   639	                                w = max(colpositions[span[2]+1]-colpositions[span[0]],w or 0)
   640	                            dW,t = self._listCellGeom(v,w or self._listValueWidth(v),s)
   641	                            if canv: canv._fontname, canv._fontsize, canv._leading = saved
   642	                            dW = dW + s.leftPadding + s.rightPadding
   643	                            if not rl_config.allowTableBoundsErrors and dW>w:
   644	                                from reportlab.platypus.doctemplate import LayoutError
   645	                                raise LayoutError("Flowable %s (%sx%s points) too wide for cell(%d,%d) (%sx* points) in\n%s" % (v[0].identity(30),fp_str(dW),fp_str(t),i,j, fp_str(w), self.identity(30)))
   646	                        else:
   647	                            v = (v is not None and str(v) or '').split("\n")
   648	                            t = (s.leading or 1.2*s.fontsize)*len(v)
   649	                        t += s.bottomPadding+s.topPadding
   650	                        if span:
   651	                            r0 = span[1]
   652	                            r1 = span[3]
   653	                            if r0!=r1:
   654	                                x = r0,r1
   655	                                spanCons[x] = max(spanCons.get(x,t),t)
   656	                                t = 0
   657	                    if t>h: h = t   #record a new maximum
   658	                # If a minimum height has been specified use that, otherwise allow the cell to grow
   659	                H[i] = max(minRowHeights[i],h) if minRowHeights else h
   660	                # we can stop if we have filled up all available room
   661	                if longTable:
   662	                    hmax = i+1      #we computed H[i] so known len == i+1
   663	                    height = sum(H[:hmax])
   664	                    if height > availHeight:
   665	                        #we can terminate if all spans are complete in H[:hmax]
   666	                        if spanCons:
   667	                            msr = max(x[1] for x in spanCons.keys())    #RS=[endrowspan,.....]
   668	                            if hmax>msr:
   669	                                break
   670	            if None not in H: hmax = lim
   671
   672	            if spanCons:
   673	                try:
   674	                    spanFixDim(H0,H,spanCons)
   675	                except:
   676	                    annotateException('\nspanning problem in %s hmax=%s lim=%s avail=%s x %s\nH0=%r H=%r\nspanCons=%r' % (self.identity(),hmax,lim,availWidth,availHeight,H0,H,spanCons))
   677
   678	        #iterate backwards through the heights to get rowpositions in reversed order
   679	        self._rowpositions = j = []
   680	        height = c = 0
   681	        for i in range(hmax-1,-1,-1):
   682	            j.append(height)
   683	            y = H[i] - c
   684	            t = height + y
   685	            c = (t - height) - y
   686	            height = t
   687	        j.append(height)
   688	        self._height = height
   689	        j.reverse()     #reverse the reversed list of row positions
   690	        self._hmax = hmax

   769	    def _calcPreliminaryWidths(self, availWidth):
   770	        """Fallback algorithm for when main one fails.
   771
   772	        Where exact width info not given but things like
   773	        paragraphs might be present, do a preliminary scan
   774	        and assign some best-guess values."""
   775
   776	        W = list(self._argW) # _calc_pc(self._argW,availWidth)
   777	        #verbose = 1
   778	        totalDefined = 0.0
   779	        percentDefined = 0
   780	        percentTotal = 0
   781	        numberUndefined = 0
   782	        numberGreedyUndefined = 0
   783	        for w in W:
   784	            if w is None:
   785	                numberUndefined += 1
   786	            elif w == '*':
   787	                numberUndefined += 1
   788	                numberGreedyUndefined += 1
   789	            elif _endswith(w,'%'):
   790	                percentDefined += 1
   791	                percentTotal += float(w[:-1])
   792	            else:
   793	                assert isinstance(w,(int,float))
   794	                totalDefined = totalDefined + w
   795	        #if verbose: print('prelim width calculation.  %d columns, %d undefined width, %0.2f units remain' % (self._ncols, numberUndefined, availWidth - totalDefined))
   796
   797	        #check columnwise in each None column to see if they are sizable.
   798	        given = []
   799	        sizeable = []
   800	        unsizeable = []
   801	        minimums = {}
   802	        totalMinimum = 0
   803	        elementWidth = self._elementWidth
   804	        for colNo in range(self._ncols):
   805	            w = W[colNo]
   806	            if w is None or w=='*' or _endswith(w,'%'):
   807	                siz = 1
   808	                final = 0
   809	                for rowNo in range(self._nrows):
   810	                    value = self._cellvalues[rowNo][colNo]
   811	                    style = self._cellStyles[rowNo][colNo]
   812	                    new = elementWidth(value,style) or 0
   813	                    new += style.leftPadding+style.rightPadding
   814	                    #if verbose: print('[%d,%d] new=%r-->%r' % (rowNo,colNo,new - style.leftPadding+style.rightPadding, new))
   815	                    final = max(final, new)
   816	                    siz = siz and self._canGetWidth(value) # irrelevant now?
   817	                if siz:
   818	                    sizeable.append(colNo)
   819	                else:
   820	                    unsizeable.append(colNo)
   821	                minimums[colNo] = final
   822	                totalMinimum += final
   823	            else:
   824	                given.append(colNo)
   825	        if len(given) == self._ncols:
   826	            return
   827	        #if verbose: print('predefined width:   ',given)
   828	        #if verbose: print('uncomputable width: ',unsizeable)
   829	        #if verbose: print('computable width:   ',sizeable)
   830	        #if verbose: print('minimums=%r' % (list(sorted(list(minimums.items()))),))
   831
   832	        # how much width is left:
   833	        remaining = availWidth - (totalMinimum + totalDefined)
   834	        if remaining > 0:
   835	            # we have some room left; fill it.
   836	            definedPercentage = (totalDefined/float(availWidth))*100
   837	            percentTotal += definedPercentage
   838	            if numberUndefined and percentTotal < 100:
   839	                undefined = numberGreedyUndefined or numberUndefined
   840	                defaultWeight = (100-percentTotal)/float(undefined)
   841	                percentTotal = 100
   842	                defaultDesired = (defaultWeight/float(percentTotal))*availWidth
   843	            else:
   844	                defaultWeight = defaultDesired = 1
   845	            # we now calculate how wide each column wanted to be, and then
   846	            # proportionately shrink that down to fit the remaining available
   847	            # space.  A column may not shrink less than its minimum width,
   848	            # however, which makes this a bit more complicated.
   849	            desiredWidths = []
   850	            totalDesired = 0
   851	            effectiveRemaining = remaining
   852	            for colNo, minimum in minimums.items():
   853	                w = W[colNo]
   854	                if _endswith(w,'%'):
   855	                    desired = (float(w[:-1])/percentTotal)*availWidth
   856	                elif w == '*':
   857	                    desired = defaultDesired
   858	                else:
   859	                    desired = not numberGreedyUndefined and defaultDesired or 1
   860	                if desired <= minimum:
   861	                    W[colNo] = minimum
   862	                else:
   863	                    desiredWidths.append(
   864	                        (desired-minimum, minimum, desired, colNo))
   865	                    totalDesired += desired
   866	                    effectiveRemaining += minimum
   867	            if desiredWidths: # else we're done
   868	                # let's say we have two variable columns.  One wanted
   869	                # 88 points, and one wanted 264 points.  The first has a
   870	                # minWidth of 66, and the second of 55.  We have 71 points
   871	                # to divide up in addition to the totalMinimum (i.e.,
   872	                # remaining==71).  Our algorithm tries to keep the proportion
   873	                # of these variable columns.
   874	                #
   875	                # To do this, we add up the minimum widths of the variable
   876	                # columns and the remaining width.  That's 192.  We add up the
   877	                # totalDesired width.  That's 352.  That means we'll try to
   878	                # shrink the widths by a proportion of 192/352--.545454.
   879	                # That would make the first column 48 points, and the second
   880	                # 144 points--adding up to the desired 192.
   881	                #
   882	                # Unfortunately, that's too small for the first column.  It
   883	                # must be 66 points.  Therefore, we go ahead and save that
   884	                # column width as 88 points.  That leaves (192-88==) 104
   885	                # points remaining.  The proportion to shrink the remaining
   886	                # column is (104/264), which, multiplied  by the desired
   887	                # width of 264, is 104: the amount assigned to the remaining
   888	                # column.
   889	                proportion = effectiveRemaining/float(totalDesired)
   890	                # we sort the desired widths by difference between desired and
   891	                # and minimum values, a value called "disappointment" in the
   892	                # code.  This means that the columns with a bigger
   893	                # disappointment will have a better chance of getting more of
   894	                # the available space.
   895	                desiredWidths.sort()
   896	                finalSet = []
   897	                for disappointment, minimum, desired, colNo in desiredWidths:
   898	                    adjusted = proportion * desired
   899	                    if adjusted < minimum:
   900	                        W[colNo] = minimum
   901	                        totalDesired -= desired
   902	                        effectiveRemaining -= minimum
   903	                        if totalDesired:
   904	                            proportion = effectiveRemaining/float(totalDesired)
   905	                    else:
   906	                        finalSet.append((minimum, desired, colNo))
   907	                for minimum, desired, colNo in finalSet:
   908	                    adjusted = proportion * desired
   909	                    assert adjusted >= minimum
   910	                    W[colNo] = adjusted
   911	        else:
   912	            for colNo, minimum in minimums.items():
   913	                W[colNo] = minimum
   914	        #if verbose: print('new widths are:', W)
   915	        self._argW = self._colWidths = W
   916	        return W

   940	    def _calcSpanRanges(self):
   941	        """Work out rects for tables which do row and column spanning.
   942
   943	        This creates some mappings to let the later code determine
   944	        if a cell is part of a "spanned" range.
   945	        self._spanRanges shows the 'coords' in integers of each
   946	        'cell range', or None if it was clobbered:
   947	        (col, row) -> (col0, row0, col1, row1)
   948
   949	        Any cell not in the key is not part of a spanned region
   950	        """
   951	        self._spanRanges = spanRanges = {}
   952	        for x in range(self._ncols):
   953	            for y in range(self._nrows):
   954	                spanRanges[x,y] = (x, y, x, y)
   955	        self._colSpanCells = []
   956	        self._rowSpanCells = []
   957	        csa = self._colSpanCells.append
   958	        rsa = self._rowSpanCells.append
   959	        for (cmd, start, stop) in self._spanCmds:
   960	            x0, y0 = start
   961	            x1, y1 = stop
   962
   963	            #normalize
   964	            if x0 < 0: x0 = x0 + self._ncols
   965	            if x1 < 0: x1 = x1 + self._ncols
   966	            if y0 < 0: y0 = y0 + self._nrows
   967	            if y1 < 0: y1 = y1 + self._nrows
   968	            if x0 > x1: x0, x1 = x1, x0
   969	            if y0 > y1: y0, y1 = y1, y0
   970
   971	            if x0!=x1 or y0!=y1:
   972	                if x0!=x1: #column span
   973	                    for y in range(y0, y1+1):
   974	                        for x in range(x0,x1+1):
   975	                            csa((x,y))
   976	                if y0!=y1: #row span
   977	                    for y in range(y0, y1+1):
   978	                        for x in range(x0,x1+1):
   979	                            rsa((x,y))
   980
   981	                for y in range(y0, y1+1):
   982	                    for x in range(x0,x1+1):
   983	                        spanRanges[x,y] = None
   984	                # set the main entry
   985	                spanRanges[x0,y0] = (x0, y0, x1, y1)

  1035	    def _calcSpanRects(self):
  1036	        """Work out rects for tables which do row and column spanning.
  1037
  1038	        Based on self._spanRanges, which is already known,
  1039	        and the widths which were given or previously calculated,
  1040	        self._spanRects shows the real coords for drawing:
  1041
  1042	            (col, row) -> (x, y, width, height)
  1043
  1044	        for each cell.  Any cell which 'does not exist' as another
  1045	        has spanned over it will get a None entry on the right
  1046	        """
  1047	        spanRects = getattr(self,'_spanRects',{})
  1048	        hmax = getattr(self,'_hmax',None)
  1049	        longTable = self._longTableOptimize
  1050	        if spanRects and (longTable and hmax==self._hmax_spanRects or not longTable):
  1051	            return
  1052	        colpositions = self._colpositions
  1053	        rowpositions = self._rowpositions
  1054	        vBlocks = {}
  1055	        hBlocks = {}
  1056	        rlim = len(rowpositions)-1
  1057	        for (coord, value) in self._spanRanges.items():
  1058	            if value is None:
  1059	                spanRects[coord] = None
  1060	            else:
  1061	                try:
  1062	                    col0, row0, col1, row1 = value
  1063	                    if row1>=rlim: continue
  1064	                    col,row = coord
  1065	                    if col1-col0>0:
  1066	                        for _ in range(col0+1,col1+1):
  1067	                            vBlocks.setdefault(colpositions[_],[]).append((rowpositions[row1+1],rowpositions[row0]))
  1068	                    if row1-row0>0:
  1069	                        for _ in range(row0+1,row1+1):
  1070	                            hBlocks.setdefault(rowpositions[_],[]).append((colpositions[col0],colpositions[col1+1]))
  1071	                    x = colpositions[col0]
  1072	                    y = rowpositions[row1+1]
  1073	                    width = colpositions[col1+1] - x
  1074	                    height = rowpositions[row0] - y
  1075	                    spanRects[coord] = (x, y, width, height)
  1076	                except:
  1077	                    annotateException('\nspanning problem in %s' % (self.identity(),))
  1078
  1079	        for _ in hBlocks, vBlocks:
  1080	            for value in _.values():
  1081	                value.sort()
  1082	        self._spanRects = spanRects
  1083	        self._vBlocks = vBlocks
  1084	        self._hBlocks = hBlocks
  1085	        self._hmax_spanRects = hmax

  1510	    def _stretchCommands(self, n, cmds, oldrowcount):
  1511	        """Stretches the commands when a row is split
  1512
  1513	        The row start is sr, the row end is er.
  1514
  1515	         sr   | er  | result
  1516	        ---------------------------------------------------------------------
  1517	          <n  |  <n | Do nothing.
  1518	              | >=n | A command that spans the break, extend end.
  1519	        ---------------------------------------------------------------------
  1520	         ==n  | ==n | Zero height. Extend the end, unless it's a LINEABOVE
  1521	              |     | commands, it's between rows so do nothing.
  1522	              |     | For LINEBELOW increase both.
  1523	              |  >n | A command that spans the break, extend end.
  1524	        ---------------------------------------------------------------------
  1525	          >n  |  >n | This command comes after the break, increase both.
  1526	        ---------------------------------------------------------------------
  1527
  1528	        Summary:
  1529	        1. If er > n then increase er
  1530	        2. If sr > n then increase sr
  1531	        3. If er == n and sr < n, increase er
  1532	        4. If er == sr == n and cmd is not line, increase er
  1533
  1534	        """
  1535	        stretched = []
  1536	        for c in cmds:
  1537	            cmd, (sc,sr), (ec,er) = c[0:3]
  1538
  1539	            if sr in ("splitlast", "splitfirst") or er in ("splitlast", "splitfirst"):
  1540	                stretched.append(c)
  1541	                continue
  1542
  1543	            if er < 0:
  1544	                er += oldrowcount
  1545	            if sr < 0:
  1546	                sr += oldrowcount
  1547
  1548	            if er > n:
  1549	                er += 1
  1550	            elif  er == n:
  1551	                if sr < n or (sr == n and cmd != "LINEABOVE"):
  1552	                    er += 1
  1553
  1554	            if sr > n or (sr == n and cmd == "LINEBELOW"):
  1555	                sr += 1
  1556
  1557	            stretched.append((c[0], (sc,sr), (ec,er)) + c[3:])
  1558
  1559	        return stretched
