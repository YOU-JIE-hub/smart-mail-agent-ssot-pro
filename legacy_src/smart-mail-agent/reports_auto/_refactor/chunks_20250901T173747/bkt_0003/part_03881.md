   248	def _findOverlapRun(B,i,wrap):
   249	    '''find overlap run containing B[i]'''
   250	    n = len(B)
   251	    R = [i]
   252	    while 1:
   253	        i = R[-1]
   254	        j = (i+1)%n
   255	        if j in R or not boundsOverlap(B[i],B[j]): break
   256	        R.append(j)
   257	    while 1:
   258	        i = R[0]
   259	        j = (i-1)%n
   260	        if j in R or not boundsOverlap(B[i],B[j]): break
   261	        R.insert(0,j)

   427	    hh = height*0.5
   428	    yhh = y+hh
   429	    m = max(mlr)
   430	    for i in (0,1):
   431	        T = LR[i]
   432	        if T:
   433	            B = []
   434	            aB = B.append
   435	            S = []
   436	            aS = S.append
   437	            T.sort(key=_fPLCF)
   438	            p = 0
   439	            yh = y+height
   440	            for l in T:
   441	                data = l._origdata
   442	                inc = x+mul*(m-data['width'])
   443	                l.x += inc
   444	                G[data['index']] = l
   445	                ly = yhh+data['smid']*hh
   446	                b = data['bounds']
   447	                b2 = (b[3]-b[1])*0.5
   448	                if ly+b2>yh: ly = yh-b2
   449	                if ly-b2<y: ly = y+b2
   450	                data['bounds'] = b = (b[0],ly-b2,b[2],ly+b2)
   451	                aB(b)
   452	                l.y = ly
   453	                aS(max(0,yh-ly-b2))
   454	                yh = ly-b2
   455	                p = max(p,data['edgePad']+data['piePad'])
   456	                mel = max(mel,abs(data['smid']*(hh+data['elbowLength']))-hh)
   457	            aS(yh-y)
   458
   459	            iter = 0
   460	            nT = len(T)
   461	            while iter<30:
   462	                R = findOverlapRun(B,wrap=0)
   463	                if not R: break
   464	                nR = len(R)
   465	                if nR==nT: break
   466	                j0 = R[0]
   467	                j1 = R[-1]
   468	                jl = j1+1
   469	                sAbove = sum(S[:j0+1])
   470	                sFree = sAbove+sum(S[jl:])
   471	                sNeed = sum([b[3]-b[1] for b in B[j0:jl]])+jl-j0-(B[j0][3]-B[j1][1])
   472	                if sNeed>sFree: break
   473	                yh = B[j0][3]+sAbove*sNeed/sFree
   474	                for r in R:

   477	                    b = data['bounds']
   478	                    b2 = (b[3]-b[1])*0.5

   481	                    B[r] = data['bounds'] = (b[0],ly-b2,b[2],yh)
   482	                    yh = ly - b2 - 0.5
   483	            mlr[i] = m+p
   484	        mul = -1
   485	    return G, mlr[0], mlr[1], mel

   487	def theta0(data, direction):
   488	    fac = (2*pi)/sum(data)
   489	    rads = [d*fac for d in data]
   490
   491	    r0 = 0
   492	    hrads = []
   493	    for r in rads:
   494	        hrads.append(r0+r*0.5)
   495	        r0 += r
   496
   497	    vstar = len(data)*1e6
   498	    rstar = 0
   499	    delta = pi/36.0
   500	    for i in range(36):
   501	        r = i*delta
   502	        v = sum([abs(sin(r+a)) for a in hrads])
   503	        if v < vstar:
   504	            if direction == 'clockwise':
   505	                rstar=-r
   506	            else:
   507	                rstar=r
   508	            vstar = v
   509	    return rstar*180/pi
