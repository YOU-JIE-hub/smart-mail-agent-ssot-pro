_class_getitem__"):
   624	        # Special cased by Python, so don't bother checking
   625	        return
   626	    if inspect.isbuiltin(runtime):
   627	        # The isinstance checks don't work reliably for builtins, e.g. datetime.datetime.now, so do
   628	        # something a little hacky that seems to work well
   629	        probably_class_method = isinstance(getattr(runtime, "__self__", None), type)
   630	        if probably_class_method and not stub.is_class:
   631	            yield "runtime is a classmethod but stub is not"
   632	        if not probably_class_method and stub.is_class:
   633	            yield "stub is a classmethod but runtime is not"
   634	        return
   635
   636	    if static_runtime is MISSING:
   637	        return
   638
   639	    if isinstance(static_runtime, classmethod) and not stub.is_class:
   640	        yield "runtime is a classmethod but stub is not"
   641	    if not isinstance(static_runtime, classmethod) and stub.is_class:
   642	        yield "stub is a classmethod but runtime is not"
   643	    if isinstance(static_runtime, staticmethod) and not stub.is_static:
   644	        yield "runtime is a staticmethod but stub is not"
   645	    if not isinstance(static_runtime, staticmethod) and stub.is_static:
   646	        yield "stub is a staticmethod but runtime is not"

   649	def _verify_arg_name(
   650	    stub_arg: nodes.Argument, runtime_arg: inspect.Parameter, function_name: str
   651	) -> Iterator[str]:
   652	    """Checks whether argument names match."""
   653	    # Ignore exact names for most dunder methods
   654	    if is_dunder(function_name, exclude_special=True):
   655	        return
   656
   657	    if (
   658	        stub_arg.variable.name == runtime_arg.name
   659	        or stub_arg.variable.name.removeprefix("__") == runtime_arg.name
   660	    ):
   661	        return
   662
   663	    nonspecific_names = {"object", "args"}
   664	    if runtime_arg.name in nonspecific_names:
   665	        return
   666
   667	    def names_approx_match(a: str, b: str) -> bool:
   668	        a = a.strip("_")
   669	        b = b.strip("_")
   670	        return a.startswith(b) or b.startswith(a) or len(a) == 1 or len(b) == 1
   671
   672	    # Be more permissive about names matching for positional-only arguments
   673	    if runtime_arg.kind == inspect.Parameter.POSITIONAL_ONLY and names_approx_match(
   674	        stub_arg.variable.name, runtime_arg.name
   675	    ):
   676	        return
   677	    # This comes up with namedtuples, so ignore
   678	    if stub_arg.variable.name == "_self":
   679	        return
   680	    yield (
   681	        f'stub argument "{stub_arg.variable.name}" '
   682	        f'differs from runtime argument "{runtime_arg.name}"'
   683	    )

   686	def _verify_arg_default_value(
   687	    stub_arg: nodes.Argument, runtime_arg: inspect.Parameter
   688	) -> Iterator[str]:
   689	    """Checks whether argument default values are compatible."""
   690	    if runtime_arg.default is not inspect.Parameter.empty:
   691	        if stub_arg.kind.is_required():
   692	            yield (
   693	                f'runtime argument "{runtime_arg.name}" '
   694	                "has a default value but stub argument does not"
   695	            )
   696	        else:
   697	            runtime_type = get_mypy_type_of_runtime_value(runtime_arg.default)
   698	            # Fallback to the type annotation type if var type is missing. The type annotation
   699	            # is an UnboundType, but I don't know enough to know what the pros and cons here are.
   700	            # UnboundTypes have ugly question marks following them, so default to var type.
   701	            # Note we do this same fallback when constructing signatures in from_overloadedfuncdef
   702	            stub_type = stub_arg.variable.type or stub_arg.type_annotation
   703	            if isinstance(stub_type, mypy.types.TypeVarType):
   704	                stub_type = stub_type.upper_bound
   705	            if (
   706	                runtime_type is not None
   707	                and stub_type is not None
   708	                # Avoid false positives for marker objects
   709	                and type(runtime_arg.default) is not object
   710	                # And ellipsis
   711	                and runtime_arg.default is not ...
   712	                and not is_subtype_helper(runtime_type, stub_type)
   713	            ):
   714	                yield (
   715	                    f'runtime argument "{runtime_arg.name}" '
   716	                    f"has a default value of type {runtime_type}, "
   717	                    f"which is incompatible with stub argument type {stub_type}"
   718	                )
   719	            if stub_arg.initializer is not None:
   720	                stub_default = evaluate_expression(stub_arg.initializer)
   721	                if (
   722	                    stub_default is not UNKNOWN
   723	                    and stub_default is not ...
   724	                    and runtime_arg.default is not UNREPRESENTABLE
   725	                ):
   726	                    defaults_match = True
   727	                    # We want the types to match exactly, e.g. in case the stub has
   728	                    # True and the runtime has 1 (or vice versa).
   729	                    if type(stub_default) is not type(runtime_arg.default):
   730	                        defaults_match = False
   731	                    else:
   732	                        try:
   733	                            defaults_match = bool(stub_default == runtime_arg.default)
   734	                        except Exception:
   735	                            # Exception can be raised in bool dunder method (e.g. numpy arrays)
   736	                            # At this point, consider the default to be different, it is probably
   737	                            # too complex to put in a stub anyway.
   738	                            defaults_match = False
   739	                    if not defaults_match:
   740	                        yield (
   741	                            f'runtime argument "{runtime_arg.name}" '
   742	                            f"has a default value of {runtime_arg.default!r}, "
   743	                            f"which is different from stub argument default {stub_default!r}"
   744	                        )
   745	    else:
   746	        if stub_arg.kind.is_optional():
   747	            yield (
   748	                f'stub argument "{stub_arg.variable.name}" has a default value '
   749	                f"but runtime argument does not"
   750	            )

   753	def maybe_strip_cls(name: str, args: list[nodes.Argument]) -> list[nodes.Argument]:
   754	    if args and name in ("__init_subclass__", "__class_getitem__"):
   755	        # These are implicitly classmethods. If the stub chooses not to have @classmethod, we
   756	        # should remove the cls argument
   757	        if args[0].variable.name == "cls":
   758	            return args[1:]
   759	    return args

   762	class Signature(Generic[T]):
   763	    def __init__(self) -> None:
   764	        self.pos: list[T] = []
   765	        self.kwonly: dict[str, T] = {}
   766	        self.varpos: T | None = None
   767	        self.varkw: T | None = None
   768
   769	    def __str__(self) -> str:
   770	        def get_name(arg: Any) -> str:
   771	            if isinstance(arg, inspect.Parameter):
   772	                return arg.name
   773	            if isinstance(arg, nodes.Argument):
   774	                return arg.variable.name
   775	            raise AssertionError
   776
   777	        def get_type(arg: Any) -> str | None:
   778	            if isinstance(arg, inspect.Parameter):
   779	                return None
   780	            if isinstance(arg, nodes.Argument):
   781	                return str(arg.variable.type or arg.type_annotation)
   782	            raise AssertionError
   783
   784	        def has_default(arg: Any) -> bool:
   785	            if isinstance(arg, inspect.Parameter):
   786	                return arg.default is not inspect.Parameter.empty
   787	            if isinstance(arg, nodes.Argument):
   788	                return arg.kind.is_optional()
   789	            raise AssertionError
   790
   791	        def get_desc(arg: Any) -> str:
   792	            arg_type = get_type(arg)
   793	            return (
   794	                get_name(arg)
   795	                + (f": {arg_type}" if arg_type else "")
   796	                + (" = ..." if has_default(arg) else "")
   797	            )
   798
   799	        kw_only = sorted(self.kwonly.values(), key=lambda a: (has_default(a), get_name(a)))
   800	        ret = "def ("
   801	        ret += ", ".join(
   802	            [get_desc(arg) for arg in self.pos]
   803	            + (["*" + get_name(self.varpos)] if self.varpos else (["*"] if self.kwonly else []))
   804	            + [get_desc(arg) for arg in kw_only]
   805	            + (["**" + get_name(self.varkw)] if self.varkw else [])
   806	        )
   807	        ret += ")"
   808	        return ret
   809
   810	    @staticmethod
   811	    def from_funcitem(stub: nodes.FuncItem) -> Signature[nodes.Argument]:
   812	        stub_sig: Signature[nodes.Argument] = Signature()
   813	        stub_args = maybe_strip_cls(stub.name, stub.arguments)
   814	        for stub_arg in stub_args:
   815	            if stub_arg.kind.is_positional():
   816	                stub_sig.pos.append(stub_arg)
   817	            elif stub_arg.kind.is_named():
   818	                stub_sig.kwonly[stub_arg.variable.name] = stub_arg
   819	            elif stub_arg.kind == nodes.ARG_STAR:
   820	                stub_sig.varpos = stub_arg
   821	            elif stub_arg.kind == nodes.ARG_STAR2:
   822	                stub_sig.varkw = stub_arg
   823	            else:
   824	                raise AssertionError
   825	        return stub_sig
   826
   827	    @staticmethod
   828	    def from_inspect_signature(signature: inspect.Signature) -> Signature[inspect.Parameter]:
   829	        runtime_sig: Signature[inspect.Parameter] = Signature()
   830	        for runtime_arg in signature.parameters.values():
   831	            if runtime_arg.kind in (
   832	                inspect.Parameter.POSITIONAL_ONLY,
   833	                inspect.Parameter.POSITIONAL_OR_KEYWORD,
   834	            ):
   835	                runtime_sig.pos.append(runtime_arg)
   836	            elif runtime_arg.kind == inspect.Parameter.KEYWORD_ONLY:
   837	                runtime_sig.kwonly[runtime_arg.name] = runtime_arg
   838	            elif runtime_arg.kind == inspect.Parameter.VAR_POSITIONAL:
   839	                runtime_sig.varpos = runtime_arg
   840	            elif runtime_arg.kind == inspect.Parameter.VAR_KEYWORD:
   841	                runtime_sig.varkw = runtime_arg
   842	            else:
   843	                raise AssertionError
   844	        return runtime_sig

   846	    @staticmethod
   847	    def from_overloadedfuncdef(stub: nodes.OverloadedFuncDef) -> Signature[nodes.Argument]:
   848	        """Returns a Signature from an OverloadedFuncDef.
   849
   850	        If life were simple, to verify_overloadedfuncdef, we'd just verify_funcitem for each of its
   851	        items. Unfortunately, life isn't simple and overloads are pretty deceitful. So instead, we
   852	        try and combine the overload's items into a single signature that is compatible with any
   853	        lies it might try to tell.
   854
   855	        """
   856	        # For most dunder methods, just assume all args are positional-only
   857	        assume_positional_only = is_dunder(stub.name, exclude_special=True)
   858
   859	        all_args: dict[str, list[tuple[nodes.Argument, int]]] = {}
   860	        for func in map(_resolve_funcitem_from_decorator, stub.items):
   861	            assert func is not None, "Failed to resolve decorated overload"
   862	            args = maybe_strip_cls(stub.name, func.arguments)
   863	            for index, arg in enumerate(args):
   864	                # For positional-only args, we allow overloads to have different names for the same
   865	                # argument. To accomplish this, we just make up a fake index-based name.
   866	                name = (
   867	                    f"__{index}"
   868	                    if arg.variable.name.startswith("__")
   869	                    or arg.pos_only
   870	                    or assume_positional_only
   871	                    or arg.variable.name.strip("_") == "self"
   872	                    else arg.variable.name
   873	                )
   874	                all_args.setdefault(name, []).append((arg, index))
   875
   876	        def get_position(arg_name: str) -> int:
   877	            # We just need this to return the positional args in the correct order.
   878	            return max(index for _, index in all_args[arg_name])
   879
   880	        def get_type(arg_name: str) -> mypy.types.ProperType:
   881	            with mypy.state.state.strict_optional_set(True):
   882	                all_types = [
   883	                    arg.variable.type or arg.type_annotation for arg, _ in all_args[arg_name]
   884	                ]
   885	                return mypy.typeops.make_simplified_union([t for t in all_types if t])
   886
   887	        def get_kind(arg_name: str) -> nodes.ArgKind:
   888	            kinds = {arg.kind for arg, _ in all_args[arg_name]}
   889	            if nodes.ARG_STAR in kinds:
   890	                return nodes.ARG_STAR
   891	            if nodes.ARG_STAR2 in kinds:
   892	                return nodes.ARG_STAR2
   893	            # The logic here is based on two tenets:
   894	            # 1) If an arg is ever optional (or unspecified), it is optional
   895	            # 2) If an arg is ever positional, it is positional
   896	            is_opt = (
   897	                len(all_args[arg_name]) < len(stub.items)
   898	                or nodes.ARG_OPT in kinds
   899	                or nodes.ARG_NAMED_OPT in kinds
   900	            )
   901	            is_pos = nodes.ARG_OPT in kinds or nodes.ARG_POS in kinds
   902	            if is_opt:
   903	                return nodes.ARG_OPT if is_pos else nodes.ARG_NAMED_OPT
   904	            return nodes.ARG_POS if is_pos else nodes.ARG_NAMED
   905
   906	        sig: Signature[nodes.Argument] = Signature()
   907	        for arg_name in sorted(all_args, key=get_position):
   908	            # example_arg_name gives us a real name (in case we had a fake index-based name)
   909	            example_arg_name = all_args[arg_name][0][0].variable.name
   910	            arg = nodes.Argument(
   911	                nodes.Var(example_arg_name, get_type(arg_name)),
   912	                type_annotation=None,
   913	                initializer=None,
   914	                kind=get_kind(arg_name),
   915	                pos_only=all(arg.pos_only for arg, _ in all_args[arg_name]),
   916	            )
   917	            if arg.kind.is_positional():
   918	                sig.pos.append(arg)
   919	            elif arg.kind.is_named():
   920	                sig.kwonly[arg.variable.name] = arg
   921	            elif arg.kind == nodes.ARG_STAR:
   922	                sig.varpos = arg
   923	            elif arg.kind == nodes.ARG_STAR2:
   924	                sig.varkw = arg
   925	            else:
   926	                raise AssertionError
   927	        return sig

   930	def _verify_signature(
   931	    stub: Signature[nodes.Argument], runtime: Signature[inspect.Parameter], function_name: str
   932	) -> Iterator[str]:
   933	    # Check positional arguments match up
   934	    for stub_arg, runtime_arg in zip(stub.pos, runtime.pos):
   935	        yield from _verify_arg_name(stub_arg, runtime_arg, function_name)
   936	        yield from _verify_arg_default_value(stub_arg, runtime_arg)
   937	        if (
   938	            runtime_arg.kind == inspect.Parameter.POSITIONAL_ONLY
   939	            and not stub_arg.pos_only
   940	            and not stub_arg.variable.name.startswith("__")
   941	            and stub_arg.variable.name.strip("_") != "self"
   942	            and not is_dunder(function_name, exclude_special=True)  # noisy for dunder methods
   943	        ):
   944	            yield (
   945	                f'stub argument "{stub_arg.variable.name}" should be positional-only '
   946	                f'(add "/", e.g. "{runtime_arg.name}, /")'
   947	            )
   948	        if (
   949	            runtime_arg.kind != inspect.Parameter.POSITIONAL_ONLY
   950	            and (stub_arg.pos_only or stub_arg.variable.name.startswith("__"))
   951	            and not runtime_arg.name.startswith("__")
   952	            and stub_arg.variable.name.strip("_") != "self"
   953	            and not is_dunder(function_name, exclude_special=True)  # noisy for dunder methods
   954	        ):
   955	            yield (
   956	                f'stub argument "{stub_arg.variable.name}" should be positional or keyword '
   957	                '(remove "/")'
   958	            )
   959
   960	    # Check unmatched positional args
   961	    if len(stub.pos) > len(runtime.pos):
   962	        # There are cases where the stub exhaustively lists out the extra parameters the function
   963	        # would take through *args. Hence, a) if runtime accepts *args, we don't check whether the
   964	        # runtime has all of the stub's parameters, b) below, we don't enforce that the stub takes
   965	        # *args, since runtime logic may prevent arbitrary arguments from actually being accepted.
   966	        if runtime.varpos is None:
   967	            for stub_arg in stub.pos[len(runtime.pos) :]:
   968	                # If the variable is in runtime.kwonly, it's just mislabelled as not a
   969	                # keyword-only argument
   970	                if stub_arg.variable.name not in runtime.kwonly:
   971	                    msg = f'runtime does not have argument "{stub_arg.variable.name}"'
   972	                    if runtime.varkw is not None:
   973	                        msg += ". Maybe you forgot to make it keyword-only in the stub?"
   974	                    yield msg
   975	                else:
   976	                    yield f'stub argument "{stub_arg.variable.name}" is not keyword-only'
   977	            if stub.varpos is not None:
   978	                yield f'runtime does not have *args argument "{stub.varpos.variable.name}"'
   979	    elif len(stub.pos) < len(runtime.pos):
   980	        for runtime_arg in runtime.pos[len(stub.pos) :]:
   981	            if runtime_arg.name not in stub.kwonly:
   982	                if not _is_private_parameter(runtime_arg):
   983	                    yield f'stub does not have argument "{runtime_arg.name}"'
   984	            else:
   985	                yield f'runtime argument "{runtime_arg.name}" is not keyword-only'
   986
   987	    # Checks involving *args
   988	    if len(stub.pos) <= len(runtime.pos) or runtime.varpos is None:
   989	        if stub.varpos is None and runtime.varpos is not None:
   990	            yield f'stub does not have *args argument "{runtime.varpos.name}"'
   991	        if stub.varpos is not None and runtime.varpos is None:
   992	            yield f'runtime does not have *args argument "{stub.varpos.variable.name}"'
   993
   994	    # Check keyword-only args
   995	    for arg in sorted(set(stub.kwonly) & set(runtime.kwonly)):
   996	        stub_arg, runtime_arg = stub.kwonly[arg], runtime.kwonly[arg]
   997	        yield from _verify_arg_name(stub_arg, runtime_arg, function_name)
   998	        yield from _verify_arg_default_value(stub_arg, runtime_arg)
   999
  1000	    # Check unmatched keyword-only args
  1001	    if runtime.varkw is None or not set(runtime.kwonly).issubset(set(stub.kwonly)):
  1002	        # There are cases where the stub exhaustively lists out the extra parameters the function
  1003	        # would take through **kwargs. Hence, a) if runtime accepts **kwargs (and the stub hasn't
  1004	        # exhaustively listed out params), we don't check whether the runtime has all of the stub's
  1005	        # parameters, b) below, we don't enforce that the stub takes **kwargs, since runtime logic
  1006	        # may prevent arbitrary keyword arguments from actually being accepted.
  1007	        for arg in sorted(set(stub.kwonly) - set(runtime.kwonly)):
  1008	            if arg in {runtime_arg.name for runtime_arg in runtime.pos}:
  1009	                # Don't report this if we've reported it before
  1010	                if arg not in {runtime_arg.name for runtime_arg in runtime.pos[len(stub.pos) :]}:
  1011	                    yield f'runtime argument "{arg}" is not keyword-only'
  1012	            else:
  1013	                yield f'runtime does not have argument "{arg}"'
  1014	    for arg in sorted(set(runtime.kwonly) - set(stub.kwonly)):
  1015	        if arg in {stub_arg.variable.name for stub_arg in stub.pos}:
  1016	            # Don't report this if we've reported it before
  1017	            if not (
  1018	                runtime.varpos is None
  1019	                and arg in {stub_arg.variable.name for stub_arg in stub.pos[len(runtime.pos) :]}
  1020	            ):
  1021	                yield f'stub argument "{arg}" is not keyword-only'
  1022	        else:
  1023	            if not _is_private_parameter(runtime.kwonly[arg]):
  1024	                yield f'stub does not have argument "{arg}"'
  1025
  1026	    # Checks involving **kwargs
  1027	    if stub.varkw is None and runtime.varkw is not None:
  1028	        # As mentioned above, don't enforce that the stub takes **kwargs.
  1029	        # Also check against positional parameters, to avoid a nitpicky message when an argument
  1030	        # isn't marked as keyword-only
  1031	        stub_pos_names = {stub_arg.variable.name for stub_arg in stub.pos}
  1032	        # Ideally we'd do a strict subset check, but in practice the errors from that aren't useful
  1033	        if not set(runtime.kwonly).issubset(set(stub.kwonly) | stub_pos_names):
  1034	            yield f'stub does not have **kwargs argument "{runtime.varkw.name}"'
  1035	    if stub.varkw is not None and runtime.varkw is None:
  1036	        yield f'runtime does not have **kwargs argument "{stub.varkw.variable.name}"'
  1037

  1039	def _is_private_parameter(arg: inspect.Parameter) -> bool:
  1040	    return (
  1041	        arg.name.startswith("_")
  1042	        and not arg.name.startswith("__")
  1043	        and arg.default is not inspect.Parameter.empty
  1044	    )

  1047	@verify.register(nodes.FuncItem)
  1048	def verify_funcitem(
  1049	    stub: nodes.FuncItem, runtime: MaybeMissing[Any], object_path: list[str]
  1050	) -> Iterator[Error]:
  1051	    if isinstance(runtime, Missing):
  1052	        yield Error(object_path, "is not present at runtime", stub, runtime)
  1053	        return
  1054
  1055	    if not is_probably_a_function(runtime):
  1056	        yield Error(object_path, "is not a function", stub, runtime)
  1057	        if not callable(runtime):
  1058	            return
  1059
  1060	    # Look the object up statically, to avoid binding by the descriptor protocol
  1061	    static_runtime = _static_lookup_runtime(object_path)
  1062
  1063	    if isinstance(stub, nodes.FuncDef):
  1064	        for error_text in _verify_abstract_status(stub, runtime):
  1065	            yield Error(object_path, error_text, stub, runtime)
  1066	        for error_text in _verify_final_method(stub, runtime, static_runtime):
  1067	            yield Error(object_path, error_text, stub, runtime)
  1068
  1069	    for message in _verify_static_class_methods(stub, runtime, static_runtime, object_path):
  1070	        yield Error(object_path, "is inconsistent, " + message, stub, runtime)
  1071
  1072	    signature = safe_inspect_signature(runtime)
  1073	    runtime_is_coroutine = inspect.iscoroutinefunction(runtime)
  1074
  1075	    if signature:
  1076	        stub_sig = Signature.from_funcitem(stub)
  1077	        runtime_sig = Signature.from_inspect_signature(signature)
  1078	        runtime_sig_desc = describe_runtime_callable(signature, is_async=runtime_is_coroutine)
  1079	        stub_desc = str(stub_sig)
  1080	    else:
  1081	        runtime_sig_desc, stub_desc = None, None
  1082
  1083	    # Don't raise an error if the stub is a coroutine, but the runtime isn't.
  1084	    # That results in false positives.
  1085	    # See https://github.com/python/typeshed/issues/7344
  1086	    if runtime_is_coroutine and not stub.is_coroutine:
  1087	        yield Error(
  1088	            object_path,
  1089	            'is an "async def" function at runtime, but not in the stub',
  1090	            stub,
  1091	            runtime,
  1092	            stub_desc=stub_desc,
  1093	            runtime_desc=runtime_sig_desc,
  1094	        )
  1095
  1096	    if not signature:
  1097	        return
  1098
  1099	    for message in _verify_signature(stub_sig, runtime_sig, function_name=stub.name):
  1100	        yield Error(
  1101	            object_path,
  1102	            "is inconsistent, " + message,
  1103	            stub,
  1104	            runtime,
  1105	            runtime_desc=runtime_sig_desc,
  1106	        )

  1109	@verify.register(Missing)
  1110	def verify_missing(
  1111	    stub: Missing, runtime: MaybeMissing[Any], object_path: list[str]

  1321	def _resolve_funcitem_from_decorator(dec: nodes.OverloadPart) -> nodes.FuncItem | None:
  1322	    """Returns a FuncItem that corresponds to the output of the decorator.
  1323
  1324	    Returns None if we can't figure out what that would be. For convenience, this function also
  1325	    accepts FuncItems.
  1326	    """
  1327	    if isinstance(dec, nodes.FuncItem):
  1328	        return dec
  1329	    if dec.func.is_property:
  1330	        return None
  1331
  1332	    def apply_decorator_to_funcitem(
  1333	        decorator: nodes.Expression, func: nodes.FuncItem
  1334	    ) -> nodes.FuncItem | None:
  1335	        if (
  1336	            isinstance(decorator, nodes.CallExpr)
  1337	            and isinstance(decorator.callee, nodes.RefExpr)
  1338	            and decorator.callee.fullname in mypy.types.DEPRECATED_TYPE_NAMES
  1339	        ):
  1340	            return func
  1341	        if not isinstance(decorator, nodes.RefExpr):
  1342	            return None
  1343	        if not decorator.fullname:
  1344	            # Happens with namedtuple
  1345	            return None
  1346	        if (
  1347	            decorator.fullname in ("builtins.staticmethod", "abc.abstractmethod")
  1348	            or decorator.fullname in mypy.types.OVERLOAD_NAMES
  1349	            or decorator.fullname in mypy.types.OVERRIDE_DECORATOR_NAMES
  1350	            or decorator.fullname in mypy.types.FINAL_DECORATOR_NAMES
  1351	        ):
  1352	            return func
  1353	        if decorator.fullname == "builtins.classmethod":
  1354	            if func.arguments[0].variable.name not in ("cls", "mcs", "metacls"):
  1355	                raise StubtestFailure(
  1356	                    f"unexpected class argument name {func.arguments[0].variable.name!r} "
  1357	                    f"in {dec.fullname}"
  1358	                )
  1359	            # FuncItem is written so that copy.copy() actually works, even when compiled
  1360	            ret = copy.copy(func)
  1361	            # Remove the cls argument, since it's not present in inspect.signature of classmethods
  1362	            ret.arguments = ret.arguments[1:]
  1363	            return ret
  1364	        # Just give up on any other decorators. After excluding properties, we don't run into
  1365	        # anything else when running on typeshed's stdlib.
  1366	        return None
  1367
  1368	    func: nodes.FuncItem = dec.func
  1369	    for decorator in dec.original_decorators:
  1370	        resulting_func = apply_decorator_to_funcitem(decorator, func)
  1371	        if resulting_func is None:
  1372	            return None
  1373	        func = resulting_func
  1374	    return func

  1411	    stub: nodes.TypeAlias, runtime: MaybeMissing[Any], object_path: list[str]
  1412	) -> Iterator[Error]:
  1413	    stub_target = mypy.types.get_proper_type(stub.target)
  1414	    stub_desc = f"Type alias for {stub_target}"
  1415	    if isinstance(runtime, Missing):
  1416	        yield Error(object_path, "is not present at runtime", stub, runtime, stub_desc=stub_desc)
  1417	        return
  1418	    runtime_origin = get_origin(runtime) or runtime
  1419	    if isinstance(stub_target, mypy.types.Instance):
  1420	        if not isinstance(runtime_origin, type):
  1421	            yield Error(
  1422	                object_path,
  1423	                "is inconsistent, runtime is not a type",
  1424	                stub,
  1425	                runtime,
  1426	                stub_desc=stub_desc,
  1427	            )
  1428	            return
  1429
  1430	        stub_origin = stub_target.type
  1431	        # Do our best to figure out the fullname of the runtime object...
  1432	        runtime_name: object
  1433	        try:
  1434	            runtime_name = runtime_origin.__qualname__
  1435	        except AttributeError:
  1436	            runtime_name = getattr(runtime_origin, "__name__", MISSING)
  1437	        if isinstance(runtime_name, str):
  1438	            runtime_module: object = getattr(runtime_origin, "__module__", MISSING)
  1439	            if isinstance(runtime_module, str):
  1440	                if runtime_module == "collections.abc" or (
  1441	                    runtime_module == "re" and runtime_name in {"Match", "Pattern"}
  1442	                ):
  1443	                    runtime_module = "typing"
  1444	                runtime_fullname = f"{runtime_module}.{runtime_name}"
  1445	                if re.fullmatch(rf"_?{re.escape(stub_origin.fullname)}", runtime_fullname):
  1446	                    # Okay, we're probably fine.
  1447	                    return
  1448
  1449	        # Okay, either we couldn't construct a fullname
  1450	        # or the fullname of the stub didn't match the fullname of the runtime.
  1451	        # Fallback to a full structural check of the runtime vis-a-vis the stub.
  1452	        yield from verify(stub_origin, runtime_origin, object_path)
  1453	        return
  1454	    if isinstance(stub_target, mypy.types.UnionType):
  1455	        # complain if runtime is not a Union or UnionType
  1456	        if runtime_origin is not Union and (
  1457	            not (sys.version_info >= (3, 10) and isinstance(runtime, types.UnionType))
  1458	        ):
  1459	            yield Error(object_path, "is not a Union", stub, runtime, stub_desc=str(stub_target))

  1461	        return

  1465	                object_path, "is not a subclass of tuple", stub, runtime, stub_desc=stub_desc

  1595	def safe_inspect_signature(runtime: Any) -> inspect.Signature | None:
  1596	    try:
  1597	        try:
  1598	            return inspect.signature(runtime)
  1599	        except ValueError:
  1600	            if (
  1601	                hasattr(runtime, "__text_signature__")
  1602	                and "<unrepresentable>" in runtime.__text_signature__
  1603	            ):
  1604	                # Try to fix up the signature. Workaround for
  1605	                # https://github.com/python/cpython/issues/87233
  1606	                sig = runtime.__text_signature__.replace("<unrepresentable>", "...")
  1607	                sig = inspect._signature_fromstr(inspect.Signature, runtime, sig)  # type: ignore[attr-defined]
  1608	                assert isinstance(sig, inspect.Signature)
  1609	                new_params = [
  1610	                    (
  1611	                        parameter.replace(default=UNREPRESENTABLE)
  1612	                        if parameter.default is ...
  1613	                        else parameter
  1614	                    )
  1615	                    for parameter in sig.parameters.values()
  1616	                ]
  1617	                return sig.replace(parameters=new_params)
  1618	            else:
  1619	                raise
  1620	    except Exception:
  1621	        # inspect.signature throws ValueError all the time
  1622	        # catch RuntimeError because of https://bugs.python.org/issue39504
  1623	        # catch TypeError because of https://github.com/python/typeshed/pull/5762
  1624	        # catch AttributeError because of inspect.signature(_curses.window.border)

  1632	def is_subtype_helper(left: mypy.types.Type, right: mypy.types.Type) -> bool:
  1633	    """Checks whether ``left`` is a subtype of ``right``."""
  1634	    left = mypy.types.get_proper_type(left)
  1635	    right = mypy.types.get_proper_type(right)
  1636	    if (
  1637	        isinstance(left, mypy.types.LiteralType)
  1638	        and isinstance(left.value, int)
  1639	        and left.value in (0, 1)
  1640	        and mypy.types.is_named_instance(right, "builtins.bool")
  1641	    ):
  1642	        # Pretend Literal[0, 1] is a subtype of bool to avoid unhelpful errors.
  1643	        return True
  1644
  1645	    if isinstance(right, mypy.types.TypedDictType) and mypy.types.is_named_instance(
  1646	        left, "builtins.dict"
  1647	    ):
  1648	        # Special case checks against TypedDicts
  1649	        return True

  1651	    with mypy.state.state.strict_optional_set(True):
  1652	        return mypy.subtypes.is_subtype(left, right)

  1655	def get_mypy_type_of_runtime_value(runtime: Any) -> mypy.types.Type | None:
  1656	    """Returns a mypy type object representing the type of ``runtime``.
  1657
  1658	    Returns None if we can't find something that works.
  1659
  1660	    """
  1661	    if runtime is None:
  1662	        return mypy.types.NoneType()
  1663	    if isinstance(runtime, property):
  1664	        # Give up on properties to avoid issues with things that are typed as attributes.
  1665	        return None
  1666
  1667	    def anytype() -> mypy.types.AnyType:
  1668	        return mypy.types.AnyType(mypy.types.TypeOfAny.unannotated)
  1669
  1670	    if isinstance(
  1671	        runtime,
  1672	        (types.FunctionType, types.BuiltinFunctionType, types.MethodType, types.BuiltinMethodType),
  1673	    ):
  1674	        builtins = get_stub("builtins")
  1675	        assert builtins is not None
  1676	        type_info = builtins.names["function"].node
  1677	        assert isinstance(type_info, nodes.TypeInfo)
  1678	        fallback = mypy.types.Instance(type_info, [anytype()])
  1679	        signature = safe_inspect_signature(runtime)
  1680	        if signature:
  1681	            arg_types = []
  1682	            arg_kinds = []
  1683	            arg_names = []
  1684	            for arg in signature.parameters.values():
  1685	                arg_types.append(anytype())
  1686	                arg_names.append(
  1687	                    None if arg.kind == inspect.Parameter.POSITIONAL_ONLY else arg.name
  1688	                )
  1689	                no_default = arg.default is inspect.Parameter.empty
  1690	                if arg.kind == inspect.Parameter.POSITIONAL_ONLY:
  1691	                    arg_kinds.append(nodes.ARG_POS if no_default else nodes.ARG_OPT)
  1692	                elif arg.kind == inspect.Parameter.POSITIONAL_OR_KEYWORD:
  1693	                    arg_kinds.append(nodes.ARG_POS if no_default else nodes.ARG_OPT)
  1694	                elif arg.kind == inspect.Parameter.KEYWORD_ONLY:
  1695	                    arg_kinds.append(nodes.ARG_NAMED if no_default else nodes.ARG_NAMED_OPT)
  1696	                elif arg.kind == inspect.Parameter.VAR_POSITIONAL:
  1697	                    arg_kinds.append(nodes.ARG_STAR)
  1698	                elif arg.kind == inspect.Parameter.VAR_KEYWORD:
  1699	                    arg_kinds.append(nodes.ARG_STAR2)
  1700	                else:
  1701	                    raise AssertionError
  1702	        else:
  1703	            arg_types = [anytype(), anytype()]
  1704	            arg_kinds = [nodes.ARG_STAR, nodes.ARG_STAR2]
  1705	            arg_names = [None, None]
  1706
  1707	        return mypy.types.CallableType(
  1708	            arg_types,
  1709	            arg_kinds,
  1710	            arg_names,
  1711	            ret_type=anytype(),
  1712	            fallback=fallback,
  1713	            is_ellipsis_args=True,
  1714	        )
  1715
  1716	    # Try and look up a stub for the runtime object
  1717	    stub = get_stub(type(runtime).__module__)
  1718	    if stub is None:
  1719	        return None
  1720	    type_name = type(runtime).__name__
  1721	    if type_name not in stub.names:
  1722	        return None
  1723	    type_info = stub.names[type_name].node
  1724	    if isinstance(type_info, nodes.Var):
  1725	        return type_info.type
  1726	    if not isinstance(type_info, nodes.TypeInfo):
  1727	        return None
  1728
  1729	    if isinstance(runtime, tuple):
  1730	        # Special case tuples so we construct a valid mypy.types.TupleType
  1731	        optional_items = [get_mypy_type_of_runtime_value(v) for v in runtime]
  1732	        items = [(i if i is not None else anytype()) for i in optional_items]
  1733	        fallback = mypy.types.Instance(type_info, [anytype()])
  1734	        return mypy.types.TupleType(items, fallback)
  1735
  1736	    fallback = mypy.types.Instance(type_info, [anytype() for _ in type_info.type_vars])
  1737
  1738	    value: bool | int | str
  1739	    if isinstance(runtime, enum.Enum) and isinstance(runtime.name, str):
  1740	        value = runtime.name
  1741	    elif isinstance(runtime, bytes):
  1742	        value = bytes_to_human_readable_repr(runtime)
  1743	    elif isinstance(runtime, (bool, int, str)):
  1744	        value = runtime
  1745	    else:
  1746	        return fallback
  1747
  1748	    return mypy.types.LiteralType(value=value, fallback=fallback)

  1759	def build_stubs(modules: list[str], options: Options, find_submodules: bool = False) -> list[str]:
  1760	    """Uses mypy to construct stub objects for the given modules.
  1761
  1762	    This sets global state that ``get_stub`` can access.
  1763
  1764	    Returns all modules we might want to check. If ``find_submodules`` is False, this is equal
  1765	    to ``modules``.
  1766
  1767	    :param modules: List of modules to build stubs for.
  1768	    :param options: Mypy options for finding and building stubs.
  1769	    :param find_submodules: Whether to attempt to find submodules of the given modules as well.
  1770
  1771	    """
  1772	    data_dir = mypy.build.default_data_dir()
  1773	    search_path = mypy.modulefinder.compute_search_paths([], options, data_dir)
  1774	    find_module_cache = mypy.modulefinder.FindModuleCache(
  1775	        search_path, fscache=None, options=options
  1776	    )
  1777
  1778	    all_modules = []
  1779	    sources = []
  1780	    for module in modules:
  1781	        all_modules.append(module)
  1782	        if not find_submodules:
  1783	            module_path = find_module_cache.find_module(module)
  1784	            if not isinstance(module_path, str):
  1785	                # test_module will yield an error later when it can't find stubs
  1786	                continue
  1787	            sources.append(mypy.modulefinder.BuildSource(module_path, module, None))
  1788	        else:
  1789	            found_sources = find_module_cache.find_modules_recursive(module)
  1790	            sources.extend(found_sources)
  1791	            # find submodules via mypy
  1792	            all_modules.extend(s.module for s in found_sources if s.module not in all_modules)
  1793	            # find submodules via pkgutil
  1794	            try:
  1795	                runtime = silent_import_module(module)
  1796	                all_modules.extend(
  1797	                    m.name
  1798	                    for m in pkgutil.walk_packages(runtime.__path__, runtime.__name__ + ".")
  1799	                    if m.name not in all_modules
  1800	                )
  1801	            except KeyboardInterrupt:
  1802	                raise
  1803	            except BaseException:
  1804	                pass
  1805
  1806	    if sources:
  1807	        try:
  1808	            res = mypy.build.build(sources=sources, options=options)
  1809	        except mypy.errors.CompileError as e:
  1810	            raise StubtestFailure(f"failed mypy compile:\n{e}") from e
  1811	        if res.errors:
  1812	            raise StubtestFailure("mypy build errors:\n" + "\n".join(res.errors))
  1813
  1814	        global _all_stubs
  1815	        _all_stubs = res.files
  1816
  1817	    return all_modules

  1825	def get_typeshed_stdlib_modules(
  1826	    custom_typeshed_dir: str | None, version_info: tuple[int, int] | None = None
  1827	) -> set[str]:
  1828	    """Returns a list of stdlib modules in typeshed (for current Python version)."""
  1829	    stdlib_py_versions = mypy.modulefinder.load_stdlib_py_versions(custom_typeshed_dir)
  1830	    if version_info is None:
  1831	        version_info = sys.version_info[0:2]
  1832
  1833	    def exists_in_version(module: str) -> bool:
  1834	        assert version_info is not None
  1835	        parts = module.split(".")
  1836	        for i in range(len(parts), 0, -1):
  1837	            current_module = ".".join(parts[:i])
  1838	            if current_module in stdlib_py_versions:
  1839	                minver, maxver = stdlib_py_versions[current_module]
  1840	                return version_info >= minver and (maxver is None or version_info <= maxver)
  1841	        return False
  1842
  1843	    if custom_typeshed_dir:
  1844	        typeshed_dir = Path(custom_typeshed_dir)
  1845	    else:
  1846	        typeshed_dir = Path(mypy.build.default_data_dir()) / "typeshed"
  1847	    stdlib_dir = typeshed_dir / "stdlib"
  1848
  1849	    modules: set[str] = set()
  1850	    for path in stdlib_dir.rglob("*.pyi"):
  1851	        if path.stem == "__init__":
  1852	            path = path.parent
  1853	        module = ".".join(path.relative_to(stdlib_dir).parts[:-1] + (path.stem,))
  1854	        if exists_in_version(module):
  1855	            modules.add(module)

  1859	def get_importable_stdlib_modules() -> set[str]:
  1860	    """Return all importable stdlib modules at runtime."""
  1861	    all_stdlib_modules: AbstractSet[str]
  1862	    if sys.version_info >= (3, 10):
  1863	        all_stdlib_modules = sys.stdlib_module_names
  1864	    else:
  1865	        all_stdlib_modules = set(sys.builtin_module_names)
  1866	        modules_by_finder: defaultdict[importlib.machinery.FileFinder, set[str]] = defaultdict(set)
  1867	        for m in pkgutil.iter_modules():
  1868	            if isinstance(m.module_finder, importlib.machinery.FileFinder):
  1869	                modules_by_finder[m.module_finder].add(m.name)
  1870	        for finder, module_group in modules_by_finder.items():
  1871	            if (
  1872	                "site-packages" not in Path(finder.path).parts
  1873	                # if "_queue" is present, it's most likely the module finder
  1874	                # for stdlib extension modules;
  1875	                # if "queue" is present, it's most likely the module finder
  1876	                # for pure-Python stdlib modules.
  1877	                # In either case, we'll want to add all the modules that the finder has to offer us.
  1878	                # This is a bit hacky, but seems to work well in a cross-platform way.
  1879	                and {"_queue", "queue"} & module_group
  1880	            ):
  1881	                all_stdlib_modules.update(module_group)
  1882
  1883	    importable_stdlib_modules: set[str] = set()
  1884	    for module_name in all_stdlib_modules:
  1885	        if module_name in ANNOYING_STDLIB_MODULES:
  1886	            continue
  1887
  1888	        try:
  1889	            runtime = silent_import_module(module_name)
  1890	        except ImportError:
  1891	            continue
  1892	        else:
  1893	            importable_stdlib_modules.add(module_name)
  1894
  1895	        try:
  1896	            # some stdlib modules (e.g. `nt`) don't have __path__ set...
  1897	            runtime_path = runtime.__path__
  1898	            runtime_name = runtime.__name__
  1899	        except AttributeError:
  1900	            continue
  1901
  1902	        for submodule in pkgutil.walk_packages(runtime_path, runtime_name + "."):
  1903	            submodule_name = submodule.name
  1904
  1905	            # There are many annoying *.__main__ stdlib modules,
  1906	            # and including stubs for them isn't really that useful anyway:
  1907	            # tkinter.__main__ opens a tkinter windows; unittest.__main__ raises SystemExit; etc.
  1908	            #
  1909	            # The idlelib.* submodules are similarly annoying in opening random tkinter windows,
  1910	            # and we're unlikely to ever add stubs for idlelib in typeshed
  1911	            # (see discussion in https://github.com/python/typeshed/pull/9193)
  1912	            #
  1913	            # test.* modules do weird things like raising exceptions in __del__ methods,
  1914	            # leading to unraisable exceptions being logged to the terminal
  1915	            # as a warning at the end of the stubtest run
  1916	            if submodule_name.endswith(".__main__") or submodule_name.startswith(
  1917	                ("idlelib.", "test.")
  1918	            ):
  1919	                continue
  1920
  1921	            try:
  1922	                silent_import_module(submodule_name)
  1923	            except KeyboardInterrupt:
  1924	                raise
  1925	            # importing multiprocessing.popen_forkserver on Windows raises AttributeError...
  1926	            # some submodules also appear to raise SystemExit as well on some Python versions
  1927	            # (not sure exactly which)
  1928	            except BaseException:
  1929	                continue
  1930	            else:
  1931	                importable_stdlib_modules.add(submodule_name)
  1932
  1933	    return importable_stdlib_modules

  1970	def test_stubs(args: _Arguments, use_builtins_fixtures: bool = False) -> int:
  1971	    """This is stubtest! It's time to test the stubs!"""
  1972	    # Load the allowlist. This is a series of strings corresponding to Error.object_desc
  1973	    # Values in the dict will store whether we used the allowlist entry or not.
  1974	    allowlist = {
  1975	        entry: False
  1976	        for allowlist_file in args.allowlist
  1977	        for entry in get_allowlist_entries(allowlist_file)
  1978	    }
  1979	    allowlist_regexes = {entry: re.compile(entry) for entry in allowlist}
  1980
  1981	    # If we need to generate an allowlist, we store Error.object_desc for each error here.
  1982	    generated_allowlist = set()
  1983
  1984	    modules = args.modules
  1985	    if args.check_typeshed:
  1986	        if args.modules:
  1987	            print(
  1988	                _style("error:", color="red", bold=True),
  1989	                "cannot pass both --check-typeshed and a list of modules",
  1990	            )
  1991	            return 1
  1992	        typeshed_modules = get_typeshed_stdlib_modules(args.custom_typeshed_dir)
  1993	        runtime_modules = get_importable_stdlib_modules()
  1994	        modules = sorted((typeshed_modules | runtime_modules) - ANNOYING_STDLIB_MODULES)
  1995
  1996	    if not modules:
  1997	        print(_style("error:", color="red", bold=True), "no modules to check")
  1998	        return 1
  1999
  2000	    options = Options()
  2001	    options.incremental = False
  2002	    options.custom_typeshed_dir = args.custom_typeshed_dir
  2003	    if options.custom_typeshed_dir:
  2004	        options.abs_custom_typeshed_dir = os.path.abspath(options.custom_typeshed_dir)
  2005	    options.config_file = args.mypy_config_file
  2006	    options.use_builtins_fixtures = use_builtins_fixtures
  2007	    options.show_traceback = args.show_traceback
  2008	    options.pdb = args.pdb
  2009
  2010	    if options.config_file:
  2011
  2012	        def set_strict_flags() -> None:  # not needed yet
  2013	            return
  2014
  2015	        parse_config_file(options, set_strict_flags, options.config_file, sys.stdout, sys.stderr)
  2016
  2017	    def error_callback(msg: str) -> typing.NoReturn:
  2018	        print(_style("error:", color="red", bold=True), msg)
  2019	        sys.exit(1)
  2020
  2021	    def warning_callback(msg: str) -> None:
  2022	        print(_style("warning:", color="yellow", bold=True), msg)
  2023
  2024	    options.process_error_codes(error_callback=error_callback)
  2025	    options.process_incomplete_features(
  2026	        error_callback=error_callback, warning_callback=warning_callback
  2027	    )
  2028	    options.process_strict_bytes()
  2029
  2030	    try:
  2031	        modules = build_stubs(modules, options, find_submodules=not args.check_typeshed)
  2032	    except StubtestFailure as stubtest_failure:
  2033	        print(
  2034	            _style("error:", color="red", bold=True),
  2035	            f"not checking stubs due to {stubtest_failure}",
  2036	        )
  2037	        return 1
  2038
  2039	    exit_code = 0
  2040	    error_count = 0
  2041	    for module in modules:
  2042	        for error in test_module(module):
  2043	            # Filter errors
  2044	            if args.ignore_missing_stub and error.is_missing_stub():
  2045	                continue
  2046	            if args.ignore_positional_only and error.is_positional_only_related():
  2047	                continue
  2048	            if error.object_desc in allowlist:
  2049	                allowlist[error.object_desc] = True
  2050	                continue
  2051	            is_allowlisted = False
  2052	            for w in allowlist:
  2053	                if allowlist_regexes[w].fullmatch(error.object_desc):
  2054	                    allowlist[w] = True
  2055	                    is_allowlisted = True
  2056	                    break
  2057	            if is_allowlisted:
  2058	                continue
  2059
  2060	            # We have errors, so change exit code, and output whatever necessary
  2061	            exit_code = 1
  2062	            if args.generate_allowlist:
  2063	                generated_allowlist.add(error.object_desc)
  2064	                continue
  2065	            print(error.get_description(concise=args.concise))
  2066	            error_count += 1
  2067
  2068	    # Print unused allowlist entries
  2069	    if not args.ignore_unused_allowlist:
  2070	        for w in allowlist:
  2071	            # Don't consider an entry unused if it regex-matches the empty string
  2072	            # This lets us allowlist errors that don't manifest at all on some systems
  2073	            if not allowlist[w] and not allowlist_regexes[w].fullmatch(""):
  2074	                exit_code = 1
  2075	                error_count += 1
  2076	                print(f"note: unused allowlist entry {w}")
  2077
  2078	    # Print the generated allowlist
  2079	    if args.generate_allowlist:
  2080	        for e in sorted(generated_allowlist):
  2081	            print(e)
  2082	        exit_code = 0
  2083	    elif not args.concise:
  2084	        if error_count:
  2085	            print(
  2086	                _style(
  2087	                    f"Found {error_count} error{plural_s(error_count)}"
  2088	                    f" (checked {len(modules)} module{plural_s(modules)})",
  2089	                    color="red",
  2090	                    bold=True,
  2091	                )
  2092	            )
  2093	        else:
  2094	            print(
  2095	                _style(
  2096	                    f"Success: no issues found in {len(modules)} module{plural_s(modules)}",
  2097	                    color="green",
  2098	                    bold=True,
  2099	                )
  2100	            )
  2101
  2102	    return exit_code

    67

   184	def remove_misplaced_type_comments(source: str | bytes) -> str | bytes:
   185	    """Remove comments from source that could be understood as misplaced type comments.
   186
   187	    Normal comments may look like misplaced type comments, and since they cause blocking
   188	    parse errors, we want to avoid them.
   189	    """
   190	    if isinstance(source, bytes):
   191	        # This gives us a 1-1 character code mapping, so it's roundtrippable.
   192	        text = source.decode("latin1")
   193	    else:
   194	        text = source
   195
   196	    # Remove something that looks like a variable type comment but that's by itself
   197	    # on a line, as it will often generate a parse error (unless it's # type: ignore).
   198	    text = re.sub(r'^[ \t]*# +type: +["\'a-zA-Z_].*$', "", text, flags=re.MULTILINE)
   199
   200	    # Remove something that looks like a function type comment after docstring,
   201	    # which will result in a parse error.
   202	    text = re.sub(r'""" *\n[ \t\n]*# +type: +\(.*$', '"""\n', text, flags=re.MULTILINE)
   203	    text = re.sub(r"''' *\n[ \t\n]*# +type: +\(.*$", "'''\n", text, flags=re.MULTILINE)
   204
   205	    # Remove something that looks like a badly formed function type comment.
   206	    text = re.sub(r"^[ \t]*# +type: +\([^()]+(\)[ \t]*)?$", "", text, flags=re.MULTILINE)
   207
   208	    if isinstance(source, bytes):
   209	        return text.encode("latin1")
   210	    else:
   211	        return text

   452	class ImportTracker:
   453	    """Record necessary imports during stub generation."""
   454
   455	    def __init__(self) -> None:
   456	        # module_for['foo'] has the module name where 'foo' was imported from, or None if
   457	        # 'foo' is a module imported directly;
   458	        # direct_imports['foo'] is the module path used when the name 'foo' was added to the
   459	        # namespace.
   460	        # reverse_alias['foo'] is the name that 'foo' had originally when imported with an
   461	        # alias; examples
   462	        #     'from pkg import mod'      ==> module_for['mod'] == 'pkg'
   463	        #     'from pkg import mod as m' ==> module_for['m'] == 'pkg'
   464	        #                                ==> reverse_alias['m'] == 'mod'
   465	        #     'import pkg.mod as m'      ==> module_for['m'] == None
   466	        #                                ==> reverse_alias['m'] == 'pkg.mod'
   467	        #     'import pkg.mod'           ==> module_for['pkg'] == None
   468	        #                                ==> module_for['pkg.mod'] == None
   469	        #                                ==> direct_imports['pkg'] == 'pkg.mod'
   470	        #                                ==> direct_imports['pkg.mod'] == 'pkg.mod'
   471	        self.module_for: dict[str, str | None] = {}
   472	        self.direct_imports: dict[str, str] = {}
   473	        self.reverse_alias: dict[str, str] = {}
   474
   475	        # required_names is the set of names that are actually used in a type annotation
   476	        self.required_names: set[str] = set()
   477
   478	        # Names that should be reexported if they come from another module
   479	        self.reexports: set[str] = set()
   480
   481	    def add_import_from(
   482	        self, module: str, names: list[tuple[str, str | None]], require: bool = False
   483	    ) -> None:
   484	        for name, alias in names:
   485	            if alias:
   486	                # 'from {module} import {name} as {alias}'
   487	                self.module_for[alias] = module
   488	                self.reverse_alias[alias] = name
   489	            else:
   490	                # 'from {module} import {name}'
   491	                self.module_for[name] = module
   492	                self.reverse_alias.pop(name, None)
   493	            if require:
   494	                self.require_name(alias or name)
   495	            self.direct_imports.pop(alias or name, None)
   496
   497	    def add_import(self, module: str, alias: str | None = None, require: bool = False) -> None:
   498	        if alias:
   499	            # 'import {module} as {alias}'
   500	            assert "." not in alias  # invalid syntax
   501	            self.module_for[alias] = None
   502	            self.reverse_alias[alias] = module
   503	            if require:
   504	                self.required_names.add(alias)
   505	        else:
   506	            # 'import {module}'
   507	            name = module
   508	            if require:
   509	                self.required_names.add(name)
   510	            # add module and its parent packages
   511	            while name:
   512	                self.module_for[name] = None
   513	                self.direct_imports[name] = module
   514	                self.reverse_alias.pop(name, None)
   515	                name = name.rpartition(".")[0]
   516
   517	    def require_name(self, name: str) -> None:
   518	        while name not in self.direct_imports and "." in name:
   519	            name = name.rsplit(".", 1)[0]
   520	        self.required_names.add(name)
   521
   522	    def reexport(self, name: str) -> None:
   523	        """Mark a given non qualified name as needed in __all__.
   524
   525	        This means that in case it comes from a module, it should be
   526	        imported with an alias even if the alias is the same as the name.
   527	        """
   528	        self.require_name(name)
   529	        self.reexports.add(name)
   530
   531	    def import_lines(self) -> list[str]:
   532	        """The list of required import lines (as strings with python code).
   533
   534	        In order for a module be included in this output, an identifier must be both
   535	        'required' via require_name() and 'imported' via add_import_from()
   536	        or add_import()
   537	        """
   538	        result = []
   539
   540	        # To summarize multiple names imported from a same module, we collect those
   541	        # in the `module_map` dictionary, mapping a module path to the list of names that should
   542	        # be imported from it. the names can also be alias in the form 'original as alias'
   543	        module_map: Mapping[str, list[str]] = defaultdict(list)
   544
   545	        for name in sorted(
   546	            self.required_names,
   547	            key=lambda n: (self.reverse_alias[n], n) if n in self.reverse_alias else (n, ""),
   548	        ):
   549	            # If we haven't seen this name in an import statement, ignore it
   550	            if name not in self.module_for:
   551	                continue
   552
   553	            m = self.module_for[name]
   554	            if m is not None:
   555	                # This name was found in a from ... import ...
   556	                # Collect the name in the module_map
   557	                if name in self.reverse_alias:
   558	                    name = f"{self.reverse_alias[name]} as {name}"
   559	                elif name in self.reexports:
   560	                    name = f"{name} as {name}"
   561	                module_map[m].append(name)
   562	            else:
   563	                # This name was found in an import ...
   564	                # We can already generate the import line
   565	                if name in self.reverse_alias:
   566	                    source = self.reverse_alias[name]
   567	                    result.append(f"import {source} as {name}\n")
   568	                elif name in self.reexports:
   569	                    assert "." not in name  # Because reexports only has nonqualified names
   570	                    result.append(f"import {name} as {name}\n")
   571	                else:
   572	                    result.append(f"import {name}\n")
   573
   574	        # Now generate all the from ... import ... lines collected in module_map
   575	        for module, names in sorted(module_map.items()):
   576	            result.append(f"from {module} import {', '.join(sorted(names))}\n")
   577	        return result
