   208	    """Make free type variables generic in the type if possible.

   226	        found_vars = []

   232	                    )
   233	                else:
   234	                    normalized = tv
   235	                if normalized in self.poly_tvars and normalized not in self.bound_tvars:
   236	                    found_vars.append(normalized)
   237	        return remove_dups(found_vars)

   239	    def visit_callable_type(self, t: CallableType) -> Type:
   240	        found_vars = self.collect_vars(t)

   292	                repl.variables = list(repl.variables) + list(found_vars)
   293	                return t.copy_modified(args=new_args)
   294	        # There is the same problem with callback protocols as with aliases
   295	        # (callback protocols are essentially more flexible aliases to callables).
   296	        if t.args and t.type.is_protocol and t.type.protocol_members == ["__call__"]:
   297	            if t.type in self.seen_aliases:
   298	                raise PolyTranslationError()
   299	            call = mypy.subtypes.find_member("__call__", t, t, is_operator=True)

   301	            return call.accept(

    60	    calculate the hashes. Note that this isn't directly related to literal
    61	    types -- the concept predates literal types.
    62	    """

    67	        self.unreachable = False
    68	        self.conditional_frame = conditional_frame

    78	class ConditionalTypeBinder:
    79	    """Keep track of conditional types of variables.
    80
    81	    NB: Variables are tracked by literal hashes of expressions, so it is
    82	    possible to confuse the binder when there is aliasing. Example:
    83
    84	        class A:
    85	            a: int | str
    86
    87	        x = A()
    88	        lst = [x]
    89	        reveal_type(x.a)      # int | str
    90	        x.a = 1
    91	        reveal_type(x.a)      # int
    92	        reveal_type(lst[0].a) # int | str
    93	        lst[0].a = 'a'
    94	        reveal_type(x.a)      # int
    95	        reveal_type(lst[0].a) # str
    96	    """
    97
    98	    # Stored assignments for situations with tuple/list lvalue and rvalue of union type.
    99	    # This maps an expression to a list of bound types for every item in the union type.
   100	    type_assignments: Assigns | None = None
   101
   102	    def __init__(self, options: Options) -> None:
   103	        # Each frame gets an increasing, distinct id.
   104	        self.next_id = 1
   105
   106	        # The stack of frames currently used.  These map
   107	        # literal_hash(expr) -- literals like 'foo.bar' --
   108	        # to types. The last element of this list is the
   109	        # top-most, current frame. Each earlier element
   110	        # records the state as of when that frame was last
   111	        # on top of the stack.
   112	        self.frames = [Frame(self._get_id())]
   113
   114	        # For frames higher in the stack, we record the set of
   115	        # Frames that can escape there, either by falling off
   116	        # the end of the frame or by a loop control construct
   117	        # or raised exception. The last element of self.frames
   118	        # has no corresponding element in this list.
   119	        self.options_on_return: list[list[Frame]] = []
   120
   121	        # Maps literal_hash(expr) to get_declaration(expr)
   122	        # for every expr stored in the binder
   123	        self.declarations: dict[Key, Type | None] = {}
   124	        # Set of other keys to invalidate if a key is changed, e.g. x -> {x.a, x[0]}
   125	        # Whenever a new key (e.g. x.a.b) is added, we update this
   126	        self.dependencies: dict[Key, set[Key]] = {}
   127
   128	        # Whether the last pop changed the newly top frame on exit
   129	        self.last_pop_changed = False
   130
   131	        # These are used to track control flow in try statements and loops.
   132	        self.try_frames: set[int] = set()
   133	        self.break_frames: list[int] = []
   134	        self.continue_frames: list[int] = []
   135
   136	        # If True, initial assignment to a simple variable (e.g. "x", but not "x.y")
   137	        # is added to the binder. This allows more precise narrowing and more
   138	        # flexible inference of variable types (--allow-redefinition-new).
   139	        self.bind_all = options.allow_redefinition_new
   140
   141	    def _get_id(self) -> int:
   142	        self.next_id += 1
   143	        return self.next_id
   144
   145	    def _add_dependencies(self, key: Key, value: Key | None = None) -> None:
   146	        if value is None:
   147	            value = key
   148	        else:
   149	            self.dependencies.setdefault(key, set()).add(value)
   150	        for elt in subkeys(key):
   151	            self._add_dependencies(elt, value)
   152
   153	    def push_frame(self, conditional_frame: bool = False) -> Frame:
   154	        """Push a new frame into the binder."""
   155	        f = Frame(self._get_id(), conditional_frame)
   156	        self.frames.append(f)
   157	        self.options_on_return.append([])
   158	        return f
   159
   160	    def _put(self, key: Key, type: Type, from_assignment: bool, index: int = -1) -> None:
   161	        self.frames[index].types[key] = CurrentType(type, from_assignment)
   162
   163	    def _get(self, key: Key, index: int = -1) -> CurrentType | None:
   164	        if index < 0:
   165	            index += len(self.frames)
   166	        for i in range(index, -1, -1):
   167	            if key in self.frames[i].types:
   168	                return self.frames[i].types[key]
   169	        return None
   170
   171	    def put(self, expr: Expression, typ: Type, *, from_assignment: bool = True) -> None:
   172	        """Directly set the narrowed type of expression (if it supports it).
   173
   174	        This is used for isinstance() etc. Assignments should go through assign_type().
   175	        """
   176	        if not isinstance(expr, (IndexExpr, MemberExpr, NameExpr)):
   177	            return
   178	        if not literal(expr):
   179	            return
   180	        key = literal_hash(expr)
   181	        assert key is not None, "Internal error: binder tried to put non-literal"
   182	        if key not in self.declarations:
   183	            self.declarations[key] = get_declaration(expr)
   184	            self._add_dependencies(key)
   185	        self._put(key, typ, from_assignment)
   186
   187	    def unreachable(self) -> None:
   188	        self.frames[-1].unreachable = True
   189
   190	    def suppress_unreachable_warnings(self) -> None:
   191	        self.frames[-1].suppress_unreachable_warnings = True
   192
   193	    def get(self, expr: Expression) -> Type | None:
   194	        key = literal_hash(expr)
   195	        assert key is not None, "Internal error: binder tried to get non-literal"
   196	        found = self._get(key)
   197	        if found is None:
   198	            return None
   199	        return found.type

   201	    def is_unreachable(self) -> bool:
   202	        # TODO: Copy the value of unreachable into new frames to avoid
   203	        # this traversal on every statement?
   204	        return any(f.unreachable for f in self.frames)
   205
   206	    def is_unreachable_warning_suppressed(self) -> bool:
   207	        return any(f.suppress_unreachable_warnings for f in self.frames)

   209	    def cleanse(self, expr: Expression) -> None:
   210	        """Remove all references to a Node from the binder."""
   211	        key = literal_hash(expr)
   212	        assert key is not None, "Internal error: binder tried cleanse non-literal"
   213	        self._cleanse_key(key)
   214
   215	    def _cleanse_key(self, key: Key) -> None:
   216	        """Remove all references to a key from the binder."""
   217	        for frame in self.frames:
   218	            if key in frame.types:
   219	                del frame.types[key]

   221	    def update_from_options(self, frames: list[Frame]) -> bool:
   222	        """Update the frame to reflect that each key will be updated
   223	        as in one of the frames.  Return whether any item changes.
   224
   225	        If a key is declared as AnyType, only update it if all the
   226	        options are the same.
   227	        """
   228	        all_reachable = all(not f.unreachable for f in frames)
   229	        frames = [f for f in frames if not f.unreachable]
   230	        changed = False
   231	        keys = {key for f in frames for key in f.types}
   232
   233	        for key in keys:
   234	            current_value = self._get(key)
   235	            resulting_values = [f.types.get(key, current_value) for f in frames]
   236	            # Keys can be narrowed using two different semantics. The new semantics
   237	            # is enabled for plain variables when bind_all is true, and it allows
   238	            # variable types to be widened using subsequent assignments. This is
   239	            # tricky to support for instance attributes (primarily due to deferrals),
   240	            # so we don't use it for them.
   241	            old_semantics = not self.bind_all or extract_var_from_literal_hash(key) is None
   242	            if old_semantics and any(x is None for x in resulting_values):
   243	                # We didn't know anything about key before
   244	                # (current_value must be None), and we still don't
   245	                # know anything about key in at least one possible frame.
   246	                continue
   247
   248	            resulting_values = [x for x in resulting_values if x is not None]
   249
   250	            if all_reachable and all(
   251	                x is not None and not x.from_assignment for x in resulting_values
   252	            ):
   253	                # Do not synthesize a new type if we encountered a conditional block
   254	                # (if, while or match-case) without assignments.
   255	                # See check-isinstance.test::testNoneCheckDoesNotMakeTypeVarOptional
   256	                # This is a safe assumption: the fact that we checked something with `is`
   257	                # or `isinstance` does not change the type of the value.
   258	                continue
   259
   260	            current_type = resulting_values[0]
   261	            assert current_type is not None
   262	            type = current_type.type
   263	            declaration_type = get_proper_type(self.declarations.get(key))
   264	            if isinstance(declaration_type, AnyType):
   265	                # At this point resulting values can't contain None, see continue above
   266	                if not all(
   267	                    t is not None and is_same_type(type, t.type) for t in resulting_values[1:]
   268	                ):
   269	                    type = AnyType(TypeOfAny.from_another_any, source_any=declaration_type)
   270	            else:
   271	                possible_types = []
   272	                for t in resulting_values:
   273	                    assert t is not None
   274	                    possible_types.append(t.type)
   275	                if len(possible_types) == 1:
   276	                    # This is to avoid calling get_proper_type() unless needed, as this may
   277	                    # interfere with our (hacky) TypeGuard support.
   278	                    type = possible_types[0]
   279	                else:
   280	                    type = make_simplified_union(possible_types)
   281	                    # Legacy guard for corner case when the original type is TypeVarType.
   282	                    if isinstance(declaration_type, TypeVarType) and not is_subtype(
   283	                        type, declaration_type
   284	                    ):
   285	                        type = declaration_type
   286	                    # Try simplifying resulting type for unions involving variadic tuples.
   287	                    # Technically, everything is still valid without this step, but if we do
   288	                    # not do this, this may create long unions after exiting an if check like:
   289	                    #     x: tuple[int, ...]
   290	                    #     if len(x) < 10:
   291	                    #         ...
   292	                    # We want the type of x to be tuple[int, ...] after this block (if it is
   293	                    # still equivalent to such type).
   294	                    if isinstance(type, UnionType):
   295	                        type = collapse_variadic_union(type)
   296	                    if (
   297	                        old_semantics
   298	                        and isinstance(type, ProperType)
   299	                        and isinstance(type, UnionType)
   300	                    ):
   301	                        # Simplify away any extra Any's that were added to the declared
   302	                        # type when popping a frame.
   303	                        simplified = UnionType.make_union(
   304	                            [t for t in type.items if not isinstance(get_proper_type(t), AnyType)]
   305	                        )
   306	                        if simplified == self.declarations[key]:
   307	                            type = simplified
   308	            if current_value is None or not is_same_type(type, current_value.type):
   309	                self._put(key, type, from_assignment=True)
   310	                changed = True
   311
   312	        self.frames[-1].unreachable = not frames
   313
   314	        return changed

   316	    def pop_frame(self, can_skip: bool, fall_through: int) -> Frame:
   317	        """Pop a frame and return it.
   318
   319	        See frame_context() for documentation of fall_through.
   320	        """
   321
   322	        if fall_through > 0:
   323	            self.allow_jump(-fall_through)
   324
   325	        result = self.frames.pop()
   326	        options = self.options_on_return.pop()
   327
   328	        if can_skip:
   329	            options.insert(0, self.frames[-1])
   330
   331	        self.last_pop_changed = self.update_from_options(options)
   332
   333	        return result

   337	        """Push a new map to collect assigned types in multiassign from union.
   338
   339	        If this map is not None, actual binding is deferred until all items in
   340	        the union are processed (a union of collected items is later bound
   341	        manually by the caller).

   344	        if self.type_assignments is not None:
   345	            old_assignments = self.type_assignments

   347	        yield self.type_assignments

   350	    def assign_type(self, expr: Expression, type: Type, declared_type: Type | None) -> None:
   351	        """Narrow type of expression through an assignment.
   352
   353	        Do nothing if the expression doesn't support narrowing.
   354
   355	        When not narrowing though an assignment (isinstance() etc.), use put()
   356	        directly. This omits some special-casing logic for assignments.
   357	        """
   358	        # We should erase last known value in binder, because if we are using it,
   359	        # it means that the target is not final, and therefore can't hold a literal.
   360	        type = remove_instance_last_known_values(type)
   361
   362	        if self.type_assignments is not None:
   363	            # We are in a multiassign from union, defer the actual binding,
   364	            # just collect the types.
   365	            self.type_assignments[expr].append((type, declared_type))
   366	            return
   367	        if not isinstance(expr, (IndexExpr, MemberExpr, NameExpr)):
   368	            return
   369	        if not literal(expr):
   370	            return
   371	        self.invalidate_dependencies(expr)
   372
   373	        if declared_type is None:
   374	            # Not sure why this happens.  It seems to mainly happen in
   375	            # member initialization.
   376	            return
   377	        if not is_subtype(type, declared_type):
   378	            # Pretty sure this is only happens when there's a type error.
   379
   380	            # Ideally this function wouldn't be called if the
   381	            # expression has a type error, though -- do other kinds of
   382	            # errors cause this function to get called at invalid
   383	            # times?
   384	            return
   385
   386	        p_declared = get_proper_type(declared_type)
   387	        p_type = get_proper_type(type)
   388	        if isinstance(p_type, AnyType):
   389	            # Any type requires some special casing, for both historical reasons,
   390	            # and to optimise user experience without sacrificing correctness too much.
   391	            if isinstance(expr, RefExpr) and isinstance(expr.node, Var) and expr.node.is_inferred:
   392	                # First case: a local/global variable without explicit annotation,
   393	                # in this case we just assign Any (essentially following the SSA logic).
   394	                self.put(expr, type)
   395	            elif isinstance(p_declared, UnionType) and any(
   396	                isinstance(get_proper_type(item), NoneType) for item in p_declared.items
   397	            ):
   398	                # Second case: explicit optional type, in this case we optimize for a common
   399	                # pattern when an untyped value used as a fallback replacing None.
   400	                new_items = [
   401	                    type if isinstance(get_proper_type(item), NoneType) else item
   402	                    for item in p_declared.items
   403	                ]
   404	                self.put(expr, UnionType(new_items))
   405	            elif isinstance(p_declared, UnionType) and any(
   406	                isinstance(get_proper_type(item), AnyType) for item in p_declared.items
   407	            ):
   408	                # Third case: a union already containing Any (most likely from an un-imported
   409	                # name), in this case we allow assigning Any as well.
   410	                self.put(expr, type)
   411	            else:
   412	                # In all other cases we don't narrow to Any to minimize false negatives.
   413	                self.put(expr, declared_type)
   414	        elif isinstance(p_declared, AnyType):
   415	            # Mirroring the first case above, we don't narrow to a precise type if the variable
   416	            # has an explicit `Any` type annotation.
   417	            if isinstance(expr, RefExpr) and isinstance(expr.node, Var) and expr.node.is_inferred:
   418	                self.put(expr, type)
   419	            else:
   420	                self.put(expr, declared_type)
   421	        else:
   422	            self.put(expr, type)
   423
   424	        for i in self.try_frames:
   425	            # XXX This should probably not copy the entire frame, but
   426	            # just copy this variable into a single stored frame.
   427	            self.allow_jump(i)
   428
   429	    def invalidate_dependencies(self, expr: BindableExpression) -> None:
   430	        """Invalidate knowledge of types that include expr, but not expr itself.
   431
   432	        For example, when expr is foo.bar, invalidate foo.bar.baz.
   433
   434	        It is overly conservative: it invalidates globally, including
   435	        in code paths unreachable from here.
   436	        """
   437	        key = literal_hash(expr)
   438	        assert key is not None
   439	        for dep in self.dependencies.get(key, set()):
   440	            self._cleanse_key(dep)
   441
   442	    def allow_jump(self, index: int) -> None:
   443	        # self.frames and self.options_on_return have different lengths
   444	        # so make sure the index is positive
   445	        if index < 0:
   446	            index += len(self.options_on_return)
   447	        frame = Frame(self._get_id())
   448	        for f in self.frames[index + 1 :]:
   449	            frame.types.update(f.types)
   450	            if f.unreachable:
   451	                frame.unreachable = True
   452	        self.options_on_return[index].append(frame)
   453
   454	    def handle_break(self) -> None:
   455	        self.allow_jump(self.break_frames[-1])
   456	        self.unreachable()

   458	    def handle_continue(self) -> None:
   459	        self.allow_jump(self.continue_frames[-1])

   462	    @contextmanager
   463	    def frame_context(
   464	        self,
   465	        *,
   466	        can_skip: bool,
   467	        fall_through: int = 1,
   468	        break_frame: int = 0,
   469	        continue_frame: int = 0,
   470	        conditional_frame: bool = False,
   471	        try_frame: bool = False,
   472	    ) -> Iterator[Frame]:
   473	        """Return a context manager that pushes/pops frames on enter/exit.
   474
   475	        If can_skip is True, control flow is allowed to bypass the
   476	        newly-created frame.
   477
   478	        If fall_through > 0, then it will allow control flow that
   479	        falls off the end of the frame to escape to its ancestor
   480	        `fall_through` levels higher. Otherwise control flow ends
   481	        at the end of the frame.
   482
   483	        If break_frame > 0, then 'break' statements within this frame
   484	        will jump out to the frame break_frame levels higher than the
   485	        frame created by this call to frame_context. Similarly for
   486	        continue_frame and 'continue' statements.
   487
   488	        If try_frame is true, then execution is allowed to jump at any
   489	        point within the newly created frame (or its descendants) to
   490	        its parent (i.e., to the frame that was on top before this
   491	        call to frame_context).
   492
   493	        After the context manager exits, self.last_pop_changed indicates
   494	        whether any types changed in the newly-topmost frame as a result
   495	        of popping this frame.
   496	        """
   497	        assert len(self.frames) > 1
   498
   499	        if break_frame:
   500	            self.break_frames.append(len(self.frames) - break_frame)
   501	        if continue_frame:
   502	            self.continue_frames.append(len(self.frames) - continue_frame)
   503	        if try_frame:
   504	            self.try_frames.add(len(self.frames) - 1)
   505
   506	        new_frame = self.push_frame(conditional_frame)
   507	        if try_frame:
   508	            # An exception may occur immediately
   509	            self.allow_jump(-1)
   510	        yield new_frame
   511	        self.pop_frame(can_skip, fall_through)
   512
   513	        if break_frame:
   514	            self.break_frames.pop()
   515	        if continue_frame:
   516	            self.continue_frames.pop()
   517	        if try_frame:
   518	            self.try_frames.remove(len(self.frames) - 1)
   519
   520	    @contextmanager
   521	    def top_frame_context(self) -> Iterator[Frame]:
   522	        """A variant of frame_context for use at the top level of
   523	        a namespace (module, function, or class).
   524	        """
   525	        assert len(self.frames) == 1
   526	        yield self.push_frame()
   527	        self.pop_frame(True, 0)
   528	        assert len(self.frames) == 1

   531	def get_declaration(expr: BindableExpression) -> Type | None:
   532	    """Get the declared or inferred type of a RefExpr expression.
   533
   534	    Return None if there is no type or the expression is not a RefExpr.
   535	    This can return None if the type hasn't been inferred yet.
   536	    """
   537	    if isinstance(expr, RefExpr):
   538	        if isinstance(expr.node, Var):
   539	            type = expr.node.type
   540	            if not isinstance(get_proper_type(type), PartialType):
   541	                return type
   542	        elif isinstance(expr.node, TypeInfo):
   543	            return TypeType(fill_typevars_with_any(expr.node))
   544	    return None

   547	def collapse_variadic_union(typ: UnionType) -> Type:
   548	    """Simplify a union involving variadic tuple if possible.
   549
   550	    This will collapse a type like e.g.
   551	        tuple[X, Z] | tuple[X, Y, Z] | tuple[X, Y, Y, *tuple[Y, ...], Z]
   552	    back to
   553	        tuple[X, *tuple[Y, ...], Z]
   554	    which is equivalent, but much simpler form of the same type.
   555	    """
   556	    tuple_items = []
   557	    other_items = []
   558	    for t in typ.items:
   559	        p_t = get_proper_type(t)
   560	        if isinstance(p_t, TupleType):
   561	            tuple_items.append(p_t)
   562	        else:
   563	            other_items.append(t)
   564	    if len(tuple_items) <= 1:
   565	        # This type cannot be simplified further.
   566	        return typ
   567	    tuple_items = sorted(tuple_items, key=lambda t: len(t.items))
   568	    first = tuple_items[0]
   569	    last = tuple_items[-1]
   570	    unpack_index = find_unpack_in_list(last.items)
   571	    if unpack_index is None:
   572	        return typ
   573	    unpack = last.items[unpack_index]
   574	    assert isinstance(unpack, UnpackType)
   575	    unpacked = get_proper_type(unpack.type)
   576	    if not isinstance(unpacked, Instance):
   577	        return typ
   578	    assert unpacked.type.fullname == "builtins.tuple"
   579	    suffix = last.items[unpack_index + 1 :]
   580
   581	    # Check that first item matches the expected pattern and infer prefix.
   582	    if len(first.items) < len(suffix):
   583	        return typ
   584	    if suffix and first.items[-len(suffix) :] != suffix:
   585	        return typ
   586	    if suffix:
   587	        prefix = first.items[: -len(suffix)]
   588	    else:
   589	        prefix = first.items
   590
   591	    # Check that all middle types match the expected pattern as well.
   592	    arg = unpacked.args[0]
   593	    for i, it in enumerate(tuple_items[1:-1]):
   594	        if it.items != prefix + [arg] * (i + 1) + suffix:
   595	            return typ
   596
   597	    # Check the last item (the one with unpack), and choose an appropriate simplified type.
   598	    if last.items != prefix + [arg] * (len(typ.items) - 1) + [unpack] + suffix:
   599	        return typ
   600	    if len(first.items) == 0:
   601	        simplified: Type = unpacked.copy_modified()
   602	    else:
   603	        simplified = TupleType(prefix + [unpack] + suffix, fallback=last.partial_fallback)
   604	    return UnionType.make_union([simplified] + other_items)

   492	def load_plugins(
   493	    options: Options, errors: Errors, stdout: TextIO, extra_plugins: Sequence[Plugin]
   494	) -> tuple[Plugin, dict[str, str]]:
   495	    """Load all configured plugins.
   496
   497	    Return a plugin that encapsulates all plugins chained together. Always
   498	    at least include the default plugin (it's last in the chain).

   500	    plugins (for cache validation).

   502	    custom_plugins, snapshot = load_plugins_from_config(options, errors, stdout)

   510	    # Custom plugins take precedence over the default plugin.
   511	    return ChainedPlugin(options, custom_plugins + [default_plugin]), snapshot

   515	    """Take plugin module snapshot by recording its version and hash.
   516
   517	    We record _both_ hash and the version to detect more possible changes
   518	    (e.g. if there is a change in modules imported by a plugin).
   519	    """
   520	    if hasattr(module, "__file__"):

   530	def find_config_file_line_number(path: str, section: str, setting_name: str) -> int:
   531	    """Return the approximate location of setting_name within mypy config file.
   532
   533	    Return -1 if can't determine the line unambiguously.
   534	    """
   535	    in_desired_section = False
   536	    try:
   537	        results = []
   538	        with open(path, encoding="UTF-8") as f:

   752	    def all_imported_modules_in_file(self, file: MypyFile) -> list[tuple[int, str, int]]:
   753	        """Find all reachable import statements in a file.
   754
   755	        Return list of tuples (priority, module id, import line number)
   756	        for all modules imported in file; lower numbers == higher priority.
   757
   758	        Can generate blocking errors on bogus relative imports.
   759	        """
   760
   761	        def correct_rel_imp(imp: ImportFrom | ImportAll) -> str:
   762	            """Function to correct for relative imports."""
   763	            file_id = file.fullname
   764	            rel = imp.relative
   765	            if rel == 0:
   766	                return imp.id
   767	            if os.path.basename(file.path).startswith("__init__."):
   768	                rel -= 1
   769	            if rel != 0:
   770	                file_id = ".".join(file_id.split(".")[:-rel])
   771	            new_id = file_id + "." + imp.id if imp.id else file_id
   772
   773	            if not new_id:
   774	                self.errors.set_file(file.path, file.name, self.options)
   775	                self.errors.report(
   776	                    imp.line, 0, "No parent module -- cannot perform relative import", blocker=True
   777	                )
   778
   779	            return new_id
   780
   781	        res: list[tuple[int, str, int]] = []
   782	        for imp in file.imports:
   783	            if not imp.is_unreachable:
   784	                if isinstance(imp, Import):
   785	                    pri = import_priority(imp, PRI_MED)
   786	                    ancestor_pri = import_priority(imp, PRI_LOW)
   787	                    for id, _ in imp.ids:
   788	                        res.append((pri, id, imp.line))
   789	                        ancestor_parts = id.split(".")[:-1]
   790	                        ancestors = []
   791	                        for part in ancestor_parts:
   792	                            ancestors.append(part)
   793	                            res.append((ancestor_pri, ".".join(ancestors), imp.line))
   794	                elif isinstance(imp, ImportFrom):
   795	                    cur_id = correct_rel_imp(imp)
   796	                    all_are_submodules = True
   797	                    # Also add any imported names that are submodules.
   798	                    pri = import_priority(imp, PRI_MED)
   799	                    for name, __ in imp.names:
   800	                        sub_id = cur_id + "." + name
   801	                        if self.is_module(sub_id):
   802	                            res.append((pri, sub_id, imp.line))
   803	                        else:
   804	                            all_are_submodules = False
   805	                    # Add cur_id as a dependency, even if all of the
   806	                    # imports are submodules. Processing import from will try
   807	                    # to look through cur_id, so we should depend on it.
   808	                    # As a workaround for for some bugs in cycle handling (#4498),
   809	                    # if all of the imports are submodules, do the import at a lower
   810	                    # priority.
   811	                    pri = import_priority(imp, PRI_HIGH if not all_are_submodules else PRI_LOW)
   812	                    res.append((pri, cur_id, imp.line))
   813	                elif isinstance(imp, ImportAll):
   814	                    pri = import_priority(imp, PRI_HIGH)
   815	                    res.append((pri, correct_rel_imp(imp), imp.line))
   816
   817	        # Sort such that module (e.g. foo.bar.baz) comes before its ancestors (e.g. foo
   818	        # and foo.bar) so that, if FindModuleCache finds the target module in a
   819	        # package marked with py.typed underneath a namespace package installed in
   820	        # site-packages, (gasp), that cache's knowledge of the ancestors
   821	        # (aka FindModuleCache.ns_ancestors) can be primed when it is asked to find
   822	        # the parent.
   823	        res.sort(key=lambda x: -x[1].count("."))
   824	        return res

   831	        self, id: str, path: str, source: str, ignore_errors: bool, options: Options
   832	    ) -> MypyFile:

   927	def write_deps_cache(
   928	    rdeps: dict[str, dict[str, set[str]]], manager: BuildManager, graph: Graph
   929	) -> None:
   930	    """Write cache files for fine-grained dependencies.
   931
   932	    Serialize fine-grained dependencies map for fine grained mode.
   933
   934	    Dependencies on some module 'm' is stored in the dependency cache
   935	    file m.deps.json.  This entails some spooky action at a distance:
   936	    if module 'n' depends on 'm', that produces entries in m.deps.json.
   937	    When there is a dependency on a module that does not exist in the
   938	    build, it is stored with its first existing parent module. If no
   939	    such module exists, it is stored with the fake module FAKE_ROOT_MODULE.
   940
   941	    This means that the validity of the fine-grained dependency caches
   942	    are a global property, so we store validity checking information for
   943	    fine-grained dependencies in a global cache file:
   944	     * We take a snapshot of current sources to later check consistency
   945	       between the fine-grained dependency cache and module cache metadata
   946	     * We store the mtime of all of the dependency files to verify they
   947	       haven't changed
   948	    """
   949	    metastore = manager.metastore
   950
   951	    error = False
   952
   953	    fg_deps_meta = manager.fg_deps_meta.copy()
   954
   955	    for id in rdeps:
   956	        if id != FAKE_ROOT_MODULE:
   957	            _, _, deps_json = get_cache_names(id, graph[id].xpath, manager.options)
   958	        else:
   959	            deps_json = DEPS_ROOT_FILE
   960	        assert deps_json
   961	        manager.log("Writing deps cache", deps_json)
   962	        if not manager.metastore.write(deps_json, deps_to_json(rdeps[id])):
   963	            manager.log(f"Error writing fine-grained deps JSON file {deps_json}")
   964	            error = True
   965	        else:
   966	            fg_deps_meta[id] = {"path": deps_json, "mtime": manager.getmtime(deps_json)}
   967
   968	    meta_snapshot: dict[str, str] = {}
   969	    for id, st in graph.items():
   970	        # If we didn't parse a file (so it doesn't have a
   971	        # source_hash), then it must be a module with a fresh cache,
   972	        # so use the hash from that.
   973	        if st.source_hash:
   974	            hash = st.source_hash
   975	        else:
   976	            if st.meta:
   977	                hash = st.meta.hash
   978	            else:
   979	                hash = ""
   980	        meta_snapshot[id] = hash
   981
   982	    meta = {"snapshot": meta_snapshot, "deps_meta": fg_deps_meta}
   983
   984	    if not metastore.write(DEPS_META_FILE, json_dumps(meta)):
   985	        manager.log(f"Error writing fine-grained deps meta JSON file {DEPS_META_FILE}")
   986	        error = True
   987
   988	    if error:
   989	        manager.errors.set_file(_cache_dir_prefix(manager.options), None, manager.options)
   990	        manager.errors.report(0, 0, "Error writing fine-grained dependencies cache", blocker=True)

   993	def invert_deps(deps: dict[str, set[str]], graph: Graph) -> dict[str, dict[str, set[str]]]:
   994	    """Splits fine-grained dependencies based on the module of the trigger.
   995
   996	    Returns a dictionary from module ids to all dependencies on that
   997	    module. Dependencies not associated with a module in the build will be
   998	    associated with the nearest parent module that is in the build, or the
   999	    fake module FAKE_ROOT_MODULE if none are.
  1000	    """
  1001	    # Lazy import to speed up startup
  1002	    from mypy.server.target import trigger_to_target
  1003
  1004	    # Prepopulate the map for all the modules that have been processed,
  1005	    # so that we always generate files for processed modules (even if
  1006	    # there aren't any dependencies to them.)
  1007	    rdeps: dict[str, dict[str, set[str]]] = {id: {} for id, st in graph.items() if st.tree}
  1008	    for trigger, targets in deps.items():
  1009	        module = module_prefix(graph, trigger_to_target(trigger))
  1010	        if not module or not graph[module].tree:
  1011	            module = FAKE_ROOT_MODULE
  1012
  1013	        mod_rdeps = rdeps.setdefault(module, {})
  1014	        mod_rdeps.setdefault(trigger, set()).update(targets)
  1015
  1016	    return rdeps

  1019	def generate_deps_for_cache(manager: BuildManager, graph: Graph) -> dict[str, dict[str, set[str]]]:
  1020	    """Generate fine-grained dependencies into a form suitable for serializing.
  1021
  1022	    This does a couple things:
  1023	    1. Splits fine-grained deps based on the module of the trigger
  1024	    2. For each module we generated fine-grained deps for, load any previous
  1025	       deps and merge them in.
  1026
  1027	    Returns a dictionary from module ids to all dependencies on that
  1028	    module. Dependencies not associated with a module in the build will be
  1029	    associated with the nearest parent module that is in the build, or the
  1030	    fake module FAKE_ROOT_MODULE if none are.
  1031	    """
  1032	    from mypy.server.deps import merge_dependencies  # Lazy import to speed up startup
  1033
  1034	    # Split the dependencies out into based on the module that is depended on.
  1035	    rdeps = invert_deps(manager.fg_deps, graph)
  1036
  1037	    # We can't just clobber existing dependency information, so we
  1038	    # load the deps for every module we've generated new dependencies
  1039	    # to and merge the new deps into them.
  1040	    for module, mdeps in rdeps.items():
  1041	        old_deps = manager.load_fine_grained_deps(module)
  1042	        merge_dependencies(old_deps, mdeps)
  1043
  1044	    return rdeps
