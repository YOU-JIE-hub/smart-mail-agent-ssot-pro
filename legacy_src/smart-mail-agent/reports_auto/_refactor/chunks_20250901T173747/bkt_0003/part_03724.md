   112	def transform_name_expr(builder: IRBuilder, expr: NameExpr) -> Value:
   113	    if isinstance(expr.node, TypeVarLikeExpr) and expr.node.is_new_style:
   114	        # Reference to Python 3.12 implicit TypeVar/TupleVarTuple/... object.
   115	        # These are stored in C statics and not visible in Python namespaces.
   116	        return builder.load_type_var(expr.node.name, expr.node.line)
   117	    if expr.node is None:
   118	        builder.add(
   119	            RaiseStandardError(
   120	                RaiseStandardError.RUNTIME_ERROR,
   121	                "mypyc internal error: should be unreachable",
   122	                expr.line,
   123	            )
   124	        )
   125	        return builder.none()
   126	    fullname = expr.node.fullname
   127	    if fullname in builtin_names:
   128	        typ, src = builtin_names[fullname]
   129	        return builder.add(LoadAddress(typ, src, expr.line))
   130	    # special cases
   131	    if fullname == "builtins.None":
   132	        return builder.none()
   133	    if fullname == "builtins.True":
   134	        return builder.true()
   135	    if fullname == "builtins.False":
   136	        return builder.false()
   137	    if fullname in ("typing.TYPE_CHECKING", "typing_extensions.TYPE_CHECKING"):
   138	        return builder.false()
   139
   140	    math_literal = transform_math_literal(builder, fullname)
   141	    if math_literal is not None:
   142	        return math_literal
   143
   144	    if isinstance(expr.node, Var) and expr.node.is_final:
   145	        value = builder.emit_load_final(
   146	            expr.node,
   147	            fullname,
   148	            expr.name,
   149	            builder.is_native_ref_expr(expr),
   150	            builder.types[expr],
   151	            expr.line,
   152	        )
   153	        if value is not None:
   154	            return value
   155
   156	    if isinstance(expr.node, MypyFile) and expr.node.fullname in builder.imports:
   157	        return builder.load_module(expr.node.fullname)
   158
   159	    # If the expression is locally defined, then read the result from the corresponding
   160	    # assignment target and return it. Otherwise if the expression is a global, load it from
   161	    # the globals dictionary.
   162	    # Except for imports, that currently always happens in the global namespace.
   163	    if expr.kind == LDEF and not (isinstance(expr.node, Var) and expr.node.is_suppressed_import):
   164	        # Try to detect and error when we hit the irritating mypy bug
   165	        # where a local variable is cast to None. (#5423)
   166	        if (
   167	            isinstance(expr.node, Var)
   168	            and is_none_rprimitive(builder.node_type(expr))
   169	            and expr.node.is_inferred
   170	        ):
   171	            builder.error(
   172	                'Local variable "{}" has inferred type None; add an annotation'.format(
   173	                    expr.node.name
   174	                ),
   175	                expr.node.line,
   176	            )
   177
   178	        # TODO: Behavior currently only defined for Var, FuncDef and MypyFile node types.
   179	        if isinstance(expr.node, MypyFile):
   180	            # Load reference to a module imported inside function from
   181	            # the modules dictionary. It would be closer to Python
   182	            # semantics to access modules imported inside functions
   183	            # via local variables, but this is tricky since the mypy
   184	            # AST doesn't include a Var node for the module. We
   185	            # instead load the module separately on each access.
   186	            mod_dict = builder.call_c(get_module_dict_op, [], expr.line)
   187	            obj = builder.primitive_op(
   188	                dict_get_item_op, [mod_dict, builder.load_str(expr.node.fullname)], expr.line
   189	            )
   190	            return obj
   191	        else:
   192	            return builder.read(builder.get_assignment_target(expr, for_read=True), expr.line)
   193
   194	    return builder.load_global(expr)

   197	def transform_member_expr(builder: IRBuilder, expr: MemberExpr) -> Value:
   198	    # Special Cases
   199	    if expr.fullname in ("typing.TYPE_CHECKING", "typing_extensions.TYPE_CHECKING"):
   200	        return builder.false()
   201
   202	    # First check if this is maybe a final attribute.
   203	    final = builder.get_final_ref(expr)
   204	    if final is not None:
   205	        fullname, final_var, native = final
   206	        value = builder.emit_load_final(
   207	            final_var, fullname, final_var.name, native, builder.types[expr], expr.line
   208	        )
   209	        if value is not None:
   210	            return value
   211
   212	    math_literal = transform_math_literal(builder, expr.fullname)
   213	    if math_literal is not None:
   214	        return math_literal
   215
   216	    if isinstance(expr.node, MypyFile) and expr.node.fullname in builder.imports:
   217	        return builder.load_module(expr.node.fullname)
   218
   219	    can_borrow = builder.is_native_attr_ref(expr)
   220	    obj = builder.accept(expr.expr, can_borrow=can_borrow)
   221	    rtype = builder.node_type(expr)
   222
   223	    # Special case: for named tuples transform attribute access to faster index access.
   224	    typ = get_proper_type(builder.types.get(expr.expr))
   225	    if isinstance(typ, TupleType) and typ.partial_fallback.type.is_named_tuple:
   226	        fields = typ.partial_fallback.type.metadata["namedtuple"]["fields"]
   227	        if expr.name in fields:
   228	            index = builder.builder.load_int(fields.index(expr.name))
   229	            return builder.gen_method_call(obj, "__getitem__", [index], rtype, expr.line)
   230
   231	    check_instance_attribute_access_through_class(builder, expr, typ)
   232
   233	    borrow = can_borrow and builder.can_borrow
   234	    return builder.builder.get_attr(obj, expr.name, rtype, expr.line, borrow=borrow)

   238	    builder: IRBuilder, expr: MemberExpr, typ: ProperType | None
   239	) -> None:
   240	    """Report error if accessing an instance attribute through class object."""
   241	    if isinstance(expr.expr, RefExpr):
   242	        node = expr.expr.node
   243	        if isinstance(typ, TypeType) and isinstance(typ.item, Instance):
   244	            # TODO: Handle other item types
   245	            node = typ.item.type
   246	        if isinstance(node, TypeInfo):
   247	            class_ir = builder.mapper.type_to_ir.get(node)
   248	            if class_ir is not None and class_ir.is_ext_class:

   251	                    sym is not None

   253	                    and not sym.node.is_classvar
   254	                    and not sym.node.is_final

   256	                    builder.error(
   257	                        'Cannot access instance attribute "{}" through class object'.format(
   258	                            expr.name
   259	                        ),
   260	                        expr.line,
   261	                    )
   262	                    builder.note(

   264	                        "a class attribute)",
   265	                        expr.line,

   269	def transform_super_expr(builder: IRBuilder, o: SuperExpr) -> Value:
   270	    # warning(builder, 'can not optimize super() expression', o.line)
   271	    sup_val = builder.load_module_attr_by_fullname("builtins.super", o.line)
   272	    if o.call.args:
   273	        args = [builder.accept(arg) for arg in o.call.args]
   274	    else:
   275	        assert o.info is not None
   276	        typ = builder.load_native_type_object(o.info.fullname)
   277	        ir = builder.mapper.type_to_ir[o.info]
   278	        iter_env = iter(builder.builder.args)
   279	        # Grab first argument
   280	        vself: Value = next(iter_env)
   281	        if builder.fn_info.is_generator:
   282	            # grab sixth argument (see comment in translate_super_method_call)
   283	            self_targ = list(builder.symtables[-1].values())[6]
   284	            vself = builder.read(self_targ, builder.fn_info.fitem.line)
   285	        elif not ir.is_ext_class:
   286	            vself = next(iter_env)  # second argument is self if non_extension class
   287	        args = [typ, vself]
   288	    res = builder.py_call(sup_val, args, o.line)
   289	    return builder.py_get_attr(res, o.name, o.line)

   295	def transform_call_expr(builder: IRBuilder, expr: CallExpr) -> Value:
   296	    callee = expr.callee
   297	    if isinstance(expr.analyzed, CastExpr):
   298	        return translate_cast_expr(builder, expr.analyzed)
   299	    elif isinstance(expr.analyzed, AssertTypeExpr):
   300	        # Compile to a no-op.
   301	        return builder.accept(expr.analyzed.expr)
   302	    elif (
   303	        isinstance(callee, (NameExpr, MemberExpr))
   304	        and isinstance(callee.node, TypeInfo)
   305	        and callee.node.is_newtype
   306	    ):
   307	        # A call to a NewType type is a no-op at runtime.
   308	        return builder.accept(expr.args[0])
   309
   310	    if isinstance(callee, IndexExpr) and isinstance(callee.analyzed, TypeApplication):
   311	        callee = callee.analyzed.expr  # Unwrap type application
   312
   313	    if isinstance(callee, MemberExpr):
   314	        if isinstance(callee.expr, RefExpr) and isinstance(callee.expr.node, MypyFile):
   315	            # Call a module-level function, not a method.
   316	            return translate_call(builder, expr, callee)
   317	        return apply_method_specialization(builder, expr, callee) or translate_method_call(
   318	            builder, expr, callee
   319	        )
   320	    elif isinstance(callee, SuperExpr):
   321	        return translate_super_method_call(builder, expr, callee)
   322	    else:
   323	        return translate_call(builder, expr, callee)

   326	def translate_call(builder: IRBuilder, expr: CallExpr, callee: Expression) -> Value:
   327	    # The common case of calls is refexprs
   328	    if isinstance(callee, RefExpr):

   330	            builder, expr, callee

   333	    function = builder.accept(callee)
   334	    args = [builder.accept(arg) for arg in expr.args]
   335	    return builder.py_call(
   336	        function, args, expr.line, arg_kinds=expr.arg_kinds, arg_names=expr.arg_names
   337	    )

   348	def translate_method_call(builder: IRBuilder, expr: CallExpr, callee: MemberExpr) -> Value:
   349	    """Generate IR for an arbitrary call of form e.m(...).
   350
   351	    This can also deal with calls to module-level functions.
   352	    """
   353	    if builder.is_native_ref_expr(callee):
   354	        # Call to module-level native function or such
   355	        return translate_call(builder, expr, callee)
   356	    elif (
   357	        isinstance(callee.expr, RefExpr)
   358	        and isinstance(callee.expr.node, TypeInfo)
   359	        and callee.expr.node in builder.mapper.type_to_ir
   360	        and builder.mapper.type_to_ir[callee.expr.node].has_method(callee.name)
   361	        and all(kind in (ARG_POS, ARG_NAMED) for kind in expr.arg_kinds)
   362	    ):
   363	        # Call a method via the *class*
   364	        assert isinstance(callee.expr.node, TypeInfo)
   365	        ir = builder.mapper.type_to_ir[callee.expr.node]
   366	        return call_classmethod(builder, ir, expr, callee)
   367	    elif builder.is_module_member_expr(callee):
   368	        # Fall back to a PyCall for non-native module calls
   369	        function = builder.accept(callee)
   370	        args = [builder.accept(arg) for arg in expr.args]
   371	        return builder.py_call(
   372	            function, args, expr.line, arg_kinds=expr.arg_kinds, arg_names=expr.arg_names
   373	        )
   374	    else:
   375	        if isinstance(callee.expr, RefExpr):
   376	            node = callee.expr.node
   377	            if isinstance(node, Var) and node.is_cls:
   378	                typ = get_proper_type(node.type)
   379	                if isinstance(typ, TypeType) and isinstance(typ.item, Instance):
   380	                    class_ir = builder.mapper.type_to_ir.get(typ.item.type)
   381	                    if class_ir and class_ir.is_ext_class and class_ir.has_no_subclasses():
   382	                        # Call a native classmethod via cls that can be statically bound,
   383	                        # since the class has no subclasses.
   384	                        return call_classmethod(builder, class_ir, expr, callee)
   385
   386	        receiver_typ = builder.node_type(callee.expr)
   387
   388	        # If there is a specializer for this method name/type, try calling it.
   389	        # We would return the first successful one.
   390	        val = apply_method_specialization(builder, expr, callee, receiver_typ)
   391	        if val is not None:
   392	            return val
   393
   394	        obj = builder.accept(callee.expr)
   395	        args = [builder.accept(arg) for arg in expr.args]
   396	        return builder.gen_method_call(
   397	            obj,
   398	            callee.name,
   399	            args,
   400	            builder.node_type(expr),
   401	            expr.line,
   402	            expr.arg_kinds,
   403	            expr.arg_names,
   404	        )

   407	def call_classmethod(builder: IRBuilder, ir: ClassIR, expr: CallExpr, callee: MemberExpr) -> Value:
   408	    decl = ir.method_decl(callee.name)
   409	    args = []
   410	    arg_kinds, arg_names = expr.arg_kinds.copy(), expr.arg_names.copy()
   411	    # Add the class argument for class methods in extension classes
   412	    if decl.kind == FUNC_CLASSMETHOD and ir.is_ext_class:
   413	        args.append(builder.load_native_type_object(ir.fullname))
   414	        arg_kinds.insert(0, ARG_POS)
   415	        arg_names.insert(0, None)

   418	    if ir.is_ext_class:
   419	        return builder.builder.call(decl, args, arg_kinds, arg_names, expr.line)
   420	    else:
   421	        obj = builder.accept(callee.expr)
   422	        return builder.gen_method_call(
   423	            obj,
   424	            callee.name,
   425	            args,
   426	            builder.node_type(expr),
   427	            expr.line,
   428	            expr.arg_kinds,
   429	            expr.arg_names,
   430	        )

   433	def translate_super_method_call(builder: IRBuilder, expr: CallExpr, callee: SuperExpr) -> Value:
   434	    if callee.info is None or (len(callee.call.args) != 0 and len(callee.call.args) != 2):
   435	        return translate_call(builder, expr, callee)
   436
   437	    # We support two-argument super but only when it is super(CurrentClass, self)
   438	    # TODO: We could support it when it is a parent class in many cases?
   439	    if len(callee.call.args) == 2:
   440	        self_arg = callee.call.args[1]
   441	        if (
   442	            not isinstance(self_arg, NameExpr)
   443	            or not isinstance(self_arg.node, Var)
   444	            or not self_arg.node.is_self
   445	        ):
   446	            return translate_call(builder, expr, callee)
   447
   448	        typ_arg = callee.call.args[0]
   449	        if (
   450	            not isinstance(typ_arg, NameExpr)
   451	            or not isinstance(typ_arg.node, TypeInfo)
   452	            or callee.info is not typ_arg.node
   453	        ):
   454	            return translate_call(builder, expr, callee)
   455
   456	    ir = builder.mapper.type_to_ir[callee.info]
   457	    # Search for the method in the mro, skipping ourselves. We
   458	    # determine targets of super calls to native methods statically.
   459	    for base in ir.mro[1:]:
   460	        if callee.name in base.method_decls:
   461	            break
   462	    else:
   463	        if (
   464	            ir.is_ext_class
   465	            and ir.builtin_base is None
   466	            and not ir.inherits_python
   467	            and callee.name == "__init__"
   468	            and len(expr.args) == 0
   469	        ):
   470	            # Call translates to object.__init__(self), which is a
   471	            # no-op, so omit the call.
   472	            return builder.none()
   473	        return translate_call(builder, expr, callee)
   474
   475	    decl = base.method_decl(callee.name)
   476	    arg_values = [builder.accept(arg) for arg in expr.args]
   477	    arg_kinds, arg_names = expr.arg_kinds.copy(), expr.arg_names.copy()
   478
   479	    if decl.kind != FUNC_STATICMETHOD:
   480	        # Grab first argument
   481	        vself: Value = builder.self()
   482	        if decl.kind == FUNC_CLASSMETHOD:
   483	            vself = builder.primitive_op(type_op, [vself], expr.line)
   484	        elif builder.fn_info.is_generator:
   485	            # For generator classes, the self target is the 6th value
   486	            # in the symbol table (which is an ordered dict). This is sort
   487	            # of ugly, but we can't search by name since the 'self' parameter
   488	            # could be named anything, and it doesn't get added to the
   489	            # environment indexes.
   490	            self_targ = list(builder.symtables[-1].values())[6]
   491	            vself = builder.read(self_targ, builder.fn_info.fitem.line)
   492	        arg_values.insert(0, vself)
   493	        arg_kinds.insert(0, ARG_POS)
   494	        arg_names.insert(0, None)
   495
   496	    return builder.builder.call(decl, arg_values, arg_kinds, arg_names, expr.line)

   516	def transform_op_expr(builder: IRBuilder, expr: OpExpr) -> Value:
   517	    if expr.op in ("and", "or"):
   518	        return builder.shortcircuit_expr(expr)
   519
   520	    # Special case for string formatting
   521	    if expr.op == "%" and isinstance(expr.left, (StrExpr, BytesExpr)):
   522	        ret = translate_printf_style_formatting(builder, expr.left, expr.right)
   523	        if ret is not None:
   524	            return ret
   525
   526	    folded = try_constant_fold(builder, expr)
   527	    if folded:
   528	        return folded
   529
   530	    borrow_left = False
   531	    borrow_right = False
   532
   533	    ltype = builder.node_type(expr.left)
   534	    rtype = builder.node_type(expr.right)
   535
   536	    # Special case some int ops to allow borrowing operands.
   537	    if is_int_rprimitive(ltype) and is_int_rprimitive(rtype):
   538	        if expr.op == "//":
   539	            expr = try_optimize_int_floor_divide(expr)
   540	        if expr.op in int_borrow_friendly_op:
   541	            borrow_left = is_borrow_friendly_expr(builder, expr.right)
   542	            borrow_right = True
   543	    elif is_fixed_width_rtype(ltype) and is_fixed_width_rtype(rtype):
   544	        borrow_left = is_borrow_friendly_expr(builder, expr.right)
   545	        borrow_right = True
   546
   547	    left = builder.accept(expr.left, can_borrow=borrow_left)
   548	    right = builder.accept(expr.right, can_borrow=borrow_right)
   549	    return builder.binary_op(left, right, expr.op, expr.line)

   552	def try_optimize_int_floor_divide(expr: OpExpr) -> OpExpr:
   553	    """Replace // with a power of two with a right shift, if possible."""
   554	    if not isinstance(expr.right, IntExpr):
   555	        return expr
   556	    divisor = expr.right.value
   557	    shift = divisor.bit_length() - 1
   558	    if 0 < shift < 28 and divisor == (1 << shift):
   559	        return OpExpr(">>", expr.left, IntExpr(shift))
   560	    return expr

   563	def transform_index_expr(builder: IRBuilder, expr: IndexExpr) -> Value:
   564	    index = expr.index
   565	    base_type = builder.node_type(expr.base)
   566	    is_list = is_list_rprimitive(base_type)
   567	    can_borrow_base = is_list and is_borrow_friendly_expr(builder, index)

   569	    base = builder.accept(expr.base, can_borrow=can_borrow_base)

   571	    if isinstance(base.type, RTuple) and isinstance(index, IntExpr):

   580	    return builder.gen_method_call(
   581	        base, "__getitem__", [index_reg], builder.node_type(expr), expr.line
   582	    )

   596	def try_gen_slice_op(builder: IRBuilder, base: Value, index: SliceExpr) -> Value | None:
   597	    """Generate specialized slice op for some index expressions.
   598

   702	def transform_comparison_expr(builder: IRBuilder, e: ComparisonExpr) -> Value:
   703	    # x in (...)/[...]
   704	    # x not in (...)/[...]
   705	    first_op = e.operators[0]
   706	    if (
   707	        first_op in ["in", "not in"]
   708	        and len(e.operators) == 1
   709	        and isinstance(e.operands[1], (TupleExpr, ListExpr))
   710	    ):
   711	        items = e.operands[1].items
   712	        n_items = len(items)
   713	        # x in y -> x == y[0] or ... or x == y[n]
   714	        # x not in y -> x != y[0] and ... and x != y[n]
   715	        # 16 is arbitrarily chosen to limit code size
   716	        if 1 < n_items < 16:
   717	            if e.operators[0] == "in":
   718	                bin_op = "or"
   719	                cmp_op = "=="
   720	            else:
   721	                bin_op = "and"
   722	                cmp_op = "!="
   723	            lhs = e.operands[0]
   724	            mypy_file = builder.graph["builtins"].tree
   725	            assert mypy_file is not None
   726	            info = mypy_file.names["bool"].node
   727	            assert isinstance(info, TypeInfo)
   728	            bool_type = Instance(info, [])
   729	            exprs = []
   730	            for item in items:
   731	                expr = ComparisonExpr([cmp_op], [lhs, item])
   732	                builder.types[expr] = bool_type
   733	                exprs.append(expr)
   734
   735	            or_expr: Expression = exprs.pop(0)
   736	            for expr in exprs:
   737	                or_expr = OpExpr(bin_op, or_expr, expr)
   738	                builder.types[or_expr] = bool_type
   739	            return builder.accept(or_expr)
   740	        # x in [y]/(y) -> x == y
   741	        # x not in [y]/(y) -> x != y
   742	        elif n_items == 1:
   743	            if e.operators[0] == "in":
   744	                cmp_op = "=="
   745	            else:
   746	                cmp_op = "!="
   747	            e.operators = [cmp_op]
   748	            e.operands[1] = items[0]
   749	        # x in []/() -> False
   750	        # x not in []/() -> True
   751	        elif n_items == 0:
   752	            if e.operators[0] == "in":
   753	                return builder.false()
   754	            else:
   755	                return builder.true()
   756
   757	    # x in {...}
   758	    # x not in {...}
   759	    if (
   760	        first_op in ("in", "not in")
   761	        and len(e.operators) == 1
   762	        and isinstance(e.operands[1], SetExpr)
   763	    ):
   764	        set_literal = precompute_set_literal(builder, e.operands[1])
   765	        if set_literal is not None:
   766	            lhs = e.operands[0]
   767	            result = builder.builder.primitive_op(
   768	                set_in_op, [builder.accept(lhs), set_literal], e.line, bool_rprimitive
   769	            )
   770	            if first_op == "not in":
   771	                return builder.unary_op(result, "not", e.line)
   772	            return result
   773
   774	    if len(e.operators) == 1:
   775	        # Special some common simple cases
   776	        if first_op in ("is", "is not"):
   777	            right_expr = e.operands[1]
   778	            if isinstance(right_expr, NameExpr) and right_expr.fullname == "builtins.None":
   779	                # Special case 'is None' / 'is not None'.
   780	                return translate_is_none(builder, e.operands[0], negated=first_op != "is")
   781	        left_expr = e.operands[0]
   782	        if is_int_rprimitive(builder.node_type(left_expr)):
   783	            right_expr = e.operands[1]
   784	            if is_int_rprimitive(builder.node_type(right_expr)):
   785	                if first_op in int_borrow_friendly_op:
   786	                    borrow_left = is_borrow_friendly_expr(builder, right_expr)
   787	                    left = builder.accept(left_expr, can_borrow=borrow_left)
   788	                    right = builder.accept(right_expr, can_borrow=True)
   789	                    return builder.binary_op(left, right, first_op, e.line)
   790
   791	    # TODO: Don't produce an expression when used in conditional context
   792	    # All of the trickiness here is due to support for chained conditionals
   793	    # (`e1 < e2 > e3`, etc). `e1 < e2 > e3` is approximately equivalent to
   794	    # `e1 < e2 and e2 > e3` except that `e2` is only evaluated once.
   795	    expr_type = builder.node_type(e)
   796
   797	    # go(i, prev) generates code for `ei opi e{i+1} op{i+1} ... en`,
   798	    # assuming that prev contains the value of `ei`.
   799	    def go(i: int, prev: Value) -> Value:
   800	        if i == len(e.operators) - 1:
   801	            return transform_basic_comparison(
   802	                builder, e.operators[i], prev, builder.accept(e.operands[i + 1]), e.line
   803	            )
   804
   805	        next = builder.accept(e.operands[i + 1])
   806	        return builder.builder.shortcircuit_helper(
   807	            "and",
   808	            expr_type,
   809	            lambda: transform_basic_comparison(builder, e.operators[i], prev, next, e.line),
   810	            lambda: go(i + 1, next),
   811	            e.line,
   812	        )
   813
   814	    return go(0, builder.accept(e.operands[0]))

   822	def transform_basic_comparison(
   823	    builder: IRBuilder, op: str, left: Value, right: Value, line: int
   824	) -> Value:
   825	    if is_fixed_width_rtype(left.type) and op in ComparisonOp.signed_ops:
   826	        if right.type == left.type:
   827	            if left.type.is_signed:
   828	                op_id = ComparisonOp.signed_ops[op]
   829	            else:
   830	                op_id = ComparisonOp.unsigned_ops[op]
   831	            return builder.builder.comparison_op(left, right, op_id, line)
   832	        elif isinstance(right, Integer):
   833	            if left.type.is_signed:
   834	                op_id = ComparisonOp.signed_ops[op]
   835	            else:
   836	                op_id = ComparisonOp.unsigned_ops[op]
   837	            return builder.builder.comparison_op(
   838	                left, builder.coerce(right, left.type, line), op_id, line
   839	            )
   840	    elif (
   841	        is_fixed_width_rtype(right.type)
   842	        and op in ComparisonOp.signed_ops
   843	        and isinstance(left, Integer)
   844	    ):
   845	        if right.type.is_signed:
   846	            op_id = ComparisonOp.signed_ops[op]
   847	        else:
   848	            op_id = ComparisonOp.unsigned_ops[op]
   849	        return builder.builder.comparison_op(
   850	            builder.coerce(left, right.type, line), right, op_id, line
   851	        )
   852
   853	    negate = False
   854	    if op == "is not":
   855	        op, negate = "is", True
   856	    elif op == "not in":
   857	        op, negate = "in", True
   858
   859	    target = builder.binary_op(left, right, op, line)
   860
   861	    if negate:
   862	        target = builder.unary_op(target, "not", line)
   863	    return target

    75	def for_loop_helper(
    76	    builder: IRBuilder,
    77	    index: Lvalue,
    78	    expr: Expression,
    79	    body_insts: GenFunc,
    80	    else_insts: GenFunc | None,
    81	    is_async: bool,
    82	    line: int,
    83	) -> None:
    84	    """Generate IR for a loop.
    85
    86	    Args:
    87	        index: the loop index Lvalue
    88	        expr: the expression to iterate over
    89	        body_insts: a function that generates the body of the loop
    90	        else_insts: a function that generates the else block instructions
    91	    """
    92	    # Body of the loop
    93	    body_block = BasicBlock()
    94	    # Block that steps to the next item
    95	    step_block = BasicBlock()
    96	    # Block for the else clause, if we need it
    97	    else_block = BasicBlock()
    98	    # Block executed after the loop
    99	    exit_block = BasicBlock()
   100
   101	    # Determine where we want to exit, if our condition check fails.
   102	    normal_loop_exit = else_block if else_insts is not None else exit_block
   103
   104	    for_gen = make_for_loop_generator(
   105	        builder, index, expr, body_block, normal_loop_exit, line, is_async=is_async
   106	    )
   107
   108	    builder.push_loop_stack(step_block, exit_block)
   109	    condition_block = BasicBlock()
   110	    builder.goto_and_activate(condition_block)
   111
   112	    # Add loop condition check.
   113	    for_gen.gen_condition()
   114
   115	    # Generate loop body.
   116	    builder.activate_block(body_block)
   117	    for_gen.begin_body()
   118	    body_insts()
   119
   120	    # We generate a separate step block (which might be empty).
   121	    builder.goto_and_activate(step_block)
   122	    for_gen.gen_step()
   123	    # Go back to loop condition.
   124	    builder.goto(condition_block)
   125
   126	    for_gen.add_cleanup(normal_loop_exit)
   127	    builder.pop_loop_stack()
   128
   129	    if else_insts is not None:
   130	        builder.activate_block(else_block)
   131	        else_insts()
   132	        builder.goto(exit_block)
   133
   134	    builder.activate_block(exit_block)

   137	def for_loop_helper_with_index(
   138	    builder: IRBuilder,
   139	    index: Lvalue,
   140	    expr: Expression,
   141	    expr_reg: Value,
   142	    body_insts: Callable[[Value], None],
   143	    line: int,
   144	) -> None:
   145	    """Generate IR for a sequence iteration.
   146
   147	    This function only works for sequence type. Compared to for_loop_helper,
   148	    it would feed iteration index to body_insts.
   149
   150	    Args:
   151	        index: the loop index Lvalue
   152	        expr: the expression to iterate over
   153	        body_insts: a function that generates the body of the loop.
   154	                    It needs a index as parameter.
   155	    """
   156	    assert is_sequence_rprimitive(expr_reg.type)
   157	    target_type = builder.get_sequence_type(expr)
   158
   159	    body_block = BasicBlock()
   160	    step_block = BasicBlock()
   161	    exit_block = BasicBlock()
   162	    condition_block = BasicBlock()
   163
   164	    for_gen = ForSequence(builder, index, body_block, exit_block, line, False)
   165	    for_gen.init(expr_reg, target_type, reverse=False)
   166
   167	    builder.push_loop_stack(step_block, exit_block)
   168
   169	    builder.goto_and_activate(condition_block)
   170	    for_gen.gen_condition()
   171
   172	    builder.activate_block(body_block)
   173	    for_gen.begin_body()
   174	    body_insts(builder.read(for_gen.index_target))
   175
   176	    builder.goto_and_activate(step_block)
   177	    for_gen.gen_step()
   178	    builder.goto(condition_block)
   179
   180	    for_gen.add_cleanup(exit_block)
   181	    builder.pop_loop_stack()
   182
   183	    builder.activate_block(exit_block)

   186	def sequence_from_generator_preallocate_helper(
   187	    builder: IRBuilder,
   188	    gen: GeneratorExpr,
   189	    empty_op_llbuilder: Callable[[Value, int], Value],
   190	    set_item_op: CFunctionDescription,
   191	) -> Value | None:
   192	    """Generate a new tuple or list from a simple generator expression.
   193
   194	    Currently we only optimize for simplest generator expression, which means that
   195	    there is no condition list in the generator and only one original sequence with
   196	    one index is allowed.
   197
   198	    e.g.  (1) tuple(f(x) for x in a_list/a_tuple)
   199	          (2) list(f(x) for x in a_list/a_tuple)
   200	          (3) [f(x) for x in a_list/a_tuple]
   201	    RTuple as an original sequence is not supported yet.
   202
   203	    Args:
   204	        empty_op_llbuilder: A function that can generate an empty sequence op when
   205	            passed in length. See `new_list_op_with_length` and `new_tuple_op_with_length`
   206	            for detailed implementation.
   207	        set_item_op: A primitive that can modify an arbitrary position of a sequence.
   208	            The op should have three arguments:
   209	                - Self
   210	                - Target position
   211	                - New Value
   212	            See `new_list_set_item_op` and `new_tuple_set_item_op` for detailed
   213	            implementation.
   214	    """
   215	    if len(gen.sequences) == 1 and len(gen.indices) == 1 and len(gen.condlists[0]) == 0:
   216	        rtype = builder.node_type(gen.sequences[0])
   217	        if is_list_rprimitive(rtype) or is_tuple_rprimitive(rtype) or is_str_rprimitive(rtype):
   218	            sequence = builder.accept(gen.sequences[0])
   219	            length = builder.builder.builtin_len(sequence, gen.line, use_pyssize_t=True)
   220	            target_op = empty_op_llbuilder(length, gen.line)
   221
   222	            def set_item(item_index: Value) -> None:
   223	                e = builder.accept(gen.left_expr)
   224	                builder.call_c(set_item_op, [target_op, item_index, e], gen.line)
   225
   226	            for_loop_helper_with_index(
   227	                builder, gen.indices[0], gen.sequences[0], sequence, set_item, gen.line
   228	            )
   229
   230	            return target_op
   231	    return None

   235	    if raise_error_if_contains_unreachable_names(builder, gen):
   236	        return builder.none()
   237
   238	    # Try simplest list comprehension, otherwise fall back to general one
   239	    val = sequence_from_generator_preallocate_helper(

   295	def comprehension_helper(
   296	    builder: IRBuilder,
   297	    loop_params: list[tuple[Lvalue, Expression, list[Expression], bool]],
   298	    gen_inner_stmts: Callable[[], None],
   299	    line: int,
   300	) -> None:
   301	    """Helper function for list comprehensions.
   302
   303	    Args:
   304	        loop_params: a list of (index, expr, [conditions]) tuples defining nested loops:
   305	            - "index" is the Lvalue indexing that loop;
   306	            - "expr" is the expression for the object to be iterated over;
   307	            - "conditions" is a list of conditions, evaluated in order with short-circuiting,
   308	                that must all be true for the loop body to be executed
   309	        gen_inner_stmts: function to generate the IR for the body of the innermost loop
   310	    """
   311
   312	    def handle_loop(loop_params: list[tuple[Lvalue, Expression, list[Expression], bool]]) -> None:
   313	        """Generate IR for a loop.
   314
   315	        Given a list of (index, expression, [conditions]) tuples, generate IR
   316	        for the nested loops the list defines.
   317	        """
   318	        index, expr, conds, is_async = loop_params[0]
   319	        for_loop_helper(
   320	            builder,
   321	            index,
   322	            expr,
   323	            lambda: loop_contents(conds, loop_params[1:]),
   324	            None,
   325	            is_async=is_async,
   326	            line=line,
   327	        )
   328
   329	    def loop_contents(
   330	        conds: list[Expression],
   331	        remaining_loop_params: list[tuple[Lvalue, Expression, list[Expression], bool]],
   332	    ) -> None:
   333	        """Generate the body of the loop.
   334
   335	        Args:
   336	            conds: a list of conditions to be evaluated (in order, with short circuiting)
   337	                to gate the body of the loop
   338	            remaining_loop_params: the parameters for any further nested loops; if it's empty
   339	                we'll instead evaluate the "gen_inner_stmts" function
   340	        """
   341	        # Check conditions, in order, short circuiting them.
   342	        for cond in conds:
   343	            cond_val = builder.accept(cond)
   344	            cont_block, rest_block = BasicBlock(), BasicBlock()
   345	            # If the condition is true we'll skip the continue.
   346	            builder.add_bool_branch(cond_val, rest_block, cont_block)
   347	            builder.activate_block(cont_block)
   348	            builder.nonlocal_control[-1].gen_continue(builder, cond.line)
   349	            builder.goto_and_activate(rest_block)
   350
   351	        if remaining_loop_params:
   352	            # There's another nested level, so the body of this loop is another loop.
   353	            return handle_loop(remaining_loop_params)
   354	        else:
   355	            # We finally reached the actual body of the generator.
   356	            # Generate the IR for the inner loop body.
   357	            gen_inner_stmts()
   358
   359	    handle_loop(loop_params)

   370	def make_for_loop_generator(
   371	    builder: IRBuilder,
   372	    index: Lvalue,
   373	    expr: Expression,
   374	    body_block: BasicBlock,
   375	    loop_exit: BasicBlock,
   376	    line: int,
   377	    is_async: bool = False,
   378	    nested: bool = False,
   379	) -> ForGenerator:
   380	    """Return helper object for generating a for loop over an iterable.
   381
   382	    If "nested" is True, this is a nested iterator such as "e" in "enumerate(e)".
   383	    """
   384
   385	    # Do an async loop if needed. async is always generic
   386	    if is_async:
   387	        expr_reg = builder.accept(expr)
   388	        async_obj = ForAsyncIterable(builder, index, body_block, loop_exit, line, nested)
   389	        item_type = builder._analyze_iterable_item_type(expr)
   390	        item_rtype = builder.type_to_rtype(item_type)
   391	        async_obj.init(expr_reg, item_rtype)
   392	        return async_obj
   393
   394	    rtyp = builder.node_type(expr)
   395	    if is_sequence_rprimitive(rtyp):
   396	        # Special case "for x in <list>".
   397	        expr_reg = builder.accept(expr)
   398	        target_type = builder.get_sequence_type(expr)
   399
   400	        for_list = ForSequence(builder, index, body_block, loop_exit, line, nested)
   401	        for_list.init(expr_reg, target_type, reverse=False)
   402	        return for_list
   403
   404	    if is_dict_rprimitive(rtyp):
   405	        # Special case "for k in <dict>".
   406	        expr_reg = builder.accept(expr)
   407	        target_type = builder.get_dict_key_type(expr)
   408
   409	        for_dict = ForDictionaryKeys(builder, index, body_block, loop_exit, line, nested)
   410	        for_dict.init(expr_reg, target_type)
   411	        return for_dict
   412
   413	    if isinstance(expr, CallExpr) and isinstance(expr.callee, RefExpr):
   414	        if (
   415	            is_range_ref(expr.callee)
   416	            and (
   417	                len(expr.args) <= 2
   418	                or (len(expr.args) == 3 and builder.extract_int(expr.args[2]) is not None)
   419	            )
   420	            and set(expr.arg_kinds) == {ARG_POS}
   421	        ):
   422	            # Special case "for x in range(...)".
   423	            # We support the 3 arg form but only for int literals, since it doesn't
   424	            # seem worth the hassle of supporting dynamically determining which
   425	            # direction of comparison to do.
   426	            if len(expr.args) == 1:
   427	                start_reg: Value = Integer(0)
   428	                end_reg = builder.accept(expr.args[0])
   429	            else:
   430	                start_reg = builder.accept(expr.args[0])
   431	                end_reg = builder.accept(expr.args[1])
   432	            if len(expr.args) == 3:
   433	                step = builder.extract_int(expr.args[2])
   434	                assert step is not None
   435	                if step == 0:
   436	                    builder.error("range() step can't be zero", expr.args[2].line)
   437	            else:
   438	                step = 1
   439
   440	            for_range = ForRange(builder, index, body_block, loop_exit, line, nested)
   441	            for_range.init(start_reg, end_reg, step)
   442	            return for_range
   443
   444	        elif (
   445	            expr.callee.fullname == "builtins.enumerate"
   446	            and len(expr.args) == 1
   447	            and expr.arg_kinds == [ARG_POS]
   448	            and isinstance(index, TupleExpr)
   449	            and len(index.items) == 2
   450	        ):
   451	            # Special case "for i, x in enumerate(y)".
   452	            lvalue1 = index.items[0]
   453	            lvalue2 = index.items[1]
   454	            for_enumerate = ForEnumerate(builder, index, body_block, loop_exit, line, nested)
   455	            for_enumerate.init(lvalue1, lvalue2, expr.args[0])
   456	            return for_enumerate
   457
   458	        elif (
   459	            expr.callee.fullname == "builtins.zip"
   460	            and len(expr.args) >= 2
   461	            and set(expr.arg_kinds) == {ARG_POS}
   462	            and isinstance(index, TupleExpr)
   463	            and len(index.items) == len(expr.args)
   464	        ):
   465	            # Special case "for x, y in zip(a, b)".
   466	            for_zip = ForZip(builder, index, body_block, loop_exit, line, nested)
   467	            for_zip.init(index.items, expr.args)
   468	            return for_zip
   469
   470	        if (
   471	            expr.callee.fullname == "builtins.reversed"
   472	            and len(expr.args) == 1
   473	            and expr.arg_kinds == [ARG_POS]
   474	            and is_sequence_rprimitive(builder.node_type(expr.args[0]))
   475	        ):
   476	            # Special case "for x in reversed(<list>)".
   477	            expr_reg = builder.accept(expr.args[0])
   478	            target_type = builder.get_sequence_type(expr)
   479
   480	            for_list = ForSequence(builder, index, body_block, loop_exit, line, nested)
   481	            for_list.init(expr_reg, target_type, reverse=True)
   482	            return for_list
   483	    if isinstance(expr, CallExpr) and isinstance(expr.callee, MemberExpr) and not expr.args:
   484	        # Special cases for dictionary iterator methods, like dict.items().
   485	        rtype = builder.node_type(expr.callee.expr)
   486	        if is_dict_rprimitive(rtype) and expr.callee.name in ("keys", "values", "items"):
   487	            expr_reg = builder.accept(expr.callee.expr)
   488	            for_dict_type: type[ForGenerator] | None = None
   489	            if expr.callee.name == "keys":
   490	                target_type = builder.get_dict_key_type(expr.callee.expr)
   491	                for_dict_type = ForDictionaryKeys
   492	            elif expr.callee.name == "values":
   493	                target_type = builder.get_dict_value_type(expr.callee.expr)
   494	                for_dict_type = ForDictionaryValues
   495	            else:
   496	                target_type = builder.get_dict_item_type(expr.callee.expr)
   497	                for_dict_type = ForDictionaryItems
   498	            for_dict_gen = for_dict_type(builder, index, body_block, loop_exit, line, nested)
   499	            for_dict_gen.init(expr_reg, target_type)
   500	            return for_dict_gen
   501
   502	    iterable_expr_reg: Value | None = None
   503	    if isinstance(expr, SetExpr):
   504	        # Special case "for x in <set literal>".
   505	        from mypyc.irbuild.expression import precompute_set_literal
   506
   507	        set_literal = precompute_set_literal(builder, expr)
   508	        if set_literal is not None:
   509	            iterable_expr_reg = set_literal
   510
   511	    # Default to a generic for loop.
   512	    if iterable_expr_reg is None:
   513	        iterable_expr_reg = builder.accept(expr)
   514	    for_obj = ForIterable(builder, index, body_block, loop_exit, line, nested)
   515	    item_type = builder._analyze_iterable_item_type(expr)
   516	    item_rtype = builder.type_to_rtype(item_type)
   517	    for_obj.init(iterable_expr_reg, item_rtype)
   518	    return for_obj

   521	class ForGenerator:
   522	    """Abstract base class for generating for loops."""
   523
   524	    def __init__(
   525	        self,
   526	        builder: IRBuilder,
   527	        index: Lvalue,
   528	        body_block: BasicBlock,
   529	        loop_exit: BasicBlock,
   530	        line: int,
   531	        nested: bool,
   532	    ) -> None:
   533	        self.builder = builder
   534	        self.index = index
   535	        self.body_block = body_block
   536	        self.line = line
   537	        # Some for loops need a cleanup block that we execute at exit. We
   538	        # create a cleanup block if needed. However, if we are generating a for
   539	        # loop for a nested iterator, such as "e" in "enumerate(e)", the
   540	        # outermost generator should generate the cleanup block -- we don't
   541	        # need to do it here.
   542	        if self.need_cleanup() and not nested:
   543	            # Create a new block to handle cleanup after loop exit.
   544	            self.loop_exit = BasicBlock()
   545	        else:
   546	            # Just use the existing loop exit block.
   547	            self.loop_exit = loop_exit
   548
   549	    def need_cleanup(self) -> bool:
   550	        """If this returns true, we need post-loop cleanup."""
   551	        return False
   552
   553	    def add_cleanup(self, exit_block: BasicBlock) -> None:
   554	        """Add post-loop cleanup, if needed."""
   555	        if self.need_cleanup():
   556	            self.builder.activate_block(self.loop_exit)
   557	            self.gen_cleanup()
   558	            self.builder.goto(exit_block)
   559
   560	    def gen_condition(self) -> None:
   561	        """Generate check for loop exit (e.g. exhaustion of iteration)."""
   562
   563	    def begin_body(self) -> None:
   564	        """Generate ops at the beginning of the body (if needed)."""
   565
   566	    def gen_step(self) -> None:
   567	        """Generate stepping to the next item (if needed)."""
   568
   569	    def gen_cleanup(self) -> None:
   570	        """Generate post-loop cleanup (if needed)."""
   571
   572	    def load_len(self, expr: Value | AssignmentTarget) -> Value:
   573	        """A helper to get collection length, used by several subclasses."""
   574	        return self.builder.builder.builtin_len(self.builder.read(expr, self.line), self.line)
   575

   577	class ForIterable(ForGenerator):
   578	    """Generate IR for a for loop over an arbitrary iterable (the general case)."""
   579
   580	    def need_cleanup(self) -> bool:
   581	        # Create a new cleanup block for when the loop is finished.
   582	        return True
   583
   584	    def init(self, expr_reg: Value, target_type: RType) -> None:
   585	        # Define targets to contain the expression, along with the iterator that will be used
   586	        # for the for-loop. If we are inside of a generator function, spill these into the
   587	        # environment class.
   588	        builder = self.builder
   589	        iter_reg = builder.primitive_op(iter_op, [expr_reg], self.line)
   590	        builder.maybe_spill(expr_reg)
   591	        self.iter_target = builder.maybe_spill(iter_reg)
   592	        self.target_type = target_type
   593
   594	    def gen_condition(self) -> None:
   595	        # We call __next__ on the iterator and check to see if the return value
   596	        # is NULL, which signals either the end of the Iterable being traversed
   597	        # or an exception being raised. Note that Branch.IS_ERROR checks only
   598	        # for NULL (an exception does not necessarily have to be raised).
   599	        builder = self.builder
   600	        line = self.line
   601	        self.next_reg = builder.call_c(next_op, [builder.read(self.iter_target, line)], line)
   602	        builder.add(Branch(self.next_reg, self.loop_exit, self.body_block, Branch.IS_ERROR))
   603
   604	    def begin_body(self) -> None:
   605	        # Assign the value obtained from __next__ to the
   606	        # lvalue so that it can be referenced by code in the body of the loop.
   607	        builder = self.builder
   608	        line = self.line
   609	        # We unbox here so that iterating with tuple unpacking generates a tuple based
   610	        # unpack instead of an iterator based one.
   611	        next_reg = builder.coerce(self.next_reg, self.target_type, line)
   612	        builder.assign(builder.get_assignment_target(self.index), next_reg, line)
   613
   614	    def gen_step(self) -> None:
   615	        # Nothing to do here, since we get the next item as part of gen_condition().
   616	        pass
   617
   618	    def gen_cleanup(self) -> None:
   619	        # We set the branch to go here if the conditional evaluates to true. If
   620	        # an exception was raised during the loop, then err_reg will be set to
   621	        # True. If no_err_occurred_op returns False, then the exception will be

   704	    Supports iterating in both forward and reverse.
   705	    """
   706
   707	    def init(self, expr_reg: Value, target_type: RType, reverse: bool) -> None:
   708	        builder = self.builder
   709	        self.reverse = reverse
   710	        # Define target to contain the expression, along with the index that will be used
   711	        # for the for-loop. If we are inside of a generator function, spill these into the
   712	        # environment class.
   713	        self.expr_target = builder.maybe_spill(expr_reg)
   714	        if not reverse:
   715	            index_reg: Value = Integer(0)
   716	        else:
   717	            index_reg = builder.binary_op(
   718	                self.load_len(self.expr_target), Integer(1), "-", self.line
   719	            )
   720	        self.index_target = builder.maybe_spill_assignable(index_reg)
   721	        self.target_type = target_type

   723	    def gen_condition(self) -> None:
   724	        builder = self.builder
   725	        line = self.line
   726	        # TODO: Don't reload the length each time when iterating an immutable sequence?
   727	        if self.reverse:
   728	            # If we are iterating in reverse order, we obviously need
   729	            # to check that the index is still positive. Somewhat less
   730	            # obviously we still need to check against the length,
   731	            # since it could shrink out from under us.
   732	            comparison = builder.binary_op(
   733	                builder.read(self.index_target, line), Integer(0), ">=", line
   734	            )
   735	            second_check = BasicBlock()
   736	            builder.add_bool_branch(comparison, second_check, self.loop_exit)
   737	            builder.activate_block(second_check)
   738	        # For compatibility with python semantics we recalculate the length
   739	        # at every iteration.
   740	        len_reg = self.load_len(self.expr_target)
   741	        comparison = builder.binary_op(builder.read(self.index_target, line), len_reg, "<", line)

   746	        line = self.line
   747	        # Read the next list item.
   748	        value_box = unsafe_index(
   749	            builder,
   750	            builder.read(self.expr_target, line),
   751	            builder.read(self.index_target, line),
   752	            line,
   753	        )
   754	        assert value_box
   755	        # We coerce to the type of list elements here so that
   756	        # iterating with tuple unpacking generates a tuple based
   757	        # unpack instead of an iterator based one.
   758	        builder.assign(
   759	            builder.get_assignment_target(self.index),
   760	            builder.coerce(value_box, self.target_type, line),
   761	            line,
   762	        )

   764	    def gen_step(self) -> None:
   765	        # Step to the next item.
   766	        builder = self.builder
   767	        line = self.line
   768	        step = 1 if not self.reverse else -1
   769	        add = builder.int_op(
   770	            short_int_rprimitive,
   771	            builder.read(self.index_target, line),
   772	            Integer(step),
   773	            IntOp.ADD,
   774	            line,
   775	        )
   776	        builder.assign(self.index_target, add, line)

   779	class ForDictionaryCommon(ForGenerator):
   780	    """Generate optimized IR for a for loop over dictionary keys/values.
   781
   782	    The logic is pretty straightforward, we use PyDict_Next() API wrapped in
   783	    a tuple, so that we can modify only a single register. The layout of the tuple:
   784	      * f0: are there more items (bool)
   785	      * f1: current offset (int)
   786	      * f2: next key (object)
   787	      * f3: next value (object)
   788	    For more info see https://docs.python.org/3/c-api/dict.html#c.PyDict_Next.
   789
   790	    Note that for subclasses we fall back to generic PyObject_GetIter() logic,
   791	    since they may override some iteration methods in subtly incompatible manner.
   792	    The fallback logic is implemented in CPy.h via dynamic type check.
   793	    """
   794
   795	    dict_next_op: ClassVar[CFunctionDescription]
   796	    dict_iter_op: ClassVar[CFunctionDescription]

   798	    def need_cleanup(self) -> bool:
   799	        # Technically, a dict subclass can raise an unrelated exception
   800	        # in __next__(), so we need this.
   801	        return True
   802
   803	    def init(self, expr_reg: Value, target_type: RType) -> None:
   804	        builder = self.builder
   805	        self.target_type = target_type
   806
   807	        # We add some variables to environment class, so they can be read across yield.
   808	        self.expr_target = builder.maybe_spill(expr_reg)
   809	        offset = Integer(0)
   810	        self.offset_target = builder.maybe_spill_assignable(offset)
   811	        self.size = builder.maybe_spill(self.load_len(self.expr_target))
   812
   813	        # For dict class (not a subclass) this is the dictionary itself.
   814	        iter_reg = builder.call_c(self.dict_iter_op, [expr_reg], self.line)
   815	        self.iter_target = builder.maybe_spill(iter_reg)
   816
   817	    def gen_condition(self) -> None:
   818	        """Get next key/value pair, set new offset, and check if we should continue."""
   819	        builder = self.builder
   820	        line = self.line
   821	        self.next_tuple = self.builder.call_c(
   822	            self.dict_next_op,
   823	            [builder.read(self.iter_target, line), builder.read(self.offset_target, line)],
   824	            line,
   825	        )
   826
   827	        # Do this here instead of in gen_step() to minimize variables in environment.
   828	        new_offset = builder.add(TupleGet(self.next_tuple, 1, line))
   829	        builder.assign(self.offset_target, new_offset, line)
   830
   831	        should_continue = builder.add(TupleGet(self.next_tuple, 0, line))
   832	        builder.add(Branch(should_continue, self.body_block, self.loop_exit, Branch.BOOL))
   833
   834	    def gen_step(self) -> None:
   835	        """Check that dictionary didn't change size during iteration.
   836
   837	        Raise RuntimeError if it is not the case to match CPython behavior.
   838	        """
   839	        builder = self.builder
   840	        line = self.line
   841	        # Technically, we don't need a new primitive for this, but it is simpler.
   842	        builder.call_c(
   843	            dict_check_size_op,
   844	            [builder.read(self.expr_target, line), builder.read(self.size, line)],
   845	            line,
   846	        )
   847
   848	    def gen_cleanup(self) -> None:
   849	        # Same as for generic ForIterable.
   850	        self.builder.call_c(no_err_occurred_op, [], self.line)

    96	def tokenizer_format_call(format_str: str) -> tuple[list[str], list[FormatOp]] | None:
    97	    """Tokenize a str.format() format string.
    98
    99	    The core function parse_format_value() is shared with mypy.
   100	    With these specifiers, we then parse the literal substrings
   101	    of the original format string and convert `ConversionSpecifier`
   102	    to `FormatOp`.
   103
   104	    Return:
   105	        A list of string literals and a list of FormatOps. The literals
   106	        are interleaved with FormatOps and the length of returned literals
   107	        should be exactly one more than FormatOps.
   108	        Return None if it cannot parse the string.
   109	    """
   110	    # Creates an empty MessageBuilder here.
   111	    # It wouldn't be used since the code has passed the type-checking.
   112	    specifiers = parse_format_value(
   113	        format_str, EMPTY_CONTEXT, MessageBuilder(Errors(Options()), {})
   114	    )
   115	    if specifiers is None:
   116	        return None
   117	    format_ops = generate_format_ops(specifiers)
   118	    if format_ops is None:
   119	        return None
   120
   121	    literals: list[str] = []
   122	    last_end = 0
   123	    for spec in specifiers:
   124	        # Skip { and }
   125	        literals.append(format_str[last_end : spec.start_pos - 1])
   126	        last_end = spec.start_pos + len(spec.whole_seq) + 1
   127	    literals.append(format_str[last_end:])
   128	    # Deal with escaped {{
   129	    literals = [x.replace("{{", "{").replace("}}", "}") for x in literals]
   130
   131	    return literals, format_ops

   134	def convert_format_expr_to_str(
   135	    builder: IRBuilder, format_ops: list[FormatOp], exprs: list[Expression], line: int
   136	) -> list[Value] | None:
   137	    """Convert expressions into string literal objects with the guidance
   138	    of FormatOps. Return None when fails."""
   139	    if len(format_ops) != len(exprs):
   140	        return None
   141
   142	    converted = []
   143	    for x, format_op in zip(exprs, format_ops):
   144	        node_type = builder.node_type(x)
   145	        if format_op == FormatOp.STR:
   146	            if is_str_rprimitive(node_type):
   147	                var_str = builder.accept(x)
   148	            elif is_int_rprimitive(node_type) or is_short_int_rprimitive(node_type):
   149	                var_str = builder.primitive_op(int_to_str_op, [builder.accept(x)], line)
   150	            else:
   151	                var_str = builder.primitive_op(str_op, [builder.accept(x)], line)
   152	        elif format_op == FormatOp.INT:
   153	            if is_int_rprimitive(node_type) or is_short_int_rprimitive(node_type):
   154	                var_str = builder.primitive_op(int_to_str_op, [builder.accept(x)], line)
   155	            else:
   156	                return None
   157	        else:
   158	            return None
   159	        converted.append(var_str)
   160	    return converted

   163	def join_formatted_strings(
   164	    builder: IRBuilder, literals: list[str] | None, substitutions: list[Value], line: int
   165	) -> Value:
   166	    """Merge the list of literals and the list of substitutions
   167	    alternatively using 'str_build_op'.
   168
   169	    `substitutions` is the result value of formatting conversions.
   170
   171	    If the `literals` is set to None, we simply join the substitutions;
   172	    Otherwise, the `literals` is the literal substrings of the original
   173	    format string and its length should be exactly one more than
   174	    substitutions.
   175
   176	    For example:
   177	    (1)    'This is a %s and the value is %d'
   178	        -> literals: ['This is a ', ' and the value is', '']
   179	    (2)    '{} and the value is {}'
   180	        -> literals: ['', ' and the value is', '']
   181	    """
   182	    # The first parameter for str_build_op is the total size of
   183	    # the following PyObject*
   184	    result_list: list[Value] = [Integer(0, c_pyssize_t_rprimitive)]
   185
   186	    if literals is not None:
   187	        for a, b in zip(literals, substitutions):
   188	            if a:
   189	                result_list.append(builder.load_str(a))
   190	            result_list.append(b)
   191	        if literals[-1]:
   192	            result_list.append(builder.load_str(literals[-1]))
   193	    else:
   194	        result_list.extend(substitutions)
   195
   196	    # Special case for empty string and literal string
   197	    if len(result_list) == 1:
   198	        return builder.load_str("")
   199	    if not substitutions and len(result_list) == 2:
   200	        return result_list[1]
   201
   202	    result_list[0] = Integer(len(result_list) - 1, c_pyssize_t_rprimitive)
   203	    return builder.call_c(str_build_op, result_list, line)
