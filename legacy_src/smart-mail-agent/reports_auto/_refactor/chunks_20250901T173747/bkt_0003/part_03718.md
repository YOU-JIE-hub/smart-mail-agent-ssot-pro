     1	"""Encode valid C string literals from Python strings.
     2
     3	If a character is not allowed in C string literals, it is either emitted
     4	as a simple escape sequence (e.g. '\\n'), or an octal escape sequence
     5	with exactly three digits ('\\oXXX'). Question marks are escaped to
     6	prevent trigraphs in the string literal from being interpreted. Note
     7	that '\\?' is an invalid escape sequence in Python.
     8
     9	Consider the string literal "AB\\xCDEF". As one would expect, Python
    10	parses it as ['A', 'B', 0xCD, 'E', 'F']. However, the C standard
    11	specifies that all hexadecimal digits immediately following '\\x' will
    12	be interpreted as part of the escape sequence. Therefore, it is
    13	unexpectedly parsed as ['A', 'B', 0xCDEF].
    14
    15	Emitting ("AB\\xCD" "EF") would avoid this behaviour. However, we opt
    16	for simplicity and use octal escape sequences instead. They do not
    17	suffer from the same issue as they are defined to parse at most three
    18	octal digits.
    19	"""
    20
    21	from __future__ import annotations
    22
    23	import string
    24	from typing import Final
    25
    26	CHAR_MAP: Final = [f"\\{i:03o}" for i in range(256)]
    27
    28	# It is safe to use string.printable as it always uses the C locale.
    29	for c in string.printable:
    30	    CHAR_MAP[ord(c)] = c
    31
    32	# These assignments must come last because we prioritize simple escape
    33	# sequences over any other representation.
    34	for c in ("'", '"', "\\", "a", "b", "f", "n", "r", "t", "v"):
    35	    escaped = f"\\{c}"
    36	    decoded = escaped.encode("ascii").decode("unicode_escape")
    37	    CHAR_MAP[ord(decoded)] = escaped
    38
    39	# This escape sequence is invalid in Python.
    40	CHAR_MAP[ord("?")] = r"\?"
    41
    42

    44	    """Produce contents of a C string literal for a byte string, without quotes."""
    45	    escaped = "".join([CHAR_MAP[i] for i in b])
    46	    return escaped

   452	                    break
   453	            else:
   454	                assert False, "not expecting tuple with error overlap"
   455	        if isinstance(item_type, RTuple):
   456	            return self.tuple_undefined_check_cond(
   457	                item_type, tuple_expr_in_c + f".f{i}", c_type_compare_val, compare
   458	            )
   459	        else:
   460	            check = f"{tuple_expr_in_c}.f{i} {compare} {c_type_compare_val(item_type)}"
   461	            if rtuple.error_overlap and check_exception:
   462	                check += " && PyErr_Occurred()"
   463	            return check
   464
   465	    def tuple_undefined_value(self, rtuple: RTuple) -> str:
   466	        """Undefined tuple value suitable in an expression."""
   467	        return f"({rtuple.struct_name}) {self.c_initializer_undefined_value(rtuple)}"
   468
   469	    def c_initializer_undefined_value(self, rtype: RType) -> str:
   470	        """Undefined value represented in a form suitable for variable initialization."""
   471	        if isinstance(rtype, RTuple):
   472	            if not rtype.types:
   473	                # Empty tuples contain a flag so that they can still indicate
   474	                # error values.
   475	                return f"{{ {int_rprimitive.c_undefined} }}"
   476	            items = ", ".join([self.c_initializer_undefined_value(t) for t in rtype.types])
   477	            return f"{{ {items} }}"
   478	        else:
   479	            return self.c_undefined_value(rtype)

   495	    def emit_inc_ref(self, dest: str, rtype: RType, *, rare: bool = False) -> None:
   496	        """Increment reference count of C expression `dest`.
   497
   498	        For composite unboxed structures (e.g. tuples) recursively
   499	        increment reference counts for each component.
   500
   501	        If rare is True, optimize for code size and compilation speed.
   502	        """
   503	        if is_int_rprimitive(rtype):
   504	            if rare:
   505	                self.emit_line("CPyTagged_IncRef(%s);" % dest)
   506	            else:
   507	                self.emit_line("CPyTagged_INCREF(%s);" % dest)
   508	        elif isinstance(rtype, RTuple):
   509	            for i, item_type in enumerate(rtype.types):
   510	                self.emit_inc_ref(f"{dest}.f{i}", item_type)
   511	        elif not rtype.is_unboxed:
   512	            # Always inline, since this is a simple but very hot op
   513	            if rtype.may_be_immortal or not HAVE_IMMORTAL:
   514	                self.emit_line("CPy_INCREF(%s);" % dest)
   515	            else:
   516	                self.emit_line("CPy_INCREF_NO_IMM(%s);" % dest)
   517	        # Otherwise assume it's an unboxed, pointerless value and do nothing.
   518
   519	    def emit_dec_ref(
   520	        self, dest: str, rtype: RType, *, is_xdec: bool = False, rare: bool = False
   521	    ) -> None:
   522	        """Decrement reference count of C expression `dest`.
   523
   524	        For composite unboxed structures (e.g. tuples) recursively
   525	        decrement reference counts for each component.
   526
   527	        If rare is True, optimize for code size and compilation speed.
   528	        """
   529	        x = "X" if is_xdec else ""
   530	        if is_int_rprimitive(rtype):
   531	            if rare:
   532	                self.emit_line(f"CPyTagged_{x}DecRef({dest});")
   533	            else:
   534	                # Inlined
   535	                self.emit_line(f"CPyTagged_{x}DECREF({dest});")
   536	        elif isinstance(rtype, RTuple):
   537	            for i, item_type in enumerate(rtype.types):
   538	                self.emit_dec_ref(f"{dest}.f{i}", item_type, is_xdec=is_xdec, rare=rare)
   539	        elif not rtype.is_unboxed:
   540	            if rare:
   541	                self.emit_line(f"CPy_{x}DecRef({dest});")
   542	            else:
   543	                # Inlined
   544	                if rtype.may_be_immortal or not HAVE_IMMORTAL:
   545	                    self.emit_line(f"CPy_{x}DECREF({dest});")
   546	                else:
   547	                    self.emit_line(f"CPy_{x}DECREF_NO_IMM({dest});")
   548	        # Otherwise assume it's an unboxed, pointerless value and do nothing.
   549
   550	    def pretty_name(self, typ: RType) -> str:
   551	        value_type = optional_value_type(typ)
   552	        if value_type is not None:
   553	            return "%s or None" % self.pretty_name(value_type)
   554	        return str(typ)

   556	    def emit_cast(
   557	        self,
   558	        src: str,
   559	        dest: str,
   560	        typ: RType,
   561	        *,
   562	        declare_dest: bool = False,
   563	        error: ErrorHandler | None = None,
   564	        raise_exception: bool = True,
   565	        optional: bool = False,
   566	        src_type: RType | None = None,
   567	        likely: bool = True,
   568	    ) -> None:
   569	        """Emit code for casting a value of given type.
   570
   571	        Somewhat strangely, this supports unboxed types but only
   572	        operates on boxed versions.  This is necessary to properly
   573	        handle types such as Optional[int] in compatibility glue.
   574
   575	        By default, assign NULL (error value) to dest if the value has
   576	        an incompatible type and raise TypeError. These can be customized
   577	        using 'error' and 'raise_exception'.
   578
   579	        Always copy/steal the reference in 'src'.
   580
   581	        Args:
   582	            src: Name of source C variable
   583	            dest: Name of target C variable
   584	            typ: Type of value
   585	            declare_dest: If True, also declare the variable 'dest'
   586	            error: What happens on error
   587	            raise_exception: If True, also raise TypeError on failure
   588	            likely: If the cast is likely to succeed (can be False for unions)
   589	        """
   590	        error = error or AssignHandler()
   591
   592	        # Special case casting *from* optional
   593	        if src_type and is_optional_type(src_type) and not is_object_rprimitive(typ):
   594	            value_type = optional_value_type(src_type)
   595	            assert value_type is not None
   596	            if is_same_type(value_type, typ):
   597	                if declare_dest:
   598	                    self.emit_line(f"PyObject *{dest};")
   599	                check = "({} != Py_None)"
   600	                if likely:
   601	                    check = f"(likely{check})"
   602	                self.emit_arg_check(src, dest, typ, check.format(src), optional)
   603	                self.emit_lines(f"    {dest} = {src};", "else {")
   604	                self.emit_cast_error_handler(error, src, dest, typ, raise_exception)
   605	                self.emit_line("}")
   606	                return
   607
   608	        # TODO: Verify refcount handling.
   609	        if (
   610	            is_list_rprimitive(typ)
   611	            or is_dict_rprimitive(typ)
   612	            or is_set_rprimitive(typ)
   613	            or is_frozenset_rprimitive(typ)
   614	            or is_str_rprimitive(typ)
   615	            or is_range_rprimitive(typ)
   616	            or is_float_rprimitive(typ)
   617	            or is_int_rprimitive(typ)
   618	            or is_bool_rprimitive(typ)
   619	            or is_bit_rprimitive(typ)
   620	            or is_fixed_width_rtype(typ)
   621	        ):
   622	            if declare_dest:
   623	                self.emit_line(f"PyObject *{dest};")
   624	            if is_list_rprimitive(typ):
   625	                prefix = "PyList"
   626	            elif is_dict_rprimitive(typ):
   627	                prefix = "PyDict"
   628	            elif is_set_rprimitive(typ):
   629	                prefix = "PySet"
   630	            elif is_frozenset_rprimitive(typ):
   631	                prefix = "PyFrozenSet"
   632	            elif is_str_rprimitive(typ):
   633	                prefix = "PyUnicode"
   634	            elif is_range_rprimitive(typ):
   635	                prefix = "PyRange"
   636	            elif is_float_rprimitive(typ):
   637	                prefix = "CPyFloat"
   638	            elif is_int_rprimitive(typ) or is_fixed_width_rtype(typ):
   639	                # TODO: Range check for fixed-width types?
   640	                prefix = "PyLong"
   641	            elif is_bool_rprimitive(typ) or is_bit_rprimitive(typ):
   642	                prefix = "PyBool"
   643	            else:
   644	                assert False, f"unexpected primitive type: {typ}"
   645	            check = "({}_Check({}))"
   646	            if likely:
   647	                check = f"(likely{check})"
   648	            self.emit_arg_check(src, dest, typ, check.format(prefix, src), optional)
   649	            self.emit_lines(f"    {dest} = {src};", "else {")
   650	            self.emit_cast_error_handler(error, src, dest, typ, raise_exception)
   651	            self.emit_line("}")
   652	        elif is_bytes_rprimitive(typ):
   653	            if declare_dest:
   654	                self.emit_line(f"PyObject *{dest};")
   655	            check = "(PyBytes_Check({}) || PyByteArray_Check({}))"
   656	            if likely:
   657	                check = f"(likely{check})"
   658	            self.emit_arg_check(src, dest, typ, check.format(src, src), optional)
   659	            self.emit_lines(f"    {dest} = {src};", "else {")
   660	            self.emit_cast_error_handler(error, src, dest, typ, raise_exception)
   661	            self.emit_line("}")
   662	        elif is_tuple_rprimitive(typ):
   663	            if declare_dest:
   664	                self.emit_line(f"{self.ctype(typ)} {dest};")
   665	            check = "(PyTuple_Check({}))"
   666	            if likely:
   667	                check = f"(likely{check})"
   668	            self.emit_arg_check(src, dest, typ, check.format(src), optional)
   669	            self.emit_lines(f"    {dest} = {src};", "else {")
   670	            self.emit_cast_error_handler(error, src, dest, typ, raise_exception)
   671	            self.emit_line("}")
   672	        elif isinstance(typ, RInstance):
   673	            if declare_dest:
   674	                self.emit_line(f"PyObject *{dest};")
   675	            concrete = all_concrete_classes(typ.class_ir)
   676	            # If there are too many concrete subclasses or we can't find any
   677	            # (meaning the code ought to be dead or we aren't doing global opts),
   678	            # fall back to a normal typecheck.
   679	            # Otherwise check all the subclasses.
   680	            if not concrete or len(concrete) > FAST_ISINSTANCE_MAX_SUBCLASSES + 1:
   681	                check = "(PyObject_TypeCheck({}, {}))".format(
   682	                    src, self.type_struct_name(typ.class_ir)
   683	                )
   684	            else:
   685	                full_str = "(Py_TYPE({src}) == {targets[0]})"
   686	                for i in range(1, len(concrete)):
   687	                    full_str += " || (Py_TYPE({src}) == {targets[%d]})" % i
   688	                if len(concrete) > 1:
   689	                    full_str = "(%s)" % full_str
   690	                check = full_str.format(
   691	                    src=src, targets=[self.type_struct_name(ir) for ir in concrete]
   692	                )
   693	            if likely:
   694	                check = f"(likely{check})"
   695	            self.emit_arg_check(src, dest, typ, check, optional)
   696	            self.emit_lines(f"    {dest} = {src};", "else {")
   697	            self.emit_cast_error_handler(error, src, dest, typ, raise_exception)
   698	            self.emit_line("}")
   699	        elif is_none_rprimitive(typ):
   700	            if declare_dest:
   701	                self.emit_line(f"PyObject *{dest};")
   702	            check = "({} == Py_None)"
   703	            if likely:
   704	                check = f"(likely{check})"
   705	            self.emit_arg_check(src, dest, typ, check.format(src), optional)
   706	            self.emit_lines(f"    {dest} = {src};", "else {")
   707	            self.emit_cast_error_handler(error, src, dest, typ, raise_exception)
   708	            self.emit_line("}")
   709	        elif is_object_rprimitive(typ):
   710	            if declare_dest:
   711	                self.emit_line(f"PyObject *{dest};")
   712	            self.emit_arg_check(src, dest, typ, "", optional)
   713	            self.emit_line(f"{dest} = {src};")
   714	            if optional:
   715	                self.emit_line("}")
   716	        elif isinstance(typ, RUnion):
   717	            self.emit_union_cast(
   718	                src, dest, typ, declare_dest, error, optional, src_type, raise_exception
   719	            )
   720	        elif isinstance(typ, RTuple):
   721	            assert not optional
   722	            self.emit_tuple_cast(src, dest, typ, declare_dest, error, src_type)
   723	        else:
   724	            assert False, "Cast not implemented: %s" % typ
   725
   726	    def emit_cast_error_handler(
   727	        self, error: ErrorHandler, src: str, dest: str, typ: RType, raise_exception: bool
   728	    ) -> None:
   729	        if raise_exception:
   730	            if isinstance(error, TracebackAndGotoHandler):
   731	                # Merge raising and emitting traceback entry into a single call.
   732	                self.emit_type_error_traceback(
   733	                    error.source_path, error.module_name, error.traceback_entry, typ=typ, src=src
   734	                )
   735	                self.emit_line("goto %s;" % error.label)
   736	                return
   737	            self.emit_line(f'CPy_TypeError("{self.pretty_name(typ)}", {src}); ')
   738	        if isinstance(error, AssignHandler):
   739	            self.emit_line("%s = NULL;" % dest)
   740	        elif isinstance(error, GotoHandler):
   741	            self.emit_line("goto %s;" % error.label)
   742	        elif isinstance(error, TracebackAndGotoHandler):
   743	            self.emit_line("%s = NULL;" % dest)
   744	            self.emit_traceback(error.source_path, error.module_name, error.traceback_entry)
   745	            self.emit_line("goto %s;" % error.label)
   746	        else:
   747	            assert isinstance(error, ReturnHandler)
   748	            self.emit_line("return %s;" % error.value)
   749
   750	    def emit_union_cast(
   751	        self,
   752	        src: str,
   753	        dest: str,
   754	        typ: RUnion,
   755	        declare_dest: bool,
   756	        error: ErrorHandler,
   757	        optional: bool,
   758	        src_type: RType | None,
   759	        raise_exception: bool,
   760	    ) -> None:
   761	        """Emit cast to a union type.
   762
   763	        The arguments are similar to emit_cast.
   764	        """
   765	        if declare_dest:
   766	            self.emit_line(f"PyObject *{dest};")
   767	        good_label = self.new_label()
   768	        if optional:
   769	            self.emit_line(f"if ({src} == NULL) {{")
   770	            self.emit_line(f"{dest} = {self.c_error_value(typ)};")
   771	            self.emit_line(f"goto {good_label};")
   772	            self.emit_line("}")
   773	        for item in typ.items:
   774	            self.emit_cast(
   775	                src,
   776	                dest,
   777	                item,
   778	                declare_dest=False,
   779	                raise_exception=False,
   780	                optional=False,
   781	                likely=False,
   782	            )
   783	            self.emit_line(f"if ({dest} != NULL) goto {good_label};")
   784	        # Handle cast failure.
   785	        self.emit_cast_error_handler(error, src, dest, typ, raise_exception)
   786	        self.emit_label(good_label)
   787
   788	    def emit_tuple_cast(
   789	        self,
   790	        src: str,
   791	        dest: str,
   792	        typ: RTuple,
   793	        declare_dest: bool,
   794	        error: ErrorHandler,
   795	        src_type: RType | None,
   796	    ) -> None:
   797	        """Emit cast to a tuple type.
   798
   799	        The arguments are similar to emit_cast.
   800	        """
   801	        if declare_dest:
   802	            self.emit_line(f"PyObject *{dest};")
   803	        # This reuse of the variable is super dodgy. We don't even
   804	        # care about the values except to check whether they are
   805	        # invalid.
   806	        out_label = self.new_label()
   807	        self.emit_lines(
   808	            "if (unlikely(!(PyTuple_Check({r}) && PyTuple_GET_SIZE({r}) == {size}))) {{".format(
   809	                r=src, size=len(typ.types)
   810	            ),
   811	            f"{dest} = NULL;",
   812	            f"goto {out_label};",
   813	            "}",
   814	        )
   815	        for i, item in enumerate(typ.types):
   816	            # Since we did the checks above this should never fail
   817	            self.emit_cast(
   818	                f"PyTuple_GET_ITEM({src}, {i})",
   819	                dest,
   820	                item,
   821	                declare_dest=False,
   822	                raise_exception=False,
   823	                optional=False,
   824	            )
   825	            self.emit_line(f"if ({dest} == NULL) goto {out_label};")
   826
   827	        self.emit_line(f"{dest} = {src};")
   828	        self.emit_label(out_label)

   830	    def emit_arg_check(self, src: str, dest: str, typ: RType, check: str, optional: bool) -> None:
   831	        if optional:
   832	            self.emit_line(f"if ({src} == NULL) {{")
   833	            self.emit_line(f"{dest} = {self.c_error_value(typ)};")
   834	        if check != "":
   835	            self.emit_line("{}if {}".format("} else " if optional else "", check))
   836	        elif optional:
   837	            self.emit_line("else {")

   839	    def emit_unbox(
   840	        self,
   841	        src: str,
   842	        dest: str,
   843	        typ: RType,
   844	        *,
   845	        declare_dest: bool = False,
   846	        error: ErrorHandler | None = None,
   847	        raise_exception: bool = True,
   848	        optional: bool = False,
   849	        borrow: bool = False,
   850	    ) -> None:
   851	        """Emit code for unboxing a value of given type (from PyObject *).
   852
   853	        By default, assign error value to dest if the value has an
   854	        incompatible type and raise TypeError. These can be customized
   855	        using 'error' and 'raise_exception'.
   856
   857	        Generate a new reference unless 'borrow' is True.
   858
   859	        Args:
   860	            src: Name of source C variable
   861	            dest: Name of target C variable
   862	            typ: Type of value
   863	            declare_dest: If True, also declare the variable 'dest'
   864	            error: What happens on error
   865	            raise_exception: If True, also raise TypeError on failure
   866	            borrow: If True, create a borrowed reference
   867
   868	        """
   869	        error = error or AssignHandler()
   870	        # TODO: Verify refcount handling.
   871	        if isinstance(error, AssignHandler):
   872	            failure = f"{dest} = {self.c_error_value(typ)};"
   873	        elif isinstance(error, GotoHandler):
   874	            failure = "goto %s;" % error.label
   875	        else:
   876	            assert isinstance(error, ReturnHandler)
   877	            failure = "return %s;" % error.value
   878	        if raise_exception:
   879	            raise_exc = f'CPy_TypeError("{self.pretty_name(typ)}", {src}); '
   880	            failure = raise_exc + failure
   881	        if is_int_rprimitive(typ) or is_short_int_rprimitive(typ):
   882	            if declare_dest:
   883	                self.emit_line(f"CPyTagged {dest};")
   884	            self.emit_arg_check(src, dest, typ, f"(likely(PyLong_Check({src})))", optional)
   885	            if borrow:
   886	                self.emit_line(f"    {dest} = CPyTagged_BorrowFromObject({src});")
   887	            else:
   888	                self.emit_line(f"    {dest} = CPyTagged_FromObject({src});")
   889	            self.emit_line("else {")
   890	            self.emit_line(failure)
   891	            self.emit_line("}")
   892	        elif is_bool_rprimitive(typ) or is_bit_rprimitive(typ):
   893	            # Whether we are borrowing or not makes no difference.
   894	            if declare_dest:
   895	                self.emit_line(f"char {dest};")
   896	            self.emit_arg_check(src, dest, typ, f"(unlikely(!PyBool_Check({src}))) {{", optional)
   897	            self.emit_line(failure)
   898	            self.emit_line("} else")
   899	            conversion = f"{src} == Py_True"
   900	            self.emit_line(f"    {dest} = {conversion};")
   901	        elif is_none_rprimitive(typ):
   902	            # Whether we are borrowing or not makes no difference.
   903	            if declare_dest:
   904	                self.emit_line(f"char {dest};")
   905	            self.emit_arg_check(src, dest, typ, f"(unlikely({src} != Py_None)) {{", optional)
   906	            self.emit_line(failure)
   907	            self.emit_line("} else")
   908	            self.emit_line(f"    {dest} = 1;")
   909	        elif is_int64_rprimitive(typ):
   910	            # Whether we are borrowing or not makes no difference.
   911	            assert not optional  # Not supported for overlapping error values
   912	            if declare_dest:
   913	                self.emit_line(f"int64_t {dest};")
   914	            self.emit_line(f"{dest} = CPyLong_AsInt64({src});")
   915	            if not isinstance(error, AssignHandler):
   916	                self.emit_unbox_failure_with_overlapping_error_value(dest, typ, failure)
   917	        elif is_int32_rprimitive(typ):
   918	            # Whether we are borrowing or not makes no difference.
   919	            assert not optional  # Not supported for overlapping error values
   920	            if declare_dest:
   921	                self.emit_line(f"int32_t {dest};")
   922	            self.emit_line(f"{dest} = CPyLong_AsInt32({src});")
   923	            if not isinstance(error, AssignHandler):
   924	                self.emit_unbox_failure_with_overlapping_error_value(dest, typ, failure)
   925	        elif is_int16_rprimitive(typ):
   926	            # Whether we are borrowing or not makes no difference.
   927	            assert not optional  # Not supported for overlapping error values
   928	            if declare_dest:
   929	                self.emit_line(f"int16_t {dest};")
   930	            self.emit_line(f"{dest} = CPyLong_AsInt16({src});")
   931	            if not isinstance(error, AssignHandler):
   932	                self.emit_unbox_failure_with_overlapping_error_value(dest, typ, failure)
   933	        elif is_uint8_rprimitive(typ):
   934	            # Whether we are borrowing or not makes no difference.
   935	            assert not optional  # Not supported for overlapping error values
   936	            if declare_dest:
   937	                self.emit_line(f"uint8_t {dest};")
   938	            self.emit_line(f"{dest} = CPyLong_AsUInt8({src});")
   939	            if not isinstance(error, AssignHandler):
   940	                self.emit_unbox_failure_with_overlapping_error_value(dest, typ, failure)
   941	        elif is_float_rprimitive(typ):
   942	            assert not optional  # Not supported for overlapping error values
   943	            if declare_dest:
   944	                self.emit_line(f"double {dest};")
   945	            # TODO: Don't use __float__ and __index__
   946	            self.emit_line(f"{dest} = PyFloat_AsDouble({src});")
   947	            self.emit_lines(f"if ({dest} == -1.0 && PyErr_Occurred()) {{", failure, "}")

   199	    finalize_name = f"{name_prefix}_finalize"
   200	    members_name = f"{name_prefix}_members"
   201	    getseters_name = f"{name_prefix}_getseters"
   202	    vtable_name = f"{name_prefix}_vtable"
   203	    traverse_name = f"{name_prefix}_traverse"
   204	    clear_name = f"{name_prefix}_clear"
   205	    dealloc_name = f"{name_prefix}_dealloc"
   206	    methods_name = f"{name_prefix}_methods"
   207	    vtable_setup_name = f"{name_prefix}_trait_vtable_setup"
   208
   209	    fields: dict[str, str] = {"tp_name": f'"{name}"'}
   210
   211	    generate_full = not cl.is_trait and not cl.builtin_base
   212	    needs_getseters = cl.needs_getseters or not cl.is_generated or cl.has_dict
   213
   214	    if not cl.builtin_base:
   215	        fields["tp_new"] = new_name
   216
   217	    if generate_full:
   218	        fields["tp_dealloc"] = f"(destructor){name_prefix}_dealloc"
   219	        fields["tp_traverse"] = f"(traverseproc){name_prefix}_traverse"
   220	        fields["tp_clear"] = f"(inquiry){name_prefix}_clear"
   221	    # Populate .tp_finalize and generate a finalize method only if __del__ is defined for this class.
   222	    del_method = next((e.method for e in cl.vtable_entries if e.name == "__del__"), None)
   223	    if del_method:
   224	        fields["tp_finalize"] = f"(destructor){finalize_name}"
   225	    if needs_getseters:
   226	        fields["tp_getset"] = getseters_name
   227	    fields["tp_methods"] = methods_name
   228
   229	    def emit_line() -> None:
   230	        emitter.emit_line()
   231
   232	    emit_line()
   233
   234	    # If the class has a method to initialize default attribute
   235	    # values, we need to call it during initialization.
   236	    defaults_fn = cl.get_method("__mypyc_defaults_setup")
   237
   238	    # If there is a __init__ method, we'll use it in the native constructor.
   239	    init_fn = cl.get_method("__init__")
   240
   241	    # Fill out slots in the type object from dunder methods.
   242	    fields.update(generate_slots(cl, SLOT_DEFS, emitter))
   243
   244	    # Fill out dunder methods that live in tables hanging off the side.
   245	    for table_name, type, slot_defs in SIDE_TABLES:
   246	        slots = generate_slots(cl, slot_defs, emitter)
   247	        if slots:
   248	            table_struct_name = generate_side_table_for_class(cl, table_name, type, slots, emitter)
   249	            fields[f"tp_{table_name}"] = f"&{table_struct_name}"
   250
   251	    richcompare_name = generate_richcompare_wrapper(cl, emitter)
   252	    if richcompare_name:
   253	        fields["tp_richcompare"] = richcompare_name
   254
   255	    # If the class inherits from python, make space for a __dict__
   256	    struct_name = cl.struct_name(emitter.names)
   257	    if cl.builtin_base:
   258	        base_size = f"sizeof({cl.builtin_base})"
   259	    elif cl.is_trait:
   260	        base_size = "sizeof(PyObject)"
   261	    else:
   262	        base_size = f"sizeof({struct_name})"
   263	    # Since our types aren't allocated using type() we need to
   264	    # populate these fields ourselves if we want them to have correct
   265	    # values. PyType_Ready will inherit the offsets from tp_base but
   266	    # that isn't what we want.
   267
   268	    # XXX: there is no reason for the __weakref__ stuff to be mixed up with __dict__
   269	    if cl.has_dict and not has_managed_dict(cl, emitter):
   270	        # __dict__ lives right after the struct and __weakref__ lives right after that
   271	        # TODO: They should get members in the struct instead of doing this nonsense.
   272	        weak_offset = f"{base_size} + sizeof(PyObject *)"
   273	        emitter.emit_lines(
   274	            f"PyMemberDef {members_name}[] = {{",
   275	            f'{{"__dict__", T_OBJECT_EX, {base_size}, 0, NULL}},',
   276	            f'{{"__weakref__", T_OBJECT_EX, {weak_offset}, 0, NULL}},',
   277	            "{0}",
   278	            "};",
   279	        )
   280
   281	        fields["tp_members"] = members_name
   282	        fields["tp_basicsize"] = f"{base_size} + 2*sizeof(PyObject *)"
   283	        if emitter.capi_version < (3, 12):
   284	            fields["tp_dictoffset"] = base_size
   285	            fields["tp_weaklistoffset"] = weak_offset
   286	    else:
   287	        fields["tp_basicsize"] = base_size
   288
   289	    if generate_full:
   290	        # Declare setup method that allocates and initializes an object. type is the
   291	        # type of the class being initialized, which could be another class if there
   292	        # is an interpreted subclass.
   293	        emitter.emit_line(f"static PyObject *{setup_name}(PyTypeObject *type);")
   294	        assert cl.ctor is not None
   295	        emitter.emit_line(native_function_header(cl.ctor, emitter) + ";")
   296
   297	        emit_line()
   298	        init_fn = cl.get_method("__init__")
   299	        generate_new_for_class(cl, new_name, vtable_name, setup_name, init_fn, emitter)
   300	        emit_line()
   301	        generate_traverse_for_class(cl, traverse_name, emitter)
   302	        emit_line()
   303	        generate_clear_for_class(cl, clear_name, emitter)
   304	        emit_line()
   305	        generate_dealloc_for_class(cl, dealloc_name, clear_name, bool(del_method), emitter)
   306	        emit_line()
   307
   308	        if cl.allow_interpreted_subclasses:
   309	            shadow_vtable_name: str | None = generate_vtables(
   310	                cl, vtable_setup_name + "_shadow", vtable_name + "_shadow", emitter, shadow=True
   311	            )
   312	            emit_line()
   313	        else:
   314	            shadow_vtable_name = None
   315	        vtable_name = generate_vtables(cl, vtable_setup_name, vtable_name, emitter, shadow=False)
   316	        emit_line()
   317	    if del_method:
   318	        generate_finalize_for_class(del_method, finalize_name, emitter)
   319	        emit_line()
   320	    if needs_getseters:
   321	        generate_getseter_declarations(cl, emitter)
   322	        emit_line()
   323	        generate_getseters_table(cl, getseters_name, emitter)
   324	        emit_line()
   325
   326	    if cl.is_trait:
   327	        generate_new_for_trait(cl, new_name, emitter)
   328
   329	    generate_methods_table(cl, methods_name, emitter)
   330	    emit_line()
   331
   332	    flags = ["Py_TPFLAGS_DEFAULT", "Py_TPFLAGS_HEAPTYPE", "Py_TPFLAGS_BASETYPE"]
   333	    if generate_full:
   334	        flags.append("Py_TPFLAGS_HAVE_GC")
   335	    if cl.has_method("__call__"):
   336	        fields["tp_vectorcall_offset"] = "offsetof({}, vectorcall)".format(
   337	            cl.struct_name(emitter.names)
   338	        )
   339	        flags.append("_Py_TPFLAGS_HAVE_VECTORCALL")
   340	        if not fields.get("tp_vectorcall"):
   341	            # This is just a placeholder to please CPython. It will be
   342	            # overridden during setup.
   343	            fields["tp_call"] = "PyVectorcall_Call"
   344	    if has_managed_dict(cl, emitter):
   345	        flags.append("Py_TPFLAGS_MANAGED_DICT")
   346	    fields["tp_flags"] = " | ".join(flags)
   347
   348	    emitter.emit_line(f"static PyTypeObject {emitter.type_struct_name(cl)}_template_ = {{")
   349	    emitter.emit_line("PyVarObject_HEAD_INIT(NULL, 0)")
   350	    for field, value in fields.items():
   351	        emitter.emit_line(f".{field} = {value},")
   352	    emitter.emit_line("};")
   353	    emitter.emit_line(
   354	        "static PyTypeObject *{t}_template = &{t}_template_;".format(
   355	            t=emitter.type_struct_name(cl)
   356	        )
   357	    )
   358
   359	    emitter.emit_line()
   360	    if generate_full:
   361	        generate_setup_for_class(
   362	            cl, setup_name, defaults_fn, vtable_name, shadow_vtable_name, emitter
   363	        )
   364	        emitter.emit_line()
   365	        generate_constructor_for_class(cl, cl.ctor, init_fn, setup_name, vtable_name, emitter)
   366	        emitter.emit_line()
   367	    if needs_getseters:
   368	        generate_getseters(cl, emitter)
   369

   379	def generate_object_struct(cl: ClassIR, emitter: Emitter) -> None:
   380	    seen_attrs: set[tuple[str, RType]] = set()
   381	    lines: list[str] = []
   382	    lines += ["typedef struct {", "PyObject_HEAD", "CPyVTableItem *vtable;"]
   383	    if cl.has_method("__call__"):
   384	        lines.append("vectorcallfunc vectorcall;")
   385	    bitmap_attrs = []
   386	    for base in reversed(cl.base_mro):
   387	        if not base.is_trait:
   388	            if base.bitmap_attrs:
   389	                # Do we need another attribute bitmap field?
   390	                if emitter.bitmap_field(len(base.bitmap_attrs) - 1) not in bitmap_attrs:
   391	                    for i in range(0, len(base.bitmap_attrs), BITMAP_BITS):
   392	                        attr = emitter.bitmap_field(i)
   393	                        if attr not in bitmap_attrs:
   394	                            lines.append(f"{BITMAP_TYPE} {attr};")
   395	                            bitmap_attrs.append(attr)
   396	            for attr, rtype in base.attributes.items():
   397	                if (attr, rtype) not in seen_attrs:
   398	                    lines.append(f"{emitter.ctype_spaced(rtype)}{emitter.attr(attr)};")
   399	                    seen_attrs.add((attr, rtype))
   400
   401	                    if isinstance(rtype, RTuple):
   402	                        emitter.declare_tuple_struct(rtype)
   403
   404	    lines.append(f"}} {cl.struct_name(emitter.names)};")
   405	    lines.append("")
   406	    emitter.context.declarations[cl.struct_name(emitter.names)] = HeaderDeclaration(
   407	        lines, is_type=True
   408	    )

   411	def generate_vtables(
   412	    base: ClassIR, vtable_setup_name: str, vtable_name: str, emitter: Emitter, shadow: bool
   413	) -> str:
   414	    """Emit the vtables and vtable setup functions for a class.
   415
   416	    This includes both the primary vtable and any trait implementation vtables.
   417	    The trait vtables go before the main vtable, and have the following layout:
   418	        {
   419	            CPyType_T1,         // pointer to type object
   420	            C_T1_trait_vtable,  // pointer to array of method pointers
   421	            C_T1_offset_table,  // pointer to array of attribute offsets
   422	            CPyType_T2,
   423	            C_T2_trait_vtable,
   424	            C_T2_offset_table,
   425	            ...
   426	        }
   427	    The method implementations are calculated at the end of IR pass, attribute
   428	    offsets are {offsetof(native__C, _x1), offsetof(native__C, _y1), ...}.
   429
   430	    To account for both dynamic loading and dynamic class creation,
   431	    vtables are populated dynamically at class creation time, so we
   432	    emit empty array definitions to store the vtables and a function to
   433	    populate them.
   434
   435	    If shadow is True, generate "shadow vtables" that point to the
   436	    shadow glue methods (which should dispatch via the Python C-API).
   437
   438	    Returns the expression to use to refer to the vtable, which might be
   439	    different than the name, if there are trait vtables.
   440	    """
   441
   442	    def trait_vtable_name(trait: ClassIR) -> str:
   443	        return "{}_{}_trait_vtable{}".format(
   444	            base.name_prefix(emitter.names),
   445	            trait.name_prefix(emitter.names),
   446	            "_shadow" if shadow else "",
   447	        )
   448
   449	    def trait_offset_table_name(trait: ClassIR) -> str:
   450	        return "{}_{}_offset_table".format(
   451	            base.name_prefix(emitter.names), trait.name_prefix(emitter.names)
   452	        )
   453
   454	    # Emit array definitions with enough space for all the entries
   455	    emitter.emit_line(
   456	        "static CPyVTableItem {}[{}];".format(
   457	            vtable_name, max(1, len(base.vtable_entries) + 3 * len(base.trait_vtables))
   458	        )
   459	    )
   460
   461	    for trait, vtable in base.trait_vtables.items():
   462	        # Trait methods entry (vtable index -> method implementation).
   463	        emitter.emit_line(
   464	            f"static CPyVTableItem {trait_vtable_name(trait)}[{max(1, len(vtable))}];"
   465	        )
   466	        # Trait attributes entry (attribute number in trait -> offset in actual struct).
   467	        emitter.emit_line(
   468	            "static size_t {}[{}];".format(
   469	                trait_offset_table_name(trait), max(1, len(trait.attributes))
   470	            )
   471	        )
   472
   473	    # Emit vtable setup function
   474	    emitter.emit_line("static bool")
   475	    emitter.emit_line(f"{NATIVE_PREFIX}{vtable_setup_name}(void)")
   476	    emitter.emit_line("{")
   477
   478	    if base.allow_interpreted_subclasses and not shadow:
   479	        emitter.emit_line(f"{NATIVE_PREFIX}{vtable_setup_name}_shadow();")
   480
   481	    subtables = []
   482	    for trait, vtable in base.trait_vtables.items():
   483	        name = trait_vtable_name(trait)
   484	        offset_name = trait_offset_table_name(trait)
   485	        generate_vtable(vtable, name, emitter, [], shadow)
   486	        generate_offset_table(offset_name, emitter, trait, base)
   487	        subtables.append((trait, name, offset_name))
   488
   489	    generate_vtable(base.vtable_entries, vtable_name, emitter, subtables, shadow)
   490
   491	    emitter.emit_line("return 1;")
   492	    emitter.emit_line("}")
   493
   494	    return vtable_name if not subtables else f"{vtable_name} + {len(subtables) * 3}"
   495

   497	def generate_offset_table(
   498	    trait_offset_table_name: str, emitter: Emitter, trait: ClassIR, cl: ClassIR
   499	) -> None:
   500	    """Generate attribute offset row of a trait vtable."""
   501	    emitter.emit_line(f"size_t {trait_offset_table_name}_scratch[] = {{")
   502	    for attr in trait.attributes:
   503	        emitter.emit_line(f"offsetof({cl.struct_name(emitter.names)}, {emitter.attr(attr)}),")
   504	    if not trait.attributes:
   505	        # This is for msvc.
   506	        emitter.emit_line("0")
   507	    emitter.emit_line("};")
   508	    emitter.emit_line(
   509	        "memcpy({name}, {name}_scratch, sizeof({name}));".format(name=trait_offset_table_name)
   510	    )

   513	def generate_vtable(
   514	    entries: VTableEntries,
   515	    vtable_name: str,
   516	    emitter: Emitter,
   517	    subtables: list[tuple[ClassIR, str, str]],
   518	    shadow: bool,
   519	) -> None:
   520	    emitter.emit_line(f"CPyVTableItem {vtable_name}_scratch[] = {{")
   521	    if subtables:
   522	        emitter.emit_line("/* Array of trait vtables */")
   523	        for trait, table, offset_table in subtables:
   524	            emitter.emit_line(
   525	                "(CPyVTableItem){}, (CPyVTableItem){}, (CPyVTableItem){},".format(
   526	                    emitter.type_struct_name(trait), table, offset_table
   527	                )
   528	            )
   529	        emitter.emit_line("/* Start of real vtable */")
   530
   531	    for entry in entries:
   532	        method = entry.shadow_method if shadow and entry.shadow_method else entry.method
   533	        emitter.emit_line(
   534	            "(CPyVTableItem){}{}{},".format(
   535	                emitter.get_group_prefix(entry.method.decl),
   536	                NATIVE_PREFIX,
   537	                method.cname(emitter.names),
   538	            )
   539	        )
   540
   541	    # msvc doesn't allow empty arrays; maybe allowing them at all is an extension?
   542	    if not entries:
   543	        emitter.emit_line("NULL")
   544	    emitter.emit_line("};")
   545	    emitter.emit_line("memcpy({name}, {name}_scratch, sizeof({name}));".format(name=vtable_name))

   548	def generate_setup_for_class(
   549	    cl: ClassIR,
   550	    func_name: str,
   551	    defaults_fn: FuncIR | None,
   552	    vtable_name: str,
   553	    shadow_vtable_name: str | None,
   554	    emitter: Emitter,
   555	) -> None:
   556	    """Generate a native function that allocates an instance of a class."""
   557	    emitter.emit_line("static PyObject *")
   558	    emitter.emit_line(f"{func_name}(PyTypeObject *type)")
   559	    emitter.emit_line("{")
   560	    emitter.emit_line(f"{cl.struct_name(emitter.names)} *self;")
   561	    emitter.emit_line(f"self = ({cl.struct_name(emitter.names)} *)type->tp_alloc(type, 0);")
   562	    emitter.emit_line("if (self == NULL)")
   563	    emitter.emit_line("    return NULL;")
   564
   565	    if shadow_vtable_name:
   566	        emitter.emit_line(f"if (type != {emitter.type_struct_name(cl)}) {{")
   567	        emitter.emit_line(f"self->vtable = {shadow_vtable_name};")
   568	        emitter.emit_line("} else {")
   569	        emitter.emit_line(f"self->vtable = {vtable_name};")
   570	        emitter.emit_line("}")
   571	    else:
   572	        emitter.emit_line(f"self->vtable = {vtable_name};")
   573
   574	    for i in range(0, len(cl.bitmap_attrs), BITMAP_BITS):
   575	        field = emitter.bitmap_field(i)
   576	        emitter.emit_line(f"self->{field} = 0;")
   577
   578	    if cl.has_method("__call__"):
   579	        name = cl.method_decl("__call__").cname(emitter.names)
   580	        emitter.emit_line(f"self->vectorcall = {PREFIX}{name};")
   581
   582	    for base in reversed(cl.base_mro):
   583	        for attr, rtype in base.attributes.items():
   584	            value = emitter.c_undefined_value(rtype)
   585
   586	            # We don't need to set this field to NULL since tp_alloc() already
   587	            # zero-initializes `self`.
   588	            if value != "NULL":
   589	                emitter.emit_line(rf"self->{emitter.attr(attr)} = {value};")
   590
   591	    # Initialize attributes to default values, if necessary
   592	    if defaults_fn is not None:
   593	        emitter.emit_lines(
   594	            "if ({}{}((PyObject *)self) == 0) {{".format(
   595	                NATIVE_PREFIX, defaults_fn.cname(emitter.names)
   596	            ),
   597	            "Py_DECREF(self);",
   598	            "return NULL;",
   599	            "}",
   600	        )
   601
   602	    emitter.emit_line("return (PyObject *)self;")
   603	    emitter.emit_line("}")

   606	def generate_constructor_for_class(
   607	    cl: ClassIR,
   608	    fn: FuncDecl,
   609	    init_fn: FuncIR | None,
   610	    setup_name: str,
   611	    vtable_name: str,
   612	    emitter: Emitter,
   613	) -> None:
   614	    """Generate a native function that allocates and initializes an instance of a class."""
   615	    emitter.emit_line(f"{native_function_header(fn, emitter)}")
   616	    emitter.emit_line("{")
   617	    emitter.emit_line(f"PyObject *self = {setup_name}({emitter.type_struct_name(cl)});")
   618	    emitter.emit_line("if (self == NULL)")
   619	    emitter.emit_line("    return NULL;")
   620	    args = ", ".join(["self"] + [REG_PREFIX + arg.name for arg in fn.sig.args])
   621	    if init_fn is not None:
   622	        emitter.emit_line(
   623	            "char res = {}{}{}({});".format(
   624	                emitter.get_group_prefix(init_fn.decl),
   625	                NATIVE_PREFIX,
   626	                init_fn.cname(emitter.names),
   627	                args,
   628	            )
   629	        )
   630	        emitter.emit_line("if (res == 2) {")
   631	        emitter.emit_line("Py_DECREF(self);")
   632	        emitter.emit_line("return NULL;")
   633	        emitter.emit_line("}")
   634
   635	    # If there is a nontrivial ctor that we didn't define, invoke it via tp_init
   636	    elif len(fn.sig.args) > 1:
   637	        emitter.emit_line(f"int res = {emitter.type_struct_name(cl)}->tp_init({args});")
   638
   639	        emitter.emit_line("if (res < 0) {")
   640	        emitter.emit_line("Py_DECREF(self);")
   641	        emitter.emit_line("return NULL;")
   642	        emitter.emit_line("}")
   643
   644	    emitter.emit_line("return self;")
   645	    emitter.emit_line("}")

   648	def generate_init_for_class(cl: ClassIR, init_fn: FuncIR, emitter: Emitter) -> str:
   649	    """Generate an init function suitable for use as tp_init.
   650
   651	    tp_init needs to be a function that returns an int, and our
   652	    __init__ methods return a PyObject. Translate NULL to -1,
   653	    everything else to 0.
   654	    """
   655	    func_name = f"{cl.name_prefix(emitter.names)}_init"
   656
   657	    emitter.emit_line("static int")
   658	    emitter.emit_line(f"{func_name}(PyObject *self, PyObject *args, PyObject *kwds)")
   659	    emitter.emit_line("{")
   660	    if cl.allow_interpreted_subclasses or cl.builtin_base:
   661	        emitter.emit_line(
   662	            "return {}{}(self, args, kwds) != NULL ? 0 : -1;".format(
   663	                PREFIX, init_fn.cname(emitter.names)
   664	            )
   665	        )
   666	    else:
   667	        emitter.emit_line("return 0;")
   668	    emitter.emit_line("}")
   669
   670	    return func_name

   673	def generate_new_for_class(
   674	    cl: ClassIR,
   675	    func_name: str,
   676	    vtable_name: str,
   677	    setup_name: str,
   678	    init_fn: FuncIR | None,
   679	    emitter: Emitter,
   680	) -> None:
   681	    emitter.emit_line("static PyObject *")
   682	    emitter.emit_line(f"{func_name}(PyTypeObject *type, PyObject *args, PyObject *kwds)")
   683	    emitter.emit_line("{")
   684	    # TODO: Check and unbox arguments
   685	    if not cl.allow_interpreted_subclasses:
   686	        emitter.emit_line(f"if (type != {emitter.type_struct_name(cl)}) {{")
   687	        emitter.emit_line(
   688	            'PyErr_SetString(PyExc_TypeError, "interpreted classes cannot inherit from compiled");'
   689	        )
   690	        emitter.emit_line("return NULL;")
   691	        emitter.emit_line("}")
   692
   693	    if not init_fn or cl.allow_interpreted_subclasses or cl.builtin_base or cl.is_serializable():
   694	        # Match Python semantics -- __new__ doesn't call __init__.
   695	        emitter.emit_line(f"return {setup_name}(type);")
   696	    else:
   697	        # __new__ of a native class implicitly calls __init__ so that we
   698	        # can enforce that instances are always properly initialized. This
   699	        # is needed to support always defined attributes.
   700	        emitter.emit_line(f"PyObject *self = {setup_name}(type);")
   701	        emitter.emit_lines("if (self == NULL)", "    return NULL;")

   726	def generate_traverse_for_class(cl: ClassIR, func_name: str, emitter: Emitter) -> None:
   727	    """Emit function that performs cycle GC traversal of an instance."""
   728	    emitter.emit_line("static int")
   729	    emitter.emit_line(
   730	        f"{func_name}({cl.struct_name(emitter.names)} *self, visitproc visit, void *arg)"
   731	    )
   732	    emitter.emit_line("{")
   733	    for base in reversed(cl.base_mro):
   734	        for attr, rtype in base.attributes.items():
   735	            emitter.emit_gc_visit(f"self->{emitter.attr(attr)}", rtype)
   736	    if has_managed_dict(cl, emitter):
   737	        emitter.emit_line("PyObject_VisitManagedDict((PyObject *)self, visit, arg);")
   738	    elif cl.has_dict:
   739	        struct_name = cl.struct_name(emitter.names)
   740	        # __dict__ lives right after the struct and __weakref__ lives right after that
   741	        emitter.emit_gc_visit(
   742	            f"*((PyObject **)((char *)self + sizeof({struct_name})))", object_rprimitive
   743	        )
   744	        emitter.emit_gc_visit(
   745	            f"*((PyObject **)((char *)self + sizeof(PyObject *) + sizeof({struct_name})))",
   746	            object_rprimitive,
   747	        )
   748	    emitter.emit_line("return 0;")
   749	    emitter.emit_line("}")

   777	    dealloc_func_name: str,
   778	    clear_func_name: str,
   779	    has_tp_finalize: bool,

   798	def generate_finalize_for_class(
   799	    del_method: FuncIR, finalize_func_name: str, emitter: Emitter
   800	) -> None:
   801	    emitter.emit_line("static void")
   802	    emitter.emit_line(f"{finalize_func_name}(PyObject *self)")
   803	    emitter.emit_line("{")
   804	    emitter.emit_line("PyObject *type, *value, *traceback;")
   805	    emitter.emit_line("PyErr_Fetch(&type, &value, &traceback);")
   806	    emitter.emit_line(
   807	        "{}{}{}(self);".format(
   808	            emitter.get_group_prefix(del_method.decl),
   809	            NATIVE_PREFIX,
   810	            del_method.cname(emitter.names),
   811	        )
   812	    )
   813	    emitter.emit_line("if (PyErr_Occurred() != NULL) {")
   814	    emitter.emit_line('PyObject *del_str = PyUnicode_FromString("__del__");')
   815	    emitter.emit_line(
   816	        "PyObject *del_method = (del_str == NULL) ? NULL : _PyType_Lookup(Py_TYPE(self), del_str);"
   817	    )
   818	    # CPython interpreter uses PyErr_WriteUnraisable: https://docs.python.org/3/c-api/exceptions.html#c.PyErr_WriteUnraisable
   819	    # However, the message is slightly different due to the way mypyc compiles classes.
   820	    # CPython interpreter prints: Exception ignored in: <function F.__del__ at 0x100aed940>
   821	    # mypyc prints: Exception ignored in: <slot wrapper '__del__' of 'F' objects>
   822	    emitter.emit_line("PyErr_WriteUnraisable(del_method);")
   823	    emitter.emit_line("Py_XDECREF(del_method);")
   824	    emitter.emit_line("Py_XDECREF(del_str);")
   825	    emitter.emit_line("}")
   826	    # PyErr_Restore also clears exception raised in __del__.
   827	    emitter.emit_line("PyErr_Restore(type, value, traceback);")
   828	    emitter.emit_line("}")

   831	def generate_methods_table(cl: ClassIR, name: str, emitter: Emitter) -> None:
   832	    emitter.emit_line(f"static PyMethodDef {name}[] = {{")
   833	    for fn in cl.methods.values():
   834	        if fn.decl.is_prop_setter or fn.decl.is_prop_getter or fn.internal:
   835	            continue
   836	        emitter.emit_line(f'{{"{fn.name}",')
   837	        emitter.emit_line(f" (PyCFunction){PREFIX}{fn.cname(emitter.names)},")
   838	        flags = ["METH_FASTCALL", "METH_KEYWORDS"]
   839	        if fn.decl.kind == FUNC_STATICMETHOD:
   840	            flags.append("METH_STATIC")
   841	        elif fn.decl.kind == FUNC_CLASSMETHOD:
   842	            flags.append("METH_CLASS")
   843
   844	        emitter.emit_line(" {}, NULL}},".format(" | ".join(flags)))
   845
   846	    # Provide a default __getstate__ and __setstate__
   847	    if not cl.has_method("__setstate__") and not cl.has_method("__getstate__"):
   848	        emitter.emit_lines(
   849	            '{"__setstate__", (PyCFunction)CPyPickle_SetState, METH_O, NULL},',
   850	            '{"__getstate__", (PyCFunction)CPyPickle_GetState, METH_NOARGS, NULL},',
   851	        )
   852
   853	    emitter.emit_line("{NULL}  /* Sentinel */")
   854	    emitter.emit_line("};")
