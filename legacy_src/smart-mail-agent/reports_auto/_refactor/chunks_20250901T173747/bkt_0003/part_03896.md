   911	class PDFCrossReferenceSubsection(PDFObject):
   912	    def __init__(self, firstentrynumber, idsequence):
   913	        self.firstentrynumber = firstentrynumber
   914	        self.idsequence = idsequence
   915	    def format(self, document):
   916	        """id sequence should represent contiguous object nums else error. free numbers not supported (yet)"""
   917	        firstentrynumber = self.firstentrynumber
   918	        idsequence = self.idsequence
   919	        entries = list(idsequence)
   920	        nentries = len(idsequence)
   921	        # special case: object number 0 is always free
   922	        taken = {}
   923	        if firstentrynumber==0:
   924	            taken[0] = "standard free entry"
   925	            nentries = nentries+1
   926	            entries.insert(0, "0000000000 65535 f ")
   927	        idToNV = document.idToObjectNumberAndVersion
   928	        idToOffset = document.idToOffset
   929	        lastentrynumber = firstentrynumber+nentries-1
   930	        for id in idsequence:
   931	            (num, version) = idToNV[id]
   932	            if num in taken:
   933	                raise ValueError("object number collision %s %s %s" % (num, repr(id), repr(taken[id])))
   934	            if num>lastentrynumber or num<firstentrynumber:
   935	                raise ValueError("object number %s not in range %s..%s" % (num, firstentrynumber, lastentrynumber))
   936	            # compute position in list
   937	            rnum = num-firstentrynumber
   938	            taken[num] = id
   939	            offset = idToOffset[id]
   940	            entries[num] = '%0.10d %0.5d n ' % (offset, version)
   941	        # now add the initial line
   942	        firstline = "%s %s" % (firstentrynumber, nentries)
   943	        entries.insert(0, firstline)
   944	        # make sure it ends with \n
   945	        entries.append("")
   946	        return pdfdocEnc('\n'.join(entries))

  1210	        """
  1211	        A PDFPageLabel changes the style of page numbering as displayed in a PDF
  1212	        viewer. PDF page labels have nothing to do with 'physical' page numbers
  1213	        printed on a canvas, but instead influence the 'logical' page numbers
  1214	        displayed by PDF viewers. However, when using roman numerals (i, ii,
  1215	        iii...) or page prefixes for appendecies (A.1, A.2...) on the physical
  1216	        pages PDF page labels are necessary to change the logical page numbers
  1217	        displayed by the PDF viewer to match up with the physical numbers. A
  1218	        PDFPageLabel changes the properties of numbering at the page on which it
  1219	        appears (see the class 'PDFPageLabels' for specifying where a PDFPageLabel
  1220	        is associated) and all subsequent pages, until a new PDFPageLabel is
  1221	        encountered.
  1222
  1223	        The arguments to this initialiser determine the properties of all
  1224	        subsequent page labels. 'style' determines the numberings style, arabic,
  1225	        roman, letters; 'start' specifies the starting number; and 'prefix' any
  1226	        prefix to be applied to the page numbers. All these arguments can be left
  1227	        out or set to None.
  1228
  1229	        * style:
  1230
  1231	            - None:                       No numbering, can be used to display the prefix only.
  1232	            - PDFPageLabel.ARABIC:        Use arabic numbers: 1, 2, 3, 4...
  1233	            - PDFPageLabel.ROMAN_UPPER:   Use upper case roman numerals: I, II, III...
  1234	            - PDFPageLabel.ROMAN_LOWER:   Use lower case roman numerals: i, ii, iii...
  1235	            - PDFPageLabel.LETTERS_UPPER: Use upper case letters: A, B, C, D...
  1236	            - PDFPageLabel.LETTERS_LOWER: Use lower case letters: a, b, c, d...

  1241	                can be used when numbering style changes to reset the page number back
  1242	                to one, ie from roman to arabic, or from arabic to appendecies. Can be
  1243	                any positive integer or None. I'm not sure what the effect of
  1244	                specifying None is, probably that page numbering continues with the
  1245	                current sequence, I'd have to check the spec to clarify though.
  1246
  1247	        * prefix:
  1248
  1249	            -   A string which is prefixed to the page numbers. Can be used to display

    44	def parseAFMFile(afmFileName):
    45	    """Quick and dirty - gives back a top-level dictionary
    46	    with top-level items, and a 'widths' key containing
    47	    a dictionary of glyph names and widths.  Just enough
    48	    needed for embedding.  A better parser would accept
    49	    options for what data you wwanted, and preserve the
    50	    order."""
    51
    52	    lines = open_and_readlines(afmFileName, 'r')
    53	    if len(lines)<=1:
    54	        #likely to be a MAC file
    55	        if lines: lines = lines[0].split('\r')
    56	        if len(lines)<=1:
    57	            raise ValueError('AFM file %s hasn\'t enough data' % afmFileName)
    58	    topLevel = {}
    59	    glyphLevel = []
    60
    61	    lines = [l.strip() for l in lines]
    62	    lines = [l for l in lines if not l.lower().startswith('comment')]
    63	    #pass 1 - get the widths
    64	    inMetrics = 0  # os 'TOP', or 'CHARMETRICS'
    65	    for line in lines:
    66	        if line[0:16] == 'StartCharMetrics':
    67	            inMetrics = 1
    68	        elif line[0:14] == 'EndCharMetrics':
    69	            inMetrics = 0
    70	        elif inMetrics:
    71	            chunks = line.split(';')
    72	            chunks = [chunk.strip() for chunk in chunks]
    73	            cidChunk, widthChunk, nameChunk = chunks[0:3]
    74
    75	            # character ID
    76	            l, r = cidChunk.split()
    77	            assert l == 'C', 'bad line in font file %s' % line
    78	            cid = int(r)
    79
    80	            # width
    81	            l, r = widthChunk.split()
    82	            assert l == 'WX', 'bad line in font file %s' % line
    83	            try:
    84	                width = int(r)
    85	            except ValueError:
    86	                width = float(r)
    87
    88	            # name
    89	            l, r = nameChunk.split()
    90	            assert l == 'N', 'bad line in font file %s' % line
    91	            name = r
    92
    93	            glyphLevel.append((cid, width, name))
    94
    95	    # pass 2 font info
    96	    inHeader = 0
    97	    for line in lines:
    98	        if line[0:16] == 'StartFontMetrics':
    99	            inHeader = 1
   100	        if line[0:16] == 'StartCharMetrics':

   108	            try:
   109	                right = int(right)
   110	            except:
   111	                pass
   112	            topLevel[left] = right
   113
   114
   115	    return (topLevel, glyphLevel)
