    19	def setup_callable_class(builder: IRBuilder) -> None:
    20	    """Generate an (incomplete) callable class representing a function.
    21
    22	    This can be a nested function or a function within a non-extension
    23	    class.  Also set up the 'self' variable for that class.
    24
    25	    This takes the most recently visited function and returns a
    26	    ClassIR to represent that function. Each callable class contains
    27	    an environment attribute which points to another ClassIR
    28	    representing the environment class where some of its variables can
    29	    be accessed.
    30
    31	    Note that some methods, such as '__call__', are not yet
    32	    created here. Use additional functions, such as
    33	    add_call_to_callable_class(), to add them.
    34
    35	    Return a newly constructed ClassIR representing the callable
    36	    class for the nested function.
    37	    """
    38	    # Check to see that the name has not already been taken. If so,
    39	    # rename the class. We allow multiple uses of the same function
    40	    # name because this is valid in if-else blocks. Example:
    41	    #
    42	    #     if True:
    43	    #         def foo():          ---->    foo_obj()
    44	    #             return True
    45	    #     else:
    46	    #         def foo():          ---->    foo_obj_0()
    47	    #             return False
    48	    name = base_name = f"{builder.fn_info.namespaced_name()}_obj"
    49	    count = 0
    50	    while name in builder.callable_class_names:
    51	        name = base_name + "_" + str(count)
    52	        count += 1
    53	    builder.callable_class_names.add(name)
    54
    55	    # Define the actual callable class ClassIR, and set its
    56	    # environment to point at the previously defined environment
    57	    # class.
    58	    callable_class_ir = ClassIR(name, builder.module_name, is_generated=True, is_final_class=True)
    59
    60	    # The functools @wraps decorator attempts to call setattr on
    61	    # nested functions, so we create a dict for these nested
    62	    # functions.
    63	    # https://github.com/python/cpython/blob/3.7/Lib/functools.py#L58
    64	    if builder.fn_info.is_nested:
    65	        callable_class_ir.has_dict = True
    66
    67	    # If the enclosing class doesn't contain nested (which will happen if
    68	    # this is a toplevel lambda), don't set up an environment.
    69	    if builder.fn_infos[-2].contains_nested:
    70	        callable_class_ir.attributes[ENV_ATTR_NAME] = RInstance(builder.fn_infos[-2].env_class)
    71	    callable_class_ir.mro = [callable_class_ir]
    72	    builder.fn_info.callable_class = ImplicitClass(callable_class_ir)
    73	    builder.classes.append(callable_class_ir)
    74
    75	    # Add a 'self' variable to the environment of the callable class,
    76	    # and store that variable in a register to be accessed later.
    77	    self_target = builder.add_self_to_env(callable_class_ir)
    78	    builder.fn_info.callable_class.self_reg = builder.read(self_target, builder.fn_info.fitem.line)

    82	    builder: IRBuilder,
    83	    args: list[Register],
    84	    blocks: list[BasicBlock],
    85	    sig: FuncSignature,
    86	    fn_info: FuncInfo,

    88	    """Generate a '__call__' method for a callable class representing a nested function.
    89
    90	    This takes the blocks and signature associated with a function
    91	    definition and uses those to build the '__call__' method of a
    92	    given callable class, used to represent that function.
    93	    """
    94	    # Since we create a method, we also add a 'self' parameter.
    95	    nargs = len(sig.args) - sig.num_bitmap_args
    96	    sig = FuncSignature(
    97	        (RuntimeArg(SELF_NAME, object_rprimitive),) + sig.args[:nargs], sig.ret_type

   102	    )

   104	    fn_info.callable_class.ir.method_decls["__call__"] = call_fn_decl
   105	    return call_fn_ir

   109	    """Generate the '__get__' method for a callable class."""
   110	    line = fn_info.fitem.line
   111	    with builder.enter_method(
   112	        fn_info.callable_class.ir,
   113	        "__get__",
   114	        object_rprimitive,
   115	        fn_info,
   116	        self_type=object_rprimitive,
   117	    ):

   119	        builder.add_argument("owner", object_rprimitive)

   122	        # object. If accessed through an object, create a new bound
   123	        # instance method object.
   124	        instance_block, class_block = BasicBlock(), BasicBlock()
   125	        comparison = builder.translate_is_op(
   126	            builder.read(instance), builder.none_object(), "is", line

   128	        builder.add_bool_branch(comparison, class_block, instance_block)

   130	        builder.activate_block(class_block)
   131	        builder.add(Return(builder.self()))
   132
   133	        builder.activate_block(instance_block)
   134	        builder.add(
   135	            Return(builder.call_c(method_new_op, [builder.self(), builder.read(instance)], line))

   139	def instantiate_callable_class(builder: IRBuilder, fn_info: FuncInfo) -> Value:
   140	    """Create an instance of a callable class for a function.
   141
   142	    Calls to the function will actually call this instance.
   143
   144	    Note that fn_info refers to the function being assigned, whereas
   145	    builder.fn_info refers to the function encapsulating the function
   146	    being turned into a callable class.
   147	    """
   148	    fitem = fn_info.fitem
   149	    func_reg = builder.add(Call(fn_info.callable_class.ir.ctor, [], fitem.line))
   150
   151	    # Set the environment attribute of the callable class to point at
   152	    # the environment class defined in the callable class' immediate
   153	    # outer scope. Note that there are three possible environment
   154	    # class registers we may use. This depends on what the encapsulating
   155	    # (parent) function is:
   156	    #
   157	    # - A nested function: the callable class is instantiated
   158	    #   from the current callable class' '__call__' function, and hence
   159	    #   the callable class' environment register is used.
   160	    # - A generator function: the callable class is instantiated
   161	    #   from the '__next__' method of the generator class, and hence the
   162	    #   environment of the generator class is used.
   163	    # - Regular function: we use the environment of the original function.
   164	    curr_env_reg = None
   165	    if builder.fn_info.is_generator:
   166	        curr_env_reg = builder.fn_info.generator_class.curr_env_reg
   167	    elif builder.fn_info.is_nested:
   168	        curr_env_reg = builder.fn_info.callable_class.curr_env_reg
   169	    elif builder.fn_info.contains_nested:
   170	        curr_env_reg = builder.fn_info.curr_env_reg
   171	    if curr_env_reg:
   172	        builder.add(SetAttr(func_reg, ENV_ATTR_NAME, curr_env_reg, fitem.line))
   173	    return func_reg

    87	def transform_class_def(builder: IRBuilder, cdef: ClassDef) -> None:
    88	    """Create IR for a class definition.
    89
    90	    This can generate both extension (native) and non-extension
    91	    classes.  These are generated in very different ways. In the
    92	    latter case we construct a Python type object at runtime by doing
    93	    the equivalent of "type(name, bases, dict)" in IR. Extension
    94	    classes are defined via C structs that are generated later in
    95	    mypyc.codegen.emitclass.
    96
    97	    This is the main entry point to this module.
    98	    """
    99	    if cdef.info not in builder.mapper.type_to_ir:
   100	        builder.error("Nested class definitions not supported", cdef.line)
   101	        return
   102
   103	    ir = builder.mapper.type_to_ir[cdef.info]
   104
   105	    # We do this check here because the base field of parent
   106	    # classes aren't necessarily populated yet at
   107	    # prepare_class_def time.
   108	    if any(ir.base_mro[i].base != ir.base_mro[i + 1] for i in range(len(ir.base_mro) - 1)):
   109	        builder.error("Multiple inheritance is not supported (except for traits)", cdef.line)
   110
   111	    if ir.allow_interpreted_subclasses:
   112	        for parent in ir.mro:
   113	            if not parent.allow_interpreted_subclasses:
   114	                builder.error(
   115	                    'Base class "{}" does not allow interpreted subclasses'.format(
   116	                        parent.fullname
   117	                    ),
   118	                    cdef.line,
   119	                )
   120
   121	    # Currently, we only create non-extension classes for classes that are
   122	    # decorated or inherit from Enum. Classes decorated with @trait do not
   123	    # apply here, and are handled in a different way.
   124	    if ir.is_ext_class:
   125	        cls_type = dataclass_type(cdef)
   126	        if cls_type is None:
   127	            cls_builder: ClassBuilder = ExtClassBuilder(builder, cdef)
   128	        elif cls_type in ["dataclasses", "attr-auto"]:
   129	            cls_builder = DataClassBuilder(builder, cdef)
   130	        elif cls_type == "attr":
   131	            cls_builder = AttrsClassBuilder(builder, cdef)
   132	        else:
   133	            raise ValueError(cls_type)
   134	    else:
   135	        cls_builder = NonExtClassBuilder(builder, cdef)
   136
   137	    for stmt in cdef.defs.body:
   138	        if isinstance(stmt, OverloadedFuncDef) and stmt.is_property:
   139	            if isinstance(cls_builder, NonExtClassBuilder):
   140	                # properties with both getters and setters in non_extension
   141	                # classes not supported
   142	                builder.error("Property setters not supported in non-extension classes", stmt.line)
   143	            for item in stmt.items:
   144	                with builder.catch_errors(stmt.line):
   145	                    cls_builder.add_method(get_func_def(item))
   146	        elif isinstance(stmt, (FuncDef, Decorator, OverloadedFuncDef)):

   148	            # bodies to compile and will need to have the bodies
   149	            # provided by some other mechanism.)
   150	            if cdef.info.names[stmt.name].plugin_generated:
   151	                continue
   152	            with builder.catch_errors(stmt.line):
   153	                cls_builder.add_method(get_func_def(stmt))
   154	        elif isinstance(stmt, PassStmt) or (
   155	            isinstance(stmt, ExpressionStmt) and isinstance(stmt.expr, EllipsisExpr)
   156	        ):
   157	            continue
   158	        elif isinstance(stmt, AssignmentStmt):
   159	            if len(stmt.lvalues) != 1:
   160	                builder.error("Multiple assignment in class bodies not supported", stmt.line)
   161	                continue
   162	            lvalue = stmt.lvalues[0]
   163	            if not isinstance(lvalue, NameExpr):
   164	                builder.error(
   165	                    "Only assignment to variables is supported in class bodies", stmt.line
   166	                )
   167	                continue
   168	            # We want to collect class variables in a dictionary for both real
   169	            # non-extension classes and fake dataclass ones.
   170	            cls_builder.add_attr(lvalue, stmt)
   171
   172	        elif isinstance(stmt, ExpressionStmt) and isinstance(stmt.expr, StrExpr):
   173	            # Docstring. Ignore
   174	            pass
   175	        else:
   176	            builder.error("Unsupported statement in class body", stmt.line)
   177
   178	    # Generate implicit property setters/getters
   179	    for name, decl in ir.method_decls.items():
   180	        if decl.implicit and decl.is_prop_getter:
   181	            getter_ir = gen_property_getter_ir(builder, decl, cdef, ir.is_trait)
   182	            builder.functions.append(getter_ir)
   183	            ir.methods[getter_ir.decl.name] = getter_ir
   184
   185	            setter_ir = None
   186	            setter_name = PROPSET_PREFIX + name
   187	            if setter_name in ir.method_decls:
   188	                setter_ir = gen_property_setter_ir(

   190	                )
   191	                builder.functions.append(setter_ir)
   192	                ir.methods[setter_name] = setter_ir
   193
   194	            ir.properties[name] = (getter_ir, setter_ir)
   195	            # TODO: Generate glue method if needed?
   196	            # TODO: Do we need interpreted glue methods? Maybe not?
   197
   198	    cls_builder.finalize(ir)

   353	            self.builder, self.non_ext, lvalue, stmt, self.cdef, self.attrs_to_cache
   354	        )
   355	        super().add_attr(lvalue, stmt)
   356
   357	    def finalize(self, ir: ClassIR) -> None:
   358	        """Generate code to finish instantiating a dataclass.
   359
   360	        This works by replacing all of the attributes on the class
   361	        (which will be descriptors) with whatever they would be in a
   362	        non-extension class, calling dataclass, then switching them back.
   363
   364	        The resulting class is an extension class and instances of it do not
   365	        have a __dict__ (unless something else requires it).
   366	        All methods written explicitly in the source are compiled and
   367	        may be called through the vtable while the methods generated
   368	        by dataclasses are interpreted and may not be.
   369
   370	        (If we just called dataclass without doing this, it would think that all
   371	        of the descriptors for our attributes are default values and generate an
   372	        incorrect constructor. We need to do the switch so that dataclass gets the
   373	        appropriate defaults.)
   374	        """
   375	        super().finalize(ir)
   376	        assert self.type_obj
   377	        add_dunders_to_non_ext_dict(
   378	            self.builder, self.non_ext, self.cdef.line, self.add_annotations_to_dict
   379	        )
   380	        dec = self.builder.accept(
   381	            next(d for d in self.cdef.decorators if is_dataclass_decorator(d))
   382	        )
   383	        dataclass_type_val = self.builder.load_str(dataclass_type(self.cdef) or "unknown")
   384	        self.builder.call_c(
   385	            dataclass_sleight_of_hand,
   386	            [dec, self.type_obj, self.non_ext.dict, self.non_ext.anns, dataclass_type_val],
   387	            self.cdef.line,
   388	        )

   391	class AttrsClassBuilder(DataClassBuilder):
   392	    """Create IR for an attrs class where auto_attribs=False (the default).
   393
   394	    When auto_attribs is enabled, attrs classes behave similarly to dataclasses
   395	    (i.e. types are stored as annotations on the class) and are thus handled
   396	    by DataClassBuilder, but when auto_attribs is disabled the types are
   397	    provided via attr.ib(type=...)
   398	    """
   399
   400	    add_annotations_to_dict = False
   401
   402	    def skip_attr_default(self, name: str, stmt: AssignmentStmt) -> bool:
   403	        return True
   404
   405	    def get_type_annotation(self, stmt: AssignmentStmt) -> TypeInfo | None:
   406	        if isinstance(stmt.rvalue, CallExpr):
   407	            # find the type arg in `attr.ib(type=str)`
   408	            callee = stmt.rvalue.callee
   409	            if (
   410	                isinstance(callee, MemberExpr)
   411	                and callee.fullname in ["attr.ib", "attr.attr"]
   412	                and "type" in stmt.rvalue.arg_names
   413	            ):
   414	                index = stmt.rvalue.arg_names.index("type")
   415	                type_name = stmt.rvalue.args[index]
   416	                if isinstance(type_name, NameExpr) and isinstance(type_name.node, TypeInfo):
   417	                    lvalue = stmt.lvalues[0]
   418	                    assert isinstance(lvalue, NameExpr)
   419	                    return type_name.node
   420	        return None

   423	def allocate_class(builder: IRBuilder, cdef: ClassDef) -> Value:
   424	    # OK AND NOW THE FUN PART
   425	    base_exprs = cdef.base_type_exprs + cdef.removed_base_type_exprs
   426	    new_style_type_args = cdef.type_args
   427	    if new_style_type_args:
   428	        bases = [make_generic_base_class(builder, cdef.fullname, new_style_type_args, cdef.line)]
   429	    else:
   430	        bases = []
   431
   432	    if base_exprs or new_style_type_args:
   433	        bases.extend([builder.accept(x) for x in base_exprs])
   434	        tp_bases = builder.new_tuple(bases, cdef.line)
   435	    else:
   436	        tp_bases = builder.add(LoadErrorValue(object_rprimitive, is_borrowed=True))
   437	    modname = builder.load_str(builder.module_name)
   438	    template = builder.add(
   439	        LoadStatic(object_rprimitive, cdef.name + "_template", builder.module_name, NAMESPACE_TYPE)
   440	    )
   441	    # Create the class
   442	    tp = builder.call_c(pytype_from_template_op, [template, tp_bases, modname], cdef.line)
   443	    # Immediately fix up the trait vtables, before doing anything with the class.
   444	    ir = builder.mapper.type_to_ir[cdef.info]
   445	    if not ir.is_trait and not ir.builtin_base:
   446	        builder.add(
   447	            Call(
   448	                FuncDecl(
   449	                    cdef.name + "_trait_vtable_setup",
   450	                    None,
   451	                    builder.module_name,
   452	                    FuncSignature([], bool_rprimitive),
   453	                ),
   454	                [],
   455	                -1,
   456	            )
   457	        )
   458	    # Populate a '__mypyc_attrs__' field containing the list of attrs
   459	    builder.primitive_op(
   460	        py_setattr_op,
   461	        [
   462	            tp,
   463	            builder.load_str("__mypyc_attrs__"),
   464	            create_mypyc_attrs_tuple(builder, builder.mapper.type_to_ir[cdef.info], cdef.line),
   465	        ],
   466	        cdef.line,
   467	    )
   468
   469	    # Save the class
   470	    builder.add(InitStatic(tp, cdef.name, builder.module_name, NAMESPACE_TYPE))
   471
   472	    # Add it to the dict
   473	    builder.primitive_op(
   474	        dict_set_item_op, [builder.load_globals_dict(), builder.load_str(cdef.name), tp], cdef.line
   475	    )
   476
   477	    return tp

   480	def make_generic_base_class(
   481	    builder: IRBuilder, fullname: str, type_args: list[TypeParam], line: int
   482	) -> Value:
   483	    """Construct Generic[...] base class object for a new-style generic class (Python 3.12)."""
   484	    mod = builder.call_c(import_op, [builder.load_str("_typing")], line)
   485	    tvs = create_type_params(builder, mod, type_args, line)
   486	    args = []
   487	    for tv, type_param in zip(tvs, type_args):
   488	        if type_param.kind == TYPE_VAR_TUPLE_KIND:
   489	            # Evaluate *Ts for a TypeVarTuple
   490	            it = builder.primitive_op(iter_op, [tv], line)
   491	            tv = builder.call_c(next_op, [it], line)
   492	        args.append(tv)
   493
   494	    gent = builder.py_get_attr(mod, "Generic", line)
   495	    if len(args) == 1:
   496	        arg = args[0]
   497	    else:
   498	        arg = builder.new_tuple(args, line)
   499
   500	    base = builder.primitive_op(py_get_item_op, [gent, arg], line)
   501	    return base

   512	def populate_non_ext_bases(builder: IRBuilder, cdef: ClassDef) -> Value:
   513	    """Create base class tuple of a non-extension class.
   514
   515	    The tuple is passed to the metaclass constructor.
   516	    """
   517	    is_named_tuple = cdef.info.is_named_tuple
   518	    ir = builder.mapper.type_to_ir[cdef.info]
   519	    bases = []
   520	    for cls in (b.type for b in cdef.info.bases):
   521	        if cls.fullname == "builtins.object":
   522	            continue
   523	        if is_named_tuple and cls.fullname in (
   524	            "typing.Sequence",
   525	            "typing.Iterable",
   526	            "typing.Collection",
   527	            "typing.Reversible",
   528	            "typing.Container",
   529	            "typing.Sized",
   530	        ):
   531	            # HAX: Synthesized base classes added by mypy don't exist at runtime, so skip them.
   532	            #      This could break if they were added explicitly, though...
   533	            continue
   534	        # Add the current class to the base classes list of concrete subclasses
   535	        if cls in builder.mapper.type_to_ir:
   536	            base_ir = builder.mapper.type_to_ir[cls]
   537	            if base_ir.children is not None:
   538	                base_ir.children.append(ir)
   539
   540	        if cls.fullname in MAGIC_TYPED_DICT_CLASSES:
   541	            # HAX: Mypy internally represents TypedDict classes differently from what
   542	            #      should happen at runtime. Replace with something that works.
   543	            module = "typing"
   544	            name = "_TypedDict"
   545	            base = builder.get_module_attr(module, name, cdef.line)
   546	        elif is_named_tuple and cls.fullname == "builtins.tuple":
   547	            name = "_NamedTuple"
   548	            base = builder.get_module_attr("typing", name, cdef.line)
   549	        else:
   550	            cls_module = cls.fullname.rsplit(".", 1)[0]
   551	            if cls_module == builder.current_module:
   552	                base = builder.load_global_str(cls.name, cdef.line)
   553	            else:
   554	                base = builder.load_module_attr_by_fullname(cls.fullname, cdef.line)
   555	        bases.append(base)
   556	        if cls.fullname in MAGIC_TYPED_DICT_CLASSES:
   557	            # The remaining base classes are synthesized by mypy and should be ignored.
   558	            break
   559	    return builder.new_tuple(bases, cdef.line)

   562	def find_non_ext_metaclass(builder: IRBuilder, cdef: ClassDef, bases: Value) -> Value:
   563	    """Find the metaclass of a class from its defs and bases."""
   564	    if cdef.metaclass:
   565	        declared_metaclass = builder.accept(cdef.metaclass)
   566	    else:
   567	        if cdef.info.typeddict_type is not None:
   568	            # In Python 3.9, the metaclass for class-based TypedDict is typing._TypedDictMeta.
   569	            # We can't easily calculate it generically, so special case it.
   570	            return builder.get_module_attr("typing", "_TypedDictMeta", cdef.line)
   571	        elif cdef.info.is_named_tuple:
   572	            # In Python 3.9, the metaclass for class-based NamedTuple is typing.NamedTupleMeta.
   573	            # We can't easily calculate it generically, so special case it.
   574	            return builder.get_module_attr("typing", "NamedTupleMeta", cdef.line)
   575
   576	        declared_metaclass = builder.add(
   577	            LoadAddress(type_object_op.type, type_object_op.src, cdef.line)
   578	        )
   579
   580	    return builder.call_c(py_calc_meta_op, [declared_metaclass, bases], cdef.line)

   583	def setup_non_ext_dict(
   584	    builder: IRBuilder, cdef: ClassDef, metaclass: Value, bases: Value
   585	) -> Value:
   586	    """Initialize the class dictionary for a non-extension class.
   587
   588	    This class dictionary is passed to the metaclass constructor.
   589	    """
   590	    # Check if the metaclass defines a __prepare__ method, and if so, call it.
   591	    has_prepare = builder.primitive_op(
   592	        py_hasattr_op, [metaclass, builder.load_str("__prepare__")], cdef.line
   593	    )
   594
   595	    non_ext_dict = Register(dict_rprimitive)
   596
   597	    true_block, false_block, exit_block = BasicBlock(), BasicBlock(), BasicBlock()
   598	    builder.add_bool_branch(has_prepare, true_block, false_block)
   599
   600	    builder.activate_block(true_block)
   601	    cls_name = builder.load_str(cdef.name)
   602	    prepare_meth = builder.py_get_attr(metaclass, "__prepare__", cdef.line)
   603	    prepare_dict = builder.py_call(prepare_meth, [cls_name, bases], cdef.line)
   604	    builder.assign(non_ext_dict, prepare_dict, cdef.line)
   605	    builder.goto(exit_block)
   606
   607	    builder.activate_block(false_block)
   608	    builder.assign(non_ext_dict, builder.call_c(dict_new_op, [], cdef.line), cdef.line)
   609	    builder.goto(exit_block)
   610	    builder.activate_block(exit_block)
   611
   612	    return non_ext_dict

   615	def add_non_ext_class_attr_ann(
   616	    builder: IRBuilder,
   617	    non_ext: NonExtClassInfo,
   618	    lvalue: NameExpr,
   619	    stmt: AssignmentStmt,
   620	    get_type_info: Callable[[AssignmentStmt], TypeInfo | None] | None = None,
   621	) -> None:
   622	    """Add a class attribute to __annotations__ of a non-extension class."""
   623	    # FIXME: try to better preserve the special forms and type parameters of generics.
   624	    typ: Value | None = None
   625	    if get_type_info is not None:
   626	        type_info = get_type_info(stmt)
   627	        if type_info:
   628	            # NOTE: Using string type information is similar to using
   629	            # `from __future__ import annotations` in standard python.
   630	            # NOTE: For string types we need to use the fullname since it
   631	            # includes the module. If string type doesn't have the module,
   632	            # @dataclass will try to get the current module and fail since the
   633	            # current module is not in sys.modules.
   634	            if builder.current_module == type_info.module_name and stmt.line < type_info.line:
   635	                typ = builder.load_str(type_info.fullname)
   636	            else:
   637	                typ = load_type(builder, type_info, stmt.unanalyzed_type, stmt.line)
   638
   639	    if typ is None:
   640	        # FIXME: if get_type_info is not provided, don't fall back to stmt.type?
   641	        ann_type = get_proper_type(stmt.type)
   642	        if (
   643	            isinstance(stmt.unanalyzed_type, UnboundType)
   644	            and stmt.unanalyzed_type.original_str_expr is not None
   645	        ):
   646	            # Annotation is a forward reference, so don't attempt to load the actual
   647	            # type and load the string instead.
   648	            #
   649	            # TODO: is it possible to determine whether a non-string annotation is
   650	            # actually a forward reference due to the __annotations__ future?
   651	            typ = builder.load_str(stmt.unanalyzed_type.original_str_expr)
   652	        elif isinstance(ann_type, Instance):
   653	            typ = load_type(builder, ann_type.type, stmt.unanalyzed_type, stmt.line)
   654	        else:
   655	            typ = builder.add(LoadAddress(type_object_op.type, type_object_op.src, stmt.line))
   656
   657	    key = builder.load_str(lvalue.name)
   658	    builder.primitive_op(dict_set_item_op, [non_ext.anns, key, typ], stmt.line)

   661	def add_non_ext_class_attr(
   662	    builder: IRBuilder,
   663	    non_ext: NonExtClassInfo,
   664	    lvalue: NameExpr,
   665	    stmt: AssignmentStmt,
   666	    cdef: ClassDef,
   667	    attr_to_cache: list[tuple[Lvalue, RType]],
   668	) -> None:
   669	    """Add a class attribute to __dict__ of a non-extension class."""
   670	    # Only add the attribute to the __dict__ if the assignment is of the form:
   671	    # x: type = value (don't add attributes of the form 'x: type' to the __dict__).
   672	    if not isinstance(stmt.rvalue, TempNode):
   673	        rvalue = builder.accept(stmt.rvalue)
   674	        builder.add_to_non_ext_dict(non_ext, lvalue.name, rvalue, stmt.line)
   675	        # We cache enum attributes to speed up enum attribute lookup since they
   676	        # are final.
   677	        if (
   678	            cdef.info.bases
   679	            # Enum class must be the last parent class.
   680	            and cdef.info.bases[-1].type.is_enum
   681	            # Skip these since Enum will remove it
   682	            and lvalue.name not in EXCLUDED_ENUM_ATTRIBUTES
   683	        ):
   684	            # Enum values are always boxed, so use object_rprimitive.
   685	            attr_to_cache.append((lvalue, object_rprimitive))

   688	def find_attr_initializers(
   689	    builder: IRBuilder, cdef: ClassDef, skip: Callable[[str, AssignmentStmt], bool] | None = None
   690	) -> tuple[set[str], list[AssignmentStmt]]:
   691	    """Find initializers of attributes in a class body.
   692
   693	    If provided, the skip arg should be a callable which will return whether
   694	    to skip generating a default for an attribute.  It will be passed the name of
   695	    the attribute and the corresponding AssignmentStmt.
   696	    """
   697	    cls = builder.mapper.type_to_ir[cdef.info]
   698	    if cls.builtin_base:
   699	        return set(), []
   700
   701	    attrs_with_defaults = set()
   702
   703	    # Pull out all assignments in classes in the mro so we can initialize them
   704	    # TODO: Support nested statements
   705	    default_assignments = []
   706	    for info in reversed(cdef.info.mro):
   707	        if info not in builder.mapper.type_to_ir:
   708	            continue
   709	        for stmt in info.defn.defs.body:
   710	            if (
   711	                isinstance(stmt, AssignmentStmt)
   712	                and isinstance(stmt.lvalues[0], NameExpr)
   713	                and not is_class_var(stmt.lvalues[0])
   714	                and not isinstance(stmt.rvalue, TempNode)
   715	            ):
   716	                name = stmt.lvalues[0].name
   717	                if name == "__slots__":
   718	                    continue
   719
   720	                if name == "__deletable__":
   721	                    check_deletable_declaration(builder, cls, stmt.line)
   722	                    continue
   723
   724	                if skip is not None and skip(name, stmt):
   725	                    continue
   726
   727	                attr_type = cls.attr_type(name)
   728
   729	                # If the attribute is initialized to None and type isn't optional,
   730	                # doesn't initialize it to anything (special case for "# type:" comments).
   731	                if isinstance(stmt.rvalue, RefExpr) and stmt.rvalue.fullname == "builtins.None":
   732	                    if (
   733	                        not is_optional_type(attr_type)
   734	                        and not is_object_rprimitive(attr_type)
   735	                        and not is_none_rprimitive(attr_type)
   736	                    ):
   737	                        continue
   738
   739	                attrs_with_defaults.add(name)
   740	                default_assignments.append(stmt)
   741
   742	    return attrs_with_defaults, default_assignments

   745	def generate_attr_defaults_init(
   746	    builder: IRBuilder, cdef: ClassDef, default_assignments: list[AssignmentStmt]
   747	) -> None:
   748	    """Generate an initialization method for default attr values (from class vars)."""
   749	    if not default_assignments:
   750	        return
   751	    cls = builder.mapper.type_to_ir[cdef.info]
   752	    if cls.builtin_base:
   753	        return
   754
   755	    with builder.enter_method(cls, "__mypyc_defaults_setup", bool_rprimitive):
   756	        self_var = builder.self()
   757	        for stmt in default_assignments:
   758	            lvalue = stmt.lvalues[0]
   759	            assert isinstance(lvalue, NameExpr)
   760	            if not stmt.is_final_def and not is_constant(stmt.rvalue):
   761	                builder.warning("Unsupported default attribute value", stmt.rvalue.line)
   762
   763	            attr_type = cls.attr_type(lvalue.name)
   764	            val = builder.coerce(builder.accept(stmt.rvalue), attr_type, stmt.line)
   765	            init = SetAttr(self_var, lvalue.name, val, -1)
   766	            init.mark_as_initializer()
   767	            builder.add(init)
   768
   769	        builder.add(Return(builder.true()))

   772	def check_deletable_declaration(builder: IRBuilder, cl: ClassIR, line: int) -> None:
   773	    for attr in cl.deletable:
   774	        if attr not in cl.attributes:
   775	            if not cl.has_attr(attr):
   776	                builder.error(f'Attribute "{attr}" not defined', line)
   777	                continue
   778	            for base in cl.mro:
   779	                if attr in base.property_types:
   780	                    builder.error(f'Cannot make property "{attr}" deletable', line)
   781	                    break
   782	            else:
   783	                _, base = cl.attr_details(attr)
   784	                builder.error(
   785	                    ('Attribute "{}" not defined in "{}" ' + '(defined in "{}")').format(
   786	                        attr, cl.name, base.name
   787	                    ),
   788	                    line,
   789	                )

   799	def gen_glue_ne_method(builder: IRBuilder, cls: ClassIR, line: int) -> None:
   800	    """Generate a "__ne__" method from a "__eq__" method."""
   801	    func_ir = cls.get_method("__eq__")
   802	    assert func_ir
   803	    eq_sig = func_ir.decl.sig
   804	    strict_typing = builder.options.strict_dunders_typing
   805	    with builder.enter_method(cls, "__ne__", eq_sig.ret_type):
   806	        rhs_type = eq_sig.args[0].type if strict_typing else object_rprimitive
   807	        rhs_arg = builder.add_argument("rhs", rhs_type)
   808	        eqval = builder.add(MethodCall(builder.self(), "__eq__", [rhs_arg], line))
   809
   810	        can_return_not_implemented = is_subtype(not_implemented_op.type, eq_sig.ret_type)
   811	        return_bool = is_subtype(eq_sig.ret_type, bool_rprimitive)
   812
   813	        if not strict_typing or can_return_not_implemented:
   814	            # If __eq__ returns NotImplemented, then __ne__ should also
   815	            not_implemented_block, regular_block = BasicBlock(), BasicBlock()
   816	            not_implemented = builder.add(
   817	                LoadAddress(not_implemented_op.type, not_implemented_op.src, line)
   818	            )
   819	            builder.add(
   820	                Branch(
   821	                    builder.translate_is_op(eqval, not_implemented, "is", line),
   822	                    not_implemented_block,
   823	                    regular_block,
   824	                    Branch.BOOL,
   825	                )
   826	            )
   827	            builder.activate_block(regular_block)
   828	            rettype = bool_rprimitive if return_bool and strict_typing else object_rprimitive
   829	            retval = builder.coerce(builder.unary_op(eqval, "not", line), rettype, line)
   830	            builder.add(Return(retval))
   831	            builder.activate_block(not_implemented_block)
   832	            builder.add(Return(not_implemented))
   833	        else:
   834	            rettype = bool_rprimitive if return_bool and strict_typing else object_rprimitive
   835	            retval = builder.coerce(builder.unary_op(eqval, "not", line), rettype, line)
   836	            builder.add(Return(retval))

     2
     3	For example, 3 + 5 can be constant folded into 8.
     4
     5	This is mostly like mypy.constant_fold, but we can bind some additional
     6	NameExpr and MemberExpr references here, since we have more knowledge

    12

    14

    17	    BytesExpr,

    19	    Expression,

    32	# All possible result types of constant folding
    33	ConstantValue = Union[int, float, complex, str, bytes]
    34	CONST_TYPES: Final = (int, float, complex, str, bytes)
    35
    36
    37	def constant_fold_expr(builder: IRBuilder, expr: Expression) -> ConstantValue | None:
    38	    """Return the constant value of an expression for supported operations.
    39
    40	    Return None otherwise.
    41	    """
    42	    if isinstance(expr, IntExpr):
    43	        return expr.value
    44	    if isinstance(expr, FloatExpr):
    45	        return expr.value
    46	    if isinstance(expr, StrExpr):
    47	        return expr.value
    48	    if isinstance(expr, BytesExpr):
    49	        return bytes_from_str(expr.value)
    50	    if isinstance(expr, ComplexExpr):
    51	        return expr.value
    52	    elif isinstance(expr, NameExpr):
    53	        node = expr.node
    54	        if isinstance(node, Var) and node.is_final:
    55	            final_value = node.final_value
    56	            if isinstance(final_value, (CONST_TYPES)):
    57	                return final_value
    58	    elif isinstance(expr, MemberExpr):
    59	        final = builder.get_final_ref(expr)
    60	        if final is not None:
    61	            fn, final_var, native = final
    62	            if final_var.is_final:
    63	                final_value = final_var.final_value
    64	                if isinstance(final_value, (CONST_TYPES)):
    65	                    return final_value
    66	    elif isinstance(expr, OpExpr):
    67	        left = constant_fold_expr(builder, expr.left)
    68	        right = constant_fold_expr(builder, expr.right)
    69	        if left is not None and right is not None:
    70	            return constant_fold_binary_op_extended(expr.op, left, right)
    71	    elif isinstance(expr, UnaryExpr):
    72	        value = constant_fold_expr(builder, expr.expr)
    73	        if value is not None and not isinstance(value, bytes):
    74	            return constant_fold_unary_op(expr.op, value)
    75	    return None
    76
    77
    78	def constant_fold_binary_op_extended(
    79	    op: str, left: ConstantValue, right: ConstantValue
    80	) -> ConstantValue | None:
    81	    """Like mypy's constant_fold_binary_op(), but includes bytes support.
    82
    83	    mypy cannot use constant folded bytes easily so it's simpler to only support them in mypyc.
    84	    """
    85	    if not isinstance(left, bytes) and not isinstance(right, bytes):
    86	        return constant_fold_binary_op(op, left, right)
    87
    88	    if op == "+" and isinstance(left, bytes) and isinstance(right, bytes):
    89	        return left + right
    90	    elif op == "*" and isinstance(left, bytes) and isinstance(right, int):
    91	        return left * right
    92	    elif op == "*" and isinstance(left, int) and isinstance(right, bytes):
    93	        return left * right
    94
    95	    return None
