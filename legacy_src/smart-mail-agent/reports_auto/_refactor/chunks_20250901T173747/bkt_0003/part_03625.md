   129	    def filter(self, generator: Iterator[PageElement]) -> Iterator[_OneElement]:
   130	        """The most generic search method offered by Beautiful Soup.
   131
   132	        Acts like Python's built-in `filter`, using
   133	        `ElementFilter.match` as the filtering function.
   134	        """
   135	        # If there are no rules at all, don't bother filtering. Let
   136	        # anything through.
   137	        if self.includes_everything:
   138	            for i in generator:
   139	                yield i
   140	        while True:
   141	            try:
   142	                i = next(generator)
   143	            except StopIteration:
   144	                break

   147	                    yield cast("_OneElement", i)

   211	class MatchRule(object):
   212	    """Each MatchRule encapsulates the logic behind a single argument
   213	    passed in to one of the Beautiful Soup find* methods.
   214	    """
   215
   216	    string: Optional[str]
   217	    pattern: Optional[_RegularExpressionProtocol]
   218	    present: Optional[bool]
   219	    exclude_everything: Optional[bool]
   220	    # TODO-TYPING: All MatchRule objects also have an attribute
   221	    # ``function``, but the type of the function depends on the
   222	    # subclass.
   223
   224	    def __init__(
   225	        self,
   226	        string: Optional[Union[str, bytes]] = None,
   227	        pattern: Optional[_RegularExpressionProtocol] = None,
   228	        function: Optional[Callable] = None,
   229	        present: Optional[bool] = None,
   230	        exclude_everything: Optional[bool] = None
   231	    ):
   232	        if isinstance(string, bytes):
   233	            string = string.decode("utf8")
   234	        self.string = string
   235	        if isinstance(pattern, bytes):
   236	            self.pattern = re.compile(pattern.decode("utf8"))
   237	        elif isinstance(pattern, str):
   238	            self.pattern = re.compile(pattern)
   239	        else:
   240	            self.pattern = pattern
   241	        self.function = function
   242	        self.present = present
   243	        self.exclude_everything = exclude_everything
   244
   245	        values = [
   246	            x
   247	            for x in (self.string, self.pattern, self.function, self.present, self.exclude_everything)
   248	            if x is not None
   249	        ]
   250	        if len(values) == 0:
   251	            raise ValueError(
   252	                "Either string, pattern, function, present, or exclude_everything must be provided."
   253	            )
   254	        if len(values) > 1:
   255	            raise ValueError(
   256	                "At most one of string, pattern, function, present, and exclude_everything must be provided."
   257	            )
   258
   259	    def _base_match(self, string: Optional[str]) -> Optional[bool]:
   260	        """Run the 'cheap' portion of a match, trying to get an answer without
   261	        calling a potentially expensive custom function.
   262
   263	        :return: True or False if we have a (positive or negative)
   264	        match; None if we need to keep trying.
   265	        """
   266	        # self.exclude_everything matches nothing.
   267	        if self.exclude_everything:
   268	            return False
   269
   270	        # self.present==True matches everything except None.
   271	        if self.present is True:
   272	            return string is not None
   273
   274	        # self.present==False matches _only_ None.
   275	        if self.present is False:
   276	            return string is None
   277
   278	        # self.string does an exact string match.
   279	        if self.string is not None:
   280	            # print(f"{self.string} ?= {string}")
   281	            return self.string == string
   282
   283	        # self.pattern does a regular expression search.
   284	        if self.pattern is not None:
   285	            # print(f"{self.pattern} ?~ {string}")
   286	            if string is None:
   287	                return False
   288	            return self.pattern.search(string) is not None
   289
   290	        return None
   291
   292	    def matches_string(self, string: Optional[str]) -> bool:
   293	        _base_result = self._base_match(string)
   294	        if _base_result is not None:
   295	            # No need to invoke the test function.
   296	            return _base_result
   297	        if self.function is not None and not self.function(string):
   298	            # print(f"{self.function}({string}) == False")
   299	            return False
   300	        return True
   301
   302	    def __repr__(self) -> str:
   303	        cls = type(self).__name__
   304	        return f"<{cls} string={self.string} pattern={self.pattern} function={self.function} present={self.present}>"
   305
   306	    def __eq__(self, other: Any) -> bool:
   307	        return (
   308	            isinstance(other, MatchRule)
   309	            and self.string == other.string
   310	            and self.pattern == other.pattern
   311	            and self.function == other.function
   312	            and self.present == other.present
   313	        )

   346	class SoupStrainer(ElementFilter):
   347	    """The `ElementFilter` subclass used internally by Beautiful Soup.
   348
   349	    A `SoupStrainer` encapsulates the logic necessary to perform the
   350	    kind of matches supported by methods such as
   351	    :py:meth:`Tag.find`. `SoupStrainer` objects are primarily created
   352	    internally, but you can create one yourself and pass it in as
   353	    ``parse_only`` to the `BeautifulSoup` constructor, to parse a
   354	    subset of a large document.
   355
   356	    Internally, `SoupStrainer` objects work by converting the
   357	    constructor arguments into `MatchRule` objects. Incoming
   358	    tags/markup are matched against those rules.
   359
   360	    :param name: One or more restrictions on the tags found in a document.
   361
   362	    :param attrs: A dictionary that maps attribute names to
   363	      restrictions on tags that use those attributes.
   364
   365	    :param string: One or more restrictions on the strings found in a
   366	      document.
   367
   368	    :param kwargs: A dictionary that maps attribute names to restrictions
   369	      on tags that use those attributes. These restrictions are additive to
   370	      any specified in ``attrs``.
   371
   372	    """
   373
   374	    name_rules: List[TagNameMatchRule]
   375	    attribute_rules: Dict[str, List[AttributeValueMatchRule]]
   376	    string_rules: List[StringMatchRule]
   377
   378	    def __init__(
   379	        self,
   380	        name: Optional[_StrainableElement] = None,
   381	        attrs: Dict[str, _StrainableAttribute] = {},
   382	        string: Optional[_StrainableString] = None,
   383	        **kwargs: _StrainableAttribute,
   384	    ):
   385	        if string is None and "text" in kwargs:
   386	            string = cast(Optional[_StrainableString], kwargs.pop("text"))
   387	            warnings.warn(
   388	                "As of version 4.11.0, the 'text' argument to the SoupStrainer constructor is deprecated. Use 'string' instead.",
   389	                DeprecationWarning,
   390	                stacklevel=2,
   391	            )
   392
   393	        if name is None and not attrs and not string and not kwargs:
   394	            # Special case for backwards compatibility. Instantiating
   395	            # a SoupStrainer with no arguments whatsoever gets you one
   396	            # that matches all Tags, and only Tags.
   397	            self.name_rules = [TagNameMatchRule(present=True)]
   398	        else:
   399	                self.name_rules = cast(
   400	                    List[TagNameMatchRule], list(self._make_match_rules(name, TagNameMatchRule))
   401	                )
   402	        self.attribute_rules = defaultdict(list)
   403
   404	        if not isinstance(attrs, dict):
   405	            # Passing something other than a dictionary as attrs is
   406	            # sugar for matching that thing against the 'class'
   407	            # attribute.
   408	            attrs = {"class": attrs}
   409
   410	        for attrdict in attrs, kwargs:
   411	            for attr, value in attrdict.items():
   412	                if attr == "class_" and attrdict is kwargs:
   413	                    # If you pass in 'class_' as part of kwargs, it's
   414	                    # because class is a Python reserved word. If you
   415	                    # pass it in as part of the attrs dict, it's
   416	                    # because you really are looking for an attribute
   417	                    # called 'class_'.
   418	                    attr = "class"
   419
   420	                if value is None:
   421	                    value = False
   422	                for rule_obj in self._make_match_rules(value, AttributeValueMatchRule):
   423	                    self.attribute_rules[attr].append(
   424	                        cast(AttributeValueMatchRule, rule_obj)
   425	                    )
   426
   427	        self.string_rules = cast(
   428	            List[StringMatchRule], list(self._make_match_rules(string, StringMatchRule))
   429	        )
   430
   431	        #: DEPRECATED 4.13.0: You shouldn't need to check this under
   432	        #: any name (.string or .text), and if you do, you're probably
   433	        #: not taking into account all of the types of values this
   434	        #: variable might have. Look at the .string_rules list instead.
   435	        self.__string = string
   436
   437	    @property
   438	    def includes_everything(self) -> bool:
   439	        """Check whether the provided rules will obviously include
   440	        everything. (They might include everything even if this returns `False`,
   441	        but not in an obvious way.)
   442	        """
   443	        return not self.name_rules and not self.string_rules and not self.attribute_rules
   444
   445	    @property
   446	    def excludes_everything(self) -> bool:
   447	        """Check whether the provided rules will obviously exclude
   448	        everything. (They might exclude everything even if this returns `False`,
   449	        but not in an obvious way.)
   450	        """
   451	        if (self.string_rules and (self.name_rules or self.attribute_rules)):
   452	            # This is self-contradictory, so the rules exclude everything.
   453	            return True
   454
   455	        # If there's a rule that ended up treated as an "exclude everything"
   456	        # rule due to creating a logical inconsistency, then the rules
   457	        # exclude everything.
   458	        if any(x.exclude_everything for x in self.string_rules):
   459	            return True
   460	        if any(x.exclude_everything for x in self.name_rules):
   461	            return True
   462	        for ruleset in self.attribute_rules.values():
   463	            if any(x.exclude_everything for x in ruleset):
   464	                return True
   465	        return False

   490	    @classmethod
   491	    def _make_match_rules(
   492	        cls,
   493	        obj: Optional[Union[_StrainableElement, _StrainableAttribute]],
   494	        rule_class: Type[MatchRule],
   495	    ) -> Iterator[MatchRule]:
   496	        """Convert a vaguely-specific 'object' into one or more well-defined
   497	        `MatchRule` objects.
   498
   499	        :param obj: Some kind of object that corresponds to one or more
   500	           matching rules.
   501	        :param rule_class: Create instances of this `MatchRule` subclass.
   502	        """
   503	        if obj is None:
   504	            return
   505	        if isinstance(obj, (str, bytes)):
   506	            yield rule_class(string=obj)
   507	        elif isinstance(obj, bool):
   508	            yield rule_class(present=obj)
   509	        elif callable(obj):
   510	            yield rule_class(function=obj)
   511	        elif isinstance(obj, _RegularExpressionProtocol):
   512	            yield rule_class(pattern=obj)
   513	        elif hasattr(obj, "__iter__"):
   514	            if not obj:
   515	                # The attribute is being matched against the null set,
   516	                # which means it should exclude everything.
   517	                yield rule_class(exclude_everything=True)
   518	            for o in obj:
   519	                if not isinstance(o, (bytes, str)) and hasattr(o, "__iter__"):
   520	                    # This is almost certainly the user's
   521	                    # mistake. This list contains another list, which
   522	                    # opens up the possibility of infinite
   523	                    # self-reference. In the interests of avoiding
   524	                    # infinite recursion, we'll treat this as an
   525	                    # impossible match and issue a rule that excludes
   526	                    # everything, rather than looking inside.
   527	                    warnings.warn(
   528	                        f"Ignoring nested list {o} to avoid the possibility of infinite recursion.",
   529	                        stacklevel=5,
   530	                    )
   531	                    yield rule_class(exclude_everything=True)
   532	                    continue
   533	                for x in cls._make_match_rules(o, rule_class):
   534	                    yield x
   535	        else:
   536	            yield rule_class(string=str(obj))

   538	    def matches_tag(self, tag: Tag) -> bool:
   539	        """Do the rules of this `SoupStrainer` trigger a match against the
   540	        given `Tag`?
   541
   542	        If the `SoupStrainer` has any `TagNameMatchRule`, at least one
   543	        must match the `Tag` or its `Tag.name`.
   544
   545	        If there are any `AttributeValueMatchRule` for a given
   546	        attribute, at least one of them must match the attribute
   547	        value.
   548
   549	        If there are any `StringMatchRule`, at least one must match,
   550	        but a `SoupStrainer` that *only* contains `StringMatchRule`
   551	        cannot match a `Tag`, only a `NavigableString`.
   552	        """
   553	        # If there are no rules at all, let anything through.
   554	        #if self.includes_everything:
   555	        #    return True
   556
   557	        # String rules cannot not match a Tag on their own.
   558	        if not self.name_rules and not self.attribute_rules:
   559	            return False
   560
   561	        # Optimization for a very common case where the user is
   562	        # searching for a tag with one specific name, and we're
   563	        # looking at a tag with a different name.
   564	        if (
   565	            not tag.prefix
   566	            and len(self.name_rules) == 1
   567	            and self.name_rules[0].string is not None
   568	            and tag.name != self.name_rules[0].string
   569	        ):
   570	            return False
   571
   572	        # If there are name rules, at least one must match. It can
   573	        # match either the Tag object itself or the prefixed name of
   574	        # the tag.
   575	        prefixed_name = None
   576	        if tag.prefix:
   577	            prefixed_name = f"{tag.prefix}:{tag.name}"
   578	        if self.name_rules:
   579	            name_matches = False
   580	            for rule in self.name_rules:
   581	                # attrs = " ".join(
   582	                #     [f"{k}={v}" for k, v in sorted(tag.attrs.items())]
   583	                # )
   584	                # print(f"Testing <{tag.name} {attrs}>{tag.string}</{tag.name}> against {rule}")
   585
   586	                # If the rule contains a function, the function will be called
   587	                # with `tag`. It will not be called a second time with
   588	                # `prefixed_name`.
   589	                if rule.matches_tag(tag) or (
   590	                        not rule.function and prefixed_name is not None and rule.matches_string(prefixed_name)
   591	                ):
   592	                    name_matches = True
   593	                    break
   594
   595	            if not name_matches:
   596	                return False
   597
   598	        # If there are attribute rules for a given attribute, at least
   599	        # one of them must match. If there are rules for multiple
   600	        # attributes, each attribute must have at least one match.
   601	        for attr, rules in self.attribute_rules.items():
   602	            attr_value = tag.get(attr, None)
   603	            this_attr_match = self._attribute_match(attr_value, rules)
   604	            if not this_attr_match:
   605	                return False
   606
   607	        # If there are string rules, at least one must match.
   608	        if self.string_rules:
   609	            _str = tag.string
   610	            if _str is None:
   611	                return False
   612	            if not self.matches_any_string_rule(_str):
   613	                return False
   614	        return True

   616	    def _attribute_match(
   617	        self,
   618	        attr_value: Optional[_AttributeValue],
   619	        rules: Iterable[AttributeValueMatchRule],
   620	    ) -> bool:
   621	        attr_values: Sequence[Optional[str]]
   622	        if isinstance(attr_value, list):
   623	            attr_values = attr_value
   624	        else:
   625	            attr_values = [cast(str, attr_value)]
   626
   627	        def _match_attribute_value_helper(attr_values: Sequence[Optional[str]]) -> bool:
   628	            for rule in rules:
   629	                for attr_value in attr_values:
   630	                    if rule.matches_string(attr_value):
   631	                        return True
   632	            return False
   633
   634	        this_attr_match = _match_attribute_value_helper(attr_values)
   635	        if not this_attr_match and len(attr_values) > 1:
   636	            # This cast converts Optional[str] to plain str.
   637	            #
   638	            # We know if there's more than one value, there can't be
   639	            # any None in the list, because Beautiful Soup never uses
   640	            # None as a value of a multi-valued attribute, and if None
   641	            # is passed in as attr_value, it's turned into a list with
   642	            # a single element (thus len(attr_values) > 1 fails).
   643	            attr_values = cast(Sequence[str], attr_values)
   644
   645	            # Try again but treat the attribute value
   646	            # as a single string.
   647	            joined_attr_value = " ".join(attr_values)
   648	            this_attr_match = _match_attribute_value_helper([joined_attr_value])
   649	        return this_attr_match

   651	    def allow_tag_creation(
   652	        self, nsprefix: Optional[str], name: str, attrs: Optional[_RawAttributeValues]
   653	    ) -> bool:
   654	        """Based on the name and attributes of a tag, see whether this
   655	        `SoupStrainer` will allow a `Tag` object to even be created.
   656
   657	        :param name: The name of the prospective tag.
   658	        :param attrs: The attributes of the prospective tag.
   659	        """
   660	        if self.string_rules:
   661	            # A SoupStrainer that has string rules can't be used to
   662	            # manage tag creation, because the string rule can't be
   663	            # evaluated until after the tag and all of its contents
   664	            # have been parsed.
   665	            return False
   666	        prefixed_name = None
   667	        if nsprefix:
   668	            prefixed_name = f"{nsprefix}:{name}"
   669	        if self.name_rules:
   670	            # At least one name rule must match.
   671	            name_match = False
   672	            for rule in self.name_rules:
   673	                for x in name, prefixed_name:
   674	                    if x is not None:

   196	                    )
   197	                    assert (
   198	                        child.previous_element is el
   199	                    ), "Bad previous_element\nNODE: {}\nPREV: {}\nEXPECTED: {}".format(
   200	                        child, child.previous_element, el
   201	                    )
   202	                    assert (
   203	                        child.previous_sibling is None
   204	                    ), "Bad previous_sibling\nNODE: {}\nPREV {}\nEXPECTED: {}".format(
   205	                        child, child.previous_sibling, None
   206	                    )
   207
   208	            # If not the first child, previous index should link as sibling to this index
   209	            # Previous element should match the last index or the last bubbled up descendant
   210	            else:
   211	                assert (
   212	                    child.previous_sibling is el.contents[idx - 1]
   213	                ), "Bad previous_sibling\nNODE: {}\nPREV {}\nEXPECTED {}".format(
   214	                    child, child.previous_sibling, el.contents[idx - 1]
   215	                )
   216	                assert (
   217	                    el.contents[idx - 1].next_sibling is child
   218	                ), "Bad next_sibling\nNODE: {}\nNEXT {}\nEXPECTED {}".format(
   219	                    el.contents[idx - 1], el.contents[idx - 1].next_sibling, child
   220	                )
   221
   222	                if last_child is not None:
   223	                    assert (
   224	                        child.previous_element is last_child
   225	                    ), "Bad previous_element\nNODE: {}\nPREV {}\nEXPECTED {}\nCONTENTS {}".format(
   226	                        child, child.previous_element, last_child, child.parent.contents
   227	                    )
   228	                    assert (
   229	                        last_child.next_element is child
   230	                    ), "Bad next_element\nNODE: {}\nNEXT {}\nEXPECTED {}".format(
   231	                        last_child, last_child.next_element, child
   232	                    )
   233
   234	            if isinstance(child, Tag) and child.contents:
   235	                descendant = self.linkage_validator(child, True)
   236	                assert descendant is not None
   237	                # A bubbled up descendant should have no next siblings
   238	                assert (
   239	                    descendant.next_sibling is None
   240	                ), "Bad next_sibling\nNODE: {}\nNEXT {}\nEXPECTED {}".format(
   241	                    descendant, descendant.next_sibling, None
   242	                )
   243
   244	            # Mark last child as either the bubbled up descendant or the current child
   245	            if descendant is not None:
   246	                last_child = descendant
   247	            else:
   248	                last_child = child
   249
   250	            # If last child, there are non next siblings
   251	            if idx == last_idx:
   252	                assert (
   253	                    child.next_sibling is None
   254	                ), "Bad next_sibling\nNODE: {}\nNEXT {}\nEXPECTED {}".format(
   255	                    child, child.next_sibling, None
   256	                )
   257	            idx += 1
   258
   259	        child = descendant if descendant is not None else child
   260	        if child is None:
   261	            child = el
   262
   263	        if not _recursive_call and child is not None:
   264	            target: Optional[Tag] = el
   265	            while True:
   266	                if target is None:
   267	                    assert (
   268	                        child.next_element is None
   269	                    ), "Bad next_element\nNODE: {}\nNEXT {}\nEXPECTED {}".format(
   270	                        child, child.next_element, None
   271	                    )
   272	                    break
   273	                elif target.next_sibling is not None:
   274	                    assert (
   275	                        child.next_element is target.next_sibling
   276	                    ), "Bad next_element\nNODE: {}\nNEXT {}\nEXPECTED {}".format(
   277	                        child, child.next_element, target.next_sibling
   278	                    )
   279	                    break
   280	                target = target.parent
   281
   282	            # We are done, so nothing to return
   283	            return None
   284	        else:
   285	            # Return the child to the recursive caller
   286	            return child
