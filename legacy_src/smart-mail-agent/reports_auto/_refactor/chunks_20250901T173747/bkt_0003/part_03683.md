    14	    name: str, modules: dict[str, MypyFile], *, raise_on_missing: bool = False
    15	) -> SymbolTableNode | None:
    16	    """Find a symbol using it fully qualified name.
    17
    18	    The algorithm has two steps: first we try splitting the name on '.' to find
    19	    the module, then iteratively look for each next chunk after a '.' (e.g. for
    20	    nested classes).

    22	    This function should *not* be used to find a module. Those should be looked
    23	    in the modules dictionary.
    24	    """
    25	    # 1. Exclude the names of ad hoc instance intersections from step 2.
    26	    i = name.find("<subclass ")
    27	    head = name if i == -1 else name[:i]
    28	    rest = []
    29	    # 2. Find a module tree in modules dictionary.
    30	    while True:
    31	        if "." not in head:
    32	            if raise_on_missing:
    33	                assert "." in head, f"Cannot find module for {name}"
    34	            return None
    35	        head, tail = head.rsplit(".", maxsplit=1)
    36	        rest.append(tail)
    37	        mod = modules.get(head)
    38	        if mod is not None:
    39	            break
    40	    names = mod.names
    41	    # 3. Find the symbol in the module tree.
    42	    if not rest:
    43	        # Looks like a module, don't use this to avoid confusions.
    44	        if raise_on_missing:
    45	            assert rest, f"Cannot find {name}, got a module symbol"
    46	        return None
    47	    if i != -1:
    48	        rest[0] += name[i:]
    49	    while True:
    50	        key = rest.pop()
    51	        if key not in names:
    52	            if raise_on_missing:
    53	                assert key in names, f"Cannot find component {key!r} for {name!r}"
    54	            return None
    55	        stnode = names[key]
    56	        if not rest:
    57	            return stnode
    58	        node = stnode.node
    59	        # In fine-grained mode, could be a cross-reference to a deleted module
    60	        # or a Var made up for a missing module.
    61	        if not isinstance(node, TypeInfo):
    62	            if raise_on_missing:
    63	                assert node, f"Cannot find {name}"
    64	            return None
    65	        names = node.names

   296	    TypeVars (in both strict-optional and non-strict-optional mode).
   297	    If 'overlap_for_overloads' is True, we check for overlaps more strictly (to avoid false
   298	    positives), for example: None only overlaps with explicitly optional types, Any
   299	    doesn't overlap with anything except object, we don't ignore positional argument names.
   300	    """
   301	    if isinstance(left, TypeGuardedType) or isinstance(  # type: ignore[misc]
   302	        right, TypeGuardedType
   303	    ):
   304	        # A type guard forces the new type even if it doesn't overlap the old.
   305	        return True
   306
   307	    if seen_types is None:
   308	        seen_types = set()
   309	    if (left, right) in seen_types:

   311	    if isinstance(left, TypeAliasType) and isinstance(right, TypeAliasType):
   312	        seen_types.add((left, right))

   315
   316	    def _is_overlapping_types(left: Type, right: Type) -> bool:
   317	        """Encode the kind of overlapping check to perform.
   318
   319	        This function mostly exists, so we don't have to repeat keyword arguments everywhere.
   320	        """
   321	        return is_overlapping_types(

   323	            right,
   324	            ignore_promotions=ignore_promotions,
   325	            prohibit_none_typevar_overlap=prohibit_none_typevar_overlap,
   326	            overlap_for_overloads=overlap_for_overloads,
   327	            seen_types=seen_types.copy(),
   328	        )
   329
   330	    # We should never encounter this type.
   331	    if isinstance(left, PartialType) or isinstance(right, PartialType):
   332	        assert False, "Unexpectedly encountered partial type"
   333
   334	    # We should also never encounter these types, but it's possible a few
   335	    # have snuck through due to unrelated bugs. For now, we handle these
   336	    # in the same way we handle 'Any'.
   337	    #
   338	    # TODO: Replace these with an 'assert False' once we are more confident.
   339	    illegal_types = (UnboundType, ErasedType, DeletedType)
   340	    if isinstance(left, illegal_types) or isinstance(right, illegal_types):
   341	        return True
   342
   343	    # When running under non-strict optional mode, simplify away types of
   344	    # the form 'Union[A, B, C, None]' into just 'Union[A, B, C]'.
   345
   346	    if not state.strict_optional:
   347	        if isinstance(left, UnionType):
   348	            left = UnionType.make_union(left.relevant_items())
   349	        if isinstance(right, UnionType):
   350	            right = UnionType.make_union(right.relevant_items())
   351	        left, right = get_proper_types((left, right))
   352
   353	    # 'Any' may or may not be overlapping with the other type
   354	    if isinstance(left, AnyType) or isinstance(right, AnyType):
   355	        return not overlap_for_overloads or is_object(left) or is_object(right)
   356
   357	    # We check for complete overlaps next as a general-purpose failsafe.
   358	    # If this check fails, we start checking to see if there exists a
   359	    # *partial* overlap between types.
   360	    #
   361	    # These checks will also handle the NoneType and UninhabitedType cases for us.
   362
   363	    # enums are sometimes expanded into an Union of Literals
   364	    # when that happens we want to make sure we treat the two as overlapping
   365	    # and crucially, we want to do that *fast* in case the enum is large
   366	    # so we do it before expanding variants below to avoid O(n**2) behavior
   367	    if (
   368	        is_enum_overlapping_union(left, right)
   369	        or is_enum_overlapping_union(right, left)
   370	        or is_literal_in_union(left, right)
   371	        or is_literal_in_union(right, left)
   372	    ):
   373	        return True
   374
   375	    def is_none_object_overlap(t1: Type, t2: Type) -> bool:
   376	        t1, t2 = get_proper_types((t1, t2))
   377	        return (
   378	            isinstance(t1, NoneType)
   379	            and isinstance(t2, Instance)
   380	            and t2.type.fullname == "builtins.object"
   381	        )
   382
   383	    if overlap_for_overloads:
   384	        if is_none_object_overlap(left, right) or is_none_object_overlap(right, left):
   385	            return False
   386
   387	    def _is_subtype(left: Type, right: Type) -> bool:
   388	        if overlap_for_overloads:
   389	            return is_proper_subtype(left, right, ignore_promotions=ignore_promotions)
   390	        else:
   391	            return is_subtype(left, right, ignore_promotions=ignore_promotions)

   394	        return True
