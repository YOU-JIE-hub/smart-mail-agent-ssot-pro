   208	        thing = contains_int_or_tuple_of_ints(expr.operands[0])
   209	        op = reverse_op[op]
   210	    if isinstance(index, int) and isinstance(thing, int):
   211	        # sys.version_info[i] <compare_op> k
   212	        if 0 <= index <= 1:
   213	            return fixed_comparison(pyversion[index], op, thing)
   214	        else:
   215	            return TRUTH_VALUE_UNKNOWN
   216	    elif isinstance(index, tuple) and isinstance(thing, tuple):
   217	        lo, hi = index
   218	        if lo is None:
   219	            lo = 0
   220	        if hi is None:
   221	            hi = 2
   222	        if 0 <= lo < hi <= 2:
   223	            val = pyversion[lo:hi]
   224	            if len(val) == len(thing) or len(val) > len(thing) and op not in ("==", "!="):
   225	                return fixed_comparison(val, op, thing)
   226	    return TRUTH_VALUE_UNKNOWN

   229	def consider_sys_platform(expr: Expression, platform: str) -> int:
   230	    """Consider whether expr is a comparison involving sys.platform.
   231
   232	    Return ALWAYS_TRUE, ALWAYS_FALSE, or TRUTH_VALUE_UNKNOWN.
   233	    """
   234	    # Cases supported:
   235	    # - sys.platform == 'linux'
   236	    # - sys.platform != 'win32'
   237	    # - sys.platform.startswith('win')
   238	    if isinstance(expr, ComparisonExpr):
   239	        # Let's not yet support chained comparisons.
   240	        if len(expr.operators) > 1:
   241	            return TRUTH_VALUE_UNKNOWN
   242	        op = expr.operators[0]
   243	        if op not in ("==", "!="):
   244	            return TRUTH_VALUE_UNKNOWN
   245	        if not is_sys_attr(expr.operands[0], "platform"):
   246	            return TRUTH_VALUE_UNKNOWN
   247	        right = expr.operands[1]
   248	        if not isinstance(right, StrExpr):
   249	            return TRUTH_VALUE_UNKNOWN
   250	        return fixed_comparison(platform, op, right.value)
   251	    elif isinstance(expr, CallExpr):
   252	        if not isinstance(expr.callee, MemberExpr):
   253	            return TRUTH_VALUE_UNKNOWN
   254	        if len(expr.args) != 1 or not isinstance(expr.args[0], StrExpr):
   255	            return TRUTH_VALUE_UNKNOWN
   256	        if not is_sys_attr(expr.callee.expr, "platform"):
   257	            return TRUTH_VALUE_UNKNOWN
   258	        if expr.callee.name != "startswith":
   259	            return TRUTH_VALUE_UNKNOWN
   260	        if platform.startswith(expr.args[0].value):
   261	            return ALWAYS_TRUE
   262	        else:
   263	            return ALWAYS_FALSE
   264	    else:
   265	        return TRUTH_VALUE_UNKNOWN

   271	def fixed_comparison(left: Targ, op: str, right: Targ) -> int:
   272	    rmap = {False: ALWAYS_FALSE, True: ALWAYS_TRUE}
   273	    if op == "==":
   274	        return rmap[left == right]
   275	    if op == "!=":
   276	        return rmap[left != right]
   277	    if op == "<=":
   278	        return rmap[left <= right]
   279	    if op == ">=":
   280	        return rmap[left >= right]
   281	    if op == "<":
   282	        return rmap[left < right]
   283	    if op == ">":
   284	        return rmap[left > right]
   285	    return TRUTH_VALUE_UNKNOWN

   288	def contains_int_or_tuple_of_ints(expr: Expression) -> None | int | tuple[int, ...]:
   289	    if isinstance(expr, IntExpr):
   290	        return expr.value
   291	    if isinstance(expr, TupleExpr):
   292	        if literal(expr) == LITERAL_YES:
   293	            thing = []
   294	            for x in expr.items:
   295	                if not isinstance(x, IntExpr):
   296	                    return None
   297	                thing.append(x.value)
   298	            return tuple(thing)
   299	    return None

   302	def contains_sys_version_info(expr: Expression) -> None | int | tuple[int | None, int | None]:
   303	    if is_sys_attr(expr, "version_info"):
   304	        return (None, None)  # Same as sys.version_info[:]
   305	    if isinstance(expr, IndexExpr) and is_sys_attr(expr.base, "version_info"):
   306	        index = expr.index
   307	        if isinstance(index, IntExpr):
   308	            return index.value
   309	        if isinstance(index, SliceExpr):
   310	            if index.stride is not None:
   311	                if not isinstance(index.stride, IntExpr) or index.stride.value != 1:
   312	                    return None
   313	            begin = end = None
   314	            if index.begin_index is not None:
   315	                if not isinstance(index.begin_index, IntExpr):
   316	                    return None
   317	                begin = index.begin_index.value
   318	            if index.end_index is not None:
   319	                if not isinstance(index.end_index, IntExpr):
   320	                    return None
   321	                end = index.end_index.value
   322	            return (begin, end)
   323	    return None

   327	    # TODO: This currently doesn't work with code like this:
