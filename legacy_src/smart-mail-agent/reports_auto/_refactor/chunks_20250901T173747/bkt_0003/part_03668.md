  1706	        self, defn: FuncItem, reverse_type: CallableType, reverse_name: str, context: Context
  1707	    ) -> None:
  1708	        """Check a reverse operator method such as __radd__."""
  1709	        # Decides whether it's worth calling check_overlapping_op_methods().
  1710
  1711	        # This used to check for some very obscure scenario.  It now
  1712	        # just decides whether it's worth calling
  1713	        # check_overlapping_op_methods().

  1715	        assert defn.info
  1716

  1718	        method_type = CallableType(
  1719	            [AnyType(TypeOfAny.special_form), AnyType(TypeOfAny.special_form)],
  1720	            [nodes.ARG_POS, nodes.ARG_POS],
  1721	            [None, None],
  1722	            AnyType(TypeOfAny.special_form),
  1723	            self.named_type("builtins.function"),
  1724	        )

  1726	            self.msg.invalid_signature(reverse_type, context)

  1730	            # These are defined for all objects => can't cause trouble.
  1731	            return

  1733	        # With 'Any' or 'object' return type we are happy, since any possible
  1734	        # return value is valid.
  1735	        ret_type = get_proper_type(reverse_type.ret_type)

  1742	            reverse_type = reverse_type.copy_modified(
  1743	                arg_types=[reverse_type.arg_types[0]] * 2,
  1744	                arg_kinds=[ARG_POS] * 2,
  1745	                arg_names=[reverse_type.arg_names[0], "_"],
  1746	            )
  1747	        assert len(reverse_type.arg_types) >= 2
  1748
  1749	        forward_name = operators.normal_from_reverse_op[reverse_name]
  1750	        forward_inst = get_proper_type(reverse_type.arg_types[1])
  1751	        if isinstance(forward_inst, TypeVarType):
  1752	            forward_inst = get_proper_type(forward_inst.upper_bound)
  1753	        elif isinstance(forward_inst, TupleType):
  1754	            forward_inst = tuple_fallback(forward_inst)
  1755	        elif isinstance(forward_inst, (FunctionLike, TypedDictType, LiteralType)):
  1756	            forward_inst = forward_inst.fallback
  1757	        if isinstance(forward_inst, TypeType):

  1774	        if not (
  1775	            isinstance(forward_inst, (Instance, UnionType))

  1780	        forward_type = self.expr_checker.analyze_external_member_access(

  1793	    def check_overlapping_op_methods(
  1794	        self,
  1795	        reverse_type: CallableType,
  1796	        reverse_name: str,
  1797	        reverse_class: TypeInfo,
  1798	        forward_type: Type,
  1799	        forward_name: str,
  1800	        forward_base: Type,
  1801	        context: Context,
  1802	    ) -> None:
  1803	        """Check for overlapping method and reverse method signatures.
  1804
  1805	        This function assumes that:
  1806
  1807	        -   The reverse method has valid argument count and kinds.
  1808	        -   If the reverse operator method accepts some argument of type
  1809	            X, the forward operator method also belong to class X.
  1810
  1811	            For example, if we have the reverse operator `A.__radd__(B)`, then the
  1812	            corresponding forward operator must have the type `B.__add__(...)`.
  1813	        """
  1814
  1815	        # Note: Suppose we have two operator methods "A.__rOP__(B) -> R1" and
  1816	        # "B.__OP__(C) -> R2". We check if these two methods are unsafely overlapping
  1817	        # by using the following algorithm:
  1818	        #
  1819	        # 1. Rewrite "B.__OP__(C) -> R1"  to "temp1(B, C) -> R1"
  1820	        #
  1821	        # 2. Rewrite "A.__rOP__(B) -> R2" to "temp2(B, A) -> R2"
  1822	        #
  1823	        # 3. Treat temp1 and temp2 as if they were both variants in the same
  1824	        #    overloaded function. (This mirrors how the Python runtime calls
  1825	        #    operator methods: we first try __OP__, then __rOP__.)
  1826	        #
  1827	        #    If the first signature is unsafely overlapping with the second,
  1828	        #    report an error.
  1829	        #
  1830	        # 4. However, if temp1 shadows temp2 (e.g. the __rOP__ method can never
  1831	        #    be called), do NOT report an error.
  1832	        #
  1833	        #    This behavior deviates from how we handle overloads -- many of the
  1834	        #    modules in typeshed seem to define __OP__ methods that shadow the
  1835	        #    corresponding __rOP__ method.
  1836	        #
  1837	        # Note: we do not attempt to handle unsafe overlaps related to multiple
  1838	        # inheritance. (This is consistent with how we handle overloads: we also
  1839	        # do not try checking unsafe overlaps due to multiple inheritance there.)
  1840
  1841	        for forward_item in flatten_nested_unions([forward_type]):
  1842	            forward_item = get_proper_type(forward_item)
  1843	            if isinstance(forward_item, CallableType):
  1844	                if self.is_unsafe_overlapping_op(forward_item, forward_base, reverse_type):
  1845	                    self.msg.operator_method_signatures_overlap(
  1846	                        reverse_class, reverse_name, forward_base, forward_name, context
  1847	                    )
  1848	            elif isinstance(forward_item, Overloaded):
  1849	                for item in forward_item.items:
  1850	                    if self.is_unsafe_overlapping_op(item, forward_base, reverse_type):
  1851	                        self.msg.operator_method_signatures_overlap(
  1852	                            reverse_class, reverse_name, forward_base, forward_name, context
  1853	                        )
  1854	            elif not isinstance(forward_item, AnyType):
  1855	                self.msg.forward_operator_not_callable(forward_name, context)
  1856
  1857	    def is_unsafe_overlapping_op(
  1858	        self, forward_item: CallableType, forward_base: Type, reverse_type: CallableType
  1859	    ) -> bool:
  1860	        # TODO: check argument kinds?
  1861	        if len(forward_item.arg_types) < 1:
  1862	            # Not a valid operator method -- can't succeed anyway.
  1863	            return False
  1864
  1865	        # Erase the type if necessary to make sure we don't have a single
  1866	        # TypeVar in forward_tweaked. (Having a function signature containing
  1867	        # just a single TypeVar can lead to unpredictable behavior.)
  1868	        forward_base_erased = forward_base
  1869	        if isinstance(forward_base, TypeVarType):
  1870	            forward_base_erased = erase_to_bound(forward_base)
  1871
  1872	        # Construct normalized function signatures corresponding to the
  1873	        # operator methods. The first argument is the left operand and the
  1874	        # second operand is the right argument -- we switch the order of
  1875	        # the arguments of the reverse method.
  1876
  1877	        # TODO: this manipulation is dangerous if callables are generic.
  1878	        # Shuffling arguments between callables can create meaningless types.
  1879	        forward_tweaked = forward_item.copy_modified(
  1880	            arg_types=[forward_base_erased, forward_item.arg_types[0]],
  1881	            arg_kinds=[nodes.ARG_POS] * 2,
  1882	            arg_names=[None] * 2,
  1883	        )
  1884	        reverse_tweaked = reverse_type.copy_modified(
  1885	            arg_types=[reverse_type.arg_types[1], reverse_type.arg_types[0]],
  1886	            arg_kinds=[nodes.ARG_POS] * 2,
  1887	            arg_names=[None] * 2,
  1888	        )
  1889
  1890	        reverse_base_erased = reverse_type.arg_types[0]
  1891	        if isinstance(reverse_base_erased, TypeVarType):
  1892	            reverse_base_erased = erase_to_bound(reverse_base_erased)
  1893
  1894	        if is_same_type(reverse_base_erased, forward_base_erased):
  1895	            return False
  1896	        elif is_subtype(reverse_base_erased, forward_base_erased):
  1897	            first = reverse_tweaked
  1898	            second = forward_tweaked
  1899	        else:
  1900	            first = forward_tweaked
  1901	            second = reverse_tweaked
  1902
  1903	        current_class = self.scope.active_class()
  1904	        type_vars = current_class.defn.type_vars if current_class else []
  1905	        return is_unsafe_overlapping_overload_signatures(
  1906	            first, second, type_vars, partial_only=False
  1907	        )

  2059	    def check_method_override(
  2060	        self, defn: FuncDef | OverloadedFuncDef | Decorator
  2061	    ) -> list[TypeInfo] | None:
  2062	        """Check if function definition is compatible with base classes.
  2063
  2064	        This may defer the method if a signature is not available in at least one base class.
  2065	        Return ``None`` if that happens.
  2066
  2067	        Return a list of base classes which contain an attribute with the method name.
  2068	        """
  2069	        # Check against definitions in base classes.
  2070	        check_override_compatibility = (
  2071	            defn.name not in ("__init__", "__new__", "__init_subclass__", "__post_init__")
  2072	            and (self.options.check_untyped_defs or not defn.is_dynamic())
  2073	            and (
  2074	                # don't check override for synthesized __replace__ methods from dataclasses
  2075	                defn.name != "__replace__"
  2076	                or defn.info.metadata.get("dataclass_tag") is None
  2077	            )
  2078	        )
  2079	        found_method_base_classes: list[TypeInfo] = []
  2080	        for base in defn.info.mro[1:]:
  2081	            result = self.check_method_or_accessor_override_for_base(
  2082	                defn, base, check_override_compatibility
  2083	            )
  2084	            if result is None:
  2085	                # Node was deferred, we will have another attempt later.
  2086	                return None
  2087	            if result:
  2088	                found_method_base_classes.append(base)
  2089	        return found_method_base_classes

  2091	    def check_method_or_accessor_override_for_base(
  2092	        self,
  2093	        defn: FuncDef | OverloadedFuncDef | Decorator,
  2094	        base: TypeInfo,
  2095	        check_override_compatibility: bool,
  2096	    ) -> bool | None:
  2097	        """Check if method definition is compatible with a base class.
  2098
  2099	        Return ``None`` if the node was deferred because one of the corresponding
  2100	        superclass nodes is not ready.
  2101
  2102	        Return ``True`` if an attribute with the method name was found in the base class.
  2103	        """
  2104	        found_base_method = False
  2105	        if base:
  2106	            name = defn.name
  2107	            base_attr = base.names.get(name)

  2343	        self,
  2344	        override: FunctionLike,
  2345	        original: FunctionLike,
  2346	        name: str,
  2347	        name_in_super: str,
  2348	        supertype: str,
  2349	        original_class_or_static: bool,
  2350	        override_class_or_static: bool,
  2351	        node: Context,

  2353	        """Check a method override with given signatures.
  2354
  2355	        Arguments:
  2356	          override:                 The signature of the overriding method.
  2357	          original:                 The signature of the original supertype method.
  2358	          name:                     The name of the overriding method.
  2359	                                    Used primarily for generating error messages.
  2360	          name_in_super:            The name of the overridden in the superclass.
  2361	                                    Used for generating error messages only.
  2362	          supertype:                The name of the supertype.
  2363	          original_class_or_static: Indicates whether the original method (from the superclass)
  2364	                                    is either a class method or a static method.
  2365	          override_class_or_static: Indicates whether the overriding method (from the subclass)
  2366	                                    is either a class method or a static method.

  2369	        # Use boolean variable to clarify code.
  2370	        fail = False
  2371	        op_method_wider_note = False
  2372	        if not is_subtype(override, original, ignore_pos_arg_names=True):
  2373	            fail = True
  2374	        elif isinstance(override, Overloaded) and self.is_forward_op_method(name):
  2375	            # Operator method overrides cannot extend the domain, as
  2376	            # this could be unsafe with reverse operator methods.
  2377	            original_domain = self.get_op_other_domain(original)
  2378	            override_domain = self.get_op_other_domain(override)
  2379	            if (
  2380	                original_domain
  2381	                and override_domain
  2382	                and not is_subtype(override_domain, original_domain)
  2383	            ):
  2384	                fail = True
  2385	                op_method_wider_note = True
  2386	        if isinstance(override, FunctionLike):
  2387	            if original_class_or_static and not override_class_or_static:
  2388	                fail = True
  2389	            elif isinstance(original, CallableType) and isinstance(override, CallableType):
  2390	                if original.type_guard is not None and override.type_guard is None:
  2391	                    fail = True
  2392	                if original.type_is is not None and override.type_is is None:
  2393	                    fail = True
  2394
  2395	        if is_private(name):
  2396	            fail = False
  2397
  2398	        if fail:

  2402	            # Normalize signatures, so we get better diagnostics.
  2403	            if isinstance(override, (CallableType, Overloaded)):

  2405	            if isinstance(original, (CallableType, Overloaded)):
  2406	                original = original.with_unpacked_kwargs()

  2408	            if (
  2409	                isinstance(override, CallableType)
  2410	                and isinstance(original, CallableType)
  2411	                and len(override.arg_types) == len(original.arg_types)
  2412	                and override.min_args == original.min_args

  2415	                # signatures having the same number of arguments and no
  2416	                # overloads.
  2417
  2418	                # override might have its own generic function type
  2419	                # variables. If an argument or return type of override
  2420	                # does not have the correct subtyping relationship
  2421	                # with the original type even after these variables
  2422	                # are erased, then it is definitely an incompatibility.

  2425	                type_name = None
  2426	                if isinstance(override.definition, FuncDef):
  2427	                    type_name = override.definition.info.name
  2428
  2429	                def erase_override(t: Type) -> Type:
  2430	                    return erase_typevars(t, ids_to_erase=override_ids)

  2433	                    zip(override.arg_kinds, original.arg_kinds)
  2434	                ):
  2435	                    if sub_kind.is_positional() and super_kind.is_positional():
  2436	                        override_arg_type = override.arg_types[i]
  2437	                        original_arg_type = original.arg_types[i]

  2439	                        arg_name = override.arg_names[i]
  2440	                        if arg_name in original.arg_names:
  2441	                            override_arg_type = override.arg_types[i]
  2442	                            original_i = original.arg_names.index(arg_name)
  2443	                            original_arg_type = original.arg_types[original_i]

  2612	    def check_init_subclass(self, defn: ClassDef) -> None:
  2613	        """Check that keywords in a class definition are valid arguments for __init_subclass__().
  2614
  2615	        In this example:
  2616	            1   class Base:
  2617	            2       def __init_subclass__(cls, thing: int):
  2618	            3           pass
  2619	            4   class Child(Base, thing=5):
  2620	            5       def __init_subclass__(cls):
  2621	            6           pass
  2622	            7   Child()
  2623
  2624	        Base.__init_subclass__(thing=5) is called at line 4. This is what we simulate here.
  2625	        Child.__init_subclass__ is never called.
  2626	        """
  2627	        if defn.info.metaclass_type and defn.info.metaclass_type.type.fullname not in (
  2628	            "builtins.type",
  2629	            "abc.ABCMeta",
  2630	        ):
  2631	            # We can't safely check situations when both __init_subclass__ and a custom
  2632	            # metaclass are present.
  2633	            return
  2634	        # At runtime, only Base.__init_subclass__ will be called, so
  2635	        # we skip the current class itself.
  2636	        for base in defn.info.mro[1:]:
  2637	            if "__init_subclass__" not in base.names:
  2638	                continue
  2639	            name_expr = NameExpr(defn.name)
  2640	            name_expr.node = base
  2641	            callee = MemberExpr(name_expr, "__init_subclass__")
  2642	            args = list(defn.keywords.values())
  2643	            arg_names: list[str | None] = list(defn.keywords.keys())
  2644	            # 'metaclass' keyword is consumed by the rest of the type machinery,
  2645	            # and is never passed to __init_subclass__ implementations
  2646	            if "metaclass" in arg_names:
  2647	                idx = arg_names.index("metaclass")
  2648	                arg_names.pop(idx)
  2649	                args.pop(idx)
  2650	            arg_kinds = [ARG_NAMED] * len(args)
  2651	            call_expr = CallExpr(callee, args, arg_kinds, arg_names)
  2652	            call_expr.line = defn.line
  2653	            call_expr.column = defn.column
  2654	            call_expr.end_line = defn.end_line
  2655	            self.expr_checker.accept(call_expr, allow_none_return=True, always_allow_any=True)
  2656	            # We are only interested in the first Base having __init_subclass__,
  2657	            # all other bases have already been checked.
  2658	            break

  2819	    def check_multiple_inheritance(self, typ: TypeInfo) -> None:
  2820	        """Check for multiple inheritance related errors."""
  2821	        if len(typ.bases) <= 1:
  2822	            # No multiple inheritance.
  2823	            return
  2824	        # Verify that inherited attributes are compatible.
  2825	        mro = typ.mro[1:]
  2826	        all_names = {name for base in mro for name in base.names}
  2827	        for name in sorted(all_names - typ.names.keys()):
  2828	            # Sort for reproducible message order.
  2829	            # Attributes defined in both the type and base are skipped.
  2830	            # Normal checks for attribute compatibility should catch any problems elsewhere.
  2831	            if is_private(name):
  2832	                continue
  2833	            # Compare the first base defining a name with the rest.
  2834	            # Remaining bases may not be pairwise compatible as the first base provides
  2835	            # the used definition.
  2836	            i, base = next((i, base) for i, base in enumerate(mro) if name in base.names)
  2837	            for base2 in mro[i + 1 :]:

  2839	                    self.check_compatibility(name, base, base2, typ)

  2841	    def check_compatibility(
  2842	        self, name: str, base1: TypeInfo, base2: TypeInfo, ctx: TypeInfo
  2843	    ) -> None:
  2844	        """Check if attribute name in base1 is compatible with base2 in multiple inheritance.
  2845
  2846	        Assume base1 comes before base2 in the MRO, and that base1 and base2 don't have
  2847	        a direct subclass relationship (i.e., the compatibility requirement only derives from
  2848	        multiple inheritance).
  2849
  2850	        This check verifies that a definition taken from base1 (and mapped to the current
  2851	        class ctx), is type compatible with the definition taken from base2 (also mapped), so
  2852	        that unsafe subclassing like this can be detected:
  2853	            class A(Generic[T]):
  2854	                def foo(self, x: T) -> None: ...
  2855
  2856	            class B:
  2857	                def foo(self, x: str) -> None: ...
  2858
  2859	            class C(B, A[int]): ...  # this is unsafe because...
  2860
  2861	            x: A[int] = C()
  2862	            x.foo  # ...runtime type is (str) -> None, while static type is (int) -> None
  2863	        """
  2864	        if name in ("__init__", "__new__", "__init_subclass__"):
  2865	            # __init__ and friends can be incompatible -- it's a special case.
  2866	            return
  2867	        first = base1.names[name]
  2868	        second = base2.names[name]
  2869	        # Specify current_class explicitly as this function is called after leaving the class.
  2870	        first_type, _ = self.node_type_from_base(name, base1, ctx, current_class=ctx)
  2871	        second_type, _ = self.node_type_from_base(name, base2, ctx, current_class=ctx)
  2872
  2873	        # TODO: use more principled logic to decide is_subtype() vs is_equivalent().
  2874	        # We should rely on mutability of superclass node, not on types being Callable.
  2875	        # (in particular handle settable properties with setter type different from getter).
  2876
  2877	        p_first_type = get_proper_type(first_type)
  2878	        p_second_type = get_proper_type(second_type)
  2879	        if isinstance(p_first_type, FunctionLike) and isinstance(p_second_type, FunctionLike):
  2880	            if p_first_type.is_type_obj() and p_second_type.is_type_obj():
  2881	                # For class objects only check the subtype relationship of the classes,
  2882	                # since we allow incompatible overrides of '__init__'/'__new__'
  2883	                ok = is_subtype(
  2884	                    left=fill_typevars_with_any(p_first_type.type_object()),
  2885	                    right=fill_typevars_with_any(p_second_type.type_object()),
  2886	                )
  2887	            else:
  2888	                assert first_type and second_type
  2889	                ok = is_subtype(first_type, second_type, ignore_pos_arg_names=True)
  2890	        elif first_type and second_type:
  2891	            if second.node is not None and not self.is_writable_attribute(second.node):
  2892	                ok = is_subtype(first_type, second_type)
  2893	            else:
  2894	                ok = is_equivalent(first_type, second_type)
  2895	            if ok:
  2896	                if (
  2897	                    first.node
  2898	                    and second.node
  2899	                    and self.is_writable_attribute(second.node)
  2900	                    and is_property(first.node)
  2901	                    and isinstance(first.node, Decorator)
  2902	                    and not isinstance(p_second_type, AnyType)
  2903	                ):
  2904	                    self.msg.fail(
  2905	                        f'Cannot override writeable attribute "{name}" in base "{base2.name}"'
  2906	                        f' with read-only property in base "{base1.name}"',
  2907	                        ctx,
  2908	                        code=codes.OVERRIDE,
  2909	                    )
  2910	        else:
  2911	            if first_type is None:
  2912	                self.msg.cannot_determine_type_in_base(name, base1.name, ctx)
  2913	            if second_type is None:
  2914	                self.msg.cannot_determine_type_in_base(name, base2.name, ctx)
  2915	            ok = True
  2916	        # Final attributes can never be overridden, but can override
  2917	        # non-final read-only attributes.
  2918	        if is_final_node(second.node) and not is_private(name):
  2919	            self.msg.cant_override_final(name, base2.name, ctx)
  2920	        if is_final_node(first.node):
  2921	            self.check_if_final_var_override_writable(name, second.node, ctx)
  2922	        # Some attributes like __slots__ and __deletable__ are special, and the type can
  2923	        # vary across class hierarchy.
  2924	        if isinstance(second.node, Var) and second.node.allow_incompatible_override:
  2925	            ok = True
  2926	        if not ok:
  2927	            self.msg.base_class_definitions_incompatible(name, base1, base2, ctx)
