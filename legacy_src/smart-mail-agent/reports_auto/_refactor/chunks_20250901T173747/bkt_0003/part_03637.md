   104	            # sys.path fakery.  If we are being run as a command, then sys.path[0]
   105	            # is the directory of the "coverage" script.  If this is so, replace
   106	            # sys.path[0] with the directory of the file we're running, or the
   107	            # current directory when running modules.  If it isn't so, then we
   108	            # don't know what's going on, and just leave it alone.

   110	            sys_path_0_abs = os.path.abspath(sys.path[0])

   112	            sys_path_0_abs = canonical_filename(sys_path_0_abs)
   113	            top_file_dir_abs = canonical_filename(top_file_dir_abs)
   114	            if sys_path_0_abs != top_file_dir_abs:

   299	def make_code_from_pyc(filename: str) -> CodeType:
   300	    """Get a code object from a .pyc file."""
   301	    try:
   302	        fpyc = open(filename, "rb")
   303	    except OSError as exc:
   304	        raise NoCode(f"No file to run: '{filename}'") from exc
   305
   306	    with fpyc:
   307	        # First four bytes are a version-specific magic number.  It has to
   308	        # match or we won't run the file.
   309	        magic = fpyc.read(4)
   310	        if magic != PYC_MAGIC_NUMBER:
   311	            raise NoCode(f"Bad magic number in .pyc file: {magic!r} != {PYC_MAGIC_NUMBER!r}")
   312
   313	        flags = struct.unpack("<L", fpyc.read(4))[0]
   314	        hash_based = flags & 0x01
   315	        if hash_based:
   316	            fpyc.read(8)    # Skip the hash.
   317	        else:

   308	# Tokenizer for _glob_to_regex.
   309	# None as a sub means disallowed.
   310	G2RX_TOKENS = [(re.compile(rx), sub) for rx, sub in [
   311	    (r"\*\*\*+", None),             # Can't have ***
   312	    (r"[^/]+\*\*+", None),          # Can't have x**
   313	    (r"\*\*+[^/]+", None),          # Can't have **x
   314	    (r"\*\*/\*\*", None),           # Can't have **/**
   315	    (r"^\*+/", r"(.*[/\\\\])?"),    # ^*/ matches any prefix-slash, or nothing.
   316	    (r"/\*+$", r"[/\\\\].*"),       # /*$ matches any slash-suffix.
   317	    (r"\*\*/", r"(.*[/\\\\])?"),    # **/ matches any subdirs, including none
   318	    (r"/", r"[/\\\\]"),             # / matches either slash or backslash
   319	    (r"\*", r"[^/\\\\]*"),          # * matches any number of non slash-likes
   320	    (r"\?", r"[^/\\\\]"),           # ? matches one non slash-like
   321	    (r"\[.*?\]", r"\g<0>"),         # [a-f] matches [a-f]
   322	    (r"[a-zA-Z0-9_-]+", r"\g<0>"),  # word chars match themselves
   323	    (r"[\[\]]", None),              # Can't have single square brackets
   324	    (r".", r"\\\g<0>"),             # Anything else is escaped to be safe
   325	]]

   327	def _glob_to_regex(pattern: str) -> str:
   328	    """Convert a file-path glob pattern into a regex."""
   329	    # Turn all backslashes into slashes to simplify the tokenizer.
   330	    pattern = pattern.replace("\\", "/")
   331	    if "/" not in pattern:
   332	        pattern = f"**/{pattern}"
   333	    path_rx = []
   334	    pos = 0
   335	    while pos < len(pattern):
   336	        for rx, sub in G2RX_TOKENS:                     # pragma: always breaks
   337	            if m := rx.match(pattern, pos=pos):
   338	                if sub is None:
   339	                    raise ConfigError(f"File pattern can't include {m[0]!r}")
   340	                path_rx.append(m.expand(sub))
   341	                pos = m.end()
   342	                break
   343	    return "".join(path_rx)

   346	def globs_to_regex(
   347	    patterns: Iterable[str],
   348	    case_insensitive: bool = False,
   349	    partial: bool = False,
   350	) -> re.Pattern[str]:
   351	    """Convert glob patterns to a compiled regex that matches any of them.
   352
   353	    Slashes are always converted to match either slash or backslash, for
   354	    Windows support, even when running elsewhere.
   355
   356	    If the pattern has no slash or backslash, then it is interpreted as
   357	    matching a file name anywhere it appears in the tree.  Otherwise, the glob
   358	    pattern must match the whole file path.
   359
   360	    If `partial` is true, then the pattern will match if the target string
   361	    starts with the pattern. Otherwise, it must match the entire string.
   362
   363	    Returns: a compiled regex object.  Use the .match method to compare target
   364	    strings.
   365
   366	    """
   367	    flags = 0
   368	    if case_insensitive:
   369	        flags |= re.IGNORECASE
   370	    rx = join_regex(map(_glob_to_regex, patterns))
   371	    if not partial:
   372	        rx = fr"(?:{rx})\Z"
   373	    compiled = re.compile(rx, flags=flags)
   374	    return compiled

   377	class PathAliases:
   378	    """A collection of aliases for paths.
   379
   380	    When combining data files from remote machines, often the paths to source
   381	    code are different, for example, due to OS differences, or because of
   382	    serialized checkouts on continuous integration machines.
   383
   384	    A `PathAliases` object tracks a list of pattern/result pairs, and can
   385	    map a path through those aliases to produce a unified path.
   386
   387	    """
   388	    def __init__(
   389	        self,
   390	        debugfn: Callable[[str], None] | None = None,
   391	        relative: bool = False,
   392	    ) -> None:
   393	        # A list of (original_pattern, regex, result)
   394	        self.aliases: list[tuple[str, re.Pattern[str], str]] = []
   395	        self.debugfn = debugfn or (lambda msg: 0)
   396	        self.relative = relative
   397	        self.pprinted = False
   398
   399	    def pprint(self) -> None:
   400	        """Dump the important parts of the PathAliases, for debugging."""
   401	        self.debugfn(f"Aliases (relative={self.relative}):")
   402	        for original_pattern, regex, result in self.aliases:
   403	            self.debugfn(f" Rule: {original_pattern!r} -> {result!r} using regex {regex.pattern!r}")
   404
   405	    def add(self, pattern: str, result: str) -> None:
   406	        """Add the `pattern`/`result` pair to the list of aliases.
   407
   408	        `pattern` is an `glob`-style pattern.  `result` is a simple
   409	        string.  When mapping paths, if a path starts with a match against
   410	        `pattern`, then that match is replaced with `result`.  This models
   411	        isomorphic source trees being rooted at different places on two
   412	        different machines.
   413
   414	        `pattern` can't end with a wildcard component, since that would
   415	        match an entire tree, and not just its root.
   416
   417	        """
   418	        original_pattern = pattern
   419	        pattern_sep = sep(pattern)
   420
   421	        if len(pattern) > 1:
   422	            pattern = pattern.rstrip(r"\/")
   423
   424	        # The pattern can't end with a wildcard component.
   425	        if pattern.endswith("*"):
   426	            raise ConfigError("Pattern must not end with wildcards.")
   427
   428	        # The pattern is meant to match a file path.  Let's make it absolute
   429	        # unless it already is, or is meant to match any prefix.
   430	        if not self.relative:
   431	            if not pattern.startswith("*") and not isabs_anywhere(pattern + pattern_sep):
   432	                pattern = abs_file(pattern)
   433	        if not pattern.endswith(pattern_sep):
   434	            pattern += pattern_sep
   435
   436	        # Make a regex from the pattern.
   437	        regex = globs_to_regex([pattern], case_insensitive=True, partial=True)
   438
   439	        # Normalize the result: it must end with a path separator.
   440	        result_sep = sep(result)
   441	        result = result.rstrip(r"\/") + result_sep
   442	        self.aliases.append((original_pattern, regex, result))
   443
   444	    def map(self, path: str, exists:Callable[[str], bool] = source_exists) -> str:
   445	        """Map `path` through the aliases.
   446
   447	        `path` is checked against all of the patterns.  The first pattern to
   448	        match is used to replace the root of the path with the result root.
   449	        Only one pattern is ever used.  If no patterns match, `path` is
   450	        returned unchanged.
   451
   452	        The separator style in the result is made to match that of the result
   453	        in the alias.
   454
   455	        `exists` is a function to determine if the resulting path actually
   456	        exists.
   457
   458	        Returns the mapped path.  If a mapping has happened, this is a
   459	        canonical path.  If no mapping has happened, it is the original value
   460	        of `path` unchanged.
   461
   462	        """
   463	        if not self.pprinted:
   464	            self.pprint()
   465	            self.pprinted = True
   466
   467	        for original_pattern, regex, result in self.aliases:
   468	            if m := regex.match(path):
   469	                new = path.replace(m[0], result)
   470	                new = new.replace(sep(path), sep(result))
   471	                if not self.relative:
   472	                    new = canonical_filename(new)
   473	                dot_start = result.startswith(("./", ".\\")) and len(result) > 2
   474	                if new.startswith(("./", ".\\")) and not dot_start:
   475	                    new = new[2:]
   476	                if not exists(new):
   477	                    self.debugfn(
   478	                        f"Rule {original_pattern!r} changed {path!r} to {new!r} " +
   479	                        "which doesn't exist, continuing",
   480	                    )
   481	                    continue
   482	                self.debugfn(
   483	                    f"Matched path {path!r} to rule {original_pattern!r} -> {result!r}, " +
   484	                    f"producing {new!r}",
   485	                )
   486	                return new
   487
   488	        # If we get here, no pattern matched.
   489
   490	        if self.relative:
   491	            path = relative_filename(path)
   492
   493	        if self.relative and not isabs_anywhere(path):
   494	            # Auto-generate a pattern to implicitly match relative files
   495	            parts = re.split(r"[/\\]", path)
   496	            if len(parts) > 1:
   497	                dir1 = parts[0]
   498	                pattern = f"*/{dir1}"
   499	                regex_pat = fr"^(.*[\\/])?{re.escape(dir1)}[\\/]"
   500	                result = f"{dir1}{os.sep}"
   501	                # Only add a new pattern if we don't already have this pattern.
   502	                if not any(p == pattern for p, _, _ in self.aliases):
   503	                    self.debugfn(
   504	                        f"Generating rule: {pattern!r} -> {result!r} using regex {regex_pat!r}",
   505	                    )
   506	                    self.aliases.append((pattern, re.compile(regex_pat), result))
   507	                    return self.map(path, exists=exists)
   508
   509	        self.debugfn(f"No rules match, path {path!r} is unchanged")
   510	        return path

   514	    """Yield all of the importable Python files in `dirname`, recursively.
   515
   516	    To be importable, the files have to be in a directory with a __init__.py,
   517	    except for `dirname` itself, which isn't required to have one.  The

   530	        if not include_namespace_packages:

   532	                # If a directory doesn't have __init__.py, then it isn't
   533	                # importable and neither are its files

   537	            # We're only interested in files that look like reasonable Python
   538	            # files: Must end with .py or .pyw, and must not have certain funny
   539	            # characters that probably mean they are editor junk.
   540	            if re.match(r"^[^.#~!$@%^&*()+=,]+\.pyw?$", filename):

   131	    def data_for_file(self, fr: FileReporter, analysis: Analysis) -> FileData:
   132	        """Produce the data needed for one file's report."""
   133	        if self.has_arcs:
   134	            missing_branch_arcs = analysis.missing_branch_arcs()
   135	            arcs_executed = analysis.arcs_executed
   136	        else:
   137	            missing_branch_arcs = {}
   138	            arcs_executed = []
   139
   140	        if self.config.show_contexts:
   141	            contexts_by_lineno = self.data.contexts_by_lineno(analysis.filename)
   142
   143	        lines = []
   144	        branch_stats = analysis.branch_stats()
   145	        multiline_map = {}
   146	        if hasattr(fr, "multiline_map"):
   147	            multiline_map = fr.multiline_map()
   148
   149	        for lineno, tokens in enumerate(fr.source_token_lines(), start=1):
   150	            # Figure out how to mark this line.
   151	            category = category2 = ""
   152	            short_annotations = []
   153	            long_annotations = []
   154
   155	            if lineno in analysis.excluded:
   156	                category = "exc"
   157	            elif lineno in analysis.missing:
   158	                category = "mis"
   159	            elif self.has_arcs and lineno in missing_branch_arcs:
   160	                category = "par"
   161	                mba = missing_branch_arcs[lineno]
   162	                if len(mba) == branch_stats[lineno][0]:
   163	                    # None of the branches were taken from this line.
   164	                    short_annotations.append("anywhere")
   165	                    long_annotations.append(
   166	                        f"line {lineno} didn't jump anywhere: it always raised an exception."
   167	                    )
   168	                else:
   169	                    for b in missing_branch_arcs[lineno]:
   170	                        if b < 0:
   171	                            short_annotations.append("exit")
   172	                        else:
   173	                            short_annotations.append(str(b))
   174	                        long_annotations.append(
   175	                            fr.missing_arc_description(lineno, b, arcs_executed)
   176	                        )
   177	            elif lineno in analysis.statements:
   178	                category = "run"
   179	            elif first_line := multiline_map.get(lineno):
   180	                if first_line in analysis.excluded:
   181	                    category2 = "exc2"
   182	                elif first_line in analysis.missing:
   183	                    category2 = "mis2"
   184	                elif self.has_arcs and first_line in missing_branch_arcs:
   185	                    category2 = "par2"
   186	                # I don't understand why this last condition is marked as
   187	                # partial.  If I add an else with an exception, the exception
   188	                # is raised.
   189	                elif first_line in analysis.statements:     # pragma: part covered
   190	                    category2 = "run2"
   191
   192	            contexts = []
   193	            contexts_label = ""
   194	            context_list = []
   195	            if category and self.config.show_contexts:
   196	                contexts = human_sorted(c or self.EMPTY for c in contexts_by_lineno.get(lineno, ()))
   197	                if contexts == [self.EMPTY]:
   198	                    contexts_label = self.EMPTY
   199	                else:
   200	                    contexts_label = f"{len(contexts)} ctx"
   201	                    context_list = contexts
   202
   203	            lines.append(LineData(
   204	                tokens=tokens,
   205	                number=lineno,
   206	                category=category or category2,
   207	                contexts=contexts,
   208	                contexts_label=contexts_label,
   209	                context_list=context_list,
   210	                short_annotations=short_annotations,
   211	                long_annotations=long_annotations,
   212	            ))
   213
   214	        file_data = FileData(
   215	            relative_filename=fr.relative_filename(),
   216	            nums=analysis.numbers,
   217	            lines=lines,
   218	        )
   219
   220	        return file_data

   233	HTML_SAFE = string.ascii_letters + string.digits + "!#$%'()*+,-./:;=?@[]^_`{|}~"

   235	@functools.cache
   236	def encode_int(n: int) -> str:
   237	    """Create a short HTML-safe string from an integer, using HTML_SAFE."""
   238	    if n == 0:
   239	        return HTML_SAFE[0]
   240
   241	    r = []
   242	    while n:
   243	        n, t = divmod(n, len(HTML_SAFE))
   244	        r.append(HTML_SAFE[t])
   245	    return "".join(r)

   248	def copy_with_cache_bust(src: str, dest_dir: str) -> str:
   249	    """Copy `src` to `dest_dir`, adding a hash to the name.
   250
   251	    Returns the updated destination file name with hash.
   252	    """
   253	    with open(src, "rb") as f:
   254	        text = f.read()
   255	    h = Hasher()
   256	    h.update(text)
   257	    cache_bust = h.hexdigest()[:8]
   258	    src_base = os.path.basename(src)
   259	    dest = src_base.replace(".", f"_cb_{cache_bust}.")
   260	    with open(os.path.join(dest_dir, dest), "wb") as f:
   261	        f.write(text)
   262	    return dest

   350	    def report(self, morfs: Iterable[TMorf] | None) -> float:
   351	        """Generate an HTML report for `morfs`.
   352
   353	        `morfs` is a list of modules or file names.
   354
   355	        """
   356	        # Read the status data and check that this run used the same
   357	        # global data as the last run.
   358	        self.incr.read()
   359	        self.incr.check_global_data(self.config, self.pyfile_html_source)
   360
   361	        # Process all the files. For each page we need to supply a link
   362	        # to the next and previous page.
   363	        files_to_report = []
   364
   365	        have_data = False
   366	        for fr, analysis in get_analysis_to_report(self.coverage, morfs):
   367	            have_data = True
   368	            ftr = FileToReport(fr, analysis)
   369	            if self.should_report(analysis, self.index_pages["file"]):
   370	                files_to_report.append(ftr)
   371	            else:
   372	                file_be_gone(os.path.join(self.directory, ftr.html_filename))
   373
   374	        if not have_data:
   375	            raise NoDataError("No data to report.")
   376
   377	        self.make_directory()
   378	        self.make_local_static_report_files()
   379
   380	        if files_to_report:
   381	            for ftr1, ftr2 in zip(files_to_report[:-1], files_to_report[1:]):
   382	                ftr1.next_html = ftr2.html_filename
   383	                ftr2.prev_html = ftr1.html_filename
   384	            files_to_report[0].prev_html = "index.html"
   385	            files_to_report[-1].next_html = "index.html"
   386
   387	        for ftr in files_to_report:
   388	            self.write_html_page(ftr)
   389	            for noun, plural_noun in ftr.fr.code_region_kinds():
   390	                if noun not in self.index_pages:
   391	                    self.index_pages[noun] = self.new_index_page(noun, plural_noun)
   392
   393	        # Write the index page.
   394	        if files_to_report:
   395	            first_html = files_to_report[0].html_filename
   396	            final_html = files_to_report[-1].html_filename
   397	        else:
   398	            first_html = final_html = "index.html"
   399	        self.write_file_index_page(first_html, final_html)
   400
   401	        # Write function and class index pages.
   402	        self.write_region_index_pages(files_to_report)
   403
   404	        return (
   405	            self.index_pages["file"].totals.n_statements
   406	            and self.index_pages["file"].totals.pc_covered
   407	        )

   441	    def should_report(self, analysis: Analysis, index_page: IndexPage) -> bool:
   442	        """Determine if we'll report this file or region."""
   443	        # Get the numbers for this file.
   444	        nums = analysis.numbers
   445	        index_page.totals += nums
   446
   447	        if self.skip_covered:
   448	            # Don't report on 100% files.
   449	            no_missing_lines = (nums.n_missing == 0)
   450	            no_missing_branches = (nums.n_partial_branches == 0)
   451	            if no_missing_lines and no_missing_branches:
   452	                index_page.skipped_covered_count += 1
   453	                return False
   454
   455	        if self.skip_empty:
   456	            # Don't report on empty files.
   457	            if nums.n_statements == 0:
   458	                index_page.skipped_empty_count += 1
   459	                return False
   460
   461	        return True

   463	    def write_html_page(self, ftr: FileToReport) -> None:
   464	        """Generate an HTML page for one source file.
   465
   466	        If the page on disk is already correct based on our incremental status
   467	        checking, then the page doesn't have to be generated, and this function
   468	        only does page summary bookkeeping.
   469
   470	        """
   471	        # Find out if the page on disk is already correct.
   472	        if self.incr.can_skip_file(self.data, ftr.fr, ftr.rootname):
   473	            self.index_pages["file"].summaries.append(self.incr.index_info(ftr.rootname))
   474	            return
   475
   476	        # Write the HTML page for this source file.
   477	        file_data = self.datagen.data_for_file(ftr.fr, ftr.analysis)
   478
   479	        contexts = collections.Counter(c for cline in file_data.lines for c in cline.contexts)
   480	        context_codes = {y: i for (i, y) in enumerate(x[0] for x in contexts.most_common())}
   481	        if context_codes:
   482	            contexts_json = json.dumps(
   483	                {encode_int(v): k for (k, v) in context_codes.items()},
   484	                indent=2,
   485	            )
   486	        else:
   487	            contexts_json = None
   488
   489	        for ldata in file_data.lines:
   490	            # Build the HTML for the line.
   491	            html_parts = []
   492	            for tok_type, tok_text in ldata.tokens:
   493	                if tok_type == "ws":
   494	                    html_parts.append(escape(tok_text))
   495	                else:
   496	                    tok_html = escape(tok_text) or "&nbsp;"
   497	                    html_parts.append(f'<span class="{tok_type}">{tok_html}</span>')
   498	            ldata.html = "".join(html_parts)
   499	            if ldata.context_list:
   500	                encoded_contexts = [
   501	                    encode_int(context_codes[c_context]) for c_context in ldata.context_list
   502	                ]
   503	                code_width = max(len(ec) for ec in encoded_contexts)
   504	                ldata.context_str = (
   505	                    str(code_width)
   506	                    + "".join(ec.ljust(code_width) for ec in encoded_contexts)
   507	                )
   508	            else:
   509	                ldata.context_str = ""
   510
   511	            if ldata.short_annotations:
   512	                # 202F is NARROW NO-BREAK SPACE.
   513	                # 219B is RIGHTWARDS ARROW WITH STROKE.
   514	                ldata.annotate = ",&nbsp;&nbsp; ".join(
   515	                    f"{ldata.number}&#x202F;&#x219B;&#x202F;{d}"
   516	                    for d in ldata.short_annotations
   517	                )
   518	            else:
   519	                ldata.annotate = None
   520
   521	            if ldata.long_annotations:
   522	                longs = ldata.long_annotations
   523	                # A line can only have two branch destinations. If there were
   524	                # two missing, we would have written one as "always raised."
   525	                assert len(longs) == 1, (
   526	                    f"Had long annotations in {ftr.fr.relative_filename()}: {longs}"
   527	                )
   528	                ldata.annotate_long = longs[0]
   529	            else:
   530	                ldata.annotate_long = None
   531
   532	            css_classes = []
   533	            if ldata.category:
   534	                css_classes.append(
   535	                    self.template_globals["category"][ldata.category],   # type: ignore[index]
   536	                )
   537	            ldata.css_class = " ".join(css_classes) or "pln"
   538
   539	        html_path = os.path.join(self.directory, ftr.html_filename)
   540	        html = self.source_tmpl.render({
   541	            **file_data.__dict__,
   542	            "contexts_json": contexts_json,
   543	            "prev_html": ftr.prev_html,
   544	            "next_html": ftr.next_html,
   545	        })
   546	        write_html(html_path, html)
   547
   548	        # Save this file's information for the index page.
   549	        index_info = IndexItem(
   550	            url = ftr.html_filename,
   551	            file = escape(ftr.fr.relative_filename()),
   552	            nums = ftr.analysis.numbers,
   553	        )
   554	        self.index_pages["file"].summaries.append(index_info)
   555	        self.incr.set_index_info(ftr.rootname, index_info)

   571	    def write_region_index_pages(self, files_to_report: Iterable[FileToReport]) -> None:
   572	        """Write the other index pages for this report."""
   573	        for ftr in files_to_report:
   574	            region_nouns = [pair[0] for pair in ftr.fr.code_region_kinds()]
   575	            num_lines = len(ftr.fr.source().splitlines())
   576	            regions = ftr.fr.code_regions()
   577
   578	            for noun in region_nouns:
   579	                page_data = self.index_pages[noun]
   580	                outside_lines = set(range(1, num_lines + 1))
   581
   582	                for region in regions:
   583	                    if region.kind != noun:
   584	                        continue
   585	                    outside_lines -= region.lines
   586	                    analysis = ftr.analysis.narrow(region.lines)
   587	                    if not self.should_report(analysis, page_data):
   588	                        continue
   589	                    sorting_name = region.name.rpartition(".")[-1].lstrip("_")
   590	                    page_data.summaries.append(IndexItem(
   591	                        url=f"{ftr.html_filename}#t{region.start}",
   592	                        file=escape(ftr.fr.relative_filename()),
   593	                        description=(
   594	                            f"<data value='{escape(sorting_name)}'>"
   595	                            + escape(region.name)
   596	                            + "</data>"
   597	                        ),
   598	                        nums=analysis.numbers,

   666	class IncrementalChecker:
   667	    """Logic and data to support incremental reporting.
   668
   669	    When generating an HTML report, often only a few of the source files have
   670	    changed since the last time we made the HTML report.  This means previously
   671	    created HTML pages can be reused without generating them again, speeding
   672	    the command.
   673
   674	    This class manages a JSON data file that captures enough information to
   675	    know whether an HTML page for a .py file needs to be regenerated or not.
   676	    The data file also needs to store all the information needed to create the
   677	    entry for the file on the index page so that if the HTML page is reused,
   678	    the index page can still be created to refer to it.
   679
   680	    The data looks like::
   681
   682	        {
   683	            "note": "This file is an internal implementation detail ...",
   684	            // A fixed number indicating the data format.  STATUS_FORMAT
   685	            "format": 5,
   686	            // The version of coverage.py
   687	            "version": "7.4.4",
   688	            // A hash of a number of global things, including the configuration
   689	            // settings and the pyfile.html template itself.
   690	            "globals": "540ee119c15d52a68a53fe6f0897346d",
   691	            "files": {
   692	                // An entry for each source file keyed by the flat_rootname().
   693	                "z_7b071bdc2a35fa80___init___py": {
   694	                    // Hash of the source, the text of the .py file.
   695	                    "hash": "e45581a5b48f879f301c0f30bf77a50c",
   696	                    // Information for the index.html file.
   697	                    "index": {
   698	                        "url": "z_7b071bdc2a35fa80___init___py.html",
   699	                        "file": "cogapp/__init__.py",
   700	                        "description": "",
   701	                        // The Numbers for this file.
   702	                        "nums": { "precision": 2, "n_files": 1, "n_statements": 43, ... }
   703	                    }
   704	                },
   705	                ...
   706	            }
   707	        }
   708
   709	    """

   711	    STATUS_FILE = "status.json"
   712	    STATUS_FORMAT = 5
   713	    NOTE = (
   714	        "This file is an internal implementation detail to speed up HTML report"
   715	        + " generation. Its format can change at any time. You might be looking"
   716	        + " for the JSON report: https://coverage.rtfd.io/cmd.html#cmd-json"
   717	    )

   719	    def __init__(self, directory: str) -> None:
   720	        self.directory = directory
   721	        self._reset()

   723	    def _reset(self) -> None:
   724	        """Initialize to empty. Causes all files to be reported."""
   725	        self.globals = ""
   726	        self.files: dict[str, FileInfo] = {}

   728	    def read(self) -> None:
   729	        """Read the information we stored last time."""
   730	        try:
   731	            status_file = os.path.join(self.directory, self.STATUS_FILE)
   732	            with open(status_file, encoding="utf-8") as fstatus:
   733	                status = json.load(fstatus)
   734	        except (OSError, ValueError):
   735	            # Status file is missing or malformed.
   736	            usable = False
   737	        else:
   738	            if status["format"] != self.STATUS_FORMAT:
   739	                usable = False
   740	            elif status["version"] != coverage.__version__:
   741	                usable = False
   742	            else:
   743	                usable = True
   744
   745	        if usable:
   746	            self.files = {}
   747	            for filename, filedict in status["files"].items():
   748	                indexdict = filedict["index"]
   749	                index_item = IndexItem(**indexdict)
   750	                index_item.nums = Numbers(**indexdict["nums"])
   751	                fileinfo = FileInfo(
   752	                    hash=filedict["hash"],
   753	                    index=index_item,
   754	                )
   755	                self.files[filename] = fileinfo
   756	            self.globals = status["globals"]
   757	        else:
   758	            self._reset()

   760	    def write(self) -> None:
   761	        """Write the current status."""
   762	        status_file = os.path.join(self.directory, self.STATUS_FILE)
   763	        status_data = {
   764	            "note": self.NOTE,
   765	            "format": self.STATUS_FORMAT,
   766	            "version": coverage.__version__,
   767	            "globals": self.globals,
   768	            "files": {
   769	                fname: dataclasses.asdict(finfo)
   770	                for fname, finfo in self.files.items()
   771	            },
   772	        }
   773	        with open(status_file, "w", encoding="utf-8") as fout:
   774	            json.dump(status_data, fout, separators=(",", ":"))

   776	    def check_global_data(self, *data: Any) -> None:
   777	        """Check the global data that can affect incremental reporting.
   778
   779	        Pass in whatever global information could affect the content of the
   780	        HTML pages.  If the global data has changed since last time, this will
   781	        clear the data so that all files are regenerated.
   782
   783	        """
   784	        m = Hasher()
   785	        for d in data:
   786	            m.update(d)
   787	        these_globals = m.hexdigest()
   788	        if self.globals != these_globals:
   789	            self._reset()
   790	            self.globals = these_globals

   792	    def can_skip_file(self, data: CoverageData, fr: FileReporter, rootname: str) -> bool:
   793	        """Can we skip reporting this file?
   794
   795	        `data` is a CoverageData object, `fr` is a `FileReporter`, and
   796	        `rootname` is the name being used for the file.
   797
   798	        Returns True if the HTML page is fine as-is, False if we need to recreate
   799	        the HTML page.
   800
   801	        """
   802	        m = Hasher()
   803	        m.update(fr.source().encode("utf-8"))
   804	        add_data_to_hash(data, fr.filename, m)
   805	        this_hash = m.hexdigest()

   807	        file_info = self.files.setdefault(rootname, FileInfo())

   809	        if this_hash == file_info.hash:
   810	            # Nothing has changed to require the file to be reported again.
   811	            return True
   812	        else:
   813	            # File has changed, record the latest hash and force regeneration.
   814	            file_info.hash = this_hash
   815	            return False
