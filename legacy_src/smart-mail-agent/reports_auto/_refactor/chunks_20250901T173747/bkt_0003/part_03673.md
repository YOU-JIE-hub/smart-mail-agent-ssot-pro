  2284	    def get_arg_infer_passes(
  2285	        self,
  2286	        callee: CallableType,
  2287	        args: list[Expression],
  2288	        arg_types: list[Type],
  2289	        formal_to_actual: list[list[int]],
  2290	        num_actuals: int,
  2291	    ) -> list[int]:
  2292	        """Return pass numbers for args for two-pass argument type inference.
  2293
  2294	        For each actual, the pass number is either 1 (first pass) or 2 (second
  2295	        pass).
  2296
  2297	        Two-pass argument type inference primarily lets us infer types of
  2298	        lambdas more effectively.
  2299	        """
  2300	        res = [1] * num_actuals
  2301	        for i, arg in enumerate(callee.arg_types):
  2302	            skip_param_spec = False
  2303	            p_formal = get_proper_type(callee.arg_types[i])
  2304	            if isinstance(p_formal, CallableType) and p_formal.param_spec():
  2305	                for j in formal_to_actual[i]:
  2306	                    p_actual = get_proper_type(arg_types[j])
  2307	                    # This is an exception from the usual logic where we put generic Callable
  2308	                    # arguments in the second pass. If we have a non-generic actual, it is
  2309	                    # likely to infer good constraints, for example if we have:
  2310	                    #   def run(Callable[P, None], *args: P.args, **kwargs: P.kwargs) -> None: ...
  2311	                    #   def test(x: int, y: int) -> int: ...
  2312	                    #   run(test, 1, 2)
  2313	                    # we will use `test` for inference, since it will allow to infer also
  2314	                    # argument *names* for P <: [x: int, y: int].
  2315	                    if isinstance(p_actual, Instance):
  2316	                        call_method = find_member("__call__", p_actual, p_actual, is_operator=True)
  2317	                        if call_method is not None:
  2318	                            p_actual = get_proper_type(call_method)
  2319	                    if (
  2320	                        isinstance(p_actual, CallableType)
  2321	                        and not p_actual.variables
  2322	                        and not isinstance(args[j], LambdaExpr)
  2323	                    ):
  2324	                        skip_param_spec = True
  2325	                        break
  2326	            if not skip_param_spec and arg.accept(ArgInferSecondPassQuery()):
  2327	                for j in formal_to_actual[i]:
  2328	                    res[j] = 2
  2329	        return res

  2351	    def check_argument_count(
  2352	        self,
  2353	        callee: CallableType,

  2356	        actual_names: Sequence[str | None] | None,
  2357	        formal_to_actual: list[list[int]],
  2358	        context: Context | None,
  2359	        object_type: Type | None = None,
  2360	        callable_name: str | None = None,

  2362	        """Check that there is a value for all required arguments to a function.
  2363
  2364	        Also check that there are no duplicate values for arguments. Report found errors
  2365	        using 'messages' if it's not None. If 'messages' is given, 'context' must also be given.
  2366
  2367	        Return False if there were any errors. Otherwise return True
  2368	        """
  2369	        if context is None:
  2370	            # Avoid "is None" checks
  2371	            context = TempNode(AnyType(TypeOfAny.special_form))
  2372
  2373	        # TODO(jukka): We could return as soon as we find an error if messages is None.
  2374
  2375	        # Collect dict of all actual arguments matched to formal arguments, with occurrence count
  2376	        all_actuals: dict[int, int] = {}
  2377	        for actuals in formal_to_actual:
  2378	            for a in actuals:
  2379	                all_actuals[a] = all_actuals.get(a, 0) + 1
  2380
  2381	        ok, is_unexpected_arg_error = self.check_for_extra_actual_arguments(
  2382	            callee, actual_types, actual_kinds, actual_names, all_actuals, context
  2383	        )
  2384
  2385	        # Check for too many or few values for formals.
  2386	        for i, kind in enumerate(callee.arg_kinds):
  2387	            mapped_args = formal_to_actual[i]
  2388	            if kind.is_required() and not mapped_args and not is_unexpected_arg_error:
  2389	                # No actual for a mandatory formal
  2390	                if kind.is_positional():

  2392	                    if object_type and callable_name and "." in callable_name:

  2394	                else:
  2395	                    argname = callee.arg_names[i] or "?"
  2396	                    self.msg.missing_named_argument(callee, context, argname)
  2397	                ok = False
  2398	            elif not kind.is_star() and is_duplicate_mapping(
  2399	                mapped_args, actual_types, actual_kinds
  2400	            ):
  2401	                if self.chk.in_checked_function() or isinstance(
  2402	                    get_proper_type(actual_types[mapped_args[0]]), TupleType
  2403	                ):
  2404	                    self.msg.duplicate_argument_value(callee, i, context)
  2405	                    ok = False
  2406	            elif (
  2407	                kind.is_named()

  2409	                and actual_kinds[mapped_args[0]] not in [nodes.ARG_NAMED, nodes.ARG_STAR2]

  2412	                self.msg.too_many_positional_arguments(callee, context)

  2414	            elif callee.param_spec() is not None:
  2415	                if not mapped_args and callee.special_sig != "partial":
  2416	                    self.msg.too_few_arguments(callee, context, actual_names)
  2417	                    ok = False
  2418	                elif len(mapped_args) > 1:
  2419	                    paramspec_entries = sum(
  2420	                        isinstance(get_proper_type(actual_types[k]), ParamSpecType)
  2421	                        for k in mapped_args
  2422	                    )
  2423	                    if actual_kinds[mapped_args[0]] == nodes.ARG_STAR and paramspec_entries > 1:
  2424	                        self.msg.fail("ParamSpec.args should only be passed once", context)
  2425	                        ok = False
  2426	                    if actual_kinds[mapped_args[0]] == nodes.ARG_STAR2 and paramspec_entries > 1:

  2431	    def check_for_extra_actual_arguments(
  2432	        self,
  2433	        callee: CallableType,
  2434	        actual_types: list[Type],
  2435	        actual_kinds: list[ArgKind],
  2436	        actual_names: Sequence[str | None] | None,
  2437	        all_actuals: dict[int, int],
  2438	        context: Context,
  2439	    ) -> tuple[bool, bool]:
  2440	        """Check for extra actual arguments.
  2441
  2442	        Return tuple (was everything ok,
  2443	                      was there an extra keyword argument error [used to avoid duplicate errors]).
  2444	        """
  2445
  2446	        is_unexpected_arg_error = False  # Keep track of errors to avoid duplicate errors
  2447	        ok = True  # False if we've found any error
  2448
  2449	        for i, kind in enumerate(actual_kinds):
  2450	            if (
  2451	                i not in all_actuals
  2452	                and
  2453	                # We accept the other iterables than tuple (including Any)
  2454	                # as star arguments because they could be empty, resulting no arguments.
  2455	                (kind != nodes.ARG_STAR or is_non_empty_tuple(actual_types[i]))
  2456	                and
  2457	                # Accept all types for double-starred arguments, because they could be empty
  2458	                # dictionaries and we can't tell it from their types
  2459	                kind != nodes.ARG_STAR2
  2460	            ):
  2461	                # Extra actual: not matched by a formal argument.
  2462	                ok = False
  2463	                if kind != nodes.ARG_NAMED:
  2464	                    self.msg.too_many_arguments(callee, context)
  2465	                else:
  2466	                    assert actual_names, "Internal error: named kinds without names given"
  2467	                    act_name = actual_names[i]
  2468	                    assert act_name is not None
  2469	                    act_type = actual_types[i]
  2470	                    self.msg.unexpected_keyword_argument(callee, act_name, act_type, context)
  2471	                    is_unexpected_arg_error = True
  2472	            elif (
  2473	                kind == nodes.ARG_STAR and nodes.ARG_STAR not in callee.arg_kinds
  2474	            ) or kind == nodes.ARG_STAR2:
  2475	                actual_type = get_proper_type(actual_types[i])
  2476	                if isinstance(actual_type, (TupleType, TypedDictType)):
  2477	                    if all_actuals.get(i, 0) < len(actual_type.items):
  2478	                        # Too many tuple/dict items as some did not match.
  2479	                        if kind != nodes.ARG_STAR2 or not isinstance(actual_type, TypedDictType):
  2480	                            self.msg.too_many_arguments(callee, context)
  2481	                        else:
  2482	                            self.msg.too_many_arguments_from_typed_dict(
  2483	                                callee, actual_type, context
  2484	                            )
  2485	                            is_unexpected_arg_error = True
  2486	                        ok = False
  2487	                # *args/**kwargs can be applied even if the function takes a fixed
  2488	                # number of positional arguments. This may succeed at runtime.
  2489
  2490	        return ok, is_unexpected_arg_error

  2518

  2520
  2521	        The check_call docstring describes some of the arguments.

  2524	        # Keep track of consumed tuple *arg items.
  2525	        mapper = ArgTypeExpander(self.argument_infer_context())
  2526

  2528	            arg_type = get_proper_type(arg_type)

  2530	                self.msg.invalid_var_arg(arg_type, context)

  2532	                is_mapping = is_subtype(

  2538	            orig_callee_arg_type = get_proper_type(callee.arg_types[i])

  2541	            # is an unpack, so this would be something like:
  2542	            # [Tuple[Unpack[Ts]], int]

  2545	            # the suffices to the tuple, e.g. a single actual like
  2546	            # Tuple[Unpack[Ts], int]

  2552	                    isinstance(p_actual_type, TupleType)
  2553	                    and len(p_actual_type.items) == 1
  2554	                    and isinstance(p_actual_type.items[0], UnpackType)

  2560	                        if isinstance(p_callee_type, TupleType):
  2561	                            assert p_callee_type.items
  2562	                            callee_arg_types = p_callee_type.items

  2564	                                len(p_callee_type.items) - 1
  2565	                            )
  2566	                            expanded_tuple = True
  2567
  2568	            if not expanded_tuple:
  2569	                actual_types = [arg_types[a] for a in actuals]
  2570	                if isinstance(orig_callee_arg_type, UnpackType):

  2585	                                callee_arg_kinds = [

  2588	                                ]

  2591	                                assert isinstance(inner_unpacked_type, Instance)
  2592	                                assert inner_unpacked_type.type.fullname == "builtins.tuple"

  2594	                                    unpacked_type.items[:inner_unpack_index]

  2597	                                    + unpacked_type.items[inner_unpack_index + 1 :]

  2599	                                callee_arg_kinds = [ARG_POS] * len(actuals)
  2600	                    elif isinstance(unpacked_type, TypeVarTupleType):

  2602	                        callee_arg_kinds = [ARG_STAR]
  2603	                    else:

  2605	                        assert unpacked_type.type.fullname == "builtins.tuple"
  2606	                        callee_arg_types = [unpacked_type.args[0]] * len(actuals)
  2607	                        callee_arg_kinds = [ARG_POS] * len(actuals)

  2689	    def check_overload_call(
  2690	        self,
  2691	        callee: Overloaded,
  2692	        args: list[Expression],
  2693	        arg_kinds: list[ArgKind],
  2694	        arg_names: Sequence[str | None] | None,
  2695	        callable_name: str | None,
  2696	        object_type: Type | None,
  2697	        context: Context,
  2698	    ) -> tuple[Type, Type]:
  2699	        """Checks a call to an overloaded function."""
  2700	        # Normalize unpacked kwargs before checking the call.
  2701	        callee = callee.with_unpacked_kwargs()
  2702	        arg_types = self.infer_arg_types_in_empty_context(args)
  2703	        # Step 1: Filter call targets to remove ones where the argument counts don't match
  2704	        plausible_targets = self.plausible_overload_call_targets(
  2705	            arg_types, arg_kinds, arg_names, callee
  2706	        )
  2707
  2708	        # Step 2: If the arguments contain a union, we try performing union math first,
  2709	        #         instead of picking the first matching overload.
  2710	        #         This is because picking the first overload often ends up being too greedy:
  2711	        #         for example, when we have a fallback alternative that accepts an unrestricted
  2712	        #         typevar. See https://github.com/python/mypy/issues/4063 for related discussion.
  2713	        erased_targets: list[CallableType] | None = None
  2714	        unioned_result: tuple[Type, Type] | None = None
  2715
  2716	        # Determine whether we need to encourage union math. This should be generally safe,
  2717	        # as union math infers better results in the vast majority of cases, but it is very
  2718	        # computationally intensive.
  2719	        none_type_var_overlap = self.possible_none_type_var_overlap(arg_types, plausible_targets)
  2720	        union_interrupted = False  # did we try all union combinations?
  2721	        if any(self.real_union(arg) for arg in arg_types):
  2722	            try:
  2723	                with self.msg.filter_errors():
  2724	                    unioned_return = self.union_overload_result(
  2725	                        plausible_targets,
  2726	                        args,
  2727	                        arg_types,
  2728	                        arg_kinds,
  2729	                        arg_names,
  2730	                        callable_name,
  2731	                        object_type,
  2732	                        none_type_var_overlap,
  2733	                        context,
  2734	                    )
  2735	            except TooManyUnions:
  2736	                union_interrupted = True
  2737	            else:
  2738	                # Record if we succeeded. Next we need to see if maybe normal procedure
  2739	                # gives a narrower type.
  2740	                if unioned_return:
  2741	                    returns, inferred_types = zip(*unioned_return)
  2742	                    # Note that we use `combine_function_signatures` instead of just returning
  2743	                    # a union of inferred callables because for example a call
  2744	                    # Union[int -> int, str -> str](Union[int, str]) is invalid and
  2745	                    # we don't want to introduce internal inconsistencies.

  3039	        the first union.

  3045	        # hang for long time because of a weird overload call. The caller will get

  3059	                    plausible_targets,

  3061	                    arg_types,
  3062	                    arg_kinds,

  3064	                    callable_name,
  3065	                    object_type,
  3066	                    context,

  3077	                    plausible_targets,
  3078	                    args,
  3079	                    arg_types,

  3087	                get_proper_type(direct[0]), (UnionType, AnyType)

  3090	                return [direct]

  3453	    def visit_op_expr(self, e: OpExpr) -> Type:
  3454	        """Type check a binary operator expression."""
  3455	        if e.analyzed:
  3456	            # It's actually a type expression X | Y.
  3457	            return self.accept(e.analyzed)
  3458	        if e.op == "and" or e.op == "or":
  3459	            return self.check_boolean_op(e, e)
  3460	        if e.op == "*" and isinstance(e.left, ListExpr):
  3461	            # Expressions of form [...] * e get special type inference.
  3462	            return self.check_list_multiply(e)
  3463	        if e.op == "%":
  3464	            if isinstance(e.left, BytesExpr):
  3465	                return self.strfrm_checker.check_str_interpolation(e.left, e.right)
  3466	            if isinstance(e.left, StrExpr):
  3467	                return self.strfrm_checker.check_str_interpolation(e.left, e.right)
  3468	        left_type = self.accept(e.left)
  3469
  3470	        proper_left_type = get_proper_type(left_type)
  3471	        if isinstance(proper_left_type, TupleType) and e.op == "+":
  3472	            left_add_method = proper_left_type.partial_fallback.type.get("__add__")
  3473	            if left_add_method and left_add_method.fullname == "builtins.tuple.__add__":
  3474	                proper_right_type = get_proper_type(self.accept(e.right))
  3475	                if isinstance(proper_right_type, TupleType):
  3476	                    right_radd_method = proper_right_type.partial_fallback.type.get("__radd__")
  3477	                    if right_radd_method is None:
  3478	                        # One cannot have two variadic items in the same tuple.
  3479	                        if (
  3480	                            find_unpack_in_list(proper_left_type.items) is None
  3481	                            or find_unpack_in_list(proper_right_type.items) is None
  3482	                        ):
  3483	                            return self.concat_tuples(proper_left_type, proper_right_type)
  3484	                elif (
  3485	                    PRECISE_TUPLE_TYPES in self.chk.options.enable_incomplete_feature
  3486	                    and isinstance(proper_right_type, Instance)
  3487	                    and self.chk.type_is_iterable(proper_right_type)
  3488	                ):
  3489	                    # Handle tuple[X, Y] + tuple[Z, ...] = tuple[X, Y, *tuple[Z, ...]].
  3490	                    right_radd_method = proper_right_type.type.get("__radd__")
  3491	                    if (
  3492	                        right_radd_method is None
  3493	                        and proper_left_type.partial_fallback.type.fullname == "builtins.tuple"
  3494	                        and find_unpack_in_list(proper_left_type.items) is None
  3495	                    ):
  3496	                        item_type = self.chk.iterable_item_type(proper_right_type, e)
  3497	                        mapped = self.chk.named_generic_type("builtins.tuple", [item_type])
  3498	                        return proper_left_type.copy_modified(
  3499	                            items=proper_left_type.items + [UnpackType(mapped)]
  3500	                        )
  3501
  3502	        use_reverse: UseReverse = USE_REVERSE_DEFAULT
  3503	        if e.op == "|":
  3504	            if is_named_instance(proper_left_type, "builtins.dict"):
  3505	                # This is a special case for `dict | TypedDict`.
  3506	                # 1. Find `dict | TypedDict` case
  3507	                # 2. Switch `dict.__or__` to `TypedDict.__ror__` (the same from both runtime and typing perspective)
  3508	                proper_right_type = get_proper_type(self.accept(e.right))
  3509	                if isinstance(proper_right_type, TypedDictType):
  3510	                    use_reverse = USE_REVERSE_ALWAYS
  3511	            if isinstance(proper_left_type, TypedDictType):
  3512	                # This is the reverse case: `TypedDict | dict`,
  3513	                # simply do not allow the reverse checking:
  3514	                # do not call `__dict__.__ror__`.
  3515	                proper_right_type = get_proper_type(self.accept(e.right))
  3516	                if is_named_instance(proper_right_type, "builtins.dict"):
  3517	                    use_reverse = USE_REVERSE_NEVER
  3518
  3519	        if PRECISE_TUPLE_TYPES in self.chk.options.enable_incomplete_feature:
  3520	            # Handle tuple[X, ...] + tuple[Y, Z] = tuple[*tuple[X, ...], Y, Z].
  3521	            if (
  3522	                e.op == "+"
  3523	                and isinstance(proper_left_type, Instance)
  3524	                and proper_left_type.type.fullname == "builtins.tuple"
  3525	            ):
  3526	                proper_right_type = get_proper_type(self.accept(e.right))
  3527	                if (
  3528	                    isinstance(proper_right_type, TupleType)
  3529	                    and proper_right_type.partial_fallback.type.fullname == "builtins.tuple"
  3530	                    and find_unpack_in_list(proper_right_type.items) is None
  3531	                ):
  3532	                    return proper_right_type.copy_modified(
  3533	                        items=[UnpackType(proper_left_type)] + proper_right_type.items
  3534	                    )
  3535
  3536	        if e.op in operators.op_methods:
  3537	            method = operators.op_methods[e.op]
  3538	            if use_reverse is UseReverse.DEFAULT or use_reverse is UseReverse.NEVER:
  3539	                result, method_type = self.check_op(
  3540	                    method,
  3541	                    base_type=left_type,
  3542	                    arg=e.right,
  3543	                    context=e,
  3544	                    allow_reverse=use_reverse is UseReverse.DEFAULT,
  3545	                )
  3546	            elif use_reverse is UseReverse.ALWAYS:
  3547	                result, method_type = self.check_op(
  3548	                    # The reverse operator here gives better error messages:
  3549	                    operators.reverse_op_methods[method],
  3550	                    base_type=self.accept(e.right),
  3551	                    arg=e.left,
  3552	                    context=e,
  3553	                    allow_reverse=False,
  3554	                )

  3559	        else:

  3562	    def visit_comparison_expr(self, e: ComparisonExpr) -> Type:
  3563	        """Type check a comparison expression.
  3564
  3565	        Comparison expressions are type checked consecutive-pair-wise
  3566	        That is, 'a < b > c == d' is check as 'a < b and b > c and c == d'
  3567	        """
  3568	        result: Type | None = None
  3569	        sub_result: Type
  3570
  3571	        # Check each consecutive operand pair and their operator
  3572	        for left, right, operator in zip(e.operands, e.operands[1:], e.operators):
  3573	            left_type = self.accept(left)
  3574
  3575	            if operator == "in" or operator == "not in":
  3576	                # This case covers both iterables and containers, which have different meanings.
  3577	                # For a container, the in operator calls the __contains__ method.
  3578	                # For an iterable, the in operator iterates over the iterable, and compares each item one-by-one.
  3579	                # We allow `in` for a union of containers and iterables as long as at least one of them matches the
  3580	                # type of the left operand, as the operation will simply return False if the union's container/iterator
  3581	                # type doesn't match the left operand.
  3582
  3583	                # If the right operand has partial type, look it up without triggering
  3584	                # a "Need type annotation ..." message, as it would be noise.
  3585	                right_type = self.find_partial_type_ref_fast_path(right)
  3586	                if right_type is None:
  3587	                    right_type = self.accept(right)  # Validate the right operand
  3588
  3589	                right_type = get_proper_type(right_type)
  3590	                item_types: Sequence[Type] = [right_type]
  3591	                if isinstance(right_type, UnionType):
  3592	                    item_types = list(right_type.relevant_items())
  3593
  3594	                sub_result = self.bool_type()
  3595
  3596	                container_types: list[Type] = []
  3597	                iterable_types: list[Type] = []
  3598	                failed_out = False
  3599	                encountered_partial_type = False
  3600
  3601	                for item_type in item_types:
  3602	                    # Keep track of whether we get type check errors (these won't be reported, they
  3603	                    # are just to verify whether something is valid typing wise).
  3604	                    with self.msg.filter_errors(save_filtered_errors=True) as container_errors:
  3605	                        _, method_type = self.check_method_call_by_name(
  3606	                            method="__contains__",
  3607	                            base_type=item_type,
  3608	                            args=[left],
  3609	                            arg_kinds=[ARG_POS],
  3610	                            context=e,
  3611	                            original_type=right_type,
  3612	                        )
  3613	                        # Container item type for strict type overlap checks. Note: we need to only
  3614	                        # check for nominal type, because a usual "Unsupported operands for in"
  3615	                        # will be reported for types incompatible with __contains__().
  3616	                        # See testCustomContainsCheckStrictEquality for an example.
  3617	                        cont_type = self.chk.analyze_container_item_type(item_type)
  3618
  3619	                    if isinstance(item_type, PartialType):
  3620	                        # We don't really know if this is an error or not, so just shut up.
  3621	                        encountered_partial_type = True
  3622	                        pass
  3623	                    elif (
  3624	                        container_errors.has_new_errors()
  3625	                        and
  3626	                        # is_valid_var_arg is True for any Iterable
  3627	                        self.is_valid_var_arg(item_type)
  3628	                    ):
  3629	                        # it's not a container, but it is an iterable
  3630	                        with self.msg.filter_errors(save_filtered_errors=True) as iterable_errors:
  3631	                            _, itertype = self.chk.analyze_iterable_item_type_without_expression(
  3632	                                item_type, e
  3633	                            )
  3634	                        if iterable_errors.has_new_errors():
  3635	                            self.msg.add_errors(iterable_errors.filtered_errors())
  3636	                            failed_out = True
  3637	                        else:
  3638	                            method_type = CallableType(
  3639	                                [left_type],
  3640	                                [nodes.ARG_POS],
  3641	                                [None],
  3642	                                self.bool_type(),
  3643	                                self.named_type("builtins.function"),
  3644	                            )
  3645	                            e.method_types.append(method_type)
  3646	                            iterable_types.append(itertype)
  3647	                    elif not container_errors.has_new_errors() and cont_type:
  3648	                        container_types.append(cont_type)
  3649	                        e.method_types.append(method_type)
  3650	                    else:
  3651	                        self.msg.add_errors(container_errors.filtered_errors())
  3652	                        failed_out = True
  3653
  3654	                if not encountered_partial_type and not failed_out:
  3655	                    iterable_type = UnionType.make_union(iterable_types)
  3656	                    if not is_subtype(left_type, iterable_type):
  3657	                        if not container_types:
  3658	                            self.msg.unsupported_operand_types("in", left_type, right_type, e)
  3659	                        else:
  3660	                            container_type = UnionType.make_union(container_types)
  3661	                            if self.dangerous_comparison(
  3662	                                left_type,
  3663	                                container_type,
  3664	                                original_container=right_type,
  3665	                                prefer_literal=False,
  3666	                            ):
  3667	                                self.msg.dangerous_comparison(
  3668	                                    left_type, container_type, "container", e
  3669	                                )
  3670
  3671	            elif operator in operators.op_methods:
  3672	                method = operators.op_methods[operator]
  3673
  3674	                with ErrorWatcher(self.msg.errors) as w:
  3675	                    sub_result, method_type = self.check_op(
  3676	                        method, left_type, right, e, allow_reverse=True
  3677	                    )
  3678	                    e.method_types.append(method_type)
  3679
  3680	                # Only show dangerous overlap if there are no other errors. See
  3681	                # testCustomEqCheckStrictEquality for an example.
  3682	                if not w.has_new_errors() and operator in ("==", "!="):
  3683	                    right_type = self.accept(right)
  3684	                    if self.dangerous_comparison(left_type, right_type):
  3685	                        # Show the most specific literal types possible
  3686	                        left_type = try_getting_literal(left_type)
  3687	                        right_type = try_getting_literal(right_type)
  3688	                        self.msg.dangerous_comparison(left_type, right_type, "equality", e)
  3689
  3690	            elif operator == "is" or operator == "is not":
  3691	                right_type = self.accept(right)  # validate the right operand
  3692	                sub_result = self.bool_type()
  3693	                if self.dangerous_comparison(left_type, right_type):
  3694	                    # Show the most specific literal types possible
  3695	                    left_type = try_getting_literal(left_type)
  3696	                    right_type = try_getting_literal(right_type)
  3697	                    self.msg.dangerous_comparison(left_type, right_type, "identity", e)
  3698	                e.method_types.append(None)
  3699	            else:
  3700	                raise RuntimeError(f"Unknown comparison operator {operator}")
  3701
  3702	            #  Determine type of boolean-and of result and sub_result
  3703	            if result is None:
  3704	                result = sub_result
  3705	            else:
  3706	                result = join.join_types(result, sub_result)
  3707
  3708	        assert result is not None
  3709	        return result

  3727	    def dangerous_comparison(
  3728	        self,
  3729	        left: Type,
  3730	        right: Type,
  3731	        *,
  3732	        original_container: Type | None = None,
  3733	        seen_types: set[tuple[Type, Type]] | None = None,
  3734	        prefer_literal: bool = True,
  3735	    ) -> bool:
  3736	        """Check for dangerous non-overlapping comparisons like 42 == 'no'.
  3737
  3738	        The original_container is the original container type for 'in' checks
  3739	        (and None for equality checks).
  3740
  3741	        Rules:
  3742	            * X and None are overlapping even in strict-optional mode. This is to allow
  3743	            'assert x is not None' for x defined as 'x = None  # type: str' in class body
  3744	            (otherwise mypy itself would have couple dozen errors because of this).
  3745	            * Optional[X] and Optional[Y] are non-overlapping if X and Y are
  3746	            non-overlapping, although technically None is overlap, it is most
  3747	            likely an error.
  3748	            * Any overlaps with everything, i.e. always safe.
  3749	            * Special case: b'abc' in b'cde' is safe.
  3750	        """
  3751	        if not self.chk.options.strict_equality:
  3752	            return False
  3753
  3754	        if seen_types is None:
