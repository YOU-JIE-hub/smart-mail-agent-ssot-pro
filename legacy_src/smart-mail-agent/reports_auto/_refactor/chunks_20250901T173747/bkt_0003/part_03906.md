  2526	# The ParaFormatter will be able to format the following
  2527	# tags:
  2528	#       < /b > - bold
  2529	#       < /i > - italics

  2534	#       < super [size="pts"] [rise="pts"]> < /super > - superscript
  2535	#       < sup ="pts"] [rise="pts"]> < /sup > - superscript
  2536	#       < sub ="pts"] [rise="pts"]> < /sub > - subscript

  2542	#       <link>link text</link>

  2544	#               size/fontSize/uwidth/uoffset=num
  2545	#               name/face/fontName=name
  2546	#               fg/textColor/color/ucolor=color
  2547	#               backcolor/backColor/bgcolor=color
  2548	#               dest/destination/target/href/link=target
  2549	#               underline=bool turn on underline
  2550	#       <a>anchor text</a>

  2552	#               fontSize=num
  2553	#               fontName=name

    72	def drawPageNumbers(canvas, style, pages, availWidth, availHeight, dot=' . ', formatter=None):
    73	    '''
    74	    Draws pagestr on the canvas using the given style.
    75	    If dot is None, pagestr is drawn at the current position in the canvas.
    76	    If dot is a string, pagestr is drawn right-aligned. If the string is not empty,
    77	    the gap is filled with it.
    78	    '''
    79	    pagestr = ', '.join([str(p) for p, _ in pages])
    80	    x, y = canvas._curr_tx_info['cur_x'], canvas._curr_tx_info['cur_y']
    81
    82	    fontSize = style.fontSize
    83	    pagestrw = stringWidth(pagestr, style.fontName, fontSize)
    84
    85	    #if it's too long to fit, we need to shrink to fit in 10% increments.
    86	    #it would be very hard to output multiline entries.
    87	    #however, we impose a minimum size of 1 point as we don't want an
    88	    #infinite loop.   Ultimately we should allow a TOC entry to spill
    89	    #over onto a second line if needed.
    90	    freeWidth = availWidth-x
    91	    while pagestrw > freeWidth and fontSize >= 1.0:
    92	        fontSize = 0.9 * fontSize
    93	        pagestrw = stringWidth(pagestr, style.fontName, fontSize)
    94
    95
    96	    if isinstance(dot, strTypes):
    97	        if dot:
    98	            dotw = stringWidth(dot, style.fontName, fontSize)
    99	            dotsn = int((availWidth-x-pagestrw)/dotw)
   100	        else:
   101	            dotsn = dotw = 0
   102	        text = '%s%s' % (dotsn * dot, pagestr)
   103	        newx = availWidth - dotsn*dotw - pagestrw
   104	        pagex = availWidth - pagestrw
   105	    elif dot is None:
   106	        text = ',  ' + pagestr
   107	        newx = x
   108	        pagex = newx
   109	    else:
   110	        raise TypeError('Argument dot should either be None or an instance of basestring.')
   111
   112	    tx = canvas.beginText(newx, y)
   113	    tx.setFont(style.fontName, fontSize)
   114	    tx.setFillColor(style.textColor)
   115	    tx.textLine(text)
   116	    canvas.drawText(tx)
   117
   118	    commaw = stringWidth(', ', style.fontName, fontSize)
   119	    for p, key in pages:
   120	        if not key:
   121	            continue
   122	        w = stringWidth(str(p), style.fontName, fontSize)
   123	        canvas.linkRect('', key, (pagex, y, pagex+w, y+style.leading), relative=1)
   124	        pagex += w + commaw

   150	class TableOfContents(IndexingFlowable):
   151	    """This creates a formatted table of contents.
   152
   153	    It presumes a correct block of data is passed in.
   154	    The data block contains a list of (level, text, pageNumber)
   155	    triplets.  You can supply a paragraph style for each level
   156	    (starting at zero).
   157	    Set dotsMinLevel to determine from which level on a line of
   158	    dots should be drawn between the text and the page number.
   159	    If dotsMinLevel is set to a negative value, no dotted lines are drawn.
   160	    """
   161
   162	    def __init__(self,**kwds):
   163	        self.rightColumnWidth = kwds.pop('rightColumnWidth',72)
   164	        self.levelStyles = kwds.pop('levelStyles',defaultLevelStyles)
   165	        self.tableStyle = kwds.pop('tableStyle',defaultTableStyle)
   166	        self.dotsMinLevel = kwds.pop('dotsMinLevel',1)
   167	        self.formatter = kwds.pop('formatter',None)
   168	        if kwds: raise ValueError('unexpected keyword arguments %s' % ', '.join(kwds.keys()))
   169	        self._table = None
   170	        self._entries = []
   171	        self._lastEntries = []
   172
   173	    def beforeBuild(self):
   174	        # keep track of the last run
   175	        self._lastEntries = self._entries[:]
   176	        self.clearEntries()
   177
   178	    def isIndexing(self):
   179	        return 1
   180
   181	    def isSatisfied(self):
   182	        return (self._entries == self._lastEntries)
   183
   184	    def notify(self, kind, stuff):
   185	        """The notification hook called to register all kinds of events.
   186
   187	        Here we are interested in 'TOCEntry' events only.
   188	        """
   189	        if kind == 'TOCEntry':
   190	            self.addEntry(*stuff)
   191
   192	    def clearEntries(self):
   193	        self._entries = []
   194
   195	    def getLevelStyle(self, n):
   196	        '''Returns the style for level n, generating and caching styles on demand if not present.'''
   197	        try:
   198	            return self.levelStyles[n]
   199	        except IndexError:
   200	            prevstyle = self.getLevelStyle(n-1)
   201	            self.levelStyles.append(ParagraphStyle(
   202	                    name='%s-%d-indented' % (prevstyle.name, n),
   203	                    parent=prevstyle,
   204	                    firstLineIndent = prevstyle.firstLineIndent+delta,
   205	                    leftIndent = prevstyle.leftIndent+delta))
   206	            return self.levelStyles[n]
   207
   208	    def addEntry(self, level, text, pageNum, key=None):
   209	        """Adds one entry to the table of contents.
   210
   211	        This allows incremental buildup by a doctemplate.
   212	        Requires that enough styles are defined."""
   213
   214	        assert type(level) == type(1), "Level must be an integer"
   215	        self._entries.append((level, text, pageNum, key))
   216
   217
   218	    def addEntries(self, listOfEntries):
   219	        """Bulk creation of entries in the table of contents.
   220
   221	        If you knew the titles but not the page numbers, you could
   222	        supply them to get sensible output on the first run."""
   223
   224	        for entryargs in listOfEntries:
   225	            self.addEntry(*entryargs)

   228	    def wrap(self, availWidth, availHeight):
   229	        "All table properties should be known by now."
   230
   231	        # makes an internal table which does all the work.
   232	        # we draw the LAST RUN's entries!  If there are
   233	        # none, we make some dummy data to keep the table
   234	        # from complaining
   235	        if len(self._lastEntries) == 0:
   236	            _tempEntries = [(0,'Placeholder for table of contents',0,None)]
   237	        else:
   238	            _tempEntries = self._lastEntries
   239
   240	        def drawTOCEntryEnd(canvas, kind, label):
   241	            '''Callback to draw dots and page numbers after each entry.'''
   242	            label = label.split(',')
   243	            page, level, key = int(label[0]), int(label[1]), literal_eval(label[2])
   244	            style = self.getLevelStyle(level)
   245	            if self.dotsMinLevel >= 0 and level >= self.dotsMinLevel:
   246	                dot = ' . '
   247	            else:
   248	                dot = ''
   249	            if self.formatter: page = self.formatter(page)
   250	            drawPageNumbers(canvas, style, [(page, key)], availWidth, availHeight, dot)
   251	        self.canv.drawTOCEntryEnd = drawTOCEntryEnd
