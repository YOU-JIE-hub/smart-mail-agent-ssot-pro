  1421	    def analyze_callable_args_for_paramspec(

  1534	    def analyze_callable_type(self, t: UnboundType) -> Type:
  1535	        fallback = self.named_type("builtins.function")
  1536	        if len(t.args) == 0:
  1537	            # Callable (bare). Treat as Callable[..., Any].
  1538	            any_type = self.get_omitted_any(t)
  1539	            ret = callable_with_ellipsis(any_type, any_type, fallback)
  1540	        elif len(t.args) == 2:
  1541	            callable_args = t.args[0]
  1542	            ret_type = t.args[1]
  1543	            if isinstance(callable_args, TypeList):
  1544	                # Callable[[ARG, ...], RET] (ordinary callable type)
  1545	                analyzed_args = self.analyze_callable_args(callable_args)
  1546	                if analyzed_args is None:
  1547	                    return AnyType(TypeOfAny.from_error)
  1548	                args, kinds, names = analyzed_args
  1549	                ret = CallableType(args, kinds, names, ret_type=ret_type, fallback=fallback)
  1550	            elif isinstance(callable_args, EllipsisType):
  1551	                # Callable[..., RET] (with literal ellipsis; accept arbitrary arguments)
  1552	                ret = callable_with_ellipsis(
  1553	                    AnyType(TypeOfAny.explicit), ret_type=ret_type, fallback=fallback
  1554	                )
  1555	            else:
  1556	                # Callable[P, RET] (where P is ParamSpec)
  1557	                with self.tvar_scope_frame(namespace=""):
  1558	                    # Temporarily bind ParamSpecs to allow code like this:
  1559	                    #     my_fun: Callable[Q, Foo[Q]]
  1560	                    # We usually do this later in visit_callable_type(), but the analysis
  1561	                    # below happens at very early stage.
  1562	                    variables = []
  1563	                    for name, tvar_expr in self.find_type_var_likes(callable_args):
  1564	                        variables.append(self.tvar_scope.bind_new(name, tvar_expr))
  1565	                    maybe_ret = self.analyze_callable_args_for_paramspec(
  1566	                        callable_args, ret_type, fallback
  1567	                    ) or self.analyze_callable_args_for_concatenate(
  1568	                        callable_args, ret_type, fallback
  1569	                    )
  1570	                    if isinstance(maybe_ret, CallableType):
  1571	                        maybe_ret = maybe_ret.copy_modified(variables=variables)
  1572	                if maybe_ret is None:
  1573	                    # Callable[?, RET] (where ? is something invalid)
  1574	                    self.fail(
  1575	                        "The first argument to Callable must be a "
  1576	                        'list of types, parameter specification, or "..."',
  1577	                        t,
  1578	                        code=codes.VALID_TYPE,
  1579	                    )
  1580	                    self.note(
  1581	                        "See https://mypy.readthedocs.io/en/stable/kinds_of_types.html#callable-types-and-lambdas",
  1582	                        t,
  1583	                    )
  1584	                    return AnyType(TypeOfAny.from_error)
  1585	                elif isinstance(maybe_ret, AnyType):
  1586	                    return maybe_ret
  1587	                ret = maybe_ret
  1588	        else:
  1589	            if self.options.disallow_any_generics:
  1590	                self.fail('Please use "Callable[[<parameters>], <return type>]"', t)
  1591	            else:
  1592	                self.fail('Please use "Callable[[<parameters>], <return type>]" or "Callable"', t)
  1593	            return AnyType(TypeOfAny.from_error)
  1594	        assert isinstance(ret, CallableType)
  1595	        return ret.accept(self)

  1604	    def analyze_callable_args(
  1605	        self, arglist: TypeList
  1606	    ) -> tuple[list[Type], list[ArgKind], list[str | None]] | None:
  1607	        args: list[Type] = []
  1608	        kinds: list[ArgKind] = []
  1609	        names: list[str | None] = []
  1610	        seen_unpack = False
  1611	        unpack_types: list[Type] = []
  1612	        invalid_unpacks: list[Type] = []
  1613	        second_unpack_last = False
  1614	        for i, arg in enumerate(arglist.items):
  1615	            if isinstance(arg, CallableArgument):
  1616	                args.append(arg.typ)
  1617	                names.append(arg.name)
  1618	                if arg.constructor is None:
  1619	                    return None
  1620	                found = self.lookup_qualified(arg.constructor, arg)
  1621	                if found is None:
  1622	                    # Looking it up already put an error message in
  1623	                    return None
  1624	                elif found.fullname not in ARG_KINDS_BY_CONSTRUCTOR:
  1625	                    self.fail(f'Invalid argument constructor "{found.fullname}"', arg)
  1626	                    return None
  1627	                else:
  1628	                    assert found.fullname is not None
  1629	                    kind = ARG_KINDS_BY_CONSTRUCTOR[found.fullname]
  1630	                    kinds.append(kind)
  1631	                    if arg.name is not None and kind.is_star():
  1632	                        self.fail(f"{arg.constructor} arguments should not have names", arg)
  1633	                        return None
  1634	            elif (
  1635	                isinstance(arg, UnboundType)
  1636	                and self.refers_to_full_names(arg, UNPACK_TYPE_NAMES)
  1637	                or isinstance(arg, UnpackType)
  1638	            ):
  1639	                if seen_unpack:
  1640	                    # Multiple unpacks, preserve them, so we can give an error later.
  1641	                    if i == len(arglist.items) - 1 and not invalid_unpacks:
  1642	                        # Special case: if there are just two unpacks, and the second one appears
  1643	                        # as last type argument, it can be still valid, if the second unpacked type
  1644	                        # is a TypedDict. This should be checked by the caller.
  1645	                        second_unpack_last = True
  1646	                    invalid_unpacks.append(arg)
  1647	                    continue
  1648	                seen_unpack = True
  1649	                unpack_types.append(arg)
  1650	            else:
  1651	                if seen_unpack:
  1652	                    unpack_types.append(arg)
  1653	                else:
  1654	                    args.append(arg)
  1655	                    kinds.append(ARG_POS)
  1656	                    names.append(None)
  1657	        if seen_unpack:
  1658	            if len(unpack_types) == 1:
  1659	                args.append(unpack_types[0])
  1660	            else:
  1661	                first = unpack_types[0]
  1662	                if isinstance(first, UnpackType):
  1663	                    # UnpackType doesn't have its own line/column numbers,
  1664	                    # so use the unpacked type for error messages.
  1665	                    first = first.type
  1666	                args.append(
  1667	                    UnpackType(self.tuple_type(unpack_types, line=first.line, column=first.column))
  1668	                )
  1669	            kinds.append(ARG_STAR)
  1670	            names.append(None)
  1671	        for arg in invalid_unpacks:
  1672	            args.append(arg)
  1673	            kinds.append(ARG_STAR2 if second_unpack_last else ARG_STAR)
  1674	            names.append(None)
  1675	        # Note that arglist below is only used for error context.
  1676	        check_arg_names(names, [arglist] * len(args), self.fail, "Callable")
  1677	        check_arg_kinds(kinds, [arglist] * len(args), self.fail)
  1678	        return args, kinds, names

  1694	    def analyze_literal_param(self, idx: int, arg: Type, ctx: Context) -> list[Type] | None:
  1695	        # This UnboundType was originally defined as a string.
  1696	        if (
  1697	            isinstance(arg, ProperType)
  1698	            and isinstance(arg, (UnboundType, UnionType))
  1699	            and arg.original_str_expr is not None
  1700	        ):
  1701	            assert arg.original_str_fallback is not None
  1702	            return [
  1703	                LiteralType(
  1704	                    value=arg.original_str_expr,
  1705	                    fallback=self.named_type(arg.original_str_fallback),
  1706	                    line=arg.line,
  1707	                    column=arg.column,
  1708	                )
  1709	            ]
  1710
  1711	        # If arg is an UnboundType that was *not* originally defined as
  1712	        # a string, try expanding it in case it's a type alias or something.
  1713	        if isinstance(arg, UnboundType):
  1714	            self.nesting_level += 1
  1715	            try:
  1716	                arg = self.visit_unbound_type(arg, defining_literal=True)
  1717	            finally:
  1718	                self.nesting_level -= 1
  1719
  1720	        # Literal[...] cannot contain Any. Give up and add an error message
  1721	        # (if we haven't already).
  1722	        arg = get_proper_type(arg)
  1723	        if isinstance(arg, AnyType):
  1724	            # Note: We can encounter Literals containing 'Any' under three circumstances:
  1725	            #
  1726	            # 1. If the user attempts use an explicit Any as a parameter
  1727	            # 2. If the user is trying to use an enum value imported from a module with
  1728	            #    no type hints, giving it an implicit type of 'Any'
  1729	            # 3. If there's some other underlying problem with the parameter.
  1730	            #
  1731	            # We report an error in only the first two cases. In the third case, we assume
  1732	            # some other region of the code has already reported a more relevant error.
  1733	            #
  1734	            # TODO: Once we start adding support for enums, make sure we report a custom
  1735	            # error for case 2 as well.
  1736	            if arg.type_of_any not in (TypeOfAny.from_error, TypeOfAny.special_form):
  1737	                self.fail(
  1738	                    f'Parameter {idx} of Literal[...] cannot be of type "Any"',
  1739	                    ctx,
  1740	                    code=codes.VALID_TYPE,
  1741	                )
  1742	            return None
  1743	        elif isinstance(arg, RawExpressionType):
  1744	            # A raw literal. Convert it directly into a literal if we can.
  1745	            if arg.literal_value is None:
  1746	                name = arg.simple_name()
  1747	                if name in ("float", "complex"):
  1748	                    msg = f'Parameter {idx} of Literal[...] cannot be of type "{name}"'
  1749	                else:
  1750	                    msg = "Invalid type: Literal[...] cannot contain arbitrary expressions"
  1751	                self.fail(msg, ctx, code=codes.VALID_TYPE)
  1752	                # Note: we deliberately ignore arg.note here: the extra info might normally be
  1753	                # helpful, but it generally won't make sense in the context of a Literal[...].
  1754	                return None
  1755
  1756	            # Remap bytes and unicode into the appropriate type for the correct Python version
  1757	            fallback = self.named_type(arg.base_type_name)
  1758	            assert isinstance(fallback, Instance)
  1759	            return [LiteralType(arg.literal_value, fallback, line=arg.line, column=arg.column)]
  1760	        elif isinstance(arg, (NoneType, LiteralType)):
  1761	            # Types that we can just add directly to the literal/potential union of literals.
  1762	            return [arg]
  1763	        elif isinstance(arg, Instance) and arg.last_known_value is not None:
  1764	            # Types generated from declarations like "var: Final = 4".
  1765	            return [arg.last_known_value]
  1766	        elif isinstance(arg, UnionType):
  1767	            out = []
  1768	            for union_arg in arg.items:
  1769	                union_result = self.analyze_literal_param(idx, union_arg, ctx)
  1770	                if union_result is None:
  1771	                    return None
  1772	                out.extend(union_result)
  1773	            return out
  1774	        else:
  1775	            self.fail(f"Parameter {idx} of Literal[...] is invalid", ctx, code=codes.VALID_TYPE)
  1776	            return None

  1799	    def infer_type_variables(
  1800	        self, type: CallableType
  1801	    ) -> tuple[list[tuple[str, TypeVarLikeExpr]], bool]:
  1802	        """Infer type variables from a callable.
  1803
  1804	        Return tuple with these items:
  1805	         - list of unique type variables referred to in a callable
  1806	         - whether there is a reference to the Self type
  1807	        """
  1808	        visitor = FindTypeVarVisitor(self.api, self.tvar_scope)
  1809	        for arg in type.arg_types:
  1810	            arg.accept(visitor)
  1811
  1812	        # When finding type variables in the return type of a function, don't
  1813	        # look inside Callable types.  Type variables only appearing in
  1814	        # functions in the return type belong to those functions, not the
  1815	        # function we're currently analyzing.

  1817	        type.ret_type.accept(visitor)

  1821	    def bind_function_type_variables(
  1822	        self, fun_type: CallableType, defn: Context
  1823	    ) -> tuple[Sequence[TypeVarLikeType], bool]:
  1824	        """Find the type variables of the function type and bind them in our tvar_scope"""
  1825	        has_self_type = False
  1826	        if fun_type.variables:
  1827	            defs = []
  1828	            for var in fun_type.variables:
  1829	                if self.api.type and self.api.type.self_type and var == self.api.type.self_type:
  1830	                    has_self_type = True
  1831	                    continue
  1832	                var_node = self.lookup_qualified(var.name, defn)
  1833	                assert var_node, "Binding for function type variable not found within function"
  1834	                var_expr = var_node.node
  1835	                assert isinstance(var_expr, TypeVarLikeExpr)
  1836	                binding = self.tvar_scope.bind_new(var.name, var_expr)
  1837	                defs.append(binding)
  1838	            return defs, has_self_type
  1839	        typevars, has_self_type = self.infer_type_variables(fun_type)
  1840	        # Do not define a new type variable if already defined in scope.
  1841	        typevars = [
  1842	            (name, tvar) for name, tvar in typevars if not self.is_defined_type_var(name, defn)
  1843	        ]
  1844	        defs = []
  1845	        for name, tvar in typevars:

  2042	def fix_instance(
  2043	    t: Instance,
  2044	    fail: MsgCallback,
  2045	    note: MsgCallback,
  2046	    disallow_any: bool,

  2048	    use_generic_error: bool = False,
  2049	    unexpanded_type: Type | None = None,

  2056	    min_tv_count = sum(not tv.has_default() for tv in t.type.defn.type_vars)
  2057	    max_tv_count = len(t.type.type_vars)
  2058	    if arg_count < min_tv_count or arg_count > max_tv_count:

  2094	def instantiate_type_alias(
  2095	    node: TypeAlias,
  2096	    args: list[Type],
  2097	    fail: MsgCallback,
  2098	    no_args: bool,
  2099	    ctx: Context,
  2100	    options: Options,
  2101	    *,
  2102	    unexpanded_type: Type | None = None,
  2103	    disallow_any: bool = False,
  2104	    use_standard_error: bool = False,
  2105	    empty_tuple_index: bool = False,
  2106	) -> Type:
  2107	    """Create an instance of a (generic) type alias from alias node and type arguments.
  2108
  2109	    We are following the rules outlined in TypeAlias docstring.
  2110	    Here:
  2111	        node: type alias node (definition)
  2112	        args: type arguments (types to be substituted in place of type variables
  2113	              when expanding the alias)
  2114	        fail: error reporter callback
  2115	        no_args: whether original definition used a bare generic `A = List`
  2116	        ctx: context where expansion happens
  2117	        unexpanded_type, disallow_any, use_standard_error: used to customize error messages
  2118	    """
  2119	    # Type aliases are special, since they can be expanded during semantic analysis,
  2120	    # so we need to normalize them as soon as possible.
  2121	    # TODO: can this cause an infinite recursion?
  2122	    args = flatten_nested_tuples(args)
  2123	    if any(unknown_unpack(a) for a in args):
  2124	        # This type is not ready to be validated, because of unknown total count.
  2125	        # Note that we keep the kind of Any for consistency.
  2126	        return set_any_tvars(node, [], ctx.line, ctx.column, options, special_form=True)
  2127
  2128	    max_tv_count = len(node.alias_tvars)
  2129	    act_len = len(args)
  2130	    if (
  2131	        max_tv_count > 0
  2132	        and act_len == 0
  2133	        and not (empty_tuple_index and node.tvar_tuple_index is not None)
  2134	    ):
  2135	        # Interpret bare Alias same as normal generic, i.e., Alias[Any, Any, ...]
  2136	        return set_any_tvars(
  2137	            node,
  2138	            args,
  2139	            ctx.line,
  2140	            ctx.column,
  2141	            options,
  2142	            disallow_any=disallow_any,
  2143	            fail=fail,
  2144	            unexpanded_type=unexpanded_type,
  2145	        )
  2146	    if max_tv_count == 0 and act_len == 0:
  2147	        if no_args:
  2148	            assert isinstance(node.target, Instance)  # type: ignore[misc]
  2149	            # Note: this is the only case where we use an eager expansion. See more info about
  2150	            # no_args aliases like L = List in the docstring for TypeAlias class.
  2151	            return Instance(node.target.type, [], line=ctx.line, column=ctx.column)
  2152	        return TypeAliasType(node, [], line=ctx.line, column=ctx.column)
  2153	    if (
  2154	        max_tv_count == 0
  2155	        and act_len > 0
  2156	        and isinstance(node.target, Instance)  # type: ignore[misc]
  2157	        and no_args
  2158	    ):
  2159	        tp = Instance(node.target.type, args)
  2160	        tp.line = ctx.line
  2161	        tp.column = ctx.column
  2162	        tp.end_line = ctx.end_line
  2163	        tp.end_column = ctx.end_column
  2164	        return tp
  2165	    if node.tvar_tuple_index is None:
  2166	        if any(isinstance(a, UnpackType) for a in args):
  2167	            # A variadic unpack in fixed size alias (fixed unpacks must be flattened by the caller)
  2168	            fail(message_registry.INVALID_UNPACK_POSITION, ctx, code=codes.VALID_TYPE)
  2169	            return set_any_tvars(node, [], ctx.line, ctx.column, options, from_error=True)
  2170	        min_tv_count = sum(not tv.has_default() for tv in node.alias_tvars)
  2171	        fill_typevars = act_len != max_tv_count
  2172	        correct = min_tv_count <= act_len <= max_tv_count
  2173	    else:
  2174	        min_tv_count = sum(
  2175	            not tv.has_default() and not isinstance(tv, TypeVarTupleType)
  2176	            for tv in node.alias_tvars
  2177	        )
  2178	        correct = act_len >= min_tv_count
  2179	        for a in args:
  2180	            if isinstance(a, UnpackType):
  2181	                unpacked = get_proper_type(a.type)
  2182	                if isinstance(unpacked, Instance) and unpacked.type.fullname == "builtins.tuple":
  2183	                    # Variadic tuple is always correct.
  2184	                    correct = True
  2185	        fill_typevars = not correct
  2186	    if fill_typevars:
  2187	        if not correct:
  2188	            if use_standard_error:
  2189	                # This is used if type alias is an internal representation of another type,
  2190	                # for example a generic TypedDict or NamedTuple.
  2191	                msg = wrong_type_arg_count(max_tv_count, max_tv_count, str(act_len), node.name)
  2192	            else:
  2193	                if node.tvar_tuple_index is not None:
  2194	                    msg = (
  2195	                        "Bad number of arguments for type alias,"
  2196	                        f" expected at least {min_tv_count}, given {act_len}"
  2197	                    )
  2198	                elif min_tv_count != max_tv_count:
  2199	                    msg = (
  2200	                        "Bad number of arguments for type alias,"
  2201	                        f" expected between {min_tv_count} and {max_tv_count}, given {act_len}"
  2202	                    )
  2203	                else:
  2204	                    msg = (
  2205	                        "Bad number of arguments for type alias,"
  2206	                        f" expected {min_tv_count}, given {act_len}"
  2207	                    )
  2208	            fail(msg, ctx, code=codes.TYPE_ARG)
  2209	            args = []
  2210	        return set_any_tvars(node, args, ctx.line, ctx.column, options, from_error=True)
  2211	    elif node.tvar_tuple_index is not None:
  2212	        # We also need to check if we are not performing a type variable tuple split.
  2213	        unpack = find_unpack_in_list(args)
  2214	        if unpack is not None:
  2215	            unpack_arg = args[unpack]
  2216	            assert isinstance(unpack_arg, UnpackType)
  2217	            if isinstance(unpack_arg.type, TypeVarTupleType):
  2218	                exp_prefix = node.tvar_tuple_index
  2219	                act_prefix = unpack
  2220	                exp_suffix = len(node.alias_tvars) - node.tvar_tuple_index - 1
  2221	                act_suffix = len(args) - unpack - 1
  2222	                if act_prefix < exp_prefix or act_suffix < exp_suffix:
  2223	                    fail("TypeVarTuple cannot be split", ctx, code=codes.TYPE_ARG)
  2224	                    return set_any_tvars(node, [], ctx.line, ctx.column, options, from_error=True)
  2225	    # TODO: we need to check args validity w.r.t alias.alias_tvars.
  2226	    # Otherwise invalid instantiations will be allowed in runtime context.
  2227	    # Note: in type context, these will be still caught by semanal_typeargs.
  2228	    typ = TypeAliasType(node, args, ctx.line, ctx.column)
  2229	    assert typ.alias is not None
  2230	    # HACK: Implement FlexibleAlias[T, typ] by expanding it to typ here.
  2231	    if (
  2232	        isinstance(typ.alias.target, Instance)  # type: ignore[misc]
  2233	        and typ.alias.target.type.fullname == "mypy_extensions.FlexibleAlias"
  2234	    ):
  2235	        exp = get_proper_type(typ)
  2236	        assert isinstance(exp, Instance)
  2237	        return exp.args[-1]

  2343	def detect_diverging_alias(
  2344	    node: TypeAlias,
  2345	    target: Type,
  2346	    lookup: Callable[[str, Context], SymbolTableNode | None],
  2347	    scope: TypeVarLikeScope,
  2348	) -> bool:
  2349	    """This detects type aliases that will diverge during type checking.
  2350
  2351	    For example F = Something[..., F[List[T]]]. At each expansion step this will produce

   164	    assert isinstance(init_method.node, (SYMBOL_FUNCBASE_TYPES, Decorator))
   165
   166	    init_index = info.mro.index(init_method.node.info)
   167	    new_index = info.mro.index(new_method.node.info)
   168
   169	    fallback = info.metaclass_type or named_type("builtins.type")
   170	    if init_index < new_index:
   171	        method: FuncBase | Decorator = init_method.node
   172	        is_new = False
   173	    elif init_index > new_index:
   174	        method = new_method.node
   175	        is_new = True
   176	    else:
   177	        if init_method.node.info.fullname == "builtins.object":
   178	            # Both are defined by object.  But if we've got a bogus
   179	            # base class, we can't know for sure, so check for that.
   180	            if info.fallback_to_any:
   181	                # Construct a universal callable as the prototype.
   182	                any_type = AnyType(TypeOfAny.special_form)
   183	                sig = CallableType(
   184	                    arg_types=[any_type, any_type],
   185	                    arg_kinds=[ARG_STAR, ARG_STAR2],
   186	                    arg_names=["_args", "_kwds"],
   187	                    ret_type=any_type,
   188	                    is_bound=True,
   189	                    fallback=named_type("builtins.function"),
   190	                )
   191	                return class_callable(sig, info, fallback, None, is_new=False)
   192
   193	        # Otherwise prefer __init__ in a tie. It isn't clear that this
   194	        # is the right thing, but __new__ caused problems with
   195	        # typeshed (#5647).
   196	        method = init_method.node
   197	        is_new = False
   198	    # Construct callable type based on signature of __init__. Adjust
   199	    # return type and insert type arguments.
   200	    if isinstance(method, FuncBase):
   201	        t = function_type(method, fallback)
   202	    else:
   203	        assert isinstance(method.type, ProperType)
   204	        assert isinstance(method.type, FunctionLike)  # is_valid_constructor() ensures this
   205	        t = method.type
   206	    return type_object_type_from_function(t, info, method.info, fallback, is_new)

   222	def type_object_type_from_function(
   223	    signature: FunctionLike, info: TypeInfo, def_info: TypeInfo, fallback: Instance, is_new: bool
   224	) -> FunctionLike:
   225	    # We first need to record all non-trivial (explicit) self types in __init__,
   226	    # since they will not be available after we bind them. Note, we use explicit
   227	    # self-types only in the defining class, similar to __new__ (but not exactly the same,
   228	    # see comment in class_callable below). This is mostly useful for annotating library
   229	    # classes such as subprocess.Popen.
   230	    default_self = fill_typevars(info)
   231	    if not is_new and not info.is_newtype:
   232	        orig_self_types = [get_self_type(it, default_self) for it in signature.items]
   233	    else:
   234	        orig_self_types = [None] * len(signature.items)
   235
   236	    # The __init__ method might come from a generic superclass 'def_info'
   237	    # with type variables that do not map identically to the type variables of
   238	    # the class 'info' being constructed. For example:
   239	    #
   240	    #   class A(Generic[T]):
   241	    #       def __init__(self, x: T) -> None: ...
   242	    #   class B(A[List[T]]):
   243	    #      ...
   244	    #
   245	    # We need to map B's __init__ to the type (List[T]) -> None.
   246	    signature = bind_self(
   247	        signature,
   248	        original_type=default_self,
   249	        is_classmethod=is_new,
   250	        # Explicit instance self annotations have special handling in class_callable(),
   251	        # we don't need to bind any type variables in them if they are generic.
   252	        ignore_instances=True,
   253	    )
   254	    signature = cast(FunctionLike, map_type_from_supertype(signature, info, def_info))
   255
   256	    special_sig: str | None = None
   257	    if def_info.fullname == "builtins.dict":
   258	        # Special signature!
   259	        special_sig = "dict"
   260
   261	    if isinstance(signature, CallableType):
   262	        return class_callable(signature, info, fallback, special_sig, is_new, orig_self_types[0])
   263	    else:
   264	        # Overloaded __init__/__new__.
   265	        assert isinstance(signature, Overloaded)
   266	        items: list[CallableType] = []
   267	        for item, orig_self in zip(signature.items, orig_self_types):
   268	            items.append(class_callable(item, info, fallback, special_sig, is_new, orig_self))
   269	        return Overloaded(items)

   272	def class_callable(
   273	    init_type: CallableType,
   274	    info: TypeInfo,
   275	    type_type: Instance,
   276	    special_sig: str | None,
   277	    is_new: bool,
   278	    orig_self_type: Type | None = None,
   279	) -> CallableType:
   280	    """Create a type object type based on the signature of __init__."""
   281	    variables: list[TypeVarLikeType] = []
   282	    variables.extend(info.defn.type_vars)
   283	    variables.extend(init_type.variables)
   284
   285	    from mypy.subtypes import is_subtype
   286
   287	    init_ret_type = get_proper_type(init_type.ret_type)
   288	    orig_self_type = get_proper_type(orig_self_type)
   289	    default_ret_type = fill_typevars(info)
   290	    explicit_type = init_ret_type if is_new else orig_self_type
   291	    if (
   292	        isinstance(explicit_type, (Instance, TupleType, UninhabitedType))
   293	        # We have to skip protocols, because it can be a subtype of a return type
   294	        # by accident. Like `Hashable` is a subtype of `object`. See #11799
   295	        and isinstance(default_ret_type, Instance)
   296	        and not default_ret_type.type.is_protocol
   297	        # Only use the declared return type from __new__ or declared self in __init__
   298	        # if it is actually returning a subtype of what we would return otherwise.
   299	        and is_subtype(explicit_type, default_ret_type, ignore_type_params=True)
   300	    ):
   301	        ret_type: Type = explicit_type
   302	    else:
   303	        ret_type = default_ret_type
   304
   305	    callable_type = init_type.copy_modified(
   306	        ret_type=ret_type,
   307	        fallback=type_type,
   308	        name=None,
   309	        variables=variables,
   310	        special_sig=special_sig,
   311	    )
   312	    c = callable_type.with_name(info.name)
   313	    return c

   316	def map_type_from_supertype(typ: Type, sub_info: TypeInfo, super_info: TypeInfo) -> Type:
   317	    """Map type variables in a type defined in a supertype context to be valid
   318	    in the subtype context. Assume that the result is unique; if more than
   319	    one type is possible, return one of the alternatives.
   320
   321	    For example, assume
   322
   323	      class D(Generic[S]): ...
   324	      class C(D[E[T]], Generic[T]): ...
   325
   326	    Now S in the context of D would be mapped to E[T] in the context of C.
   327	    """
   328	    # Create the type of self in subtype, of form t[a1, ...].
   329	    inst_type = fill_typevars(sub_info)
   330	    if isinstance(inst_type, TupleType):
   331	        inst_type = tuple_fallback(inst_type)
   332	    # Map the type of self to supertype. This gets us a description of the
   333	    # supertype type variables in terms of subtype variables, i.e. t[t1, ...]
   334	    # so that any type variables in tN are to be interpreted in subtype
   335	    # context.
   336	    inst_type = map_instance_to_supertype(inst_type, super_info)
   337	    # Finally expand the type variables in type with those in the previously
   338	    # constructed type. Note that both type and inst_type may have type
   339	    # variables, but in type they are interpreted in supertype context while
   340	    # in inst_type they are interpreted in subtype context. This works even if
   341	    # the names of type variables in supertype and subtype overlap.
   342	    return expand_type_by_instance(typ, inst_type)

   345	def supported_self_type(
   346	    typ: ProperType, allow_callable: bool = True, allow_instances: bool = True
   347	) -> bool:
   348	    """Is this a supported kind of explicit self-types?
   349
   350	    Currently, this means an X or Type[X], where X is an instance or
   351	    a type variable with an instance upper bound.
   352	    """
   353	    if isinstance(typ, TypeType):
   354	        return supported_self_type(typ.item)
   355	    if allow_callable and isinstance(typ, CallableType):
   356	        # Special case: allow class callable instead of Type[...] as cls annotation,
   357	        # as well as callable self for callback protocols.
   358	        return True
   359	    return isinstance(typ, TypeVarType) or (
   360	        allow_instances and isinstance(typ, Instance) and typ != fill_typevars(typ.type)
   361	    )

   367	def bind_self(
   368	    method: F,
   369	    original_type: Type | None = None,
   370	    is_classmethod: bool = False,
   371	    ignore_instances: bool = False,
   372	) -> F:
   373	    """Return a copy of `method`, with the type of its first parameter (usually
   374	    self or cls) bound to original_type.
   375
   376	    If the type of `self` is a generic type (T, or Type[T] for classmethods),
   377	    instantiate every occurrence of type with original_type in the rest of the
   378	    signature and in the return type.
   379
   380	    original_type is the type of E in the expression E.copy(). It is None in
   381	    compatibility checks. In this case we treat it as the erasure of the
   382	    declared type of self.
   383
   384	    This way we can express "the type of self". For example:
   385
   386	    T = TypeVar('T', bound='A')
   387	    class A:
   388	        def copy(self: T) -> T: ...
   389
   390	    class B(A): pass
   391
   392	    b = B().copy()  # type: B
   393
   394	    """
   395	    if isinstance(method, Overloaded):
   396	        items = []
   397	        original_type = get_proper_type(original_type)
   398	        for c in method.items:
   399	            if isinstance(original_type, Instance):
   400	                # Filter based on whether declared self type can match actual object type.
   401	                # For example, if self has type C[int] and method is accessed on a C[str] value,
   402	                # omit this item. This is best effort since bind_self can be called in many
   403	                # contexts, and doing complete validation might trigger infinite recursion.
   404	                #
   405	                # Note that overload item filtering normally happens elsewhere. This is needed
   406	                # at least during constraint inference.
   407	                keep = is_valid_self_type_best_effort(c, original_type)
   408	            else:
   409	                keep = True
   410	            if keep:
   411	                items.append(bind_self(c, original_type, is_classmethod, ignore_instances))
   412	        if len(items) == 0:
   413	            # If no item matches, returning all items helps avoid some spurious errors
   414	            items = [
   415	                bind_self(c, original_type, is_classmethod, ignore_instances) for c in method.items
   416	            ]
   417	        return cast(F, Overloaded(items))
   418	    assert isinstance(method, CallableType)
   419	    func: CallableType = method
   420	    if not func.arg_types:
   421	        # Invalid method, return something.
   422	        return method
   423	    if func.arg_kinds[0] in (ARG_STAR, ARG_STAR2):
   424	        # The signature is of the form 'def foo(*args, ...)'.
   425	        # In this case we shouldn't drop the first arg,
   426	        # since func will be absorbed by the *args.
   427	        # TODO: infer bounds on the type of *args?
   428
   429	        # In the case of **kwargs we should probably emit an error, but
   430	        # for now we simply skip it, to avoid crashes down the line.
   431	        return method
   432	    self_param_type = get_proper_type(func.arg_types[0])
   433
   434	    variables: Sequence[TypeVarLikeType]
   435	    # Having a def __call__(self: Callable[...], ...) can cause infinite recursion. Although
   436	    # this special-casing looks not very principled, there is nothing meaningful we can infer
   437	    # from such definition, since it is inherently indefinitely recursive.
   438	    allow_callable = func.name is None or not func.name.startswith("__call__ of")
   439	    if func.variables and supported_self_type(
   440	        self_param_type, allow_callable=allow_callable, allow_instances=not ignore_instances
   441	    ):
   442	        from mypy.infer import infer_type_arguments
   443
   444	        if original_type is None:
   445	            # TODO: type check method override (see #7861).
   446	            original_type = erase_to_bound(self_param_type)
   447	        original_type = get_proper_type(original_type)
   448
   449	        # Find which of method type variables appear in the type of "self".
   450	        self_ids = {tv.id for tv in get_all_type_vars(self_param_type)}
   451	        self_vars = [tv for tv in func.variables if tv.id in self_ids]
   452
   453	        # Solve for these type arguments using the actual class or instance type.
   454	        typeargs = infer_type_arguments(
   455	            self_vars, self_param_type, original_type, is_supertype=True
   456	        )
   457	        if (
   458	            is_classmethod
   459	            and any(isinstance(get_proper_type(t), UninhabitedType) for t in typeargs)
   460	            and isinstance(original_type, (Instance, TypeVarType, TupleType))
   461	        ):
   462	            # In case we call a classmethod through an instance x, fallback to type(x).
   463	            typeargs = infer_type_arguments(
   464	                self_vars, self_param_type, TypeType(original_type), is_supertype=True
   465	            )
   466
   467	        # Update the method signature with the solutions found.
   468	        # Technically, some constraints might be unsolvable, make them Never.
   469	        to_apply = [t if t is not None else UninhabitedType() for t in typeargs]
   470	        func = expand_type(func, {tv.id: arg for tv, arg in zip(self_vars, to_apply)})
   471	        variables = [v for v in func.variables if v not in self_vars]
   472	    else:
   473	        variables = func.variables
   474
   475	    original_type = get_proper_type(original_type)
   476	    if isinstance(original_type, CallableType) and original_type.is_type_obj():
   477	        original_type = TypeType.make_normalized(original_type.ret_type)
   478	    res = func.copy_modified(
   479	        arg_types=func.arg_types[1:],
   480	        arg_kinds=func.arg_kinds[1:],
   481	        arg_names=func.arg_names[1:],
   482	        variables=variables,
   483	        is_bound=True,
   484	    )
   485	    return cast(F, res)

   488	def is_valid_self_type_best_effort(c: CallableType, self_type: Instance) -> bool:
   489	    """Quickly check if self_type might match the self in a callable.
   490
   491	    Avoid performing any complex type operations. This is performance-critical.
   492
   493	    Default to returning True if we don't know (or it would be too expensive).
   494	    """
   495	    if (
   496	        self_type.args
   497	        and c.arg_types
   498	        and isinstance((arg_type := get_proper_type(c.arg_types[0])), Instance)
   499	        and c.arg_kinds[0] in (ARG_POS, ARG_OPT)
   500	        and arg_type.args
   501	        and self_type.type.fullname != "functools._SingleDispatchCallable"
   502	    ):
   503	        if self_type.type is not arg_type.type:
   504	            # We can't map to supertype, since it could trigger expensive checks for
   505	            # protocol types, so we consevatively assume this is fine.
   506	            return True
   507
   508	        # Fast path: no explicit annotation on self
   509	        if all(
   510	            (
   511	                type(arg) is TypeVarType
   512	                and type(arg.upper_bound) is Instance
   513	                and arg.upper_bound.type.fullname == "builtins.object"
   514	            )
   515	            for arg in arg_type.args
   516	        ):
   517	            return True
   518
   519	        from mypy.meet import is_overlapping_types
   520
   521	        return is_overlapping_types(self_type, c.arg_types[0])
   522	    return True

   525	def erase_to_bound(t: Type) -> Type:
   526	    # TODO: use value restrictions to produce a union?
   527	    t = get_proper_type(t)
   528	    if isinstance(t, TypeVarType):
   529	        return t.upper_bound
   530	    if isinstance(t, TypeType):
   531	        if isinstance(t.item, TypeVarType):
   532	            return TypeType.make_normalized(t.item.upper_bound)
   533	    return t

   536	def callable_corresponding_argument(
   537	    typ: NormalizedCallableType | Parameters, model: FormalArgument
   538	) -> FormalArgument | None:
   539	    """Return the argument a function that corresponds to `model`"""
   540
   541	    by_name = typ.argument_by_name(model.name)
   542	    by_pos = typ.argument_by_position(model.pos)
   543	    if by_name is None and by_pos is None:
   544	        return None
   545	    if by_name is not None and by_pos is not None:
   546	        if by_name == by_pos:
   547	            return by_name
   548	        # If we're dealing with an optional pos-only and an optional
   549	        # name-only arg, merge them.  This is the case for all functions
   550	        # taking both *args and **args, or a pair of functions like so:
   551
   552	        # def right(a: int = ...) -> None: ...
   553	        # def left(__a: int = ..., *, a: int = ...) -> None: ...
   554	        from mypy.subtypes import is_equivalent
   555
   556	        if (
   557	            not (by_name.required or by_pos.required)
   558	            and by_pos.name is None
   559	            and by_name.pos is None
   560	            and is_equivalent(by_name.typ, by_pos.typ)
   561	        ):
   562	            return FormalArgument(by_name.name, by_pos.pos, by_name.typ, False)
   563	    return by_name if by_name is not None else by_pos

   583	def make_simplified_union(
   584	    items: Sequence[Type],
   585	    line: int = -1,
   586	    column: int = -1,
   587	    *,
   588	    keep_erased: bool = False,
   589	    contract_literals: bool = True,
   590	    handle_recursive: bool = True,
   591	) -> ProperType:
   592	    """Build union type with redundant union items removed.
   593
   594	    If only a single item remains, this may return a non-union type.
   595
   596	    Examples:
   597
   598	    * [int, str] -> Union[int, str]
   599	    * [int, object] -> object
   600	    * [int, int] -> int
   601	    * [int, Any] -> Union[int, Any] (Any types are not simplified away!)
   602	    * [Any, Any] -> Any
   603	    * [int, Union[bytes, str]] -> Union[int, bytes, str]
   604
   605	    Note: This must NOT be used during semantic analysis, since TypeInfos may not
   606	          be fully initialized.
   607
   608	    The keep_erased flag is used for type inference against union types
   609	    containing type variables. If set to True, keep all ErasedType items.
   610
   611	    The contract_literals flag indicates whether we need to contract literal types
   612	    back into a sum type. Set it to False when called by try_expanding_sum_type_
   613	    to_union().
   614	    """
   615	    # Step 1: expand all nested unions
   616	    items = flatten_nested_unions(items, handle_recursive=handle_recursive)
   617
   618	    # Step 2: fast path for single item
   619	    if len(items) == 1:
   620	        return get_proper_type(items[0])
   621
   622	    # Step 3: remove redundant unions
   623	    simplified_set: Sequence[Type] = _remove_redundant_union_items(items, keep_erased)
   624
   625	    # Step 4: If more than one literal exists in the union, try to simplify
   626	    if (
   627	        contract_literals
   628	        and sum(isinstance(get_proper_type(item), LiteralType) for item in simplified_set) > 1
   629	    ):
   630	        simplified_set = try_contracting_literals_in_union(simplified_set)
   631
   632	    result = get_proper_type(UnionType.make_union(simplified_set, line, column))
   633
   634	    nitems = len(items)
   635	    if nitems > 1 and (
   636	        nitems > 2 or not (type(items[0]) is NoneType or type(items[1]) is NoneType)
   637	    ):
   638	        # Step 5: At last, we erase any (inconsistent) extra attributes on instances.
   639
   640	        # Initialize with None instead of an empty set as a micro-optimization. The set
   641	        # is needed very rarely, so we try to avoid constructing it.
   642	        extra_attrs_set: set[ExtraAttrs] | None = None
   643	        for item in items:
   644	            instance = try_getting_instance_fallback(item)
   645	            if instance and instance.extra_attrs:
   646	                if extra_attrs_set is None:
   647	                    extra_attrs_set = {instance.extra_attrs}
   648	                else:
   649	                    extra_attrs_set.add(instance.extra_attrs)
   650
   651	        if extra_attrs_set is not None and len(extra_attrs_set) > 1:
   652	            fallback = try_getting_instance_fallback(result)
   653	            if fallback:
   654	                fallback.extra_attrs = None
   655
   656	    return result
   657

   659	def _remove_redundant_union_items(items: list[Type], keep_erased: bool) -> list[Type]:
   660	    from mypy.subtypes import is_proper_subtype
   661
   662	    # The first pass through this loop, we check if later items are subtypes of earlier items.
   663	    # The second pass through this loop, we check if earlier items are subtypes of later items
   664	    # (by reversing the remaining items)
   665	    for _direction in range(2):
   666	        new_items: list[Type] = []
   667	        # seen is a map from a type to its index in new_items
   668	        seen: dict[ProperType, int] = {}
   669	        unduplicated_literal_fallbacks: set[Instance] | None = None
   670	        for ti in items:
   671	            proper_ti = get_proper_type(ti)
   672
   673	            # UninhabitedType is always redundant
   674	            if isinstance(proper_ti, UninhabitedType):
   675	                continue
   676
   677	            duplicate_index = -1
   678	            # Quickly check if we've seen this type
   679	            if proper_ti in seen:
   680	                duplicate_index = seen[proper_ti]
   681	            elif (
   682	                isinstance(proper_ti, LiteralType)
   683	                and unduplicated_literal_fallbacks is not None
   684	                and proper_ti.fallback in unduplicated_literal_fallbacks
   685	            ):
   686	                # This is an optimisation for unions with many LiteralType
   687	                # We've already checked for exact duplicates. This means that any super type of
   688	                # the LiteralType must be a super type of its fallback. If we've gone through
   689	                # the expensive loop below and found no super type for a previous LiteralType
   690	                # with the same fallback, we can skip doing that work again and just add the type
   691	                # to new_items
   692	                pass
   693	            else:
   694	                # If not, check if we've seen a supertype of this type
   695	                for j, tj in enumerate(new_items):
   696	                    tj = get_proper_type(tj)
   697	                    # If tj is an Instance with a last_known_value, do not remove proper_ti
   698	                    # (unless it's an instance with the same last_known_value)
   699	                    if (
   700	                        isinstance(tj, Instance)
   701	                        and tj.last_known_value is not None
   702	                        and not (
   703	                            isinstance(proper_ti, Instance)
   704	                            and tj.last_known_value == proper_ti.last_known_value
   705	                        )
   706	                    ):
   707	                        continue
   708
   709	                    if is_proper_subtype(
   710	                        ti, tj, keep_erased_types=keep_erased, ignore_promotions=True
   711	                    ):
   712	                        duplicate_index = j
   713	                        break
   714	            if duplicate_index != -1:
   715	                # If deleted subtypes had more general truthiness, use that
   716	                orig_item = new_items[duplicate_index]
   717	                if not orig_item.can_be_true and ti.can_be_true:
   718	                    new_items[duplicate_index] = true_or_false(orig_item)
   719	                elif not orig_item.can_be_false and ti.can_be_false:
   720	                    new_items[duplicate_index] = true_or_false(orig_item)
   721	            else:
   722	                # We have a non-duplicate item, add it to new_items
   723	                seen[proper_ti] = len(new_items)
   724	                new_items.append(ti)
   725	                if isinstance(proper_ti, LiteralType):
   726	                    if unduplicated_literal_fallbacks is None:
   727	                        unduplicated_literal_fallbacks = set()
   728	                    unduplicated_literal_fallbacks.add(proper_ti.fallback)
   729
   730	        items = new_items
   731	        if len(items) <= 1:
   732	            break
   733	        items.reverse()
   734
   735	    return items

   739	    # For Enum literals the ret_type can change based on the Enum

   750
   751	    return None

   754	def true_only(t: Type) -> ProperType:
   755	    """
   756	    Restricted version of t with only True-ish values
   757	    """
   758	    t = get_proper_type(t)
   759
   760	    if not t.can_be_true:
   761	        # All values of t are False-ish, so there are no true values in it
   762	        return UninhabitedType(line=t.line, column=t.column)
   763	    elif not t.can_be_false:
   764	        # All values of t are already True-ish, so true_only is idempotent in this case
   765	        return t
   766	    elif isinstance(t, UnionType):
   767	        # The true version of a union type is the union of the true versions of its components
   768	        new_items = [true_only(item) for item in t.items]
   769	        can_be_true_items = [item for item in new_items if item.can_be_true]
   770	        return make_simplified_union(can_be_true_items, line=t.line, column=t.column)
   771	    else:
   772	        ret_type = _get_type_method_ret_type(t, name="__bool__") or _get_type_method_ret_type(
   773	            t, name="__len__"
   774	        )
   775
   776	        if ret_type and not ret_type.can_be_true:
   777	            return UninhabitedType(line=t.line, column=t.column)
   778
   779	        new_t = copy_type(t)
   780	        new_t.can_be_false = False
   781	        return new_t
