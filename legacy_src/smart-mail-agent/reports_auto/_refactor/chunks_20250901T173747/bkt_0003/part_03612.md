  1322	def fixture(
  1323	    fixture_function: FixtureFunction | None = None,
  1324	    *,
  1325	    scope: _ScopeName | Callable[[str, Config], _ScopeName] = "function",
  1326	    params: Iterable[object] | None = None,
  1327	    autouse: bool = False,
  1328	    ids: Sequence[object | None] | Callable[[Any], object | None] | None = None,
  1329	    name: str | None = None,
  1330	) -> FixtureFunctionMarker | FixtureFunctionDefinition:
  1331	    """Decorator to mark a fixture factory function.
  1332
  1333	    This decorator can be used, with or without parameters, to define a
  1334	    fixture function.
  1335
  1336	    The name of the fixture function can later be referenced to cause its
  1337	    invocation ahead of running tests: test modules or classes can use the
  1338	    ``pytest.mark.usefixtures(fixturename)`` marker.
  1339
  1340	    Test functions can directly use fixture names as input arguments in which
  1341	    case the fixture instance returned from the fixture function will be
  1342	    injected.
  1343
  1344	    Fixtures can provide their values to test functions using ``return`` or
  1345	    ``yield`` statements. When using ``yield`` the code block after the
  1346	    ``yield`` statement is executed as teardown code regardless of the test
  1347	    outcome, and must yield exactly once.
  1348
  1349	    :param scope:
  1350	        The scope for which this fixture is shared; one of ``"function"``
  1351	        (default), ``"class"``, ``"module"``, ``"package"`` or ``"session"``.
  1352
  1353	        This parameter may also be a callable which receives ``(fixture_name, config)``
  1354	        as parameters, and must return a ``str`` with one of the values mentioned above.
  1355
  1356	        See :ref:`dynamic scope` in the docs for more information.
  1357
  1358	    :param params:
  1359	        An optional list of parameters which will cause multiple invocations
  1360	        of the fixture function and all of the tests using it. The current
  1361	        parameter is available in ``request.param``.
  1362
  1363	    :param autouse:
  1364	        If True, the fixture func is activated for all tests that can see it.
  1365	        If False (the default), an explicit reference is needed to activate
  1366	        the fixture.
  1367
  1368	    :param ids:
  1369	        Sequence of ids each corresponding to the params so that they are
  1370	        part of the test id. If no ids are provided they will be generated
  1371	        automatically from the params.
  1372
  1373	    :param name:
  1374	        The name of the fixture. This defaults to the name of the decorated
  1375	        function. If a fixture is used in the same module in which it is
  1376	        defined, the function name of the fixture will be shadowed by the
  1377	        function arg that requests the fixture; one way to resolve this is to
  1378	        name the decorated function ``fixture_<fixturename>`` and then use
  1379	        ``@pytest.fixture(name='<fixturename>')``.
  1380	    """
  1381	    fixture_marker = FixtureFunctionMarker(
  1382	        scope=scope,
  1383	        params=tuple(params) if params is not None else None,
  1384	        autouse=autouse,
  1385	        ids=None if ids is None else ids if callable(ids) else tuple(ids),
  1386	        name=name,
  1387	        _ispytest=True,
  1388	    )
  1389
  1390	    # Direct decoration.
  1391	    if fixture_function:
  1392	        return fixture_marker(fixture_function)
  1393
  1394	    return fixture_marker

  1499	class FixtureManager:
  1500	    """pytest fixture definitions and information is stored and managed
  1501	    from this class.
  1502
  1503	    During collection fm.parsefactories() is called multiple times to parse
  1504	    fixture function definitions into FixtureDef objects and internal
  1505	    data structures.
  1506
  1507	    During collection of test functions, metafunc-mechanics instantiate
  1508	    a FuncFixtureInfo object which is cached per node/func-name.
  1509	    This FuncFixtureInfo object is later retrieved by Function nodes
  1510	    which themselves offer a fixturenames attribute.
  1511
  1512	    The FuncFixtureInfo object holds information about fixtures and FixtureDefs
  1513	    relevant for a particular function. An initial list of fixtures is
  1514	    assembled like this:
  1515
  1516	    - ini-defined usefixtures
  1517	    - autouse-marked fixtures along the collection chain up from the function
  1518	    - usefixtures markers at module/class/function level
  1519	    - test function funcargs
  1520
  1521	    Subsequently the funcfixtureinfo.fixturenames attribute is computed
  1522	    as the closure of the fixtures needed to setup the initial fixtures,
  1523	    i.e. fixtures needed by fixture functions themselves are appended
  1524	    to the fixturenames list.
  1525
  1526	    Upon the test-setup phases all fixturenames are instantiated, retrieved
  1527	    by a lookup of their FuncFixtureInfo.
  1528	    """
  1529
  1530	    def __init__(self, session: Session) -> None:
  1531	        self.session = session
  1532	        self.config: Config = session.config
  1533	        # Maps a fixture name (argname) to all of the FixtureDefs in the test
  1534	        # suite/plugins defined with this name. Populated by parsefactories().
  1535	        # TODO: The order of the FixtureDefs list of each arg is significant,
  1536	        #       explain.
  1537	        self._arg2fixturedefs: Final[dict[str, list[FixtureDef[Any]]]] = {}
  1538	        self._holderobjseen: Final[set[object]] = set()

  1540	        self._nodeid_autousenames: Final[dict[str, list[str]]] = {
  1541	            "": self.config.getini("usefixtures"),
  1542	        }
  1543	        session.config.pluginmanager.register(self, "funcmanage")

  1552
  1553	        If ``func`` is None, or if the item sets an attribute
  1554	        ``nofuncargs = True``, then ``func`` is not examined at all.

  1556	        :param node:
  1557	            The item requesting the fixtures.
  1558	        :param func:

  1623	    def getfixtureclosure(
  1624	        self,
  1625	        parentnode: nodes.Node,
  1626	        initialnames: tuple[str, ...],
  1627	        ignore_args: AbstractSet[str],
  1628	    ) -> tuple[list[str], dict[str, Sequence[FixtureDef[Any]]]]:
  1629	        # Collect the closure of all fixtures, starting with the given
  1630	        # fixturenames as the initial set.  As we have to visit all
  1631	        # factory definitions anyway, we also return an arg2fixturedefs
  1632	        # mapping so that the caller can reuse it and does not have
  1633	        # to re-discover fixturedefs again for each fixturename
  1634	        # (discovering matching fixtures for a given name/node is expensive).
  1635
  1636	        fixturenames_closure = list(initialnames)
  1637
  1638	        arg2fixturedefs: dict[str, Sequence[FixtureDef[Any]]] = {}
  1639	        lastlen = -1
  1640	        while lastlen != len(fixturenames_closure):
  1641	            lastlen = len(fixturenames_closure)
  1642	            for argname in fixturenames_closure:
  1643	                if argname in ignore_args:
  1644	                    continue
  1645	                if argname in arg2fixturedefs:
  1646	                    continue
  1647	                fixturedefs = self.getfixturedefs(argname, parentnode)
  1648	                if fixturedefs:
  1649	                    arg2fixturedefs[argname] = fixturedefs
  1650	                    for arg in fixturedefs[-1].argnames:
  1651	                        if arg not in fixturenames_closure:
  1652	                            fixturenames_closure.append(arg)
  1653
  1654	        def sort_by_scope(arg_name: str) -> Scope:
  1655	            try:
  1656	                fixturedefs = arg2fixturedefs[arg_name]
  1657	            except KeyError:
  1658	                return Scope.Function
  1659	            else:
  1660	                return fixturedefs[-1]._scope
  1661
  1662	        fixturenames_closure.sort(key=sort_by_scope, reverse=True)
  1663	        return fixturenames_closure, arg2fixturedefs

   653	@hookspec(firstresult=True)
   654	def pytest_runtest_protocol(item: Item, nextitem: Item | None) -> object | None:
   655	    """Perform the runtest protocol for a single test item.
   656
   657	    The default runtest protocol is this (see individual hooks for full details):
   658
   659	    - ``pytest_runtest_logstart(nodeid, location)``
   660
   661	    - Setup phase:
   662	        - ``call = pytest_runtest_setup(item)`` (wrapped in ``CallInfo(when="setup")``)
   663	        - ``report = pytest_runtest_makereport(item, call)``
   664	        - ``pytest_runtest_logreport(report)``
   665	        - ``pytest_exception_interact(call, report)`` if an interactive exception occurred
   666
   667	    - Call phase, if the setup passed and the ``setuponly`` pytest option is not set:
   668	        - ``call = pytest_runtest_call(item)`` (wrapped in ``CallInfo(when="call")``)
   669	        - ``report = pytest_runtest_makereport(item, call)``
   670	        - ``pytest_runtest_logreport(report)``
   671	        - ``pytest_exception_interact(call, report)`` if an interactive exception occurred
   672
   673	    - Teardown phase:
   674	        - ``call = pytest_runtest_teardown(item, nextitem)`` (wrapped in ``CallInfo(when="teardown")``)
   675	        - ``report = pytest_runtest_makereport(item, call)``
   676	        - ``pytest_runtest_logreport(report)``
   677	        - ``pytest_exception_interact(call, report)`` if an interactive exception occurred
   678
   679	    - ``pytest_runtest_logfinish(nodeid, location)``
   680
   681	    :param item: Test item for which the runtest protocol is performed.
   682	    :param nextitem: The scheduled-to-be-next test item (or None if this is the end my friend).
   683
   684	    Stops at first non-None result, see :ref:`firstresult`.
   685	    The return value is not used, but only stops further processing.
   686
   687	    Use in conftest plugins
   688	    =======================
   689
   690	    Any conftest file can implement this hook.
   691	    """
