    94	def run_setup(script_name: str, script_args: list[str]) -> bool:
    95	    """Run a setup script in a somewhat controlled environment.
    96
    97	    This is adapted from code in distutils and our goal here is that is
    98	    faster to not need to spin up a python interpreter to run it.
    99
   100	    We had to fork it because the real run_setup swallows errors
   101	    and KeyboardInterrupt with no way to recover them (!).
   102	    The real version has some extra features that we removed since
   103	    we weren't using them.
   104
   105	    Returns whether the setup succeeded.
   106	    """
   107	    save_argv = sys.argv.copy()
   108	    g = {"__file__": script_name}
   109	    try:
   110	        try:
   111	            sys.argv[0] = script_name
   112	            sys.argv[1:] = script_args
   113	            with open(script_name, "rb") as f:
   114	                exec(f.read(), g)
   115	        finally:
   116	            sys.argv = save_argv
   117	    except SystemExit as e:
   118	        # distutils converts KeyboardInterrupt into a SystemExit with
   119	        # "interrupted" as the argument. Convert it back so that
   120	        # pytest will exit instead of just failing the test.
   121	        if e.code == "interrupted":
   122	            raise KeyboardInterrupt from e
   123
   124	        return e.code == 0 or e.code is None
   125
   126	    return True

   438	def fix_native_line_number(message: str, fnam: str, delta: int) -> str:
   439	    """Update code locations in test case output to point to the .test file.
   440
   441	    The description of the test case is written to native.py, and line numbers
   442	    in test case output often are relative to native.py. This translates the
   443	    line numbers to be relative to the .test file that contains the test case
   444	    description, and also updates the file name to the .test file name.
   445
   446	    Args:
   447	        message: message to update
   448	        fnam: path of the .test file
   449	        delta: line number of the beginning of the test case in the .test file
   450
   451	    Returns updated message (or original message if we couldn't find anything).
   452	    """
   453	    fnam = os.path.basename(fnam)
   454	    message = re.sub(
   455	        r"native\.py:([0-9]+):", lambda m: "%s:%d:" % (fnam, int(m.group(1)) + delta), message
   456	    )
   457	    message = re.sub(
   458	        r'"native.py", line ([0-9]+),',
   459	        lambda m: '"%s", line %d,' % (fnam, int(m.group(1)) + delta),
   460	        message,
   461	    )
   462	    return message

    34	def assert_blobs_same(x: Any, y: Any, trail: tuple[Any, ...]) -> None:
    35	    """Compare two blobs of IR as best we can.
    36
    37	    FuncDecls, FuncIRs, and ClassIRs are compared by fullname to avoid
    38	    infinite recursion.
    39	    (More detailed comparisons should be done manually.)
    40
    41	    Types and signatures are compared using mypyc.sametype.
    42
    43	    Containers are compared recursively.
    44
    45	    Anything else is compared with ==.
    46
    47	    The `trail` argument is used in error messages.
    48	    """
    49
    50	    assert type(x) is type(y), (f"Type mismatch at {trail}", type(x), type(y))
    51	    if isinstance(x, (FuncDecl, FuncIR, ClassIR)):
    52	        assert x.fullname == y.fullname, f"Name mismatch at {trail}"
    53	    elif isinstance(x, dict):
    54	        assert len(x.keys()) == len(y.keys()), f"Keys mismatch at {trail}"
    55	        for (xk, xv), (yk, yv) in zip(x.items(), y.items()):
    56	            assert_blobs_same(xk, yk, trail + ("keys",))
    57	            assert_blobs_same(xv, yv, trail + (xk,))
    58	    elif isinstance(x, dict):
    59	        assert x.keys() == y.keys(), f"Keys mismatch at {trail}"
    60	        for k in x.keys():
    61	            assert_blobs_same(x[k], y[k], trail + (k,))
    62	    elif isinstance(x, Iterable) and not isinstance(x, (str, set)):
    63	        # Special case iterables to generate better assert error messages.
    64	        # We can't use this for sets since the ordering is unpredictable,
    65	        # and strings should be treated as atomic values.
    66	        for i, (xv, yv) in enumerate(zip(x, y)):
    67	            assert_blobs_same(xv, yv, trail + (i,))
    68	    elif isinstance(x, RType):
    69	        assert is_same_type(x, y), f"RType mismatch at {trail}"
    70	    elif isinstance(x, FuncSignature):
    71	        assert is_same_signature(x, y), f"Signature mismatch at {trail}"
    72	    else:
    73	        assert x == y, f"Value mismatch at {trail}"

     1	"""Simple copy propagation optimization.
     2
     3	Example input:
     4
     5	    x = f()
     6	    y = x
     7
     8	The register x is redundant and we can directly assign its value to y:
     9
    10	    y = f()
    11
    12	This can optimize away registers that are assigned to once.
    13	"""
    14
    15	from __future__ import annotations
    16
    17	from mypyc.ir.func_ir import FuncIR
    18	from mypyc.ir.ops import Assign, AssignMulti, LoadAddress, LoadErrorValue, Register, Value
    19	from mypyc.irbuild.ll_builder import LowLevelIRBuilder

    22	from mypyc.transform.ir_transform import IRTransform

    24
    25	def do_copy_propagation(fn: FuncIR, options: CompilerOptions) -> None:
    26	    """Perform copy propagation optimization for fn."""
    27
    28	    # Anything with an assignment count >1 will not be optimized
    29	    # here, as it would be require data flow analysis and we want to
    30	    # keep this simple and fast, at least until we've made data flow
    31	    # analysis much faster.
    32	    counts: dict[Value, int] = {}
    33	    replacements: dict[Value, Value] = {}
    34	    for arg in fn.arg_regs:
    35	        # Arguments are always assigned to initially
    36	        counts[arg] = 1
    37
    38	    for block in fn.blocks:
    39	        for op in block.ops:
    40	            if isinstance(op, Assign):
    41	                c = counts.get(op.dest, 0)
    42	                counts[op.dest] = c + 1
    43	                # Does this look like a supported assignment?
    44	                # TODO: Something needs LoadErrorValue assignments to be preserved?
    45	                if (
    46	                    c == 0
    47	                    and is_same_type(op.dest.type, op.src.type)
    48	                    and not isinstance(op.src, LoadErrorValue)
    49	                ):
    50	                    replacements[op.dest] = op.src
    51	                elif c == 1:
    52	                    # Too many assignments -- don't replace this one
    53	                    replacements.pop(op.dest, 0)
    54	            elif isinstance(op, AssignMulti):
    55	                # Copy propagation not supported for AssignMulti destinations
    56	                counts[op.dest] = 2
    57	                replacements.pop(op.dest, 0)
    58	            elif isinstance(op, LoadAddress):
    59	                # We don't support taking the address of an arbitrary Value,
    60	                # so we'll need to preserve the operands of LoadAddress.
    61	                if isinstance(op.src, Register):
    62	                    counts[op.src] = 2
    63	                    replacements.pop(op.src, 0)
    64
    65	    # Follow chains of propagation with more than one assignment.
    66	    for src, dst in list(replacements.items()):
    67	        if counts.get(dst, 0) > 1:
    68	            # Not supported
    69	            del replacements[src]
    70	        else:
    71	            while dst in replacements:
    72	                dst = replacements[dst]
    73	                if counts.get(dst, 0) > 1:
    74	                    # Not supported
    75	                    del replacements[src]
    76	        if src in replacements:
    77	            replacements[src] = dst
    78
    79	    builder = LowLevelIRBuilder(None, options)
    80	    transform = CopyPropagationTransform(builder, replacements)
    81	    transform.transform_blocks(fn.blocks)
    82	    fn.blocks = builder.blocks
    83

    94	        return self.add(op)

     2
     3	Example input:
     4
     5	  L1:
     6	    r0 = f()
     7	    b = r0
     8	    goto L3

    38	def do_flag_elimination(fn: FuncIR, options: CompilerOptions) -> None:
    39	    # Find registers that are used exactly once as source, and in a branch.
    40	    counts: dict[Register, int] = {}
    41	    branches: dict[Register, Branch] = {}
    42	    labels: dict[Register, BasicBlock] = {}
    43	    for block in fn.blocks:
    44	        for i, op in enumerate(block.ops):
    45	            for src in op.sources():
    46	                if isinstance(src, Register):
    47	                    counts[src] = counts.get(src, 0) + 1
    48	            if i == 0 and isinstance(op, Branch) and isinstance(op.value, Register):
    49	                branches[op.value] = op
    50	                labels[op.value] = block
    51
    52	    # Based on these we can find the candidate registers.
    53	    candidates: set[Register] = {
    54	        r for r in branches if counts.get(r, 0) == 1 and r not in fn.arg_regs
    55	    }
    56
    57	    # Remove candidates with invalid assignments.
    58	    for block in fn.blocks:
    59	        for i, op in enumerate(block.ops):
    60	            if isinstance(op, Assign) and op.dest in candidates:
    61	                next_op = block.ops[i + 1]
    62	                if not (isinstance(next_op, Goto) and next_op.label is labels[op.dest]):
    63	                    # Not right
    64	                    candidates.remove(op.dest)

    68	        builder, {x: y for x, y in branches.items() if x in candidates}
    69	    )
    70	    transform.transform_blocks(fn.blocks)
    71	    fn.blocks = builder.blocks
