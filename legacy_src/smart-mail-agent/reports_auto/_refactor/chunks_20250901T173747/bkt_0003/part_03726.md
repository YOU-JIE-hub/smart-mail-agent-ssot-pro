), we check
   756	            the flag to determine whether to append the argument to
   757	            the *args list or add it to the **kwargs dict. What a
   758	            mess!
   759
   760	            This is what really makes everything here such a tangle;
   761	            otherwise the *args and **kwargs code could be separated.
   762
   763	        The arguments has_star and has_star2 indicate whether the target function
   764	        takes an ARG_STAR and ARG_STAR2 argument, respectively.
   765	        (These will always be true when making a pycall, and be based
   766	        on the actual target signature for a native call.)
   767	        """
   768
   769	        star_result: Value | None = None
   770	        star2_result: Value | None = None
   771	        # We aggregate values that need to go into *args and **kwargs
   772	        # in these lists. Once all arguments are processed (in the
   773	        # happiest case), or we encounter an ARG_STAR/ARG_STAR2 or a
   774	        # nullable arg, then we create the list and/or dict.
   775	        star_values: list[Value] = []
   776	        star2_keys: list[Value] = []
   777	        star2_values: list[Value] = []
   778
   779	        seen_empty_reg: Register | None = None
   780
   781	        for value, kind, name in args:
   782	            if kind == ARG_STAR:
   783	                if star_result is None:
   784	                    star_result = self.new_list_op(star_values, line)
   785	                self.primitive_op(list_extend_op, [star_result, value], line)
   786	            elif kind == ARG_STAR2:
   787	                if star2_result is None:
   788	                    star2_result = self._create_dict(star2_keys, star2_values, line)
   789
   790	                self.call_c(dict_update_in_display_op, [star2_result, value], line=line)
   791	            else:
   792	                nullable = kind.is_optional()
   793	                maybe_pos = kind.is_positional() and has_star
   794	                maybe_named = kind.is_named() or (kind.is_optional() and name and has_star2)
   795
   796	                # If the argument is nullable, we need to create the
   797	                # relevant args/kwargs objects so that we can
   798	                # conditionally modify them.
   799	                if nullable:
   800	                    if maybe_pos and star_result is None:
   801	                        star_result = self.new_list_op(star_values, line)
   802	                    if maybe_named and star2_result is None:
   803	                        star2_result = self._create_dict(star2_keys, star2_values, line)
   804
   805	                # Easy cases: just collect the argument.
   806	                if maybe_pos and star_result is None:
   807	                    star_values.append(value)
   808	                    continue
   809
   810	                if maybe_named and star2_result is None:
   811	                    assert name is not None
   812	                    key = self.load_str(name)
   813	                    star2_keys.append(key)
   814	                    star2_values.append(value)
   815	                    continue
   816
   817	                # OK, anything that is nullable or *after* a nullable arg needs to be here
   818	                # TODO: We could try harder to avoid creating basic blocks in the common case
   819	                new_seen_empty_reg = seen_empty_reg
   820
   821	                out = BasicBlock()
   822	                if nullable:
   823	                    # If this is the first nullable positional arg we've seen, create
   824	                    # a register to track whether anything has been null.
   825	                    # (We won't *check* the register until the next argument, though.)
   826	                    if maybe_pos and not seen_empty_reg:
   827	                        new_seen_empty_reg = Register(bool_rprimitive)
   828	                        self.add(Assign(new_seen_empty_reg, self.false(), line))
   829
   830	                    skip = BasicBlock() if maybe_pos else out
   831	                    keep = BasicBlock()
   832	                    self.add(Branch(value, skip, keep, Branch.IS_ERROR))
   833	                    self.activate_block(keep)
   834
   835	                # If this could be positional or named and we /might/ have seen a missing
   836	                # positional arg, then we need to compile *both* a positional and named
   837	                # version! What a pain!
   838	                if maybe_pos and maybe_named and seen_empty_reg:
   839	                    pos_block, named_block = BasicBlock(), BasicBlock()
   840	                    self.add(Branch(seen_empty_reg, named_block, pos_block, Branch.BOOL))
   841	                else:
   842	                    pos_block = named_block = BasicBlock()
   843	                    self.goto(pos_block)
   844
   845	                if maybe_pos:
   846	                    self.activate_block(pos_block)
   847	                    assert star_result
   848	                    self.translate_special_method_call(
   849	                        star_result, "append", [value], result_type=None, line=line
   850	                    )
   851	                    self.goto(out)
   852
   853	                if maybe_named and (not maybe_pos or seen_empty_reg):
   854	                    self.activate_block(named_block)
   855	                    assert name is not None
   856	                    key = self.load_str(name)
   857	                    assert star2_result
   858	                    self.translate_special_method_call(
   859	                        star2_result, "__setitem__", [key, value], result_type=None, line=line
   860	                    )
   861	                    self.goto(out)
   862
   863	                if nullable and maybe_pos and new_seen_empty_reg:
   864	                    assert skip is not out
   865	                    self.activate_block(skip)
   866	                    self.add(Assign(new_seen_empty_reg, self.true(), line))
   867	                    self.goto(out)
   868
   869	                self.activate_block(out)
   870
   871	                seen_empty_reg = new_seen_empty_reg
   872
   873	        assert not (star_result or star_values) or has_star
   874	        assert not (star2_result or star2_values) or has_star2
   875	        if has_star:
   876	            # If we managed to make it this far without creating a
   877	            # *args list, then we can directly create a
   878	            # tuple. Otherwise create the tuple from the list.
   879	            if star_result is None:
   880	                star_result = self.new_tuple(star_values, line)
   881	            else:
   882	                star_result = self.primitive_op(list_tuple_op, [star_result], line)
   883	        if has_star2 and star2_result is None:
   884	            star2_result = self._create_dict(star2_keys, star2_values, line)
   885
   886	        return star_result, star2_result

   888	    def py_call(
   889	        self,
   890	        function: Value,
   891	        arg_values: list[Value],
   892	        line: int,
   893	        arg_kinds: list[ArgKind] | None = None,
   894	        arg_names: Sequence[str | None] | None = None,
   895	    ) -> Value:
   896	        """Call a Python function (non-native and slow).
   897
   898	        Use py_call_op or py_call_with_kwargs_op for Python function call.
   899	        """
   900	        result = self._py_vector_call(function, arg_values, line, arg_kinds, arg_names)
   901	        if result is not None:
   902	            return result
   903
   904	        # If all arguments are positional, we can use py_call_op.
   905	        if arg_kinds is None or all(kind == ARG_POS for kind in arg_kinds):
   906	            return self.call_c(py_call_op, [function] + arg_values, line)
   907
   908	        # Otherwise fallback to py_call_with_kwargs_op.
   909	        assert arg_names is not None
   910
   911	        pos_args_tuple, kw_args_dict = self._construct_varargs(
   912	            list(zip(arg_values, arg_kinds, arg_names)), line, has_star=True, has_star2=True
   913	        )
   914	        assert pos_args_tuple and kw_args_dict
   915
   916	        return self.call_c(py_call_with_kwargs_op, [function, pos_args_tuple, kw_args_dict], line)

   918	    def _py_vector_call(
   919	        self,
   920	        function: Value,
   921	        arg_values: list[Value],
   922	        line: int,
   923	        arg_kinds: list[ArgKind] | None = None,
   924	        arg_names: Sequence[str | None] | None = None,
   925	    ) -> Value | None:
   926	        """Call function using the vectorcall API if possible.
   927
   928	        Return the return value if successful. Return None if a non-vectorcall
   929	        API should be used instead.
   930	        """
   931	        # We can do this if all args are positional or named (no *args or **kwargs, not optional).
   932	        if arg_kinds is None or all(
   933	            not kind.is_star() and not kind.is_optional() for kind in arg_kinds
   934	        ):
   935	            if arg_values:
   936	                # Create a C array containing all arguments as boxed values.
   937	                coerced_args = [self.coerce(arg, object_rprimitive, line) for arg in arg_values]
   938	                arg_ptr = self.setup_rarray(object_rprimitive, coerced_args, object_ptr=True)
   939	            else:
   940	                arg_ptr = Integer(0, object_pointer_rprimitive)
   941	            num_pos = num_positional_args(arg_values, arg_kinds)
   942	            keywords = self._vectorcall_keywords(arg_names)
   943	            value = self.call_c(
   944	                py_vectorcall_op,
   945	                [function, arg_ptr, Integer(num_pos, c_size_t_rprimitive), keywords],
   946	                line,
   947	            )
   948	            if arg_values:
   949	                # Make sure arguments won't be freed until after the call.
   950	                # We need this because RArray doesn't support automatic
   951	                # memory management.
   952	                self.add(KeepAlive(coerced_args))
   953	            return value
   954	        return None

   956	    def _vectorcall_keywords(self, arg_names: Sequence[str | None] | None) -> Value:
   957	        """Return a reference to a tuple literal with keyword argument names.
   958
   959	        Return null pointer if there are no keyword arguments.
   960	        """
   961	        if arg_names:
   962	            kw_list = [name for name in arg_names if name is not None]
   963	            if kw_list:
   964	                return self.add(LoadLiteral(tuple(kw_list), object_rprimitive))
   965	        return Integer(0, object_rprimitive)

  1033	    def call(
  1034	        self,
  1035	        decl: FuncDecl,
  1036	        args: Sequence[Value],
  1037	        arg_kinds: list[ArgKind],
  1038	        arg_names: Sequence[str | None],
  1039	        line: int,
  1040	        *,
  1041	        bitmap_args: list[Register] | None = None,
  1042	    ) -> Value:
  1043	        """Call a native function.
  1044
  1045	        If bitmap_args is given, they override the values of (some) of the bitmap
  1046	        arguments used to track the presence of values for certain arguments. By
  1047	        default, the values of the bitmap arguments are inferred from args.
  1048	        """
  1049	        # Normalize args to positionals.
  1050	        args = self.native_args_to_positional(
  1051	            args, arg_kinds, arg_names, decl.sig, line, bitmap_args=bitmap_args
  1052	        )
  1053	        return self.add(Call(decl, args, line))

  1055	    def native_args_to_positional(
  1056	        self,
  1057	        args: Sequence[Value],
  1058	        arg_kinds: list[ArgKind],
  1059	        arg_names: Sequence[str | None],
  1060	        sig: FuncSignature,
  1061	        line: int,
  1062	        *,
  1063	        bitmap_args: list[Register] | None = None,
  1064	    ) -> list[Value]:
  1065	        """Prepare arguments for a native call.
  1066
  1067	        Given args/kinds/names and a target signature for a native call, map
  1068	        keyword arguments to their appropriate place in the argument list,
  1069	        fill in error values for unspecified default arguments,
  1070	        package arguments that will go into *args/**kwargs into a tuple/dict,
  1071	        and coerce arguments to the appropriate type.
  1072	        """
  1073
  1074	        sig_args = sig.args
  1075	        n = sig.num_bitmap_args
  1076	        if n:
  1077	            sig_args = sig_args[:-n]
  1078
  1079	        sig_arg_kinds = [arg.kind for arg in sig_args]
  1080	        sig_arg_names = [arg.name for arg in sig_args]
  1081
  1082	        concrete_kinds = [concrete_arg_kind(arg_kind) for arg_kind in arg_kinds]
  1083	        formal_to_actual = map_actuals_to_formals(
  1084	            concrete_kinds,
  1085	            arg_names,
  1086	            sig_arg_kinds,
  1087	            sig_arg_names,
  1088	            lambda n: AnyType(TypeOfAny.special_form),
  1089	        )
  1090
  1091	        # First scan for */** and construct those
  1092	        has_star = has_star2 = False
  1093	        star_arg_entries = []
  1094	        for lst, arg in zip(formal_to_actual, sig_args):
  1095	            if arg.kind.is_star():
  1096	                star_arg_entries.extend([(args[i], arg_kinds[i], arg_names[i]) for i in lst])
  1097	            has_star = has_star or arg.kind == ARG_STAR
  1098	            has_star2 = has_star2 or arg.kind == ARG_STAR2
  1099
  1100	        star_arg, star2_arg = self._construct_varargs(
  1101	            star_arg_entries, line, has_star=has_star, has_star2=has_star2
  1102	        )
  1103
  1104	        # Flatten out the arguments, loading error values for default
  1105	        # arguments, constructing tuples/dicts for star args, and
  1106	        # coercing everything to the expected type.
  1107	        output_args: list[Value] = []
  1108	        for lst, arg in zip(formal_to_actual, sig_args):
  1109	            if arg.kind == ARG_STAR:
  1110	                assert star_arg
  1111	                output_arg = star_arg
  1112	            elif arg.kind == ARG_STAR2:
  1113	                assert star2_arg
  1114	                output_arg = star2_arg
  1115	            elif not lst:
  1116	                if is_fixed_width_rtype(arg.type):
  1117	                    output_arg = Integer(0, arg.type)
  1118	                elif is_float_rprimitive(arg.type):
  1119	                    output_arg = Float(0.0)
  1120	                else:
  1121	                    output_arg = self.add(LoadErrorValue(arg.type, is_borrowed=True))
  1122	            else:
  1123	                base_arg = args[lst[0]]
  1124
  1125	                if arg_kinds[lst[0]].is_optional():
  1126	                    output_arg = self.coerce_nullable(base_arg, arg.type, line)
  1127	                else:
  1128	                    output_arg = self.coerce(base_arg, arg.type, line)
  1129
  1130	            output_args.append(output_arg)
  1131
  1132	        for i in reversed(range(n)):
  1133	            if bitmap_args and i < len(bitmap_args):
  1134	                # Use override provided by caller
  1135	                output_args.append(bitmap_args[i])
  1136	                continue
  1137	            # Infer values of bitmap args
  1138	            bitmap = 0
  1139	            c = 0
  1140	            for lst, arg in zip(formal_to_actual, sig_args):
  1141	                if arg.kind.is_optional() and arg.type.error_overlap:
  1142	                    if i * BITMAP_BITS <= c < (i + 1) * BITMAP_BITS:
  1143	                        if lst:
  1144	                            bitmap |= 1 << (c & (BITMAP_BITS - 1))
  1145	                    c += 1
  1146	            output_args.append(Integer(bitmap, bitmap_rprimitive))
  1147
  1148	        return output_args

  1150	    def gen_method_call(
  1151	        self,
  1152	        base: Value,
  1153	        name: str,
  1154	        arg_values: list[Value],
  1155	        result_type: RType | None,
  1156	        line: int,
  1157	        arg_kinds: list[ArgKind] | None = None,
  1158	        arg_names: list[str | None] | None = None,
  1159	        can_borrow: bool = False,
  1160	    ) -> Value:
  1161	        """Generate either a native or Python method call."""
  1162	        # If we have *args, then fallback to Python method call.
  1163	        if arg_kinds is not None and any(kind.is_star() for kind in arg_kinds):
  1164	            return self.py_method_call(base, name, arg_values, line, arg_kinds, arg_names)
  1165
  1166	        # If the base type is one of ours, do a MethodCall
  1167	        if (
  1168	            isinstance(base.type, RInstance)
  1169	            and base.type.class_ir.is_ext_class
  1170	            and not base.type.class_ir.builtin_base
  1171	        ):
  1172	            if base.type.class_ir.has_method(name):
  1173	                decl = base.type.class_ir.method_decl(name)
  1174	                if arg_kinds is None:
  1175	                    assert arg_names is None, "arg_kinds not present but arg_names is"
  1176	                    arg_kinds = [ARG_POS for _ in arg_values]
  1177	                    arg_names = [None for _ in arg_values]
  1178	                else:
  1179	                    assert arg_names is not None, "arg_kinds present but arg_names is not"
  1180
  1181	                # Normalize args to positionals.
  1182	                assert decl.bound_sig
  1183	                arg_values = self.native_args_to_positional(
  1184	                    arg_values, arg_kinds, arg_names, decl.bound_sig, line
  1185	                )
  1186	                return self.add(MethodCall(base, name, arg_values, line))
  1187	            elif base.type.class_ir.has_attr(name):
  1188	                function = self.add(GetAttr(base, name, line))
  1189	                return self.py_call(
  1190	                    function, arg_values, line, arg_kinds=arg_kinds, arg_names=arg_names
  1191	                )
  1192
  1193	        elif isinstance(base.type, RUnion):
  1194	            return self.union_method_call(
  1195	                base, base.type, name, arg_values, result_type, line, arg_kinds, arg_names
  1196	            )
  1197
  1198	        # Try to do a special-cased method call
  1199	        if not arg_kinds or arg_kinds == [ARG_POS] * len(arg_values):
  1200	            target = self.translate_special_method_call(
  1201	                base, name, arg_values, result_type, line, can_borrow=can_borrow
  1202	            )
  1203	            if target:
  1204	                return target
  1205
  1206	        # Fall back to Python method call
  1207	        return self.py_method_call(base, name, arg_values, line, arg_kinds, arg_names)

  1311	    def binary_op(self, lreg: Value, rreg: Value, op: str, line: int) -> Value:
  1312	        """Perform a binary operation.
  1313
  1314	        Generate specialized operations based on operand types, with a fallback
  1315	        to generic operations.
  1316	        """
  1317	        ltype = lreg.type
  1318	        rtype = rreg.type
  1319
  1320	        # Special case tuple comparison here so that nested tuples can be supported
  1321	        if isinstance(ltype, RTuple) and isinstance(rtype, RTuple) and op in ("==", "!="):
  1322	            return self.compare_tuples(lreg, rreg, op, line)
  1323
  1324	        # Special case == and != when we can resolve the method call statically
  1325	        if op in ("==", "!="):
  1326	            value = self.translate_eq_cmp(lreg, rreg, op, line)
  1327	            if value is not None:
  1328	                return value
  1329
  1330	        # Special case various ops
  1331	        if op in ("is", "is not"):
  1332	            return self.translate_is_op(lreg, rreg, op, line)
  1333	        # TODO: modify 'str' to use same interface as 'compare_bytes' as it avoids
  1334	        # call to PyErr_Occurred()
  1335	        if is_str_rprimitive(ltype) and is_str_rprimitive(rtype) and op in ("==", "!="):
  1336	            return self.compare_strings(lreg, rreg, op, line)
  1337	        if is_bytes_rprimitive(ltype) and is_bytes_rprimitive(rtype) and op in ("==", "!="):
  1338	            return self.compare_bytes(lreg, rreg, op, line)
  1339	        if is_bool_rprimitive(ltype) and is_bool_rprimitive(rtype) and op in BOOL_BINARY_OPS:
  1340	            if op in ComparisonOp.signed_ops:
  1341	                return self.bool_comparison_op(lreg, rreg, op, line)
  1342	            else:
  1343	                return self.bool_bitwise_op(lreg, rreg, op[0], line)
  1344	        if isinstance(rtype, RInstance) and op in ("in", "not in"):
  1345	            return self.translate_instance_contains(rreg, lreg, op, line)
  1346	        if is_fixed_width_rtype(ltype):
  1347	            if op in FIXED_WIDTH_INT_BINARY_OPS:
  1348	                op = op.removesuffix("=")
  1349	                if op != "//":
  1350	                    op_id = int_op_to_id[op]
  1351	                else:
  1352	                    op_id = IntOp.DIV
  1353	                if is_bool_rprimitive(rtype) or is_bit_rprimitive(rtype):
  1354	                    rreg = self.coerce(rreg, ltype, line)
  1355	                    rtype = ltype
  1356	                if is_fixed_width_rtype(rtype) or is_tagged(rtype):
  1357	                    return self.fixed_width_int_op(ltype, lreg, rreg, op_id, line)
  1358	                if isinstance(rreg, Integer):
  1359	                    return self.fixed_width_int_op(
  1360	                        ltype, lreg, self.coerce(rreg, ltype, line), op_id, line
  1361	                    )
  1362	            elif op in ComparisonOp.signed_ops:
  1363	                if is_int_rprimitive(rtype):
  1364	                    rreg = self.coerce_int_to_fixed_width(rreg, ltype, line)
  1365	                elif is_bool_rprimitive(rtype) or is_bit_rprimitive(rtype):
  1366	                    rreg = self.coerce(rreg, ltype, line)
  1367	                op_id = ComparisonOp.signed_ops[op]
  1368	                if is_fixed_width_rtype(rreg.type):
  1369	                    return self.comparison_op(lreg, rreg, op_id, line)
  1370	                if isinstance(rreg, Integer):
  1371	                    return self.comparison_op(lreg, self.coerce(rreg, ltype, line), op_id, line)
  1372	        elif is_fixed_width_rtype(rtype):
  1373	            if op in FIXED_WIDTH_INT_BINARY_OPS:
  1374	                op = op.removesuffix("=")
  1375	                if op != "//":
  1376	                    op_id = int_op_to_id[op]
  1377	                else:
  1378	                    op_id = IntOp.DIV
  1379	                if isinstance(lreg, Integer):
  1380	                    return self.fixed_width_int_op(
  1381	                        rtype, self.coerce(lreg, rtype, line), rreg, op_id, line
  1382	                    )
  1383	                if is_tagged(ltype):
  1384	                    return self.fixed_width_int_op(rtype, lreg, rreg, op_id, line)
  1385	                if is_bool_rprimitive(ltype) or is_bit_rprimitive(ltype):
  1386	                    lreg = self.coerce(lreg, rtype, line)
  1387	                    return self.fixed_width_int_op(rtype, lreg, rreg, op_id, line)
  1388	            elif op in ComparisonOp.signed_ops:
  1389	                if is_int_rprimitive(ltype):
  1390	                    lreg = self.coerce_int_to_fixed_width(lreg, rtype, line)
  1391	                elif is_bool_rprimitive(ltype) or is_bit_rprimitive(ltype):
  1392	                    lreg = self.coerce(lreg, rtype, line)
  1393	                op_id = ComparisonOp.signed_ops[op]
  1394	                if isinstance(lreg, Integer):
  1395	                    return self.comparison_op(self.coerce(lreg, rtype, line), rreg, op_id, line)
  1396	                if is_fixed_width_rtype(lreg.type):
  1397	                    return self.comparison_op(lreg, rreg, op_id, line)
  1398
  1399	        if is_float_rprimitive(ltype) or is_float_rprimitive(rtype):
  1400	            if isinstance(lreg, Integer):
  1401	                lreg = Float(float(lreg.numeric_value()))
  1402	            elif isinstance(rreg, Integer):
  1403	                rreg = Float(float(rreg.numeric_value()))
  1404	            elif is_int_rprimitive(lreg.type):
  1405	                lreg = self.int_to_float(lreg, line)
  1406	            elif is_int_rprimitive(rreg.type):
  1407	                rreg = self.int_to_float(rreg, line)
  1408	            if is_float_rprimitive(lreg.type) and is_float_rprimitive(rreg.type):
  1409	                if op in float_comparison_op_to_id:
  1410	                    return self.compare_floats(lreg, rreg, float_comparison_op_to_id[op], line)
  1411	                if op.endswith("="):
  1412	                    base_op = op[:-1]
  1413	                else:
  1414	                    base_op = op
  1415	                if base_op in float_op_to_id:
  1416	                    return self.float_op(lreg, rreg, base_op, line)
  1417
  1418	        dunder_op = self.dunder_op(lreg, rreg, op, line)
  1419	        if dunder_op:
  1420	            return dunder_op
  1421
  1422	        primitive_ops_candidates = binary_ops.get(op, [])
  1423	        target = self.matching_primitive_op(primitive_ops_candidates, [lreg, rreg], line)
  1424	        assert target, "Unsupported binary operation: %s" % op
  1425	        return target

  1427	    def dunder_op(self, lreg: Value, rreg: Value | None, op: str, line: int) -> Value | None:
  1428	        """
  1429	        Dispatch a dunder method if applicable.
  1430	        For example for `a + b` it will use `a.__add__(b)` which can lead to higher performance
  1431	        due to the fact that the method could be already compiled and optimized instead of going
  1432	        all the way through `PyNumber_Add(a, b)` python api (making a jump into the python DL).
  1433	        """
  1434	        ltype = lreg.type
  1435	        if not isinstance(ltype, RInstance):
  1436	            return None
  1437
  1438	        method_name = op_methods.get(op) if rreg else unary_op_methods.get(op)
  1439	        if method_name is None:
  1440	            return None
  1441
  1442	        if not ltype.class_ir.has_method(method_name):
  1443	            return None
  1444
  1445	        decl = ltype.class_ir.method_decl(method_name)
  1446	        if not rreg and len(decl.sig.args) != 1:
  1447	            return None
  1448
  1449	        if rreg and (len(decl.sig.args) != 2 or not is_subtype(rreg.type, decl.sig.args[1].type)):
  1450	            return None
  1451
  1452	        if rreg and is_subtype(not_implemented_op.type, decl.sig.ret_type):
  1453	            # If the method is able to return NotImplemented, we should not optimize it.
  1454	            # We can just let go so it will be handled through the python api.
  1455	            return None
  1456
  1457	        args = [rreg] if rreg else []
  1458	        return self.gen_method_call(lreg, method_name, args, decl.sig.ret_type, line)

  1504	    def compare_tuples(self, lhs: Value, rhs: Value, op: str, line: int = -1) -> Value:
  1505	        """Compare two tuples item by item"""
  1506	        # type cast to pass mypy check
  1507	        assert isinstance(lhs.type, RTuple) and isinstance(rhs.type, RTuple)
  1508	        equal = True if op == "==" else False
  1509	        result = Register(bool_rprimitive)
  1510	        # empty tuples
  1511	        if len(lhs.type.types) == 0 and len(rhs.type.types) == 0:
  1512	            self.add(Assign(result, self.true() if equal else self.false(), line))
  1513	            return result
  1514	        length = len(lhs.type.types)
  1515	        false_assign, true_assign, out = BasicBlock(), BasicBlock(), BasicBlock()
  1516	        check_blocks = [BasicBlock() for _ in range(length)]
  1517	        lhs_items = [self.add(TupleGet(lhs, i, line)) for i in range(length)]
  1518	        rhs_items = [self.add(TupleGet(rhs, i, line)) for i in range(length)]
  1519
  1520	        if equal:
  1521	            early_stop, final = false_assign, true_assign
  1522	        else:
  1523	            early_stop, final = true_assign, false_assign
  1524
  1525	        for i in range(len(lhs.type.types)):
  1526	            if i != 0:
  1527	                self.activate_block(check_blocks[i])
  1528	            lhs_item = lhs_items[i]
  1529	            rhs_item = rhs_items[i]
  1530	            compare = self.binary_op(lhs_item, rhs_item, op, line)
  1531	            # Cast to bool if necessary since most types uses comparison returning a object type
  1532	            # See generic_ops.py for more information
  1533	            if not is_bool_rprimitive(compare.type):
  1534	                compare = self.primitive_op(bool_op, [compare], line)
  1535	            if i < len(lhs.type.types) - 1:
  1536	                branch = Branch(compare, early_stop, check_blocks[i + 1], Branch.BOOL)
  1537	            else:
  1538	                branch = Branch(compare, early_stop, final, Branch.BOOL)
  1539	            # if op is ==, we branch on false, else branch on true
  1540	            branch.negated = equal
  1541	            self.add(branch)
  1542	        self.activate_block(false_assign)
  1543	        self.add(Assign(result, self.false(), line))
  1544	        self.goto(out)
  1545	        self.activate_block(true_assign)
  1546	        self.add(Assign(result, self.true(), line))
  1547	        self.goto_and_activate(out)
  1548	        return result

  1577	    def unary_op(self, value: Value, expr_op: str, line: int) -> Value:
  1578	        typ = value.type
  1579	        if is_bool_rprimitive(typ) or is_bit_rprimitive(typ):
  1580	            if expr_op == "not":
  1581	                return self.unary_not(value, line)
  1582	            if expr_op == "+":
  1583	                return value
  1584	        if is_fixed_width_rtype(typ):
  1585	            if expr_op == "-":
  1586	                # Translate to '0 - x'
  1587	                return self.int_op(typ, Integer(0, typ), value, IntOp.SUB, line)
  1588	            elif expr_op == "~":
  1589	                if typ.is_signed:
  1590	                    # Translate to 'x ^ -1'
  1591	                    return self.int_op(typ, value, Integer(-1, typ), IntOp.XOR, line)
  1592	                else:
  1593	                    # Translate to 'x ^ 0xff...'
  1594	                    mask = (1 << (typ.size * 8)) - 1
  1595	                    return self.int_op(typ, value, Integer(mask, typ), IntOp.XOR, line)
  1596	            elif expr_op == "+":
  1597	                return value
  1598	        if is_float_rprimitive(typ):
  1599	            if expr_op == "-":
  1600	                return self.add(FloatNeg(value, line))
  1601	            elif expr_op == "+":
  1602	                return value
  1603
  1604	        if isinstance(value, Integer):
  1605	            # TODO: Overflow? Unsigned?
  1606	            num = value.value
  1607	            if is_short_int_rprimitive(typ):
  1608	                num >>= 1
  1609	            return Integer(-num, typ, value.line)
  1610	        if is_tagged(typ) and expr_op == "+":
  1611	            return value
  1612	        if isinstance(value, Float):
  1613	            return Float(-value.value, value.line)
  1614	        if isinstance(typ, RInstance):
  1615	            result = self.dunder_op(value, None, expr_op, line)
  1616	            if result is not None:
  1617	                return result
  1618	        primitive_ops_candidates = unary_ops.get(expr_op, [])
  1619	        target = self.matching_primitive_op(primitive_ops_candidates, [value], line)
  1620	        assert target, "Unsupported unary operation: %s" % expr_op
  1621	        return target

  1664	    def new_list_op(self, values: list[Value], line: int) -> Value:

  1667	            return self.call_c(list_build_op, length + values, line)
  1668
  1669	        # If the length of the list is less than the threshold,
  1670	        # LIST_BUILDING_EXPANSION_THRESHOLD, we directly expand the
  1671	        # for-loop and inline the SetMem operation, which is faster
  1672	        # than list_build_op, however generates more code.

  1680	                buf_init_item, [ob_item_base, Integer(i, c_pyssize_t_rprimitive), args[i]], line
  1681	            )

  1708	        # left_body takes the value of the left side, right_body the right
  1709	        left_body, right_body, next_block = BasicBlock(), BasicBlock(), BasicBlock()
  1710	        # true_body is taken if the left is true, false_body if it is false.
  1711	        # For 'and' the value is the right side if the left is true, and for 'or'
  1712	        # it is the right side if the left is false.
  1713	        true_body, false_body = (right_body, left_body) if op == "and" else (left_body, right_body)
  1714
  1715	        left_value = left()
  1716	        self.add_bool_branch(left_value, true_body, false_body)
  1717
  1718	        self.activate_block(left_body)
  1719	        left_coerced = self.coerce(left_value, expr_type, line)
  1720	        self.add(Assign(target, left_coerced))
  1721	        self.goto(next_block)
  1722
  1723	        self.activate_block(right_body)
  1724	        right_value = right()
  1725	        right_coerced = self.coerce(right_value, expr_type, line)
  1726	        self.add(Assign(target, right_coerced))
  1727	        self.goto(next_block)
  1728
  1729	        self.activate_block(next_block)
  1730	        return target

  1732	    def bool_value(self, value: Value) -> Value:
  1733	        """Return bool(value).
  1734
  1735	        The result type can be bit_rprimitive or bool_rprimitive.
  1736	        """
  1737	        if is_bool_rprimitive(value.type) or is_bit_rprimitive(value.type):
  1738	            result = value
  1739	        elif is_runtime_subtype(value.type, int_rprimitive):
  1740	            zero = Integer(0, short_int_rprimitive)
  1741	            result = self.comparison_op(value, zero, ComparisonOp.NEQ, value.line)
  1742	        elif is_fixed_width_rtype(value.type):
  1743	            zero = Integer(0, value.type)
  1744	            result = self.add(ComparisonOp(value, zero, ComparisonOp.NEQ))
  1745	        elif is_same_type(value.type, str_rprimitive):
  1746	            result = self.call_c(str_check_if_true, [value], value.line)
  1747	        elif is_same_type(value.type, list_rprimitive) or is_same_type(
  1748	            value.type, dict_rprimitive
  1749	        ):
  1750	            length = self.builtin_len(value, value.line)
  1751	            zero = Integer(0)
  1752	            result = self.binary_op(length, zero, "!=", value.line)
  1753	        elif (
  1754	            isinstance(value.type, RInstance)
  1755	            and value.type.class_ir.is_ext_class
  1756	            and value.type.class_ir.has_method("__bool__")
  1757	        ):
  1758	            # Directly call the __bool__ method on classes that have it.
  1759	            result = self.gen_method_call(value, "__bool__", [], bool_rprimitive, value.line)
  1760	        elif is_float_rprimitive(value.type):
  1761	            result = self.compare_floats(value, Float(0.0), FloatComparisonOp.NEQ, value.line)
  1762	        else:
  1763	            value_type = optional_value_type(value.type)
  1764	            if value_type is not None:
  1765	                not_none = self.translate_is_op(value, self.none_object(), "is not", value.line)
  1766	                always_truthy = False
  1767	                if isinstance(value_type, RInstance):
  1768	                    # check whether X.__bool__ is always just the default (object.__bool__)
  1769	                    if not value_type.class_ir.has_method(
  1770	                        "__bool__"
  1771	                    ) and value_type.class_ir.is_method_final("__bool__"):
  1772	                        always_truthy = True
  1773
  1774	                if always_truthy:
  1775	                    result = not_none
  1776	                else:
  1777	                    # "X | None" where X may be falsey and requires a check
  1778	                    result = Register(bit_rprimitive)
  1779	                    true, false, end = BasicBlock(), BasicBlock(), BasicBlock()
  1780	                    branch = Branch(not_none, true, false, Branch.BOOL)
  1781	                    self.add(branch)
  1782	                    self.activate_block(true)
  1783	                    # unbox_or_cast instead of coerce because we want the
  1784	                    # type to change even if it is a subtype.
  1785	                    remaining = self.unbox_or_cast(value, value_type, value.line)
  1786	                    as_bool = self.bool_value(remaining)
  1787	                    self.add(Assign(result, as_bool))
  1788	                    self.goto(end)
  1789	                    self.activate_block(false)
  1790	                    self.add(Assign(result, Integer(0, bit_rprimitive)))
  1791	                    self.goto(end)
  1792	                    self.activate_block(end)
  1793	            else:
  1794	                result = self.primitive_op(bool_op, [value], value.line)
  1795	        return result

  1797	    def add_bool_branch(self, value: Value, true: BasicBlock, false: BasicBlock) -> None:
  1798	        opt_value_type = optional_value_type(value.type)
  1799	        if opt_value_type is None:
  1800	            bool_value = self.bool_value(value)
  1801	            self.add(Branch(bool_value, true, false, Branch.BOOL))
  1802	        else:
  1803	            # Special-case optional types
  1804	            is_none = self.translate_is_op(value, self.none_object(), "is not", value.line)
  1805	            branch = Branch(is_none, true, false, Branch.BOOL)
  1806	            self.add(branch)
  1807	            always_truthy = False
  1808	            if isinstance(opt_value_type, RInstance):
  1809	                # check whether X.__bool__ is always just the default (object.__bool__)
  1810	                if not opt_value_type.class_ir.has_method(

  1812	                ) and opt_value_type.class_ir.is_method_final("__bool__"):

  1814
  1815	            if not always_truthy:

  1824	    def call_c(
  1825	        self,
  1826	        desc: CFunctionDescription,
  1827	        args: list[Value],
  1828	        line: int,
  1829	        result_type: RType | None = None,
  1830	    ) -> Value:
  1831	        """Call function using C/native calling convention (not a Python callable)."""
  1832	        # Handle void function via singleton RVoid instance
  1833	        coerced = []
  1834	        # Coerce fixed number arguments
  1835	        for i in range(min(len(args), len(desc.arg_types))):
  1836	            formal_type = desc.arg_types[i]
  1837	            arg = args[i]
  1838	            arg = self.coerce(arg, formal_type, line)
  1839	            coerced.append(arg)
  1840	        # Reorder args if necessary
  1841	        if desc.ordering is not None:
  1842	            assert desc.var_arg_type is None
  1843	            coerced = [coerced[i] for i in desc.ordering]
  1844	        # Coerce any var_arg
  1845	        var_arg_idx = -1
  1846	        if desc.var_arg_type is not None:
  1847	            var_arg_idx = len(desc.arg_types)
  1848	            for i in range(len(desc.arg_types), len(args)):
  1849	                arg = args[i]
  1850	                arg = self.coerce(arg, desc.var_arg_type, line)
  1851	                coerced.append(arg)
  1852	        # Add extra integer constant if any
  1853	        for item in desc.extra_int_constants:
  1854	            val, typ = item
  1855	            extra_int_constant = Integer(val, typ, line)
  1856	            coerced.append(extra_int_constant)
  1857	        error_kind = desc.error_kind
  1858	        if error_kind == ERR_NEG_INT:
  1859	            # Handled with an explicit comparison
  1860	            error_kind = ERR_NEVER
  1861	        target = self.add(
  1862	            CallC(
  1863	                desc.c_function_name,
  1864	                coerced,
  1865	                desc.return_type,
  1866	                desc.steals,
  1867	                desc.is_borrowed,
  1868	                error_kind,
  1869	                line,
  1870	                var_arg_idx,
  1871	                is_pure=desc.is_pure,
  1872	            )
  1873	        )
  1874	        if desc.is_borrowed:
  1875	            # If the result is borrowed, force the arguments to be
  1876	            # kept alive afterwards, as otherwise the result might be
  1877	            # immediately freed, at the risk of a dangling pointer.
  1878	            for arg in coerced:
  1879	                if not isinstance(arg, (Integer, LoadLiteral)):
  1880	                    self.keep_alives.append(arg)
  1881	        if desc.error_kind == ERR_NEG_INT:
  1882	            comp = ComparisonOp(target, Integer(0, desc.return_type, line), ComparisonOp.SGE, line)
  1883	            comp.error_kind = ERR_FALSE
  1884	            self.add(comp)
  1885
  1886	        if desc.truncated_type is None:
  1887	            result = target
  1888	        else:
  1889	            truncate = self.add(Truncate(target, desc.truncated_type))
  1890	            result = truncate
  1891	        if result_type and not is_runtime_subtype(result.type, result_type):
  1892	            if is_none_rprimitive(result_type):
  1893	                # Special case None return. The actual result may actually be a bool
  1894	                # and so we can't just coerce it.
  1895	                result = self.none()
  1896	            else:
  1897	                result = self.coerce(target, result_type, line, can_borrow=desc.is_borrowed)
  1898	        return result

  2037	    def float_op(self, lhs: Value, rhs: Value, op: str, line: int) -> Value:
  2038	        """Generate a native float binary arithmetic operation.
  2039
  2040	        This follows Python semantics (e.g. raise exception on division by zero).
  2041	        Add a FloatOp directly if you want low-level semantics.
  2042
  2043	        Args:
  2044	            op: Binary operator (e.g. '+' or '*')
  2045	        """
  2046	        op_id = float_op_to_id[op]
  2047	        if op_id in (FloatOp.DIV, FloatOp.MOD):
  2048	            if not (isinstance(rhs, Float) and rhs.value != 0.0):
  2049	                c = self.compare_floats(rhs, Float(0.0), FloatComparisonOp.EQ, line)
  2050	                err, ok = BasicBlock(), BasicBlock()

  2052	                self.activate_block(err)
  2053	                if op_id == FloatOp.DIV:
  2054	                    msg = "float division by zero"
  2055	                else:

  2067	        """Perform x % y on floats using Python semantics."""
  2068	        mod = self.add(FloatOp(lhs, rhs, FloatOp.MOD, line))
