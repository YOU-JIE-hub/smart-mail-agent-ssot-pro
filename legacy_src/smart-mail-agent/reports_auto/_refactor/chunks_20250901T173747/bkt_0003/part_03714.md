  1029	def try_expanding_sum_type_to_union(typ: Type, target_fullname: str) -> ProperType:
  1030	    """Attempts to recursively expand any enum Instances with the given target_fullname
  1031	    into a Union of all of its component LiteralTypes.
  1032
  1033	    For example, if we have:
  1034
  1035	        class Color(Enum):
  1036	            RED = 1
  1037	            BLUE = 2

  1041	            SUCCESS = 1
  1042	            FAILURE = 2
  1043	            UNKNOWN = 3
  1044
  1045	    ...and if we call `try_expanding_sum_type_to_union(Union[Color, Status], 'module.Color')`,
  1046	    this function will return Literal[Color.RED, Color.BLUE, Color.YELLOW, Status].
  1047	    """
  1048	    typ = get_proper_type(typ)
  1049
  1050	    if isinstance(typ, UnionType):
  1051	        items = [
  1052	            try_expanding_sum_type_to_union(item, target_fullname) for item in typ.relevant_items()
  1053	        ]
  1054	        return make_simplified_union(items, contract_literals=False)
  1055
  1056	    if isinstance(typ, Instance) and typ.type.fullname == target_fullname:
  1057	        if typ.type.fullname == "builtins.bool":
  1058	            items = [LiteralType(True, typ), LiteralType(False, typ)]
  1059	            return make_simplified_union(items, contract_literals=False)
  1060
  1061	        if typ.type.is_enum:
  1062	            items = [LiteralType(name, typ) for name in typ.type.enum_members]
  1063	            if not items:
  1064	                return typ
  1065	            return make_simplified_union(items, contract_literals=False)

  1070	def try_contracting_literals_in_union(types: Sequence[Type]) -> list[ProperType]:
  1071	    """Contracts any literal types back into a sum type if possible.
  1072
  1073	    Requires a flattened union and does not descend into children.
  1074
  1075	    Will replace the first instance of the literal with the sum type and
  1076	    remove all others.
  1077
  1078	    If we call `try_contracting_union(Literal[Color.RED, Color.BLUE, Color.YELLOW])`,
  1079	    this function will return Color.
  1080
  1081	    We also treat `Literal[True, False]` as `bool`.
  1082	    """
  1083	    proper_types = [get_proper_type(typ) for typ in types]
  1084	    sum_types: dict[str, tuple[set[Any], list[int]]] = {}
  1085	    marked_for_deletion = set()
  1086	    for idx, typ in enumerate(proper_types):
  1087	        if isinstance(typ, LiteralType):
  1088	            fullname = typ.fallback.type.fullname
  1089	            if typ.fallback.type.is_enum or isinstance(typ.value, bool):
  1090	                if fullname not in sum_types:
  1091	                    sum_types[fullname] = (
  1092	                        (
  1093	                            set(typ.fallback.type.enum_members)
  1094	                            if typ.fallback.type.is_enum
  1095	                            else {True, False}
  1096	                        ),
  1097	                        [],
  1098	                    )
  1099	                literals, indexes = sum_types[fullname]
  1100	                literals.discard(typ.value)
  1101	                indexes.append(idx)
  1102	                if not literals:
  1103	                    first, *rest = indexes
  1104	                    proper_types[first] = typ.fallback
  1105	                    marked_for_deletion |= set(rest)
  1106	    return list(
  1107	        itertools.compress(
  1108	            proper_types, [(i not in marked_for_deletion) for i in range(len(proper_types))]
  1109	        )
  1110	    )

   238	    # when inferring the type of logic expressions like `x and y`.

   240	    # For example:
   241	    #   * the literal `False` can't be true while `True` can.

   298	class TypeAliasType(Type):
   299	    """A type alias to another type.
   300
   301	    To support recursive type aliases we don't immediately expand a type alias
   302	    during semantic analysis, but create an instance of this type that records the target alias
   303	    definition node (mypy.nodes.TypeAlias) and type arguments (for generic aliases).
   304
   305	    This is very similar to how TypeInfo vs Instance interact, where a recursive class-based
   306	    structure like
   307	        class Node:
   308	            value: int
   309	            children: List[Node]
   310	    can be represented in a tree-like manner.
   311	    """
   312
   313	    __slots__ = ("alias", "args", "type_ref")
   314
   315	    def __init__(
   316	        self,
   317	        alias: mypy.nodes.TypeAlias | None,
   318	        args: list[Type],
   319	        line: int = -1,
   320	        column: int = -1,
   321	    ) -> None:
   322	        super().__init__(line, column)
   323	        self.alias = alias
   324	        self.args = args
   325	        self.type_ref: str | None = None
   326
   327	    def _expand_once(self) -> Type:
   328	        """Expand to the target type exactly once.
   329
   330	        This doesn't do full expansion, i.e. the result can contain another
   331	        (or even this same) type alias. Use this internal helper only when really needed,
   332	        its public wrapper mypy.types.get_proper_type() is preferred.
   333	        """
   334	        assert self.alias is not None
   335	        if self.alias.no_args:
   336	            # We know that no_args=True aliases like L = List must have an instance
   337	            # as their target.
   338	            assert isinstance(self.alias.target, Instance)  # type: ignore[misc]
   339	            return self.alias.target.copy_modified(args=self.args)
   340
   341	        # TODO: this logic duplicates the one in expand_type_by_instance().
   342	        if self.alias.tvar_tuple_index is None:
   343	            mapping = {v.id: s for (v, s) in zip(self.alias.alias_tvars, self.args)}
   344	        else:
   345	            prefix = self.alias.tvar_tuple_index
   346	            suffix = len(self.alias.alias_tvars) - self.alias.tvar_tuple_index - 1
   347	            start, middle, end = split_with_prefix_and_suffix(tuple(self.args), prefix, suffix)
   348	            tvar = self.alias.alias_tvars[prefix]
   349	            assert isinstance(tvar, TypeVarTupleType)
   350	            mapping = {tvar.id: TupleType(list(middle), tvar.tuple_fallback)}
   351	            for tvar, sub in zip(
   352	                self.alias.alias_tvars[:prefix] + self.alias.alias_tvars[prefix + 1 :], start + end
   353	            ):
   354	                mapping[tvar.id] = sub
   355
   356	        new_tp = self.alias.target.accept(InstantiateAliasVisitor(mapping))
   357	        new_tp.accept(LocationSetter(self.line, self.column))
   358	        new_tp.line = self.line
   359	        new_tp.column = self.column
   360	        return new_tp
   361
   362	    def _partial_expansion(self, nothing_args: bool = False) -> tuple[ProperType, bool]:
   363	        # Private method mostly for debugging and testing.
   364	        unroller = UnrollAliasVisitor(set(), {})
   365	        if nothing_args:
   366	            alias = self.copy_modified(args=[UninhabitedType()] * len(self.args))
   367	        else:
   368	            alias = self
   369	        unrolled = alias.accept(unroller)
   370	        assert isinstance(unrolled, ProperType)
   371	        return unrolled, unroller.recursed
   372
   373	    def expand_all_if_possible(self, nothing_args: bool = False) -> ProperType | None:
   374	        """Attempt a full expansion of the type alias (including nested aliases).
   375
   376	        If the expansion is not possible, i.e. the alias is (mutually-)recursive,
   377	        return None. If nothing_args is True, replace all type arguments with an
   378	        UninhabitedType() (used to detect recursively defined aliases).
   379	        """
   380	        unrolled, recursed = self._partial_expansion(nothing_args=nothing_args)
   381	        if recursed:
   382	            return None
   383	        return unrolled
   384
   385	    @property
   386	    def is_recursive(self) -> bool:
   387	        """Whether this type alias is recursive.
   388
   389	        Note this doesn't check generic alias arguments, but only if this alias
   390	        *definition* is recursive. The property value thus can be cached on the
   391	        underlying TypeAlias node. If you want to include all nested types, use
   392	        has_recursive_types() function.
   393	        """
   394	        assert self.alias is not None, "Unfixed type alias"
   395	        is_recursive = self.alias._is_recursive
   396	        if is_recursive is None:
   397	            is_recursive = self.expand_all_if_possible(nothing_args=True) is None
   398	            # We cache the value on the underlying TypeAlias node as an optimization,
   399	            # since the value is the same for all instances of the same alias.
   400	            self.alias._is_recursive = is_recursive
   401	        return is_recursive
   402
   403	    def can_be_true_default(self) -> bool:
   404	        if self.alias is not None:
   405	            return self.alias.target.can_be_true
   406	        return super().can_be_true_default()
   407
   408	    def can_be_false_default(self) -> bool:
   409	        if self.alias is not None:
   410	            return self.alias.target.can_be_false
   411	        return super().can_be_false_default()
   412
   413	    def accept(self, visitor: TypeVisitor[T]) -> T:
   414	        return visitor.visit_type_alias_type(self)
   415
   416	    def __hash__(self) -> int:
   417	        return hash((self.alias, tuple(self.args)))
   418
   419	    def __eq__(self, other: object) -> bool:
   420	        # Note: never use this to determine subtype relationships, use is_subtype().
   421	        if not isinstance(other, TypeAliasType):
   422	            return NotImplemented
   423	        return self.alias == other.alias and self.args == other.args
   424
   425	    def serialize(self) -> JsonDict:
   426	        assert self.alias is not None
   427	        data: JsonDict = {
   428	            ".class": "TypeAliasType",
   429	            "type_ref": self.alias.fullname,
   430	            "args": [arg.serialize() for arg in self.args],
   431	        }
   432	        return data
   433
   434	    @classmethod
   435	    def deserialize(cls, data: JsonDict) -> TypeAliasType:
   436	        assert data[".class"] == "TypeAliasType"
   437	        args: list[Type] = []
   438	        if "args" in data:
   439	            args_list = data["args"]
   440	            assert isinstance(args_list, list)
   441	            args = [deserialize_type(arg) for arg in args_list]
   442	        alias = TypeAliasType(None, args)
   443	        alias.type_ref = data["type_ref"]
   444	        return alias
   445
   446	    def copy_modified(self, *, args: list[Type] | None = None) -> TypeAliasType:
   447	        return TypeAliasType(
   448	            self.alias, args if args is not None else self.args.copy(), self.line, self.column
   449	        )

   511	class TypeVarId:
   512	    # A type variable is uniquely identified by its raw id and meta level.
   513
   514	    # For plain variables (type parameters of generic classes and
   515	    # functions) raw ids are allocated by semantic analysis, using
   516	    # positive ids 1, 2, ... for generic class parameters and negative
   517	    # ids -1, ... for generic function type arguments. A special value 0
   518	    # is reserved for Self type variable (autogenerated). This convention
   519	    # is only used to keep type variable ids distinct when allocating
   520	    # them; the type checker makes no distinction between class and
   521	    # function type variables.
   522
   523	    # Metavariables are allocated unique ids starting from 1.
   524	    raw_id: int
   525
   526	    # Level of the variable in type inference. Currently either 0 for
   527	    # declared types, or 1 for type inference metavariables.
   528	    meta_level: int = 0
   529
   530	    # Class variable used for allocating fresh ids for metavariables.
   531	    next_raw_id: ClassVar[int] = 1
   532
   533	    # Fullname of class or function/method which declares this type
   534	    # variable (not the fullname of the TypeVar definition!), or ''
   535	    namespace: str
   536
   537	    def __init__(self, raw_id: int, meta_level: int = 0, *, namespace: str = "") -> None:
   538	        self.raw_id = raw_id
   539	        self.meta_level = meta_level
   540	        self.namespace = namespace
   541
   542	    @staticmethod
   543	    def new(meta_level: int) -> TypeVarId:
   544	        raw_id = TypeVarId.next_raw_id
   545	        TypeVarId.next_raw_id += 1
   546	        return TypeVarId(raw_id, meta_level)
   547
   548	    def __repr__(self) -> str:
   549	        return self.raw_id.__repr__()
   550
   551	    def __eq__(self, other: object) -> bool:
   552	        return (
   553	            isinstance(other, TypeVarId)
   554	            and self.raw_id == other.raw_id
   555	            and self.meta_level == other.meta_level
   556	            and self.namespace == other.namespace
   557	        )
   558
   559	    def __ne__(self, other: object) -> bool:
   560	        return not (self == other)
   561
   562	    def __hash__(self) -> int:
   563	        return hash((self.raw_id, self.meta_level, self.namespace))
   564
   565	    def is_meta_var(self) -> bool:
   566	        return self.meta_level > 0
   567
   568	    def is_self(self) -> bool:
   569	        # This is a special value indicating typing.Self variable.
   570	        return self.raw_id == 0

  1405	class Instance(ProperType):
  1406	    """An instance type of form C[T1, ..., Tn].
  1407
  1408	    The list of type variables may be empty.
  1409
  1410	    Several types have fallbacks to `Instance`, because in Python everything is an object
  1411	    and this concept is impossible to express without intersection types. We therefore use
  1412	    fallbacks for all "non-special" (like UninhabitedType, ErasedType etc) types.
  1413	    """
  1414
  1415	    __slots__ = ("type", "args", "invalid", "type_ref", "last_known_value", "_hash", "extra_attrs")
  1416
  1417	    def __init__(
  1418	        self,
  1419	        typ: mypy.nodes.TypeInfo,
  1420	        args: Sequence[Type],
  1421	        line: int = -1,
  1422	        column: int = -1,
  1423	        *,
  1424	        last_known_value: LiteralType | None = None,
  1425	        extra_attrs: ExtraAttrs | None = None,
  1426	    ) -> None:
  1427	        super().__init__(line, column)
  1428	        self.type = typ
  1429	        self.args = tuple(args)
  1430	        self.type_ref: str | None = None
  1431
  1432	        # True if recovered after incorrect number of type arguments error
  1433	        self.invalid = False
  1434
  1435	        # This field keeps track of the underlying Literal[...] value associated with
  1436	        # this instance, if one is known.
  1437	        #
  1438	        # This field is set whenever possible within expressions, but is erased upon
  1439	        # variable assignment (see erasetype.remove_instance_last_known_values) unless
  1440	        # the variable is declared to be final.
  1441	        #
  1442	        # For example, consider the following program:
  1443	        #
  1444	        #     a = 1
  1445	        #     b: Final[int] = 2
  1446	        #     c: Final = 3
  1447	        #     print(a + b + c + 4)
  1448	        #
  1449	        # The 'Instance' objects associated with the expressions '1', '2', '3', and '4' will
  1450	        # have last_known_values of type Literal[1], Literal[2], Literal[3], and Literal[4]
  1451	        # respectively. However, the Instance object assigned to 'a' and 'b' will have their
  1452	        # last_known_value erased: variable 'a' is mutable; variable 'b' was declared to be
  1453	        # specifically an int.
  1454	        #
  1455	        # Or more broadly, this field lets this Instance "remember" its original declaration
  1456	        # when applicable. We want this behavior because we want implicit Final declarations
  1457	        # to act pretty much identically with constants: we should be able to replace any
  1458	        # places where we use some Final variable with the original value and get the same
  1459	        # type-checking behavior. For example, we want this program:
  1460	        #
  1461	        #    def expects_literal(x: Literal[3]) -> None: pass
  1462	        #    var: Final = 3
  1463	        #    expects_literal(var)
  1464	        #
  1465	        # ...to type-check in the exact same way as if we had written the program like this:
  1466	        #
  1467	        #    def expects_literal(x: Literal[3]) -> None: pass
  1468	        #    expects_literal(3)
  1469	        #
  1470	        # In order to make this work (especially with literal types), we need var's type
  1471	        # (an Instance) to remember the "original" value.
  1472	        #
  1473	        # Preserving this value within expressions is useful for similar reasons.
  1474	        #
  1475	        # Currently most of mypy will ignore this field and will continue to treat this type like
  1476	        # a regular Instance. We end up using this field only when we are explicitly within a
  1477	        # Literal context.
  1478	        self.last_known_value = last_known_value
  1479
  1480	        # Cached hash value
  1481	        self._hash = -1
  1482
  1483	        # Additional attributes defined per instance of this type. For example modules
  1484	        # have different attributes per instance of types.ModuleType.
  1485	        self.extra_attrs = extra_attrs
  1486
  1487	    def accept(self, visitor: TypeVisitor[T]) -> T:
  1488	        return visitor.visit_instance(self)
  1489
  1490	    def __hash__(self) -> int:
  1491	        if self._hash == -1:
  1492	            self._hash = hash((self.type, self.args, self.last_known_value, self.extra_attrs))
  1493	        return self._hash
  1494
  1495	    def __eq__(self, other: object) -> bool:
  1496	        if not isinstance(other, Instance):
  1497	            return NotImplemented
  1498	        return (
  1499	            self.type == other.type
  1500	            and self.args == other.args
  1501	            and self.last_known_value == other.last_known_value
  1502	            and self.extra_attrs == other.extra_attrs
  1503	        )
  1504
  1505	    def serialize(self) -> JsonDict | str:
  1506	        assert self.type is not None
  1507	        type_ref = self.type.fullname
  1508	        if not self.args and not self.last_known_value:
  1509	            return type_ref
  1510	        data: JsonDict = {
  1511	            ".class": "Instance",
  1512	            "type_ref": type_ref,
  1513	            "args": [arg.serialize() for arg in self.args],
  1514	        }
  1515	        if self.last_known_value is not None:
  1516	            data["last_known_value"] = self.last_known_value.serialize()
  1517	        data["extra_attrs"] = self.extra_attrs.serialize() if self.extra_attrs else None
  1518	        return data
  1519
  1520	    @classmethod
  1521	    def deserialize(cls, data: JsonDict | str) -> Instance:
  1522	        if isinstance(data, str):
  1523	            inst = Instance(NOT_READY, [])
  1524	            inst.type_ref = data
  1525	            return inst
  1526	        assert data[".class"] == "Instance"
  1527	        args: list[Type] = []
  1528	        if "args" in data:
  1529	            args_list = data["args"]
  1530	            assert isinstance(args_list, list)
  1531	            args = [deserialize_type(arg) for arg in args_list]
  1532	        inst = Instance(NOT_READY, args)
  1533	        inst.type_ref = data["type_ref"]  # Will be fixed up by fixup.py later.
  1534	        if "last_known_value" in data:
  1535	            inst.last_known_value = LiteralType.deserialize(data["last_known_value"])
  1536	        if data.get("extra_attrs") is not None:
  1537	            inst.extra_attrs = ExtraAttrs.deserialize(data["extra_attrs"])
  1538	        return inst
  1539
  1540	    def copy_modified(
  1541	        self,
  1542	        *,
  1543	        args: Bogus[list[Type]] = _dummy,
  1544	        last_known_value: Bogus[LiteralType | None] = _dummy,
  1545	    ) -> Instance:
  1546	        new = Instance(
  1547	            typ=self.type,
  1548	            args=args if args is not _dummy else self.args,
  1549	            line=self.line,
  1550	            column=self.column,
  1551	            last_known_value=(
  1552	                last_known_value if last_known_value is not _dummy else self.last_known_value
  1553	            ),
  1554	            extra_attrs=self.extra_attrs,
  1555	        )
  1556	        # We intentionally don't copy the extra_attrs here, so they will be erased.

  1558	        new.can_be_false = self.can_be_false
  1559	        return new

  1561	    def copy_with_extra_attr(self, name: str, typ: Type) -> Instance:
  1562	        if self.extra_attrs:
  1563	            existing_attrs = self.extra_attrs.copy()
  1564	        else:
  1565	            existing_attrs = ExtraAttrs({}, set(), None)
  1566	        existing_attrs.attrs[name] = typ
  1567	        new = self.copy_modified()
  1568	        new.extra_attrs = existing_attrs
  1569	        return new

  1571	    def is_singleton_type(self) -> bool:
  1572	        # TODO:

  1702	        for position, (type, kind) in enumerate(zip(self.arg_types, self.arg_kinds)):

  1707	    def formal_arguments(self, include_star_args: bool = False) -> list[FormalArgument]:
  1708	        """Yields the formal arguments corresponding to this callable, ignoring *arg and **kwargs.
  1709
  1710	        To handle *args and **kwargs, use the 'callable.var_args' and 'callable.kw_args' fields,
  1711	        if they are not None.
  1712
  1713	        If you really want to include star args in the yielded output, set the
  1714	        'include_star_args' parameter to 'True'."""
  1715	        args = []
  1716	        done_with_positional = False
  1717	        for i in range(len(self.arg_types)):
  1718	            kind = self.arg_kinds[i]
  1719	            if kind.is_named() or kind.is_star():
  1720	                done_with_positional = True
  1721	            if not include_star_args and kind.is_star():
  1722	                continue
  1723
  1724	            required = kind.is_required()
  1725	            pos = None if done_with_positional else i
  1726	            arg = FormalArgument(self.arg_names[i], pos, self.arg_types[i], required)
  1727	            args.append(arg)
  1728	        return args

  1730	    def argument_by_name(self, name: str | None) -> FormalArgument | None:
  1731	        if name is None:
  1732	            return None
  1733	        seen_star = False
  1734	        for i, (arg_name, kind, typ) in enumerate(
  1735	            zip(self.arg_names, self.arg_kinds, self.arg_types)
  1736	        ):
  1737	            # No more positional arguments after these.
  1738	            if kind.is_named() or kind.is_star():
  1739	                seen_star = True
  1740	            if kind.is_star():
  1741	                continue
  1742	            if arg_name == name:
  1743	                position = None if seen_star else i
  1744	                return FormalArgument(name, position, typ, kind.is_required())
  1745	        return self.try_synthesizing_arg_from_kwarg(name)

  1747	    def argument_by_position(self, position: int | None) -> FormalArgument | None:
  1748	        if position is None:
  1749	            return None
  1750	        if position >= len(self.arg_names):
  1751	            return self.try_synthesizing_arg_from_vararg(position)
  1752	        name, kind, typ = (
  1753	            self.arg_names[position],
  1754	            self.arg_kinds[position],
  1755	            self.arg_types[position],
  1756	        )
  1757	        if kind.is_positional():
  1758	            return FormalArgument(name, position, typ, kind == ARG_POS)
  1759	        else:
  1760	            return self.try_synthesizing_arg_from_vararg(position)

  1763	        kw_arg = self.kw_arg()
  1764	        if kw_arg is not None:
  1765	            return FormalArgument(name, None, kw_arg.typ, False)
  1766	        else:
  1767	            return None

  1769	    def try_synthesizing_arg_from_vararg(self, position: int | None) -> FormalArgument | None:
  1770	        var_arg = self.var_arg()
  1771	        if var_arg is not None:
  1772	            return FormalArgument(None, position, var_arg.typ, False)
  1773	        else:
  1774	            return None
