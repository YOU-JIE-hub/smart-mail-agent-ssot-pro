   327	def all_values(args: list[Register], blocks: list[BasicBlock]) -> list[Value]:
   328	    """Return the set of all values that may be initialized in the blocks.
   329
   330	    This omits registers that are only read.
   331	    """
   332	    values: list[Value] = list(args)
   333	    seen_registers = set(args)
   334
   335	    for block in blocks:
   336	        for op in block.ops:
   337	            if not isinstance(op, ControlOp):
   338	                if isinstance(op, (Assign, AssignMulti)):
   339	                    if op.dest not in seen_registers:
   340	                        values.append(op.dest)
   341	                        seen_registers.add(op.dest)
   342	                elif op.is_void:
   343	                    continue
   344	                else:
   345	                    # If we take the address of a register, it might get initialized.
   346	                    if (
   347	                        isinstance(op, LoadAddress)
   348	                        and isinstance(op.src, Register)
   349	                        and op.src not in seen_registers
   350	                    ):
   351	                        values.append(op.src)
   352	                        seen_registers.add(op.src)
   353	                    values.append(op)
   354
   355	    return values

   375	                        seen_registers.add(op.dest)

    50	class BasicBlock:
    51	    """IR basic block.
    52
    53	    Contains a sequence of Ops and ends with a ControlOp (Goto,
    54	    Branch, Return or Unreachable). Only the last op can be a
    55	    ControlOp.
    56
    57	    All generated Ops live in basic blocks. Basic blocks determine the
    58	    order of evaluation and control flow within a function. A basic
    59	    block is always associated with a single function/method (FuncIR).
    60
    61	    When building the IR, ops that raise exceptions can be included in
    62	    the middle of a basic block, but the exceptions aren't checked.
    63	    Afterwards we perform a transform that inserts explicit checks for
    64	    all error conditions and splits basic blocks accordingly to preserve
    65	    the invariant that a jump, branch or return can only ever appear
    66	    as the final op in a block. Manually inserting error checking ops
    67	    would be boring and error-prone.
    68
    69	    BasicBlocks have an error_handler attribute that determines where
    70	    to jump if an error occurs. If none is specified, an error will
    71	    propagate up out of the function. This is compiled away by the
    72	    `exceptions` module.
    73

    75	    get filled in by those passes.

   363	class Branch(ControlOp):
   364	    """Branch based on a value.
   365
   366	    If op is BOOL, branch based on a bit/bool value:
   367	       if [not] r1 goto L1 else goto L2
   368
   369	    If op is IS_ERROR, branch based on whether there is an error value:
   370	       if [not] is_error(r1) goto L1 else goto L2
   371	    """
   372
   373	    # Branch ops never raise an exception.
   374	    error_kind = ERR_NEVER
   375
   376	    BOOL: Final = 100
   377	    IS_ERROR: Final = 101
   378
   379	    def __init__(
   380	        self,
   381	        value: Value,
   382	        true_label: BasicBlock,
   383	        false_label: BasicBlock,
   384	        op: int,
   385	        line: int = -1,
   386	        *,
   387	        rare: bool = False,
   388	    ) -> None:
   389	        super().__init__(line)
   390	        # Target value being checked
   391	        self.value = value
   392	        # Branch here if the condition is true
   393	        self.true = true_label
   394	        # Branch here if the condition is false
   395	        self.false = false_label
   396	        # Branch.BOOL (boolean check) or Branch.IS_ERROR (error value check)
   397	        self.op = op
   398	        # If True, the condition is negated
   399	        self.negated = False
   400	        # If not None, the true label should generate a traceback entry (func name, line number)
   401	        self.traceback_entry: tuple[str, int] | None = None
   402	        # If True, we expect to usually take the false branch (for optimization purposes);
   403	        # this is implicitly treated as true if there is a traceback entry
   404	        self.rare = rare
   405
   406	    def targets(self) -> Sequence[BasicBlock]:
   407	        return (self.true, self.false)
   408
   409	    def set_target(self, i: int, new: BasicBlock) -> None:
   410	        assert i == 0 or i == 1
   411	        if i == 0:
   412	            self.true = new
   413	        else:
   414	            self.false = new
   415
   416	    def sources(self) -> list[Value]:
   417	        return [self.value]
   418
   419	    def set_sources(self, new: list[Value]) -> None:
   420	        (self.value,) = new
   421
   422	    def invert(self) -> None:
   423	        self.negated = not self.negated
   424
   425	    def accept(self, visitor: OpVisitor[T]) -> T:
   426	        return visitor.visit_branch(self)

   621	class PrimitiveDescription:
   622	    """Description of a primitive op.
   623
   624	    Primitives get lowered into lower-level ops before code generation.
   625
   626	    If c_function_name is provided, a primitive will be lowered into a CallC op.
   627	    Otherwise custom logic will need to be implemented to transform the
   628	    primitive into lower-level ops.
   629	    """
   630
   631	    def __init__(
   632	        self,
   633	        name: str,
   634	        arg_types: list[RType],
   635	        return_type: RType,  # TODO: What about generic?
   636	        var_arg_type: RType | None,
   637	        truncated_type: RType | None,
   638	        c_function_name: str | None,
   639	        error_kind: int,
   640	        steals: StealsDescription,
   641	        is_borrowed: bool,
   642	        ordering: list[int] | None,
   643	        extra_int_constants: list[tuple[int, RType]],
   644	        priority: int,
   645	        is_pure: bool,
   646	    ) -> None:
   647	        # Each primitive much have a distinct name, but otherwise they are arbitrary.
   648	        self.name: Final = name
   649	        self.arg_types: Final = arg_types
   650	        self.return_type: Final = return_type
   651	        self.var_arg_type: Final = var_arg_type
   652	        self.truncated_type: Final = truncated_type
   653	        # If non-None, this will map to a call of a C helper function; if None,
   654	        # there must be a custom handler function that gets invoked during the lowering
   655	        # pass to generate low-level IR for the primitive (in the mypyc.lower package)
   656	        self.c_function_name: Final = c_function_name
   657	        self.error_kind: Final = error_kind
   658	        self.steals: Final = steals
   659	        self.is_borrowed: Final = is_borrowed
   660	        self.ordering: Final = ordering
   661	        self.extra_int_constants: Final = extra_int_constants
   662	        self.priority: Final = priority
   663	        # Pure primitives have no side effects, take immutable arguments, and
   664	        # never fail. They support additional optimizations.
   665	        self.is_pure: Final = is_pure
   666	        if is_pure:
   667	            assert error_kind == ERR_NEVER
   668
   669	    def __repr__(self) -> str:
   670	        return f"<PrimitiveDescription {self.name!r}: {self.arg_types}>"

   673	class PrimitiveOp(RegisterOp):
   674	    """A higher-level primitive operation.
   675
   676	    Some of these have special compiler support. These will be lowered
   677	    (transformed) into lower-level IR ops before code generation, and after
   678	    reference counting op insertion. Others will be transformed into CallC
   679	    ops.
   680
   681	    Tagged integer equality is a typical primitive op with non-trivial
   682	    lowering. It gets transformed into a tag check, followed by different
   683	    code paths for short and long representations.
   684	    """
   685
   686	    def __init__(self, args: list[Value], desc: PrimitiveDescription, line: int = -1) -> None:
   687	        self.args = args
   688	        self.type = desc.return_type
   689	        self.error_kind = desc.error_kind
   690	        self.desc = desc
   691
   692	    def sources(self) -> list[Value]:
   693	        return self.args
   694
   695	    def set_sources(self, new: list[Value]) -> None:
   696	        self.args = new[:]
   697
   698	    def stolen(self) -> list[Value]:
   699	        steals = self.desc.steals
   700	        if isinstance(steals, list):
   701	            assert len(steals) == len(self.args)
   702	            return [arg for arg, steal in zip(self.args, steals) if steal]
   703	        else:
   704	            return [] if not steals else self.sources()

   706	    def accept(self, visitor: OpVisitor[T]) -> T:
   707	        return visitor.visit_primitive_op(self)

  1603	class KeepAlive(RegisterOp):
  1604	    """A no-op operation that ensures source values aren't freed.
  1605
  1606	    This is sometimes useful to avoid decref when a reference is still
  1607	    being held but not seen by the compiler.
  1608
  1609	    A typical use case is like this (C-like pseudocode):
  1610
  1611	      ptr = &x.item
  1612	      r = *ptr
  1613	      keep_alive x  # x must not be freed here
  1614	      # x may be freed here
  1615
  1616	    If we didn't have "keep_alive x", x could be freed immediately
  1617	    after taking the address of 'item', resulting in a read after free
  1618	    on the second line.
  1619
  1620	    If 'steal' is true, the value is considered to be stolen at
  1621	    this op, i.e. it won't be decref'd. You need to ensure that
  1622	    the value is freed otherwise, perhaps by using borrowing
  1623	    followed by Unborrow.
  1624
  1625	    Be careful with steal=True -- this can cause memory leaks.
  1626	    """
  1627
  1628	    error_kind = ERR_NEVER
  1629
  1630	    def __init__(self, src: list[Value], *, steal: bool = False) -> None:
  1631	        assert src
  1632	        self.src = src
  1633	        self.steal = steal
  1634
  1635	    def sources(self) -> list[Value]:
  1636	        return self.src.copy()
  1637
  1638	    def stolen(self) -> list[Value]:
  1639	        if self.steal:
  1640	            return self.src.copy()
  1641	        return []
  1642
  1643	    def set_sources(self, new: list[Value]) -> None:
  1644	        self.src = new[:]

  1646	    def accept(self, visitor: OpVisitor[T]) -> T:

  1650	class Unborrow(RegisterOp):
  1651	    """A no-op op to create a regular reference from a borrowed one.
  1652
  1653	    Borrowed references can only be used temporarily and the reference
  1654	    counts won't be managed. This value will be refcounted normally.
  1655
  1656	    This is mainly useful if you split an aggregate value, such as
  1657	    a tuple, into components using borrowed values (to avoid increfs),
  1658	    and want to treat the components as sharing the original managed
  1659	    reference. You'll also need to use KeepAlive with steal=True to
  1660	    "consume" the original tuple reference:
  1661
  1662	      # t is a 2-tuple
  1663	      r0 = borrow t[0]
  1664	      r1 = borrow t[1]
  1665	      keep_alive steal t
  1666	      r2 = unborrow r0
  1667	      r3 = unborrow r1
  1668	      # now (r2, r3) represent the tuple as separate items, that are
  1669	      # managed again. (Note we need to steal before unborrow, to avoid
  1670	      # refcount briefly touching zero if r2 or r3 are unused.)

  1673	    """
  1674
  1675	    error_kind = ERR_NEVER

  1680	        self.src = src
  1681	        self.type = src.type

  1686	    def set_sources(self, new: list[Value]) -> None:
  1687	        (self.src,) = new
  1688
  1689	    def stolen(self) -> list[Value]:
  1690	        return []

  1854	# We do a three-pass deserialization scheme in order to resolve name
  1855	# references.
  1856	#  1. Create an empty ClassIR for each class in an SCC.
  1857	#  2. Deserialize all of the functions, which can contain references
  1858	#     to ClassIRs in their types
  1859	#  3. Deserialize all of the classes, which contain lots of references
  1860	#     to the functions they contain. (And to other classes.)
  1861	#
  1862	# Note that this approach differs from how we deserialize ASTs in mypy itself,
  1863	# where everything is deserialized in one pass then a second pass cleans up
  1864	# 'cross_refs'. We don't follow that approach here because it seems to be more
  1865	# code for not a lot of gain since it is easy in mypyc to identify all the objects
  1866	# we might need to reference.
  1867	#
  1868	# Because of these references, we need to maintain maps from class
  1869	# names to ClassIRs and func IDs to FuncIRs.
  1870	#
  1871	# These are tracked in a DeserMaps which is passed to every
  1872	# deserialization function.
  1873	#
  1874	# (Serialization and deserialization *will* be used for incremental
  1875	# compilation but so far it is not hooked up to anything.)
  1876	class DeserMaps(NamedTuple):
  1877	    classes: dict[str, ClassIR]
  1878	    functions: dict[str, FuncIR]

   300	    def format(self, fmt: str, *args: Any) -> str:
   301	        """Helper for formatting strings.
   302
   303	        These format sequences are supported in fmt:
   304
   305	          %s: arbitrary object converted to string using str()
   306	          %r: name of IR value/register
   307	          %d: int
   308	          %f: float
   309	          %l: BasicBlock (formatted as label 'Ln')
   310	          %t: RType
   311	        """
   312	        result = []
   313	        i = 0
   314	        arglist = list(args)
   315	        while i < len(fmt):
   316	            n = fmt.find("%", i)
   317	            if n < 0:
   318	                n = len(fmt)
   319	            result.append(fmt[i:n])
   320	            if n < len(fmt):
   321	                typespec = fmt[n + 1]
   322	                arg = arglist.pop(0)
   323	                if typespec == "r":
   324	                    # Register/value
   325	                    assert isinstance(arg, Value)
   326	                    if isinstance(arg, Integer):
   327	                        result.append(str(arg.value))
   328	                    elif isinstance(arg, Float):
   329	                        result.append(repr(arg.value))
   330	                    else:
   331	                        result.append(self.names[arg])
   332	                elif typespec == "d":
   333	                    # Integer
   334	                    result.append("%d" % arg)
   335	                elif typespec == "f":
   336	                    # Float
   337	                    result.append("%f" % arg)
   338	                elif typespec == "l":
   339	                    # Basic block (label)
   340	                    assert isinstance(arg, BasicBlock)
   341	                    result.append("L%s" % arg.label)
   342	                elif typespec == "t":
   343	                    # RType
   344	                    assert isinstance(arg, RType)
   345	                    result.append(arg.name)
   346	                elif typespec == "s":
   347	                    # String
   348	                    result.append(str(arg))
   349	                else:
   350	                    raise ValueError(f"Invalid format sequence %{typespec}")
   351	                i = n + 2
   352	            else:
   353	                i = n
   354	        return "".join(result)

   357	def format_registers(func_ir: FuncIR, names: dict[Value, str]) -> list[str]:
   358	    result = []
   359	    i = 0
   360	    regs = all_values_full(func_ir.arg_regs, func_ir.blocks)
   361	    while i < len(regs):
   362	        i0 = i
   363	        group = [names[regs[i0]]]
   364	        while i + 1 < len(regs) and regs[i + 1].type == regs[i0].type:
   365	            i += 1
   366	            group.append(names[regs[i]])
   367	        i += 1
   368	        result.append("{} :: {}".format(", ".join(group), regs[i0].type))
   369	    return result
