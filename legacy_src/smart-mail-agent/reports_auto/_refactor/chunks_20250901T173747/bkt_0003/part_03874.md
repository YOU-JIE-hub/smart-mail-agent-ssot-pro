    31	    term_choices = ['term', 'term-missing']

   412	def _parse_marker(marker):
   413	    """Return (timeout, method) tuple from marker.
   414
   415	    Either could be None.  The values are not interpreted, so
   416	    could still be bogus and even the wrong type.
   417	    """
   418	    if not marker.args and not marker.kwargs:
   419	        raise TypeError("Timeout marker must have at least one argument")
   420	    timeout = method = func_only = NOTSET = object()
   421	    for kw, val in marker.kwargs.items():
   422	        if kw == "timeout":
   423	            timeout = val
   424	        elif kw == "method":
   425	            method = val
   426	        elif kw == "func_only":
   427	            func_only = val
   428	        else:
   429	            raise TypeError("Invalid keyword argument for timeout marker: %s" % kw)
   430	    if len(marker.args) >= 1 and timeout is not NOTSET:
   431	        raise TypeError("Multiple values for timeout argument of timeout marker")
   432	    elif len(marker.args) >= 1:
   433	        timeout = marker.args[0]
   434	    if len(marker.args) >= 2 and method is not NOTSET:
   435	        raise TypeError("Multiple values for method argument of timeout marker")
   436	    elif len(marker.args) >= 2:
   437	        method = marker.args[1]
   438	    if len(marker.args) > 2:
   439	        raise TypeError("Too many arguments for timeout marker")
   440	    if timeout is NOTSET:
   441	        timeout = None
   442	    if method is NOTSET:
   443	        method = None
   444	    if func_only is NOTSET:
   445	        func_only = None
   446	    return Settings(timeout, method, func_only, None)

   547	    current_ident = threading.current_thread().ident

   558	        terminal.write("".join(traceback.format_stack(frame)))
