     4	when a value of type T is expected without requiring any runtime conversions.

   827	    def assert_emit(
   828	        self,
   829	        op: Op | list[Op],
   830	        expected: str,
   831	        next_block: BasicBlock | None = None,
   832	        *,
   833	        rare: bool = False,
   834	        next_branch: Branch | None = None,
   835	        skip_next: bool = False,
   836	    ) -> None:
   837	        block = BasicBlock(0)
   838	        if isinstance(op, Op):
   839	            block.ops.append(op)
   840	        else:
   841	            block.ops.extend(op)
   842	            op = op[-1]
   843	        value_names = generate_names_for_ir(self.registers, [block])
   844	        emitter = Emitter(self.context, value_names)
   845	        declarations = Emitter(self.context, value_names)
   846	        emitter.fragments = []
   847	        declarations.fragments = []
   848
   849	        visitor = FunctionEmitterVisitor(emitter, declarations, "prog.py", "prog")
   850	        visitor.next_block = next_block
   851	        visitor.rare = rare
   852	        if next_branch:
   853	            visitor.ops = [op, next_branch]
   854	        else:
   855	            visitor.ops = [op]
   856	        visitor.op_index = 0
   857
   858	        op.accept(visitor)
   859	        frags = declarations.fragments + emitter.fragments
   860	        actual_lines = [line.strip(" ") for line in frags]
   861	        assert all(line.endswith("\n") for line in actual_lines)
   862	        actual_lines = [line.rstrip("\n") for line in actual_lines]
   863	        if not expected.strip():
   864	            expected_lines = []
   865	        else:
   866	            expected_lines = expected.rstrip().split("\n")
   867	        expected_lines = [line.strip(" ") for line in expected_lines]
   868	        assert_string_arrays_equal(
   869	            expected_lines, actual_lines, msg="Generated code unexpected", traceback=True
   870	        )
   871	        if skip_next:
