    78	        :param params: Tuple of types the class . Given a generic class

   103	        if not hasattr(cls, '__parameters__'):

   109	        if all_identical(typevars_map.keys(), typevars_map.values()) and typevars_map:

   112	        # Create new model with original model as parent inserting fields with DeferredType.
   113	        model_name = cls.__concrete_name__(params)

   116	        type_hints = get_all_type_hints(cls).items()

   186	    @classmethod
   187	    def __parameterized_bases__(cls, typevars_map: Parametrization) -> Iterator[Type[Any]]:
   188	        """
   189	        Returns unbound bases of cls parameterised to given type variables
   190
   191	        :param typevars_map: Dictionary of type applications for binding subclasses.
   192	            Given a generic class `Model` with 2 type variables [S, T]
   193	            and a concrete model `Model[str, int]`,
   194	            the value `{S: str, T: int}` would be passed to `typevars_map`.
   195	        :return: an iterator of generic sub classes, parameterised by `typevars_map`
   196	            and other assigned parameters of `cls`
   197
   198	        e.g.:
   199	        ```
   200	        class A(GenericModel, Generic[T]):
   201	            ...
   202
   203	        class B(A[V], Generic[V]):
   204	            ...
   205
   206	        assert A[int] in B.__parameterized_bases__({V: int})
   207	        ```
   208	        """

   210	        def build_base_model(
   211	            base_model: Type[GenericModel], mapped_types: Parametrization
   212	        ) -> Iterator[Type[GenericModel]]:
   213	            base_parameters = tuple(mapped_types[param] for param in base_model.__parameters__)
   214	            parameterized_base = base_model.__class_getitem__(base_parameters)
   215	            if parameterized_base is base_model or parameterized_base is cls:
   216	                # Avoid duplication in MRO
   217	                return
   218	            yield parameterized_base

   220	        for base_model in cls.__bases__:
   221	            if not issubclass(base_model, GenericModel):
   222	                # not a class that can be meaningfully parameterized
   223	                continue
   224	            elif not getattr(base_model, '__parameters__', None):
   225	                # base_model is "GenericModel"  (and has no __parameters__)
   226	                # or
   227	                # base_model is already concrete, and will be included transitively via cls.
   228	                continue

   232	                    # e.g. cls = B[S], base_model = A[S]
   233	                    # B[S][int] should subclass A[int],  (and will be transitively via B[int])
   234	                    # but it's not viable to consistently subclass types with arbitrary construction
   235	                    # So don't attempt to include A[S][int]
   236	                    continue
   237	                else:  # base_model not in _assigned_parameters:
   238	                    # cls is partially parameterized, base_model is original generic
   239	                    # e.g.  cls = B[str, T], base_model = B[S, T]
   240	                    # Need to determine the mapping for the base_model parameters
   241	                    mapped_types: Parametrization = {
   242	                        key: typevars_map.get(value, value) for key, value in _assigned_parameters[cls].items()
   243	                    }
   244	                    yield from build_base_model(base_model, mapped_types)
   245	            else:
   246	                # cls is base generic, so base_class has a distinct base
   247	                # can construct the Parameterised base model using typevars_map directly
   248	                yield from build_base_model(base_model, typevars_map)

   251	def replace_types(type_: Any, type_map: Mapping[Any, Any]) -> Any:
   252	    """Return type with all occurrences of `type_map` keys recursively replaced with their values.
   253
   254	    :param type_: Any type, class or generic alias
   255	    :param type_map: Mapping from `TypeVar` instance to concrete types.
   256	    :return: New type representing the basic structure of `type_` with all
   257	        `typevar_map` keys recursively replaced.
   258
   259	    >>> replace_types(Tuple[str, Union[List[str], float]], {str: int})
   260	    Tuple[int, Union[List[int], float]]
   261
   262	    """
   263	    if not type_map:
   264	        return type_
   265
   266	    type_args = get_args(type_)
   267	    origin_type = get_origin(type_)
   268
   269	    if origin_type is Annotated:
   270	        annotated_type, *annotations = type_args
   271	        return Annotated[replace_types(annotated_type, type_map), tuple(annotations)]
   272
   273	    if (origin_type is ExtLiteral) or (sys.version_info >= (3, 8) and origin_type is Literal):
   274	        return type_map.get(type_, type_)
   275	    # Having type args is a good indicator that this is a typing module
   276	    # class instantiation or a generic alias of some sort.
   277	    if type_args:
   278	        resolved_type_args = tuple(replace_types(arg, type_map) for arg in type_args)
   279	        if all_identical(type_args, resolved_type_args):
   280	            # If all arguments are the same, there is no need to modify the
   281	            # type or create a new object at all
   282	            return type_
   283	        if (
   284	            origin_type is not None
   285	            and isinstance(type_, typing_base)
   286	            and not isinstance(origin_type, typing_base)
   287	            and getattr(type_, '_name', None) is not None
   288	        ):
   289	            # In python < 3.9 generic aliases don't exist so any of these like `list`,
   290	            # `type` or `collections.abc.Callable` need to be translated.
   291	            # See: https://www.python.org/dev/peps/pep-0585
   292	            origin_type = getattr(typing, type_._name)
   293	        assert origin_type is not None
   294	        # PEP-604 syntax (Ex.: list | str) is represented with a types.UnionType object that does not have __getitem__.
   295	        # We also cannot use isinstance() since we have to compare types.
   296	        if sys.version_info >= (3, 10) and origin_type is types.UnionType:  # noqa: E721
   297	            return _UnionGenericAlias(origin_type, resolved_type_args)
   298	        return origin_type[resolved_type_args]
   299
   300	    # We handle pydantic generic models separately as they don't have the same
   301	    # semantics as "typing" classes or generic aliases
   302	    if not origin_type and lenient_issubclass(type_, GenericModel) and not type_.__concrete__:
   303	        type_args = type_.__parameters__
   304	        resolved_type_args = tuple(replace_types(t, type_map) for t in type_args)
   305	        if all_identical(type_args, resolved_type_args):
   306	            return type_
   307	        return type_[resolved_type_args]
   308
   309	    # Handle special case for typehints that can have lists as arguments.
   310	    # `typing.Callable[[int, str], int]` is an example for this.
   311	    if isinstance(type_, (List, list)):
   312	        resolved_list = list(replace_types(element, type_map) for element in type_)
   313	        if all_identical(type_, resolved_list):
   314	            return type_
   315	        return resolved_list
   316
   317	    # For JsonWrapperValue, need to handle its inner type to allow correct parsing
   318	    # of generic Json arguments like Json[T]
   319	    if not origin_type and lenient_issubclass(type_, JsonWrapper):
   320	        type_.inner_type = replace_types(type_.inner_type, type_map)
   321	        return type_
   322
   323	    # If all else fails, we try to resolve the type directly and otherwise just
   324	    # return the input with no modifications.
   325	    new_type = type_map.get(type_, type_)
   326	    # Convert string to ForwardRef
   327	    if isinstance(new_type, str):
   328	        return ForwardRef(new_type)
   329	    else:
   330	        return new_type

   109	# If a field is of type `Callable`, its default value should be a function and cannot to ignored.
   110	ANNOTATED_FIELD_UNTOUCHED_TYPES: Tuple[Any, ...] = (property, type, classmethod, staticmethod)
   111	# When creating a `BaseModel` instance, we bypass all the methods, properties... added to the model
   112	UNTOUCHED_TYPES: Tuple[Any, ...] = (FunctionType,) + ANNOTATED_FIELD_UNTOUCHED_TYPES
   113	# Note `ModelMetaclass` refers to `BaseModel`, but is also used to *create* `BaseModel`, so we need to add this extra
   114	# (somewhat hacky) boolean to keep track of whether we've created the `BaseModel` class yet, and therefore whether it's
   115	# safe to refer to it. If it *hasn't* been created, we assume that the `__new__` call we're in the middle of is for
   116	# the `BaseModel` class, since that's defined immediately after the metaclass.
   117	_is_base_model_class_defined = False

   120	@dataclass_transform(kw_only_default=True, field_specifiers=(Field,))
   121	class ModelMetaclass(ABCMeta):
   122	    @no_type_check  # noqa C901
   123	    def __new__(mcs, name, bases, namespace, **kwargs):  # noqa C901
   124	        fields: Dict[str, ModelField] = {}
   125	        config = BaseConfig
   126	        validators: 'ValidatorListDict' = {}
   127
   128	        pre_root_validators, post_root_validators = [], []
   129	        private_attributes: Dict[str, ModelPrivateAttr] = {}
   130	        base_private_attributes: Dict[str, ModelPrivateAttr] = {}
   131	        slots: SetStr = namespace.get('__slots__', ())
   132	        slots = {slots} if isinstance(slots, str) else set(slots)
   133	        class_vars: SetStr = set()
   134	        hash_func: Optional[Callable[[Any], int]] = None
   135
   136	        for base in reversed(bases):
   137	            if _is_base_model_class_defined and issubclass(base, BaseModel) and base != BaseModel:
   138	                fields.update(smart_deepcopy(base.__fields__))
   139	                config = inherit_config(base.__config__, config)
   140	                validators = inherit_validators(base.__validators__, validators)
   141	                pre_root_validators += base.__pre_root_validators__
   142	                post_root_validators += base.__post_root_validators__
   143	                base_private_attributes.update(base.__private_attributes__)
   144	                class_vars.update(base.__class_vars__)
   145	                hash_func = base.__hash__
   146
   147	        resolve_forward_refs = kwargs.pop('__resolve_forward_refs__', True)
   148	        allowed_config_kwargs: SetStr = {
   149	            key
   150	            for key in dir(config)
   151	            if not (key.startswith('__') and key.endswith('__'))  # skip dunder methods and attributes
   152	        }
   153	        config_kwargs = {key: kwargs.pop(key) for key in kwargs.keys() & allowed_config_kwargs}
   154	        config_from_namespace = namespace.get('Config')
   155	        if config_kwargs and config_from_namespace:
   156	            raise TypeError('Specifying config in two places is ambiguous, use either Config attribute or class kwargs')
   157	        config = inherit_config(config_from_namespace, config, **config_kwargs)
   158
   159	        validators = inherit_validators(extract_validators(namespace), validators)
   160	        vg = ValidatorGroup(validators)
   161
   162	        for f in fields.values():
   163	            f.set_config(config)
   164	            extra_validators = vg.get_validators(f.name)
   165	            if extra_validators:
   166	                f.class_validators.update(extra_validators)
   167	                # re-run prepare to add extra validators
   168	                f.populate_validators()
   169
   170	        prepare_config(config, name)
   171
   172	        untouched_types = ANNOTATED_FIELD_UNTOUCHED_TYPES
   173
   174	        def is_untouched(v: Any) -> bool:
   175	            return isinstance(v, untouched_types) or v.__class__.__name__ == 'cython_function_or_method'
   176
   177	        if (namespace.get('__module__'), namespace.get('__qualname__')) != ('pydantic.main', 'BaseModel'):
   178	            annotations = resolve_annotations(namespace.get('__annotations__', {}), namespace.get('__module__', None))
   179	            # annotation only fields need to come first in fields
   180	            for ann_name, ann_type in annotations.items():
   181	                if is_classvar(ann_type):
   182	                    class_vars.add(ann_name)
   183	                elif is_finalvar_with_default_val(ann_type, namespace.get(ann_name, Undefined)):
   184	                    class_vars.add(ann_name)
   185	                elif is_valid_field(ann_name):
   186	                    validate_field_name(bases, ann_name)
   187	                    value = namespace.get(ann_name, Undefined)
   188	                    allowed_types = get_args(ann_type) if is_union(get_origin(ann_type)) else (ann_type,)
   189	                    if (
   190	                        is_untouched(value)
   191	                        and ann_type != PyObject
   192	                        and not any(
   193	                            lenient_issubclass(get_origin(allowed_type), Type) for allowed_type in allowed_types
   194	                        )
   195	                    ):
   196	                        continue
   197	                    fields[ann_name] = ModelField.infer(
   198	                        name=ann_name,
   199	                        value=value,
   200	                        annotation=ann_type,
   201	                        class_validators=vg.get_validators(ann_name),
   202	                        config=config,
   203	                    )
   204	                elif ann_name not in namespace and config.underscore_attrs_are_private:
   205	                    private_attributes[ann_name] = PrivateAttr()
   206
   207	            untouched_types = UNTOUCHED_TYPES + config.keep_untouched
   208	            for var_name, value in namespace.items():
   209	                can_be_changed = var_name not in class_vars and not is_untouched(value)
   210	                if isinstance(value, ModelPrivateAttr):
   211	                    if not is_valid_private_name(var_name):
   212	                        raise NameError(
   213	                            f'Private attributes "{var_name}" must not be a valid field name; '
   214	                            f'Use sunder or dunder names, e. g. "_{var_name}" or "__{var_name}__"'
   215	                        )
   216	                    private_attributes[var_name] = value
   217	                elif config.underscore_attrs_are_private and is_valid_private_name(var_name) and can_be_changed:
   218	                    private_attributes[var_name] = PrivateAttr(default=value)
   219	                elif is_valid_field(var_name) and var_name not in annotations and can_be_changed:
   220	                    validate_field_name(bases, var_name)
   221	                    inferred = ModelField.infer(
   222	                        name=var_name,
   223	                        value=value,
   224	                        annotation=annotations.get(var_name, Undefined),
   225	                        class_validators=vg.get_validators(var_name),
   226	                        config=config,
   227	                    )
   228	                    if var_name in fields:
   229	                        if lenient_issubclass(inferred.type_, fields[var_name].type_):
   230	                            inferred.type_ = fields[var_name].type_
   231	                        else:
   232	                            raise TypeError(
   233	                                f'The type of {name}.{var_name} differs from the new default value; '
   234	                                f'if you wish to change the type of this field, please use a type annotation'
   235	                            )
   236	                    fields[var_name] = inferred
   237
   238	        _custom_root_type = ROOT_KEY in fields
   239	        if _custom_root_type:
   240	            validate_custom_root_type(fields)
   241	        vg.check_for_unused()
   242	        if config.json_encoders:
   243	            json_encoder = partial(custom_pydantic_encoder, config.json_encoders)
   244	        else:
   245	            json_encoder = pydantic_encoder
   246	        pre_rv_new, post_rv_new = extract_root_validators(namespace)
   247
   248	        if hash_func is None:
   249	            hash_func = generate_hash_function(config.frozen)
   250
   251	        exclude_from_namespace = fields | private_attributes.keys() | {'__slots__'}
   252	        new_namespace = {
   253	            '__config__': config,
   254	            '__fields__': fields,
   255	            '__exclude_fields__': {
   256	                name: field.field_info.exclude for name, field in fields.items() if field.field_info.exclude is not None
   257	            }
   258	            or None,
   259	            '__include_fields__': {
   260	                name: field.field_info.include for name, field in fields.items() if field.field_info.include is not None
   261	            }
   262	            or None,
   263	            '__validators__': vg.validators,
   264	            '__pre_root_validators__': unique_list(
   265	                pre_root_validators + pre_rv_new,
   266	                name_factory=lambda v: v.__name__,
   267	            ),
   268	            '__post_root_validators__': unique_list(
   269	                post_root_validators + post_rv_new,
   270	                name_factory=lambda skip_on_failure_and_v: skip_on_failure_and_v[1].__name__,
   271	            ),
   272	            '__schema_cache__': {},
   273	            '__json_encoder__': staticmethod(json_encoder),
   274	            '__custom_root_type__': _custom_root_type,
   275	            '__private_attributes__': {**base_private_attributes, **private_attributes},
   276	            '__slots__': slots | private_attributes.keys(),
   277	            '__hash__': hash_func,
   278	            '__class_vars__': class_vars,
   279	            **{n: v for n, v in namespace.items() if n not in exclude_from_namespace},
   280	        }
   281
   282	        cls = super().__new__(mcs, name, bases, new_namespace, **kwargs)
   283	        # set __signature__ attr only for model class, but not for its instances
   284	        cls.__signature__ = ClassAttribute('__signature__', generate_model_signature(cls.__init__, fields, config))
   285
   286	        if not _is_base_model_class_defined:
   287	            # Cython does not understand the `if TYPE_CHECKING:` condition in the
   288	            # BaseModel's body (where annotations are set), so clear them manually:
   289	            getattr(cls, '__annotations__', {}).clear()
   290
   291	        if resolve_forward_refs:
   292	            cls.__try_update_forward_refs__()
   293
   294	        # preserve `__set_name__` protocol defined in https://peps.python.org/pep-0487
   295	        # for attributes not in `new_namespace` (e.g. private attributes)
   296	        for name, obj in namespace.items():
   297	            if name not in new_namespace:
   298	                set_name = getattr(obj, '__set_name__', None)
   299	                if callable(set_name):
   300	                    set_name(cls, name)
   301
   302	        return cls

   339	        """
   340	        Create a new model by parsing and validating input data from keyword arguments.
   341
   342	        Raises ValidationError if the input data cannot be parsed to form a valid model.
   343	        """
   344	        # Uses something other than `self` the first arg to allow "self" as a settable attribute
   345	        values, fields_set, validation_error = validate_model(__pydantic_self__.__class__, data)
   346	        if validation_error:
   347	            raise validation_error

   349	            object_setattr(__pydantic_self__, '__dict__', values)

   357	    @no_type_check
   358	    def __setattr__(self, name, value):  # noqa: C901 (ignore complexity)
   359	        if name in self.__private_attributes__ or name in DUNDER_ATTRIBUTES:
   360	            return object_setattr(self, name, value)
   361
   362	        if self.__config__.extra is not Extra.allow and name not in self.__fields__:
   363	            raise ValueError(f'"{self.__class__.__name__}" object has no field "{name}"')
   364	        elif not self.__config__.allow_mutation or self.__config__.frozen:
   365	            raise TypeError(f'"{self.__class__.__name__}" is immutable and does not support item assignment')
   366	        elif name in self.__fields__ and self.__fields__[name].final:
   367	            raise TypeError(
   368	                f'"{self.__class__.__name__}" object "{name}" field is final and does not support reassignment'
   369	            )
   370	        elif self.__config__.validate_assignment:
   371	            new_values = {**self.__dict__, name: value}
   372
   373	            for validator in self.__pre_root_validators__:
   374	                try:
   375	                    new_values = validator(self.__class__, new_values)
   376	                except (ValueError, TypeError, AssertionError) as exc:
   377	                    raise ValidationError([ErrorWrapper(exc, loc=ROOT_KEY)], self.__class__)
   378
   379	            known_field = self.__fields__.get(name, None)
   380	            if known_field:
   381	                # We want to
   382	                # - make sure validators are called without the current value for this field inside `values`
   383	                # - keep other values (e.g. submodels) untouched (using `BaseModel.dict()` will change them into dicts)
   384	                # - keep the order of the fields
   385	                if not known_field.field_info.allow_mutation:
   386	                    raise TypeError(f'"{known_field.name}" has allow_mutation set to False and cannot be assigned')
   387	                dict_without_original_value = {k: v for k, v in self.__dict__.items() if k != name}
   388	                value, error_ = known_field.validate(value, dict_without_original_value, loc=name, cls=self.__class__)
   389	                if error_:
   390	                    raise ValidationError([error_], self.__class__)
   391	                else:
   392	                    new_values[name] = value
   393
   394	            errors = []
   395	            for skip_on_failure, validator in self.__post_root_validators__:
   396	                if skip_on_failure and errors:
   397	                    continue
   398	                try:
   399	                    new_values = validator(self.__class__, new_values)
   400	                except (ValueError, TypeError, AssertionError) as exc:
   401	                    errors.append(ErrorWrapper(exc, loc=ROOT_KEY))
   402	            if errors:
   403	                raise ValidationError(errors, self.__class__)
   404
   405	            # update the whole __dict__ as other values than just `value`
   406	            # may be changed (e.g. with `root_validator`)
   407	            object_setattr(self, '__dict__', new_values)
   408	        else:
   409	            self.__dict__[name] = value
   410
   411	        self.__fields_set__.add(name)

   468	        self,
   469	        *,
   470	        include: Optional[Union['AbstractSetIntStr', 'MappingIntStrAny']] = None,
   471	        exclude: Optional[Union['AbstractSetIntStr', 'MappingIntStrAny']] = None,
   472	        by_alias: bool = False,
   473	        skip_defaults: Optional[bool] = None,
   474	        exclude_unset: bool = False,
   475	        exclude_defaults: bool = False,
   476	        exclude_none: bool = False,
   477	        encoder: Optional[Callable[[Any], Any]] = None,
   478	        models_as_dict: bool = True,
   479	        **dumps_kwargs: Any,
   480	    ) -> str:
   481	        """
   482	        Generate a JSON representation of the model, `include` and `exclude` arguments as per `dict()`.

   484	        `encoder` is an optional function to supply as `default` to json.dumps(), other arguments as per `json.dumps()`.
   485	        """

   487	            warnings.warn(
   488	                f'{self.__class__.__name__}.json(): "skip_defaults" is deprecated and replaced by "exclude_unset"',
   489	                DeprecationWarning,
   490	            )
   491	            exclude_unset = skip_defaults

   495	        # because we want to be able to keep raw `BaseModel` instances and not as `dict`.
   496	        # This allows users to write custom JSON encoders for given `BaseModel` classes.
   497	        data = dict(
   498	            self._iter(
   499	                to_dict=models_as_dict,
   500	                by_alias=by_alias,
   501	                include=include,
   502	                exclude=exclude,
   503	                exclude_unset=exclude_unset,
   504	                exclude_defaults=exclude_defaults,
   505	                exclude_none=exclude_none,
   506	            )
   507	        )
   508	        if self.__custom_root_type__:
   509	            data = data[ROOT_KEY]
   510	        return self.__config__.json_dumps(data, default=encoder, **dumps_kwargs)

   831	        by_alias: bool = False,
   832	        include: Optional[Union['AbstractSetIntStr', 'MappingIntStrAny']] = None,
   833	        exclude: Optional[Union['AbstractSetIntStr', 'MappingIntStrAny']] = None,
   834	        exclude_unset: bool = False,
   835	        exclude_defaults: bool = False,
   836	        exclude_none: bool = False,
   837	    ) -> 'TupleGenerator':
   838	        # Merge field set excludes with explicit exclude parameter with explicit overriding field set options.
   839	        # The extra "is not None" guards are not logically necessary but optimizes performance for the simple case.
   840	        if exclude is not None or self.__exclude_fields__ is not None:
   841	            exclude = ValueItems.merge(self.__exclude_fields__, exclude)
   842
   843	        if include is not None or self.__include_fields__ is not None:
   844	            include = ValueItems.merge(self.__include_fields__, include, intersect=True)
   845
   846	        allowed_keys = self._calculate_keys(
   847	            include=include, exclude=exclude, exclude_unset=exclude_unset  # type: ignore
   848	        )
   849	        if allowed_keys is None and not (to_dict or by_alias or exclude_unset or exclude_defaults or exclude_none):
   850	            # huge boost for plain _iter()
   851	            yield from self.__dict__.items()
   852	            return
   853
   854	        value_exclude = ValueItems(self, exclude) if exclude is not None else None
   855	        value_include = ValueItems(self, include) if include is not None else None

   857	        for field_key, v in self.__dict__.items():

   859	                continue

   861	            if exclude_defaults:

   863	                if not getattr(model_field, 'required', True) and getattr(model_field, 'default', _missing) == v:
   864	                    continue

   867	                dict_key = self.__fields__[field_key].alias
   868	            else:
   869	                dict_key = field_key

   871	            if to_dict or value_include or value_exclude:

   877	                    exclude=value_exclude and value_exclude.for_element(field_key),
   878	                    exclude_unset=exclude_unset,
   879	                    exclude_defaults=exclude_defaults,
   880	                    exclude_none=exclude_none,
   881	                )
   882	            yield dict_key, v

   171	    def _pydantic_field_callback(self, ctx: FunctionContext) -> 'Type':
   172	        """
   173	        Extract the type of the `default` argument from the Field function, and use it as the return type.
   174
   175	        In particular:
   176	        * Check whether the default and default_factory argument is specified.
   177	        * Output an error if both are specified.
   178	        * Retrieve the type of the argument which is specified, and use it as return type for the function.
   179	        """
   180	        default_any_type = ctx.default_return_type
   181
   182	        assert ctx.callee_arg_names[0] == 'default', '"default" is no longer first argument in Field()'
   183	        assert ctx.callee_arg_names[1] == 'default_factory', '"default_factory" is no longer second argument in Field()'
   184	        default_args = ctx.args[0]
   185	        default_factory_args = ctx.args[1]
   186
   187	        if default_args and default_factory_args:
   188	            error_default_and_default_factory_specified(ctx.api, ctx.context)
   189	            return default_any_type
   190
   191	        if default_args:
   192	            default_type = ctx.arg_types[0][0]
   193	            default_arg = default_args[0]
   194
   195	            # Fallback to default Any type if the field is required
   196	            if not isinstance(default_arg, EllipsisExpr):
   197	                return default_type
   198
   199	        elif default_factory_args:
   200	            default_factory_type = ctx.arg_types[1][0]
   201
   202	            # Functions which use `ParamSpec` can be overloaded, exposing the callable's types as a parameter
   203	            # Pydantic calls the default factory without any argument, so we retrieve the first item
   204	            if isinstance(default_factory_type, Overloaded):
   205	                if MYPY_VERSION_TUPLE > (0, 910):
   206	                    default_factory_type = default_factory_type.items[0]
   207	                else:
   208	                    # Mypy0.910 exposes the items of overloaded types in a function
   209	                    default_factory_type = default_factory_type.items()[0]  # type: ignore[operator]
   210
   211	            if isinstance(default_factory_type, CallableType):
   212	                ret_type = default_factory_type.ret_type
   213	                # mypy doesn't think `ret_type` has `args`, you'd think mypy should know,
   214	                # add this check in case it varies by version
   215	                args = getattr(ret_type, 'args', None)
   216	                if args:
   217	                    if all(isinstance(arg, TypeVarType) for arg in args):
   218	                        # Looks like the default factory is a type like `list` or `dict`, replace all args with `Any`
   219	                        ret_type.args = tuple(default_any_type for _ in args)  # type: ignore[attr-defined]
   220	                return ret_type
   221
   222	        return default_any_type

   225	class PydanticPluginConfig:
   226	    __slots__ = (
   227	        'init_forbid_extra',
   228	        'init_typed',
   229	        'warn_required_dynamic_aliases',
   230	        'warn_untyped_fields',
   231	        'debug_dataclass_transform',
   232	    )
   233	    init_forbid_extra: bool
   234	    init_typed: bool
   235	    warn_required_dynamic_aliases: bool
   236	    warn_untyped_fields: bool
   237	    debug_dataclass_transform: bool  # undocumented
   238
   239	    def __init__(self, options: Options) -> None:
   240	        if options.config_file is None:  # pragma: no cover
   241	            return
   242
   243	        toml_config = parse_toml(options.config_file)
   244	        if toml_config is not None:
   245	            config = toml_config.get('tool', {}).get('pydantic-mypy', {})
   246	            for key in self.__slots__:
   247	                setting = config.get(key, False)
   248	                if not isinstance(setting, bool):
   249	                    raise ValueError(f'Configuration value must be a boolean for key: {key}')
   250	                setattr(self, key, setting)
   251	        else:
   252	            plugin_config = ConfigParser()
   253	            plugin_config.read(options.config_file)
   254	            for key in self.__slots__:
   255	                setting = plugin_config.getboolean(CONFIGFILE_KEY, key, fallback=False)
   256	                setattr(self, key, setting)
   257
   258	    def to_data(self) -> Dict[str, Any]:
   259	        return {key: getattr(self, key) for key in self.__slots__}

   262	def from_orm_callback(ctx: MethodContext) -> Type:
   263	    """
   264	    Raise an error if orm_mode is not enabled
   265	    """
   266	    model_type: Instance
   267	    ctx_type = ctx.type
   268	    if isinstance(ctx_type, TypeType):
   269	        ctx_type = ctx_type.item
   270	    if isinstance(ctx_type, CallableType) and isinstance(ctx_type.ret_type, Instance):
   271	        model_type = ctx_type.ret_type  # called on the class
   272	    elif isinstance(ctx_type, Instance):
   273	        model_type = ctx_type  # called on an instance (unusual, but still valid)
   274	    else:  # pragma: no cover
   275	        detail = f'ctx.type: {ctx_type} (of type {ctx_type.__class__.__name__})'
   276	        error_unexpected_behavior(detail, ctx.api, ctx.context)
   277	        return ctx.default_return_type
   278	    pydantic_metadata = model_type.type.metadata.get(METADATA_KEY)
   279	    if pydantic_metadata is None:
   280	        return ctx.default_return_type
   281	    orm_mode = pydantic_metadata.get('config', {}).get('orm_mode')
   282	    if orm_mode is not True:
   283	        error_from_orm(get_name(model_type.type), ctx.api, ctx.context)
   284	    return ctx.default_return_type

   287	class PydanticModelTransformer:
   288	    tracked_config_fields: Set[str] = {
   289	        'extra',
   290	        'allow_mutation',
   291	        'frozen',
   292	        'orm_mode',
   293	        'allow_population_by_field_name',
   294	        'alias_generator',
   295	    }
   296
   297	    def __init__(self, ctx: ClassDefContext, plugin_config: PydanticPluginConfig) -> None:
   298	        self._ctx = ctx
   299	        self.plugin_config = plugin_config

   301	    def transform(self) -> None:
   302	        """
   303	        Configures the BaseModel subclass according to the plugin settings.
   304
   305	        In particular:
   306	        * determines the model config and fields,
   307	        * adds a fields-aware signature for the initializer and construct methods
   308	        * freezes the class if allow_mutation = False or frozen = True
   309	        * stores the fields, config, and if the class is settings in the mypy metadata for access by subclasses
   310	        """
   311	        ctx = self._ctx
   312	        info = ctx.cls.info
   313
   314	        self.adjust_validator_signatures()
   315	        config = self.collect_config()
   316	        fields = self.collect_fields(config)
   317	        is_settings = any(get_fullname(base) == BASESETTINGS_FULLNAME for base in info.mro[:-1])
   318	        self.add_initializer(fields, config, is_settings)
   319	        self.add_construct_method(fields)
   320	        self.set_frozen(fields, frozen=config.allow_mutation is False or config.frozen is True)
   321	        info.metadata[METADATA_KEY] = {
   322	            'fields': {field.name: field.serialize() for field in fields},
   323	            'config': config.set_values_dict(),
   324	        }
   325
   326	    def adjust_validator_signatures(self) -> None:
   327	        """When we decorate a function `f` with `pydantic.validator(...), mypy sees
   328	        `f` as a regular method taking a `self` instance, even though pydantic
   329	        internally wraps `f` with `classmethod` if necessary.
   330
   331	        Teach mypy this by marking any function whose outermost decorator is a
   332	        `validator()` call as a classmethod.
   333	        """
   334	        for name, sym in self._ctx.cls.info.names.items():
   335	            if isinstance(sym.node, Decorator):
   336	                first_dec = sym.node.original_decorators[0]
   337	                if (
   338	                    isinstance(first_dec, CallExpr)
   339	                    and isinstance(first_dec.callee, NameExpr)
   340	                    and first_dec.callee.fullname == f'{_NAMESPACE}.class_validators.validator'
   341	                ):
   342	                    sym.node.func.is_class = True
   343
   344	    def collect_config(self) -> 'ModelConfigData':
   345	        """
   346	        Collects the values of the config attributes that are used by the plugin, accounting for parent classes.
   347	        """
   348	        ctx = self._ctx
   349	        cls = ctx.cls
   350	        config = ModelConfigData()
   351	        for stmt in cls.defs.body:
   352	            if not isinstance(stmt, ClassDef):
   353	                continue
   354	            if stmt.name == 'Config':
   355	                for substmt in stmt.defs.body:
   356	                    if not isinstance(substmt, AssignmentStmt):
   357	                        continue
   358	                    config.update(self.get_config_update(substmt))
   359	                if (
   360	                    config.has_alias_generator
   361	                    and not config.allow_population_by_field_name
   362	                    and self.plugin_config.warn_required_dynamic_aliases
   363	                ):
   364	                    error_required_dynamic_aliases(ctx.api, stmt)
   365	        for info in cls.info.mro[1:]:  # 0 is the current class
   366	            if METADATA_KEY not in info.metadata:
   367	                continue
   368
   369	            # Each class depends on the set of fields in its ancestors
   370	            ctx.api.add_plugin_dependency(make_wildcard_trigger(get_fullname(info)))
   371	            for name, value in info.metadata[METADATA_KEY]['config'].items():
   372	                config.setdefault(name, value)
   373	        return config

   375	    def collect_fields(self, model_config: 'ModelConfigData') -> List['PydanticModelField']:
   376	        """
   377	        Collects the fields for the model, accounting for parent classes
   378	        """
   379	        # First, collect fields belonging to the current class.
   380	        ctx = self._ctx
   381	        cls = self._ctx.cls
   382	        fields = []  # type: List[PydanticModelField]
   383	        known_fields = set()  # type: Set[str]
   384	        for stmt in cls.defs.body:
   385	            if not isinstance(stmt, AssignmentStmt):  # `and stmt.new_syntax` to require annotation
   386	                continue
   387
   388	            lhs = stmt.lvalues[0]
   389	            if not isinstance(lhs, NameExpr) or not is_valid_field(lhs.name):
   390	                continue
   391
   392	            if not stmt.new_syntax and self.plugin_config.warn_untyped_fields:
   393	                error_untyped_fields(ctx.api, stmt)
   394
   395	            # if lhs.name == '__config__':  # BaseConfig not well handled; I'm not sure why yet
   396	            #     continue
   397
   398	            sym = cls.info.names.get(lhs.name)
   399	            if sym is None:  # pragma: no cover
   400	                # This is likely due to a star import (see the dataclasses plugin for a more detailed explanation)
   401	                # This is the same logic used in the dataclasses plugin
   402	                continue
   403
   404	            node = sym.node
   405	            if isinstance(node, PlaceholderNode):  # pragma: no cover
   406	                # See the PlaceholderNode docstring for more detail about how this can occur
   407	                # Basically, it is an edge case when dealing with complex import logic
   408	                # This is the same logic used in the dataclasses plugin
   409	                continue
   410	            if not isinstance(node, Var):  # pragma: no cover
   411	                # Don't know if this edge case still happens with the `is_valid_field` check above
   412	                # but better safe than sorry
   413	                continue
   414
   415	            # x: ClassVar[int] is ignored by dataclasses.
   416	            if node.is_classvar:
   417	                continue
   418
   419	            is_required = self.get_is_required(cls, stmt, lhs)
   420	            alias, has_dynamic_alias = self.get_alias_info(stmt)
   421	            if (
   422	                has_dynamic_alias
   423	                and not model_config.allow_population_by_field_name
   424	                and self.plugin_config.warn_required_dynamic_aliases
   425	            ):
   426	                error_required_dynamic_aliases(ctx.api, stmt)
   427	            fields.append(
   428	                PydanticModelField(
   429	                    name=lhs.name,
   430	                    is_required=is_required,
   431	                    alias=alias,
   432	                    has_dynamic_alias=has_dynamic_alias,
   433	                    line=stmt.line,
   434	                    column=stmt.column,
   435	                )
   436	            )
   437	            known_fields.add(lhs.name)
   438	        all_fields = fields.copy()
   439	        for info in cls.info.mro[1:]:  # 0 is the current class, -2 is BaseModel, -1 is object
   440	            if METADATA_KEY not in info.metadata:
   441	                continue
   442
   443	            superclass_fields = []
   444	            # Each class depends on the set of fields in its ancestors
   445	            ctx.api.add_plugin_dependency(make_wildcard_trigger(get_fullname(info)))
   446
   447	            for name, data in info.metadata[METADATA_KEY]['fields'].items():
   448	                if name not in known_fields:
   449	                    field = PydanticModelField.deserialize(info, data)
   450	                    known_fields.add(name)
   451	                    superclass_fields.append(field)
   452	                else:
   453	                    (field,) = (a for a in all_fields if a.name == name)
   454	                    all_fields.remove(field)
   455	                    superclass_fields.append(field)
   456	            all_fields = superclass_fields + all_fields
   457	        return all_fields

   459	    def add_initializer(self, fields: List['PydanticModelField'], config: 'ModelConfigData', is_settings: bool) -> None:
   460	        """
   461	        Adds a fields-aware `__init__` method to the class.
   462
   463	        The added `__init__` will be annotated with types vs. all `Any` depending on the plugin settings.
   464	        """
   465	        ctx = self._ctx
   466	        typed = self.plugin_config.init_typed
   467	        use_alias = config.allow_population_by_field_name is not True
   468	        force_all_optional = is_settings or bool(
   469	            config.has_alias_generator and not config.allow_population_by_field_name
   470	        )
   471	        init_arguments = self.get_field_arguments(
   472	            fields, typed=typed, force_all_optional=force_all_optional, use_alias=use_alias
   473	        )
   474	        if not self.should_init_forbid_extra(fields, config):
   475	            var = Var('kwargs')
   476	            init_arguments.append(Argument(var, AnyType(TypeOfAny.explicit), None, ARG_STAR2))
   477
   478	        if '__init__' not in ctx.cls.info.names:
   479	            add_method(ctx, '__init__', init_arguments, NoneType())

   481	    def add_construct_method(self, fields: List['PydanticModelField']) -> None:
   482	        """
   483	        Adds a fully typed `construct` classmethod to the class.
   484
   485	        Similar to the fields-aware __init__ method, but always uses the field names (not aliases),
   486	        and does not treat settings fields as optional.
   487	        """
   488	        ctx = self._ctx
   489	        set_str = ctx.api.named_type(f'{BUILTINS_NAME}.set', [ctx.api.named_type(f'{BUILTINS_NAME}.str')])
   490	        optional_set_str = UnionType([set_str, NoneType()])
   491	        fields_set_argument = Argument(Var('_fields_set', optional_set_str), optional_set_str, None, ARG_OPT)
   492	        construct_arguments = self.get_field_arguments(fields, typed=True, force_all_optional=False, use_alias=False)
   493	        construct_arguments = [fields_set_argument] + construct_arguments
   494
   495	        obj_type = ctx.api.named_type(f'{BUILTINS_NAME}.object')
   496	        self_tvar_name = '_PydanticBaseModel'  # Make sure it does not conflict with other names in the class
   497	        tvar_fullname = ctx.cls.fullname + '.' + self_tvar_name
   498	        if MYPY_VERSION_TUPLE >= (1, 4):
   499	            tvd = TypeVarType(
   500	                self_tvar_name,
   501	                tvar_fullname,
   502	                (
   503	                    TypeVarId(-1, namespace=ctx.cls.fullname + '.construct')
   504	                    if MYPY_VERSION_TUPLE >= (1, 11)
   505	                    else TypeVarId(-1)
   506	                ),
   507	                [],
   508	                obj_type,
   509	                AnyType(TypeOfAny.from_omitted_generics),  # type: ignore[arg-type]
   510	            )
   511	            self_tvar_expr = TypeVarExpr(
   512	                self_tvar_name,
   513	                tvar_fullname,
   514	                [],
   515	                obj_type,
   516	                AnyType(TypeOfAny.from_omitted_generics),  # type: ignore[arg-type]
   517	            )
   518	        else:
   519	            tvd = TypeVarDef(self_tvar_name, tvar_fullname, -1, [], obj_type)
   520	            self_tvar_expr = TypeVarExpr(self_tvar_name, tvar_fullname, [], obj_type)
   521	        ctx.cls.info.names[self_tvar_name] = SymbolTableNode(MDEF, self_tvar_expr)
   522
   523	        # Backward-compatible with TypeVarDef from Mypy 0.910.
   524	        if isinstance(tvd, TypeVarType):
   525	            self_type = tvd
   526	        else:
   527	            self_type = TypeVarType(tvd)
   528
   529	        add_method(
   530	            ctx,
   531	            'construct',
   532	            construct_arguments,
   533	            return_type=self_type,
   534	            self_type=self_type,
   535	            tvar_def=tvd,
   536	            is_classmethod=True,
   537	        )

   539	    def set_frozen(self, fields: List['PydanticModelField'], frozen: bool) -> None:
   540	        """
   541	        Marks all fields as properties so that attempts to set them trigger mypy errors.
   542
   543	        This is the same approach used by the attrs and dataclasses plugins.
   544	        """
   545	        ctx = self._ctx
   546	        info = ctx.cls.info
   547	        for field in fields:
   548	            sym_node = info.names.get(field.name)
   549	            if sym_node is not None:
   550	                var = sym_node.node
   551	                if isinstance(var, Var):
   552	                    var.is_property = frozen
