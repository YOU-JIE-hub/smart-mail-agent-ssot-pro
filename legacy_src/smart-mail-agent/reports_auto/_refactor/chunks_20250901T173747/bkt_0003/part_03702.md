   270	    def get_default_function_sig(self, func: object, ctx: FunctionContext) -> FunctionSig:
   271	        argspec = None
   272	        if not self.is_c_module:
   273	            # Get the full argument specification of the function
   274	            try:
   275	                argspec = inspect.getfullargspec(func)
   276	            except TypeError:
   277	                # some callables cannot be inspected, e.g. functools.partial
   278	                pass
   279	        if argspec is None:
   280	            if ctx.class_info is not None:
   281	                # method:
   282	                return FunctionSig(
   283	                    name=ctx.name,
   284	                    args=infer_c_method_args(ctx.name, ctx.class_info.self_var),
   285	                    ret_type=infer_method_ret_type(ctx.name),
   286	                )
   287	            else:
   288	                # function:
   289	                return FunctionSig(
   290	                    name=ctx.name,
   291	                    args=[ArgSig(name="*args"), ArgSig(name="**kwargs")],
   292	                    ret_type=None,
   293	                )
   294
   295	        # Extract the function arguments, defaults, and varargs
   296	        args = argspec.args
   297	        defaults = argspec.defaults
   298	        varargs = argspec.varargs
   299	        kwargs = argspec.varkw
   300	        annotations = argspec.annotations
   301	        kwonlyargs = argspec.kwonlyargs
   302	        kwonlydefaults = argspec.kwonlydefaults
   303
   304	        def get_annotation(key: str) -> str | None:
   305	            if key not in annotations:
   306	                return None
   307	            argtype = annotations[key]
   308	            if argtype is None:
   309	                return "None"
   310	            if not isinstance(argtype, str):
   311	                return self.get_type_fullname(argtype)
   312	            return argtype
   313
   314	        arglist: list[ArgSig] = []
   315
   316	        # Add the arguments to the signature
   317	        def add_args(
   318	            args: list[str], get_default_value: Callable[[int, str], object | _Missing]
   319	        ) -> None:
   320	            for i, arg in enumerate(args):
   321	                # Check if the argument has a default value
   322	                default_value = get_default_value(i, arg)
   323	                if default_value is not _Missing.VALUE:
   324	                    if arg in annotations:
   325	                        argtype = annotations[arg]
   326	                    else:
   327	                        argtype = self.get_type_annotation(default_value)
   328	                        if argtype == "None":
   329	                            # None is not a useful annotation, but we can infer that the arg
   330	                            # is optional
   331	                            incomplete = self.add_name("_typeshed.Incomplete")
   332	                            argtype = f"{incomplete} | None"
   333
   334	                    arglist.append(ArgSig(arg, argtype, default=True))
   335	                else:
   336	                    arglist.append(ArgSig(arg, get_annotation(arg), default=False))
   337
   338	        def get_pos_default(i: int, _arg: str) -> Any | _Missing:
   339	            if defaults and i >= len(args) - len(defaults):
   340	                return defaults[i - (len(args) - len(defaults))]
   341	            else:
   342	                return _Missing.VALUE
   343
   344	        add_args(args, get_pos_default)
   345
   346	        # Add *args if present
   347	        if varargs:
   348	            arglist.append(ArgSig(f"*{varargs}", get_annotation(varargs)))
   349	        # if we have keyword only args, then we need to add "*"
   350	        elif kwonlyargs:
   351	            arglist.append(ArgSig("*"))
   352
   353	        def get_kw_default(_i: int, arg: str) -> Any | _Missing:
   354	            if kwonlydefaults and arg in kwonlydefaults:
   355	                return kwonlydefaults[arg]
   356	            else:
   357	                return _Missing.VALUE
   358
   359	        add_args(kwonlyargs, get_kw_default)
   360
   361	        # Add **kwargs if present
   362	        if kwargs:
   363	            arglist.append(ArgSig(f"**{kwargs}", get_annotation(kwargs)))
   364
   365	        # add types for known special methods
   366	        if ctx.class_info is not None and all(
   367	            arg.type is None and arg.default is False for arg in arglist
   368	        ):
   369	            new_args = infer_method_arg_types(
   370	                ctx.name, ctx.class_info.self_var, [arg.name for arg in arglist if arg.name]
   371	            )
   372	            if new_args is not None:
   373	                arglist = new_args
   374
   375	        ret_type = get_annotation("return") or infer_method_ret_type(ctx.name)
   376	        return FunctionSig(ctx.name, arglist, ret_type)

   388	    def strip_or_import(self, type_name: str) -> str:
   389	        """Strips unnecessary module names from typ.
   390
   391	        If typ represents a type that is inside module or is a type coming from builtins, remove
   392	        module declaration from it. Return stripped name of the type.
   393
   394	        Arguments:
   395	            typ: name of the type
   396	        """
   397	        local_modules = ["builtins", self.module_name]
   398	        parsed_type = parse_type_comment(type_name, 0, 0, None)[1]
   399	        assert parsed_type is not None, type_name
   400	        return self.print_annotation(parsed_type, self.known_modules, local_modules)

   411	    def generate_module(self) -> None:
   412	        all_items = self.get_members(self.module)
   413	        if self.resort_members:
   414	            all_items = sorted(all_items, key=lambda x: x[0])
   415	        items = []
   416	        for name, obj in all_items:
   417	            if inspect.ismodule(obj) and obj.__name__ in self.known_modules:
   418	                module_name = obj.__name__
   419	                if module_name.startswith(self.module_name + "."):
   420	                    # from {.rel_name} import {mod_name} as {name}
   421	                    pkg_name, mod_name = module_name.rsplit(".", 1)
   422	                    rel_module = pkg_name[len(self.module_name) :] or "."
   423	                    self.import_tracker.add_import_from(rel_module, [(mod_name, name)])
   424	                    self.import_tracker.reexport(name)
   425	                else:
   426	                    # import {module_name} as {name}
   427	                    self.import_tracker.add_import(module_name, name)
   428	                    self.import_tracker.reexport(name)
   429	            elif self.is_defined_in_module(obj) and not inspect.ismodule(obj):
   430	                # process this below
   431	                items.append((name, obj))
   432	            else:
   433	                # from {obj_module} import {obj_name}
   434	                obj_module_name = self.get_obj_module(obj)
   435	                if obj_module_name:
   436	                    self.import_tracker.add_import_from(obj_module_name, [(name, None)])
   437	                    if self.should_reexport(name, obj_module_name, name_is_alias=False):
   438	                        self.import_tracker.reexport(name)
   439
   440	        self.set_defined_names({name for name, obj in all_items if not inspect.ismodule(obj)})
   441
   442	        if self.resort_members:
   443	            functions: list[str] = []
   444	            types: list[str] = []
   445	            variables: list[str] = []
   446	        else:
   447	            output: list[str] = []
   448	            functions = types = variables = output
   449
   450	        for name, obj in items:
   451	            if self.is_function(obj):
   452	                self.generate_function_stub(name, obj, output=functions)
   453	            elif inspect.isclass(obj):
   454	                self.generate_class_stub(name, obj, output=types)
   455	            else:
   456	                self.generate_variable_stub(name, obj, output=variables)
   457
   458	        self._output = []
   459
   460	        if self.resort_members:
   461	            for line in variables:
   462	                self._output.append(line + "\n")
   463	            for line in types:
   464	                if line.startswith("class") and self._output and self._output[-1]:
   465	                    self._output.append("\n")
   466	                self._output.append(line + "\n")
   467	            if self._output and functions:
   468	                self._output.append("\n")
   469	            for line in functions:
   470	                self._output.append(line + "\n")
   471	        else:
   472	            for i, line in enumerate(output):
   473	                if (
   474	                    self._output
   475	                    and line.startswith("class")
   476	                    and (
   477	                        not self._output[-1].startswith("class")
   478	                        or (len(output) > i + 1 and output[i + 1].startswith("    "))
   479	                    )
   480	                ) or (
   481	                    self._output
   482	                    and self._output[-1].startswith("def")
   483	                    and not line.startswith("def")
   484	                ):
   485	                    self._output.append("\n")
   486	                self._output.append(line + "\n")
   487	        self.check_undefined_names()

   490	        return (
   491	            attr
   492	            in (

   494	                "__getattribute__",

   496	                "__repr__",
   497	                "__doc__",

   501	                "__annotations__",

   503	                "__static_attributes__",
   504	                "__annotate__",
   505	            )

   511	    def get_members(self, obj: object) -> list[tuple[str, Any]]:
   512	        obj_dict: Mapping[str, Any] = getattr(obj, "__dict__")  # noqa: B009
   513	        results = []
   514	        for name in obj_dict:
   515	            if self.is_skipped_attribute(name):
   516	                continue
   517	            # Try to get the value via getattr
   518	            try:
   519	                value = getattr(obj, name)
   520	            except AttributeError:
   521	                continue
   522	            else:
   523	                results.append((name, value))
   524	        return results

   526	    def get_type_annotation(self, obj: object) -> str:
   527	        """
   528	        Given an instance, return a string representation of its type that is valid
   529	        to use as a type annotation.
   530	        """
   531	        if obj is None or obj is type(None):
   532	            return "None"
   533	        elif inspect.isclass(obj):

   537	        elif isinstance(obj, ModuleType):

   539	        else:

   606	    def generate_function_stub(
   607	        self, name: str, obj: object, *, output: list[str], class_info: ClassInfo | None = None
   608	    ) -> None:
   609	        """Generate stub for a single function or method.
   610
   611	        The result (always a single line) will be appended to 'output'.
   612	        If necessary, any required names will be added to 'imports'.
   613	        The 'class_name' is used to find signature of __init__ or __new__ in
   614	        'class_sigs'.
   615	        """
   616	        docstring: Any = getattr(obj, "__doc__", None)
   617	        if not isinstance(docstring, str):
   618	            docstring = None
   619
   620	        ctx = FunctionContext(
   621	            self.module_name,
   622	            name,
   623	            docstring=docstring,
   624	            is_abstract=self.is_abstract_method(obj),
   625	            class_info=class_info,
   626	        )
   627	        if self.is_private_name(name, ctx.fullname) or self.is_not_in_all(name):
   628	            return
   629
   630	        self.record_name(ctx.name)
   631	        default_sig = self.get_default_function_sig(obj, ctx)
   632	        inferred = self.get_signatures(default_sig, self.sig_generators, ctx)
   633	        self.process_inferred_sigs(inferred)
   634
   635	        decorators = []
   636	        if len(inferred) > 1:
   637	            decorators.append("@{}".format(self.add_name("typing.overload")))
   638
   639	        if ctx.is_abstract:
   640	            decorators.append("@{}".format(self.add_name("abc.abstractmethod")))
   641
   642	        if class_info is not None:
   643	            if self.is_staticmethod(class_info, name, obj):
   644	                decorators.append("@staticmethod")
   645	            else:
   646	                for sig in inferred:
   647	                    if not sig.args or sig.args[0].name not in ("self", "cls"):
   648	                        sig.args.insert(0, ArgSig(name=class_info.self_var))
   649	                # a sig generator indicates @classmethod by specifying the cls arg.
   650	                if inferred[0].args and inferred[0].args[0].name == "cls":
   651	                    decorators.append("@classmethod")
   652
   653	        docstring = self._indent_docstring(ctx.docstring) if ctx.docstring else None
   654	        output.extend(self.format_func_def(inferred, decorators=decorators, docstring=docstring))
   655	        self._fix_iter(ctx, inferred, output)
   656
   657	    def _indent_docstring(self, docstring: str) -> str:
   658	        """Fix indentation of docstring extracted from pybind11 or other binding generators."""
   659	        lines = docstring.splitlines(keepends=True)
   660	        indent = self._indent + "    "
   661	        if len(lines) > 1:
   662	            if not all(line.startswith(indent) or not line.strip() for line in lines):
   663	                # if the docstring is not indented, then indent all but the first line
   664	                for i, line in enumerate(lines[1:]):
   665	                    if line.strip():
   666	                        lines[i + 1] = indent + line
   667	        # if there's a trailing newline, add a final line to visually indent the quoted docstring
   668	        if lines[-1].endswith("\n"):
   669	            if len(lines) > 1:
   670	                lines.append(indent)
   671	            else:
   672	                lines[-1] = lines[-1][:-1]
   673	        return "".join(lines)

   675	    def _fix_iter(
   676	        self, ctx: FunctionContext, inferred: list[FunctionSig], output: list[str]
   677	    ) -> None:
   678	        """Ensure that objects which implement old-style iteration via __getitem__
   679	        are considered iterable.
   680	        """
   681	        if (
   682	            ctx.class_info
   683	            and ctx.class_info.cls is not None
   684	            and ctx.name == "__getitem__"
   685	            and "__iter__" not in ctx.class_info.cls.__dict__
   686	        ):
   687	            item_type: str | None = None
   688	            for sig in inferred:
   689	                if sig.args and sig.args[-1].type == "int":
   690	                    item_type = sig.ret_type
   691	                    break
   692	            if item_type is None:
   693	                return
   694	            obj = CFunctionStub(
   695	                "__iter__", f"def __iter__(self) -> typing.Iterator[{item_type}]\n"
   696	            )
   697	            self.generate_function_stub("__iter__", obj, output=output, class_info=ctx.class_info)

   699	    def generate_property_stub(
   700	        self,
   701	        name: str,
   702	        raw_obj: object,
   703	        obj: object,
   704	        static_properties: list[str],
   705	        rw_properties: list[str],
   706	        ro_properties: list[str],
   707	        class_info: ClassInfo | None = None,
   708	    ) -> None:
   709	        """Generate property stub using introspection of 'obj'.
   710
   711	        Try to infer type from docstring, append resulting lines to 'output'.
   712
   713	        raw_obj : object before evaluation of descriptor (if any)
   714	        obj : object after evaluation of descriptor
   715	        """
   716
   717	        docstring = getattr(raw_obj, "__doc__", None)
   718	        fget = getattr(raw_obj, "fget", None)
   719	        if fget:
   720	            alt_docstr = getattr(fget, "__doc__", None)
   721	            if alt_docstr and docstring:
   722	                docstring += "\n" + alt_docstr
   723	            elif alt_docstr:
   724	                docstring = alt_docstr
   725
   726	        ctx = FunctionContext(
   727	            self.module_name, name, docstring=docstring, is_abstract=False, class_info=class_info
   728	        )
   729
   730	        if self.is_private_name(name, ctx.fullname) or self.is_not_in_all(name):
   731	            return
   732
   733	        self.record_name(ctx.name)
   734	        static = self.is_static_property(raw_obj)
   735	        readonly = self.is_property_readonly(raw_obj)
   736	        if static:
   737	            ret_type: str | None = self.strip_or_import(self.get_type_annotation(obj))
   738	        else:
   739	            default_sig = self.get_default_function_sig(raw_obj, ctx)
   740	            ret_type = default_sig.ret_type
   741
   742	        inferred_type = self.get_property_type(ret_type, self.sig_generators, ctx)
   743	        if inferred_type is not None:
   744	            inferred_type = self.strip_or_import(inferred_type)
   745
   746	        if static:
   747	            classvar = self.add_name("typing.ClassVar")
   748	            trailing_comment = "  # read-only" if readonly else ""
   749	            if inferred_type is None:
   750	                inferred_type = self.add_name("_typeshed.Incomplete")
   751
   752	            static_properties.append(
   753	                f"{self._indent}{name}: {classvar}[{inferred_type}] = ...{trailing_comment}"
   754	            )
   755	        else:  # regular property
   756	            if readonly:
   757	                docstring = self._indent_docstring(ctx.docstring) if ctx.docstring else None
   758	                ro_properties.append(f"{self._indent}@property")

   492	            and not mypy.types.is_named_instance(
   493	                stub.metaclass_type, ("abc.ABCMeta", "builtins.type")
   494	            )
   495	        ):
   496	            # This means that our stub has a metaclass that is not present at runtime.
   497	            yield Error(
   498	                object_path,
   499	                "metaclass mismatch",
   500	                stub,
   501	                runtime,
   502	                stub_desc=f"{stub.metaclass_type.type.fullname}",

   507	@verify.register(nodes.TypeInfo)
   508	def verify_typeinfo(
   509	    stub: nodes.TypeInfo, runtime: MaybeMissing[type[Any]], object_path: list[str]
   510	) -> Iterator[Error]:
   511	    if stub.is_type_check_only:
   512	        # This type only exists in stubs, we only check that the runtime part
   513	        # is missing. Other checks are not required.
   514	        if not isinstance(runtime, Missing):
   515	            yield Error(
   516	                object_path,
   517	                'is marked as "@type_check_only", but also exists at runtime',
   518	                stub,
   519	                runtime,
   520	                stub_desc=repr(stub),
   521	            )
   522	        return
   523
   524	    if isinstance(runtime, Missing):
   525	        yield Error(object_path, "is not present at runtime", stub, runtime, stub_desc=repr(stub))
   526	        return
   527	    if not isinstance(runtime, type):
   528	        # Yes, some runtime objects can be not types, no way to tell mypy about that.
   529	        yield Error(object_path, "is not a type", stub, runtime, stub_desc=repr(stub))  # type: ignore[unreachable]
   530	        return
   531
   532	    yield from _verify_final(stub, runtime, object_path)
   533	    is_runtime_typeddict = stub.typeddict_type is not None and is_typeddict(runtime)
   534	    yield from _verify_metaclass(
   535	        stub, runtime, object_path, is_runtime_typeddict=is_runtime_typeddict
   536	    )
   537
   538	    # Check everything already defined on the stub class itself (i.e. not inherited)
   539	    #
   540	    # Filter out non-identifier names, as these are (hopefully always?) whacky/fictional things
   541	    # (like __mypy-replace or __mypy-post_init, etc.) that don't exist at runtime,
   542	    # and exist purely for internal mypy reasons
   543	    to_check = {name for name in stub.names if name.isidentifier()}
   544	    # Check all public things on the runtime class
   545	    to_check.update(
   546	        m for m in vars(runtime) if not is_probably_private(m) and m not in IGNORABLE_CLASS_DUNDERS
   547	    )
   548	    # Special-case the __init__ method for Protocols and the __new__ method for TypedDicts
   549	    #
   550	    # TODO: On Python <3.11, __init__ methods on Protocol classes
   551	    # are silently discarded and replaced.
   552	    # However, this is not the case on Python 3.11+.
   553	    # Ideally, we'd figure out a good way of validating Protocol __init__ methods on 3.11+.
   554	    if stub.is_protocol:
   555	        to_check.discard("__init__")
   556	    if is_runtime_typeddict:
   557	        to_check.discard("__new__")
   558
   559	    for entry in sorted(to_check):
   560	        mangled_entry = entry
   561	        if entry.startswith("__") and not entry.endswith("__"):
   562	            mangled_entry = f"_{stub.name.lstrip('_')}{entry}"
   563	        stub_to_verify = next((t.names[entry].node for t in stub.mro if entry in t.names), MISSING)
   564	        assert stub_to_verify is not None
   565	        try:
   566	            try:
   567	                runtime_attr = getattr(runtime, mangled_entry)
   568	            except AttributeError:
   569	                runtime_attr = inspect.getattr_static(runtime, mangled_entry, MISSING)
   570	        except Exception:
   571	            # Catch all exceptions in case the runtime raises an unexpected exception
   572	            # from __getattr__ or similar.
   573	            continue
   574
   575	        # If it came from the metaclass, consider the runtime_attr to be MISSING
   576	        # for a more accurate message
   577	        if (
   578	            runtime_attr is not MISSING
   579	            and type(runtime) is not runtime
   580	            and getattr(runtime_attr, "__objclass__", None) is type(runtime)
   581	        ):
   582	            runtime_attr = MISSING
   583
   584	        # __setattr__ and __delattr__ on object are a special case,
   585	        # so if we only have these methods inherited from there, pretend that
   586	        # we don't have them. See python/typeshed#7385.
   587	        if (
   588	            entry in ("__setattr__", "__delattr__")
   589	            and runtime_attr is not MISSING
   590	            and runtime is not object
   591	            and getattr(runtime_attr, "__objclass__", None) is object
   592	        ):
   593	            runtime_attr = MISSING
   594
   595	        # Do not error for an object missing from the stub
   596	        # If the runtime object is a types.WrapperDescriptorType object
   597	        # and has a non-special dunder name.
   598	        # The vast majority of these are false positives.
   599	        if not (
   600	            isinstance(stub_to_verify, Missing)
   601	            and isinstance(runtime_attr, types.WrapperDescriptorType)
   602	            and is_dunder(mangled_entry, exclude_special=True)
   603	        ):
   604	            yield from verify(stub_to_verify, runtime_attr, object_path + [entry])
