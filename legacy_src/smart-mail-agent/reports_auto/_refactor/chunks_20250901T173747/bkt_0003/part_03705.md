  1768	        # (*Any) a supertype of all callables with positional arguments. This is needed in
  1769	        # particular because we often refuse to try type inference if actual type is not

  1778	    # or both. If L's corresponding argument is ambiguous, L is not a subtype of R.

  1784	    # properly stand in for the full domain of formal arguments that they're
  1785	    # used for in practice.
  1786
  1787	    # Every argument in R must have a corresponding argument in L, and every
  1788	    # required argument in L must have a corresponding argument in R.

  1792	    # Phase 1a: If left and right can both accept an infinite number of args,
  1793	    #           their types must be compatible.

  1840	    if right_star is not None and not trivial_vararg_suffix:

  1849	                break

  1866	    #           they're more general than the corresponding member in right.

  1901	    #          (Every required argument in L must have a corresponding argument in R)

  1903	    for left_arg in left.formal_arguments():

  1909	            right.argument_by_position(left_arg.pos) if left_arg.pos is not None else None

  1913	        # neither of them can be required.
  1914	        if (
  1915	            right_by_name is not None
  1916	            and right_by_pos is not None
  1917	            and right_by_name != right_by_pos
  1918	            and (right_by_pos.required or right_by_name.required)
  1919	            and strict_concatenate_check
  1920	            and not right.imprecise_arg_kinds
  1921	        ):

  1925	        # right-hand argument.  Optional args do not matter.
  1926	        if left_arg.required and right_by_pos is None and right_by_name is None:
  1927	            return False

  1932	def are_args_compatible(
  1933	    left: FormalArgument,
  1934	    right: FormalArgument,
  1935	    is_compat: Callable[[Type, Type], bool],
  1936	    *,
  1937	    ignore_pos_arg_names: bool,
  1938	    allow_partial_overlap: bool,
  1939	    allow_imprecise_kinds: bool = False,
  1940	) -> bool:
  1941	    if left.required and right.required:
  1942	        # If both arguments are required allow_partial_overlap has no effect.
  1943	        allow_partial_overlap = False
  1944
  1945	    def is_different(
  1946	        left_item: object | None, right_item: object | None, allow_overlap: bool
  1947	    ) -> bool:
  1948	        """Checks if the left and right items are different.
  1949
  1950	        If the right item is unspecified (e.g. if the right callable doesn't care
  1951	        about what name or position its arg has), we default to returning False.
  1952
  1953	        If we're allowing partial overlap, we also default to returning False
  1954	        if the left callable also doesn't care."""
  1955	        if right_item is None:
  1956	            return False
  1957	        if allow_overlap and left_item is None:
  1958	            return False
  1959	        return left_item != right_item
  1960
  1961	    # If right has a specific name it wants this argument to be, left must
  1962	    # have the same.
  1963	    if is_different(left.name, right.name, allow_partial_overlap):
  1964	        # But pay attention to whether we're ignoring positional arg names
  1965	        if not ignore_pos_arg_names or right.pos is None:
  1966	            return False
  1967
  1968	    # If right is at a specific position, left must have the same.
  1969	    # TODO: partial overlap logic is flawed for positions.
  1970	    # We disable it to avoid false positives at a cost of few false negatives.
  1971	    if is_different(left.pos, right.pos, allow_overlap=False) and not allow_imprecise_kinds:
  1972	        return False
  1973
  1974	    # If right's argument is optional, left's must also be
  1975	    # (unless we're relaxing the checks to allow potential
  1976	    # rather than definite compatibility).
  1977	    if not allow_partial_overlap and not right.required and left.required:
  1978	        return False
  1979
  1980	    # If we're allowing partial overlaps and neither arg is required,
  1981	    # the types don't actually need to be the same
  1982	    if allow_partial_overlap and not left.required and not right.required:
  1983	        return True
  1984
  1985	    # Left must have a more general type
  1986	    return is_compat(right.typ, left.typ)

     3	Currently works by collecting all argument types at callsites,
     4	synthesizing a list of possible function types from that, trying them
     5	all, and picking the one with the fewest errors that we think is the
     6	"best".
     7
     8	Can return JSON that pyannotate can use to apply the annotations to code.
     9
    10	There are a bunch of TODOs here:
    11	 * Maybe want a way to surface the choices not selected??
    12	 * We can generate an exponential number of type suggestions, and probably want
    13	   a way to not always need to check them all.
    14	 * Our heuristics for what types to try are primitive and not yet
    15	   supported by real practice.
    16	 * More!
    17
    18	Other things:
    19	 * This is super brute force. Could we integrate with the typechecker
    20	   more to understand more about what is going on?
    21	 * Like something with tracking constraints/unification variables?
    22	 * No understanding of type variables at *all*
    23	"""

   101	class SuggestionPlugin(Plugin):
   102	    """Plugin that records all calls to a given target."""
   103
   104	    def __init__(self, target: str) -> None:
   105	        if target.endswith((".__new__", ".__init__")):
   106	            target = target.rsplit(".", 1)[0]
   107
   108	        self.target = target
   109	        # List of call sites found by dmypy suggest:
   110	        # (path, line, <arg kinds>, <arg names>, <arg types>)
   111	        self.mystery_hits: list[Callsite] = []
   112
   113	    def get_function_hook(self, fullname: str) -> Callable[[FunctionContext], Type] | None:
   114	        if fullname == self.target:
   115	            return self.log
   116	        else:
   117	            return None
   118
   119	    def get_method_hook(self, fullname: str) -> Callable[[MethodContext], Type] | None:
   120	        if fullname == self.target:
   121	            return self.log
   122	        else:
   123	            return None
   124
   125	    def log(self, ctx: FunctionContext | MethodContext) -> Type:
   126	        self.mystery_hits.append(
   127	            Callsite(
   128	                ctx.api.path,
   129	                ctx.context.line,
   130	                ctx.arg_kinds,
   131	                ctx.callee_arg_names,
   132	                ctx.arg_names,
   133	                ctx.arg_types,
   134	            )
   135	        )
   136	        return ctx.default_return_type

   141	class ReturnFinder(TraverserVisitor):
   142	    """Visitor for finding all types returned from a function."""
   143
   144	    def __init__(self, typemap: dict[Expression, Type]) -> None:
   145	        self.typemap = typemap
   146	        self.return_types: list[Type] = []
   147
   148	    def visit_return_stmt(self, o: ReturnStmt) -> None:
   149	        if o.expr is not None and o.expr in self.typemap:
   150	            self.return_types.append(self.typemap[o.expr])
   151
   152	    def visit_func_def(self, o: FuncDef) -> None:
   153	        # Skip nested functions
   154	        pass

   157	def get_return_types(typemap: dict[Expression, Type], func: FuncDef) -> list[Type]:
   158	    """Find all the types returned by return statements in func."""
   159	    finder = ReturnFinder(typemap)
   160	    func.body.accept(finder)
   161	    return finder.return_types

   164	class ArgUseFinder(TraverserVisitor):
   165	    """Visitor for finding all the types of arguments that each arg is passed to.
   166
   167	    This is extremely simple minded but might be effective anyways.
   168	    """
   169
   170	    def __init__(self, func: FuncDef, typemap: dict[Expression, Type]) -> None:
   171	        self.typemap = typemap
   172	        self.arg_types: dict[SymbolNode, list[Type]] = {arg.variable: [] for arg in func.arguments}
   173
   174	    def visit_call_expr(self, o: CallExpr) -> None:
   175	        if not any(isinstance(e, RefExpr) and e.node in self.arg_types for e in o.args):
   176	            return
   177
   178	        typ = get_proper_type(self.typemap.get(o.callee))
   179	        if not isinstance(typ, CallableType):
   180	            return
   181
   182	        formal_to_actual = map_actuals_to_formals(
   183	            o.arg_kinds,
   184	            o.arg_names,
   185	            typ.arg_kinds,
   186	            typ.arg_names,
   187	            lambda n: AnyType(TypeOfAny.special_form),
   188	        )
   189
   190	        for i, args in enumerate(formal_to_actual):
   191	            for arg_idx in args:
   192	                arg = o.args[arg_idx]
   193	                if isinstance(arg, RefExpr) and arg.node in self.arg_types:
   194	                    self.arg_types[arg.node].append(typ.arg_types[i])
   195

   197	def get_arg_uses(typemap: dict[Expression, Type], func: FuncDef) -> list[list[Type]]:
   198	    """Find all the types of arguments that each arg is passed to.
   199
   200	    For example, given
   201	      def foo(x: int) -> None: ...
   202	      def bar(x: str) -> None: ...
   203	      def test(x, y):
   204	          foo(x)
   205	          bar(y)
   206
   207	    this will return [[int], [str]].
   208	    """
   209	    finder = ArgUseFinder(func, typemap)
   210	    func.body.accept(finder)
   211	    return [finder.arg_types[arg.variable] for arg in func.arguments]

   218	def is_explicit_any(typ: AnyType) -> bool:
   219	    # Originally I wanted to count as explicit anything derived from an explicit any, but that
   220	    # seemed too strict in some testing.
   221	    # return (typ.type_of_any == TypeOfAny.explicit
   222	    #         or (typ.source_any is not None and typ.source_any.type_of_any == TypeOfAny.explicit))
   223	    # Important question: what should we do with source_any stuff? Does that count?
   224	    # And actually should explicit anys count at all?? Maybe not!
   225	    return typ.type_of_any == TypeOfAny.explicit

   228	def is_implicit_any(typ: Type) -> bool:
   229	    typ = get_proper_type(typ)
   230	    return isinstance(typ, AnyType) and not is_explicit_any(typ)

   233	class SuggestionEngine:
   234	    """Engine for finding call sites and suggesting signatures."""
   235
   236	    def __init__(
   237	        self,
   238	        fgmanager: FineGrainedBuildManager,
   239	        *,
   240	        json: bool,
   241	        no_errors: bool = False,
   242	        no_any: bool = False,
   243	        flex_any: float | None = None,
   244	        use_fixme: str | None = None,
   245	        max_guesses: int | None = None,
   246	    ) -> None:
   247	        self.fgmanager = fgmanager
   248	        self.manager = fgmanager.manager
   249	        self.plugin = self.manager.plugin
   250	        self.graph = fgmanager.graph
   251	        self.finder = SourceFinder(self.manager.fscache, self.manager.options)
   252
   253	        self.give_json = json
   254	        self.no_errors = no_errors
   255	        self.flex_any = flex_any
   256	        if no_any:
   257	            self.flex_any = 1.0
   258
   259	        self.max_guesses = max_guesses or 64
   260	        self.use_fixme = use_fixme
   261
   262	    def suggest(self, function: str) -> str:
   263	        """Suggest an inferred type for function."""
   264	        mod, func_name, node = self.find_node(function)
   265
   266	        with self.restore_after(mod):
   267	            with self.with_export_types():
   268	                suggestion = self.get_suggestion(mod, node)
   269
   270	        if self.give_json:
   271	            return self.json_suggestion(mod, func_name, node, suggestion)
   272	        else:
   273	            return self.format_signature(suggestion)
   274
   275	    def suggest_callsites(self, function: str) -> str:
   276	        """Find a list of call sites of function."""
   277	        mod, _, node = self.find_node(function)
   278	        with self.restore_after(mod):
   279	            callsites, _ = self.get_callsites(node)
   280
   281	        return "\n".join(
   282	            dedup(
   283	                [
   284	                    f"{path}:{line}: {self.format_args(arg_kinds, arg_names, arg_types)}"
   285	                    for path, line, arg_kinds, _, arg_names, arg_types in callsites
   286	                ]
   287	            )
   288	        )

   290	    @contextmanager
   291	    def restore_after(self, module: str) -> Iterator[None]:
   292	        """Context manager that reloads a module after executing the body.
   293
   294	        This should undo any damage done to the module state while mucking around.
   295	        """
   296	        try:
   297	            yield
   298	        finally:
   299	            self.reload(self.graph[module])

   301	    @contextmanager
   302	    def with_export_types(self) -> Iterator[None]:
   303	        """Context manager that enables the export_types flag in the body.
   304
   305	        This causes type information to be exported into the manager's all_types variable.
   306	        """
   307	        old = self.manager.options.export_types
   308	        self.manager.options.export_types = True
   309	        try:
   310	            yield
   311	        finally:
   312	            self.manager.options.export_types = old

   315	        """Generate a trivial callable type from a func def, with all Anys"""
   316	        # The Anys are marked as being from the suggestion engine

   323	            AnyType(TypeOfAny.suggestion_engine),
   324	            self.named_type("builtins.function"),
   325	        )

   333	    def get_args(
   334	        self,
   335	        is_method: bool,
   336	        base: CallableType,
   337	        defaults: list[Type | None],
   338	        callsites: list[Callsite],
   339	        uses: list[list[Type]],
   340	    ) -> list[list[Type]]:
   341	        """Produce a list of type suggestions for each argument type."""
   342	        types: list[list[Type]] = []
   343	        for i in range(len(base.arg_kinds)):
   344	            # Make self args Any but this will get overridden somewhere in the checker
   345	            if i == 0 and is_method:
   346	                types.append([AnyType(TypeOfAny.suggestion_engine)])
   347	                continue
   348
   349	            all_arg_types = []
   350	            for call in callsites:
   351	                for typ in call.arg_types[i - is_method]:
   352	                    # Collect all the types except for implicit anys
   353	                    if not is_implicit_any(typ):
   354	                        all_arg_types.append(typ)
   355	            all_use_types = []
   356	            for typ in uses[i]:
   357	                # Collect all the types except for implicit anys
   358	                if not is_implicit_any(typ):
   359	                    all_use_types.append(typ)
   360	            # Add in any default argument types
   361	            default = defaults[i]
   362	            if default:
   363	                all_arg_types.append(default)
   364	                if all_use_types:
   365	                    all_use_types.append(default)
   366
   367	            arg_types = []
   368
   369	            if all_arg_types and all(
   370	                isinstance(get_proper_type(tp), NoneType) for tp in all_arg_types
   371	            ):
   372	                arg_types.append(
   373	                    UnionType.make_union([all_arg_types[0], AnyType(TypeOfAny.explicit)])
   374	                )
   375	            elif all_arg_types:
   376	                arg_types.extend(generate_type_combinations(all_arg_types))
   377	            else:
   378	                arg_types.append(AnyType(TypeOfAny.explicit))
   379
   380	            if all_use_types:
   381	                # This is a meet because the type needs to be compatible with all the uses
   382	                arg_types.append(meet_type_list(all_use_types))
   383
   384	            types.append(arg_types)
   385	        return types

   393	    def get_guesses(
   394	        self,
   395	        is_method: bool,
   396	        base: CallableType,
   397	        defaults: list[Type | None],
   398	        callsites: list[Callsite],
   399	        uses: list[list[Type]],
   400	    ) -> list[CallableType]:
   401	        """Compute a list of guesses for a function's type.
   402
   403	        This focuses just on the argument types, and doesn't change the provided return type.
   404	        """
   405	        options = self.get_args(is_method, base, defaults, callsites, uses)
   406
   407	        # Take the first `max_guesses` guesses.
   408	        product = itertools.islice(itertools.product(*options), 0, self.max_guesses)
   409	        return [refine_callable(base, base.copy_modified(arg_types=list(x))) for x in product]

   411	    def get_callsites(self, func: FuncDef) -> tuple[list[Callsite], list[str]]:
   412	        """Find all call sites of a function."""
   413	        new_type = self.get_starting_type(func)
   414
   415	        collector_plugin = SuggestionPlugin(func.fullname)
   416
   417	        self.plugin._plugins.insert(0, collector_plugin)
   418	        try:
   419	            errors = self.try_type(func, new_type)
   420	        finally:
   421	            self.plugin._plugins.pop(0)
   422
   423	        return collector_plugin.mystery_hits, errors

   425	    def filter_options(
   426	        self, guesses: list[CallableType], is_method: bool, ignore_return: bool
   427	    ) -> list[CallableType]:
   428	        """Apply any configured filters to the possible guesses.
   429
   430	        Currently the only option is filtering based on Any prevalance."""
   431	        return [
   432	            t
   433	            for t in guesses
   434	            if self.flex_any is None
   435	            or any_score_callable(t, is_method, ignore_return) >= self.flex_any
   436	        ]

   438	    def find_best(self, func: FuncDef, guesses: list[CallableType]) -> tuple[CallableType, int]:
   439	        """From a list of possible function types, find the best one.
   440
   441	        For best, we want the fewest errors, then the best "score" from score_callable.
   442	        """
   443	        if not guesses:
   444	            raise SuggestionFailure("No guesses that match criteria!")
   445	        errors = {guess: self.try_type(func, guess) for guess in guesses}
   446	        best = min(guesses, key=lambda s: (count_errors(errors[s]), self.score_callable(s)))
   447	        return best, count_errors(errors[best])

   449	    def get_guesses_from_parent(self, node: FuncDef) -> list[CallableType]:
   450	        """Try to get a guess of a method type from a parent class."""
   451	        if not node.info:
   452	            return []
   453
   454	        for parent in node.info.mro[1:]:
   455	            pnode = parent.names.get(node.name)
   456	            if pnode and isinstance(pnode.node, (FuncDef, Decorator)):
   457	                typ = get_proper_type(pnode.node.type)
   458	                # FIXME: Doesn't work right with generic types
   459	                if isinstance(typ, CallableType) and len(typ.arg_types) == len(node.arguments):
   460	                    # Return the first thing we find, since it probably doesn't make sense
   461	                    # to grab things further up in the chain if an earlier parent has it.
   462	                    return [typ]
   463
   464	        return []

   466	    def get_suggestion(self, mod: str, node: FuncDef) -> PyAnnotateSignature:
   467	        """Compute a suggestion for a function.
   468
   469	        Return the type and whether the first argument should be ignored.
   470	        """
   471	        graph = self.graph
   472	        callsites, orig_errors = self.get_callsites(node)
   473	        uses = get_arg_uses(self.manager.all_types, node)
   474
   475	        if self.no_errors and orig_errors:
   476	            raise SuggestionFailure("Function does not typecheck.")
   477
   478	        is_method = bool(node.info) and not node.is_static
   479
   480	        with state.strict_optional_set(graph[mod].options.strict_optional):
   481	            guesses = self.get_guesses(
   482	                is_method,
   483	                self.get_starting_type(node),
   484	                self.get_default_arg_types(node),
   485	                callsites,
   486	                uses,
   487	            )
   488	        guesses += self.get_guesses_from_parent(node)
   489	        guesses = self.filter_options(guesses, is_method, ignore_return=True)
   490	        best, _ = self.find_best(node, guesses)
   491
   492	        # Now try to find the return type!
   493	        self.try_type(node, best)
   494	        returns = get_return_types(self.manager.all_types, node)
   495	        with state.strict_optional_set(graph[mod].options.strict_optional):
   496	            if returns:
   497	                ret_types = generate_type_combinations(returns)
   498	            else:
   499	                ret_types = [NoneType()]
   500
   501	        guesses = [best.copy_modified(ret_type=refine_type(best.ret_type, t)) for t in ret_types]
   502	        guesses = self.filter_options(guesses, is_method, ignore_return=False)
   503	        best, errors = self.find_best(node, guesses)
   504
   505	        if self.no_errors and errors:
   506	            raise SuggestionFailure("No annotation without errors")
   507
   508	        return self.pyannotate_signature(mod, is_method, best)

   510	    def format_args(
   511	        self,
   512	        arg_kinds: list[list[ArgKind]],
   513	        arg_names: list[list[str | None]],
   514	        arg_types: list[list[Type]],
   515	    ) -> str:
   516	        args: list[str] = []
   517	        for i in range(len(arg_types)):
   518	            for kind, name, typ in zip(arg_kinds[i], arg_names[i], arg_types[i]):
   519	                arg = self.format_type(None, typ)
   520	                if kind == ARG_STAR:
   521	                    arg = "*" + arg
   522	                elif kind == ARG_STAR2:
   523	                    arg = "**" + arg
   524	                elif kind.is_named():
   525	                    if name:
   526	                        arg = f"{name}={arg}"
   527	            args.append(arg)
   528	        return f"({', '.join(args)})"

   530	    def find_node(self, key: str) -> tuple[str, str, FuncDef]:
   531	        """From a target name, return module/target names and the func def.
   532
   533	        The 'key' argument can be in one of two formats:
   534	        * As the function full name, e.g., package.module.Cls.method
   535	        * As the function location as file and line separated by column,
   536	          e.g., path/to/file.py:42
   537	        """
   538	        # TODO: Also return OverloadedFuncDef -- currently these are ignored.
   539	        node: SymbolNode | None = None
   540	        if ":" in key:
   541	            # A colon might be part of a drive name on Windows (like `C:/foo/bar`)
   542	            # and is also used as a delimiter between file path and lineno.
   543	            # If a colon is there for any of those reasons, it must be a file+line
   544	            # reference.
   545	            platform_key_count = 2 if sys.platform == "win32" else 1
   546	            if key.count(":") > platform_key_count:
   547	                raise SuggestionFailure(
   548	                    "Malformed location for function: {}. Must be either"
   549	                    " package.module.Class.method or path/to/file.py:line".format(key)
   550	                )
   551	            file, line = key.rsplit(":", 1)
   552	            if not line.isdigit():
   553	                raise SuggestionFailure(f"Line number must be a number. Got {line}")
   554	            line_number = int(line)
   555	            modname, node = self.find_node_by_file_and_line(file, line_number)
   556	            tail = node.fullname[len(modname) + 1 :]  # add one to account for '.'
   557	        else:
   558	            target = split_target(self.fgmanager.graph, key)
   559	            if not target:
   560	                raise SuggestionFailure(f"Cannot find module for {key}")
   561	            modname, tail = target
   562	            node = self.find_node_by_module_and_name(modname, tail)
   563
   564	        if isinstance(node, Decorator):
   565	            node = self.extract_from_decorator(node)
   566	            if not node:
   567	                raise SuggestionFailure(f"Object {key} is a decorator we can't handle")
   568
   569	        if not isinstance(node, FuncDef):
   570	            raise SuggestionFailure(f"Object {key} is not a function")
   571
   572	        return modname, tail, node

   574	    def find_node_by_module_and_name(self, modname: str, tail: str) -> SymbolNode | None:
   575	        """Find symbol node by module id and qualified name.
   576
   577	        Raise SuggestionFailure if can't find one.
   578	        """
   579	        tree = self.ensure_loaded(self.fgmanager.graph[modname])

   751	    def score_type(self, t: Type, arg_pos: bool) -> int:
   752	        """Generate a score for a type that we use to pick which type to use.
   753
   754	        Lower is better, prefer non-union/non-any types. Don't penalize optionals.
   755	        """
   756	        t = get_proper_type(t)
   757	        if isinstance(t, AnyType):
   758	            return 20
   759	        if arg_pos and isinstance(t, NoneType):
   760	            return 20

   762	            if any(isinstance(get_proper_type(x), AnyType) for x in t.items):
   763	                return 20

   765	                return 15
   766	            if not is_overlapping_none(t):
   767	                return 10

   778	def any_score_type(ut: Type, arg_pos: bool) -> float:
   779	    """Generate a very made up number representing the Anyness of a type.
   780
   781	    Higher is better, 1.0 is max
   782	    """
   783	    t = get_proper_type(ut)
   784	    if isinstance(t, AnyType) and t.type_of_any != TypeOfAny.suggestion_engine:
   785	        return 0
   786	    if isinstance(t, NoneType) and arg_pos:
   787	        return 0.5
   788	    if isinstance(t, UnionType):
   789	        if any(isinstance(get_proper_type(x), AnyType) for x in t.items):
   790	            return 0.5
   791	        if any(has_any_type(x) for x in t.items):
   792	            return 0.25
   793	    if isinstance(t, CallableType) and is_tricky_callable(t):
   794	        return 0.5
   795	    if has_any_type(t):
   796	        return 0.5
   797
   798	    return 1.0

   801	def any_score_callable(t: CallableType, is_method: bool, ignore_return: bool) -> float:
   802	    # Ignore the first argument of methods
   803	    scores = [any_score_type(x, arg_pos=True) for x in t.arg_types[int(is_method) :]]
   804	    # Return type counts twice (since it spreads type information), unless it is
   805	    # None in which case it does not count at all. (Though it *does* still count
   806	    # if there are no arguments.)
   807	    if not isinstance(get_proper_type(t.ret_type), NoneType) or not scores:
   808	        ret = 1.0 if ignore_return else any_score_type(t.ret_type, arg_pos=False)
   809	        scores += [ret, ret]
   810
   811	    return sum(scores) / len(scores)

   941	def refine_type(ti: Type, si: Type) -> Type:
   942	    """Refine `ti` by replacing Anys in it with information taken from `si`
   943
   944	    This basically works by, when the types have the same structure,
   945	    traversing both of them in parallel and replacing Any on the left
   946	    with whatever the type on the right is. If the types don't have the
   947	    same structure (or aren't supported), the left type is chosen.
   948
   949	    For example:
   950	      refine(Any, T) = T,  for all T
   951	      refine(float, int) = float
   952	      refine(List[Any], List[int]) = List[int]
   953	      refine(Dict[int, Any], Dict[Any, int]) = Dict[int, int]
   954	      refine(Tuple[int, Any], Tuple[Any, int]) = Tuple[int, int]
   955
   956	      refine(Callable[[Any], Any], Callable[[int], int]) = Callable[[int], int]
   957	      refine(Callable[..., int], Callable[[int, float], Any]) = Callable[[int, float], int]
   958
   959	      refine(Optional[Any], int) = Optional[int]
   960	      refine(Optional[Any], Optional[int]) = Optional[int]
   961	      refine(Optional[Any], Union[int, str]) = Optional[Union[int, str]]
   962	      refine(Optional[List[Any]], List[int]) = List[int]
   963
   964	    """
   965	    t = get_proper_type(ti)
   966	    s = get_proper_type(si)
   967
   968	    if isinstance(t, AnyType):
   969	        # If s is also an Any, we return if it is a missing_import Any
   970	        return t if isinstance(s, AnyType) and t.missing_import_name else s
   971
   972	    if isinstance(t, Instance) and isinstance(s, Instance) and t.type == s.type:
   973	        return t.copy_modified(args=[refine_type(ta, sa) for ta, sa in zip(t.args, s.args)])
   974
   975	    if (
   976	        isinstance(t, TupleType)
   977	        and isinstance(s, TupleType)
   978	        and t.partial_fallback == s.partial_fallback
   979	        and len(t.items) == len(s.items)
   980	    ):
   981	        return t.copy_modified(items=[refine_type(ta, sa) for ta, sa in zip(t.items, s.items)])
   982
   983	    if isinstance(t, CallableType) and isinstance(s, CallableType):
   984	        return refine_callable(t, s)
   985
   986	    if isinstance(t, UnionType):
   987	        return refine_union(t, s)
   988
   989	    # TODO: Refining of builtins.tuple, Type?
   990
   991	    return t

   994	def refine_union(t: UnionType, s: ProperType) -> Type:
   995	    """Refine a union type based on another type.
   996
   997	    This is done by refining every component of the union against the
   998	    right hand side type (or every component of its union if it is
   999	    one). If an element of the union is successfully refined, we drop it
  1000	    from the union in favor of the refined versions.
  1001	    """
  1002	    # Don't try to do any union refining if the types are already the
  1003	    # same.  This prevents things like refining Optional[Any] against
  1004	    # itself and producing None.
  1005	    if t == s:
  1006	        return t
  1007
  1008	    rhs_items = s.items if isinstance(s, UnionType) else [s]
  1009
  1010	    new_items = []
  1011	    for lhs in t.items:
  1012	        refined = False
  1013	        for rhs in rhs_items:
  1014	            new = refine_type(lhs, rhs)
  1015	            if new != lhs:
  1016	                new_items.append(new)
  1017	                refined = True
  1018	        if not refined:
  1019	            new_items.append(lhs)
  1020
  1021	    # Turn strict optional on when simplifying the union since we
  1022	    # don't want to drop Nones.
  1023	    with state.strict_optional_set(True):
  1024	        return make_simplified_union(new_items)

  1027	def refine_callable(t: CallableType, s: CallableType) -> CallableType:
  1028	    """Refine a callable based on another.
  1029
  1030	    See comments for refine_type.
  1031	    """
  1032	    if t.fallback != s.fallback:
  1033	        return t
  1034
  1035	    if t.is_ellipsis_args and not is_tricky_callable(s):
  1036	        return s.copy_modified(ret_type=refine_type(t.ret_type, s.ret_type))

  1039	        return t
  1040
  1041	    return t.copy_modified(
  1042	        arg_types=[refine_type(ta, sa) for ta, sa in zip(t.arg_types, s.arg_types)],
  1043	        ret_type=refine_type(t.ret_type, s.ret_type),

    49	def _file_arg_to_module(filename: str) -> str:
    50	    filename, _ = os.path.splitext(filename)

    57	def parse_test_case(case: DataDrivenTestCase) -> None:
    58	    """Parse and prepare a single case from suite with test case descriptions.
    59
    60	    This method is part of the setup phase, just before the test case is run.
    61	    """
    62	    test_items = parse_test_data(case.data, case.name)
    63	    base_path = case.suite.base_path
    64	    if case.suite.native_sep:
    65	        join = os.path.join
    66	    else:
    67	        join = posixpath.join
    68
    69	    out_section_missing = case.suite.required_out_section
    70
    71	    files: list[tuple[str, str]] = []  # path and contents
    72	    output_files: list[tuple[str, str | Pattern[str]]] = []  # output path and contents
    73	    output: list[str] = []  # Regular output errors
    74	    output2: dict[int, list[str]] = {}  # Output errors for incremental, runs 2+
    75	    deleted_paths: dict[int, set[str]] = {}  # from run number of paths
    76	    stale_modules: dict[int, set[str]] = {}  # from run number to module names
    77	    rechecked_modules: dict[int, set[str]] = {}  # from run number module names
    78	    triggered: list[str] = []  # Active triggers (one line per incremental step)
    79	    targets: dict[int, list[str]] = {}  # Fine-grained targets (per fine-grained update)
    80	    test_modules: list[str] = []  # Modules which are deemed "test" (vs "fixture")
    81
    82	    def _case_fail(msg: str) -> NoReturn:
    83	        pytest.fail(f"{case.file}:{case.line}: {msg}", pytrace=False)
    84
    85	    # Process the parsed items. Each item has a header of form [id args],
    86	    # optionally followed by lines of text.
    87	    item = first_item = test_items[0]
    88	    test_modules.append("__main__")
    89	    for item in test_items[1:]:
    90
    91	        def _item_fail(msg: str) -> NoReturn:
    92	            item_abs_line = case.line + item.line - 2
    93	            pytest.fail(f"{case.file}:{item_abs_line}: {msg}", pytrace=False)
    94
    95	        if item.id in {"file", "fixture", "outfile", "outfile-re"}:
    96	            # Record an extra file needed for the test case.
    97	            assert item.arg is not None
    98	            contents = expand_variables("\n".join(item.data))
    99	            path = join(base_path, item.arg)
   100	            if item.id != "fixture":
   101	                test_modules.append(_file_arg_to_module(item.arg))
   102	            if item.id in {"file", "fixture"}:
   103	                files.append((path, contents))
   104	            elif item.id == "outfile-re":
   105	                output_files.append((path, re.compile(contents.rstrip(), re.S)))
   106	            elif item.id == "outfile":
   107	                output_files.append((path, contents))
   108	        elif item.id == "builtins":
   109	            # Use an alternative stub file for the builtins module.
   110	            assert item.arg is not None
   111	            mpath = join(os.path.dirname(case.file), item.arg)
   112	            with open(mpath, encoding="utf8") as f:
   113	                files.append((join(base_path, "builtins.pyi"), f.read()))
   114	        elif item.id == "typing":
   115	            # Use an alternative stub file for the typing module.
   116	            assert item.arg is not None
   117	            src_path = join(os.path.dirname(case.file), item.arg)
   118	            with open(src_path, encoding="utf8") as f:
   119	                files.append((join(base_path, "typing.pyi"), f.read()))
   120	        elif item.id == "_typeshed":
   121	            # Use an alternative stub file for the _typeshed module.
   122	            assert item.arg is not None
   123	            src_path = join(os.path.dirname(case.file), item.arg)
   124	            with open(src_path, encoding="utf8") as f:
   125	                files.append((join(base_path, "_typeshed.pyi"), f.read()))
   126	        elif re.match(r"stale[0-9]*$", item.id):
   127	            passnum = 1 if item.id == "stale" else int(item.id[len("stale") :])
   128	            assert passnum > 0
   129	            modules = set() if item.arg is None else {t.strip() for t in item.arg.split(",")}
   130	            stale_modules[passnum] = modules
   131	        elif re.match(r"rechecked[0-9]*$", item.id):
   132	            passnum = 1 if item.id == "rechecked" else int(item.id[len("rechecked") :])
   133	            assert passnum > 0
   134	            modules = set() if item.arg is None else {t.strip() for t in item.arg.split(",")}
   135	            rechecked_modules[passnum] = modules
   136	        elif re.match(r"targets[0-9]*$", item.id):
   137	            passnum = 1 if item.id == "targets" else int(item.id[len("targets") :])
   138	            assert passnum > 0
   139	            reprocessed = [] if item.arg is None else [t.strip() for t in item.arg.split(",")]
   140	            targets[passnum] = reprocessed
   141	        elif item.id == "delete":
   142	            # File/directory to delete during a multi-step test case
   143	            assert item.arg is not None
   144	            m = re.match(r"(.*)\.([0-9]+)$", item.arg)
   145	            if m is None:
   146	                _item_fail(f"Invalid delete section {item.arg!r}")
   147	            num = int(m.group(2))
   148	            if num < 2:
   149	                _item_fail(f"Can't delete during step {num}")
   150	            full = join(base_path, m.group(1))
   151	            deleted_paths.setdefault(num, set()).add(full)
   152	        elif re.match(r"out[0-9]*$", item.id):
   153	            if item.arg is None:
   154	                args = []
   155	            else:
   156	                args = item.arg.split(",")
   157
   158	            version_check = True
   159	            for arg in args:
   160	                if arg.startswith("version"):
   161	                    compare_op = arg[7:9]
   162	                    if compare_op not in {">=", "=="}:
   163	                        _item_fail("Only >= and == version checks are currently supported")
   164	                    version_str = arg[9:]
   165	                    try:
   166	                        version = tuple(int(x) for x in version_str.split("."))
   167	                    except ValueError:
   168	                        _item_fail(f"{version_str!r} is not a valid python version")
   169	                    if compare_op == ">=":
   170	                        if version <= defaults.PYTHON3_VERSION:
   171	                            _item_fail(
   172	                                f"{arg} always true since minimum runtime version is {defaults.PYTHON3_VERSION}"
   173	                            )
   174	                        version_check = sys.version_info >= version
   175	                    elif compare_op == "==":
   176	                        if version < defaults.PYTHON3_VERSION:
   177	                            _item_fail(
   178	                                f"{arg} always false since minimum runtime version is {defaults.PYTHON3_VERSION}"
   179	                            )
   180	                        if not 1 < len(version) < 4:
   181	                            _item_fail(
   182	                                f'Only minor or patch version checks are currently supported with "==": {version_str!r}'
   183	                            )
   184	                        version_check = sys.version_info[: len(version)] == version
   185	            if version_check:
   186	                tmp_output = [expand_variables(line) for line in item.data]
   187	                if os.path.sep == "\\" and case.normalize_output:
   188	                    tmp_output = [fix_win_path(line) for line in tmp_output]
   189	                if item.id == "out" or item.id == "out1":
   190	                    output = tmp_output
   191	                else:
   192	                    passnum = int(item.id[len("out") :])
   193	                    assert passnum > 1
   194	                    output2[passnum] = tmp_output
   195	                out_section_missing = False
   196	        elif item.id == "triggered" and item.arg is None:
   197	            triggered = item.data
   198	        else:
   199	            section_str = item.id + (f" {item.arg}" if item.arg else "")
   200	            _item_fail(f"Invalid section header [{section_str}] in case {case.name!r}")
   201
   202	    if out_section_missing:
   203	        _case_fail(f"Required output section not found in case {case.name!r}")
   204
   205	    for passnum in stale_modules.keys():
   206	        if passnum not in rechecked_modules:
   207	            # If the set of rechecked modules isn't specified, make it the same as the set
   208	            # of modules with a stale public interface.
   209	            rechecked_modules[passnum] = stale_modules[passnum]
   210	        if (
   211	            passnum in stale_modules
   212	            and passnum in rechecked_modules
   213	            and not stale_modules[passnum].issubset(rechecked_modules[passnum])
   214	        ):
   215	            _case_fail(f"Stale modules after pass {passnum} must be a subset of rechecked modules")
   216
   217	    output_inline_start = len(output)
   218	    input = first_item.data
   219	    expand_errors(input, output, "main")
   220	    for file_path, contents in files:
   221	        expand_errors(contents.split("\n"), output, file_path)
   222
   223	    seen_files = set()
   224	    for file, _ in files:
   225	        if file in seen_files:
   226	            _case_fail(f"Duplicated filename {file}. Did you include it multiple times?")
   227
   228	        seen_files.add(file)
   229
   230	    case.input = input
   231	    case.output = output
   232	    case.output_inline_start = output_inline_start
   233	    case.output2 = output2
   234	    case.last_line = case.line + item.line + len(item.data) - 2
   235	    case.files = files
   236	    case.output_files = output_files
   237	    case.expected_stale_modules = stale_modules
   238	    case.expected_rechecked_modules = rechecked_modules
   239	    case.deleted_paths = deleted_paths
   240	    case.triggered = triggered or []
   241	    case.expected_fine_grained_targets = targets
   242	    case.test_modules = test_modules

   245	class DataDrivenTestCase(pytest.Item):
   246	    """Holds parsed data-driven test cases, and handles directory setup and teardown."""
   247
   248	    # Override parent member type
   249	    parent: DataFileCollector
   250
   251	    input: list[str]
   252	    output: list[str]  # Output for the first pass
   253	    output_inline_start: int
   254	    output2: dict[int, list[str]]  # Output for runs 2+, indexed by run number
   255
   256	    # full path of test suite
   257	    file = ""
   258	    line = 0
   259
   260	    # (file path, file content) tuples
   261	    files: list[tuple[str, str]]
   262	    # Modules which is to be considered "test" rather than "fixture"
   263	    test_modules: list[str]
   264	    expected_stale_modules: dict[int, set[str]]
   265	    expected_rechecked_modules: dict[int, set[str]]
   266	    expected_fine_grained_targets: dict[int, list[str]]
   267
   268	    # Whether or not we should normalize the output to standardize things like
   269	    # forward vs backward slashes in file paths for Windows vs Linux.
   270	    normalize_output: bool
   271
   272	    # Extra attributes used by some tests.
   273	    last_line: int
   274	    output_files: list[tuple[str, str | Pattern[str]]]  # Path and contents for output files
   275	    deleted_paths: dict[int, set[str]]  # Mapping run number -> paths
   276	    triggered: list[str]  # Active triggers (one line per incremental step)
   277
   278	    def __init__(
   279	        self,
   280	        parent: DataFileCollector,
   281	        suite: DataSuite,
   282	        *,
   283	        file: str,
   284	        name: str,
   285	        writescache: bool,
   286	        only_when: str,
   287	        normalize_output: bool,
   288	        platform: str | None,
   289	        skip: bool,
   290	        xfail: bool,
   291	        data: str,
   292	        line: int,
   293	    ) -> None:
   294	        assert isinstance(parent, DataFileCollector)
   295	        super().__init__(name, parent)
   296	        self.suite = suite
   297	        self.file = file
   298	        self.writescache = writescache
   299	        self.only_when = only_when
   300	        self.normalize_output = normalize_output
   301	        if (platform == "windows" and sys.platform != "win32") or (
   302	            platform == "posix" and sys.platform == "win32"
   303	        ):
   304	            skip = True
   305	        self.skip = skip
   306	        self.xfail = xfail
   307	        self.data = data
   308	        self.line = line
   309	        self.old_cwd: str | None = None
   310	        self.tmpdir: str | None = None

   312	    def runtest(self) -> None:
   313	        if self.skip:
   314	            pytest.skip()
   315	        # TODO: add a better error message for when someone uses skip and xfail at the same time
   316	        elif self.xfail:
   317	            self.add_marker(pytest.mark.xfail)
   318	        parent = self.getparent(DataSuiteCollector)
   319	        assert parent is not None, "Should not happen"
   320	        suite = parent.obj()
   321	        suite.setup()
   322	        try:
   323	            suite.run_case(self)
   324	        except Exception:
   325	            # As a debugging aid, support copying the contents of the tmp directory somewhere
   326	            save_dir: str | None = self.config.getoption("--save-failures-to", None)
   327	            if save_dir:
   328	                assert self.tmpdir is not None
   329	                target_dir = os.path.join(save_dir, os.path.basename(self.tmpdir))
   330	                print(f"Copying data from test {self.name} to {target_dir}")
   331	                if not os.path.isabs(target_dir):
   332	                    assert self.old_cwd
   333	                    target_dir = os.path.join(self.old_cwd, target_dir)
   334	                shutil.copytree(self.tmpdir, target_dir)
   335	            raise

   337	    def setup(self) -> None:
   338	        parse_test_case(case=self)
   339	        self.old_cwd = os.getcwd()
   340	        self.tmpdir = tempfile.mkdtemp(prefix="mypy-test-")
   341	        os.chdir(self.tmpdir)
   342	        os.mkdir(test_temp_dir)
   343
   344	        # Precalculate steps for find_steps()
   345	        steps: dict[int, list[FileOperation]] = {}
   346
   347	        for path, content in self.files:
   348	            m = re.match(r".*\.([0-9]+)$", path)
   349	            if m:
   350	                # Skip writing subsequent incremental steps - rather
   351	                # store them as operations.
   352	                num = int(m.group(1))
   353	                assert num >= 2
   354	                target_path = re.sub(r"\.[0-9]+$", "", path)
   355	                module = module_from_path(target_path)
   356	                operation = UpdateFile(module, content, target_path)
   357	                steps.setdefault(num, []).append(operation)
   358	            else:
   359	                # Write the first incremental steps
   360	                dir = os.path.dirname(path)
   361	                os.makedirs(dir, exist_ok=True)
   362	                with open(path, "w", encoding="utf8") as f:
   363	                    f.write(content)
   364
   365	        for num, paths in self.deleted_paths.items():
   366	            assert num >= 2
   367	            for path in paths:
   368	                module = module_from_path(path)
   369	                steps.setdefault(num, []).append(DeleteFile(module, path))
   370	        max_step = max(steps) if steps else 2
   371	        self.steps = [steps.get(num, []) for num in range(2, max_step + 1)]
   372
   373	    def teardown(self) -> None:
   374	        if self.old_cwd is not None:
   375	            os.chdir(self.old_cwd)
   376	        if self.tmpdir is not None:
   377	            shutil.rmtree(self.tmpdir, ignore_errors=True)
   378	        self.old_cwd = None
   379	        self.tmpdir = None

   381	    def reportinfo(self) -> tuple[str, int, str]:
   382	        return self.file, self.line, self.name

   384	    def repr_failure(
   385	        self, excinfo: pytest.ExceptionInfo[BaseException], style: Any | None = None
   386	    ) -> str:
   387	        excrepr: object
   388	        if isinstance(excinfo.value, SystemExit):
   389	            # We assume that before doing exit() (which raises SystemExit) we've printed
   390	            # enough context about what happened so that a stack trace is not useful.
   391	            # In particular, uncaught exceptions during semantic analysis or type checking
   392	            # call exit() and they already print out a stack trace.
   393	            excrepr = excinfo.exconly()
   394	        elif isinstance(excinfo.value, pytest.fail.Exception) and not excinfo.value.pytrace:
   395	            excrepr = excinfo.exconly()
   396	        else:
   397	            excinfo.traceback = self.parent._traceback_filter(excinfo)
   398	            excrepr = excinfo.getrepr(style="short")
   399
   400	        return f"data: {self.file}:{self.line}:\n{excrepr}"

   402	    def find_steps(self) -> list[list[FileOperation]]:
   403	        """Return a list of descriptions of file operations for each incremental step.
   404
   405	        The first list item corresponds to the first incremental step, the second for the
   406	        second step, etc. Each operation can either be a file modification/creation (UpdateFile)
   407	        or deletion (DeleteFile).

   409	        Defaults to having two steps if there aern't any operations.

   414	def module_from_path(path: str) -> str:
   415	    path = re.sub(r"\.pyi?$", "", path)
   416	    # We can have a mix of Unix-style and Windows-style separators.
   417	    parts = re.split(r"[/\\]", path)
   418	    del parts[0]
   419	    module = ".".join(parts)
   420	    module = re.sub(r"\.__init__$", "", module)
   421	    return module

   424	@dataclass
   425	class TestItem:
   426	    """Parsed test caseitem.
   427
   428	    An item is of the form
   429	      [id arg]
   430	      .. data ..
   431	    """
   432
   433	    id: str
   434	    arg: str | None
   435	    # Processed, collapsed text data
   436	    data: list[str]
   437	    # Start line: 1-based, inclusive, relative to testcase
   438	    line: int
   439	    # End line: 1-based, exclusive, relative to testcase; not same as `line + len(test_item.data)` due to collapsing
   440	    end_line: int
   441
   442	    @property
   443	    def trimmed_newlines(self) -> int:  # compensates for strip_list
   444	        return self.end_line - self.line - len(self.data)

   447	def parse_test_data(raw_data: str, name: str) -> list[TestItem]:
   448	    """Parse a list of lines that represent a sequence of test items."""
   449
   450	    lines = ["", "[case " + name + "]"] + raw_data.split("\n")
   451	    ret: list[TestItem] = []
   452	    data: list[str] = []
   453
   454	    id: str | None = None
   455	    arg: str | None = None
   456
   457	    i = 0
   458	    i0 = 0
   459	    while i < len(lines):
   460	        s = lines[i].strip()
   461
   462	        if lines[i].startswith("[") and s.endswith("]"):
   463	            if id:
   464	                data = collapse_line_continuation(data)
   465	                data = strip_list(data)
   466	                ret.append(TestItem(id, arg, data, i0 + 1, i))
   467
   468	            i0 = i
   469	            id = s[1:-1]
   470	            arg = None
   471	            if " " in id:
   472	                arg = id[id.index(" ") + 1 :]
   473	                id = id[: id.index(" ")]
   474	            data = []
   475	        elif lines[i].startswith("\\["):
   476	            data.append(lines[i][1:])
   477	        elif not lines[i].startswith("--"):
   478	            data.append(lines[i])
   479	        elif lines[i].startswith("----"):
   480	            data.append(lines[i][2:])
   481	        i += 1
   482
   483	    # Process the last item.
   484	    if id:
   485	        data = collapse_line_continuation(data)
   486	        data = strip_list(data)
   487	        ret.append(TestItem(id, arg, data, i0 + 1, i - 1))
   488
   489	    return ret

   492	def strip_list(l: list[str]) -> list[str]:
   493	    """Return a stripped copy of l.
   494
   495	    Strip whitespace at the end of all lines, and strip all empty
   496	    lines from the end of the array.
   497	    """
   498
   499	    r: list[str] = []
   500	    for s in l:
   501	        # Strip spaces at end of line
   502	        r.append(re.sub(r"\s+$", "", s))
   503
   504	    while r and r[-1] == "":
   505	        r.pop()
   506
   507	    return r

   510	def collapse_line_continuation(l: list[str]) -> list[str]:
   511	    r: list[str] = []
   512	    cont = False
   513	    for s in l:
   514	        ss = re.sub(r"\\$", "", s)
   515	        if cont:
   516	            r[-1] += re.sub("^ +", "", ss)
   517	        else:
   518	            r.append(ss)
   519	        cont = s.endswith("\\")
   520	    return r

   527	def expand_errors(input: list[str], output: list[str], fnam: str) -> None:
   528	    """Transform comments such as '# E: message' or
   529	    '# E:3: message' in input.
   530
   531	    The result is lines like 'fnam:line: error: message'.
   532	    """
   533
   534	    for i in range(len(input)):
   535	        # The first in the split things isn't a comment
   536	        for possible_err_comment in input[i].split(" # ")[1:]:
   537	            m = re.search(
   538	                r"^([ENW]):((?P<col>\d+):)? (?P<message>.*)$", possible_err_comment.strip()
   539	            )
   540	            if m:
   541	                if m.group(1) == "E":
   542	                    severity = "error"
   543	                elif m.group(1) == "N":
   544	                    severity = "note"
   545	                elif m.group(1) == "W":
   546	                    severity = "warning"
   547	                col = m.group("col")
   548	                message = m.group("message")
   549	                message = message.replace("\\#", "#")  # adds back escaped # character
   550	                if col is None:
   551	                    output.append(f"{fnam}:{i + 1}: {severity}: {message}")
   552	                else:
   553	                    output.append(f"{fnam}:{i + 1}:{col}: {severity}: {message}")

   556	def fix_win_path(line: str) -> str:
   557	    r"""Changes Windows paths to Linux paths in error messages.
   558
   559	    E.g. foo\bar.py -> foo/bar.py.
   560	    """
   561	    line = line.replace(root_dir, root_dir.replace("\\", "/"))
   562	    m = re.match(r"^([\S/]+):(\d+:)?(\s+.*)", line)
   563	    if not m:
   564	        return line
   565	    else:
   566	        filename, lineno, message = m.groups()
   567	        return "{}:{}{}".format(filename.replace("\\", "/"), lineno or "", message)

   570	def fix_cobertura_filename(line: str) -> str:
   571	    r"""Changes filename paths to Linux paths in Cobertura output files.
   572
   573	    E.g. filename="pkg\subpkg\a.py" -> filename="pkg/subpkg/a.py".
   574	    """
   575	    m = re.search(r'<class .* filename="(?P<filename>.*?)"', line)
   576	    if not m:
   577	        return line
   578	    return "{}{}{}".format(
   579	        line[: m.start(1)], m.group("filename").replace("\\", "/"), line[m.end(1) :]
   580	    )
