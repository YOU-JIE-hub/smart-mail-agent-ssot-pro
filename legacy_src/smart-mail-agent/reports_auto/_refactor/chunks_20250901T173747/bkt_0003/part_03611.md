   594	def _init_checker_class() -> type[doctest.OutputChecker]:
   595	    import doctest
   596
   597	    class LiteralsOutputChecker(doctest.OutputChecker):
   598	        # Based on doctest_nose_plugin.py from the nltk project
   599	        # (https://github.com/nltk/nltk) and on the "numtest" doctest extension
   600	        # by Sebastien Boisgerault (https://github.com/boisgera/numtest).
   601
   602	        _unicode_literal_re = re.compile(r"(\W|^)[uU]([rR]?[\'\"])", re.UNICODE)
   603	        _bytes_literal_re = re.compile(r"(\W|^)[bB]([rR]?[\'\"])", re.UNICODE)
   604	        _number_re = re.compile(
   605	            r"""
   606	            (?P<number>
   607	              (?P<mantissa>
   608	                (?P<integer1> [+-]?\d*)\.(?P<fraction>\d+)
   609	                |
   610	                (?P<integer2> [+-]?\d+)\.
   611	              )
   612	              (?:
   613	                [Ee]
   614	                (?P<exponent1> [+-]?\d+)
   615	              )?
   616	              |
   617	              (?P<integer3> [+-]?\d+)
   618	              (?:
   619	                [Ee]
   620	                (?P<exponent2> [+-]?\d+)
   621	              )
   622	            )
   623	            """,
   624	            re.VERBOSE,
   625	        )
   626
   627	        def check_output(self, want: str, got: str, optionflags: int) -> bool:
   628	            if super().check_output(want, got, optionflags):
   629	                return True
   630
   631	            allow_unicode = optionflags & _get_allow_unicode_flag()
   632	            allow_bytes = optionflags & _get_allow_bytes_flag()
   633	            allow_number = optionflags & _get_number_flag()
   634
   635	            if not allow_unicode and not allow_bytes and not allow_number:
   636	                return False
   637
   638	            def remove_prefixes(regex: re.Pattern[str], txt: str) -> str:
   639	                return re.sub(regex, r"\1\2", txt)
   640
   641	            if allow_unicode:
   642	                want = remove_prefixes(self._unicode_literal_re, want)
   643	                got = remove_prefixes(self._unicode_literal_re, got)
   644
   645	            if allow_bytes:
   646	                want = remove_prefixes(self._bytes_literal_re, want)
   647	                got = remove_prefixes(self._bytes_literal_re, got)
   648
   649	            if allow_number:
   650	                got = self._remove_unwanted_precision(want, got)
   651
   652	            return super().check_output(want, got, optionflags)
   653
   654	        def _remove_unwanted_precision(self, want: str, got: str) -> str:
   655	            wants = list(self._number_re.finditer(want))
   656	            gots = list(self._number_re.finditer(got))
   657	            if len(wants) != len(gots):
   658	                return got
   659	            offset = 0
   660	            for w, g in zip(wants, gots):
   661	                fraction: str | None = w.group("fraction")
   662	                exponent: str | None = w.group("exponent1")
   663	                if exponent is None:
   664	                    exponent = w.group("exponent2")
   665	                precision = 0 if fraction is None else len(fraction)
   666	                if exponent is not None:
   667	                    precision -= int(exponent)
   668	                if float(w.group()) == approx(float(g.group()), abs=10**-precision):
   669	                    # They're close enough. Replace the text we actually
   670	                    # got with the text we want, so that it will match when we
   671	                    # check the string literally.
   672	                    got = (
   673	                        got[: g.start() + offset] + w.group() + got[g.end() + offset :]
   674	                    )
   675	                    offset += w.end() - w.start() - (g.end() - g.start())
   676	            return got
   677
   678	    return LiteralsOutputChecker

   249	    argkeys_by_item: Mapping[Scope, Mapping[nodes.Item, OrderedSet[ParamArgKey]]],
   250	    items_by_argkey: Mapping[

   256	        return items
   257
   258	    scoped_items_by_argkey = items_by_argkey[scope]
   259	    scoped_argkeys_by_item = argkeys_by_item[scope]

   261	    ignore: set[ParamArgKey] = set()

   273	            )

   275	                no_argkey_items[item] = None

   289	                            argkey_dict = other_scoped_items_by_argkey[argkey]
   290	                            if not hasattr(sys, "pypy_version_info"):
   291	                                argkey_dict[i] = None
   292	                                argkey_dict.move_to_end(i, last=False)
   293	                            else:
   294	                                # Work around a bug in PyPy:
   295	                                # https://github.com/pypy/pypy/issues/5257

   297	                                bkp = argkey_dict.copy()

   313	class FuncFixtureInfo:
   314	    """Fixture-related information for a fixture-requesting item (e.g. test
   315	    function).
   316
   317	    This is used to examine the fixtures which an item requests statically
   318	    (known during collection). This includes autouse fixtures, fixtures
   319	    requested by the `usefixtures` marker, fixtures requested in the function
   320	    parameters, and the transitive closure of these.
   321
   322	    An item may also request fixtures dynamically (using `request.getfixturevalue`);
   323	    these are not reflected here.
   324	    """
   325
   326	    __slots__ = ("argnames", "initialnames", "name2fixturedefs", "names_closure")
   327
   328	    # Fixture names that the item requests directly by function parameters.
   329	    argnames: tuple[str, ...]
   330	    # Fixture names that the item immediately requires. These include
   331	    # argnames + fixture names specified via usefixtures and via autouse=True in
   332	    # fixture definitions.
   333	    initialnames: tuple[str, ...]
   334	    # The transitive closure of the fixture names that the item requires.
   335	    # Note: can't include dynamic dependencies (`request.getfixturevalue` calls).
   336	    names_closure: list[str]
   337	    # A map from a fixture name in the transitive closure to the FixtureDefs
   338	    # matching the name which are applicable to this function.
   339	    # There may be multiple overriding fixtures with the same name. The
   340	    # sequence is ordered from furthest to closes to the function.
   341	    name2fixturedefs: dict[str, Sequence[FixtureDef[Any]]]
   342
   343	    def prune_dependency_tree(self) -> None:
   344	        """Recompute names_closure from initialnames and name2fixturedefs.
   345
   346	        Can only reduce names_closure, which means that the new closure will
   347	        always be a subset of the old one. The order is preserved.
   348
   349	        This method is needed because direct parametrization may shadow some
   350	        of the fixtures that were included in the originally built dependency
   351	        tree. In this way the dependency tree can get pruned, and the closure
   352	        of argnames may get reduced.
   353	        """
   354	        closure: set[str] = set()
   355	        working_set = set(self.initialnames)
   356	        while working_set:
   357	            argname = working_set.pop()
   358	            # Argname may be something not included in the original names_closure,
   359	            # in which case we ignore it. This currently happens with pseudo
   360	            # FixtureDefs which wrap 'get_direct_param_fixture_func(request)'.
   361	            # So they introduce the new dependency 'request' which might have
   362	            # been missing in the original tree (closure).
   363	            if argname not in closure and argname in self.names_closure:
   364	                closure.add(argname)
   365	                if argname in self.name2fixturedefs:
   366	                    working_set.update(self.name2fixturedefs[argname][-1].argnames)
   367
   368	        self.names_closure[:] = sorted(closure, key=self.names_closure.index)

   539	            The fixture name.
   540	        :raises pytest.FixtureLookupError:
   541	            If the given fixture could not be found.
   542	        """
   543	        # Note that in addition to the use case described in the docstring,
   544	        # getfixturevalue() is also called by pytest itself during item and fixture
   545	        # setup to evaluate the fixtures that are requested statically
   546	        # (using function parameters, autouse, etc).
   547
   548	        fixturedef = self._get_active_fixturedef(argname)
   549	        assert fixturedef.cached_result is not None, (
   550	            f'The fixture value for "{argname}" is not available.  '
   551	            "This can happen when the fixture has already been torn down."
   552	        )
   553	        return fixturedef.cached_result[0]
   554
   555	    def _iter_chain(self) -> Iterator[SubRequest]:
   556	        """Yield all SubRequests in the chain, from self up.
   557
   558	        Note: does *not* yield the TopRequest.
   559	        """
   560	        current = self
   561	        while isinstance(current, SubRequest):
   562	            yield current
   563	            current = current._parent_request
   564
   565	    def _get_active_fixturedef(
   566	        self, argname: str
   567	    ) -> FixtureDef[object] | PseudoFixtureDef[object]:
   568	        if argname == "request":
   569	            cached_result = (self, [0], None)
   570	            return PseudoFixtureDef(cached_result, Scope.Function)
   571
   572	        # If we already finished computing a fixture by this name in this item,
   573	        # return it.
   574	        fixturedef = self._fixture_defs.get(argname)
   575	        if fixturedef is not None:
   576	            self._check_scope(fixturedef, fixturedef._scope)
   577	            return fixturedef
   578
   579	        # Find the appropriate fixturedef.
   580	        fixturedefs = self._arg2fixturedefs.get(argname, None)
   581	        if fixturedefs is None:
   582	            # We arrive here because of a dynamic call to
   583	            # getfixturevalue(argname) which was naturally
   584	            # not known at parsing/collection time.
   585	            fixturedefs = self._fixturemanager.getfixturedefs(argname, self._pyfuncitem)
   586	            if fixturedefs is not None:
   587	                self._arg2fixturedefs[argname] = fixturedefs
   588	        # No fixtures defined with this name.
   589	        if fixturedefs is None:
   590	            raise FixtureLookupError(argname, self)
   591	        # The are no fixtures with this name applicable for the function.
   592	        if not fixturedefs:
   593	            raise FixtureLookupError(argname, self)
   594
   595	        # A fixture may override another fixture with the same name, e.g. a
   596	        # fixture in a module can override a fixture in a conftest, a fixture in
   597	        # a class can override a fixture in the module, and so on.
   598	        # An overriding fixture can request its own name (possibly indirectly);
   599	        # in this case it gets the value of the fixture it overrides, one level
   600	        # up.
   601	        # Check how many `argname`s deep we are, and take the next one.
   602	        # `fixturedefs` is sorted from furthest to closest, so use negative
   603	        # indexing to go in reverse.
   604	        index = -1
   605	        for request in self._iter_chain():
   606	            if request.fixturename == argname:
   607	                index -= 1
   608	        # If already consumed all of the available levels, fail.
   609	        if -index > len(fixturedefs):
   610	            raise FixtureLookupError(argname, self)
   611	        fixturedef = fixturedefs[index]
   612
   613	        # Prepare a SubRequest object for calling the fixture.
   614	        try:
   615	            callspec = self._pyfuncitem.callspec
   616	        except AttributeError:
   617	            callspec = None
   618	        if callspec is not None and argname in callspec.params:
   619	            param = callspec.params[argname]
   620	            param_index = callspec.indices[argname]
   621	            # The parametrize invocation scope overrides the fixture's scope.
   622	            scope = callspec._arg2scope[argname]
   623	        else:
   624	            param = NOTSET
   625	            param_index = 0
   626	            scope = fixturedef._scope
   627	            self._check_fixturedef_without_param(fixturedef)
   628	        # The parametrize invocation scope only controls caching behavior while
   629	        # allowing wider-scoped fixtures to keep depending on the parametrized
   630	        # fixture. Scope control is enforced for parametrized fixtures
   631	        # by recreating the whole fixture tree on parameter change.
   632	        # Hence `fixturedef._scope`, not `scope`.
   633	        self._check_scope(fixturedef, fixturedef._scope)
   634	        subrequest = SubRequest(
   635	            self, scope, param, param_index, fixturedef, _ispytest=True
   636	        )
   637
   638	        # Make sure the fixture value is cached, running it if it isn't
   639	        fixturedef.execute(request=subrequest)
   640
   641	        self._fixture_defs[argname] = fixturedef
   642	        return fixturedef
   643
   644	    def _check_fixturedef_without_param(self, fixturedef: FixtureDef[object]) -> None:
   645	        """Check that this request is allowed to execute this fixturedef without
   646	        a param."""
   647	        funcitem = self._pyfuncitem
   648	        has_params = fixturedef.params is not None
   649	        fixtures_not_supported = getattr(funcitem, "nofuncargs", False)
   650	        if has_params and fixtures_not_supported:
   651	            msg = (
   652	                f"{funcitem.name} does not support fixtures, maybe unittest.TestCase subclass?\n"
   653	                f"Node id: {funcitem.nodeid}\n"
   654	                f"Function type: {type(funcitem).__name__}"
   655	            )
   656	            fail(msg, pytrace=False)
   657	        if has_params:
   658	            frame = inspect.stack()[3]
   659	            frameinfo = inspect.getframeinfo(frame[0])
   660	            source_path = absolutepath(frameinfo.filename)
   661	            source_lineno = frameinfo.lineno
   662	            try:
   663	                source_path_str = str(source_path.relative_to(funcitem.config.rootpath))
   664	            except ValueError:
   665	                source_path_str = str(source_path)
   666	            location = getlocation(fixturedef.func, funcitem.config.rootpath)
   667	            msg = (
   668	                "The requested fixture has no parameter defined for test:\n"
   669	                f"    {funcitem.nodeid}\n\n"
   670	                f"Requested fixture '{fixturedef.argname}' defined in:\n"
   671	                f"{location}\n\n"
   672	                f"Requested here:\n"
   673	                f"{source_path_str}:{source_lineno}"
   674	            )
   675	            fail(msg, pytrace=False)
   676
   677	    def _get_fixturestack(self) -> list[FixtureDef[Any]]:
   678	        values = [request._fixturedef for request in self._iter_chain()]
   679	        values.reverse()
   680	        return values

   726	class SubRequest(FixtureRequest):
   727	    """The type of the ``request`` fixture in a fixture function requested
   728	    (transitively) by a test function."""
   729
   730	    def __init__(
   731	        self,
   732	        request: FixtureRequest,
   733	        scope: Scope,
   734	        param: Any,
   735	        param_index: int,
   736	        fixturedef: FixtureDef[object],
   737	        *,
   738	        _ispytest: bool = False,
   739	    ) -> None:
   740	        super().__init__(
   741	            pyfuncitem=request._pyfuncitem,
   742	            fixturename=fixturedef.argname,
   743	            fixture_defs=request._fixture_defs,
   744	            arg2fixturedefs=request._arg2fixturedefs,
   745	            _ispytest=_ispytest,
   746	        )
   747	        self._parent_request: Final[FixtureRequest] = request
   748	        self._scope_field: Final = scope
   749	        self._fixturedef: Final[FixtureDef[object]] = fixturedef
   750	        if param is not NOTSET:
   751	            self.param = param
   752	        self.param_index: Final = param_index
   753
   754	    def __repr__(self) -> str:
   755	        return f"<SubRequest {self.fixturename!r} for {self._pyfuncitem!r}>"
   756
   757	    @property
   758	    def _scope(self) -> Scope:
   759	        return self._scope_field
   760
   761	    @property
   762	    def node(self):
   763	        scope = self._scope
   764	        if scope is Scope.Function:
   765	            # This might also be a non-function Item despite its attribute name.
   766	            node: nodes.Node | None = self._pyfuncitem
   767	        elif scope is Scope.Package:
   768	            node = get_scope_package(self._pyfuncitem, self._fixturedef)
   769	        else:
   770	            node = get_scope_node(self._pyfuncitem, scope)
   771	        if node is None and scope is Scope.Class:
   772	            # Fallback to function item itself.
   773	            node = self._pyfuncitem
   774	        assert node, (
   775	            f'Could not obtain a node for scope "{scope}" for function {self._pyfuncitem!r}'
   776	        )
   777	        return node
   778
   779	    def _check_scope(
   780	        self,
   781	        requested_fixturedef: FixtureDef[object] | PseudoFixtureDef[object],
   782	        requested_scope: Scope,
   783	    ) -> None:
   784	        if isinstance(requested_fixturedef, PseudoFixtureDef):
   785	            return
   786	        if self._scope > requested_scope:
   787	            # Try to report something helpful.
   788	            argname = requested_fixturedef.argname
   789	            fixture_stack = "\n".join(
   790	                self._format_fixturedef_line(fixturedef)
   791	                for fixturedef in self._get_fixturestack()
   792	            )
   793	            requested_fixture = self._format_fixturedef_line(requested_fixturedef)
   794	            fail(
   795	                f"ScopeMismatch: You tried to access the {requested_scope.value} scoped "
   796	                f"fixture {argname} with a {self._scope.value} scoped request object. "
   797	                f"Requesting fixture stack:\n{fixture_stack}\n"
   798	                f"Requested fixture:\n{requested_fixture}",
   799	                pytrace=False,
   800	            )
   801
   802	    def _format_fixturedef_line(self, fixturedef: FixtureDef[object]) -> str:
   803	        factory = fixturedef.func
   804	        path, lineno = getfslineno(factory)
   805	        if isinstance(path, Path):
   806	            path = bestrelpath(self._pyfuncitem.session.path, path)
   807	        signature = inspect.signature(factory)
   808	        return f"{path}:{lineno + 1}:  def {factory.__name__}{signature}"
   809
   810	    def addfinalizer(self, finalizer: Callable[[], object]) -> None:
   811	        self._fixturedef.addfinalizer(finalizer)

   814	@final
   815	class FixtureLookupError(LookupError):
   816	    """Could not return a requested fixture (missing or invalid)."""
   817
   818	    def __init__(
   819	        self, argname: str | None, request: FixtureRequest, msg: str | None = None
   820	    ) -> None:
   821	        self.argname = argname
   822	        self.request = request
   823	        self.fixturestack = request._get_fixturestack()
   824	        self.msg = msg
   825
   826	    def formatrepr(self) -> FixtureLookupErrorRepr:
   827	        tblines: list[str] = []
   828	        addline = tblines.append
   829	        stack = [self.request._pyfuncitem.obj]
   830	        stack.extend(map(lambda x: x.func, self.fixturestack))
   831	        msg = self.msg
   832	        # This function currently makes an assumption that a non-None msg means we
   833	        # have a non-empty `self.fixturestack`. This is currently true, but if
   834	        # somebody at some point want to extend the use of FixtureLookupError to
   835	        # new cases it might break.
   836	        # Add the assert to make it clearer to developer that this will fail, otherwise
   837	        # it crashes because `fspath` does not get set due to `stack` being empty.
   838	        assert self.msg is None or self.fixturestack, (
   839	            "formatrepr assumptions broken, rewrite it to handle it"
   840	        )
   841	        if msg is not None:
   842	            # The last fixture raise an error, let's present
   843	            # it at the requesting side.
   844	            stack = stack[:-1]
   845	        for function in stack:
   846	            fspath, lineno = getfslineno(function)
   847	            try:
   848	                lines, _ = inspect.getsourcelines(get_real_func(function))
   849	            except (OSError, IndexError, TypeError):
   850	                error_msg = "file %s, line %s: source code not available"
   851	                addline(error_msg % (fspath, lineno + 1))
   852	            else:
   853	                addline(f"file {fspath}, line {lineno + 1}")
   854	                for i, line in enumerate(lines):
   855	                    line = line.rstrip()
   856	                    addline("  " + line)
   857	                    if line.lstrip().startswith("def"):
   858	                        break
   859
   860	        if msg is None:
   861	            fm = self.request._fixturemanager
   862	            available = set()
   863	            parent = self.request._pyfuncitem.parent
   864	            assert parent is not None
   865	            for name, fixturedefs in fm._arg2fixturedefs.items():
   866	                faclist = list(fm._matchfactories(fixturedefs, parent))
   867	                if faclist:
   868	                    available.add(name)
   869	            if self.argname in available:
   870	                msg = (
   871	                    f" recursive dependency involving fixture '{self.argname}' detected"
   872	                )
   873	            else:
   874	                msg = f"fixture '{self.argname}' not found"
   875	            msg += "\n available fixtures: {}".format(", ".join(sorted(available)))
   876	            msg += "\n use 'pytest --fixtures [testpath]' for help on them."
   877
   878	        return FixtureLookupErrorRepr(fspath, lineno, tblines, msg, self.argname)

   915	def call_fixture_func(
   916	    fixturefunc: _FixtureFunc[FixtureValue], request: FixtureRequest, kwargs
   917	) -> FixtureValue:
   918	    if inspect.isgeneratorfunction(fixturefunc):
   919	        fixturefunc = cast(Callable[..., Generator[FixtureValue]], fixturefunc)
   920	        generator = fixturefunc(**kwargs)
   921	        try:
   922	            fixture_result = next(generator)
   923	        except StopIteration:
   924	            raise ValueError(f"{request.fixturename} did not yield a value") from None
   925	        finalizer = functools.partial(_teardown_yield_fixture, fixturefunc, generator)
   926	        request.addfinalizer(finalizer)
   927	    else:
   928	        fixturefunc = cast(Callable[..., FixtureValue], fixturefunc)
   929	        fixture_result = fixturefunc(**kwargs)
   930	    return fixture_result

   933	def _teardown_yield_fixture(fixturefunc, it) -> None:
   934	    """Execute the teardown of a fixture function by advancing the iterator
   935	    after the yield and ensure the iteration ends (if not it means there is
   936	    more than one yield in the function)."""
   937	    try:
   938	        next(it)
   939	    except StopIteration:
   940	        pass
   941	    else:
   942	        fs, lineno = getfslineno(fixturefunc)
   943	        fail(
   944	            f"fixture function has more than one 'yield':\n\n"
   945	            f"{Source(fixturefunc).indent()}\n"
   946	            f"{fs}:{lineno + 1}",
   947	            pytrace=False,
   948	        )

   951	def _eval_scope_callable(
   952	    scope_callable: Callable[[str, Config], _ScopeName],
   953	    fixture_name: str,
   954	    config: Config,
   955	) -> _ScopeName:
   956	    try:
   957	        # Type ignored because there is no typing mechanism to specify
   958	        # keyword arguments, currently.
   959	        result = scope_callable(fixture_name=fixture_name, config=config)  # type: ignore[call-arg]
   960	    except Exception as e:
   961	        raise TypeError(
   962	            f"Error evaluating {scope_callable} while defining fixture '{fixture_name}'.\n"
   963	            "Expected a function with the signature (*, fixture_name, config)"
   964	        ) from e
   965	    if not isinstance(result, str):
   966	        fail(
   967	            f"Expected {scope_callable} to return a 'str' while defining fixture '{fixture_name}', but it returned:\n"
   968	            f"{result!r}",
   969	            pytrace=False,
   970	        )
   971	    return result

   974	@final
   975	class FixtureDef(Generic[FixtureValue]):
   976	    """A container for a fixture definition.
   977
   978	    Note: At this time, only explicitly documented fields and methods are
   979	    considered public stable API.
   980	    """
   981
   982	    def __init__(
   983	        self,
   984	        config: Config,
   985	        baseid: str | None,
   986	        argname: str,
   987	        func: _FixtureFunc[FixtureValue],
   988	        scope: Scope | _ScopeName | Callable[[str, Config], _ScopeName] | None,
   989	        params: Sequence[object] | None,
   990	        ids: tuple[object | None, ...] | Callable[[Any], object | None] | None = None,
   991	        *,
   992	        _ispytest: bool = False,
   993	        # only used in a deprecationwarning msg, can be removed in pytest9
   994	        _autouse: bool = False,
   995	    ) -> None:
   996	        check_ispytest(_ispytest)
   997	        # The "base" node ID for the fixture.
   998	        #
   999	        # This is a node ID prefix. A fixture is only available to a node (e.g.
  1000	        # a `Function` item) if the fixture's baseid is a nodeid of a parent of
  1001	        # node.
  1002	        #
  1003	        # For a fixture found in a Collector's object (e.g. a `Module`s module,
  1004	        # a `Class`'s class), the baseid is the Collector's nodeid.
  1005	        #
  1006	        # For a fixture found in a conftest plugin, the baseid is the conftest's
  1007	        # directory path relative to the rootdir.
  1008	        #
  1009	        # For other plugins, the baseid is the empty string (always matches).
  1010	        self.baseid: Final = baseid or ""
  1011	        # Whether the fixture was found from a node or a conftest in the
  1012	        # collection tree. Will be false for fixtures defined in non-conftest
  1013	        # plugins.
  1014	        self.has_location: Final = baseid is not None
  1015	        # The fixture factory function.
  1016	        self.func: Final = func
  1017	        # The name by which the fixture may be requested.
  1018	        self.argname: Final = argname
  1019	        if scope is None:
  1020	            scope = Scope.Function
  1021	        elif callable(scope):
  1022	            scope = _eval_scope_callable(scope, argname, config)
  1023	        if isinstance(scope, str):
  1024	            scope = Scope.from_user(
  1025	                scope, descr=f"Fixture '{func.__name__}'", where=baseid
  1026	            )
  1027	        self._scope: Final = scope
  1028	        # If the fixture is directly parametrized, the parameter values.
  1029	        self.params: Final = params
  1030	        # If the fixture is directly parametrized, a tuple of explicit IDs to
  1031	        # assign to the parameter values, or a callable to generate an ID given
  1032	        # a parameter value.
  1033	        self.ids: Final = ids
  1034	        # The names requested by the fixtures.
  1035	        self.argnames: Final = getfuncargnames(func, name=argname)
  1036	        # If the fixture was executed, the current value of the fixture.
  1037	        # Can change if the fixture is executed with different parameters.
  1038	        self.cached_result: _FixtureCachedResult[FixtureValue] | None = None
  1039	        self._finalizers: Final[list[Callable[[], object]]] = []
  1040
  1041	        # only used to emit a deprecationwarning, can be removed in pytest9
  1042	        self._autouse = _autouse
  1043
  1044	    @property
  1045	    def scope(self) -> _ScopeName:
  1046	        """Scope string, one of "function", "class", "module", "package", "session"."""
  1047	        return self._scope.value
  1048
  1049	    def addfinalizer(self, finalizer: Callable[[], object]) -> None:
  1050	        self._finalizers.append(finalizer)
  1051
  1052	    def finish(self, request: SubRequest) -> None:
  1053	        exceptions: list[BaseException] = []
  1054	        while self._finalizers:
  1055	            fin = self._finalizers.pop()
  1056	            try:
  1057	                fin()
  1058	            except BaseException as e:
  1059	                exceptions.append(e)
  1060	        node = request.node
  1061	        node.ihook.pytest_fixture_post_finalizer(fixturedef=self, request=request)
  1062	        # Even if finalization fails, we invalidate the cached fixture
  1063	        # value and remove all finalizers because they may be bound methods
  1064	        # which will keep instances alive.
  1065	        self.cached_result = None
  1066	        self._finalizers.clear()
  1067	        if len(exceptions) == 1:
  1068	            raise exceptions[0]
  1069	        elif len(exceptions) > 1:
  1070	            msg = f'errors while tearing down fixture "{self.argname}" of {node}'
  1071	            raise BaseExceptionGroup(msg, exceptions[::-1])

  1078	        # parametrization, they finalize themselves and fixtures depending on it

  1095	            cache_key = self.cached_result[1]
  1096	            try:

  1098	                # which do not implement the standard comparison (like numpy arrays -- #6497).
  1099	                cache_hit = bool(request_cache_key == cache_key)
  1100	            except (ValueError, RuntimeError):

  1102	                cache_hit = request_cache_key is cache_key

  1115

  1120	        for parent_fixture in requested_fixtures_that_should_finalize_us:
  1121	            parent_fixture.addfinalizer(finalizer)

  1126	            # in self.cached_result

  1128	        finally:

  1130	            request.node.addfinalizer(finalizer)
