   227	        assert right_type.is_protocol
   228	        self._rechecked_types.add(left_type)
   229	        self._attempted_protocols.setdefault(left_type.fullname, set()).add(right_type.fullname)
   230	        self._checked_against_members.setdefault(left_type.fullname, set()).update(
   231	            right_type.protocol_members
   232	        )

   234	    def _snapshot_protocol_deps(self) -> dict[str, set[str]]:
   235	        """Collect protocol attribute dependencies found so far from registered subtype checks.
   236
   237	        There are three kinds of protocol dependencies. For example, after a subtype check:
   238
   239	            x: Proto = C()
   240
   241	        the following dependencies will be generated:
   242	            1. ..., <SuperProto[wildcard]>, <Proto[wildcard]> -> <Proto>
   243	            2. ..., <B.attr>, <C.attr> -> <C> [for every attr in Proto members]
   244	            3. <C> -> Proto  # this one to invalidate the subtype cache
   245
   246	        The first kind is generated immediately per-module in deps.py (see also an example there
   247	        for motivation why it is needed). While two other kinds are generated here after all
   248	        modules are type checked and we have recorded all the subtype checks. To understand these
   249	        two kinds, consider a simple example:
   250
   251	            class A:
   252	                def __iter__(self) -> Iterator[int]:
   253	                    ...
   254
   255	            it: Iterable[int] = A()
   256
   257	        We add <a.A.__iter__> -> <a.A> to invalidate the assignment (module target in this case),
   258	        whenever the signature of a.A.__iter__ changes. We also add <a.A> -> typing.Iterable,
   259	        to invalidate the subtype caches of the latter. (Note that the same logic applies to
   260	        proper subtype checks, and calculating meets and joins, if this involves calling
   261	        'subtypes.is_protocol_implementation').
   262	        """
   263	        deps: dict[str, set[str]] = {}
   264	        for info in self._rechecked_types:
   265	            for attr in self._checked_against_members[info.fullname]:
   266	                # The need for full MRO here is subtle, during an update, base classes of
   267	                # a concrete class may not be reprocessed, so not all <B.x> -> <C.x> deps
   268	                # are added.
   269	                for base_info in info.mro[:-1]:
   270	                    trigger = make_trigger(f"{base_info.fullname}.{attr}")
   271	                    if "typing" in trigger or "builtins" in trigger:
   272	                        # TODO: avoid everything from typeshed
   273	                        continue
   274	                    deps.setdefault(trigger, set()).add(make_trigger(info.fullname))
   275	            for proto in self._attempted_protocols[info.fullname]:
   276	                trigger = make_trigger(info.fullname)
   277	                if "typing" in trigger or "builtins" in trigger:
   278	                    continue
   279	                # If any class that was checked against a protocol changes,
   280	                # we need to reset the subtype cache for the protocol.
   281	                #
   282	                # Note: strictly speaking, the protocol doesn't need to be
   283	                # re-checked, we only need to reset the cache, and its uses
   284	                # elsewhere are still valid (unless invalidated by other deps).
   285	                deps.setdefault(trigger, set()).add(proto)
   286	        return deps
   287
   288	    def update_protocol_deps(self, second_map: dict[str, set[str]] | None = None) -> None:
   289	        """Update global protocol dependency map.
   290
   291	        We update the global map incrementally, using a snapshot only from recently
   292	        type checked types. If second_map is given, update it as well. This is currently used
   293	        by FineGrainedBuildManager that maintains normal (non-protocol) dependencies.
   294	        """
   295	        assert self.proto_deps is not None, "This should not be called after failed cache load"
   296	        new_deps = self._snapshot_protocol_deps()
   297	        for trigger, targets in new_deps.items():
   298	            self.proto_deps.setdefault(trigger, set()).update(targets)
   299	        if second_map is not None:
   300	            for trigger, targets in new_deps.items():
   301	                second_map.setdefault(trigger, set()).update(targets)
   302	        self._rechecked_types.clear()
   303	        self._attempted_protocols.clear()
   304	        self._checked_against_members.clear()
   305
   306	    def add_all_protocol_deps(self, deps: dict[str, set[str]]) -> None:

   312	        self.update_protocol_deps()  # just in case
   313	        if self.proto_deps is not None:
   314	            for trigger, targets in self.proto_deps.items():
   315	                deps.setdefault(trigger, set()).update(targets)

   198	    if max_len < 2 * min_width + 1:
   199	        # In case the window is too tiny it is better to still show something.
   200	        max_len = 2 * min_width + 1
   201
   202	    # Trivial case: line already fits in.
   203	    if len(line) <= max_len:
   204	        return line, 0
   205
   206	    # If column is not too large so that there is still min_width after it,
   207	    # the line doesn't need to be trimmed at the start.
   208	    if col + min_width < max_len:
   209	        return line[:max_len] + "...", 0
   210
   211	    # Otherwise, if the column is not too close to the end, trim both sides.
   212	    if col < len(line) - min_width - 1:
   213	        offset = col - max_len + min_width + 1
   214	        return "..." + line[offset : col + min_width + 1] + "...", offset - 3
   215
   216	    # Finally, if the column is near the end, just trim the start.
   217	    return "..." + line[-max_len:], len(line) - max_len - 3

   381	    new: object, old: object, copy_dict: bool = False, skip_slots: tuple[str, ...] = ()
   382	) -> None:
   383	    """Copy state of old node to the new node.
   384

   405	        # writes from ones that don't (I think?), so we just ignore
   406	        # AttributeErrors if we need to.
   407	        # TODO: What about getsetdescriptors that act like properties???
   408	        except AttributeError:

   413	    """Given two paths, return if path is a sub-path of dir.

   415	    Moral equivalent of: Path(dir) in Path(path).parents

   526	def soft_wrap(msg: str, max_len: int, first_offset: int, num_indent: int = 0) -> str:
   527	    """Wrap a long error message into few lines.
   528
   529	    Breaks will only happen between words, and never inside a quoted group
   530	    (to avoid breaking types such as "Union[int, str]"). The 'first_offset' is
   531	    the width before the start of first line.
   532
   533	    Pad every next line with 'num_indent' spaces. Every line will be at most 'max_len'
   534	    characters, except if it is a single word or quoted group.
   535
   536	    For example:
   537	               first_offset
   538	        ------------------------
   539	        path/to/file: error: 58: Some very long error message
   540	            that needs to be split in separate lines.
   541	            "Long[Type, Names]" are never split.
   542	        ^^^^--------------------------------------------------
   543	        num_indent           max_len
   544	    """
   545	    words = split_words(msg)
   546	    next_line = words.pop(0)
   547	    lines: list[str] = []
   548	    while words:
   549	        next_word = words.pop(0)
   550	        max_line_len = max_len - num_indent if lines else max_len - first_offset
   551	        # Add 1 to account for space between words.
   552	        if len(next_line) + len(next_word) + 1 <= max_line_len:
   553	            next_line += " " + next_word
   554	        else:
   555	            lines.append(next_line)
   556	            next_line = next_word
   557	    lines.append(next_line)
   558	    padding = "\n" + " " * num_indent
   559	    return padding.join(lines)

   666	        """Return True if initialization was successful and we can use colors, False otherwise"""
   667	        is_win = sys.platform == "win32"
   668	        if is_win or not CURSES_ENABLED:
   669	            return False
   670	        try:
   671	            # setupterm wants a fd to potentially write an "initialization sequence".
   672	            # We override sys.stdout for the daemon API so if stdout doesn't have an fd,
   673	            # just give it /dev/null.
   674	            try:
   675	                fd = sys.stdout.fileno()
   676	            except io.UnsupportedOperation:

   678	                    curses.setupterm(fd=f.fileno())
   679	            else:
   680	                curses.setupterm(fd=fd)

   682	            # Most likely terminfo not found.
   683	            return False

   685	        under = curses.tigetstr("smul")
   686	        set_color = curses.tigetstr("setaf")
   687	        set_eseq = curses.tigetstr("cup")
   688	        normal = curses.tigetstr("sgr0")
   689
   690	        if not (bold and under and set_color and set_eseq and normal):
   691	            return False

   694	        self.BOLD = bold.decode()
   695	        self.UNDER = under.decode()
   696	        self.DIM = parse_gray_color(set_eseq)
   697	        self.BLUE = curses.tparm(set_color, curses.COLOR_BLUE).decode()
   698	        self.GREEN = curses.tparm(set_color, curses.COLOR_GREEN).decode()
   699	        self.RED = curses.tparm(set_color, curses.COLOR_RED).decode()
   700	        self.YELLOW = curses.tparm(set_color, curses.COLOR_YELLOW).decode()

   724	    def fit_in_terminal(
   725	        self, messages: list[str], fixed_terminal_width: int | None = None
   726	    ) -> list[str]:
   727	        """Improve readability by wrapping error messages and trimming source code."""
   728	        width = fixed_terminal_width or get_terminal_width()
   729	        new_messages = messages.copy()
   730	        for i, error in enumerate(messages):
   731	            if ": error:" in error:
   732	                loc, msg = error.split("error:", maxsplit=1)
   733	                msg = soft_wrap(msg, width, first_offset=len(loc) + len("error: "))
   734	                new_messages[i] = loc + "error:" + msg
   735	            if error.startswith(" " * DEFAULT_SOURCE_OFFSET) and "^" not in error:
   736	                # TODO: detecting source code highlights through an indent can be surprising.
   737	                # Restore original error message and error location.
   738	                error = error[DEFAULT_SOURCE_OFFSET:]
   739	                marker_line = messages[i + 1]
   740	                marker_column = marker_line.index("^")
   741	                column = marker_column - DEFAULT_SOURCE_OFFSET
   742	                if "~" not in marker_line:
   743	                    marker = "^"
   744	                else:
   745	                    # +1 because both ends are included
   746	                    marker = marker_line[marker_column : marker_line.rindex("~") + 1]
   747
   748	                # Let source have some space also on the right side, plus 6
   749	                # to accommodate ... on each side.
   750	                max_len = width - DEFAULT_SOURCE_OFFSET - 6
   751	                source_line, offset = trim_source_line(error, max_len, column, MINIMUM_WIDTH)

   753	                new_messages[i] = " " * DEFAULT_SOURCE_OFFSET + source_line
   754	                # Also adjust the error marker position and trim error marker is needed.
   755	                new_marker_line = " " * (DEFAULT_SOURCE_OFFSET + column - offset) + marker
   756	                if len(new_marker_line) > len(new_messages[i]) and len(marker) > 3:
   757	                    new_marker_line = new_marker_line[: len(new_messages[i]) - 3] + "..."
   758	                new_messages[i + 1] = new_marker_line

   903	    """Returns docstring correctly encapsulated in a single or double quoted form."""
   904	    # Uses repr to get hint on the correct quotes and escape everything properly.
   905	    # Creating multiline string for prettier output.

   913	            return f'"""{docstr_repr[1:-1]}"""'

     1	"""Always defined attribute analysis.
     2
     3	An always defined attribute has some statements in __init__ or the
     4	class body that cause the attribute to be always initialized when an
     5	instance is constructed. It must also not be possible to read the
     6	attribute before initialization, and it can't be deletable.
     7
     8	We can assume that the value is always defined when reading an always
     9	defined attribute. Otherwise we'll need to raise AttributeError if the
    10	value is undefined (i.e. has the error value).
    11
    12	We use data flow analysis to figure out attributes that are always
    13	defined. Example:
    14
    15	  class C:
    16	      def __init__(self) -> None:
    17	          self.x = 0
    18	          if func():
    19	              self.y = 1
    20	          else:
    21	              self.y = 2
    22	              self.z = 3
    23
    24	In this example, the attributes 'x' and 'y' are always defined, but 'z'
    25	is not. The analysis assumes that we know that there won't be any subclasses.
    26
    27	The analysis also works if there is a known, closed set of subclasses.
    28	An attribute defined in a base class can only be always defined if it's
    29	also always defined in all subclasses.
    30
    31	As soon as __init__ contains an op that can 'leak' self to another
    32	function, we will stop inferring always defined attributes, since the
    33	analysis is mostly intra-procedural and only looks at __init__ methods.
    34	The called code could read an uninitialized attribute. Example:
    35
    36	  class C:
    37	      def __init__(self) -> None:
    38	          self.x = self.foo()
    39
    40	      def foo(self) -> int:
    41	          ...
    42
    43	Now we won't infer 'x' as always defined, since 'foo' might read 'x'
    44	before initialization.
    45
    46	As an exception to the above limitation, we perform inter-procedural
    47	analysis of super().__init__ calls, since these are very common.
    48
    49	Our analysis is somewhat optimistic. We assume that nobody calls a
    50	method of a partially uninitialized object through gc.get_objects(), in
    51	particular. Code like this could potentially cause a segfault with a null
    52	pointer dereference. This seems very unlikely to be an issue in practice,
    53	however.
    54
    55	Accessing an attribute via getattr always checks for undefined attributes
    56	and thus works if the object is partially uninitialized. This can be used
    57	as a workaround if somebody ever needs to inspect partially uninitialized
    58	objects via gc.get_objects().
    59
    60	The analysis runs after IR building as a separate pass. Since we only
    61	run this on __init__ methods, this analysis pass will be fairly quick.
    62	"""

   113	    # First pass: only look at target class and classes in MRO
   114	    for cl in class_irs:
   115	        analyze_always_defined_attrs_in_class(cl, seen)
   116
   117	    # Second pass: look at all derived class
   118	    seen = set()
   119	    for cl in class_irs:

   121

   123	    seen = set()
   124	    for cl in class_irs:

   128	def analyze_always_defined_attrs_in_class(cl: ClassIR, seen: set[ClassIR]) -> None:
   129	    if cl in seen:
   130	        return
   131
   132	    seen.add(cl)
   133
   134	    if (
   135	        cl.is_trait
   136	        or cl.inherits_python
   137	        or cl.allow_interpreted_subclasses
   138	        or cl.builtin_base is not None
   139	        or cl.children is None
   140	        or cl.is_serializable()
   141	    ):
   142	        # Give up -- we can't enforce that attributes are always defined.
   143	        return
   144
   145	    # First analyze all base classes. Track seen classes to avoid duplicate work.
   146	    for base in cl.mro[1:]:
   147	        analyze_always_defined_attrs_in_class(base, seen)
   148
   149	    m = cl.get_method("__init__")
   150	    if m is None:
   151	        cl._always_initialized_attrs = cl.attrs_with_defaults.copy()
   152	        cl._sometimes_initialized_attrs = cl.attrs_with_defaults.copy()
   153	        return
   154	    self_reg = m.arg_regs[0]
   155	    cfg = get_cfg(m.blocks)
   156	    dirty = analyze_self_leaks(m.blocks, self_reg, cfg)
   157	    maybe_defined = analyze_maybe_defined_attrs_in_init(
   158	        m.blocks, self_reg, cl.attrs_with_defaults, cfg
   159	    )
   160	    all_attrs: set[str] = set()
   161	    for base in cl.mro:
   162	        all_attrs.update(base.attributes)
   163	    maybe_undefined = analyze_maybe_undefined_attrs_in_init(
   164	        m.blocks, self_reg, initial_undefined=all_attrs - cl.attrs_with_defaults, cfg=cfg
   165	    )
   166
   167	    always_defined = find_always_defined_attributes(
   168	        m.blocks, self_reg, all_attrs, maybe_defined, maybe_undefined, dirty
   169	    )
   170	    always_defined = {a for a in always_defined if not cl.is_deletable(a)}
   171
   172	    cl._always_initialized_attrs = always_defined
   173	    if dump_always_defined:
   174	        print(cl.name, sorted(always_defined))
   175	    cl._sometimes_initialized_attrs = find_sometimes_defined_attributes(
   176	        m.blocks, self_reg, maybe_defined, dirty
   177	    )
   178
   179	    mark_attr_initialization_ops(m.blocks, self_reg, maybe_defined, dirty)
   180
   181	    # Check if __init__ can run unpredictable code (leak 'self').
   182	    any_dirty = False
   183	    for b in m.blocks:
   184	        for i, op in enumerate(b.ops):
   185	            if dirty.after[b, i] and not isinstance(op, Return):
   186	                any_dirty = True
   187	                break
   188	    cl.init_self_leak = any_dirty

   191	def find_always_defined_attributes(
   192	    blocks: list[BasicBlock],
   193	    self_reg: Register,
   194	    all_attrs: set[str],
   195	    maybe_defined: AnalysisResult[str],
   196	    maybe_undefined: AnalysisResult[str],
   197	    dirty: AnalysisResult[None],
   198	) -> set[str]:
   199	    """Find attributes that are always initialized in some basic blocks.
   200
   201	    The analysis results are expected to be up-to-date for the blocks.
   202
   203	    Return a set of always defined attributes.
   204	    """
   205	    attrs = all_attrs.copy()
   206	    for block in blocks:
   207	        for i, op in enumerate(block.ops):
   208	            # If an attribute we *read* may be undefined, it isn't always defined.
   209	            if isinstance(op, GetAttr) and op.obj is self_reg:
   210	                if op.attr in maybe_undefined.before[block, i]:
   211	                    attrs.discard(op.attr)
   212	            # If an attribute we *set* may be sometimes undefined and
   213	            # sometimes defined, don't consider it always defined. Unlike
   214	            # the get case, it's fine for the attribute to be undefined.
   215	            # The set operation will then be treated as initialization.
   216	            if isinstance(op, SetAttr) and op.obj is self_reg:
   217	                if (
   218	                    op.attr in maybe_undefined.before[block, i]
   219	                    and op.attr in maybe_defined.before[block, i]
   220	                ):
   221	                    attrs.discard(op.attr)
   222	            # Treat an op that might run arbitrary code as an "exit"
   223	            # in terms of the analysis -- we can't do any inference
   224	            # afterwards reliably.
   225	            if dirty.after[block, i]:
   226	                if not dirty.before[block, i]:
   227	                    attrs = attrs & (
   228	                        maybe_defined.after[block, i] - maybe_undefined.after[block, i]
   229	                    )
   230	                break
   231	            if isinstance(op, ControlOp):
   232	                for target in op.targets():
   233	                    # Gotos/branches can also be "exits".
   234	                    if not dirty.after[block, i] and dirty.before[target, 0]:
   235	                        attrs = attrs & (
   236	                            maybe_defined.after[target, 0] - maybe_undefined.after[target, 0]
   237	                        )
   238	    return attrs

   241	def find_sometimes_defined_attributes(
   242	    blocks: list[BasicBlock],
   243	    self_reg: Register,
   244	    maybe_defined: AnalysisResult[str],
   245	    dirty: AnalysisResult[None],
   246	) -> set[str]:
   247	    """Find attributes that are sometimes initialized in some basic blocks."""
   248	    attrs: set[str] = set()
   249	    for block in blocks:
   250	        for i, op in enumerate(block.ops):
   251	            # Only look at possibly defined attributes at exits.
   252	            if dirty.after[block, i]:
   253	                if not dirty.before[block, i]:
   254	                    attrs = attrs | maybe_defined.after[block, i]
   255	                break
   256	            if isinstance(op, ControlOp):
   257	                for target in op.targets():
   258	                    if not dirty.after[block, i] and dirty.before[target, 0]:
   259	                        attrs = attrs | maybe_defined.after[target, 0]
   260	    return attrs
