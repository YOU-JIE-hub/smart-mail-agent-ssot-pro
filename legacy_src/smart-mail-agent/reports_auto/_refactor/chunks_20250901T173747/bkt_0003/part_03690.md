lt"))
   651	    attr_has_factory = bool(_get_argument(rvalue, "factory"))
   652
   653	    if attr_has_default and attr_has_factory:
   654	        ctx.api.fail('Can\'t pass both "default" and "factory".', rvalue)
   655	    elif attr_has_factory:
   656	        attr_has_default = True
   657
   658	    # If the type isn't set through annotation but is passed through `type=` use that.
   659	    type_arg = _get_argument(rvalue, "type")
   660	    if type_arg and not init_type:
   661	        try:
   662	            un_type = expr_to_unanalyzed_type(type_arg, ctx.api.options, ctx.api.is_stub_file)
   663	        except TypeTranslationError:
   664	            ctx.api.fail("Invalid argument to type", type_arg)
   665	        else:
   666	            init_type = ctx.api.anal_type(un_type)
   667	            if init_type and isinstance(lhs.node, Var) and not lhs.node.type:
   668	                # If there is no annotation, add one.
   669	                lhs.node.type = init_type
   670	                lhs.is_inferred_def = False
   671
   672	    # Note: convert is deprecated but works the same as converter.
   673	    converter = _get_argument(rvalue, "converter")
   674	    convert = _get_argument(rvalue, "convert")
   675	    if convert and converter:
   676	        ctx.api.fail('Can\'t pass both "convert" and "converter".', rvalue)
   677	    elif convert:
   678	        ctx.api.fail("convert is deprecated, use converter", rvalue)
   679	        converter = convert
   680	    converter_info = _parse_converter(ctx, converter)
   681
   682	    # Custom alias might be defined:
   683	    alias = None
   684	    alias_expr = _get_argument(rvalue, "alias")
   685	    if alias_expr:
   686	        alias = ctx.api.parse_str_literal(alias_expr)
   687	        if alias is None:
   688	            ctx.api.fail(
   689	                '"alias" argument to attrs field must be a string literal',
   690	                rvalue,
   691	                code=LITERAL_REQ,
   692	            )
   693	    name = unmangle(lhs.name)
   694	    return Attribute(
   695	        name, alias, ctx.cls.info, attr_has_default, init, kw_only, converter_info, stmt, init_type
   696	    )

   699	def _parse_converter(
   700	    ctx: mypy.plugin.ClassDefContext, converter_expr: Expression | None
   701	) -> Converter | None:
   702	    """Return the Converter object from an Expression."""
   703	    # TODO: Support complex converters, e.g. lambdas, calls, etc.
   704	    if not converter_expr:
   705	        return None
   706	    converter_info = Converter()
   707	    if (
   708	        isinstance(converter_expr, CallExpr)
   709	        and isinstance(converter_expr.callee, RefExpr)
   710	        and converter_expr.callee.fullname in attr_optional_converters
   711	        and converter_expr.args
   712	        and converter_expr.args[0]
   713	    ):
   714	        # Special handling for attr.converters.optional(type)
   715	        # We extract the type and add make the init_args Optional in Attribute.argument
   716	        converter_expr = converter_expr.args[0]
   717	        is_attr_converters_optional = True
   718	    else:
   719	        is_attr_converters_optional = False
   720
   721	    converter_type: Type | None = None
   722	    if isinstance(converter_expr, RefExpr) and converter_expr.node:
   723	        if isinstance(converter_expr.node, FuncDef):
   724	            if converter_expr.node.type and isinstance(converter_expr.node.type, FunctionLike):
   725	                converter_type = converter_expr.node.type
   726	            else:  # The converter is an unannotated function.
   727	                converter_info.init_type = AnyType(TypeOfAny.unannotated)
   728	                return converter_info
   729	        elif isinstance(converter_expr.node, OverloadedFuncDef) and is_valid_overloaded_converter(
   730	            converter_expr.node
   731	        ):
   732	            converter_type = converter_expr.node.type
   733	        elif isinstance(converter_expr.node, TypeInfo):
   734	            converter_type = type_object_type(converter_expr.node, ctx.api.named_type)
   735	    elif (
   736	        isinstance(converter_expr, IndexExpr)
   737	        and isinstance(converter_expr.analyzed, TypeApplication)
   738	        and isinstance(converter_expr.base, RefExpr)
   739	        and isinstance(converter_expr.base.node, TypeInfo)
   740	    ):
   741	        # The converter is a generic type.
   742	        converter_type = type_object_type(converter_expr.base.node, ctx.api.named_type)
   743	        if isinstance(converter_type, CallableType):
   744	            converter_type = apply_generic_arguments(
   745	                converter_type,
   746	                converter_expr.analyzed.types,
   747	                ctx.api.msg.incompatible_typevar_value,
   748	                converter_type,
   749	            )
   750	        else:
   751	            converter_type = None
   752
   753	    if isinstance(converter_expr, LambdaExpr):
   754	        # TODO: should we send a fail if converter_expr.min_args > 1?
   755	        converter_info.init_type = AnyType(TypeOfAny.unannotated)
   756	        return converter_info
   757
   758	    if not converter_type:
   759	        # Signal that we have an unsupported converter.
   760	        ctx.api.fail(
   761	            "Unsupported converter, only named functions, types and lambdas are currently "
   762	            "supported",
   763	            converter_expr,
   764	        )
   765	        converter_info.init_type = AnyType(TypeOfAny.from_error)
   766	        return converter_info
   767
   768	    converter_type = get_proper_type(converter_type)
   769	    if isinstance(converter_type, CallableType) and converter_type.arg_types:
   770	        converter_info.init_type = converter_type.arg_types[0]
   771	        if not is_attr_converters_optional:
   772	            converter_info.ret_type = converter_type.ret_type
   773	    elif isinstance(converter_type, Overloaded):
   774	        types: list[Type] = []
   775	        for item in converter_type.items:
   776	            # Walk the overloads looking for methods that can accept one argument.
   777	            num_arg_types = len(item.arg_types)
   778	            if not num_arg_types:
   779	                continue
   780	            if num_arg_types > 1 and any(kind == ARG_POS for kind in item.arg_kinds[1:]):
   781	                continue
   782	            types.append(item.arg_types[0])
   783	        # Make a union of all the valid types.
   784	        if types:
   785	            converter_info.init_type = make_simplified_union(types)
   786
   787	    if is_attr_converters_optional and converter_info.init_type:
   788	        # If the converter was attr.converter.optional(type) then add None to
   789	        # the allowed init_type.
   790	        converter_info.init_type = UnionType.make_union([converter_info.init_type, NoneType()])
   791
   792	    return converter_info
   793

   803	    lvalue: Expression, stmt: AssignmentStmt
   804	) -> tuple[list[NameExpr], list[Expression]]:
   805	    """Convert a possibly complex assignment expression into lists of lvalues and rvalues."""
   806	    lvalues: list[NameExpr] = []
   807	    rvalues: list[Expression] = []
   808	    if isinstance(lvalue, (TupleExpr, ListExpr)):
   809	        if all(isinstance(item, NameExpr) for item in lvalue.items):
   810	            lvalues = cast(list[NameExpr], lvalue.items)

   815	        rvalues = [stmt.rvalue]
   816	    return lvalues, rvalues

   819	def _add_order(ctx: mypy.plugin.ClassDefContext, adder: MethodAdder) -> None:
   820	    """Generate all the ordering methods for this class."""
   821	    bool_type = ctx.api.named_type("builtins.bool")
   822	    object_type = ctx.api.named_type("builtins.object")
   823	    # Make the types be:
   824	    #    AT = TypeVar('AT')
   825	    #    def __lt__(self: AT, other: AT) -> bool
   826	    # This way comparisons with subclasses will work correctly.
   827	    fullname = f"{ctx.cls.info.fullname}.{SELF_TVAR_NAME}"
   828	    tvd = TypeVarType(
   829	        SELF_TVAR_NAME,
   830	        fullname,
   831	        # Namespace is patched per-method below.
   832	        id=TypeVarId(-1, namespace=""),
   833	        values=[],
   834	        upper_bound=object_type,
   835	        default=AnyType(TypeOfAny.from_omitted_generics),
   836	    )
   837	    self_tvar_expr = TypeVarExpr(
   838	        SELF_TVAR_NAME, fullname, [], object_type, AnyType(TypeOfAny.from_omitted_generics)
   839	    )
   840	    ctx.cls.info.names[SELF_TVAR_NAME] = SymbolTableNode(MDEF, self_tvar_expr)
   841
   842	    for method in ["__lt__", "__le__", "__gt__", "__ge__"]:
   843	        namespace = f"{ctx.cls.info.fullname}.{method}"
   844	        tvd = tvd.copy_modified(id=TypeVarId(tvd.id.raw_id, namespace=namespace))
   845	        args = [Argument(Var("other", tvd), tvd, None, ARG_POS)]
   846	        adder.add_method(method, args, bool_type, self_type=tvd, tvd=tvd)

   849	def _make_frozen(ctx: mypy.plugin.ClassDefContext, attributes: list[Attribute]) -> None:
   850	    """Turn all the attributes into properties to simulate frozen classes."""
   851	    for attribute in attributes:
   852	        if attribute.name in ctx.cls.info.names:
   853	            # This variable belongs to this class so we can modify it.
   854	            node = ctx.cls.info.names[attribute.name].node
   855	            if not isinstance(node, Var):
   856	                # The superclass attribute was overridden with a non-variable.
   857	                # No need to do anything here, override will be verified during
   858	                # type checking.
   859	                continue
   860	            node.is_property = True
   861	        else:
   862	            # This variable belongs to a super class so create new Var so we
   863	            # can modify it.
   864	            var = Var(attribute.name, attribute.init_type)
   865	            var.info = ctx.cls.info
   866	            var._fullname = f"{ctx.cls.info.fullname}.{var.name}"
   867	            ctx.cls.info.names[var.name] = SymbolTableNode(MDEF, var)
   868	            var.is_property = True

   871	def _add_init(
   872	    ctx: mypy.plugin.ClassDefContext,
   873	    attributes: list[Attribute],
   874	    adder: MethodAdder,
   875	    method_name: Literal["__init__", "__attrs_init__"],
   876	) -> None:
   877	    """Generate an __init__ method for the attributes and add it to the class."""
   878	    # Convert attributes to arguments with kw_only arguments at the end of
   879	    # the argument list
   880	    pos_args = []
   881	    kw_only_args = []
   882	    sym_table = ctx.cls.info.names
   883	    for attribute in attributes:
   884	        if not attribute.init:
   885	            continue
   886	        if attribute.kw_only:
   887	            kw_only_args.append(attribute.argument(ctx))
   888	        else:
   889	            pos_args.append(attribute.argument(ctx))
   890
   891	        # If the attribute is Final, present in `__init__` and has
   892	        # no default, make sure it doesn't error later.
   893	        if not attribute.has_default and attribute.name in sym_table:
   894	            sym_node = sym_table[attribute.name].node
   895	            if isinstance(sym_node, Var) and sym_node.is_final:
   896	                sym_node.final_set_in_init = True
   897	    args = pos_args + kw_only_args
   898	    if all(
   899	        # We use getattr rather than instance checks because the variable.type
   900	        # might be wrapped into a Union or some other type, but even non-Any
   901	        # types reliably track the fact that the argument was not annotated.
   902	        getattr(arg.variable.type, "type_of_any", None) == TypeOfAny.unannotated
   903	        for arg in args
   904	    ):
   905	        # This workaround makes --disallow-incomplete-defs usable with attrs,
   906	        # but is definitely suboptimal as a long-term solution.
   907	        # See https://github.com/python/mypy/issues/5954 for discussion.
   908	        for a in args:
   909	            a.variable.type = AnyType(TypeOfAny.implementation_artifact)
   910	            a.type_annotation = AnyType(TypeOfAny.implementation_artifact)
   911	    adder.add_method(method_name, args, NoneType())

  1057	def _get_expanded_attr_types(
  1058	    ctx: mypy.plugin.FunctionSigContext,
  1059	    typ: ProperType,
  1060	    display_typ: ProperType,
  1061	    parent_typ: ProperType,
  1062	) -> list[Mapping[str, Type]] | None:
  1063	    """
  1064	    For a given type, determine what attrs classes it can be: for each class, return the field types.
  1065	    For generic classes, the field types are expanded.
  1066	    If the type contains Any or a non-attrs type, returns None; in the latter case, also reports an error.
  1067	    """
  1068	    if isinstance(typ, AnyType):
  1069	        return None
  1070	    elif isinstance(typ, UnionType):
  1071	        ret: list[Mapping[str, Type]] | None = []
  1072	        for item in typ.relevant_items():
  1073	            item = get_proper_type(item)
  1074	            item_types = _get_expanded_attr_types(ctx, item, item, parent_typ)
  1075	            if ret is not None and item_types is not None:
  1076	                ret += item_types
  1077	            else:
  1078	                ret = None  # but keep iterating to emit all errors
  1079	        return ret
  1080	    elif isinstance(typ, TypeVarType):
  1081	        return _get_expanded_attr_types(
  1082	            ctx, get_proper_type(typ.upper_bound), display_typ, parent_typ
  1083	        )
  1084	    elif isinstance(typ, Instance):
  1085	        init_func = _get_attrs_init_type(typ)
  1086	        if init_func is None:
  1087	            _fail_not_attrs_class(ctx, display_typ, parent_typ)
  1088	            return None
  1089	        init_func = expand_type_by_instance(init_func, typ)
  1090	        # [1:] to skip the self argument of AttrClass.__init__
  1091	        field_names = cast(list[str], init_func.arg_names[1:])
  1092	        field_types = init_func.arg_types[1:]
  1093	        return [dict(zip(field_names, field_types))]
  1094	    else:
  1095	        _fail_not_attrs_class(ctx, display_typ, parent_typ)
  1096	        return None

  1099	def _meet_fields(types: list[Mapping[str, Type]]) -> Mapping[str, Type]:
  1100	    """
  1101	    "Meet" the fields of a list of attrs classes, i.e. for each field, its new type will be the lower bound.
  1102	    """
  1103	    field_to_types = defaultdict(list)
  1104	    for fields in types:
  1105	        for name, typ in fields.items():
  1106	            field_to_types[name].append(typ)
  1107
  1108	    return {
  1109	        name: (
  1110	            get_proper_type(reduce(meet_types, f_types))
  1111	            if len(f_types) == len(types)
  1112	            else UninhabitedType()
  1113	        )
  1114	        for name, f_types in field_to_types.items()
  1115	    }

  1118	def evolve_function_sig_callback(ctx: mypy.plugin.FunctionSigContext) -> CallableType:
  1119	    """
  1120	    Generate a signature for the 'attr.evolve' function that's specific to the call site
  1121	    and dependent on the type of the first argument.
  1122	    """
  1123	    if len(ctx.args) != 2:
  1124	        # Ideally the name and context should be callee's, but we don't have it in FunctionSigContext.
  1125	        ctx.api.fail(f'"{ctx.default_signature.name}" has unexpected type annotation', ctx.context)
  1126	        return ctx.default_signature
  1127
  1128	    if len(ctx.args[0]) != 1:
  1129	        return ctx.default_signature  # leave it to the type checker to complain
  1130
  1131	    inst_arg = ctx.args[0][0]
  1132	    inst_type = get_proper_type(ctx.api.get_expression_type(inst_arg))
  1133	    inst_type_str = format_type_bare(inst_type, ctx.api.options)
  1134
  1135	    attr_types = _get_expanded_attr_types(ctx, inst_type, inst_type, inst_type)
  1136	    if attr_types is None:
  1137	        return ctx.default_signature
  1138	    fields = _meet_fields(attr_types)

   101	def find_shallow_matching_overload_item(overload: Overloaded, call: CallExpr) -> CallableType:
   102	    """Perform limited lookup of a matching overload item.
   103
   104	    Full overload resolution is only supported during type checking, but plugins
   105	    sometimes need to resolve overloads. This can be used in some such use cases.
   106
   107	    Resolve overloads based on these things only:
   108
   109	    * Match using argument kinds and names
   110	    * If formal argument has type None, only accept the "None" expression in the callee
   111	    * If formal argument has type Literal[True] or Literal[False], only accept the
   112	      relevant bool literal
   113
   114	    Return the first matching overload item, or the last one if nothing matches.
   115	    """
   116	    for item in overload.items[:-1]:
   117	        ok = True
   118	        mapped = map_actuals_to_formals(
   119	            call.arg_kinds,
   120	            call.arg_names,
   121	            item.arg_kinds,
   122	            item.arg_names,
   123	            lambda i: AnyType(TypeOfAny.special_form),
   124	        )
   125
   126	        # Look for extra actuals
   127	        matched_actuals = set()
   128	        for actuals in mapped:
   129	            matched_actuals.update(actuals)
   130	        if any(i not in matched_actuals for i in range(len(call.args))):
   131	            ok = False
   132
   133	        for arg_type, kind, actuals in zip(item.arg_types, item.arg_kinds, mapped):
   134	            if kind.is_required() and not actuals:
   135	                # Missing required argument
   136	                ok = False
   137	                break
   138	            elif actuals:
   139	                args = [call.args[i] for i in actuals]
   140	                arg_type = get_proper_type(arg_type)
   141	                arg_none = any(isinstance(arg, NameExpr) and arg.name == "None" for arg in args)
   142	                if isinstance(arg_type, NoneType):
   143	                    if not arg_none:
   144	                        ok = False
   145	                        break
   146	                elif (
   147	                    arg_none
   148	                    and not is_overlapping_none(arg_type)
   149	                    and not (
   150	                        isinstance(arg_type, Instance)
   151	                        and arg_type.type.fullname == "builtins.object"
   152	                    )
   153	                    and not isinstance(arg_type, AnyType)
   154	                ):
   155	                    ok = False
   156	                    break
   157	                elif isinstance(arg_type, LiteralType) and isinstance(arg_type.value, bool):
   158	                    if not any(parse_bool(arg) == arg_type.value for arg in args):
   159	                        ok = False
   160	                        break
   161	        if ok:
   162	            return item
   163	    return overload.items[-1]

   369
   370	        if decorator_arguments["slots"]:
   371	            self.add_slots(info, attributes, correct_version=py_version >= (3, 10))
   372
   373	        self.reset_init_only_vars(info, attributes)
   374
   375	        if (
   376	            decorator_arguments["match_args"]
   377	            and (
   378	                "__match_args__" not in info.names or info.names["__match_args__"].plugin_generated
   379	            )
   380	            and py_version >= (3, 10)
   381	        ):
   382	            str_type = self._api.named_type("builtins.str")
   383	            literals: list[Type] = [
   384	                LiteralType(attr.name, str_type)
   385	                for attr in attributes
   386	                if attr.is_in_init and not attr.kw_only
   387	            ]
   388	            match_args_type = TupleType(literals, self._api.named_type("builtins.tuple"))
   389	            add_attribute_to_class(self._api, self._cls, "__match_args__", match_args_type)
   390
   391	        self._add_dataclass_fields_magic_attribute()
   392	        self._add_internal_replace_method(attributes)
   393	        if self._api.options.python_version >= (3, 13):
   394	            self._add_dunder_replace(attributes)
   395
   396	        if "__post_init__" in info.names:
   397	            self._add_internal_post_init_method(attributes)
   398
   399	        info.metadata["dataclass"] = {
   400	            "attributes": [attr.serialize() for attr in attributes],
   401	            "frozen": decorator_arguments["frozen"],
   402	        }
   403
   404	        return True

   449	    def add_slots(
   450	        self, info: TypeInfo, attributes: list[DataclassAttribute], *, correct_version: bool
   451	    ) -> None:
   452	        if not correct_version:
   453	            # This means that version is lower than `3.10`,
   454	            # it is just a non-existent argument for `dataclass` function.
   455	            self._api.fail(
   456	                'Keyword argument "slots" for "dataclass" is only valid in Python 3.10 and higher',
   457	                self._reason,
   458	            )
   459	            return
   460
   461	        generated_slots = {attr.name for attr in attributes}
   462	        if (info.slots is not None and info.slots != generated_slots) or info.names.get(
   463	            "__slots__"
   464	        ):
   465	            # This means we have a slots conflict.
   466	            # Class explicitly specifies a different `__slots__` field.
   467	            # And `@dataclass(slots=True)` is used.
   468	            # In runtime this raises a type error.
   469	            self._api.fail(
   470	                '"{}" both defines "__slots__" and is used with "slots=True"'.format(
   471	                    self._cls.name
   472	                ),
   473	                self._cls,
   474	            )
   475	            return
   476
   477	        if any(p.slots is None for p in info.mro[1:-1]):
   478	            # At least one type in mro (excluding `self` and `object`)
   479	            # does not have concrete `__slots__` defined. Ignoring.
   480	            return
   481
   482	        info.slots = generated_slots
   483
   484	        # Now, insert `.__slots__` attribute to class namespace:
   485	        slots_type = TupleType(
   486	            [self._api.named_type("builtins.str") for _ in generated_slots],
   487	            self._api.named_type("builtins.tuple"),
   488	        )
   489	        add_attribute_to_class(self._api, self._cls, "__slots__", slots_type)

   491	    def reset_init_only_vars(self, info: TypeInfo, attributes: list[DataclassAttribute]) -> None:
   492	        """Remove init-only vars from the class and reset init var declarations."""
   493	        for attr in attributes:
   494	            if attr.is_init_var:
   495	                if attr.name in info.names:
   496	                    del info.names[attr.name]
   497	                else:
   498	                    # Nodes of superclass InitVars not used in __init__ cannot be reached.
   499	                    assert attr.is_init_var
   500	                for stmt in info.defn.defs.body:
   501	                    if isinstance(stmt, AssignmentStmt) and stmt.unanalyzed_type:
   502	                        lvalue = stmt.lvalues[0]
   503	                        if isinstance(lvalue, NameExpr) and lvalue.name == attr.name:
   504	                            # Reset node so that another semantic analysis pass will
   505	                            # recreate a symbol node for this attribute.
   506	                            lvalue.node = None

   508	    def _get_assignment_statements_from_if_statement(
   509	        self, stmt: IfStmt
   510	    ) -> Iterator[AssignmentStmt]:
   511	        for body in stmt.body:
   512	            if not body.is_unreachable:
   513	                yield from self._get_assignment_statements_from_block(body)
   514	        if stmt.else_body is not None and not stmt.else_body.is_unreachable:
   515	            yield from self._get_assignment_statements_from_block(stmt.else_body)
   516
   517	    def _get_assignment_statements_from_block(self, block: Block) -> Iterator[AssignmentStmt]:
   518	        for stmt in block.body:
   519	            if isinstance(stmt, AssignmentStmt):
   520	                yield stmt
   521	            elif isinstance(stmt, IfStmt):
   522	                yield from self._get_assignment_statements_from_if_statement(stmt)

   524	    def collect_attributes(self) -> list[DataclassAttribute] | None:
   525	        """Collect all attributes declared in the dataclass and its parents.
   526
   527	        All assignments of the form
   528
   529	          a: SomeType
   530	          b: SomeOtherType = ...
   531
   532	        are collected.
   533
   534	        Return None if some dataclass base class hasn't been processed
   535	        yet and thus we'll need to ask for another pass.
   536	        """
   537	        cls = self._cls
   538
   539	        # First, collect attributes belonging to any class in the MRO, ignoring duplicates.
   540	        #
   541	        # We iterate through the MRO in reverse because attrs defined in the parent must appear
   542	        # earlier in the attributes list than attrs defined in the child. See:
   543	        # https://docs.python.org/3/library/dataclasses.html#inheritance
   544	        #
   545	        # However, we also want attributes defined in the subtype to override ones defined
   546	        # in the parent. We can implement this via a dict without disrupting the attr order
   547	        # because dicts preserve insertion order in Python 3.7+.
   548	        found_attrs: dict[str, DataclassAttribute] = {}
   549	        for info in reversed(cls.info.mro[1:-1]):
   550	            if "dataclass_tag" in info.metadata and "dataclass" not in info.metadata:
   551	                # We haven't processed the base class yet. Need another pass.
   552	                return None
   553	            if "dataclass" not in info.metadata:
   554	                continue
   555
   556	            # Each class depends on the set of attributes in its dataclass ancestors.
   557	            self._api.add_plugin_dependency(make_wildcard_trigger(info.fullname))
   558
   559	            for data in info.metadata["dataclass"]["attributes"]:
   560	                name: str = data["name"]
   561
   562	                attr = DataclassAttribute.deserialize(info, data, self._api)
   563	                # TODO: We shouldn't be performing type operations during the main
   564	                #       semantic analysis pass, since some TypeInfo attributes might
   565	                #       still be in flux. This should be performed in a later phase.
   566	                attr.expand_typevar_from_subtype(cls.info)
   567	                found_attrs[name] = attr
   568
   569	                sym_node = cls.info.names.get(name)
   570	                if sym_node and sym_node.node and not isinstance(sym_node.node, Var):
   571	                    self._api.fail(
   572	                        "Dataclass attribute may only be overridden by another attribute",
   573	                        sym_node.node,
   574	                    )
   575
   576	        # Second, collect attributes belonging to the current class.
   577	        current_attr_names: set[str] = set()
   578	        kw_only = self._get_bool_arg("kw_only", self._spec.kw_only_default)
   579	        for stmt in self._get_assignment_statements_from_block(cls.defs):
   580	            # Any assignment that doesn't use the new type declaration
   581	            # syntax can be ignored out of hand.
   582	            if not stmt.new_syntax:
   583	                continue
   584
   585	            # a: int, b: str = 1, 'foo' is not supported syntax so we
   586	            # don't have to worry about it.
   587	            lhs = stmt.lvalues[0]
   588	            if not isinstance(lhs, NameExpr):
   589	                continue
   590
   591	            sym = cls.info.names.get(lhs.name)
   592	            if sym is None:
   593	                # There was probably a semantic analysis error.
   594	                continue
   595
   596	            node = sym.node
   597	            assert not isinstance(node, PlaceholderNode)
   598
   599	            if isinstance(node, TypeAlias):
   600	                self._api.fail(
   601	                    ("Type aliases inside dataclass definitions are not supported at runtime"),
   602	                    node,
   603	                )
   604	                # Skip processing this node. This doesn't match the runtime behaviour,
   605	                # but the only alternative would be to modify the SymbolTable,
   606	                # and it's a little hairy to do that in a plugin.
   607	                continue
   608	            if isinstance(node, Decorator):
   609	                # This might be a property / field name clash.
   610	                # We will issue an error later.
   611	                continue
   612
   613	            assert isinstance(node, Var)
   614
   615	            # x: ClassVar[int] is ignored by dataclasses.
   616	            if node.is_classvar:
   617	                continue
   618
   619	            # x: InitVar[int] is turned into x: int and is removed from the class.
   620	            is_init_var = False
   621	            node_type = get_proper_type(node.type)
   622	            if (
   623	                isinstance(node_type, Instance)
   624	                and node_type.type.fullname == "dataclasses.InitVar"
   625	            ):
   626	                is_init_var = True
   627	                node.type = node_type.args[0]
   628
   629	            if self._is_kw_only_type(node_type):
   630	                kw_only = True
   631
   632	            has_field_call, field_args = self._collect_field_args(stmt.rvalue)
   633
   634	            is_in_init_param = field_args.get("init")
   635	            if is_in_init_param is None:
   636	                is_in_init = self._get_default_init_value_for_field_specifier(stmt.rvalue)
   637	            else:
   638	                is_in_init = bool(self._api.parse_bool(is_in_init_param))
   639
   640	            has_default = False
   641	            # Ensure that something like x: int = field() is rejected
   642	            # after an attribute with a default.
   643	            if has_field_call:
   644	                has_default = (
   645	                    "default" in field_args
   646	                    or "default_factory" in field_args
   647	                    # alias for default_factory defined in PEP 681
   648	                    or "factory" in field_args
   649	                )
   650
   651	            # All other assignments are already type checked.
   652	            elif not isinstance(stmt.rvalue, TempNode):
   653	                has_default = True
   654
   655	            if not has_default and self._spec is _TRANSFORM_SPEC_FOR_DATACLASSES:
   656	                # Make all non-default dataclass attributes implicit because they are de-facto
   657	                # set on self in the generated __init__(), not in the class body. On the other
   658	                # hand, we don't know how custom dataclass transforms initialize attributes,
   659	                # so we don't treat them as implicit. This is required to support descriptors
   660	                # (https://github.com/python/mypy/issues/14868).
   661	                sym.implicit = True
   662
   663	            is_kw_only = kw_only
   664	            # Use the kw_only field arg if it is provided. Otherwise use the
   665	            # kw_only value from the decorator parameter.
   666	            field_kw_only_param = field_args.get("kw_only")
   667	            if field_kw_only_param is not None:
   668	                value = self._api.parse_bool(field_kw_only_param)
   669	                if value is not None:
   670	                    is_kw_only = value
   671	                else:
   672	                    self._api.fail('"kw_only" argument must be a boolean literal', stmt.rvalue)
   673
   674	            if sym.type is None and node.is_final and node.is_inferred:
   675	                # This is a special case, assignment like x: Final = 42 is classified
   676	                # annotated above, but mypy strips the `Final` turning it into x = 42.
   677	                # We do not support inferred types in dataclasses, so we can try inferring
   678	                # type for simple literals, and otherwise require an explicit type
   679	                # argument for Final[...].
   680	                typ = self._api.analyze_simple_literal_type(stmt.rvalue, is_final=True)
   681	                if typ:
   682	                    node.type = typ
   683	                else:
   684	                    self._api.fail(
   685	                        "Need type argument for Final[...] with non-literal default in dataclass",
   686	                        stmt,
   687	                    )
   688	                    node.type = AnyType(TypeOfAny.from_error)
   689
   690	            alias = None
   691	            if "alias" in field_args:
   692	                alias = self._api.parse_str_literal(field_args["alias"])
   693	                if alias is None:
   694	                    self._api.fail(
   695	                        message_registry.DATACLASS_FIELD_ALIAS_MUST_BE_LITERAL,
   696	                        stmt.rvalue,
   697	                        code=errorcodes.LITERAL_REQ,
   698	                    )
   699
   700	            current_attr_names.add(lhs.name)
   701	            with state.strict_optional_set(self._api.options.strict_optional):
   702	                init_type = self._infer_dataclass_attr_init_type(sym, lhs.name, stmt)
   703	            found_attrs[lhs.name] = DataclassAttribute(
   704	                name=lhs.name,
   705	                alias=alias,
   706	                is_in_init=is_in_init,
   707	                is_init_var=is_init_var,
   708	                has_default=has_default,
   709	                line=stmt.line,
   710	                column=stmt.column,
   711	                type=init_type,
   712	                info=cls.info,
   713	                kw_only=is_kw_only,
   714	                is_neither_frozen_nor_nonfrozen=_has_direct_dataclass_transform_metaclass(
   715	                    cls.info
   716	                ),
   717	                api=self._api,
   718	            )
   719
   720	        all_attrs = list(found_attrs.values())
   721	        all_attrs.sort(key=lambda a: a.kw_only)
   722
   723	        # Third, ensure that arguments without a default don't follow
   724	        # arguments that have a default and that the KW_ONLY sentinel
   725	        # is only provided once.
   726	        found_default = False
   727	        found_kw_sentinel = False
   728	        for attr in all_attrs:
   729	            # If we find any attribute that is_in_init, not kw_only, and that
   730	            # doesn't have a default after one that does have one,
   731	            # then that's an error.
   732	            if found_default and attr.is_in_init and not attr.has_default and not attr.kw_only:
   733	                # If the issue comes from merging different classes, report it
   734	                # at the class definition point.
   735	                context: Context = cls
   736	                if attr.name in current_attr_names:
   737	                    context = Context(line=attr.line, column=attr.column)
   738	                self._api.fail(
   739	                    "Attributes without a default cannot follow attributes with one", context
   740	                )
   741
   742	            found_default = found_default or (attr.has_default and attr.is_in_init)
   743	            if found_kw_sentinel and self._is_kw_only_type(attr.type):
   744	                context = cls
   745	                if attr.name in current_attr_names:
   746	                    context = Context(line=attr.line, column=attr.column)
   747	                self._api.fail(
   748	                    "There may not be more than one field with the KW_ONLY type", context
   749	                )
   750	            found_kw_sentinel = found_kw_sentinel or self._is_kw_only_type(attr.type)
   751	        return all_attrs

   753	    def _freeze(self, attributes: list[DataclassAttribute]) -> None:
   754	        """Converts all attributes to @property methods in order to
   755	        emulate frozen classes.
   756	        """
   757	        info = self._cls.info
   758	        for attr in attributes:
   759	            # Classes that directly specify a dataclass_transform metaclass must be neither frozen
   760	            # non non-frozen per PEP681. Though it is surprising, this means that attributes from
   761	            # such a class must be writable even if the rest of the class hierarchy is frozen. This
   762	            # matches the behavior of Pyright (the reference implementation).
   763	            if attr.is_neither_frozen_nor_nonfrozen:
   764	                continue
   765
   766	            sym_node = info.names.get(attr.name)
   767	            if sym_node is not None:
   768	                var = sym_node.node
   769	                if isinstance(var, Var):
   770	                    if var.is_final:
   771	                        continue  # do not turn `Final` attrs to `@property`
   772	                    var.is_property = True
   773	            else:
   774	                var = attr.to_var(info)
   775	                var.info = info
   776	                var.is_property = True
   777	                var._fullname = info.fullname + "." + var.name
   778	                info.names[var.name] = SymbolTableNode(MDEF, var)

   780	    def _propertize_callables(
   781	        self, attributes: list[DataclassAttribute], settable: bool = True
   782	    ) -> None:
   783	        """Converts all attributes with callable types to @property methods.
   784
   785	        This avoids the typechecker getting confused and thinking that
   786	        `my_dataclass_instance.callable_attr(foo)` is going to receive a
   787	        `self` argument (it is not).
   788
   789	        """
   790	        info = self._cls.info
   791	        for attr in attributes:
   792	            if isinstance(get_proper_type(attr.type), CallableType):
   793	                var = attr.to_var(info)
   794	                var.info = info
   795	                var.is_property = True
   796	                var.is_settable_property = settable
   797	                var._fullname = info.fullname + "." + var.name
   798	                info.names[var.name] = SymbolTableNode(MDEF, var)

   800	    def _is_kw_only_type(self, node: Type | None) -> bool:
   801	        """Checks if the type of the node is the KW_ONLY sentinel value."""
   802	        if node is None:
   803	            return False
   804	        node_type = get_proper_type(node)
   805	        if not isinstance(node_type, Instance):
   806	            return False
   807	        return node_type.type.fullname == "dataclasses.KW_ONLY"

   809	    def _add_dataclass_fields_magic_attribute(self) -> None:
   810	        attr_name = "__dataclass_fields__"
   811	        any_type = AnyType(TypeOfAny.explicit)
   812	        # For `dataclasses`, use the type `dict[str, Field[Any]]` for accuracy. For dataclass
   813	        # transforms, it's inaccurate to use `Field` since a given transform may use a completely
   814	        # different type (or none); fall back to `Any` there.
   815	        #
   816	        # In either case, we're aiming to match the Typeshed stub for `is_dataclass`, which expects
   817	        # the instance to have a `__dataclass_fields__` attribute of type `dict[str, Field[Any]]`.
   818	        if self._spec is _TRANSFORM_SPEC_FOR_DATACLASSES:
   819	            field_type = self._api.named_type_or_none("dataclasses.Field", [any_type]) or any_type
   820	        else:
   821	            field_type = any_type
   822	        attr_type = self._api.named_type(
   823	            "builtins.dict", [self._api.named_type("builtins.str"), field_type]
   824	        )
   825	        var = Var(name=attr_name, type=attr_type)
   826	        var.info = self._cls.info
   827	        var._fullname = self._cls.info.fullname + "." + attr_name
   828	        var.is_classvar = True
   829	        self._cls.info.names[attr_name] = SymbolTableNode(
   830	            kind=MDEF, node=var, plugin_generated=True
   831	        )

   833	    def _collect_field_args(self, expr: Expression) -> tuple[bool, dict[str, Expression]]:
   834	        """Returns a tuple where the first value represents whether or not
   835	        the expression is a call to dataclass.field and the second is a
   836	        dictionary of the keyword arguments that field() was called with.
   837	        """
   838	        if (
   839	            isinstance(expr, CallExpr)
   840	            and isinstance(expr.callee, RefExpr)
   841	            and expr.callee.fullname in self._spec.field_specifiers
   842	        ):
   843	            # field() only takes keyword arguments.
   844	            args = {}
   845	            for name, arg, kind in zip(expr.arg_names, expr.args, expr.arg_kinds):
   846	                if not kind.is_named():
   847	                    if kind.is_named(star=True):
   848	                        # This means that `field` is used with `**` unpacking,
   849	                        # the best we can do for now is not to fail.
   850	                        # TODO: we can infer what's inside `**` and try to collect it.
   851	                        message = 'Unpacking **kwargs in "field()" is not supported'
   852	                    elif self._spec is not _TRANSFORM_SPEC_FOR_DATACLASSES:
   853	                        # dataclasses.field can only be used with keyword args, but this
   854	                        # restriction is only enforced for the *standardized* arguments to
   855	                        # dataclass_transform field specifiers. If this is not a
   856	                        # dataclasses.dataclass class, we can just skip positional args safely.
   857	                        continue
   858	                    else:
   859	                        message = '"field()" does not accept positional arguments'
   860	                    self._api.fail(message, expr)
   861	                    return True, {}
   862	                assert name is not None
   863	                args[name] = arg
   864	            return True, args
   865	        return False, {}

   867	    def _get_bool_arg(self, name: str, default: bool) -> bool:
   868	        # Expressions are always CallExprs (either directly or via a wrapper like Decorator), so
   869	        # we can use the helpers from common
   870	        if isinstance(self._reason, Expression):
   871	            return _get_decorator_bool_argument(
   872	                ClassDefContext(self._cls, self._reason, self._api), name, default
   873	            )
   874
   875	        # Subclass/metaclass use of `typing.dataclass_transform` reads the parameters from the
   876	        # class's keyword arguments (ie `class Subclass(Parent, kwarg1=..., kwarg2=...)`)
   877	        expression = self._cls.keywords.get(name)
   878	        if expression is not None:
   879	            return require_bool_literal_argument(self._api, expression, name, default)
   880	        return default

   882	    def _get_default_init_value_for_field_specifier(self, call: Expression) -> bool:
   883	        """
   884	        Find a default value for the `init` parameter of the specifier being called. If the
   885	        specifier's type signature includes an `init` parameter with a type of `Literal[True]` or
   886	        `Literal[False]`, return the appropriate boolean value from the literal. Otherwise,
   887	        fall back to the standard default of `True`.
   888	        """
   889	        if not isinstance(call, CallExpr):
   890	            return True
   891
   892	        specifier_type = _get_callee_type(call)
   893	        if specifier_type is None:
   894	            return True
   895
   896	        parameter = specifier_type.argument_by_name("init")
   897	        if parameter is None:
   898	            return True
   899
   900	        literals = try_getting_literals_from_type(parameter.typ, bool, "builtins.bool")
   901	        if literals is None or len(literals) != 1:
   902	            return True
   903
   904	        return literals[0]

   906	    def _infer_dataclass_attr_init_type(
   907	        self, sym: SymbolTableNode, name: str, context: Context
   908	    ) -> Type | None:
   909	        """Infer __init__ argument type for an attribute.
   910
   911	        In particular, possibly use the signature of __set__.
   912	        """
   913	        default = sym.type
   914	        if sym.implicit:
   915	            return default
   916	        t = get_proper_type(sym.type)
   917
   918	        # Perform a simple-minded inference from the signature of __set__, if present.
   919	        # We can't use mypy.checkmember here, since this plugin runs before type checking.
   920	        # We only support some basic scanerios here, which is hopefully sufficient for
   921	        # the vast majority of use cases.
   922	        if not isinstance(t, Instance):
   923	            return default
   924	        setter = t.type.get("__set__")
   925	        if setter:
   926	            if isinstance(setter.node, FuncDef):
   927	                super_info = t.type.get_containing_type_info("__set__")
   928	                assert super_info
   929	                if setter.type:
   930	                    setter_type = get_proper_type(
   931	                        map_type_from_supertype(setter.type, t.type, super_info)
   932	                    )
   933	                else:
   934	                    return AnyType(TypeOfAny.unannotated)
   935	                if isinstance(setter_type, CallableType) and setter_type.arg_kinds == [
   936	                    ARG_POS,
   937	                    ARG_POS,
   938	                    ARG_POS,
   939	                ]:
   940	                    return expand_type_by_instance(setter_type.arg_types[2], t)
   941	                else:
   942	                    self._api.fail(
   943	                        f'Unsupported signature for "__set__" in "{t.type.name}"', context
   944	                    )
   945	            else:
   946	                self._api.fail(f'Unsupported "__set__" in "{t.type.name}"', context)
   947
   948	        return default
