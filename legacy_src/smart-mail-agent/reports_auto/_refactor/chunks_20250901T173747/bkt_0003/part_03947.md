   107	            model_label = str(result.get("label", "unknown"))
   108	            confidence = float(result.get("score", 0.0))
   109	        except Exception as e:

   115	                "subject": subject,

   117	            }
   118
   119	        # ===== Fallback 決策：規則 > 情緒 > 低信心泛用 =====
