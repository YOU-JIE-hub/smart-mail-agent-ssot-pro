    83	                suffix = s.type.type_var_tuple_suffix
    84	                tvt = s.type.defn.type_vars[prefix]
    85	                assert isinstance(tvt, TypeVarTupleType)
    86	                fallback = tvt.tuple_fallback
    87	                s_prefix, s_middle, s_suffix = split_with_prefix_and_suffix(s.args, prefix, suffix)
    88	                t_prefix, t_middle, t_suffix = split_with_prefix_and_suffix(t.args, prefix, suffix)
    89	                s_args = s_prefix + (TupleType(list(s_middle), fallback),) + s_suffix
    90	                t_args = t_prefix + (TupleType(list(t_middle), fallback),) + t_suffix
    91	            else:
    92	                t_args = t.args
    93	                s_args = s.args
    94	            for ta, sa, type_var in zip(t_args, s_args, t.type.defn.type_vars):
    95	                ta_proper = get_proper_type(ta)
    96	                sa_proper = get_proper_type(sa)
    97	                new_type: Type | None = None
    98	                if isinstance(ta_proper, AnyType):
    99	                    new_type = AnyType(TypeOfAny.from_another_any, ta_proper)
   100	                elif isinstance(sa_proper, AnyType):
   101	                    new_type = AnyType(TypeOfAny.from_another_any, sa_proper)
   102	                elif isinstance(type_var, TypeVarType):
   103	                    if type_var.variance in (COVARIANT, VARIANCE_NOT_READY):
   104	                        new_type = join_types(ta, sa, self)
   105	                        if len(type_var.values) != 0 and new_type not in type_var.values:
   106	                            self.seen_instances.pop()
   107	                            return object_from_instance(t)
   108	                        if not is_subtype(new_type, type_var.upper_bound):
   109	                            self.seen_instances.pop()
   110	                            return object_from_instance(t)
   111	                    # TODO: contravariant case should use meet but pass seen instances as
   112	                    # an argument to keep track of recursive checks.
   113	                    elif type_var.variance in (INVARIANT, CONTRAVARIANT):
   114	                        if isinstance(ta_proper, UninhabitedType) and ta_proper.ambiguous:
   115	                            new_type = sa
   116	                        elif isinstance(sa_proper, UninhabitedType) and sa_proper.ambiguous:
   117	                            new_type = ta
   118	                        elif not is_equivalent(ta, sa):
   119	                            self.seen_instances.pop()
   120	                            return object_from_instance(t)
   121	                        else:
   122	                            # If the types are different but equivalent, then an Any is involved
   123	                            # so using a join in the contravariant case is also OK.
   124	                            new_type = join_types(ta, sa, self)
   125	                elif isinstance(type_var, TypeVarTupleType):
   126	                    new_type = get_proper_type(join_types(ta, sa, self))
   127	                    # Put the joined arguments back into instance in the normal form:
   128	                    #   a) Tuple[X, Y, Z] -> [X, Y, Z]
   129	                    #   b) tuple[X, ...] -> [*tuple[X, ...]]
   130	                    if isinstance(new_type, Instance):
   131	                        assert new_type.type.fullname == "builtins.tuple"
   132	                        new_type = UnpackType(new_type)
   133	                    else:
   134	                        assert isinstance(new_type, TupleType)
   135	                        args.extend(new_type.items)
   136	                        continue
   137	                else:
   138	                    # ParamSpec type variables behave the same, independent of variance
   139	                    if not is_equivalent(ta, sa):
   140	                        return get_proper_type(type_var.upper_bound)
   141	                    new_type = join_types(ta, sa, self)
   142	                assert new_type is not None
   143	                args.append(new_type)
   144	            result: ProperType = Instance(t.type, args)
   145	        elif t.type.bases and is_proper_subtype(
   146	            t, s, subtype_context=SubtypeContext(ignore_type_params=True)
   147	        ):
   148	            result = self.join_instances_via_supertype(t, s)
   149	        else:
   150	            # Now t is not a subtype of s, and t != s. Now s could be a subtype
   151	            # of t; alternatively, we need to find a common supertype. This works
   152	            # in of the both cases.
   153	            result = self.join_instances_via_supertype(s, t)
   154
   155	        self.seen_instances.pop()
   156	        return result
   157
   158	    def join_instances_via_supertype(self, t: Instance, s: Instance) -> ProperType:
   159	        # Give preference to joins via duck typing relationship, so that
   160	        # join(int, float) == float, for example.
   161	        for p in t.type._promote:
   162	            if is_subtype(p, s):
   163	                return join_types(p, s, self)
   164	        for p in s.type._promote:
   165	            if is_subtype(p, t):
   166	                return join_types(t, p, self)
   167
   168	        # Compute the "best" supertype of t when joined with s.
   169	        # The definition of "best" may evolve; for now it is the one with
   170	        # the longest MRO.  Ties are broken by using the earlier base.
   171
   172	        # Go over both sets of bases in case there's an explicit Protocol base. This is important
   173	        # to ensure commutativity of join (although in cases where both classes have relevant
   174	        # Protocol bases this maybe might still not be commutative)
   175	        base_types: dict[TypeInfo, None] = {}  # dict to deduplicate but preserve order
   176	        for base in t.type.bases:
   177	            base_types[base.type] = None
   178	        for base in s.type.bases:
   179	            if base.type.is_protocol and is_subtype(t, base):
   180	                base_types[base.type] = None
   181
   182	        best: ProperType | None = None
   183	        for base_type in base_types:
   184	            mapped = map_instance_to_supertype(t, base_type)
   185	            res = self.join_instances(mapped, s)
   186	            if best is None or is_better(res, best):
   187	                best = res
   188	        assert best is not None
   189	        for promote in t.type._promote:
   190	            if isinstance(promote, Instance):
   191	                res = self.join_instances(promote, s)
   192	                if is_better(res, best):
   193	                    best = res
   194	        return best

   198	    """Return one of types (expanded) if it is a supertype of other, otherwise top type."""
   199	    if is_subtype(s, t):
   200	        return t
   201	    elif is_subtype(t, s):
   202	        return s
   203	    else:
   204	        return object_or_any_from_type(get_proper_type(t))

   217	def join_types(s: Type, t: Type, instance_joiner: InstanceJoiner | None = None) -> Type:
   218	    """Return the least upper bound of s and t.
   219
   220	    For example, the join of 'int' and 'object' is 'object'.
   221	    """
   222	    if mypy.typeops.is_recursive_pair(s, t):
   223	        # This case can trigger an infinite recursion, general support for this will be
   224	        # tricky so we use a trivial join (like for protocols).
   225	        return trivial_join(s, t)
   226	    s = get_proper_type(s)
   227	    t = get_proper_type(t)
   228
   229	    if (s.can_be_true, s.can_be_false) != (t.can_be_true, t.can_be_false):
   230	        # if types are restricted in different ways, use the more general versions
   231	        s = mypy.typeops.true_or_false(s)
   232	        t = mypy.typeops.true_or_false(t)
   233
   234	    if isinstance(s, UnionType) and not isinstance(t, UnionType):
   235	        s, t = t, s
   236
   237	    if isinstance(s, AnyType):
   238	        return s
   239
   240	    if isinstance(s, ErasedType):
   241	        return t
   242
   243	    if isinstance(s, NoneType) and not isinstance(t, NoneType):
   244	        s, t = t, s
   245
   246	    if isinstance(s, UninhabitedType) and not isinstance(t, UninhabitedType):
   247	        s, t = t, s
   248
   249	    # Meets/joins require callable type normalization.
   250	    s, t = normalize_callables(s, t)
   251
   252	    # Use a visitor to handle non-trivial cases.
   253	    return t.accept(TypeJoinVisitor(s, instance_joiner))

   339	    def visit_instance(self, t: Instance) -> ProperType:
   340	        if isinstance(self.s, Instance):
   341	            if self.instance_joiner is None:
   342	                self.instance_joiner = InstanceJoiner()
   343	            nominal = self.instance_joiner.join_instances(t, self.s)
   344	            structural: Instance | None = None
   345	            if t.type.is_protocol and is_protocol_implementation(self.s, t):
   346	                structural = t
   347	            elif self.s.type.is_protocol and is_protocol_implementation(t, self.s):
   348	                structural = self.s

   350	            # structural and nominal and they have same MRO length (see two comments
   351	            # in join_instances_via_supertype). Otherwise, just return the nominal join.
   352	            if not structural or is_better(nominal, structural):
   353	                return nominal
   354	            return structural
   355	        elif isinstance(self.s, FunctionLike):
   356	            if t.type.is_protocol:
   357	                call = unpack_callback_protocol(t)
   358	                if call:
   359	                    return join_types(call, self.s)
   360	            return join_types(t, self.s.fallback)
   361	        elif isinstance(self.s, TypeType):
   362	            return join_types(t, self.s)
   363	        elif isinstance(self.s, TypedDictType):
   364	            return join_types(t, self.s)
   365	        elif isinstance(self.s, TupleType):
   366	            return join_types(t, self.s)
   367	        elif isinstance(self.s, LiteralType):
   368	            return join_types(t, self.s)
   369	        elif isinstance(self.s, TypeVarTupleType) and is_subtype(self.s.upper_bound, t):
   370	            return t
   371	        else:
   372	            return self.default(self.s)

   374	    def visit_callable_type(self, t: CallableType) -> ProperType:
   375	        if isinstance(self.s, CallableType) and is_similar_callables(t, self.s):
   376	            if is_equivalent(t, self.s):
   377	                return combine_similar_callables(t, self.s)
   378	            result = join_similar_callables(t, self.s)
   379	            # We set the from_type_type flag to suppress error when a collection of
   380	            # concrete class objects gets inferred as their common abstract superclass.
   381	            if not (
   382	                (t.is_type_obj() and t.type_object().is_abstract)
   383	                or (self.s.is_type_obj() and self.s.type_object().is_abstract)
   384	            ):
   385	                result.from_type_type = True
   386	            if any(
   387	                isinstance(tp, (NoneType, UninhabitedType))
   388	                for tp in get_proper_types(result.arg_types)
   389	            ):
   390	                # We don't want to return unusable Callable, attempt fallback instead.
   391	                return join_types(t.fallback, self.s)
   392	            return result
   393	        elif isinstance(self.s, Overloaded):
   394	            # Switch the order of arguments to that we'll get to visit_overloaded.
   395	            return join_types(t, self.s)
   396	        elif isinstance(self.s, Instance) and self.s.type.is_protocol:
   397	            call = unpack_callback_protocol(self.s)
   398	            if call:
   399	                return join_types(t, call)
   400	        return join_types(t.fallback, self.s)

   402	    def visit_overloaded(self, t: Overloaded) -> ProperType:
   403	        # This is more complex than most other cases. Here are some
   404	        # examples that illustrate how this works.
   405	        #
   406	        # First let's define a concise notation:
   407	        #  - Cn are callable types (for n in 1, 2, ...)
   408	        #  - Ov(C1, C2, ...) is an overloaded type with items C1, C2, ...
   409	        #  - Callable[[T, ...], S] is written as [T, ...] -> S.
   410	        #
   411	        # We want some basic properties to hold (assume Cn are all
   412	        # unrelated via Any-similarity):
   413	        #
   414	        #   join(Ov(C1, C2), C1) == C1
   415	        #   join(Ov(C1, C2), Ov(C1, C2)) == Ov(C1, C2)
   416	        #   join(Ov(C1, C2), Ov(C1, C3)) == C1
   417	        #   join(Ov(C2, C2), C3) == join of fallback types
   418	        #
   419	        # The presence of Any types makes things more interesting. The join is the
   420	        # most general type we can get with respect to Any:
   421	        #
   422	        #   join(Ov([int] -> int, [str] -> str), [Any] -> str) == Any -> str
   423	        #
   424	        # We could use a simplification step that removes redundancies, but that's not
   425	        # implemented right now. Consider this example, where we get a redundancy:
   426	        #
   427	        #   join(Ov([int, Any] -> Any, [str, Any] -> Any), [Any, int] -> Any) ==
   428	        #       Ov([Any, int] -> Any, [Any, int] -> Any)
   429	        #
   430	        # TODO: Consider more cases of callable subtyping.
   431	        result: list[CallableType] = []
   432	        s = self.s
   433	        if isinstance(s, FunctionLike):
   434	            # The interesting case where both types are function types.
   435	            for t_item in t.items:
   436	                for s_item in s.items:
   437	                    if is_similar_callables(t_item, s_item):
   438	                        if is_equivalent(t_item, s_item):
   439	                            result.append(combine_similar_callables(t_item, s_item))
   440	                        elif is_subtype(t_item, s_item):
   441	                            result.append(s_item)
   442	            if result:
   443	                # TODO: Simplify redundancies from the result.
   444	                if len(result) == 1:
   445	                    return result[0]
   446	                else:
   447	                    return Overloaded(result)
   448	            return join_types(t.fallback, s.fallback)
   449	        elif isinstance(s, Instance) and s.type.is_protocol:
   450	            call = unpack_callback_protocol(s)
   451	            if call:
   452	                return join_types(t, call)
   453	        return join_types(t.fallback, s)

   455	    def join_tuples(self, s: TupleType, t: TupleType) -> list[Type] | None:
   456	        """Join two tuple types while handling variadic entries.
   457
   458	        This is surprisingly tricky, and we don't handle some tricky corner cases.
   459	        Most of the trickiness comes from the variadic tuple items like *tuple[X, ...]
   460	        since they can have arbitrary partial overlaps (while *Ts can't be split).
   461	        """
   462	        s_unpack_index = find_unpack_in_list(s.items)
   463	        t_unpack_index = find_unpack_in_list(t.items)
   464	        if s_unpack_index is None and t_unpack_index is None:
   465	            if s.length() == t.length():
   466	                items: list[Type] = []
   467	                for i in range(t.length()):
   468	                    items.append(join_types(t.items[i], s.items[i]))
   469	                return items
   470	            return None
   471	        if s_unpack_index is not None and t_unpack_index is not None:
   472	            # The most complex case: both tuples have an unpack item.
   473	            s_unpack = s.items[s_unpack_index]
   474	            assert isinstance(s_unpack, UnpackType)
   475	            s_unpacked = get_proper_type(s_unpack.type)
   476	            t_unpack = t.items[t_unpack_index]
   477	            assert isinstance(t_unpack, UnpackType)
   478	            t_unpacked = get_proper_type(t_unpack.type)
   479	            if s.length() == t.length() and s_unpack_index == t_unpack_index:
   480	                # We can handle a case where arity is perfectly aligned, e.g.
   481	                # join(Tuple[X1, *tuple[Y1, ...], Z1], Tuple[X2, *tuple[Y2, ...], Z2]).
   482	                # We can essentially perform the join elementwise.
   483	                prefix_len = t_unpack_index
   484	                suffix_len = t.length() - t_unpack_index - 1
   485	                items = []
   486	                for si, ti in zip(s.items[:prefix_len], t.items[:prefix_len]):
   487	                    items.append(join_types(si, ti))
   488	                joined = join_types(s_unpacked, t_unpacked)
   489	                if isinstance(joined, TypeVarTupleType):
   490	                    items.append(UnpackType(joined))
   491	                elif isinstance(joined, Instance) and joined.type.fullname == "builtins.tuple":
   492	                    items.append(UnpackType(joined))
   493	                else:
   494	                    if isinstance(t_unpacked, Instance):
   495	                        assert t_unpacked.type.fullname == "builtins.tuple"
   496	                        tuple_instance = t_unpacked
   497	                    else:
   498	                        assert isinstance(t_unpacked, TypeVarTupleType)
   499	                        tuple_instance = t_unpacked.tuple_fallback
   500	                    items.append(
   501	                        UnpackType(
   502	                            tuple_instance.copy_modified(
   503	                                args=[object_from_instance(tuple_instance)]
   504	                            )
   505	                        )
   506	                    )
   507	                if suffix_len:
   508	                    for si, ti in zip(s.items[-suffix_len:], t.items[-suffix_len:]):
   509	                        items.append(join_types(si, ti))
   510	                return items
   511	            if s.length() == 1 or t.length() == 1:
   512	                # Another case we can handle is when one of tuple is purely variadic
   513	                # (i.e. a non-normalized form of tuple[X, ...]), in this case the join
   514	                # will be again purely variadic.
   515	                if not (isinstance(s_unpacked, Instance) and isinstance(t_unpacked, Instance)):
   516	                    return None
   517	                assert s_unpacked.type.fullname == "builtins.tuple"
   518	                assert t_unpacked.type.fullname == "builtins.tuple"
   519	                mid_joined = join_types(s_unpacked.args[0], t_unpacked.args[0])
   520	                t_other = [a for i, a in enumerate(t.items) if i != t_unpack_index]
   521	                s_other = [a for i, a in enumerate(s.items) if i != s_unpack_index]
   522	                other_joined = join_type_list(s_other + t_other)
   523	                mid_joined = join_types(mid_joined, other_joined)
   524	                return [UnpackType(s_unpacked.copy_modified(args=[mid_joined]))]
   525	            # TODO: are there other case we can handle (e.g. both prefix/suffix are shorter)?
   526	            return None
   527	        if s_unpack_index is not None:
   528	            variadic = s
   529	            unpack_index = s_unpack_index
   530	            fixed = t
   531	        else:
   532	            assert t_unpack_index is not None
   533	            variadic = t
   534	            unpack_index = t_unpack_index
   535	            fixed = s
   536	        # Case where one tuple has variadic item and the other one doesn't. The join will
   537	        # be variadic, since fixed tuple is a subtype of variadic, but not vice versa.
   538	        unpack = variadic.items[unpack_index]
   539	        assert isinstance(unpack, UnpackType)
   540	        unpacked = get_proper_type(unpack.type)
   541	        if not isinstance(unpacked, Instance):
   542	            return None
   543	        if fixed.length() < variadic.length() - 1:
   544	            # There are no non-trivial types that are supertype of both.
   545	            return None
   546	        prefix_len = unpack_index
   547	        suffix_len = variadic.length() - prefix_len - 1
   548	        prefix, middle, suffix = split_with_prefix_and_suffix(
   549	            tuple(fixed.items), prefix_len, suffix_len
   550	        )
   551	        items = []
   552	        for fi, vi in zip(prefix, variadic.items[:prefix_len]):
   553	            items.append(join_types(fi, vi))
   554	        mid_joined = join_type_list(list(middle))
   555	        mid_joined = join_types(mid_joined, unpacked.args[0])
   556	        items.append(UnpackType(unpacked.copy_modified(args=[mid_joined])))
   557	        if suffix_len:
   558	            for fi, vi in zip(suffix, variadic.items[-suffix_len:]):
   559	                items.append(join_types(fi, vi))
   560	        return items

   562	    def visit_tuple_type(self, t: TupleType) -> ProperType:
   563	        # When given two fixed-length tuples:
   564	        # * If they have the same length, join their subtypes item-wise:
   565	        #   Tuple[int, bool] + Tuple[bool, bool] becomes Tuple[int, bool]
   566	        # * If lengths do not match, return a variadic tuple:
   567	        #   Tuple[bool, int] + Tuple[bool] becomes Tuple[int, ...]
   568	        #
   569	        # Otherwise, `t` is a fixed-length tuple but `self.s` is NOT:
   570	        # * Joining with a variadic tuple returns variadic tuple:
   571	        #   Tuple[int, bool] + Tuple[bool, ...] becomes Tuple[int, ...]
   572	        # * Joining with any Sequence also returns a Sequence:
   573	        #   Tuple[int, bool] + List[bool] becomes Sequence[int]

   591	                    return t

    61	# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
    62	#
    63	# Mypy uses the term "literal" to refer to any expression built out of
    64	# the following:
    65	#
    66	# * Plain literal expressions, like `1` (integer, float, string, etc.)
    67	#
    68	# * Compound literal expressions, like `(lit1, lit2)` (list, dict,
    69	#   set, or tuple)
    70	#
    71	# * Operator expressions, like `lit1 + lit2`
    72	#
    73	# * Variable references, like `x`
    74	#
    75	# * Member references, like `lit.m`
    76	#
    77	# * Index expressions, like `lit[0]`
    78	#
    79	# A typical "literal" looks like `x[(i,j+1)].m`.
    80	#
    81	# An expression that is a literal has a `literal_hash`, with the
    82	# following properties.
    83	#
    84	# * `literal_hash` is a Key: a tuple containing basic data types and
    85	#   possibly other Keys. So it can be used as a key in a dictionary
    86	#   that will be compared by value (as opposed to the Node itself,
    87	#   which is compared by identity).
    88	#
    89	# * Two expressions have equal `literal_hash`es if and only if they
    90	#   are syntactically equal expressions. (NB: Actually, we also
    91	#   identify as equal expressions like `3` and `3.0`; is this a good
    92	#   idea?)
    93	#
    94	# * The elements of `literal_hash` that are tuples are exactly the
    95	#   subexpressions of the original expression (e.g. the base and index
    96	#   of an index expression, or the operands of an operator expression).
    97

   117	def literal(e: Expression) -> int:
   118	    """Return the literal kind for an expression."""
   119
   120	    if isinstance(e, ComparisonExpr):
   121	        return min(literal(o) for o in e.operands)
   122
   123	    elif isinstance(e, OpExpr):
   124	        return min(literal(e.left), literal(e.right))
   125
   126	    elif isinstance(e, (MemberExpr, UnaryExpr, StarExpr)):
   127	        return literal(e.expr)
   128
   129	    elif isinstance(e, AssignmentExpr):
   130	        return literal(e.target)
   131
   132	    elif isinstance(e, IndexExpr):
   133	        if literal(e.index) == LITERAL_YES:
   134	            return literal(e.base)
   135	        else:
   136	            return LITERAL_NO

   138	    elif isinstance(e, NameExpr):
   139	        if isinstance(e.node, Var) and e.node.is_final and e.node.final_value is not None:
   140	            return LITERAL_YES
   141	        return LITERAL_TYPE

   143	    if isinstance(e, (IntExpr, FloatExpr, ComplexExpr, StrExpr, BytesExpr)):
   144	        return LITERAL_YES

   146	    if literal_hash(e):

   199	    def visit_comparison_expr(self, e: ComparisonExpr) -> Key:
   200	        rest: tuple[str | Key | None, ...] = tuple(e.operators)
   201	        rest += tuple(literal_hash(o) for o in e.operands)
   202	        return ("Comparison",) + rest

   204	    def visit_unary_expr(self, e: UnaryExpr) -> Key:

   207	    def seq_expr(self, e: ListExpr | TupleExpr | SetExpr, name: str) -> Key | None:
   208	        if all(literal(x) == LITERAL_YES for x in e.items):
   209	            rest: tuple[Key | None, ...] = tuple(literal_hash(x) for x in e.items)
   210	            return (name,) + rest

   213	    def visit_list_expr(self, e: ListExpr) -> Key | None:

   218	            rest: tuple[Key | None, ...] = tuple(
   219	                (literal_hash(a) if a else None, literal_hash(b)) for a, b in e.items
