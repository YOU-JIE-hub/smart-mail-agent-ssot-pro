   358	    def visit_name_expr(self, e: NameExpr) -> Type:
   359	        """Type check a name expression.
   360
   361	        It can be of any kind: local, member or global.
   362	        """
   363	        self.chk.module_refs.update(extract_refexpr_names(e))
   364	        result = self.analyze_ref_expr(e)
   365	        narrowed = self.narrow_type_from_binder(e, result)
   366	        self.chk.check_deprecated(e.node, e)
   367	        return narrowed

   369	    def analyze_ref_expr(self, e: RefExpr, lvalue: bool = False) -> Type:
   370	        result: Type | None = None
   371	        node = e.node
   372
   373	        if isinstance(e, NameExpr) and e.is_special_form:
   374	            # A special form definition, nothing to check here.
   375	            return AnyType(TypeOfAny.special_form)
   376
   377	        if isinstance(node, Var):
   378	            # Variable reference.
   379	            result = self.analyze_var_ref(node, e)
   380	            if isinstance(result, PartialType):
   381	                result = self.chk.handle_partial_var_type(result, lvalue, node, e)
   382	        elif isinstance(node, Decorator):
   383	            result = self.analyze_var_ref(node.var, e)
   384	        elif isinstance(node, OverloadedFuncDef):
   385	            if node.type is None:
   386	                if self.chk.in_checked_function() and node.items:
   387	                    self.chk.handle_cannot_determine_type(node.name, e)
   388	                result = AnyType(TypeOfAny.from_error)
   389	            else:
   390	                result = node.type
   391	        elif isinstance(node, (FuncDef, TypeInfo, TypeAlias, MypyFile, TypeVarLikeExpr)):
   392	            result = self.analyze_static_reference(node, e, e.is_alias_rvalue or lvalue)
   393	        else:
   394	            if isinstance(node, PlaceholderNode):
   395	                assert False, f"PlaceholderNode {node.fullname!r} leaked to checker"
   396	            # Unknown reference; use any type implicitly to avoid
   397	            # generating extra type errors.
   398	            result = AnyType(TypeOfAny.from_error)
   399	        if isinstance(node, TypeInfo):
   400	            if isinstance(result, CallableType) and isinstance(  # type: ignore[misc]
   401	                result.ret_type, Instance
   402	            ):
   403	                # We need to set correct line and column
   404	                # TODO: always do this in type_object_type by passing the original context
   405	                result.ret_type.line = e.line
   406	                result.ret_type.column = e.column
   407	            if is_type_type_context(self.type_context[-1]):
   408	                # This is the type in a type[] expression, so substitute type
   409	                # variables with Any.
   410	                result = erasetype.erase_typevars(result)
   411	        assert result is not None
   412	        return result

   414	    def analyze_static_reference(
   415	        self,
   416	        node: SymbolNode,
   417	        ctx: Context,
   418	        is_lvalue: bool,
   419	        *,
   420	        include_modules: bool = True,
   421	        suppress_errors: bool = False,
   422	    ) -> Type:
   423	        """
   424	        This is the version of analyze_ref_expr() that doesn't do any deferrals.
   425
   426	        This function can be used by member access to "static" attributes. For example,
   427	        when accessing module attributes in protocol checks, or accessing attributes of
   428	        special kinds (like TypeAlias, TypeInfo, etc.) on an instance or class object.
   429	        # TODO: merge with analyze_ref_expr() when we are confident about performance.
   430	        """
   431	        if isinstance(node, (Var, Decorator, OverloadedFuncDef)):
   432	            return node.type or AnyType(TypeOfAny.special_form)
   433	        elif isinstance(node, FuncDef):
   434	            return function_type(node, self.named_type("builtins.function"))
   435	        elif isinstance(node, TypeInfo):
   436	            # Reference to a type object.
   437	            if node.typeddict_type:
   438	                # We special-case TypedDict, because they don't define any constructor.
   439	                return self.typeddict_callable(node)
   440	            elif node.fullname == "types.NoneType":
   441	                # We special case NoneType, because its stub definition is not related to None.
   442	                return TypeType(NoneType())
   443	            else:
   444	                return type_object_type(node, self.named_type)
   445	        elif isinstance(node, TypeAlias):
   446	            # Something that refers to a type alias appears in runtime context.
   447	            # Note that we suppress bogus errors for alias redefinitions,
   448	            # they are already reported in semanal.py.
   449	            with self.msg.filter_errors() if suppress_errors else nullcontext():
   450	                return self.alias_type_in_runtime_context(
   451	                    node, ctx=ctx, alias_definition=is_lvalue
   452	                )
   453	        elif isinstance(node, TypeVarExpr):
   454	            return self.named_type("typing.TypeVar")
   455	        elif isinstance(node, (ParamSpecExpr, TypeVarTupleExpr)):
   456	            return self.object_type()
   457	        elif isinstance(node, MypyFile):
   458	            # Reference to a module object.
   459	            return self.module_type(node) if include_modules else AnyType(TypeOfAny.special_form)
   460	        return AnyType(TypeOfAny.from_error)

   462	    def analyze_var_ref(self, var: Var, context: Context) -> Type:
   463	        if var.type:
   464	            var_type = get_proper_type(var.type)
   465	            if isinstance(var_type, Instance):
   466	                if var.fullname == "typing.Any":
   467	                    # The typeshed type is 'object'; give a more useful type in runtime context
   468	                    return self.named_type("typing._SpecialForm")
   469	                if self.is_literal_context() and var_type.last_known_value is not None:
   470	                    return var_type.last_known_value
   471	                if var.name in {"True", "False"}:
   472	                    return self.infer_literal_expr_type(var.name == "True", "builtins.bool")
   473	            return var.type
   474	        else:
   475	            if not var.is_ready and self.chk.in_checked_function():
   476	                self.chk.handle_cannot_determine_type(var.name, context)
   477	            # Implicit 'Any' type.
   478	            return AnyType(TypeOfAny.special_form)

   480	    def module_type(self, node: MypyFile) -> Instance:
   481	        try:
   482	            result = self.named_type("types.ModuleType")
   483	        except KeyError:
   484	            # In test cases might 'types' may not be available.
   485	            # Fall back to a dummy 'object' type instead to
   486	            # avoid a crash.
   487	            result = self.named_type("builtins.object")
   488	        module_attrs: dict[str, Type] = {}
   489	        immutable = set()
   490	        for name, n in node.names.items():
   491	            if not n.module_public:
   492	                continue
   493	            if isinstance(n.node, Var) and n.node.is_final:
   494	                immutable.add(name)
   495	            if n.node is None:
   496	                module_attrs[name] = AnyType(TypeOfAny.from_error)
   497	            else:
   498	                # TODO: what to do about nested module references?
   499	                # They are non-trivial because there may be import cycles.
   500	                module_attrs[name] = self.analyze_static_reference(
   501	                    n.node, n.node, False, include_modules=False, suppress_errors=True
   502	                )
   503	        result.extra_attrs = ExtraAttrs(module_attrs, immutable, node.fullname)
   504	        return result

   506	    def visit_call_expr(self, e: CallExpr, allow_none_return: bool = False) -> Type:
   507	        """Type check a call expression."""
   508	        if e.analyzed:
   509	            if isinstance(e.analyzed, NamedTupleExpr) and not e.analyzed.is_typed:
   510	                # Type check the arguments, but ignore the results. This relies
   511	                # on the typeshed stubs to type check the arguments.
   512	                self.visit_call_expr_inner(e)
   513	            # It's really a special form that only looks like a call.
   514	            return self.accept(e.analyzed, self.type_context[-1])
   515	        return self.visit_call_expr_inner(e, allow_none_return=allow_none_return)

   517	    def refers_to_typeddict(self, base: Expression) -> bool:
   518	        if not isinstance(base, RefExpr):
   519	            return False
   520	        if isinstance(base.node, TypeInfo) and base.node.typeddict_type is not None:
   521	            # Direct reference.
   522	            return True
   523	        return isinstance(base.node, TypeAlias) and isinstance(
   524	            get_proper_type(base.node.target), TypedDictType
   525	        )

   527	    def visit_call_expr_inner(self, e: CallExpr, allow_none_return: bool = False) -> Type:
   528	        if (
   529	            self.refers_to_typeddict(e.callee)
   530	            or isinstance(e.callee, IndexExpr)
   531	            and self.refers_to_typeddict(e.callee.base)
   532	        ):
   533	            typeddict_callable = get_proper_type(self.accept(e.callee, is_callee=True))
   534	            if isinstance(typeddict_callable, CallableType):
   535	                typeddict_type = get_proper_type(typeddict_callable.ret_type)
   536	                assert isinstance(typeddict_type, TypedDictType)
   537	                return self.check_typeddict_call(
   538	                    typeddict_type, e.arg_kinds, e.arg_names, e.args, e, typeddict_callable
   539	                )
   540	        if (
   541	            isinstance(e.callee, NameExpr)
   542	            and e.callee.name in ("isinstance", "issubclass")
   543	            and len(e.args) == 2
   544	        ):
   545	            for typ in mypy.checker.flatten(e.args[1]):
   546	                node = None
   547	                if isinstance(typ, NameExpr):
   548	                    try:
   549	                        node = self.chk.lookup_qualified(typ.name)
   550	                    except KeyError:
   551	                        # Undefined names should already be reported in semantic analysis.
   552	                        pass
   553	                if is_expr_literal_type(typ):
   554	                    self.msg.cannot_use_function_with_type(e.callee.name, "Literal", e)
   555	                    continue
   556	                if node and isinstance(node.node, TypeAlias):
   557	                    target = get_proper_type(node.node.target)
   558	                    if isinstance(target, AnyType):
   559	                        self.msg.cannot_use_function_with_type(e.callee.name, "Any", e)
   560	                        continue
   561	                    if isinstance(target, NoneType):
   562	                        continue
   563	                if (
   564	                    isinstance(typ, IndexExpr)
   565	                    and isinstance(typ.analyzed, (TypeApplication, TypeAliasExpr))
   566	                ) or (
   567	                    isinstance(typ, NameExpr)
   568	                    and node
   569	                    and isinstance(node.node, TypeAlias)
   570	                    and not node.node.no_args
   571	                    and not (
   572	                        isinstance(union_target := get_proper_type(node.node.target), UnionType)
   573	                        and union_target.uses_pep604_syntax
   574	                    )
   575	                ):
   576	                    self.msg.type_arguments_not_allowed(e)
   577	                if isinstance(typ, RefExpr) and isinstance(typ.node, TypeInfo):
   578	                    if typ.node.typeddict_type:
   579	                        self.msg.cannot_use_function_with_type(e.callee.name, "TypedDict", e)
   580	                    elif typ.node.is_newtype:
   581	                        self.msg.cannot_use_function_with_type(e.callee.name, "NewType", e)
   582	        self.try_infer_partial_type(e)
   583	        type_context = None
   584	        if isinstance(e.callee, LambdaExpr):
   585	            formal_to_actual = map_actuals_to_formals(
   586	                e.arg_kinds,
   587	                e.arg_names,
   588	                e.callee.arg_kinds,
   589	                e.callee.arg_names,
   590	                lambda i: self.accept(e.args[i]),
   591	            )
   592
   593	            arg_types = [
   594	                join.join_type_list([self.accept(e.args[j]) for j in formal_to_actual[i]])
   595	                for i in range(len(e.callee.arg_kinds))
   596	            ]
   597	            type_context = CallableType(
   598	                arg_types,
   599	                e.callee.arg_kinds,
   600	                e.callee.arg_names,
   601	                ret_type=self.object_type(),
   602	                fallback=self.named_type("builtins.function"),
   603	            )
   604	        callee_type = get_proper_type(
   605	            self.accept(e.callee, type_context, always_allow_any=True, is_callee=True)
   606	        )
   607
   608	        # Figure out the full name of the callee for plugin lookup.
   609	        object_type = None
   610	        member = None
   611	        fullname = None
   612	        if isinstance(e.callee, RefExpr):
   613	            # There are two special cases where plugins might act:
   614	            # * A "static" reference/alias to a class or function;
   615	            #   get_function_hook() will be invoked for these.
   616	            fullname = e.callee.fullname or None
   617	            if isinstance(e.callee.node, TypeAlias):
   618	                target = get_proper_type(e.callee.node.target)
   619	                if isinstance(target, Instance):
   620	                    fullname = target.type.fullname
   621	            # * Call to a method on object that has a full name (see
   622	            #   method_fullname() for details on supported objects);
   623	            #   get_method_hook() and get_method_signature_hook() will
   624	            #   be invoked for these.
   625	            if (
   626	                not fullname
   627	                and isinstance(e.callee, MemberExpr)
   628	                and self.chk.has_type(e.callee.expr)
   629	            ):
   630	                member = e.callee.name
   631	                object_type = self.chk.lookup_type(e.callee.expr)
   632
   633	        if (
   634	            self.chk.options.disallow_untyped_calls
   635	            and self.chk.in_checked_function()
   636	            and isinstance(callee_type, CallableType)
   637	            and callee_type.implicit
   638	            and callee_type.name != LAMBDA_NAME
   639	        ):
   640	            if fullname is None and member is not None:
   641	                assert object_type is not None
   642	                fullname = self.method_fullname(object_type, member)
   643	            if not fullname or not any(
   644	                fullname == p or fullname.startswith(f"{p}.")
   645	                for p in self.chk.options.untyped_calls_exclude
   646	            ):
   647	                self.msg.untyped_function_call(callee_type, e)
   648
   649	        ret_type = self.check_call_expr_with_callee_type(
   650	            callee_type, e, fullname, object_type, member
   651	        )
   652	        if isinstance(e.callee, RefExpr) and len(e.args) == 2:
   653	            if e.callee.fullname in ("builtins.isinstance", "builtins.issubclass"):
   654	                self.check_runtime_protocol_test(e)
   655	            if e.callee.fullname == "builtins.issubclass":
   656	                self.check_protocol_issubclass(e)
   657	        if isinstance(e.callee, MemberExpr) and e.callee.name == "format":
   658	            self.check_str_format_call(e)
   659	        ret_type = get_proper_type(ret_type)
   660	        if isinstance(ret_type, UnionType):
   661	            ret_type = make_simplified_union(ret_type.items)
   662	        if isinstance(ret_type, UninhabitedType) and not ret_type.ambiguous:
   663	            self.chk.binder.unreachable()
   664	        # Warn on calls to functions that always return None. The check
   665	        # of ret_type is both a common-case optimization and prevents reporting
   666	        # the error in dynamic functions (where it will be Any).
   667	        if (
   668	            not allow_none_return
   669	            and isinstance(ret_type, NoneType)
   670	            and self.always_returns_none(e.callee)
   671	        ):
   672	            self.chk.msg.does_not_return_value(callee_type, e)
   673	            return AnyType(TypeOfAny.from_error)
   674	        return ret_type

   676	    def check_str_format_call(self, e: CallExpr) -> None:
   677	        """More precise type checking for str.format() calls on literals."""
   678	        assert isinstance(e.callee, MemberExpr)
   679	        format_value = None
   680	        if isinstance(e.callee.expr, StrExpr):
   681	            format_value = e.callee.expr.value
   682	        elif self.chk.has_type(e.callee.expr):
   683	            typ = get_proper_type(self.chk.lookup_type(e.callee.expr))
   684	            if (
   685	                isinstance(typ, Instance)
   686	                and typ.type.is_enum
   687	                and isinstance(typ.last_known_value, LiteralType)
   688	                and isinstance(typ.last_known_value.value, str)
   689	            ):
   690	                value_type = typ.type.names[typ.last_known_value.value].type
   691	                if isinstance(value_type, Type):
   692	                    typ = get_proper_type(value_type)
   693	            base_typ = try_getting_literal(typ)
   694	            if isinstance(base_typ, LiteralType) and isinstance(base_typ.value, str):
   695	                format_value = base_typ.value
   696	        if format_value is not None:
   697	            self.strfrm_checker.check_str_format_call(e, format_value)

   699	    def method_fullname(self, object_type: Type, method_name: str) -> str | None:
   700	        """Convert a method name to a fully qualified name, based on the type of the object that
   701	        it is invoked on. Return `None` if the name of `object_type` cannot be determined.
   702	        """
   703	        object_type = get_proper_type(object_type)
   704
   705	        if isinstance(object_type, CallableType) and object_type.is_type_obj():
   706	            # For class method calls, object_type is a callable representing the class object.
   707	            # We "unwrap" it to a regular type, as the class/instance method difference doesn't
   708	            # affect the fully qualified name.
   709	            object_type = get_proper_type(object_type.ret_type)
   710	        elif isinstance(object_type, TypeType):
   711	            object_type = object_type.item
   712
   713	        type_name = None
   714	        if isinstance(object_type, Instance):
   715	            type_name = object_type.type.fullname
   716	        elif isinstance(object_type, (TypedDictType, LiteralType)):
   717	            info = object_type.fallback.type.get_containing_type_info(method_name)
   718	            type_name = info.fullname if info is not None else None
   719	        elif isinstance(object_type, TupleType):
   720	            type_name = tuple_fallback(object_type).type.fullname
   721
   722	        if type_name:
   723	            return f"{type_name}.{method_name}"
   724	        else:
   725	            return None

   727	    def always_returns_none(self, node: Expression) -> bool:
   728	        """Check if `node` refers to something explicitly annotated as only returning None."""
   729	        if isinstance(node, RefExpr):
   730	            if self.defn_returns_none(node.node):
   731	                return True
   732	        if isinstance(node, MemberExpr) and node.node is None:  # instance or class attribute
   733	            typ = get_proper_type(self.chk.lookup_type(node.expr))
   734	            if isinstance(typ, Instance):
   735	                info = typ.type
   736	            elif isinstance(typ, CallableType) and typ.is_type_obj():
   737	                ret_type = get_proper_type(typ.ret_type)
   738	                if isinstance(ret_type, Instance):
   739	                    info = ret_type.type
   740	                else:
   741	                    return False
   742	            else:
   743	                return False
   744	            sym = info.get(node.name)
   745	            if sym and self.defn_returns_none(sym.node):
   746	                return True
   747	        return False

   749	    def defn_returns_none(self, defn: SymbolNode | None) -> bool:
   750	        """Check if `defn` can _only_ return None."""
   751	        if isinstance(defn, FuncDef):
   752	            return isinstance(defn.type, CallableType) and isinstance(
   753	                get_proper_type(defn.type.ret_type), NoneType
   754	            )
   755	        if isinstance(defn, OverloadedFuncDef):
   756	            return all(self.defn_returns_none(item) for item in defn.items)
   757	        if isinstance(defn, Var):
   758	            typ = get_proper_type(defn.type)
   759	            if (
   760	                not defn.is_inferred
   761	                and isinstance(typ, CallableType)
   762	                and isinstance(get_proper_type(typ.ret_type), NoneType)
   763	            ):
   764	                return True
   765	            if isinstance(typ, Instance):
   766	                sym = typ.type.get("__call__")
   767	                if sym and self.defn_returns_none(sym.node):
   768	                    return True
   769	        return False

   771	    def check_runtime_protocol_test(self, e: CallExpr) -> None:
   772	        for expr in mypy.checker.flatten(e.args[1]):
   773	            tp = get_proper_type(self.chk.lookup_type(expr))
   774	            if (
   775	                isinstance(tp, FunctionLike)
   776	                and tp.is_type_obj()
   777	                and tp.type_object().is_protocol
   778	                and not tp.type_object().runtime_protocol
   779	            ):
   780	                self.chk.fail(message_registry.RUNTIME_PROTOCOL_EXPECTED, e)
   781
   782	    def check_protocol_issubclass(self, e: CallExpr) -> None:
   783	        for expr in mypy.checker.flatten(e.args[1]):
   784	            tp = get_proper_type(self.chk.lookup_type(expr))
   785	            if isinstance(tp, FunctionLike) and tp.is_type_obj() and tp.type_object().is_protocol:
   786	                attr_members = non_method_protocol_members(tp.type_object())
   787	                if attr_members:
   788	                    self.chk.msg.report_non_method_protocol(tp.type_object(), attr_members, e)

   790	    def check_typeddict_call(
   791	        self,
   792	        callee: TypedDictType,
   793	        arg_kinds: list[ArgKind],
   794	        arg_names: Sequence[str | None],
   795	        args: list[Expression],
   796	        context: Context,
   797	        orig_callee: Type | None,
   798	    ) -> Type:
   799	        if args and all(ak in (ARG_NAMED, ARG_STAR2) for ak in arg_kinds):
   800	            # ex: Point(x=42, y=1337, **extras)
   801	            # This is a bit ugly, but this is a price for supporting all possible syntax
   802	            # variants for TypedDict constructors.
   803	            kwargs = zip([StrExpr(n) if n is not None else None for n in arg_names], args)
   804	            result = self.validate_typeddict_kwargs(kwargs=kwargs, callee=callee)
   805	            if result is not None:
   806	                validated_kwargs, always_present_keys = result
   807	                return self.check_typeddict_call_with_kwargs(
   808	                    callee, validated_kwargs, context, orig_callee, always_present_keys
   809	                )
   810	            return AnyType(TypeOfAny.from_error)
   811
   812	        if len(args) == 1 and arg_kinds[0] == ARG_POS:
   813	            unique_arg = args[0]
   814	            if isinstance(unique_arg, DictExpr):
   815	                # ex: Point({'x': 42, 'y': 1337, **extras})
   816	                return self.check_typeddict_call_with_dict(
   817	                    callee, unique_arg.items, context, orig_callee
   818	                )
   819	            if isinstance(unique_arg, CallExpr) and isinstance(unique_arg.analyzed, DictExpr):
   820	                # ex: Point(dict(x=42, y=1337, **extras))
   821	                return self.check_typeddict_call_with_dict(
   822	                    callee, unique_arg.analyzed.items, context, orig_callee
   823	                )
   824
   825	        if not args:
   826	            # ex: EmptyDict()
   827	            return self.check_typeddict_call_with_kwargs(callee, {}, context, orig_callee, set())
   828
   829	        self.chk.fail(message_registry.INVALID_TYPEDDICT_ARGS, context)
   830	        return AnyType(TypeOfAny.from_error)
   831
   832	    def validate_typeddict_kwargs(
   833	        self, kwargs: Iterable[tuple[Expression | None, Expression]], callee: TypedDictType
   834	    ) -> tuple[dict[str, list[Expression]], set[str]] | None:
   835	        # All (actual or mapped from ** unpacks) expressions that can match given key.
   836	        result = defaultdict(list)
   837	        # Keys that are guaranteed to be present no matter what (e.g. for all items of a union)
   838	        always_present_keys = set()
   839	        # Indicates latest encountered ** unpack among items.
   840	        last_star_found = None
   841
   842	        for item_name_expr, item_arg in kwargs:
   843	            if item_name_expr:
   844	                key_type = self.accept(item_name_expr)
   845	                values = try_getting_str_literals(item_name_expr, key_type)
   846	                literal_value = None
   847	                if values and len(values) == 1:
   848	                    literal_value = values[0]
   849	                if literal_value is None:
   850	                    key_context = item_name_expr or item_arg
   851	                    self.chk.fail(
   852	                        message_registry.TYPEDDICT_KEY_MUST_BE_STRING_LITERAL,
   853	                        key_context,
   854	                        code=codes.LITERAL_REQ,
   855	                    )
   856	                    return None
   857	                else:
   858	                    # A directly present key unconditionally shadows all previously found
   859	                    # values from ** items.
   860	                    # TODO: for duplicate keys, type-check all values.
   861	                    result[literal_value] = [item_arg]
   862	                    always_present_keys.add(literal_value)
   863	            else:
   864	                last_star_found = item_arg
   865	                if not self.validate_star_typeddict_item(
   866	                    item_arg, callee, result, always_present_keys
   867	                ):
   868	                    return None
   869	        if self.chk.options.extra_checks and last_star_found is not None:
   870	            absent_keys = []
   871	            for key in callee.items:
   872	                if key not in callee.required_keys and key not in result:
   873	                    absent_keys.append(key)
   874	            if absent_keys:
   875	                # Having an optional key not explicitly declared by a ** unpacked
   876	                # TypedDict is unsafe, it may be an (incompatible) subtype at runtime.
   877	                # TODO: catch the cases where a declared key is overridden by a subsequent
   878	                # ** item without it (and not again overridden with complete ** item).
   879	                self.msg.non_required_keys_absent_with_star(absent_keys, last_star_found)
   880	        return result, always_present_keys

   882	    def validate_star_typeddict_item(
   883	        self,
   884	        item_arg: Expression,
   885	        callee: TypedDictType,
   886	        result: dict[str, list[Expression]],
   887	        always_present_keys: set[str],
   888	    ) -> bool:
   889	        """Update keys/expressions from a ** expression in TypedDict constructor.
   890
   891	        Note `result` and `always_present_keys` are updated in place. Return true if the
   892	        expression `item_arg` may valid in `callee` TypedDict context.
   893	        """
   894	        inferred = get_proper_type(self.accept(item_arg, type_context=callee))
   895	        possible_tds = []
   896	        if isinstance(inferred, TypedDictType):
   897	            possible_tds = [inferred]
   898	        elif isinstance(inferred, UnionType):
   899	            for item in get_proper_types(inferred.relevant_items()):
   900	                if isinstance(item, TypedDictType):
   901	                    possible_tds.append(item)
   902	                elif not self.valid_unpack_fallback_item(item):
   903	                    self.msg.unsupported_target_for_star_typeddict(item, item_arg)
   904	                    return False
   905	        elif not self.valid_unpack_fallback_item(inferred):
   906	            self.msg.unsupported_target_for_star_typeddict(inferred, item_arg)
   907	            return False
   908	        all_keys: set[str] = set()
   909	        for td in possible_tds:
   910	            all_keys |= td.items.keys()
   911	        for key in all_keys:
   912	            arg = TempNode(
   913	                UnionType.make_union([td.items[key] for td in possible_tds if key in td.items])
   914	            )
   915	            arg.set_line(item_arg)
   916	            if all(key in td.required_keys for td in possible_tds):
   917	                always_present_keys.add(key)
   918	                # Always present keys override previously found values. This is done
   919	                # to support use cases like `Config({**defaults, **overrides})`, where
   920	                # some `overrides` types are narrower that types in `defaults`, and
   921	                # former are too wide for `Config`.
   922	                if result[key]:
   923	                    first = result[key][0]
   924	                    if not isinstance(first, TempNode):
   925	                        # We must always preserve any non-synthetic values, so that
   926	                        # we will accept them even if they are shadowed.
   927	                        result[key] = [first, arg]
   928	                    else:
   929	                        result[key] = [arg]
   930	                else:
   931	                    result[key] = [arg]
   932	            else:
   933	                # If this key is not required at least in some item of a union
   934	                # it may not shadow previous item, so we need to type check both.
   935	                result[key].append(arg)
   936	        return True

   938	    def valid_unpack_fallback_item(self, typ: ProperType) -> bool:
   939	        if isinstance(typ, AnyType):
   940	            return True
   941	        if not isinstance(typ, Instance) or not typ.type.has_base("typing.Mapping"):
   942	            return False
   943	        mapped = map_instance_to_supertype(typ, self.chk.lookup_typeinfo("typing.Mapping"))
   944	        return all(isinstance(a, AnyType) for a in get_proper_types(mapped.args))

   980	        """Construct a reasonable type for a TypedDict type in runtime context.
   981
   982	        If it appears as a callee, it will be special-cased anyway, e.g. it is
   983	        also allowed to accept a single positional argument if it is a dict literal.

  1009	    def check_typeddict_call_with_kwargs(
  1010	        self,
  1011	        callee: TypedDictType,
  1012	        kwargs: dict[str, list[Expression]],
  1013	        context: Context,
  1014	        orig_callee: Type | None,
  1015	        always_present_keys: set[str],
  1016	    ) -> Type:
  1017	        actual_keys = kwargs.keys()
  1018	        if callee.to_be_mutated:
  1019	            assigned_readonly_keys = actual_keys & callee.readonly_keys
  1020	            if assigned_readonly_keys:
  1021	                self.msg.readonly_keys_mutated(assigned_readonly_keys, context=context)
  1022	        if not (
  1023	            callee.required_keys <= always_present_keys and actual_keys <= callee.items.keys()
  1024	        ):
  1025	            if not (actual_keys <= callee.items.keys()):
  1026	                self.msg.unexpected_typeddict_keys(
  1027	                    callee,
  1028	                    expected_keys=[
  1029	                        key
  1030	                        for key in callee.items.keys()
  1031	                        if key in callee.required_keys or key in actual_keys
  1032	                    ],
  1033	                    actual_keys=list(actual_keys),
  1034	                    context=context,
  1035	                )
  1036	            if not (callee.required_keys <= always_present_keys):
  1037	                self.msg.unexpected_typeddict_keys(
  1038	                    callee,
  1039	                    expected_keys=[
  1040	                        key for key in callee.items.keys() if key in callee.required_keys
  1041	                    ],
  1042	                    actual_keys=[
  1043	                        key for key in always_present_keys if key in callee.required_keys
  1044	                    ],
  1045	                    context=context,
  1046	                )
  1047	            if callee.required_keys > actual_keys:
  1048	                # found_set is a sub-set of the required_keys
  1049	                # This means we're missing some keys and as such, we can't
  1050	                # properly type the object
  1051	                return AnyType(TypeOfAny.from_error)
  1052
  1053	        orig_callee = get_proper_type(orig_callee)
  1054	        if isinstance(orig_callee, CallableType):
  1055	            infer_callee = orig_callee
  1056	        else:
  1057	            # Try reconstructing from type context.
  1058	            if callee.fallback.type.special_alias is not None:
  1059	                infer_callee = self.typeddict_callable(callee.fallback.type)
  1060	            else:
  1061	                # Likely a TypedDict type generated by a plugin.
  1062	                infer_callee = self.typeddict_callable_from_context(callee)
  1063
  1064	        # We don't show any errors, just infer types in a generic TypedDict type,
  1065	        # a custom error message will be given below, if there are errors.
  1066	        with self.msg.filter_errors(), self.chk.local_type_map():
  1067	            orig_ret_type, _ = self.check_callable_call(
  1068	                infer_callee,
  1069	                # We use first expression for each key to infer type variables of a generic
  1070	                # TypedDict. This is a bit arbitrary, but in most cases will work better than
  1071	                # trying to infer a union or a join.
  1072	                [args[0] for args in kwargs.values()],
  1073	                [ArgKind.ARG_NAMED] * len(kwargs),
  1074	                context,
  1075	                list(kwargs.keys()),
  1076	                None,
  1077	                None,
  1078	                None,
  1079	            )
  1080
  1081	        ret_type = get_proper_type(orig_ret_type)
  1082	        if not isinstance(ret_type, TypedDictType):
  1083	            # If something went really wrong, type-check call with original type,
  1084	            # this may give a better error message.
  1085	            ret_type = callee
  1086
  1087	        for item_name, item_expected_type in ret_type.items.items():
  1088	            if item_name in kwargs:
  1089	                item_values = kwargs[item_name]
  1090	                for item_value in item_values:
  1091	                    self.chk.check_simple_assignment(
  1092	                        lvalue_type=item_expected_type,
  1093	                        rvalue=item_value,
  1094	                        context=item_value,
  1095	                        msg=ErrorMessage(
  1096	                            message_registry.INCOMPATIBLE_TYPES.value, code=codes.TYPEDDICT_ITEM
  1097	                        ),
  1098	                        lvalue_name=f'TypedDict item "{item_name}"',
  1099	                        rvalue_name="expression",
  1100	                    )
  1101
  1102	        return orig_ret_type

  1105	        """Get variable node for a partial self attribute.
  1106
  1107	        If the expression is not a self attribute, or attribute is not variable,
  1108	        or variable is not partial, return None.
  1109	        """
  1110	        if not (
  1111	            isinstance(expr.expr, NameExpr)
  1112	            and isinstance(expr.expr.node, Var)
  1113	            and expr.expr.node.is_self

  1116	            return None

  1138	    def try_infer_partial_type(self, e: CallExpr) -> None:
  1139	        """Try to make partial type precise from a call."""
  1140	        if not isinstance(e.callee, MemberExpr):
  1141	            return
  1142	        callee = e.callee
  1143	        if isinstance(callee.expr, RefExpr):
  1144	            # Call a method with a RefExpr callee, such as 'x.method(...)'.
  1145	            ret = self.get_partial_var(callee.expr)
  1146	            if ret is None:
  1147	                return
  1148	            var, partial_types = ret
  1149	            typ = self.try_infer_partial_value_type_from_call(e, callee.name, var)
  1150	            # Var may be deleted from partial_types in try_infer_partial_value_type_from_call
  1151	            if typ is not None and var in partial_types:
  1152	                self.chk.replace_partial_type(var, typ, partial_types)
  1153	        elif isinstance(callee.expr, IndexExpr) and isinstance(callee.expr.base, RefExpr):
  1154	            # Call 'x[y].method(...)'; may infer type of 'x' if it's a partial defaultdict.
  1155	            if callee.expr.analyzed is not None:
  1156	                return  # A special form
  1157	            base = callee.expr.base
  1158	            index = callee.expr.index
  1159	            ret = self.get_partial_var(base)
  1160	            if ret is None:
  1161	                return
  1162	            var, partial_types = ret
  1163	            partial_type = get_partial_instance_type(var.type)
  1164	            if partial_type is None or partial_type.value_type is None:
  1165	                return
  1166	            value_type = self.try_infer_partial_value_type_from_call(e, callee.name, var)
  1167	            if value_type is not None:
  1168	                # Infer key type.
  1169	                key_type = self.accept(index)
  1170	                if mypy.checker.is_valid_inferred_type(key_type, self.chk.options):
  1171	                    # Store inferred partial type.
  1172	                    assert partial_type.type is not None
  1173	                    typename = partial_type.type.fullname
  1174	                    new_type = self.chk.named_generic_type(typename, [key_type, value_type])
  1175	                    self.chk.replace_partial_type(var, new_type, partial_types)
  1176
  1177	    def get_partial_var(self, ref: RefExpr) -> tuple[Var, dict[Var, Context]] | None:
  1178	        var = ref.node
  1179	        if var is None and isinstance(ref, MemberExpr):
  1180	            var = self.get_partial_self_var(ref)
  1181	        if not isinstance(var, Var):
  1182	            return None
  1183	        partial_types = self.chk.find_partial_types(var)
  1184	        if partial_types is None:
  1185	            return None
  1186	        return var, partial_types

  1188	    def try_infer_partial_value_type_from_call(
  1189	        self, e: CallExpr, methodname: str, var: Var
  1190	    ) -> Instance | None:
  1191	        """Try to make partial type precise from a call such as 'x.append(y)'."""
  1192	        if self.chk.current_node_deferred:
  1193	            return None
  1194	        partial_type = get_partial_instance_type(var.type)
  1195	        if partial_type is None:
  1196	            return None
  1197	        if partial_type.value_type:
  1198	            typename = partial_type.value_type.type.fullname
  1199	        else:
  1200	            assert partial_type.type is not None
  1201	            typename = partial_type.type.fullname
  1202	        # Sometimes we can infer a full type for a partial List, Dict or Set type.
  1203	        # TODO: Don't infer argument expression twice.
  1204	        if (
  1205	            typename in self.item_args
  1206	            and methodname in self.item_args[typename]
  1207	            and e.arg_kinds == [ARG_POS]
  1208	        ):
  1209	            item_type = self.accept(e.args[0])
  1210	            if mypy.checker.is_valid_inferred_type(item_type, self.chk.options):
  1211	                return self.chk.named_generic_type(typename, [item_type])
  1212	        elif (
  1213	            typename in self.container_args
  1214	            and methodname in self.container_args[typename]
  1215	            and e.arg_kinds == [ARG_POS]
  1216	        ):
  1217	            arg_type = get_proper_type(self.accept(e.args[0]))
  1218	            if isinstance(arg_type, Instance):
  1219	                arg_typename = arg_type.type.fullname
  1220	                if arg_typename in self.container_args[typename][methodname]:
  1221	                    if all(
  1222	                        mypy.checker.is_valid_inferred_type(item_type, self.chk.options)
  1223	                        for item_type in arg_type.args
  1224	                    ):
  1225	                        return self.chk.named_generic_type(typename, list(arg_type.args))
  1226	            elif isinstance(arg_type, AnyType):
  1227	                return self.chk.named_type(typename)
  1228
  1229	        return None

  1231	    def apply_function_plugin(
  1232	        self,
  1233	        callee: CallableType,
  1234	        arg_kinds: list[ArgKind],
  1235	        arg_types: list[Type],
  1236	        arg_names: Sequence[str | None] | None,
  1237	        formal_to_actual: list[list[int]],
  1238	        args: list[Expression],
  1239	        fullname: str,
  1240	        object_type: Type | None,
  1241	        context: Context,
  1242	    ) -> Type:
  1243	        """Use special case logic to infer the return type of a specific named function/method.
  1244
  1245	        Caller must ensure that a plugin hook exists. There are two different cases:
  1246
  1247	        - If object_type is None, the caller must ensure that a function hook exists
  1248	          for fullname.
  1249	        - If object_type is not None, the caller must ensure that a method hook exists
  1250	          for fullname.
  1251
  1252	        Return the inferred return type.
  1253	        """
  1254	        num_formals = len(callee.arg_types)
  1255	        formal_arg_types: list[list[Type]] = [[] for _ in range(num_formals)]
  1256	        formal_arg_exprs: list[list[Expression]] = [[] for _ in range(num_formals)]
  1257	        formal_arg_names: list[list[str | None]] = [[] for _ in range(num_formals)]
  1258	        formal_arg_kinds: list[list[ArgKind]] = [[] for _ in range(num_formals)]
  1259	        for formal, actuals in enumerate(formal_to_actual):
  1260	            for actual in actuals:
  1261	                formal_arg_types[formal].append(arg_types[actual])
  1262	                formal_arg_exprs[formal].append(args[actual])
  1263	                if arg_names:
  1264	                    formal_arg_names[formal].append(arg_names[actual])
  1265	                else:
  1266	                    formal_arg_names[formal].append(None)
  1267	                formal_arg_kinds[formal].append(arg_kinds[actual])
  1268
  1269	        if object_type is None:
  1270	            # Apply function plugin
  1271	            callback = self.plugin.get_function_hook(fullname)
  1272	            assert callback is not None  # Assume that caller ensures this
  1273	            return callback(
  1274	                FunctionContext(
  1275	                    arg_types=formal_arg_types,
  1276	                    arg_kinds=formal_arg_kinds,
  1277	                    callee_arg_names=callee.arg_names,
  1278	                    arg_names=formal_arg_names,
  1279	                    default_return_type=callee.ret_type,
  1280	                    args=formal_arg_exprs,
  1281	                    context=context,
  1282	                    api=self.chk,
  1283	                )
  1284	            )
  1285	        else:
  1286	            # Apply method plugin
  1287	            method_callback = self.plugin.get_method_hook(fullname)
  1288	            assert method_callback is not None  # Assume that caller ensures this
  1289	            object_type = get_proper_type(object_type)
  1290	            return method_callback(
  1291	                MethodContext(
  1292	                    type=object_type,
  1293	                    arg_types=formal_arg_types,
  1294	                    arg_kinds=formal_arg_kinds,
  1295	                    callee_arg_names=callee.arg_names,
  1296	                    arg_names=formal_arg_names,
  1297	                    default_return_type=callee.ret_type,
  1298	                    args=formal_arg_exprs,
  1299	                    context=context,
  1300	                    api=self.chk,
  1301	                )
  1302	            )

  1304	    def apply_signature_hook(
  1305	        self,
  1306	        callee: FunctionLike,
  1307	        args: list[Expression],
  1308	        arg_kinds: list[ArgKind],
  1309	        arg_names: Sequence[str | None] | None,
  1310	        hook: Callable[[list[list[Expression]], CallableType], FunctionLike],
  1311	    ) -> FunctionLike:
  1312	        """Helper to apply a signature hook for either a function or method"""
  1313	        if isinstance(callee, CallableType):
  1314	            num_formals = len(callee.arg_kinds)
  1315	            formal_to_actual = map_actuals_to_formals(
  1316	                arg_kinds,
  1317	                arg_names,
  1318	                callee.arg_kinds,
  1319	                callee.arg_names,
  1320	                lambda i: self.accept(args[i]),
  1321	            )
  1322	            formal_arg_exprs: list[list[Expression]] = [[] for _ in range(num_formals)]
  1323	            for formal, actuals in enumerate(formal_to_actual):
  1324	                for actual in actuals:
  1325	                    formal_arg_exprs[formal].append(args[actual])
  1326	            return hook(formal_arg_exprs, callee)
  1327	        else:
  1328	            assert isinstance(callee, Overloaded)
  1329	            items = []
  1330	            for item in callee.items:
  1331	                adjusted = self.apply_signature_hook(item, args, arg_kinds, arg_names, hook)
  1332	                assert isinstance(adjusted, CallableType)
  1333	                items.append(adjusted)
  1334	            return Overloaded(items)

  1340	        arg_kinds: list[ArgKind],

  1342	        arg_names: Sequence[str | None] | None,
  1343	        signature_hook: Callable[[FunctionSigContext], FunctionLike],

  1345	        """Apply a plugin hook that may infer a more precise signature for a function."""
  1346	        return self.apply_signature_hook(
  1347	            callee,
  1348	            args,
  1349	            arg_kinds,
  1350	            arg_names,
  1351	            (lambda args, sig: signature_hook(FunctionSigContext(args, sig, context, self.chk))),

  1378	    def transform_callee_type(
  1379	        self,
  1380	        callable_name: str | None,
  1381	        callee: Type,
  1382	        args: list[Expression],
  1383	        arg_kinds: list[ArgKind],
  1384	        context: Context,
  1385	        arg_names: Sequence[str | None] | None = None,
  1386	        object_type: Type | None = None,
  1387	    ) -> Type:
  1388	        """Attempt to determine a more accurate signature for a method call.
  1389
  1390	        This is done by looking up and applying a method signature hook (if one exists for the
  1391	        given method name).
  1392
  1393	        If no matching method signature hook is found, callee is returned unmodified. The same
  1394	        happens if the arguments refer to a non-method callable (this is allowed so that the code
  1395	        calling transform_callee_type needs to perform fewer boilerplate checks).
  1396
  1397	        Note: this method is *not* called automatically as part of check_call, because in some
  1398	        cases check_call is called multiple times while checking a single call (for example when
  1399	        dealing with overloads). Instead, this method needs to be called explicitly
  1400	        (if appropriate) before the signature is passed to check_call.
  1401	        """
  1402	        callee = get_proper_type(callee)
  1403	        if callable_name is not None and isinstance(callee, FunctionLike):
  1404	            if object_type is not None:
  1405	                method_sig_hook = self.plugin.get_method_signature_hook(callable_name)
  1406	                if method_sig_hook:
  1407	                    return self.apply_method_signature_hook(
  1408	                        callee, args, arg_kinds, context, arg_names, object_type, method_sig_hook
  1409	                    )
  1410	            else:
  1411	                function_sig_hook = self.plugin.get_function_signature_hook(callable_name)
  1412	                if function_sig_hook:
  1413	                    return self.apply_function_signature_hook(
  1414	                        callee, args, arg_kinds, context, arg_names, function_sig_hook
  1415	                    )
  1416
  1417	        return callee

  1419	    def is_generic_decorator_overload_call(
  1420	        self, callee_type: CallableType, args: list[Expression]
  1421	    ) -> Overloaded | None:
  1422	        """Check if this looks like an application of a generic function to overload argument."""
  1423	        assert callee_type.variables
  1424	        if len(callee_type.arg_types) != 1 or len(args) != 1:
  1425	            # TODO: can we handle more general cases?
  1426	            return None
  1427	        if not isinstance(get_proper_type(callee_type.arg_types[0]), CallableType):
  1428	            return None
  1429	        if not isinstance(get_proper_type(callee_type.ret_type), CallableType):
  1430	            return None
  1431	        with self.chk.local_type_map():
  1432	            with self.msg.filter_errors():
  1433	                arg_type = get_proper_type(self.accept(args[0], type_context=None))
  1434	        if isinstance(arg_type, Overloaded):
  1435	            return arg_type
  1436	        return None

  1438	    def handle_decorator_overload_call(
  1439	        self, callee_type: CallableType, overloaded: Overloaded, ctx: Context
  1440	    ) -> tuple[Type, Type] | None:
  1441	        """Type-check application of a generic callable to an overload.
  1442
  1443	        We check call on each individual overload item, and then combine results into a new
  1444	        overload. This function should be only used if callee_type takes and returns a Callable.
  1445	        """
  1446	        result = []
  1447	        inferred_args = []
  1448	        for item in overloaded.items:
  1449	            arg = TempNode(typ=item)
  1450	            with self.msg.filter_errors() as err:
  1451	                item_result, inferred_arg = self.check_call(callee_type, [arg], [ARG_POS], ctx)
  1452	            if err.has_new_errors():
  1453	                # This overload doesn't match.
  1454	                continue
  1455	            p_item_result = get_proper_type(item_result)
  1456	            if not isinstance(p_item_result, CallableType):
  1457	                continue
  1458	            p_inferred_arg = get_proper_type(inferred_arg)
  1459	            if not isinstance(p_inferred_arg, CallableType):
  1460	                continue
  1461	            inferred_args.append(p_inferred_arg)
  1462	            result.append(p_item_result)
  1463	        if not result or not inferred_args:
  1464	            # None of the overload matched (or overload was initially malformed).
  1465	            return None
  1466	        return Overloaded(result), Overloaded(inferred_args)

  1468	    def check_call_expr_with_callee_type(
  1469	        self,
  1470	        callee_type: Type,
  1471	        e: CallExpr,
  1472	        callable_name: str | None,
  1473	        object_type: Type | None,
  1474	        member: str | None = None,
  1475	    ) -> Type:
  1476	        """Type check call expression.
  1477
  1478	        The callee_type should be used as the type of callee expression. In particular,
  1479	        in case of a union type this can be a particular item of the union, so that we can
  1480	        apply plugin hooks to each item.
  1481
  1482	        The 'member', 'callable_name' and 'object_type' are only used to call plugin hooks.
  1483	        If 'callable_name' is None but 'member' is not None (member call), try constructing
  1484	        'callable_name' using 'object_type' (the base type on which the method is called),
  1485	        for example 'typing.Mapping.get'.
  1486	        """
  1487	        if callable_name is None and member is not None:
  1488	            assert object_type is not None
  1489	            callable_name = self.method_fullname(object_type, member)
  1490	        object_type = get_proper_type(object_type)
  1491	        if callable_name:
  1492	            # Try to refine the call signature using plugin hooks before checking the call.
  1493	            callee_type = self.transform_callee_type(
  1494	                callable_name, callee_type, e.args, e.arg_kinds, e, e.arg_names, object_type
  1495	            )
  1496	        # Unions are special-cased to allow plugins to act on each item in the union.
  1497	        elif member is not None and isinstance(object_type, UnionType):
  1498	            return self.check_union_call_expr(e, object_type, member)
  1499	        ret_type, callee_type = self.check_call(
  1500	            callee_type,
  1501	            e.args,
  1502	            e.arg_kinds,
  1503	            e,
  1504	            e.arg_names,
  1505	            callable_node=e.callee,
  1506	            callable_name=callable_name,
  1507	            object_type=object_type,
  1508	        )
  1509	        proper_callee = get_proper_type(callee_type)
  1510	        if isinstance(e.callee, (NameExpr, MemberExpr)):
  1511	            node = e.callee.node
  1512	            if node is None and member is not None and isinstance(object_type, Instance):
  1513	                if (symbol := object_type.type.get(member)) is not None:
  1514	                    node = symbol.node
  1515	            self.chk.check_deprecated(node, e)
  1516	            self.chk.warn_deprecated_overload_item(
  1517	                node, e, target=callee_type, selftype=object_type
  1518	            )
  1519	        if isinstance(e.callee, RefExpr) and isinstance(proper_callee, CallableType):
  1520	            # Cache it for find_isinstance_check()
  1521	            if proper_callee.type_guard is not None:
  1522	                e.callee.type_guard = proper_callee.type_guard
  1523	            if proper_callee.type_is is not None:
  1524	                e.callee.type_is = proper_callee.type_is
  1525	        return ret_type

  1527	    def check_union_call_expr(self, e: CallExpr, object_type: UnionType, member: str) -> Type:
  1528	        """Type check calling a member expression where the base type is a union."""
  1529	        res: list[Type] = []
  1530	        for typ in object_type.relevant_items():
  1531	            # Member access errors are already reported when visiting the member expression.
  1532	            with self.msg.filter_errors():
  1533	                item = analyze_member_access(
  1534	                    member,
  1535	                    typ,
  1536	                    e,
  1537	                    is_lvalue=False,
  1538	                    is_super=False,
  1539	                    is_operator=False,
  1540	                    original_type=object_type,
  1541	                    chk=self.chk,
  1542	                    in_literal_context=self.is_literal_context(),
  1543	                    self_type=typ,
  1544	                )
  1545	            narrowed = self.narrow_type_from_binder(e.callee, item, skip_non_overlapping=True)
  1546	            if narrowed is None:
  1547	                continue
  1548	            callable_name = self.method_fullname(typ, member)
  1549	            item_object_type = typ if callable_name else None

  1555	    def check_call(
  1556	        self,
  1557	        callee: Type,
  1558	        args: list[Expression],
  1559	        arg_kinds: list[ArgKind],
  1560	        context: Context,
  1561	        arg_names: Sequence[str | None] | None = None,
  1562	        callable_node: Expression | None = None,
  1563	        callable_name: str | None = None,
  1564	        object_type: Type | None = None,
  1565	        original_type: Type | None = None,
  1566	    ) -> tuple[Type, Type]:
  1567	        """Type check a call.
  1568
  1569	        Also infer type arguments if the callee is a generic function.
  1570
  1571	        Return (result type, inferred callee type).
  1572
  1573	        Arguments:
  1574	            callee: type of the called value
  1575	            args: actual argument expressions
  1576	            arg_kinds: contains nodes.ARG_* constant for each argument in args
  1577	                 describing whether the argument is positional, *arg, etc.
  1578	            context: current expression context, used for inference.
  1579	            arg_names: names of arguments (optional)
  1580	            callable_node: associate the inferred callable type to this node,
  1581	                if specified
  1582	            callable_name: Fully-qualified name of the function/method to call,
  1583	                or None if unavailable (examples: 'builtins.open', 'typing.Mapping.get')
  1584	            object_type: If callable_name refers to a method, the type of the object
  1585	                on which the method is being called
  1586	        """
  1587	        callee = get_proper_type(callee)
  1588
  1589	        if isinstance(callee, CallableType):
  1590	            if callee.variables:
  1591	                overloaded = self.is_generic_decorator_overload_call(callee, args)
  1592	                if overloaded is not None:
  1593	                    # Special casing for inline application of generic callables to overloads.
  1594	                    # Supporting general case would be tricky, but this should cover 95% of cases.
  1595	                    overloaded_result = self.handle_decorator_overload_call(
  1596	                        callee, overloaded, context
  1597	                    )
  1598	                    if overloaded_result is not None:
  1599	                        return overloaded_result
  1600
  1601	            return self.check_callable_call(
  1602	                callee,
  1603	                args,
  1604	                arg_kinds,
  1605	                context,
  1606	                arg_names,
  1607	                callable_node,
  1608	                callable_name,
  1609	                object_type,
  1610	            )
  1611	        elif isinstance(callee, Overloaded):
  1612	            return self.check_overload_call(
  1613	                callee, args, arg_kinds, arg_names, callable_name, object_type, context
  1614	            )
  1615	        elif isinstance(callee, AnyType) or not self.chk.in_checked_function():
  1616	            return self.check_any_type_call(args, callee)
  1617	        elif isinstance(callee, UnionType):
  1618	            return self.check_union_call(callee, args, arg_kinds, arg_names, context)
  1619	        elif isinstance(callee, Instance):
  1620	            call_function = analyze_member_access(
  1621	                "__call__",
  1622	                callee,
  1623	                context,
  1624	                is_lvalue=False,
  1625	                is_super=False,
  1626	                is_operator=True,
  1627	                original_type=original_type or callee,
  1628	                chk=self.chk,
  1629	                in_literal_context=self.is_literal_context(),
  1630	            )
  1631	            callable_name = callee.type.fullname + ".__call__"
  1632	            # Apply method signature hook, if one exists
  1633	            call_function = self.transform_callee_type(
  1634	                callable_name, call_function, args, arg_kinds, context, arg_names, callee
  1635	            )
  1636	            result = self.check_call(
  1637	                call_function,
  1638	                args,
  1639	                arg_kinds,
  1640	                context,
  1641	                arg_names,
  1642	                callable_node,
  1643	                callable_name,
  1644	                callee,
  1645	            )
  1646	            if callable_node:
  1647	                # check_call() stored "call_function" as the type, which is incorrect.
  1648	                # Override the type.
  1649	                self.chk.store_type(callable_node, callee)
  1650	            return result
  1651	        elif isinstance(callee, TypeVarType):
  1652	            return self.check_call(
  1653	                callee.upper_bound, args, arg_kinds, context, arg_names, callable_node
  1654	            )
  1655	        elif isinstance(callee, TypeType):
  1656	            item = self.analyze_type_type_callee(callee.item, context)
  1657	            return self.check_call(item, args, arg_kinds, context, arg_names, callable_node)
  1658	        elif isinstance(callee, TupleType):
  1659	            return self.check_call(
  1660	                tuple_fallback(callee),
  1661	                args,
  1662	                arg_kinds,
  1663	                context,
  1664	                arg_names,
  1665	                callable_node,
  1666	                callable_name,
  1667	                object_type,
  1668	                original_type=callee,
  1669	            )
  1670	        else:
  1671	            return self.msg.not_callable(callee, context), AnyType(TypeOfAny.from_error)
  1672
  1673	    def check_callable_call(
  1674	        self,
  1675	        callee: CallableType,
  1676	        args: list[Expression],
  1677	        arg_kinds: list[ArgKind],
  1678	        context: Context,
  1679	        arg_names: Sequence[str | None] | None,
  1680	        callable_node: Expression | None,
  1681	        callable_name: str | None,
  1682	        object_type: Type | None,
  1683	    ) -> tuple[Type, Type]:
  1684	        """Type check a call that targets a callable value.
  1685
  1686	        See the docstring of check_call for more information.
  1687	        """
  1688	        # Always unpack **kwargs before checking a call.
  1689	        callee = callee.with_unpacked_kwargs().with_normalized_var_args()
  1690	        if callable_name is None and callee.name:
  1691	            callable_name = callee.name
  1692	        ret_type = get_proper_type(callee.ret_type)
  1693	        if callee.is_type_obj() and isinstance(ret_type, Instance):
  1694	            callable_name = ret_type.type.fullname
  1695	        if isinstance(callable_node, RefExpr) and callable_node.fullname in ENUM_BASES:
  1696	            # An Enum() call that failed SemanticAnalyzerPass2.check_enum_call().
  1697	            return callee.ret_type, callee
  1698
  1699	        if (
  1700	            callee.is_type_obj()
  1701	            and callee.type_object().is_protocol
  1702	            # Exception for Type[...]
  1703	            and not callee.from_type_type
  1704	        ):
  1705	            self.chk.fail(
  1706	                message_registry.CANNOT_INSTANTIATE_PROTOCOL.format(callee.type_object().name),
  1707	                context,
  1708	            )
  1709	        elif (
  1710	            callee.is_type_obj()
  1711	            and callee.type_object().is_abstract
  1712	            # Exception for Type[...]
  1713	            and not callee.from_type_type
  1714	            and not callee.type_object().fallback_to_any
  1715	        ):
  1716	            type = callee.type_object()
  1717	            # Determine whether the implicitly abstract attributes are functions with
  1718	            # None-compatible return types.
  1719	            abstract_attributes: dict[str, bool] = {}
  1720	            for attr_name, abstract_status in type.abstract_attributes:
  1721	                if abstract_status == IMPLICITLY_ABSTRACT:
  1722	                    abstract_attributes[attr_name] = self.can_return_none(type, attr_name)
  1723	                else:
  1724	                    abstract_attributes[attr_name] = False
  1725	            self.msg.cannot_instantiate_abstract_class(
  1726	                callee.type_object().name, abstract_attributes, context
  1727	            )
  1728
  1729	        var_arg = callee.var_arg()
  1730	        if var_arg and isinstance(var_arg.typ, UnpackType):
  1731	            # It is hard to support multiple variadic unpacks (except for old-style *args: int),
  1732	            # fail gracefully to avoid crashes later.
  1733	            seen_unpack = False
  1734	            for arg, arg_kind in zip(args, arg_kinds):
  1735	                if arg_kind != ARG_STAR:
  1736	                    continue
  1737	                arg_type = get_proper_type(self.accept(arg))
  1738	                if not isinstance(arg_type, TupleType) or any(
  1739	                    isinstance(t, UnpackType) for t in arg_type.items
  1740	                ):
  1741	                    if seen_unpack:
  1742	                        self.msg.fail(
  1743	                            "Passing multiple variadic unpacks in a call is not supported",
  1744	                            context,
  1745	                            code=codes.CALL_ARG,
  1746	                        )
  1747	                        return AnyType(TypeOfAny.from_error), callee
  1748	                    seen_unpack = True
  1749
  1750	        formal_to_actual = map_actuals_to_formals(
  1751	            arg_kinds,
  1752	            arg_names,
  1753	            callee.arg_kinds,
  1754	            callee.arg_names,
  1755	            lambda i: self.accept(args[i]),
  1756	        )
  1757
  1758	        # This is tricky: return type may contain its own type variables, like in
  1759	        # def [S] (S) -> def [T] (T) -> tuple[S, T], so we need to update their ids
  1760	        # to avoid possible id clashes if this call itself appears in a generic
  1761	        # function body.
  1762	        ret_type = get_proper_type(callee.ret_type)
  1763	        if isinstance(ret_type, CallableType) and ret_type.variables:
  1764	            fresh_ret_type = freshen_all_functions_type_vars(callee.ret_type)
  1765	            freeze_all_type_vars(fresh_ret_type)
  1766	            callee = callee.copy_modified(ret_type=fresh_ret_type)
  1767
  1768	        if callee.is_generic():
  1769	            need_refresh = any(
  1770	                isinstance(v, (ParamSpecType, TypeVarTupleType)) for v in callee.variables
  1771	            )
  1772	            callee = freshen_function_type_vars(callee)
  1773	            callee = self.infer_function_type_arguments_using_context(callee, context)
  1774	            if need_refresh:
  1775	                # Argument kinds etc. may have changed due to
  1776	                # ParamSpec or TypeVarTuple variables being replaced with an arbitrary
  1777	                # number of arguments; recalculate actual-to-formal map
  1778	                formal_to_actual = map_actuals_to_formals(
  1779	                    arg_kinds,
  1780	                    arg_names,
  1781	                    callee.arg_kinds,
  1782	                    callee.arg_names,
  1783	                    lambda i: self.accept(args[i]),
  1784	                )
  1785	            callee = self.infer_function_type_arguments(
  1786	                callee, args, arg_kinds, arg_names, formal_to_actual, need_refresh, context
  1787	            )
  1788	            if need_refresh:
  1789	                formal_to_actual = map_actuals_to_formals(
  1790	                    arg_kinds,
  1791	                    arg_names,
  1792	                    callee.arg_kinds,
  1793	                    callee.arg_names,
  1794	                    lambda i: self.accept(args[i]),
  1795	                )
  1796
  1797	        param_spec = callee.param_spec()
  1798	        if (
  1799	            param_spec is not None
  1800	            and arg_kinds == [ARG_STAR, ARG_STAR2]
  1801	            and len(formal_to_actual) == 2

  1810	                and arg1.id == arg2.id == param_spec.id

  1829	        )
  1830
  1831	        if (
  1832	            callee.is_type_obj()
  1833	            and (len(arg_types) == 1)
  1834	            and is_equivalent(callee.ret_type, self.named_type("builtins.type"))
  1835	        ):

  1837
  1838	        if callable_node:
  1839	            # Store the inferred callable type.
  1840	            self.chk.store_type(callable_node, callee)
  1841
  1842	        if callable_name and (
  1843	            (object_type is None and self.plugin.get_function_hook(callable_name))
  1844	            or (object_type is not None and self.plugin.get_method_hook(callable_name))
  1845	        ):

  1847	                callee,
  1848	                arg_kinds,
  1849	                arg_types,
  1850	                arg_names,
  1851	                formal_to_actual,
  1852	                args,
  1853	                callable_name,
  1854	                object_type,
  1855	                context,
  1856	            )

  1860	    def can_return_none(self, type: TypeInfo, attr_name: str) -> bool:
  1861	        """Is the given attribute a method with a None-compatible return type?
  1862
  1863	        Overloads are only checked if there is an implementation.
  1864	        """
  1865	        if not state.strict_optional:
  1866	            # If strict-optional is not set, is_subtype(NoneType(), T) is always True.
  1867	            # So, we cannot do anything useful here in that case.
  1868	            return False
  1869	        for base in type.mro:
  1870	            symnode = base.names.get(attr_name)
  1871	            if symnode is None:
  1872	                continue
  1873	            node = symnode.node
  1874	            if isinstance(node, OverloadedFuncDef):
  1875	                node = node.impl
  1876	            if isinstance(node, Decorator):
  1877	                node = node.func
  1878	            if isinstance(node, FuncDef):
  1879	                if node.type is not None:
  1880	                    assert isinstance(node.type, CallableType)

  1884	    def analyze_type_type_callee(self, item: ProperType, context: Context) -> Type:
  1885	        """Analyze the callee X in X(...) where X is Type[item].
  1886
  1887	        Return a Y that we can pass to check_call(Y, ...).
  1888	        """
  1889	        if isinstance(item, AnyType):
  1890	            return AnyType(TypeOfAny.from_another_any, source_any=item)
  1891	        if isinstance(item, Instance):
  1892	            res = type_object_type(item.type, self.named_type)
  1893	            if isinstance(res, CallableType):
  1894	                res = res.copy_modified(from_type_type=True)
  1895	            expanded = expand_type_by_instance(res, item)
  1896	            if isinstance(expanded, CallableType):
  1897	                # Callee of the form Type[...] should never be generic, only
  1898	                # proper class objects can be.
  1899	                expanded = expanded.copy_modified(variables=[])
  1900	            return expanded
  1901	        if isinstance(item, UnionType):
  1902	            return UnionType(
  1903	                [
  1904	                    self.analyze_type_type_callee(get_proper_type(tp), context)
  1905	                    for tp in item.relevant_items()
  1906	                ],
  1907	                item.line,
  1908	            )
  1909	        if isinstance(item, TypeVarType):
  1910	            # Pretend we're calling the typevar's upper bound,
  1911	            # i.e. its constructor (a poor approximation for reality,
  1912	            # but better than AnyType...), but replace the return type
  1913	            # with typevar.
  1914	            callee = self.analyze_type_type_callee(get_proper_type(item.upper_bound), context)
  1915	            callee = get_proper_type(callee)
  1916	            if isinstance(callee, CallableType):
  1917	                callee = callee.copy_modified(ret_type=item)
  1918	            elif isinstance(callee, Overloaded):
  1919	                callee = Overloaded([c.copy_modified(ret_type=item) for c in callee.items])
  1920	            return callee
  1921	        # We support Type of namedtuples but not of tuples in general
  1922	        if isinstance(item, TupleType) and tuple_fallback(item).type.fullname != "builtins.tuple":
  1923	            return self.analyze_type_type_callee(tuple_fallback(item), context)
  1924	        if isinstance(item, TypedDictType):
  1925	            return self.typeddict_callable_from_context(item)
  1926
  1927	        self.msg.unsupported_type_type(item, context)
  1928	        return AnyType(TypeOfAny.from_error)

  1962	    def infer_arg_types_in_context(
  1963	        self,
  1964	        callee: CallableType,
  1965	        args: list[Expression],
  1966	        arg_kinds: list[ArgKind],
  1967	        formal_to_actual: list[list[int]],
  1968	    ) -> list[Type]:
  1969	        """Infer argument expression types using a callable type as context.
  1970
  1971	        For example, if callee argument 2 has type List[int], infer the
  1972	        argument expression with List[int] type context.
  1973
  1974	        Returns the inferred types of *actual arguments*.
  1975	        """
  1976	        res: list[Type | None] = [None] * len(args)
  1977
  1978	        for i, actuals in enumerate(formal_to_actual):
  1979	            for ai in actuals:
  1980	                if not arg_kinds[ai].is_star():
  1981	                    arg_type = callee.arg_types[i]
  1982	                    # When the outer context for a function call is known to be recursive,
  1983	                    # we solve type constraints inferred from arguments using unions instead
  1984	                    # of joins. This is a bit arbitrary, but in practice it works for most
  1985	                    # cases. A cleaner alternative would be to switch to single bin type
  1986	                    # inference, but this is a lot of work.
  1987	                    old = self.infer_more_unions_for_recursive_type(arg_type)
  1988	                    res[ai] = self.accept(args[ai], arg_type)
  1989	                    # We need to manually restore union inference state, ugh.
  1990	                    type_state.infer_unions = old
  1991
  1992	        # Fill in the rest of the argument types.
  1993	        for i, t in enumerate(res):
  1994	            if not t:
  1995	                res[i] = self.accept(args[i])
  1996	        assert all(tp is not None for tp in res)
  1997	        return cast(list[Type], res)

  1999	    def infer_function_type_arguments_using_context(
  2000	        self, callable: CallableType, error_context: Context
  2001	    ) -> CallableType:
  2002	        """Unify callable return type to type context to infer type vars.
  2003
  2004	        For example, if the return type is set[t] where 't' is a type variable
  2005	        of callable, and if the context is set[int], return callable modified
  2006	        by substituting 't' with 'int'.
  2007	        """
  2008	        ctx = self.type_context[-1]
  2009	        if not ctx:
  2010	            return callable
  2011	        # The return type may have references to type metavariables that
  2012	        # we are inferring right now. We must consider them as indeterminate
  2013	        # and they are not potential results; thus we replace them with the
  2014	        # special ErasedType type. On the other hand, class type variables are
  2015	        # valid results.
  2016	        erased_ctx = replace_meta_vars(ctx, ErasedType())
  2017	        ret_type = callable.ret_type
  2018	        if is_overlapping_none(ret_type) and is_overlapping_none(ctx):
  2019	            # If both the context and the return type are optional, unwrap the optional,
  2020	            # since in 99% cases this is what a user expects. In other words, we replace
  2021	            #     Optional[T] <: Optional[int]
  2022	            # with
  2023	            #     T <: int
  2024	            # while the former would infer T <: Optional[int].
  2025	            ret_type = remove_optional(ret_type)
  2026	            erased_ctx = remove_optional(erased_ctx)
  2027	            #
  2028	            # TODO: Instead of this hack and the one below, we need to use outer and
  2029	            # inner contexts at the same time. This is however not easy because of two
  2030	            # reasons:
  2031	            #   * We need to support constraints like [1 <: 2, 2 <: X], i.e. with variables
  2032	            #     on both sides. (This is not too hard.)
  2033	            #   * We need to update all the inference "infrastructure", so that all
  2034	            #     variables in an expression are inferred at the same time.
  2035	            #     (And this is hard, also we need to be careful with lambdas that require
  2036	            #     two passes.)
  2037	        proper_ret = get_proper_type(ret_type)
  2038	        if (
  2039	            isinstance(proper_ret, TypeVarType)
  2040	            or isinstance(proper_ret, UnionType)
  2041	            and all(isinstance(get_proper_type(u), TypeVarType) for u in proper_ret.items)
  2042	        ):
  2043	            # Another special case: the return type is a type variable. If it's unrestricted,
  2044	            # we could infer a too general type for the type variable if we use context,
  2045	            # and this could result in confusing and spurious type errors elsewhere.
  2046	            #
  2047	            # So we give up and just use function arguments for type inference, with just two
  2048	            # exceptions:
  2049	            #
  2050	            # 1. If the context is a generic instance type, actually use it as context, as
  2051	            #    this *seems* to usually be the reasonable thing to do.
  2052	            #
  2053	            #    See also github issues #462 and #360.
  2054	            #
  2055	            # 2. If the context is some literal type, we want to "propagate" that information
  2056	            #    down so that we infer a more precise type for literal expressions. For example,
  2057	            #    the expression `3` normally has an inferred type of `builtins.int`: but if it's
  2058	            #    in a literal context like below, we want it to infer `Literal[3]` instead.
  2059	            #
  2060	            #        def expects_literal(x: Literal[3]) -> None: pass
  2061	            #        def identity(x: T) -> T: return x
  2062	            #
  2063	            #        expects_literal(identity(3))  # Should type-check
  2064	            # TODO: we may want to add similar exception if all arguments are lambdas, since
  2065	            # in this case external context is almost everything we have.
  2066	            if not is_generic_instance(ctx) and not is_literal_type_like(ctx):
  2067	                return callable.copy_modified()
  2068	        args = infer_type_arguments(
  2069	            callable.variables, ret_type, erased_ctx, skip_unsatisfied=True
  2070	        )
  2071	        # Only substitute non-Uninhabited and non-erased types.
  2072	        new_args: list[Type | None] = []
  2073	        for arg in args:
  2074	            if has_uninhabited_component(arg) or has_erased_component(arg):
  2075	                new_args.append(None)
  2076	            else:
  2077	                new_args.append(arg)
  2078	        # Don't show errors after we have only used the outer context for inference.
  2079	        # We will use argument context to infer more variables.
  2080	        return self.apply_generic_arguments(
  2081	            callable, new_args, error_context, skip_unsatisfied=True
  2082	        )

  2084	    def infer_function_type_arguments(
  2085	        self,
  2086	        callee_type: CallableType,
  2087	        args: list[Expression],
  2088	        arg_kinds: list[ArgKind],
  2089	        arg_names: Sequence[str | None] | None,
  2090	        formal_to_actual: list[list[int]],
  2091	        need_refresh: bool,
  2092	        context: Context,
  2093	    ) -> CallableType:
  2094	        """Infer the type arguments for a generic callee type.
  2095
  2096	        Infer based on the types of arguments.
  2097
  2098	        Return a derived callable type that has the arguments applied.
  2099	        """
  2100	        if self.chk.in_checked_function():
  2101	            # Disable type errors during type inference. There may be errors
  2102	            # due to partial available context information at this time, but
  2103	            # these errors can be safely ignored as the arguments will be
  2104	            # inferred again later.
  2105	            with self.msg.filter_errors():
  2106	                arg_types = self.infer_arg_types_in_context(
  2107	                    callee_type, args, arg_kinds, formal_to_actual
  2108	                )
  2109
  2110	            arg_pass_nums = self.get_arg_infer_passes(
  2111	                callee_type, args, arg_types, formal_to_actual, len(args)
  2112	            )
  2113
  2114	            pass1_args: list[Type | None] = []
  2115	            for i, arg in enumerate(arg_types):
  2116	                if arg_pass_nums[i] > 1:
  2117	                    pass1_args.append(None)
  2118	                else:
  2119	                    pass1_args.append(arg)
  2120
  2121	            inferred_args, _ = infer_function_type_arguments(
  2122	                callee_type,
  2123	                pass1_args,
  2124	                arg_kinds,
  2125	                arg_names,
  2126	                formal_to_actual,
  2127	                context=self.argument_infer_context(),
  2128	                strict=self.chk.in_checked_function(),
  2129	            )
  2130
  2131	            if 2 in arg_pass_nums:
  2132	                # Second pass of type inference.
  2133	                (callee_type, inferred_args) = self.infer_function_type_arguments_pass2(
  2134	                    callee_type,
  2135	                    args,
  2136	                    arg_kinds,
  2137	                    arg_names,
  2138	                    formal_to_actual,
  2139	                    inferred_args,
  2140	                    need_refresh,
  2141	                    context,
  2142	                )
  2143
  2144	            if (
  2145	                callee_type.special_sig == "dict"
  2146	                and len(inferred_args) == 2
  2147	                and (ARG_NAMED in arg_kinds or ARG_STAR2 in arg_kinds)
  2148	            ):
  2149	                # HACK: Infer str key type for dict(...) with keyword args. The type system
  2150	                #       can't represent this so we special case it, as this is a pretty common
  2151	                #       thing. This doesn't quite work with all possible subclasses of dict
  2152	                #       if they shuffle type variables around, as we assume that there is a 1-1
  2153	                #       correspondence with dict type variables. This is a marginal issue and
  2154	                #       a little tricky to fix so it's left unfixed for now.
  2155	                first_arg = get_proper_type(inferred_args[0])
  2156	                if isinstance(first_arg, (NoneType, UninhabitedType)):
  2157	                    inferred_args[0] = self.named_type("builtins.str")
  2158	                elif not first_arg or not is_subtype(self.named_type("builtins.str"), first_arg):
  2159	                    self.chk.fail(message_registry.KEYWORD_ARGUMENT_REQUIRES_STR_KEY_TYPE, context)
  2160
  2161	            if not self.chk.options.old_type_inference and any(
  2162	                a is None
  2163	                or isinstance(get_proper_type(a), UninhabitedType)
  2164	                or set(get_type_vars(a)) & set(callee_type.variables)
  2165	                for a in inferred_args
  2166	            ):
  2167	                if need_refresh:
  2168	                    # Technically we need to refresh formal_to_actual after *each* inference pass,
  2169	                    # since each pass can expand ParamSpec or TypeVarTuple. Although such situations
  2170	                    # are very rare, not doing this can cause crashes.
  2171	                    formal_to_actual = map_actuals_to_formals(
  2172	                        arg_kinds,
  2173	                        arg_names,
  2174	                        callee_type.arg_kinds,
  2175	                        callee_type.arg_names,
  2176	                        lambda a: self.accept(args[a]),
  2177	                    )
  2178	                # If the regular two-phase inference didn't work, try inferring type
  2179	                # variables while allowing for polymorphic solutions, i.e. for solutions
  2180	                # potentially involving free variables.
  2181	                # TODO: support the similar inference for return type context.
  2182	                poly_inferred_args, free_vars = infer_function_type_arguments(
  2183	                    callee_type,
  2184	                    arg_types,
  2185	                    arg_kinds,
  2186	                    arg_names,
  2187	                    formal_to_actual,
  2188	                    context=self.argument_infer_context(),
  2189	                    strict=self.chk.in_checked_function(),
  2190	                    allow_polymorphic=True,
  2191	                )
  2192	                poly_callee_type = self.apply_generic_arguments(
  2193	                    callee_type, poly_inferred_args, context
  2194	                )
  2195	                # Try applying inferred polymorphic type if possible, e.g. Callable[[T], T] can
  2196	                # be interpreted as def [T] (T) -> T, but dict[T, T] cannot be expressed.
  2197	                applied = applytype.apply_poly(poly_callee_type, free_vars)
  2198	                if applied is not None and all(
  2199	                    a is not None and not isinstance(get_proper_type(a), UninhabitedType)
  2200	                    for a in poly_inferred_args
  2201	                ):

  2203	                    return applied
  2204	                # If it didn't work, erase free variables as uninhabited, to avoid confusing errors.
  2205	                unknown = UninhabitedType()
  2206	                unknown.ambiguous = True
  2207	                inferred_args = [
  2208	                    (
  2209	                        expand_type(

  2211	                        )
  2212	                        if a is not None
  2213	                        else None
  2214	                    )
  2215	                    for a in poly_inferred_args
  2216	                ]
  2217	        else:

  2219	            # type variables.
  2220	            inferred_args = [AnyType(TypeOfAny.unannotated)] * len(callee_type.variables)

  2223	    def infer_function_type_arguments_pass2(
  2224	        self,
  2225	        callee_type: CallableType,
  2226	        args: list[Expression],
  2227	        arg_kinds: list[ArgKind],
  2228	        arg_names: Sequence[str | None] | None,
  2229	        formal_to_actual: list[list[int]],
  2230	        old_inferred_args: Sequence[Type | None],
  2231	        need_refresh: bool,
  2232	        context: Context,
  2233	    ) -> tuple[CallableType, list[Type | None]]:
  2234	        """Perform second pass of generic function type argument inference.
  2235
  2236	        The second pass is needed for arguments with types such as Callable[[T], S],
  2237	        where both T and S are type variables, when the actual argument is a
  2238	        lambda with inferred types.  The idea is to infer the type variable T
  2239	        in the first pass (based on the types of other arguments).  This lets
  2240	        us infer the argument and return type of the lambda expression and
  2241	        thus also the type variable S in this second pass.
  2242
  2243	        Return (the callee with type vars applied, inferred actual arg types).
  2244	        """
  2245	        # None or erased types in inferred types mean that there was not enough
  2246	        # information to infer the argument. Replace them with None values so
  2247	        # that they are not applied yet below.
  2248	        inferred_args = list(old_inferred_args)
  2249	        for i, arg in enumerate(get_proper_types(inferred_args)):
  2250	            if isinstance(arg, (NoneType, UninhabitedType)) or has_erased_component(arg):
  2251	                inferred_args[i] = None
  2252	        callee_type = self.apply_generic_arguments(callee_type, inferred_args, context)
  2253	        if need_refresh:
  2254	            formal_to_actual = map_actuals_to_formals(
  2255	                arg_kinds,
  2256	                arg_names,
  2257	                callee_type.arg_kinds,
  2258	                callee_type.arg_names,
  2259	                lambda a: self.accept(args[a]),
  2260	            )
  2261
  2262	        # Same as during first pass, disable type errors (we still have partial context).
  2263	        with self.msg.filter_errors():
  2264	            arg_types = self.infer_arg_types_in_context(
  2265	                callee_type, args, arg_kinds, formal_to_actual
  2266	            )
  2267
  2268	        inferred_args, _ = infer_function_type_arguments(
  2269	            callee_type,
  2270	            arg_types,
  2271	            arg_kinds,
  2272	            arg_names,
  2273	            formal_to_actual,
  2274	            context=self.argument_infer_context(),
  2275	        )
  2276
  2277	        return callee_type, inferred_args
