   559	def show_help(
   560	    error: str | None = None,
   561	    topic: str | None = None,
   562	    parser: optparse.OptionParser | None = None,
   563	) -> None:
   564	    """Display an error message, or the named topic."""
   565	    assert error or topic or parser
   566
   567	    program_path = sys.argv[0]
   568	    if program_path.endswith(os.path.sep + "__main__.py"):
   569	        # The path is the main module of a package; get that path instead.
   570	        program_path = os.path.dirname(program_path)
   571	    program_name = os.path.basename(program_path)
   572	    if env.WINDOWS:
   573	        # entry_points={"console_scripts":...} on Windows makes files
   574	        # called coverage.exe, coverage3.exe, and coverage-3.5.exe. These
   575	        # invoke coverage-script.py, coverage3-script.py, and
   576	        # coverage-3.5-script.py.  argv[0] is the .py file, but we want to
   577	        # get back to the original form.
   578	        auto_suffix = "-script.py"
   579	        if program_name.endswith(auto_suffix):
   580	            program_name = program_name[:-len(auto_suffix)]
   581
   582	    help_params = dict(coverage.__dict__)
   583	    help_params["__url__"] = __url__
   584	    help_params["program_name"] = program_name
   585	    if CTRACER_FILE:
   586	        help_params["extension_modifier"] = "with C extension"
   587	    else:
   588	        help_params["extension_modifier"] = "without C extension"
   589
   590	    if error:
   591	        print(error, file=sys.stderr)
   592	        print(f"Use '{program_name} help' for help.", file=sys.stderr)
   593	    elif parser:
   594	        print(parser.format_help().strip())
   595	        print()
   596	    else:
   597	        assert topic is not None
   598	        help_msg = textwrap.dedent(HELP_TOPICS.get(topic, "")).strip()
   599	        if help_msg:
   600	            print(help_msg.format(**help_params))
   601	        else:
   602	            print(f"Don't know topic {topic!r}")
   603	    print("Full documentation is at {__url__}".format(**help_params))

   609	class CoverageScript:
   610	    """The command-line interface to coverage.py."""
   611
   612	    def __init__(self) -> None:
   613	        self.global_option = False
   614	        self.coverage: Coverage
   615
   616	    def command_line(self, argv: list[str]) -> int:
   617	        """The bulk of the command line interface to coverage.py.
   618
   619	        `argv` is the argument list to process.
   620
   621	        Returns 0 if all is well, 1 if something went wrong.
   622
   623	        """
   624	        # Collect the command-line options.
   625	        if not argv:
   626	            show_help(topic="minimum_help")
   627	            return OK
   628
   629	        # The command syntax we parse depends on the first argument.  Global
   630	        # switch syntax always starts with an option.
   631	        parser: optparse.OptionParser | None
   632	        self.global_option = argv[0].startswith("-")
   633	        if self.global_option:
   634	            parser = GlobalOptionParser()
   635	        else:
   636	            parser = COMMANDS.get(argv[0])
   637	            if not parser:
   638	                show_help(f"Unknown command: {argv[0]!r}")
   639	                return ERR
   640	            argv = argv[1:]
   641
   642	        ok, options, args = parser.parse_args_ok(argv)
   643	        if not ok:
   644	            return ERR
   645	        assert options is not None
   646
   647	        # Handle help and version.
   648	        if self.do_help(options, args, parser):
   649	            return OK
   650
   651	        # Listify the list options.
   652	        source = unshell_list(options.source)
   653	        omit = unshell_list(options.omit)
   654	        include = unshell_list(options.include)
   655	        debug = unshell_list(options.debug)
   656	        contexts = unshell_list(options.contexts)
   657
   658	        if options.concurrency is not None:
   659	            concurrency = options.concurrency.split(",")
   660	        else:
   661	            concurrency = None
   662
   663	        # Do something.
   664	        self.coverage = Coverage(
   665	            data_file=options.data_file or DEFAULT_DATAFILE,
   666	            data_suffix=options.parallel_mode,
   667	            cover_pylib=options.pylib,
   668	            timid=options.timid,
   669	            branch=options.branch,
   670	            config_file=options.rcfile,
   671	            source=source,
   672	            omit=omit,
   673	            include=include,
   674	            debug=debug,
   675	            concurrency=concurrency,
   676	            check_preimported=True,
   677	            context=options.context,
   678	            messages=not options.quiet,
   679	        )
   680
   681	        if options.action == "debug":
   682	            return self.do_debug(args)
   683
   684	        elif options.action == "erase":
   685	            self.coverage.erase()
   686	            return OK
   687
   688	        elif options.action == "run":
   689	            return self.do_run(options, args)
   690
   691	        elif options.action == "combine":
   692	            if options.append:
   693	                self.coverage.load()
   694	            data_paths = args or None
   695	            self.coverage.combine(data_paths, strict=True, keep=bool(options.keep))
   696	            self.coverage.save()
   697	            return OK
   698
   699	        # Remaining actions are reporting, with some common options.
   700	        report_args = dict(
   701	            morfs=unglob_args(args),
   702	            ignore_errors=options.ignore_errors,
   703	            omit=omit,
   704	            include=include,
   705	            contexts=contexts,
   706	        )
