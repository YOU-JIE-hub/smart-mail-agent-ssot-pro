  5613	        intersection type. Instead, it returns an instance of a new ad-hoc
  5614	        subclass of the given instances.

  5617	        theoretical subclass of the instances the user may be trying to use

  5622	        twice will return the same subclass of inst_1 and inst_2.

  5632	        # for isinstance() checks, etc). This way we will still type-check branches like
  5633	        # x: complex = 1
  5634	        # if isinstance(x, int):
  5635	        #     ...

  5637	        if is_proper_subtype(left, right, ignore_promotions=False):
  5638	            return left
  5639	        if is_proper_subtype(right, left, ignore_promotions=False):
  5640	            return right

  5748	    def partition_by_callable(
  5749	        self, typ: Type, unsound_partition: bool
  5750	    ) -> tuple[list[Type], list[Type]]:
  5751	        """Partitions a type into callable subtypes and uncallable subtypes.
  5752
  5753	        Thus, given:
  5754	        `callables, uncallables = partition_by_callable(type)`
  5755
  5756	        If we assert `callable(type)` then `type` has type Union[*callables], and
  5757	        If we assert `not callable(type)` then `type` has type Union[*uncallables]
  5758
  5759	        If unsound_partition is set, assume that anything that is not
  5760	        clearly callable is in fact not callable. Otherwise we generate a
  5761	        new subtype that *is* callable.
  5762
  5763	        Guaranteed to not return [], [].
  5764	        """
  5765	        typ = get_proper_type(typ)
  5766
  5767	        if isinstance(typ, (FunctionLike, TypeType)):
  5768	            return [typ], []
  5769
  5770	        if isinstance(typ, AnyType):
  5771	            return [typ], [typ]
  5772
  5773	        if isinstance(typ, NoneType):
  5774	            return [], [typ]
  5775
  5776	        if isinstance(typ, UnionType):
  5777	            callables = []
  5778	            uncallables = []
  5779	            for subtype in typ.items:
  5780	                # Use unsound_partition when handling unions in order to
  5781	                # allow the expected type discrimination.
  5782	                subcallables, subuncallables = self.partition_by_callable(
  5783	                    subtype, unsound_partition=True
  5784	                )
  5785	                callables.extend(subcallables)
  5786	                uncallables.extend(subuncallables)
  5787	            return callables, uncallables
  5788
  5789	        if isinstance(typ, TypeVarType):
  5790	            # We could do better probably?
  5791	            # Refine the type variable's bound as our type in the case that
  5792	            # callable() is true. This unfortunately loses the information that
  5793	            # the type is a type variable in that branch.
  5794	            # This matches what is done for isinstance, but it may be possible to
  5795	            # do better.
  5796	            # If it is possible for the false branch to execute, return the original
  5797	            # type to avoid losing type information.
  5798	            callables, uncallables = self.partition_by_callable(
  5799	                erase_to_union_or_bound(typ), unsound_partition
  5800	            )
  5801	            uncallables = [typ] if uncallables else []
  5802	            return callables, uncallables
  5803
  5804	        # A TupleType is callable if its fallback is, but needs special handling
  5805	        # when we dummy up a new type.
  5806	        ityp = typ
  5807	        if isinstance(typ, TupleType):
  5808	            ityp = tuple_fallback(typ)
  5809
  5810	        if isinstance(ityp, Instance):
  5811	            method = ityp.type.get_method("__call__")
  5812	            if method and method.type:
  5813	                callables, uncallables = self.partition_by_callable(
  5814	                    method.type, unsound_partition=False
  5815	                )
  5816	                if callables and not uncallables:
  5817	                    # Only consider the type callable if its __call__ method is
  5818	                    # definitely callable.
  5819	                    return [typ], []
  5820
  5821	            if not unsound_partition:
  5822	                fake = self.make_fake_callable(ityp)
  5823	                if isinstance(typ, TupleType):
  5824	                    fake.type.tuple_type = TupleType(typ.items, fake)
  5825	                    return [fake.type.tuple_type], [typ]
  5826	                return [fake], [typ]
  5827
  5828	        if unsound_partition:
  5829	            return [], [typ]
  5830	        else:
  5831	            # We don't know how properly make the type callable.
  5832	            return [typ], [typ]

  5834	    def conditional_callable_type_map(

  5837	        """Takes in an expression and the current type of the expression.
  5838
  5839	        Returns a 2-tuple: The first element is a map from the expression to
  5840	        the restricted type if it were callable. The second element is a
  5841	        map from the expression to the type it would hold if it weren't
  5842	        callable.
  5843	        """
  5844	        if not current_type:
  5845	            return {}, {}
  5846
  5847	        if isinstance(get_proper_type(current_type), AnyType):
  5848	            return {}, {}

  5854	            uncallable_map = {expr: UnionType.make_union(uncallables)} if uncallables else None
  5855	            return callable_map, uncallable_map

  5857	        elif callables:

  5860	        return None, {}

  5917	    def check_for_truthy_type(self, t: Type, expr: Expression) -> None:
  5918	        """

  5920
  5921	        Used in checks like::
  5922
  5923	            if x: # <---
  5924
  5925	            not x  # <---
  5926	        """
  5927	        if not state.strict_optional:
  5928	            return  # if everything can be None, all bets are off
  5929
  5930	        t = get_proper_type(t)
  5931	        if not self._is_truthy_type(t):
  5932	            return
  5933
  5934	        def format_expr_type() -> str:
  5935	            typ = format_type(t, self.options)
  5936	            if isinstance(expr, MemberExpr):
  5937	                return f'Member "{expr.name}" has type {typ}'
  5938	            elif isinstance(expr, RefExpr) and expr.fullname:
  5939	                return f'"{expr.fullname}" has type {typ}'
  5940	            elif isinstance(expr, CallExpr):
  5941	                if isinstance(expr.callee, MemberExpr):
  5942	                    return f'"{expr.callee.name}" returns {typ}'
  5943	                elif isinstance(expr.callee, RefExpr) and expr.callee.fullname:
  5944	                    return f'"{expr.callee.fullname}" returns {typ}'

  5950	            if isinstance(expr, (NameExpr, MemberExpr)):
  5951	                return f'"{expr.name}"'

  5971	    def find_type_equals_check(
  5972	        self, node: ComparisonExpr, expr_indices: list[int]
  5973	    ) -> tuple[TypeMap, TypeMap]:
  5974	        """Narrow types based on any checks of the type ``type(x) == T``
  5975
  5976	        Args:
  5977	            node: The node that might contain the comparison
  5978	            expr_indices: The list of indices of expressions in ``node`` that are being
  5979	                compared
  5980	        """
  5981
  5982	        def is_type_call(expr: CallExpr) -> bool:
  5983	            """Is expr a call to type with one argument?"""
  5984	            return refers_to_fullname(expr.callee, "builtins.type") and len(expr.args) == 1
  5985
  5986	        # exprs that are being passed into type
  5987	        exprs_in_type_calls: list[Expression] = []
  5988	        # type that is being compared to type(expr)
  5989	        type_being_compared: list[TypeRange] | None = None
  5990	        # whether the type being compared to is final
  5991	        is_final = False
  5992
  5993	        for index in expr_indices:
  5994	            expr = node.operands[index]
  5995
  5996	            if isinstance(expr, CallExpr) and is_type_call(expr):
  5997	                exprs_in_type_calls.append(expr.args[0])
  5998	            else:
  5999	                current_type = self.get_isinstance_type(expr)
  6000	                if current_type is None:
  6001	                    continue
  6002	                if type_being_compared is not None:
  6003	                    # It doesn't really make sense to have several types being
  6004	                    # compared to the output of type (like type(x) == int == str)
  6005	                    # because whether that's true is solely dependent on what the
  6006	                    # types being compared are, so we don't try to narrow types any
  6007	                    # further because we can't really get any information about the
  6008	                    # type of x from that check
  6009	                    return {}, {}
  6010	                else:
  6011	                    if isinstance(expr, RefExpr) and isinstance(expr.node, TypeInfo):
  6012	                        is_final = expr.node.is_final
  6013	                    type_being_compared = current_type
  6014
  6015	        if not exprs_in_type_calls:
  6016	            return {}, {}
  6017
  6018	        if_maps: list[TypeMap] = []
  6019	        else_maps: list[TypeMap] = []
  6020	        for expr in exprs_in_type_calls:
  6021	            current_if_type, current_else_type = self.conditional_types_with_intersection(
  6022	                self.lookup_type(expr), type_being_compared, expr
  6023	            )
  6024	            current_if_map, current_else_map = conditional_types_to_typemaps(
  6025	                expr, current_if_type, current_else_type
  6026	            )
  6027	            if_maps.append(current_if_map)
  6028	            else_maps.append(current_else_map)
  6029
  6030	        def combine_maps(list_maps: list[TypeMap]) -> TypeMap:
  6031	            """Combine all typemaps in list_maps into one typemap"""
  6032	            if all(m is None for m in list_maps):
  6033	                return None
  6034	            result_map = {}
  6035	            for d in list_maps:
  6036	                if d is not None:
  6037	                    result_map.update(d)
  6038	            return result_map
  6039
  6040	        if_map = combine_maps(if_maps)
  6041	        # type(x) == T is only true when x has the same type as T, meaning
  6042	        # that it can be false if x is an instance of a subclass of T. That means
  6043	        # we can't do any narrowing in the else case unless T is final, in which
  6044	        # case T can't be subclassed
  6045	        if is_final:
  6046	            else_map = combine_maps(else_maps)
  6047	        else:
  6048	            else_map = {}
  6049	        return if_map, else_map

  6051	    def find_isinstance_check(
  6052	        self, node: Expression, *, in_boolean_context: bool = True
  6053	    ) -> tuple[TypeMap, TypeMap]:
  6054	        """Find any isinstance checks (within a chain of ands).  Includes
  6055	        implicit and explicit checks for None and calls to callable.
  6056	        Also includes TypeGuard and TypeIs functions.
  6057
  6058	        Return value is a map of variables to their types if the condition
  6059	        is true and a map of variables to their types if the condition is false.

  6061	        If either of the values in the tuple is None, then that particular
  6062	        branch can never occur.

  6073	            node, in_boolean_context=in_boolean_context
  6074	        )
  6075	        new_if_map = self.propagate_up_typemap_info(if_map)
  6076	        new_else_map = self.propagate_up_typemap_info(else_map)
  6077	        return new_if_map, new_else_map

  6079	    def find_isinstance_check_helper(
  6080	        self, node: Expression, *, in_boolean_context: bool = True
  6081	    ) -> tuple[TypeMap, TypeMap]:
  6082	        if is_true_literal(node):
  6083	            return {}, None
  6084	        if is_false_literal(node):
  6085	            return None, {}
  6086
  6087	        if isinstance(node, CallExpr) and len(node.args) != 0:
  6088	            expr = collapse_walrus(node.args[0])
  6089	            if refers_to_fullname(node.callee, "builtins.isinstance"):
  6090	                if len(node.args) != 2:  # the error will be reported elsewhere
  6091	                    return {}, {}
  6092	                if literal(expr) == LITERAL_TYPE:
  6093	                    return conditional_types_to_typemaps(
  6094	                        expr,
  6095	                        *self.conditional_types_with_intersection(
  6096	                            self.lookup_type(expr), self.get_isinstance_type(node.args[1]), expr
  6097	                        ),
  6098	                    )
  6099	            elif refers_to_fullname(node.callee, "builtins.issubclass"):
  6100	                if len(node.args) != 2:  # the error will be reported elsewhere
  6101	                    return {}, {}
  6102	                if literal(expr) == LITERAL_TYPE:
  6103	                    return self.infer_issubclass_maps(node, expr)
  6104	            elif refers_to_fullname(node.callee, "builtins.callable"):
  6105	                if len(node.args) != 1:  # the error will be reported elsewhere
  6106	                    return {}, {}
  6107	                if literal(expr) == LITERAL_TYPE:
  6108	                    vartype = self.lookup_type(expr)
  6109	                    return self.conditional_callable_type_map(expr, vartype)
  6110	            elif refers_to_fullname(node.callee, "builtins.hasattr"):
  6111	                if len(node.args) != 2:  # the error will be reported elsewhere
  6112	                    return {}, {}
  6113	                attr = try_getting_str_literals(node.args[1], self.lookup_type(node.args[1]))
  6114	                if literal(expr) == LITERAL_TYPE and attr and len(attr) == 1:
  6115	                    return self.hasattr_type_maps(expr, self.lookup_type(expr), attr[0])
  6116	            elif isinstance(node.callee, RefExpr):
  6117	                if node.callee.type_guard is not None or node.callee.type_is is not None:
  6118	                    # TODO: Follow *args, **kwargs
  6119	                    if node.arg_kinds[0] != nodes.ARG_POS:
  6120	                        # the first argument might be used as a kwarg
  6121	                        called_type = get_proper_type(self.lookup_type(node.callee))
  6122
  6123	                        # TODO: there are some more cases in check_call() to handle.
  6124	                        if isinstance(called_type, Instance):
  6125	                            call = find_member(
  6126	                                "__call__", called_type, called_type, is_operator=True
  6127	                            )
  6128	                            if call is not None:
  6129	                                called_type = get_proper_type(call)
  6130
  6131	                        # *assuming* the overloaded function is correct, there's a couple cases:
  6132	                        #  1) The first argument has different names, but is pos-only. We don't
  6133	                        #     care about this case, the argument must be passed positionally.
  6134	                        #  2) The first argument allows keyword reference, therefore must be the
  6135	                        #     same between overloads.
  6136	                        if isinstance(called_type, (CallableType, Overloaded)):
  6137	                            name = called_type.items[0].arg_names[0]
  6138	                            if name in node.arg_names:
  6139	                                idx = node.arg_names.index(name)
  6140	                                # we want the idx-th variable to be narrowed
  6141	                                expr = collapse_walrus(node.args[idx])
  6142	                            else:
  6143	                                kind = (
  6144	                                    "guard" if node.callee.type_guard is not None else "narrower"
  6145	                                )
  6146	                                self.fail(
  6147	                                    message_registry.TYPE_GUARD_POS_ARG_REQUIRED.format(kind), node
  6148	                                )
  6149	                                return {}, {}
  6150	                    if literal(expr) == LITERAL_TYPE:
  6151	                        # Note: we wrap the target type, so that we can special case later.
  6152	                        # Namely, for isinstance() we use a normal meet, while TypeGuard is
  6153	                        # considered "always right" (i.e. even if the types are not overlapping).
  6154	                        # Also note that a care must be taken to unwrap this back at read places
  6155	                        # where we use this to narrow down declared type.
  6156	                        if node.callee.type_guard is not None:
  6157	                            return {expr: TypeGuardedType(node.callee.type_guard)}, {}
  6158	                        else:
  6159	                            assert node.callee.type_is is not None
  6160	                            return conditional_types_to_typemaps(
  6161	                                expr,
  6162	                                *self.conditional_types_with_intersection(
  6163	                                    self.lookup_type(expr),
  6164	                                    [TypeRange(node.callee.type_is, is_upper_bound=False)],
  6165	                                    expr,
  6166	                                ),
  6167	                            )
  6168	        elif isinstance(node, ComparisonExpr):
  6169	            return self.comparison_type_narrowing_helper(node)
  6170	        elif isinstance(node, AssignmentExpr):
  6171	            if_map: dict[Expression, Type] | None
  6172	            else_map: dict[Expression, Type] | None
  6173	            if_map = {}
  6174	            else_map = {}
  6175
  6176	            if_assignment_map, else_assignment_map = self.find_isinstance_check(node.target)
  6177
  6178	            if if_assignment_map is not None:
  6179	                if_map.update(if_assignment_map)
  6180	            if else_assignment_map is not None:
  6181	                else_map.update(else_assignment_map)
  6182
  6183	            if_condition_map, else_condition_map = self.find_isinstance_check(
  6184	                node.value, in_boolean_context=False
  6185	            )
  6186
  6187	            if if_condition_map is not None:
  6188	                if_map.update(if_condition_map)
  6189	            if else_condition_map is not None:
  6190	                else_map.update(else_condition_map)
  6191
  6192	            return (
  6193	                (None if if_assignment_map is None or if_condition_map is None else if_map),
  6194	                (None if else_assignment_map is None or else_condition_map is None else else_map),
  6195	            )
  6196	        elif isinstance(node, OpExpr) and node.op == "and":
  6197	            left_if_vars, left_else_vars = self.find_isinstance_check(node.left)
  6198	            right_if_vars, right_else_vars = self.find_isinstance_check(node.right)

  6200	            # (e1 and e2) is true if both e1 and e2 are true,
  6201	            # and false if at least one of e1 and e2 is false.
  6202	            return (
  6203	                and_conditional_maps(left_if_vars, right_if_vars),
  6204	                # Note that if left else type is Any, we can't add any additional
  6205	                # types to it, since the right maps were computed assuming
  6206	                # the left is True, which may be not the case in the else branch.
  6207	                or_conditional_maps(left_else_vars, right_else_vars, coalesce_any=True),
  6208	            )
  6209	        elif isinstance(node, OpExpr) and node.op == "or":
  6210	            left_if_vars, left_else_vars = self.find_isinstance_check(node.left)
  6211	            right_if_vars, right_else_vars = self.find_isinstance_check(node.right)
  6212
  6213	            # (e1 or e2) is true if at least one of e1 or e2 is true,
  6214	            # and false if both e1 and e2 are false.
  6215	            return (
  6216	                or_conditional_maps(left_if_vars, right_if_vars),
  6217	                and_conditional_maps(left_else_vars, right_else_vars),
  6218	            )
  6219	        elif isinstance(node, UnaryExpr) and node.op == "not":
  6220	            left, right = self.find_isinstance_check(node.expr)
  6221	            return right, left
  6222	        elif (
  6223	            literal(node) == LITERAL_TYPE
  6224	            and self.has_type(node)
  6225	            and self.can_be_narrowed_with_len(self.lookup_type(node))

  6227	            and not custom_special_method(self.lookup_type(node), "__bool__")
  6228	            and self.options.strict_optional
  6229	        ):
  6230	            # Combine a `len(x) > 0` check with the default logic below.
  6231	            yes_type, no_type = self.narrow_with_len(self.lookup_type(node), ">", 0)
  6232	            if yes_type is not None:
  6233	                yes_type = true_only(yes_type)
  6234	            else:
  6235	                yes_type = UninhabitedType()
  6236	            if no_type is not None:
  6237	                no_type = false_only(no_type)
  6238	            else:
  6239	                no_type = UninhabitedType()
  6240	            if_map = {node: yes_type} if not isinstance(yes_type, UninhabitedType) else None
  6241	            else_map = {node: no_type} if not isinstance(no_type, UninhabitedType) else None
  6242	            return if_map, else_map

  6244	        # Restrict the type of the variable to True-ish/False-ish in the if and else branches
  6245	        # respectively
  6246	        original_vartype = self.lookup_type(node)
  6247	        if in_boolean_context:
  6248	            # We don't check `:=` values in expressions like `(a := A())`,
  6249	            # because they produce two error messages.
  6250	            self.check_for_truthy_type(original_vartype, node)
  6251	        vartype = try_expanding_sum_type_to_union(original_vartype, "builtins.bool")
  6252
  6253	        if_type = true_only(vartype)
  6254	        else_type = false_only(vartype)

  6256	        else_map = {node: else_type} if not isinstance(else_type, UninhabitedType) else None
  6257	        return if_map, else_map

  6259	    def comparison_type_narrowing_helper(self, node: ComparisonExpr) -> tuple[TypeMap, TypeMap]:
  6260	        """Infer type narrowing from a comparison expression."""
  6261	        # Step 1: Obtain the types of each operand and whether or not we can
  6262	        # narrow their types. (For example, we shouldn't try narrowing the
  6263	        # types of literal string or enum expressions).
  6264
  6265	        operands = [collapse_walrus(x) for x in node.operands]
  6266	        operand_types = []
  6267	        narrowable_operand_index_to_hash = {}
  6268	        for i, expr in enumerate(operands):
  6269	            if not self.has_type(expr):
  6270	                return {}, {}
  6271	            expr_type = self.lookup_type(expr)
  6272	            operand_types.append(expr_type)
  6273
  6274	            if (
  6275	                literal(expr) == LITERAL_TYPE
  6276	                and not is_literal_none(expr)
  6277	                and not self.is_literal_enum(expr)
  6278	            ):
  6279	                h = literal_hash(expr)
  6280	                if h is not None:
  6281	                    narrowable_operand_index_to_hash[i] = h
  6282
  6283	        # Step 2: Group operands chained by either the 'is' or '==' operands
  6284	        # together. For all other operands, we keep them in groups of size 2.
  6285	        # So the expression:
  6286	        #
  6287	        #   x0 == x1 == x2 < x3 < x4 is x5 is x6 is not x7 is not x8
  6288	        #
  6289	        # ...is converted into the simplified operator list:
  6290	        #
  6291	        #  [("==", [0, 1, 2]), ("<", [2, 3]), ("<", [3, 4]),
  6292	        #   ("is", [4, 5, 6]), ("is not", [6, 7]), ("is not", [7, 8])]
  6293	        #
  6294	        # We group identity/equality expressions so we can propagate information
  6295	        # we discover about one operand across the entire chain. We don't bother
  6296	        # handling 'is not' and '!=' chains in a special way: those are very rare
  6297	        # in practice.
  6298
  6299	        simplified_operator_list = group_comparison_operands(
  6300	            node.pairwise(), narrowable_operand_index_to_hash, {"==", "is"}
  6301	        )
  6302
  6303	        # Step 3: Analyze each group and infer more precise type maps for each
  6304	        # assignable operand, if possible. We combine these type maps together
  6305	        # in the final step.
  6306
  6307	        partial_type_maps = []
  6308	        for operator, expr_indices in simplified_operator_list:
  6309	            if operator in {"is", "is not", "==", "!="}:
  6310	                if_map, else_map = self.equality_type_narrowing_helper(
  6311	                    node,
  6312	                    operator,
  6313	                    operands,
  6314	                    operand_types,
  6315	                    expr_indices,
  6316	                    narrowable_operand_index_to_hash,
  6317	                )
  6318	            elif operator in {"in", "not in"}:
  6319	                assert len(expr_indices) == 2
  6320	                left_index, right_index = expr_indices
  6321	                item_type = operand_types[left_index]
  6322	                iterable_type = operand_types[right_index]
  6323
  6324	                if_map, else_map = {}, {}
  6325
  6326	                if left_index in narrowable_operand_index_to_hash:
  6327	                    # We only try and narrow away 'None' for now
  6328	                    if is_overlapping_none(item_type):
  6329	                        collection_item_type = get_proper_type(builtin_item_type(iterable_type))
  6330	                        if (
  6331	                            collection_item_type is not None
  6332	                            and not is_overlapping_none(collection_item_type)
  6333	                            and not (
  6334	                                isinstance(collection_item_type, Instance)
  6335	                                and collection_item_type.type.fullname == "builtins.object"
  6336	                            )
  6337	                            and is_overlapping_erased_types(item_type, collection_item_type)
  6338	                        ):
  6339	                            if_map[operands[left_index]] = remove_optional(item_type)
  6340
  6341	                if right_index in narrowable_operand_index_to_hash:
  6342	                    if_type, else_type = self.conditional_types_for_iterable(
  6343	                        item_type, iterable_type
  6344	                    )
  6345	                    expr = operands[right_index]
  6346	                    if if_type is None:
  6347	                        if_map = None
  6348	                    else:
  6349	                        if_map[expr] = if_type
  6350	                    if else_type is None:
  6351	                        else_map = None
  6352	                    else:
  6353	                        else_map[expr] = else_type
  6354
  6355	            else:
  6356	                if_map = {}
  6357	                else_map = {}
  6358
  6359	            if operator in {"is not", "!=", "not in"}:
  6360	                if_map, else_map = else_map, if_map
  6361
  6362	            partial_type_maps.append((if_map, else_map))
  6363
  6364	        # If we have found non-trivial restrictions from the regular comparisons,
  6365	        # then return soon. Otherwise try to infer restrictions involving `len(x)`.
  6366	        # TODO: support regular and len() narrowing in the same chain.
  6367	        if any(m != ({}, {}) for m in partial_type_maps):
  6368	            return reduce_conditional_maps(partial_type_maps)
  6369	        else:
  6370	            # Use meet for `and` maps to get correct results for chained checks
  6371	            # like `if 1 < len(x) < 4: ...`
  6372	            return reduce_conditional_maps(self.find_tuple_len_narrowing(node), use_meet=True)

  6374	    def equality_type_narrowing_helper(
  6375	        self,
  6376	        node: ComparisonExpr,
  6377	        operator: str,
  6378	        operands: list[Expression],
  6379	        operand_types: list[Type],
  6380	        expr_indices: list[int],
  6381	        narrowable_operand_index_to_hash: dict[int, tuple[Key, ...]],
  6382	    ) -> tuple[TypeMap, TypeMap]:
  6383	        """Calculate type maps for '==', '!=', 'is' or 'is not' expression."""
  6384	        # is_valid_target:
  6385	        #   Controls which types we're allowed to narrow exprs to. Note that
  6386	        #   we cannot use 'is_literal_type_like' in both cases since doing
  6387	        #   'x = 10000 + 1; x is 10001' is not always True in all Python
  6388	        #   implementations.
  6389	        #
  6390	        # coerce_only_in_literal_context:
  6391	        #   If true, coerce types into literal types only if one or more of
  6392	        #   the provided exprs contains an explicit Literal type. This could
  6393	        #   technically be set to any arbitrary value, but it seems being liberal
  6394	        #   with narrowing when using 'is' and conservative when using '==' seems
  6395	        #   to break the least amount of real-world code.
  6396	        #
  6397	        # should_narrow_by_identity:
  6398	        #   Set to 'false' only if the user defines custom __eq__ or __ne__ methods
  6399	        #   that could cause identity-based narrowing to produce invalid results.
  6400	        if operator in {"is", "is not"}:
  6401	            is_valid_target: Callable[[Type], bool] = is_singleton_type
  6402	            coerce_only_in_literal_context = False
  6403	            should_narrow_by_identity = True
  6404	        else:
  6405
  6406	            def is_exactly_literal_type(t: Type) -> bool:
  6407	                return isinstance(get_proper_type(t), LiteralType)
  6408
  6409	            def has_no_custom_eq_checks(t: Type) -> bool:
  6410	                return not custom_special_method(
  6411	                    t, "__eq__", check_all=False
  6412	                ) and not custom_special_method(t, "__ne__", check_all=False)
  6413
  6414	            is_valid_target = is_exactly_literal_type
  6415	            coerce_only_in_literal_context = True
  6416

  6431	                coerce_only_in_literal_context,

  6445	    def propagate_up_typemap_info(self, new_types: TypeMap) -> TypeMap:
  6446	        """Attempts refining parent expressions of any MemberExpr or IndexExprs in new_types.
  6447
  6448	        Specifically, this function accepts two mappings of expression to original types:
  6449	        the original mapping (existing_types), and a new mapping (new_types) intended to
  6450	        update the original.
  6451
  6452	        This function iterates through new_types and attempts to use the information to try
  6453	        refining any parent types that happen to be unions.
  6454
  6455	        For example, suppose there are two types "A = Tuple[int, int]" and "B = Tuple[str, str]".
  6456	        Next, suppose that 'new_types' specifies the expression 'foo[0]' has a refined type
  6457	        of 'int' and that 'foo' was previously deduced to be of type Union[A, B].
  6458
  6459	        Then, this function will observe that since A[0] is an int and B[0] is not, the type of
  6460	        'foo' can be further refined from Union[A, B] into just B.
  6461
  6462	        We perform this kind of "parent narrowing" for member lookup expressions and indexing
  6463	        expressions into tuples, namedtuples, and typeddicts. We repeat this narrowing
  6464	        recursively if the parent is also a "lookup expression". So for example, if we have
  6465	        the expression "foo['bar'].baz[0]", we'd potentially end up refining types for the
  6466	        expressions "foo", "foo['bar']", and "foo['bar'].baz".
  6467
  6468	        We return the newly refined map. This map is guaranteed to be a superset of 'new_types'.
  6469	        """
  6470	        if new_types is None:
  6471	            return None
  6472	        output_map = {}
  6473	        for expr, expr_type in new_types.items():
  6474	            # The original inferred type should always be present in the output map, of course
  6475	            output_map[expr] = expr_type
  6476
  6477	            # Next, try using this information to refine the parent types, if applicable.
  6478	            new_mapping = self.refine_parent_types(expr, expr_type)
  6479	            for parent_expr, proposed_parent_type in new_mapping.items():
  6480	                # We don't try inferring anything if we've already inferred something for
  6481	                # the parent expression.
  6482	                # TODO: Consider picking the narrower type instead of always discarding this?
  6483	                if parent_expr in new_types:
  6484	                    continue
  6485	                output_map[parent_expr] = proposed_parent_type
  6486	        return output_map

  6488	    def refine_parent_types(self, expr: Expression, expr_type: Type) -> Mapping[Expression, Type]:
  6489	        """Checks if the given expr is a 'lookup operation' into a union and iteratively refines
  6490	        the parent types based on the 'expr_type'.
  6491
  6492	        For example, if 'expr' is an expression like 'a.b.c.d', we'll potentially return refined
  6493	        types for expressions 'a', 'a.b', and 'a.b.c'.
  6494
  6495	        For more details about what a 'lookup operation' is and how we use the expr_type to refine
  6496	        the parent types of lookup_expr, see the docstring in 'propagate_up_typemap_info'.
  6497	        """
  6498	        output: dict[Expression, Type] = {}
  6499
  6500	        # Note: parent_expr and parent_type are progressively refined as we crawl up the
  6501	        # parent lookup chain.
  6502	        while True:
  6503	            # First, check if this expression is one that's attempting to
  6504	            # "lookup" some key in the parent type. If so, save the parent type
  6505	            # and create function that will try replaying the same lookup
  6506	            # operation against arbitrary types.
  6507	            if isinstance(expr, MemberExpr):
  6508	                parent_expr = self._propagate_walrus_assignments(expr.expr, output)
  6509	                parent_type = self.lookup_type_or_none(parent_expr)
  6510	                member_name = expr.name
  6511
  6512	                def replay_lookup(new_parent_type: ProperType) -> Type | None:
  6513	                    with self.msg.filter_errors() as w:
  6514	                        member_type = analyze_member_access(
  6515	                            name=member_name,
  6516	                            typ=new_parent_type,
  6517	                            context=parent_expr,
  6518	                            is_lvalue=False,
  6519	                            is_super=False,
  6520	                            is_operator=False,
  6521	                            original_type=new_parent_type,
  6522	                            chk=self,
  6523	                            in_literal_context=False,
  6524	                        )
  6525	                    if w.has_new_errors():
  6526	                        return None
  6527	                    else:
  6528	                        return member_type
  6529
  6530	            elif isinstance(expr, IndexExpr):
  6531	                parent_expr = self._propagate_walrus_assignments(expr.base, output)
  6532	                parent_type = self.lookup_type_or_none(parent_expr)
  6533
  6534	                self._propagate_walrus_assignments(expr.index, output)
  6535	                index_type = self.lookup_type_or_none(expr.index)
  6536	                if index_type is None:
  6537	                    return output
  6538
  6539	                str_literals = try_getting_str_literals_from_type(index_type)
  6540	                if str_literals is not None:
  6541	                    # Refactoring these two indexing replay functions is surprisingly
  6542	                    # tricky -- see https://github.com/python/mypy/pull/7917, which
  6543	                    # was blocked by https://github.com/mypyc/mypyc/issues/586
  6544	                    def replay_lookup(new_parent_type: ProperType) -> Type | None:
  6545	                        if not isinstance(new_parent_type, TypedDictType):
  6546	                            return None
  6547	                        try:
  6548	                            assert str_literals is not None
  6549	                            member_types = [new_parent_type.items[key] for key in str_literals]
  6550	                        except KeyError:
  6551	                            return None
  6552	                        return make_simplified_union(member_types)
  6553
  6554	                else:
  6555	                    int_literals = try_getting_int_literals_from_type(index_type)
  6556	                    if int_literals is not None:
  6557
  6558	                        def replay_lookup(new_parent_type: ProperType) -> Type | None:
  6559	                            if not isinstance(new_parent_type, TupleType):
  6560	                                return None
  6561	                            try:
  6562	                                assert int_literals is not None
  6563	                                member_types = [new_parent_type.items[key] for key in int_literals]
  6564	                            except IndexError:
  6565	                                return None
  6566	                            return make_simplified_union(member_types)
  6567
  6568	                    else:
  6569	                        return output
  6570	            else:
  6571	                return output
  6572
  6573	            # If we somehow didn't previously derive the parent type, abort completely
  6574	            # with what we have so far: something went wrong at an earlier stage.
  6575	            if parent_type is None:
  6576	                return output
  6577
  6578	            # We currently only try refining the parent type if it's a Union.
  6579	            # If not, there's no point in trying to refine any further parents
  6580	            # since we have no further information we can use to refine the lookup
  6581	            # chain, so we end early as an optimization.
  6582	            parent_type = get_proper_type(parent_type)
  6583	            if not isinstance(parent_type, UnionType):
  6584	                return output
  6585
  6586	            # Take each element in the parent union and replay the original lookup procedure
  6587	            # to figure out which parents are compatible.
  6588	            new_parent_types = []
  6589	            for item in flatten_nested_unions(parent_type.items):
  6590	                member_type = replay_lookup(get_proper_type(item))
  6591	                if member_type is None:
  6592	                    # We were unable to obtain the member type. So, we give up on refining this
  6593	                    # parent type entirely and abort.
  6594	                    return output
  6595
  6596	                if is_overlapping_types(member_type, expr_type):
  6597	                    new_parent_types.append(item)
  6598
  6599	            # If none of the parent types overlap (if we derived an empty union), something
  6600	            # went wrong. We should never hit this case, but deriving the uninhabited type or
  6601	            # reporting an error both seem unhelpful. So we abort.
  6602	            if not new_parent_types:
  6603	                return output
  6604
  6605	            expr = parent_expr
  6606	            expr_type = output[parent_expr] = make_simplified_union(new_parent_types)

  6626	    def refine_identity_comparison_expression(
  6627	        self,
  6628	        operands: list[Expression],
  6629	        operand_types: list[Type],
  6630	        chain_indices: list[int],
  6631	        narrowable_operand_indices: AbstractSet[int],
  6632	        is_valid_target: Callable[[ProperType], bool],
  6633	        coerce_only_in_literal_context: bool,
  6634	    ) -> tuple[TypeMap, TypeMap]:
  6635	        """Produce conditional type maps refining expressions by an identity/equality comparison.
  6636
  6637	        The 'operands' and 'operand_types' lists should be the full list of operands used
  6638	        in the overall comparison expression. The 'chain_indices' list is the list of indices
  6639	        actually used within this identity comparison chain.
  6640
  6641	        So if we have the expression:
  6642
  6643	            a <= b is c is d <= e
  6644
  6645	        ...then 'operands' and 'operand_types' would be lists of length 5 and 'chain_indices'
  6646	        would be the list [1, 2, 3].
  6647
  6648	        The 'narrowable_operand_indices' parameter is the set of all indices we are allowed
  6649	        to refine the types of: that is, all operands that will potentially be a part of
  6650	        the output TypeMaps.
  6651
  6652	        Although this function could theoretically try setting the types of the operands
  6653	        in the chains to the meet, doing that causes too many issues in real-world code.
  6654	        Instead, we use 'is_valid_target' to identify which of the given chain types
  6655	        we could plausibly use as the refined type for the expressions in the chain.
  6656
  6657	        Similarly, 'coerce_only_in_literal_context' controls whether we should try coercing
  6658	        expressions in the chain to a Literal type. Performing this coercion is sometimes
  6659	        too aggressive of a narrowing, depending on context.
  6660	        """
  6661	        should_coerce = True
  6662	        if coerce_only_in_literal_context:
  6663
  6664	            def should_coerce_inner(typ: Type) -> bool:
  6665	                typ = get_proper_type(typ)
  6666	                return is_literal_type_like(typ) or (
  6667	                    isinstance(typ, Instance) and typ.type.is_enum
  6668	                )
  6669
  6670	            should_coerce = any(should_coerce_inner(operand_types[i]) for i in chain_indices)
  6671
  6672	        target: Type | None = None
  6673	        possible_target_indices = []
  6674	        for i in chain_indices:
  6675	            expr_type = operand_types[i]
  6676	            if should_coerce:
  6677	                expr_type = coerce_to_literal(expr_type)
  6678	            if not is_valid_target(get_proper_type(expr_type)):
  6679	                continue
  6680	            if target and not is_same_type(target, expr_type):
  6681	                # We have multiple disjoint target types. So the 'if' branch
  6682	                # must be unreachable.
  6683	                return None, {}
  6684	            target = expr_type
  6685	            possible_target_indices.append(i)
  6686
  6687	        # There's nothing we can currently infer if none of the operands are valid targets,
  6688	        # so we end early and infer nothing.
  6689	        if target is None:
  6690	            return {}, {}
  6691
  6692	        # If possible, use an unassignable expression as the target.
  6693	        # We skip refining the type of the target below, so ideally we'd
  6694	        # want to pick an expression we were going to skip anyways.
  6695	        singleton_index = -1
  6696	        for i in possible_target_indices:
  6697	            if i not in narrowable_operand_indices:
  6698	                singleton_index = i
  6699
  6700	        # But if none of the possible singletons are unassignable ones, we give up
  6701	        # and arbitrarily pick the last item, mostly because other parts of the
  6702	        # type narrowing logic bias towards picking the rightmost item and it'd be
  6703	        # nice to stay consistent.
  6704	        #
  6705	        # That said, it shouldn't matter which index we pick. For example, suppose we
  6706	        # have this if statement, where 'x' and 'y' both have singleton types:
  6707	        #
  6708	        #     if x is y:
  6709	        #         reveal_type(x)
  6710	        #         reveal_type(y)
  6711	        #     else:
  6712	        #         reveal_type(x)
  6713	        #         reveal_type(y)
  6714	        #
  6715	        # At this point, 'x' and 'y' *must* have the same singleton type: we would have
  6716	        # ended early in the first for-loop in this function if they weren't.
  6717	        #
  6718	        # So, we should always get the same result in the 'if' case no matter which
  6719	        # index we pick. And while we do end up getting different results in the 'else'
  6720	        # case depending on the index (e.g. if we pick 'y', then its type stays the same
  6721	        # while 'x' is narrowed to '<uninhabited>'), this distinction is also moot: mypy
  6722	        # currently will just mark the whole branch as unreachable if either operand is
  6723	        # narrowed to <uninhabited>.
  6724	        if singleton_index == -1:
  6725	            singleton_index = possible_target_indices[-1]
  6726
  6727	        sum_type_name = None
  6728	        target = get_proper_type(target)
  6729	        if isinstance(target, LiteralType) and (
  6730	            target.is_enum_literal() or isinstance(target.value, bool)
  6731	        ):
  6732	            sum_type_name = target.fallback.type.fullname
  6733
  6734	        target_type = [TypeRange(target, is_upper_bound=False)]
  6735
  6736	        partial_type_maps = []
  6737	        for i in chain_indices:
  6738	            # If we try refining a type against itself, conditional_type_map
  6739	            # will end up assuming that the 'else' branch is unreachable. This is
  6740	            # typically not what we want: generally the user will intend for the
  6741	            # target type to be some fixed 'sentinel' value and will want to refine
  6742	            # the other exprs against this one instead.
  6743	            if i == singleton_index:
  6744	                continue
  6745
  6746	            # Naturally, we can't refine operands which are not permitted to be refined.
  6747	            if i not in narrowable_operand_indices:
  6748	                continue
  6749
  6750	            expr = operands[i]
  6751	            expr_type = coerce_to_literal(operand_types[i])
  6752
  6753	            if sum_type_name is not None:
  6754	                expr_type = try_expanding_sum_type_to_union(expr_type, sum_type_name)
  6755
  6756	            # We intentionally use 'conditional_types' directly here instead of
  6757	            # 'self.conditional_types_with_intersection': we only compute ad-hoc
  6758	            # intersections when working with pure instances.
  6759	            types = conditional_types(expr_type, target_type)
  6760	            partial_type_maps.append(conditional_types_to_typemaps(expr, *types))
  6761
  6762	        return reduce_conditional_maps(partial_type_maps)

  6764	    def refine_away_none_in_comparison(
  6765	        self,
  6766	        operands: list[Expression],
  6767	        operand_types: list[Type],
  6768	        chain_indices: list[int],
  6769	        narrowable_operand_indices: AbstractSet[int],

  6771	        """Produces conditional type maps refining away None in an identity/equality chain.

  6777	        non_optional_types = []

  6860	    def find_tuple_len_narrowing(self, node: ComparisonExpr) -> list[tuple[TypeMap, TypeMap]]:
  6861	        """Top-level logic to find type restrictions from a length check on tuples.
  6862
  6863	        We try to detect `if` checks like the following:
  6864	            x: tuple[int, int] | tuple[int, int, int]
  6865	            y: tuple[int, int] | tuple[int, int, int]
  6866	            if len(x) == len(y) == 2:
  6867	                a, b = x  # OK
  6868	                c, d = y  # OK
  6869
  6870	            z: tuple[int, ...]
  6871	            if 1 < len(z) < 4:
  6872	                x = z  # OK
  6873	        and report corresponding type restrictions to the binder.
  6874	        """
  6875	        # First step: group consecutive `is` and `==` comparisons together.
  6876	        # This is essentially a simplified version of group_comparison_operands(),
  6877	        # tuned to the len()-like checks. Note that we don't propagate indirect
  6878	        # restrictions like e.g. `len(x) > foo() > 1` yet, since it is tricky.
  6879	        # TODO: propagate indirect len() comparison restrictions.
  6880	        chained = []
  6881	        last_group = set()
  6882	        for op, left, right in node.pairwise():
  6883	            if isinstance(left, AssignmentExpr):
  6884	                left = left.value
  6885	            if isinstance(right, AssignmentExpr):
  6886	                right = right.value
  6887	            if op in ("is", "=="):
  6888	                last_group.add(left)
  6889	                last_group.add(right)
  6890	            else:
  6891	                if last_group:
  6892	                    chained.append(("==", list(last_group)))
  6893	                    last_group = set()
  6894	                if op in {"is not", "!=", "<", "<=", ">", ">="}:
  6895	                    chained.append((op, [left, right]))
  6896	        if last_group:
  6897	            chained.append(("==", list(last_group)))
  6898
  6899	        # Second step: infer type restrictions from each group found above.
  6900	        type_maps = []
  6901	        for op, items in chained:
  6902	            # TODO: support unions of literal types as len() comparison targets.
  6903	            if not any(self.literal_int_expr(it) is not None for it in items):
  6904	                continue
  6905	            if not any(self.is_len_of_tuple(it) for it in items):
  6906	                continue
  6907
  6908	            # At this step we know there is at least one len(x) and one literal in the group.
  6909	            if op in ("is", "=="):
  6910	                literal_values = set()
  6911	                tuples = []
  6912	                for it in items:
  6913	                    lit = self.literal_int_expr(it)
  6914	                    if lit is not None:
  6915	                        literal_values.add(lit)
  6916	                        continue
  6917	                    if self.is_len_of_tuple(it):
  6918	                        assert isinstance(it, CallExpr)
  6919	                        tuples.append(it.args[0])
  6920	                if len(literal_values) > 1:
  6921	                    # More than one different literal value found, like 1 == len(x) == 2,
  6922	                    # so the corresponding branch is unreachable.
  6923	                    return [(None, {})]
  6924	                size = literal_values.pop()
  6925	                if size > MAX_PRECISE_TUPLE_SIZE:
  6926	                    # Avoid creating huge tuples from checks like if len(x) == 300.
  6927	                    continue
  6928	                for tpl in tuples:
  6929	                    yes_type, no_type = self.narrow_with_len(self.lookup_type(tpl), op, size)
  6930	                    yes_map = None if yes_type is None else {tpl: yes_type}
  6931	                    no_map = None if no_type is None else {tpl: no_type}
  6932	                    type_maps.append((yes_map, no_map))
  6933	            else:
  6934	                left, right = items
  6935	                if self.is_len_of_tuple(right):
  6936	                    # Normalize `1 < len(x)` and similar as `len(x) > 1`.
  6937	                    left, right = right, left
  6938	                    op = flip_ops.get(op, op)
  6939	                r_size = self.literal_int_expr(right)
  6940	                assert r_size is not None
  6941	                if r_size > MAX_PRECISE_TUPLE_SIZE:
  6942	                    # Avoid creating huge unions from checks like if len(x) > 300.
  6943	                    continue
  6944	                assert isinstance(left, CallExpr)
  6945	                yes_type, no_type = self.narrow_with_len(
  6946	                    self.lookup_type(left.args[0]), op, r_size
  6947	                )
  6948	                yes_map = None if yes_type is None else {left.args[0]: yes_type}
  6949	                no_map = None if no_type is None else {left.args[0]: no_type}
  6950	                type_maps.append((yes_map, no_map))
  6951	        return type_maps

  6953	    def narrow_with_len(self, typ: Type, op: str, size: int) -> tuple[Type | None, Type | None]:
  6954	        """Dispatch tuple type narrowing logic depending on the kind of type we got."""
  6955	        typ = get_proper_type(typ)
  6956	        if isinstance(typ, TupleType):
  6957	            return self.refine_tuple_type_with_len(typ, op, size)
  6958	        elif isinstance(typ, Instance):
  6959	            return self.refine_instance_type_with_len(typ, op, size)
  6960	        elif isinstance(typ, UnionType):
  6961	            yes_types = []
  6962	            no_types = []
  6963	            other_types = []
  6964	            for t in typ.items:
  6965	                if not self.can_be_narrowed_with_len(t):
  6966	                    other_types.append(t)
  6967	                    continue
  6968	                yt, nt = self.narrow_with_len(t, op, size)
  6969	                if yt is not None:
  6970	                    yes_types.append(yt)
  6971	                if nt is not None:
  6972	                    no_types.append(nt)
  6973	            yes_types += other_types
  6974	            no_types += other_types
  6975	            if yes_types:
  6976	                yes_type = make_simplified_union(yes_types)
  6977	            else:
  6978	                yes_type = None
  6979	            if no_types:
  6980	                no_type = make_simplified_union(no_types)
  6981	            else:
  6982	                no_type = None
  6983	            return yes_type, no_type
  6984	        else:
  6985	            assert False, "Unsupported type for len narrowing"

  6987	    def refine_tuple_type_with_len(
  6988	        self, typ: TupleType, op: str, size: int
  6989	    ) -> tuple[Type | None, Type | None]:
  6990	        """Narrow a TupleType using length restrictions."""
  6991	        unpack_index = find_unpack_in_list(typ.items)
  6992	        if unpack_index is None:
  6993	            # For fixed length tuple situation is trivial, it is either reachable or not,
  6994	            # depending on the current length, expected length, and the comparison op.
  6995	            method = int_op_to_method[op]
  6996	            if method(typ.length(), size):
  6997	                return typ, None
  6998	            return None, typ
  6999	        unpack = typ.items[unpack_index]
  7000	        assert isinstance(unpack, UnpackType)
  7001	        unpacked = get_proper_type(unpack.type)
  7002	        if isinstance(unpacked, TypeVarTupleType):
  7003	            # For tuples involving TypeVarTuple unpack we can't do much except
  7004	            # inferring reachability, and recording the restrictions on TypeVarTuple
  7005	            # for further "manual" use elsewhere.
  7006	            min_len = typ.length() - 1 + unpacked.min_len
  7007	            if op in ("==", "is"):
  7008	                if min_len <= size:
  7009	                    return typ, typ
  7010	                return None, typ
  7011	            elif op in ("<", "<="):
  7012	                if op == "<=":
  7013	                    size += 1
  7014	                if min_len < size:
  7015	                    prefix = typ.items[:unpack_index]
  7016	                    suffix = typ.items[unpack_index + 1 :]
  7017	                    # TODO: also record max_len to avoid false negatives?
  7018	                    unpack = UnpackType(unpacked.copy_modified(min_len=size - typ.length() + 1))
  7019	                    return typ, typ.copy_modified(items=prefix + [unpack] + suffix)
  7020	                return None, typ
  7021	            else:
  7022	                yes_type, no_type = self.refine_tuple_type_with_len(typ, neg_ops[op], size)
  7023	                return no_type, yes_type
  7024	        # Homogeneous variadic item is the case where we are most flexible. Essentially,
  7025	        # we adjust the variadic item by "eating away" from it to satisfy the restriction.
  7026	        assert isinstance(unpacked, Instance) and unpacked.type.fullname == "builtins.tuple"
  7027	        min_len = typ.length() - 1
  7028	        arg = unpacked.args[0]
  7029	        prefix = typ.items[:unpack_index]
  7030	        suffix = typ.items[unpack_index + 1 :]
  7031	        if op in ("==", "is"):
  7032	            if min_len <= size:
  7033	                # TODO: return fixed union + prefixed variadic tuple for no_type?
  7034	                return typ.copy_modified(items=prefix + [arg] * (size - min_len) + suffix), typ
  7035	            return None, typ
  7036	        elif op in ("<", "<="):
  7037	            if op == "<=":
  7038	                size += 1
  7039	            if min_len < size:
  7040	                # Note: there is some ambiguity w.r.t. to where to put the additional
  7041	                # items: before or after the unpack. However, such types are equivalent,
  7042	                # so we always put them before for consistency.
  7043	                no_type = typ.copy_modified(
  7044	                    items=prefix + [arg] * (size - min_len) + [unpack] + suffix
  7045	                )
  7046	                yes_items = []
  7047	                for n in range(size - min_len):
  7048	                    yes_items.append(typ.copy_modified(items=prefix + [arg] * n + suffix))
  7049	                return UnionType.make_union(yes_items, typ.line, typ.column), no_type
  7050	            return None, typ
  7051	        else:
  7052	            yes_type, no_type = self.refine_tuple_type_with_len(typ, neg_ops[op], size)
  7053	            return no_type, yes_type
  7054
  7055	    def refine_instance_type_with_len(
  7056	        self, typ: Instance, op: str, size: int
  7057	    ) -> tuple[Type | None, Type | None]:
  7058	        """Narrow a homogeneous tuple using length restrictions."""
  7059	        base = map_instance_to_supertype(typ, self.lookup_typeinfo("builtins.tuple"))
  7060	        arg = base.args[0]
  7061	        # Again, we are conservative about subclasses until we gain more confidence.
  7062	        allow_precise = (
  7063	            PRECISE_TUPLE_TYPES in self.options.enable_incomplete_feature
  7064	        ) and typ.type.fullname == "builtins.tuple"
  7065	        if op in ("==", "is"):
  7066	            # TODO: return fixed union + prefixed variadic tuple for no_type?
  7067	            return TupleType(items=[arg] * size, fallback=typ), typ
  7068	        elif op in ("<", "<="):
  7069	            if op == "<=":
  7070	                size += 1
  7071	            if allow_precise:
  7072	                unpack = UnpackType(self.named_generic_type("builtins.tuple", [arg]))
  7073	                no_type: Type | None = TupleType(items=[arg] * size + [unpack], fallback=typ)
  7074	            else:
  7075	                no_type = typ
  7076	            if allow_precise:
  7077	                items = []
  7078	                for n in range(size):
  7079	                    items.append(TupleType([arg] * n, fallback=typ))
  7080	                yes_type: Type | None = UnionType.make_union(items, typ.line, typ.column)
  7081	            else:
  7082	                yes_type = typ
  7083	            return yes_type, no_type
  7084	        else:
  7085	            yes_type, no_type = self.refine_instance_type_with_len(typ, neg_ops[op], size)
  7086	            return no_type, yes_type

  7120	    def check_subtype(
  7121	        self,
  7122	        subtype: Type,
  7123	        supertype: Type,
  7124	        context: Context,
  7125	        msg: str | ErrorMessage,
  7126	        subtype_label: str | None = None,
  7127	        supertype_label: str | None = None,
  7128	        *,
  7129	        notes: list[str] | None = None,
  7130	        code: ErrorCode | None = None,
  7131	        outer_context: Context | None = None,
  7132	    ) -> bool:
  7133	        """Generate an error if the subtype is not compatible with supertype."""
  7134	        if is_subtype(subtype, supertype, options=self.options):
  7135	            return True
  7136
  7137	        if isinstance(msg, str):
  7138	            msg = ErrorMessage(msg, code=code)
  7139
  7140	        if self.msg.prefer_simple_messages():
  7141	            self.fail(msg, context)  # Fast path -- skip all fancy logic
  7142	            return False
  7143
  7144	        orig_subtype = subtype
  7145	        subtype = get_proper_type(subtype)
  7146	        orig_supertype = supertype
  7147	        supertype = get_proper_type(supertype)
  7148	        if self.msg.try_report_long_tuple_assignment_error(
  7149	            subtype, supertype, context, msg, subtype_label, supertype_label
  7150	        ):
  7151	            return False
  7152	        extra_info: list[str] = []
  7153	        note_msg = ""
  7154	        notes = notes or []
  7155	        if subtype_label is not None or supertype_label is not None:
  7156	            subtype_str, supertype_str = format_type_distinctly(
  7157	                orig_subtype, orig_supertype, options=self.options
  7158	            )
  7159	            if subtype_label is not None:
  7160	                extra_info.append(subtype_label + " " + subtype_str)
  7161	            if supertype_label is not None:
  7162	                extra_info.append(supertype_label + " " + supertype_str)
  7163	            note_msg = make_inferred_type_note(
  7164	                outer_context or context, subtype, supertype, supertype_str
  7165	            )
  7166	            if isinstance(subtype, Instance) and isinstance(supertype, Instance):
  7167	                notes = append_invariance_notes(notes, subtype, supertype)
  7168	            if isinstance(subtype, UnionType) and isinstance(supertype, UnionType):
  7169	                notes = append_union_note(notes, subtype, supertype, self.options)
  7170	        if extra_info:
  7171	            msg = msg.with_additional_msg(" (" + ", ".join(extra_info) + ")")
  7172
  7173	        error = self.fail(msg, context)
  7174	        for note in notes:
  7175	            self.msg.note(note, context, code=msg.code)
  7176	        if note_msg:
  7177	            self.note(note_msg, context, code=msg.code)
  7178	        self.msg.maybe_note_concatenate_pos_args(subtype, supertype, context, code=msg.code)
  7179	        if (
  7180	            isinstance(supertype, Instance)
  7181	            and supertype.type.is_protocol
  7182	            and isinstance(subtype, (CallableType, Instance, TupleType, TypedDictType, TypeType))
  7183	        ):
  7184	            self.msg.report_protocol_problems(subtype, supertype, context, parent_error=error)
  7185	        if isinstance(supertype, CallableType) and isinstance(subtype, Instance):
  7186	            call = find_member("__call__", subtype, subtype, is_operator=True)
  7187	            if call:
  7188	                self.msg.note_call(subtype, call, context, code=msg.code)
  7189	        if isinstance(subtype, (CallableType, Overloaded)) and isinstance(supertype, Instance):
  7190	            if supertype.type.is_protocol and "__call__" in supertype.type.protocol_members:
  7191	                call = find_member("__call__", supertype, subtype, is_operator=True)
  7192	                assert call is not None
  7193	                if not is_subtype(subtype, call, options=self.options):
  7194	                    self.msg.note_call(supertype, call, context, code=msg.code)
  7195	        self.check_possible_missing_await(subtype, supertype, context, code=msg.code)
  7196	        return False

  7198	    def get_precise_awaitable_type(self, typ: Type, local_errors: ErrorWatcher) -> Type | None:
  7199	        """If type implements Awaitable[X] with non-Any X, return X.
  7200
  7201	        In all other cases return None. This method must be called in context
  7202	        of local_errors.
  7203	        """
  7204	        if isinstance(get_proper_type(typ), PartialType):
  7205	            # Partial types are special, ignore them here.
  7206	            return None
  7207	        try:
  7208	            aw_type = self.expr_checker.check_awaitable_expr(
  7209	                typ, Context(), "", ignore_binder=True
  7210	            )
  7211	        except KeyError:
  7212	            # This is a hack to speed up tests by not including Awaitable in all typing stubs.
  7213	            return None
  7214	        if local_errors.has_new_errors():
  7215	            return None
  7216	        if isinstance(get_proper_type(aw_type), (AnyType, UnboundType)):
  7217	            return None
  7218	        return aw_type
