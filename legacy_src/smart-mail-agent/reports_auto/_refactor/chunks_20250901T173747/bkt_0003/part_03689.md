t_be_defined.update(state.must_be_defined)
   124	        current_branch.may_be_defined.update(state.may_be_defined)
   125	        current_branch.may_be_defined.difference_update(current_branch.must_be_defined)
   126
   127	    def skip_branch(self) -> None:
   128	        assert len(self.branches) > 0
   129	        self.branches[-1].skipped = True
   130
   131	    def is_possibly_undefined(self, name: str) -> bool:
   132	        assert len(self.branches) > 0
   133	        return name in self.branches[-1].may_be_defined
   134
   135	    def is_undefined(self, name: str) -> bool:
   136	        assert len(self.branches) > 0
   137	        branch = self.branches[-1]
   138	        return name not in branch.may_be_defined and name not in branch.must_be_defined
   139
   140	    def is_defined_in_a_branch(self, name: str) -> bool:
   141	        assert len(self.branches) > 0
   142	        for b in self.branches:
   143	            if name in b.must_be_defined or name in b.may_be_defined:
   144	                return True
   145	        return False
   146
   147	    def done(self) -> BranchState:
   148	        # First, compute all vars, including skipped branches. We include skipped branches
   149	        # because our goal is to capture all variables that semantic analyzer would
   150	        # consider defined.
   151	        all_vars = set()
   152	        for b in self.branches:
   153	            all_vars.update(b.may_be_defined)
   154	            all_vars.update(b.must_be_defined)
   155	        # For the rest of the things, we only care about branches that weren't skipped.
   156	        non_skipped_branches = [b for b in self.branches if not b.skipped]
   157	        if non_skipped_branches:
   158	            must_be_defined = non_skipped_branches[0].must_be_defined
   159	            for b in non_skipped_branches[1:]:
   160	                must_be_defined.intersection_update(b.must_be_defined)
   161	        else:
   162	            must_be_defined = set()
   163	        # Everything that wasn't defined in all branches but was defined
   164	        # in at least one branch should be in `may_be_defined`!
   165	        may_be_defined = all_vars.difference(must_be_defined)
   166	        return BranchState(
   167	            must_be_defined=must_be_defined,
   168	            may_be_defined=may_be_defined,
   169	            skipped=len(non_skipped_branches) == 0,
   170	        )

   180	class Scope:
   181	    def __init__(self, stmts: list[BranchStatement], scope_type: ScopeType) -> None:
   182	        self.branch_stmts: list[BranchStatement] = stmts
   183	        self.scope_type = scope_type
   184	        self.undefined_refs: dict[str, set[NameExpr]] = {}
   185
   186	    def copy(self) -> Scope:
   187	        result = Scope([s.copy() for s in self.branch_stmts], self.scope_type)
   188	        result.undefined_refs = self.undefined_refs.copy()
   189	        return result
   190
   191	    def record_undefined_ref(self, o: NameExpr) -> None:
   192	        if o.name not in self.undefined_refs:
   193	            self.undefined_refs[o.name] = set()
   194	        self.undefined_refs[o.name].add(o)
   195
   196	    def pop_undefined_ref(self, name: str) -> set[NameExpr]:
   197	        return self.undefined_refs.pop(name, set())

   200	class DefinedVariableTracker:
   201	    """DefinedVariableTracker manages the state and scope for the UndefinedVariablesVisitor."""
   202
   203	    def __init__(self) -> None:
   204	        # There's always at least one scope. Within each scope, there's at least one "global" BranchingStatement.
   205	        self.scopes: list[Scope] = [Scope([BranchStatement()], ScopeType.Global)]
   206	        # disable_branch_skip is used to disable skipping a branch due to a return/raise/etc. This is useful
   207	        # in things like try/except/finally statements.
   208	        self.disable_branch_skip = False
   209
   210	    def copy(self) -> DefinedVariableTracker:
   211	        result = DefinedVariableTracker()
   212	        result.scopes = [s.copy() for s in self.scopes]
   213	        result.disable_branch_skip = self.disable_branch_skip
   214	        return result
   215
   216	    def _scope(self) -> Scope:
   217	        assert len(self.scopes) > 0
   218	        return self.scopes[-1]
   219
   220	    def enter_scope(self, scope_type: ScopeType) -> None:
   221	        assert len(self._scope().branch_stmts) > 0
   222	        initial_state = None
   223	        if scope_type == ScopeType.Generator:
   224	            # Generators are special because they inherit the outer scope.
   225	            initial_state = self._scope().branch_stmts[-1].branches[-1]
   226	        self.scopes.append(Scope([BranchStatement(initial_state)], scope_type))
   227
   228	    def exit_scope(self) -> None:
   229	        self.scopes.pop()
   230
   231	    def in_scope(self, scope_type: ScopeType) -> bool:
   232	        return self._scope().scope_type == scope_type
   233
   234	    def start_branch_statement(self) -> None:
   235	        assert len(self._scope().branch_stmts) > 0
   236	        self._scope().branch_stmts.append(
   237	            BranchStatement(self._scope().branch_stmts[-1].branches[-1])
   238	        )
   239
   240	    def next_branch(self) -> None:
   241	        assert len(self._scope().branch_stmts) > 1
   242	        self._scope().branch_stmts[-1].next_branch()
   243
   244	    def end_branch_statement(self) -> None:
   245	        assert len(self._scope().branch_stmts) > 1
   246	        result = self._scope().branch_stmts.pop().done()
   247	        self._scope().branch_stmts[-1].record_nested_branch(result)
   248
   249	    def skip_branch(self) -> None:
   250	        # Only skip branch if we're outside of "root" branch statement.
   251	        if len(self._scope().branch_stmts) > 1 and not self.disable_branch_skip:
   252	            self._scope().branch_stmts[-1].skip_branch()
   253
   254	    def record_definition(self, name: str) -> None:
   255	        assert len(self.scopes) > 0
   256	        assert len(self.scopes[-1].branch_stmts) > 0
   257	        self._scope().branch_stmts[-1].record_definition(name)
   258
   259	    def delete_var(self, name: str) -> None:
   260	        assert len(self.scopes) > 0
   261	        assert len(self.scopes[-1].branch_stmts) > 0
   262	        self._scope().branch_stmts[-1].delete_var(name)
   263
   264	    def record_undefined_ref(self, o: NameExpr) -> None:
   265	        """Records an undefined reference. These can later be retrieved via `pop_undefined_ref`."""
   266	        assert len(self.scopes) > 0
   267	        self._scope().record_undefined_ref(o)
   268
   269	    def pop_undefined_ref(self, name: str) -> set[NameExpr]:
   270	        """If name has previously been reported as undefined, the NameExpr that was called will be returned."""
   271	        assert len(self.scopes) > 0
   272	        return self._scope().pop_undefined_ref(name)
   273
   274	    def is_possibly_undefined(self, name: str) -> bool:
   275	        assert len(self._scope().branch_stmts) > 0
   276	        # A variable is undefined if it's in a set of `may_be_defined` but not in `must_be_defined`.
   277	        return self._scope().branch_stmts[-1].is_possibly_undefined(name)
   278
   279	    def is_defined_in_different_branch(self, name: str) -> bool:
   280	        """This will return true if a variable is defined in a branch that's not the current branch."""
   281	        assert len(self._scope().branch_stmts) > 0
   282	        stmt = self._scope().branch_stmts[-1]
   283	        if not stmt.is_undefined(name):
   284	            return False
   285	        for stmt in self._scope().branch_stmts:
   286	            if stmt.is_defined_in_a_branch(name):
   287	                return True
   288	        return False
   289
   290	    def is_undefined(self, name: str) -> bool:
   291	        assert len(self._scope().branch_stmts) > 0
   292	        return self._scope().branch_stmts[-1].is_undefined(name)

   300	class PossiblyUndefinedVariableVisitor(ExtendedTraverserVisitor):
   301	    """Detects the following cases:
   302	    - A variable that's defined only part of the time.
   303	    - If a variable is used before definition
   304
   305	    An example of a partial definition:
   306	    if foo():
   307	        x = 1
   308	    print(x)  # Error: "x" may be undefined.
   309
   310	    Example of a used before definition:
   311	    x = y
   312	    y: int = 2
   313
   314	    Note that this code does not detect variables not defined in any of the branches -- that is
   315	    handled by the semantic analyzer.
   316	    """
   317
   318	    def __init__(
   319	        self,
   320	        msg: MessageBuilder,
   321	        type_map: dict[Expression, Type],
   322	        options: Options,
   323	        names: SymbolTable,
   324	    ) -> None:
   325	        self.msg = msg
   326	        self.type_map = type_map
   327	        self.options = options
   328	        self.builtins = SymbolTable()
   329	        builtins_mod = names.get("__builtins__", None)
   330	        if builtins_mod:
   331	            assert isinstance(builtins_mod.node, MypyFile)
   332	            self.builtins = builtins_mod.node.names
   333	        self.loops: list[Loop] = []
   334	        self.try_depth = 0
   335	        self.tracker = DefinedVariableTracker()
   336	        for name in implicit_module_attrs:
   337	            self.tracker.record_definition(name)
   338
   339	    def var_used_before_def(self, name: str, context: Context) -> None:
   340	        if self.msg.errors.is_error_code_enabled(errorcodes.USED_BEFORE_DEF):
   341	            self.msg.var_used_before_def(name, context)
   342
   343	    def variable_may_be_undefined(self, name: str, context: Context) -> None:
   344	        if self.msg.errors.is_error_code_enabled(errorcodes.POSSIBLY_UNDEFINED):
   345	            self.msg.variable_may_be_undefined(name, context)
   346
   347	    def process_definition(self, name: str) -> None:
   348	        # Was this name previously used? If yes, it's a used-before-definition error.
   349	        if not self.tracker.in_scope(ScopeType.Class):
   350	            refs = self.tracker.pop_undefined_ref(name)
   351	            for ref in refs:
   352	                if self.loops:
   353	                    self.variable_may_be_undefined(name, ref)
   354	                else:
   355	                    self.var_used_before_def(name, ref)
   356	        else:
   357	            # Errors in class scopes are caught by the semantic analyzer.
   358	            pass
   359	        self.tracker.record_definition(name)
   360
   361	    def visit_global_decl(self, o: GlobalDecl) -> None:
   362	        for name in o.names:
   363	            self.process_definition(name)
   364	        super().visit_global_decl(o)

   366	    def visit_nonlocal_decl(self, o: NonlocalDecl) -> None:

   369	        super().visit_nonlocal_decl(o)

   371	    def process_lvalue(self, lvalue: Lvalue | None) -> None:
   372	        if isinstance(lvalue, NameExpr):
   373	            self.process_definition(lvalue.name)
   374	        elif isinstance(lvalue, StarExpr):
   375	            self.process_lvalue(lvalue.expr)
   376	        elif isinstance(lvalue, (ListExpr, TupleExpr)):
   377	            for item in lvalue.items:
   378	                self.process_lvalue(item)

   380	    def visit_assignment_stmt(self, o: AssignmentStmt) -> None:
   381	        for lvalue in o.lvalues:
   382	            self.process_lvalue(lvalue)
   383	        super().visit_assignment_stmt(o)

   389	    def visit_if_stmt(self, o: IfStmt) -> None:
   390	        for e in o.expr:
   391	            e.accept(self)
   392	        self.tracker.start_branch_statement()
   393	        for b in o.body:
   394	            if b.is_unreachable:
   395	                continue
   396	            b.accept(self)
   397	            self.tracker.next_branch()
   398	        if o.else_body:
   399	            if not o.else_body.is_unreachable:
   400	                o.else_body.accept(self)
   401	            else:
   402	                self.tracker.skip_branch()
   403	        self.tracker.end_branch_statement()
   404
   405	    def visit_match_stmt(self, o: MatchStmt) -> None:
   406	        o.subject.accept(self)
   407	        self.tracker.start_branch_statement()
   408	        for i in range(len(o.patterns)):
   409	            pattern = o.patterns[i]
   410	            pattern.accept(self)
   411	            guard = o.guards[i]
   412	            if guard is not None:
   413	                guard.accept(self)
   414	            if not o.bodies[i].is_unreachable:
   415	                o.bodies[i].accept(self)
   416	            else:
   417	                self.tracker.skip_branch()
   418	            is_catchall = infer_pattern_value(pattern) == ALWAYS_TRUE
   419	            if not is_catchall:
   420	                self.tracker.next_branch()
   421	        self.tracker.end_branch_statement()

   425	        super().visit_func_def(o)

   427	    def visit_func(self, o: FuncItem) -> None:
   428	        if o.is_dynamic() and not self.options.check_untyped_defs:
   429	            return

   431	        args = o.arguments or []
   432	        # Process initializers (defaults) outside the function scope.
   433	        for arg in args:
   434	            if arg.initializer is not None:
   435	                arg.initializer.accept(self)

   438	        for arg in args:
   439	            self.process_definition(arg.variable.name)
   440	            super().visit_var(arg.variable)
   441	        o.body.accept(self)
   442	        self.tracker.exit_scope()

   458	    def visit_for_stmt(self, o: ForStmt) -> None:
   459	        o.expr.accept(self)
   460	        self.process_lvalue(o.index)
   461	        o.index.accept(self)
   462	        self.tracker.start_branch_statement()
   463	        loop = Loop()
   464	        self.loops.append(loop)
   465	        o.body.accept(self)
   466	        self.tracker.next_branch()
   467	        self.tracker.end_branch_statement()
   468	        if o.else_body is not None:
   469	            # If the loop has a `break` inside, `else` is executed conditionally.
   470	            # If the loop doesn't have a `break` either the function will return or
   471	            # execute the `else`.
   472	            has_break = loop.has_break
   473	            if has_break:
   474	                self.tracker.start_branch_statement()
   475	                self.tracker.next_branch()
   476	            o.else_body.accept(self)
   477	            if has_break:
   478	                self.tracker.end_branch_statement()
   479	        self.loops.pop()

   483	        self.tracker.skip_branch()

   507	        self.tracker.skip_branch()

   509	    def visit_expression_stmt(self, o: ExpressionStmt) -> None:

   511	        if typ is None or isinstance(get_proper_type(typ), UninhabitedType):
   512	            self.tracker.skip_branch()

   515	    def visit_try_stmt(self, o: TryStmt) -> None:
   516	        """
   517	        Note that finding undefined vars in `finally` requires different handling from
   518	        the rest of the code. In particular, we want to disallow skipping branches due to jump
   519	        statements in except/else clauses for finally but not for other cases. Imagine a case like:
   520	        def f() -> int:
   521	            try:
   522	                x = 1
   523	            except:
   524	                # This jump statement needs to be handled differently depending on whether or
   525	                # not we're trying to process `finally` or not.
   526	                return 0
   527	            finally:
   528	                # `x` may be undefined here.
   529	                pass
   530	            # `x` is always defined here.
   531	            return x
   532	        """
   533	        self.try_depth += 1
   534	        if o.finally_body is not None:
   535	            # In order to find undefined vars in `finally`, we need to
   536	            # process try/except with branch skipping disabled. However, for the rest of the code
   537	            # after finally, we need to process try/except with branch skipping enabled.
   538	            # Therefore, we need to process try/finally twice.
   539	            # Because processing is not idempotent, we should make a copy of the tracker.
   540	            old_tracker = self.tracker.copy()
   541	            self.tracker.disable_branch_skip = True
   542	            self.process_try_stmt(o)
   543	            self.tracker = old_tracker
   544	        self.process_try_stmt(o)
   545	        self.try_depth -= 1
   546
   547	    def process_try_stmt(self, o: TryStmt) -> None:
   548	        """
   549	        Processes try statement decomposing it into the following:
   550	        if ...:
   551	            body
   552	            else_body
   553	        elif ...:
   554	            except 1
   555	        elif ...:
   556	            except 2
   557	        else:
   558	            except n
   559	        finally
   560	        """
   561	        self.tracker.start_branch_statement()
   562	        o.body.accept(self)
   563	        if o.else_body is not None:
   564	            o.else_body.accept(self)
   565	        if len(o.handlers) > 0:
   566	            assert len(o.handlers) == len(o.vars) == len(o.types)
   567	            for i in range(len(o.handlers)):
   568	                self.tracker.next_branch()
   569	                exc_type = o.types[i]
   570	                if exc_type is not None:
   571	                    exc_type.accept(self)
   572	                var = o.vars[i]
   573	                if var is not None:
   574	                    self.process_definition(var.name)
   575	                    var.accept(self)
   576	                o.handlers[i].accept(self)
   577	                if var is not None:
   578	                    self.tracker.delete_var(var.name)
   579	        self.tracker.end_branch_statement()
   580
   581	        if o.finally_body is not None:
   582	            o.finally_body.accept(self)
   583
   584	    def visit_while_stmt(self, o: WhileStmt) -> None:
   585	        o.expr.accept(self)
   586	        self.tracker.start_branch_statement()
   587	        loop = Loop()
   588	        self.loops.append(loop)
   589	        o.body.accept(self)
   590	        has_break = loop.has_break
   591	        if not checker.is_true_literal(o.expr):
   592	            # If this is a loop like `while True`, we can consider the body to be
   593	            # a single branch statement (we're guaranteed that the body is executed at least once).
   594	            # If not, call next_branch() to make all variables defined there conditional.
   595	            self.tracker.next_branch()
   596	        self.tracker.end_branch_statement()
   597	        if o.else_body is not None:
   598	            # If the loop has a `break` inside, `else` is executed conditionally.
   599	            # If the loop doesn't have a `break` either the function will return or
   600	            # execute the `else`.
   601	            if has_break:
   602	                self.tracker.start_branch_statement()
   603	                self.tracker.next_branch()
   604	            if o.else_body:
   605	                o.else_body.accept(self)
   606	            if has_break:
   607	                self.tracker.end_branch_statement()
   608	        self.loops.pop()

   610	    def visit_as_pattern(self, o: AsPattern) -> None:
   611	        if o.name is not None:
   612	            self.process_lvalue(o.name)
   613	        super().visit_as_pattern(o)
   614
   615	    def visit_starred_pattern(self, o: StarredPattern) -> None:
   616	        if o.capture is not None:
   617	            self.process_lvalue(o.capture)
   618	        super().visit_starred_pattern(o)
   619
   620	    def visit_name_expr(self, o: NameExpr) -> None:
   621	        if o.name in self.builtins and self.tracker.in_scope(ScopeType.Global):
   622	            return
   623	        if self.tracker.is_possibly_undefined(o.name):
   624	            # A variable is only defined in some branches.
   625	            self.variable_may_be_undefined(o.name, o)
   626	            # We don't want to report the error on the same variable multiple times.
   627	            self.tracker.record_definition(o.name)
   628	        elif self.tracker.is_defined_in_different_branch(o.name):
   629	            # A variable is defined in one branch but used in a different branch.
   630	            if self.loops or self.try_depth > 0:
   631	                # If we're in a loop or in a try, we can't be sure that this variable
   632	                # is undefined. Report it as "may be undefined".
   633	                self.variable_may_be_undefined(o.name, o)
   634	            else:
   635	                self.var_used_before_def(o.name, o)
   636	        elif self.tracker.is_undefined(o.name):
   637	            # A variable is undefined. It could be due to two things:
   638	            # 1. A variable is just totally undefined
   639	            # 2. The variable is defined later in the code.
   640	            # Case (1) will be caught by semantic analyzer. Case (2) is a forward ref that should
   641	            # be caught by this visitor. Save the ref for later, so that if we see a definition,
   642	            # we know it's a used-before-definition scenario.
   643	            self.tracker.record_undefined_ref(o)
   644	        super().visit_name_expr(o)
   645
   646	    def visit_with_stmt(self, o: WithStmt) -> None:
   647	        for expr, idx in zip(o.expr, o.target):

     1	"""Plugin system for extending mypy.
     2
     3	At large scale the plugin system works as following:
     4
     5	* Plugins are collected from the corresponding mypy config file option
     6	  (either via paths to Python files, or installed Python modules)
     7	  and imported using importlib.
     8
     9	* Every module should get an entry point function (called 'plugin' by default,
    10	  but may be overridden in the config file) that should accept a single string
    11	  argument that is a full mypy version (includes git commit hash for dev
    12	  versions) and return a subclass of mypy.plugins.Plugin.
    13
    14	* All plugin class constructors should match the signature of mypy.plugin.Plugin
    15	  (i.e. should accept an mypy.options.Options object), and *must* call
    16	  super().__init__().
    17
    18	* At several steps during semantic analysis and type checking mypy calls
    19	  special `get_xxx` methods on user plugins with a single string argument that
    20	  is a fully qualified name (full name) of a relevant definition
    21	  (see mypy.plugin.Plugin method docstrings for details).
    22
    23	* The plugins are called in the order they are passed in the config option.
    24	  Every plugin must decide whether to act on a given full name. The first
    25	  plugin that returns non-None object will be used.
    26
    27	* The above decision should be made using the limited common API specified by
    28	  mypy.plugin.CommonPluginApi.
    29
    30	* The callback returned by the plugin will be called with a larger context that
    31	  includes relevant current state (e.g. a default return type, or a default
    32	  attribute type) and a wider relevant API provider (e.g.
    33	  SemanticAnalyzerPluginInterface or CheckerPluginInterface).
    34
    35	* The result of this is used for further processing. See various `XxxContext`
    36	  named tuples for details about which information is given to each hook.
    37
    38	Plugin developers should ensure that their plugins work well in incremental and
    39	daemon modes. In particular, plugins should not hold global state, and should
    40	always call add_plugin_dependency() in plugin hooks called during semantic
    41	analysis. See the method docstring for more details.
    42
    43	There is no dedicated cache storage for plugins, but plugins can store
    44	per-TypeInfo data in a special .metadata attribute that is serialized to the
    45	mypy caches between incremental runs. To avoid collisions between plugins, they
    46	are encouraged to store their state under a dedicated key coinciding with
    47	plugin name in the metadata dictionary. Every value stored there must be
    48	JSON-serializable.
    49
    50	## Notes about the semantic analyzer
    51
    52	Mypy 0.710 introduced a new semantic analyzer that changed how plugins are
    53	expected to work in several notable ways (from mypy 0.730 the old semantic
    54	analyzer is no longer available):
    55
    56	1. The order of processing AST nodes in modules is different. The old semantic
    57	   analyzer processed modules in textual order, one module at a time. The new
    58	   semantic analyzer first processes the module top levels, including bodies of
    59	   any top-level classes and classes nested within classes. ("Top-level" here
    60	   means "not nested within a function/method".) Functions and methods are
    61	   processed only after module top levels have been finished. If there is an
    62	   import cycle, all module top levels in the cycle are processed before
    63	   processing any functions or methods. Each unit of processing (a module top
    64	   level or a function/method) is called a *target*.
    65
    66	   This also means that function signatures in the same module have not been
    67	   analyzed yet when analyzing the module top level. If you need access to
    68	   a function signature, you'll need to explicitly analyze the signature first
    69	   using `anal_type()`.
    70
    71	2. Each target can be processed multiple times. This may happen if some forward
    72	   references are not ready yet, for example. This means that semantic analyzer
    73	   related plugin hooks can be called multiple times for the same full name.
    74	   These plugin methods must thus be idempotent.
    75
    76	3. The `anal_type` API function returns None if some part of the type is not
    77	   available yet. If this happens, the current target being analyzed will be
    78	   *deferred*, which means that it will be processed again soon, in the hope
    79	   that additional dependencies will be available. This may happen if there are
    80	   forward references to types or inter-module references to types within an
    81	   import cycle.
    82
    83	   Note that if there is a circular definition, mypy may decide to stop
    84	   processing to avoid an infinite number of iterations. When this happens,
    85	   `anal_type` will generate an error and return an `AnyType` type object
    86	   during the final iteration (instead of None).
    87
    88	4. There is a new API method `defer()`. This can be used to explicitly request
    89	   the current target to be reprocessed one more time. You don't need this
    90	   to call this if `anal_type` returns None, however.
    91
    92	5. There is a new API property `final_iteration`, which is true once mypy
    93	   detected no progress during the previous iteration or if the maximum
    94	   semantic analysis iteration count has been reached. You must never
    95	   defer during the final iteration, as it will cause a crash.
    96
    97	6. The `node` attribute of SymbolTableNode objects may contain a reference to
    98	   a PlaceholderNode object. This object means that this definition has not
    99	   been fully processed yet. If you encounter a PlaceholderNode, you should
   100	   defer unless it's the final iteration. If it's the final iteration, you
   101	   should generate an error message. It usually means that there's a cyclic
   102	   definition that cannot be resolved by mypy. PlaceholderNodes can only refer
   103	   to references inside an import cycle. If you are looking up things from
   104	   another module, such as the builtins, that is outside the current module or
   105	   import cycle, you can safely assume that you won't receive a placeholder.
   106
   107	When testing your plugin, you should have a test case that forces a module top
   108	level to be processed multiple times. The easiest way to do this is to include
   109	a forward reference to a class in a top-level annotation. Example:
   110
   111	    c: C  # Forward reference causes second analysis pass
   112	    class C: pass
   113
   114	Note that a forward reference in a function signature won't trigger another
   115	pass, since all functions are processed only after the top level has been fully
   116	analyzed.

   109	class Attribute:
   110	    """The value of an attr.ib() call."""
   111
   112	    def __init__(
   113	        self,
   114	        name: str,
   115	        alias: str | None,
   116	        info: TypeInfo,
   117	        has_default: bool,
   118	        init: bool,
   119	        kw_only: bool,
   120	        converter: Converter | None,
   121	        context: Context,
   122	        init_type: Type | None,
   123	    ) -> None:
   124	        self.name = name
   125	        self.alias = alias
   126	        self.info = info
   127	        self.has_default = has_default
   128	        self.init = init
   129	        self.kw_only = kw_only
   130	        self.converter = converter
   131	        self.context = context
   132	        self.init_type = init_type
   133
   134	    def argument(self, ctx: mypy.plugin.ClassDefContext) -> Argument:
   135	        """Return this attribute as an argument to __init__."""
   136	        assert self.init
   137	        init_type: Type | None = None
   138	        if self.converter:
   139	            if self.converter.init_type:
   140	                init_type = self.converter.init_type
   141	                if init_type and self.init_type and self.converter.ret_type:
   142	                    # The converter return type should be the same type as the attribute type.
   143	                    # Copy type vars from attr type to converter.
   144	                    converter_vars = get_type_vars(self.converter.ret_type)
   145	                    init_vars = get_type_vars(self.init_type)
   146	                    if converter_vars and len(converter_vars) == len(init_vars):
   147	                        variables = {
   148	                            binder.id: arg for binder, arg in zip(converter_vars, init_vars)
   149	                        }
   150	                        init_type = expand_type(init_type, variables)
   151	            else:
   152	                ctx.api.fail("Cannot determine __init__ type from converter", self.context)
   153	                init_type = AnyType(TypeOfAny.from_error)
   154	        else:  # There is no converter, the init type is the normal type.
   155	            init_type = self.init_type or self.info[self.name].type
   156
   157	        unannotated = False
   158	        if init_type is None:
   159	            unannotated = True
   160	            # Convert type not set to Any.
   161	            init_type = AnyType(TypeOfAny.unannotated)
   162	        else:
   163	            proper_type = get_proper_type(init_type)
   164	            if isinstance(proper_type, AnyType):
   165	                if proper_type.type_of_any == TypeOfAny.unannotated:
   166	                    unannotated = True
   167
   168	        if unannotated and ctx.api.options.disallow_untyped_defs:
   169	            # This is a compromise.  If you don't have a type here then the
   170	            # __init__ will be untyped. But since the __init__ is added it's
   171	            # pointing at the decorator. So instead we also show the error in the
   172	            # assignment, which is where you would fix the issue.
   173	            node = self.info[self.name].node
   174	            assert node is not None
   175	            ctx.api.msg.need_annotation_for_var(node, self.context)
   176
   177	        if self.kw_only:
   178	            arg_kind = ARG_NAMED_OPT if self.has_default else ARG_NAMED
   179	        else:
   180	            arg_kind = ARG_OPT if self.has_default else ARG_POS
   181
   182	        # Attrs removes leading underscores when creating the __init__ arguments.
   183	        name = self.alias or self.name.lstrip("_")
   184	        return Argument(Var(name, init_type), init_type, None, arg_kind)
   185
   186	    def serialize(self) -> JsonDict:
   187	        """Serialize this object so it can be saved and restored."""
   188	        return {
   189	            "name": self.name,
   190	            "alias": self.alias,
   191	            "has_default": self.has_default,
   192	            "init": self.init,
   193	            "kw_only": self.kw_only,
   194	            "has_converter": self.converter is not None,
   195	            "converter_init_type": (
   196	                self.converter.init_type.serialize()
   197	                if self.converter and self.converter.init_type
   198	                else None
   199	            ),
   200	            "context_line": self.context.line,
   201	            "context_column": self.context.column,
   202	            "init_type": self.init_type.serialize() if self.init_type else None,
   203	        }
   204
   205	    @classmethod
   206	    def deserialize(
   207	        cls, info: TypeInfo, data: JsonDict, api: SemanticAnalyzerPluginInterface
   208	    ) -> Attribute:
   209	        """Return the Attribute that was serialized."""
   210	        raw_init_type = data["init_type"]
   211	        init_type = deserialize_and_fixup_type(raw_init_type, api) if raw_init_type else None
   212	        raw_converter_init_type = data["converter_init_type"]
   213	        converter_init_type = (
   214	            deserialize_and_fixup_type(raw_converter_init_type, api)
   215	            if raw_converter_init_type
   216	            else None
   217	        )
   218
   219	        return Attribute(
   220	            data["name"],
   221	            data["alias"],
   222	            info,
   223	            data["has_default"],
   224	            data["init"],
   225	            data["kw_only"],
   226	            Converter(converter_init_type) if data["has_converter"] else None,
   227	            Context(line=data["context_line"], column=data["context_column"]),
   228	            init_type,
   229	        )

   231	    def expand_typevar_from_subtype(self, sub_type: TypeInfo) -> None:
   232	        """Expands type vars in the context of a subtype when an attribute is inherited
   233	        from a generic super type."""
   234	        if self.init_type:
   235	            self.init_type = map_type_from_supertype(self.init_type, sub_type, self.info)
   236	        else:
   237	            self.init_type = None

   240	def _determine_eq_order(ctx: mypy.plugin.ClassDefContext) -> bool:
   241	    """
   242	    Validate the combination of *cmp*, *eq*, and *order*. Derive the effective
   243	    value of order.
   244	    """
   245	    cmp = _get_decorator_optional_bool_argument(ctx, "cmp")
   246	    eq = _get_decorator_optional_bool_argument(ctx, "eq")
   247	    order = _get_decorator_optional_bool_argument(ctx, "order")
   248
   249	    if cmp is not None and any((eq is not None, order is not None)):
   250	        ctx.api.fail('Don\'t mix "cmp" with "eq" and "order"', ctx.reason)
   251
   252	    # cmp takes precedence due to bw-compatibility.
   253	    if cmp is not None:
   254	        return cmp
   255
   256	    # If left None, equality is on and ordering mirrors equality.
   257	    if eq is None:
   258	        eq = True
   259
   260	    if order is None:
   261	        order = eq
   262
   263	    if eq is False and order is True:
   264	        ctx.api.fail("eq must be True if order is True", ctx.reason)
   265
   266	    return order

   269	def _get_decorator_optional_bool_argument(
   270	    ctx: mypy.plugin.ClassDefContext, name: str, default: bool | None = None
   271	) -> bool | None:
   272	    """Return the Optional[bool] argument for the decorator.
   273
   274	    This handles both @decorator(...) and @decorator.
   275	    """
   276	    if isinstance(ctx.reason, CallExpr):
   277	        attr_value = _get_argument(ctx.reason, name)
   278	        if attr_value:
   279	            if isinstance(attr_value, NameExpr):
   280	                if attr_value.fullname == "builtins.True":
   281	                    return True
   282	                if attr_value.fullname == "builtins.False":
   283	                    return False
   284	                if attr_value.fullname == "builtins.None":
   285	                    return None
   286	            ctx.api.fail(
   287	                f'"{name}" argument must be a True, False, or None literal',
   288	                ctx.reason,
   289	                code=LITERAL_REQ,
   290	            )
   291	            return default
   292	        return default
   293	    else:
   294	        return default

   307	def attr_class_maker_callback(
   308	    ctx: mypy.plugin.ClassDefContext,
   309	    auto_attribs_default: bool | None = False,
   310	    frozen_default: bool = False,
   311	    slots_default: bool = False,
   312	) -> bool:
   313	    """Add necessary dunder methods to classes decorated with attr.s.
   314
   315	    attrs is a package that lets you define classes without writing dull boilerplate code.
   316
   317	    At a quick glance, the decorator searches the class body for assignments of `attr.ib`s (or
   318	    annotated variables if auto_attribs=True), then depending on how the decorator is called,
   319	    it will add an __init__ or all the compare methods.
   320	    For frozen=True it will turn the attrs into properties.
   321
   322	    Hashability will be set according to https://www.attrs.org/en/stable/hashing.html.
   323
   324	    See https://www.attrs.org/en/stable/how-does-it-work.html for information on how attrs works.
   325
   326	    If this returns False, some required metadata was not ready yet, and we need another
   327	    pass.
   328	    """
   329	    with state.strict_optional_set(ctx.api.options.strict_optional):
   330	        # This hook is called during semantic analysis, but it uses a bunch of
   331	        # type-checking ops, so it needs the strict optional set properly.
   332	        return attr_class_maker_callback_impl(
   333	            ctx, auto_attribs_default, frozen_default, slots_default
   334	        )

   337	def attr_class_maker_callback_impl(
   338	    ctx: mypy.plugin.ClassDefContext,
   339	    auto_attribs_default: bool | None,
   340	    frozen_default: bool,
   341	    slots_default: bool,
   342	) -> bool:
   343	    info = ctx.cls.info
   344
   345	    init = _get_decorator_bool_argument(ctx, "init", True)
   346	    frozen = _get_frozen(ctx, frozen_default)
   347	    order = _determine_eq_order(ctx)
   348	    slots = _get_decorator_bool_argument(ctx, "slots", slots_default)
   349
   350	    auto_attribs = _get_decorator_optional_bool_argument(ctx, "auto_attribs", auto_attribs_default)
   351	    kw_only = _get_decorator_bool_argument(ctx, "kw_only", False)
   352	    match_args = _get_decorator_bool_argument(ctx, "match_args", True)
   353
   354	    for super_info in ctx.cls.info.mro[1:-1]:
   355	        if "attrs_tag" in super_info.metadata and "attrs" not in super_info.metadata:
   356	            # Super class is not ready yet. Request another pass.
   357	            return False
   358
   359	    attributes = _analyze_class(ctx, auto_attribs, kw_only)
   360
   361	    # Check if attribute types are ready.
   362	    for attr in attributes:
   363	        node = info.get(attr.name)
   364	        if node is None:
   365	            # This name is likely blocked by some semantic analysis error that
   366	            # should have been reported already.
   367	            _add_empty_metadata(info)
   368	            return True
   369
   370	    _add_attrs_magic_attribute(ctx, [(attr.name, info[attr.name].type) for attr in attributes])
   371	    if slots:
   372	        _add_slots(ctx, attributes)
   373	    if match_args and ctx.api.options.python_version[:2] >= (3, 10):
   374	        # `.__match_args__` is only added for python3.10+, but the argument
   375	        # exists for earlier versions as well.
   376	        _add_match_args(ctx, attributes)
   377
   378	    # Save the attributes so that subclasses can reuse them.
   379	    ctx.cls.info.metadata["attrs"] = {
   380	        "attributes": [attr.serialize() for attr in attributes],
   381	        "frozen": frozen,
   382	    }
   383
   384	    adder = MethodAdder(ctx)
   385	    # If  __init__ is not being generated, attrs still generates it as __attrs_init__ instead.
   386	    _add_init(ctx, attributes, adder, "__init__" if init else ATTRS_INIT_NAME)
   387
   388	    if order:
   389	        _add_order(ctx, adder)
   390	    if frozen:
   391	        _make_frozen(ctx, attributes)
   392	        # Frozen classes are hashable by default, even if inheriting from non-frozen ones.
   393	        hashable: bool | None = _get_decorator_bool_argument(
   394	            ctx, "hash", True
   395	        ) and _get_decorator_bool_argument(ctx, "unsafe_hash", True)
   396	    else:
   397	        hashable = _get_decorator_optional_bool_argument(ctx, "unsafe_hash")
   398	        if hashable is None:  # unspecified
   399	            hashable = _get_decorator_optional_bool_argument(ctx, "hash")
   400
   401	    eq = _get_decorator_optional_bool_argument(ctx, "eq")
   402	    has_own_hash = "__hash__" in ctx.cls.info.names
   403
   404	    if has_own_hash or (hashable is None and eq is False):
   405	        pass  # Do nothing.
   406	    elif hashable:
   407	        # We copy the `__hash__` signature from `object` to make them hashable.
   408	        ctx.cls.info.names["__hash__"] = ctx.cls.info.mro[-1].names["__hash__"]
   409	    else:
   410	        _remove_hashability(ctx)
   411
   412	    return True
   413

   415	def _get_frozen(ctx: mypy.plugin.ClassDefContext, frozen_default: bool) -> bool:
   416	    """Return whether this class is frozen."""
   417	    if _get_decorator_bool_argument(ctx, "frozen", frozen_default):
   418	        return True
   419	    # Subclasses of frozen classes are frozen so check that.
   420	    for super_info in ctx.cls.info.mro[1:-1]:
   421	        if "attrs" in super_info.metadata and super_info.metadata["attrs"]["frozen"]:
   422	            return True
   423	    return False

   426	def _analyze_class(
   427	    ctx: mypy.plugin.ClassDefContext, auto_attribs: bool | None, kw_only: bool
   428	) -> list[Attribute]:
   429	    """Analyze the class body of an attr maker, its parents, and return the Attributes found.
   430
   431	    auto_attribs=True means we'll generate attributes from type annotations also.
   432	    auto_attribs=None means we'll detect which mode to use.
   433	    kw_only=True means that all attributes created here will be keyword only args in __init__.
   434	    """
   435	    own_attrs: dict[str, Attribute] = {}
   436	    if auto_attribs is None:
   437	        auto_attribs = _detect_auto_attribs(ctx)
   438
   439	    # Walk the body looking for assignments and decorators.
   440	    for stmt in ctx.cls.defs.body:
   441	        if isinstance(stmt, AssignmentStmt):
   442	            for attr in _attributes_from_assignment(ctx, stmt, auto_attribs, kw_only):
   443	                # When attrs are defined twice in the same body we want to use the 2nd definition
   444	                # in the 2nd location. So remove it from the OrderedDict.
   445	                # Unless it's auto_attribs in which case we want the 2nd definition in the
   446	                # 1st location.
   447	                if not auto_attribs and attr.name in own_attrs:
   448	                    del own_attrs[attr.name]
   449	                own_attrs[attr.name] = attr
   450	        elif isinstance(stmt, Decorator):
   451	            _cleanup_decorator(stmt, own_attrs)
   452
   453	    for attribute in own_attrs.values():
   454	        # Even though these look like class level assignments we want them to look like
   455	        # instance level assignments.
   456	        if attribute.name in ctx.cls.info.names:
   457	            node = ctx.cls.info.names[attribute.name].node
   458	            if isinstance(node, PlaceholderNode):
   459	                # This node is not ready yet.
   460	                continue
   461	            assert isinstance(node, Var)
   462	            node.is_initialized_in_class = False
   463
   464	    # Traverse the MRO and collect attributes from the parents.
   465	    taken_attr_names = set(own_attrs)
   466	    super_attrs = []
   467	    for super_info in ctx.cls.info.mro[1:-1]:
   468	        if "attrs" in super_info.metadata:
   469	            # Each class depends on the set of attributes in its attrs ancestors.
   470	            ctx.api.add_plugin_dependency(make_wildcard_trigger(super_info.fullname))
   471
   472	            for data in super_info.metadata["attrs"]["attributes"]:
   473	                # Only add an attribute if it hasn't been defined before.  This
   474	                # allows for overwriting attribute definitions by subclassing.
   475	                if data["name"] not in taken_attr_names:
   476	                    a = Attribute.deserialize(super_info, data, ctx.api)
   477	                    a.expand_typevar_from_subtype(ctx.cls.info)
   478	                    super_attrs.append(a)
   479	                    taken_attr_names.add(a.name)
   480	    attributes = super_attrs + list(own_attrs.values())
   481
   482	    # Check the init args for correct default-ness.  Note: This has to be done after all the
   483	    # attributes for all classes have been read, because subclasses can override parents.
   484	    last_default = False
   485
   486	    for i, attribute in enumerate(attributes):
   487	        if not attribute.init:
   488	            continue
   489
   490	        if attribute.kw_only:
   491	            # Keyword-only attributes don't care whether they are default or not.
   492	            continue
   493
   494	        # If the issue comes from merging different classes, report it
   495	        # at the class definition point.
   496	        context = attribute.context if i >= len(super_attrs) else ctx.cls
   497
   498	        if not attribute.has_default and last_default:
   499	            ctx.api.fail("Non-default attributes not allowed after default attributes.", context)
   500	        last_default |= attribute.has_default
   501
   502	    return attributes
   503

   510	def _detect_auto_attribs(ctx: mypy.plugin.ClassDefContext) -> bool:
   511	    """Return whether auto_attribs should be enabled or disabled.
   512
   513	    It's disabled if there are any unannotated attribs()
   514	    """
   515	    for stmt in ctx.cls.defs.body:
   516	        if isinstance(stmt, AssignmentStmt):
   517	            for lvalue in stmt.lvalues:
   518	                lvalues, rvalues = _parse_assignments(lvalue, stmt)
   519
   520	                if len(lvalues) != len(rvalues):
   521	                    # This means we have some assignment that isn't 1 to 1.
   522	                    # It can't be an attrib.
   523	                    continue
   524
   525	                for lhs, rvalue in zip(lvalues, rvalues):
   526	                    # Check if the right hand side is a call to an attribute maker.
   527	                    if (
   528	                        isinstance(rvalue, CallExpr)
   529	                        and isinstance(rvalue.callee, RefExpr)
   530	                        and rvalue.callee.fullname in attr_attrib_makers
   531	                        and not stmt.new_syntax
   532	                    ):
   533	                        # This means we have an attrib without an annotation and so
   534	                        # we can't do auto_attribs=True
   535	                        return False
   536	    return True

   539	def _attributes_from_assignment(
   540	    ctx: mypy.plugin.ClassDefContext, stmt: AssignmentStmt, auto_attribs: bool, kw_only: bool
   541	) -> Iterable[Attribute]:
   542	    """Return Attribute objects that are created by this assignment.
   543
   544	    The assignments can look like this:
   545	        x = attr.ib()
   546	        x = y = attr.ib()
   547	        x, y = attr.ib(), attr.ib()
   548	    or if auto_attribs is enabled also like this:
   549	        x: type
   550	        x: type = default_value
   551	        x: type = attr.ib(...)
   552	    """
   553	    for lvalue in stmt.lvalues:
   554	        lvalues, rvalues = _parse_assignments(lvalue, stmt)
   555
   556	        if len(lvalues) != len(rvalues):
   557	            # This means we have some assignment that isn't 1 to 1.
   558	            # It can't be an attrib.
   559	            continue
   560
   561	        for lhs, rvalue in zip(lvalues, rvalues):
   562	            # Check if the right hand side is a call to an attribute maker.
   563	            if (
   564	                isinstance(rvalue, CallExpr)
   565	                and isinstance(rvalue.callee, RefExpr)
   566	                and rvalue.callee.fullname in attr_attrib_makers
   567	            ):
   568	                attr = _attribute_from_attrib_maker(ctx, auto_attribs, kw_only, lhs, rvalue, stmt)
   569	                if attr:
   570	                    yield attr
   571	            elif auto_attribs and stmt.type and stmt.new_syntax and not is_class_var(lhs):
   572	                yield _attribute_from_auto_attrib(ctx, kw_only, lhs, rvalue, stmt)
   573

   575	def _cleanup_decorator(stmt: Decorator, attr_map: dict[str, Attribute]) -> None:
   576	    """Handle decorators in class bodies.
   577
   578	    `x.default` will set a default value on x
   579	    `x.validator` and `x.default` will get removed to avoid throwing a type error.
   580	    """
   581	    remove_me = []
   582	    for func_decorator in stmt.decorators:
   583	        if (
   584	            isinstance(func_decorator, MemberExpr)
   585	            and isinstance(func_decorator.expr, NameExpr)
   586	            and func_decorator.expr.name in attr_map
   587	        ):
   588	            if func_decorator.name == "default":
   589	                attr_map[func_decorator.expr.name].has_default = True
   590
   591	            if func_decorator.name in ("default", "validator"):
   592	                # These are decorators on the attrib object that only exist during
   593	                # class creation time.  In order to not trigger a type error later we
   594	                # just remove them.  This might leave us with a Decorator with no
   595	                # decorators (Emperor's new clothes?)
   596	                # TODO: It would be nice to type-check these rather than remove them.
   597	                #       default should be Callable[[], T]
   598	                #       validator should be Callable[[Any, 'Attribute', T], Any]
   599	                #       where T is the type of the attribute.
   600	                remove_me.append(func_decorator)
   601	    for dec in remove_me:
   602	        stmt.decorators.remove(dec)

   605	def _attribute_from_auto_attrib(
   606	    ctx: mypy.plugin.ClassDefContext,
   607	    kw_only: bool,
   608	    lhs: NameExpr,
   609	    rvalue: Expression,
   610	    stmt: AssignmentStmt,
   611	) -> Attribute:
   612	    """Return an Attribute for a new type assignment."""
   613	    name = unmangle(lhs.name)
   614	    # `x: int` (without equal sign) assigns rvalue to TempNode(AnyType())
   615	    has_rhs = not isinstance(rvalue, TempNode)
   616	    sym = ctx.cls.info.names.get(name)
   617	    init_type = sym.type if sym else None
   618	    return Attribute(name, None, ctx.cls.info, has_rhs, True, kw_only, None, stmt, init_type)

   621	def _attribute_from_attrib_maker(
   622	    ctx: mypy.plugin.ClassDefContext,
   623	    auto_attribs: bool,
   624	    kw_only: bool,
   625	    lhs: NameExpr,
   626	    rvalue: CallExpr,
   627	    stmt: AssignmentStmt,
   628	) -> Attribute | None:
   629	    """Return an Attribute from the assignment or None if you can't make one."""
   630	    if auto_attribs and not stmt.new_syntax:
   631	        # auto_attribs requires an annotation on *every* attr.ib.
   632	        assert lhs.node is not None
   633	        ctx.api.msg.need_annotation_for_var(lhs.node, stmt)
   634	        return None
   635
   636	    if len(stmt.lvalues) > 1:
   637	        ctx.api.fail("Too many names for one attribute", stmt)
   638	        return None
   639
   640	    # This is the type that belongs in the __init__ method for this attrib.
   641	    init_type = stmt.type
