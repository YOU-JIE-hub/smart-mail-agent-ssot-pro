   315	class LineCoverageVisitor(TraverserVisitor):
   316	    def __init__(self, source: list[str]) -> None:
   317	        self.source = source
   318
   319	        # For each line of source, we maintain a pair of
   320	        #  * the indentation level of the surrounding function
   321	        #    (-1 if not inside a function), and
   322	        #  * whether the surrounding function is typed.
   323	        # Initially, everything is covered at indentation level -1.
   324	        self.lines_covered = [(-1, True) for l in source]
   325
   326	    # The Python AST has position information for the starts of
   327	    # elements, but not for their ends. Fortunately the
   328	    # indentation-based syntax makes it pretty easy to find where a
   329	    # block ends without doing any real parsing.
   330
   331	    # TODO: Handle line continuations (explicit and implicit) and
   332	    # multi-line string literals. (But at least line continuations
   333	    # are normally more indented than their surrounding block anyways,
   334	    # by PEP 8.)
   335
   336	    def indentation_level(self, line_number: int) -> int | None:
   337	        """Return the indentation of a line of the source (specified by
   338	        zero-indexed line number). Returns None for blank lines or comments."""
   339	        line = self.source[line_number]
   340	        indent = 0
   341	        for char in list(line):
   342	            if char == " ":
   343	                indent += 1
   344	            elif char == "\t":
   345	                indent = 8 * ((indent + 8) // 8)
   346	            elif char == "#":
   347	                # Line is a comment; ignore it
   348	                return None
   349	            elif char == "\n":
   350	                # Line is entirely whitespace; ignore it
   351	                return None
   352	            # TODO line continuation (\)
   353	            else:
   354	                # Found a non-whitespace character
   355	                return indent
   356	        # Line is entirely whitespace, and at end of file
   357	        # with no trailing newline; ignore it
   358	        return None
   359
   360	    def visit_func_def(self, defn: FuncDef) -> None:
   361	        start_line = defn.line - 1
   362	        start_indent = None
   363	        # When a function is decorated, sometimes the start line will point to
   364	        # whitespace or comments between the decorator and the function, so
   365	        # we have to look for the start.
   366	        while start_line < len(self.source):
   367	            start_indent = self.indentation_level(start_line)
   368	            if start_indent is not None:
   369	                break
   370	            start_line += 1
   371	        # If we can't find the function give up and don't annotate anything.
   372	        # Our line numbers are not reliable enough to be asserting on.
   373	        if start_indent is None:
   374	            return
   375
   376	        cur_line = start_line + 1
   377	        end_line = cur_line
   378	        # After this loop, function body will be lines [start_line, end_line)
   379	        while cur_line < len(self.source):
   380	            cur_indent = self.indentation_level(cur_line)
   381	            if cur_indent is None:
   382	                # Consume the line, but don't mark it as belonging to the function yet.
   383	                cur_line += 1
   384	            elif cur_indent > start_indent:
   385	                # A non-blank line that belongs to the function.
   386	                cur_line += 1
   387	                end_line = cur_line
   388	            else:
   389	                # We reached a line outside the function definition.
   390	                break
   391
   392	        is_typed = defn.type is not None
   393	        for line in range(start_line, end_line):
   394	            old_indent, _ = self.lines_covered[line]
   395	            # If there was an old indent level for this line, and the new
   396	            # level isn't increasing the indentation, ignore it.
   397	            # This is to be defensive against funniness in our line numbers,
   398	            # which are not always reliable.
   399	            if old_indent <= start_indent:
   400	                self.lines_covered[line] = (start_indent, is_typed)
   401
   402	        # Visit the body, in case there are nested functions
   403	        super().visit_func_def(defn)

   528	    def inside_except_star_block_set(
   529	        self, value: bool, entering_loop: bool = False
   530	    ) -> Iterator[None]:
   531	        old = self.inside_except_star_block
   532	        self.inside_except_star_block = value
   533
   534	        # Return statement would still be in except* scope if entering loops
   535	        if not entering_loop:
   536	            old_return_stmt_flag = self.return_stmt_inside_except_star_block
   537	            self.return_stmt_inside_except_star_block = value
   538
   539	        try:
   540	            yield
   541	        finally:
   542	            self.inside_except_star_block = old
   543	            if not entering_loop:
   544	                self.return_stmt_inside_except_star_block = old_return_stmt_flag

   550	    def prepare_file(self, file_node: MypyFile) -> None:
   551	        """Prepare a freshly parsed file for semantic analysis."""
   552	        if "builtins" in self.modules:
   553	            file_node.names["__builtins__"] = SymbolTableNode(GDEF, self.modules["builtins"])
   554	        if file_node.fullname == "builtins":
   555	            self.prepare_builtins_namespace(file_node)
   556	        if file_node.fullname == "typing":
   557	            self.prepare_typing_namespace(file_node, type_aliases)
   558	        if file_node.fullname == "typing_extensions":
   559	            self.prepare_typing_namespace(file_node, typing_extensions_aliases)

   561	    def prepare_typing_namespace(self, file_node: MypyFile, aliases: dict[str, str]) -> None:
   562	        """Remove dummy alias definitions such as List = TypeAlias(object) from typing.
   563
   564	        They will be replaced with real aliases when corresponding targets are ready.
   565	        """
   566
   567	        # This is all pretty unfortunate. typeshed now has a
   568	        # sys.version_info check for OrderedDict, and we shouldn't
   569	        # take it out, because it is correct and a typechecker should
   570	        # use that as a source of truth. But instead we rummage
   571	        # through IfStmts to remove the info first.  (I tried to
   572	        # remove this whole machinery and ran into issues with the
   573	        # builtins/typing import cycle.)
   574	        def helper(defs: list[Statement]) -> None:
   575	            for stmt in defs.copy():
   576	                if isinstance(stmt, IfStmt):
   577	                    for body in stmt.body:
   578	                        helper(body.body)
   579	                    if stmt.else_body:
   580	                        helper(stmt.else_body.body)
   581	                if (
   582	                    isinstance(stmt, AssignmentStmt)
   583	                    and len(stmt.lvalues) == 1
   584	                    and isinstance(stmt.lvalues[0], NameExpr)
   585	                ):
   586	                    # Assignment to a simple name, remove it if it is a dummy alias.
   587	                    if f"{file_node.fullname}.{stmt.lvalues[0].name}" in aliases:
   588	                        defs.remove(stmt)
   589
   590	        helper(file_node.defs)

   592	    def prepare_builtins_namespace(self, file_node: MypyFile) -> None:
   593	        """Add certain special-cased definitions to the builtins module.
   594
   595	        Some definitions are too special or fundamental to be processed
   596	        normally from the AST.
   597	        """
   598	        names = file_node.names
   599
   600	        # Add empty definition for core built-in classes, since they are required for basic
   601	        # operation. These will be completed later on.
   602	        for name in CORE_BUILTIN_CLASSES:
   603	            cdef = ClassDef(name, Block([]))  # Dummy ClassDef, will be replaced later
   604	            info = TypeInfo(SymbolTable(), cdef, "builtins")
   605	            info._fullname = f"builtins.{name}"
   606	            names[name] = SymbolTableNode(GDEF, info)
   607
   608	        bool_info = names["bool"].node
   609	        assert isinstance(bool_info, TypeInfo)
   610	        bool_type = Instance(bool_info, [])
   611
   612	        special_var_types: list[tuple[str, Type]] = [
   613	            ("None", NoneType()),
   614	            # reveal_type is a mypy-only function that gives an error with
   615	            # the type of its arg.
   616	            ("reveal_type", AnyType(TypeOfAny.special_form)),
   617	            # reveal_locals is a mypy-only function that gives an error with the types of
   618	            # locals
   619	            ("reveal_locals", AnyType(TypeOfAny.special_form)),
   620	            ("True", bool_type),
   621	            ("False", bool_type),
   622	            ("__debug__", bool_type),
   623	        ]
   624
   625	        for name, typ in special_var_types:
   626	            v = Var(name, typ)
   627	            v._fullname = f"builtins.{name}"
   628	            file_node.names[name] = SymbolTableNode(GDEF, v)
