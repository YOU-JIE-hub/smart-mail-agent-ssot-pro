   526	def get_args(builder: IRBuilder, rt_args: Sequence[RuntimeArg], line: int) -> ArgInfo:
   527	    # The environment operates on Vars, so we make some up
   528	    fake_vars = [(Var(arg.name), arg.type) for arg in rt_args]

   531	        for var, type in fake_vars

   534	        arg.name if arg.kind.is_named() or (arg.kind.is_optional() and not arg.pos_only) else None
   535	        for arg in rt_args

   537	    arg_kinds = [arg.kind for arg in rt_args]
   538	    return ArgInfo(args, arg_names, arg_kinds)

   541	def gen_glue_method(
   542	    builder: IRBuilder,
   543	    base_sig: FuncSignature,
   544	    target: FuncIR,
   545	    cls: ClassIR,
   546	    base: ClassIR,
   547	    line: int,
   548	    do_pycall: bool,
   549	) -> FuncIR:
   550	    """Generate glue methods that mediate between different method types in subclasses.
   551
   552	    For example, if we have:
   553
   554	    class A:
   555	        def f(builder: IRBuilder, x: int) -> object: ...
   556
   557	    then it is totally permissible to have a subclass
   558
   559	    class B(A):
   560	        def f(builder: IRBuilder, x: object) -> int: ...
   561
   562	    since '(object) -> int' is a subtype of '(int) -> object' by the usual
   563	    contra/co-variant function subtyping rules.
   564
   565	    The trickiness here is that int and object have different
   566	    runtime representations in mypyc, so A.f and B.f have
   567	    different signatures at the native C level. To deal with this,
   568	    we need to generate glue methods that mediate between the
   569	    different versions by coercing the arguments and return
   570	    values.
   571
   572	    If do_pycall is True, then make the call using the C API
   573	    instead of a native call.
   574	    """
   575	    check_native_override(builder, base_sig, target.decl.sig, line)
   576
   577	    builder.enter()
   578	    builder.ret_types[-1] = base_sig.ret_type
   579
   580	    rt_args = list(base_sig.args)
   581	    if target.decl.kind == FUNC_NORMAL:
   582	        rt_args[0] = RuntimeArg(base_sig.args[0].name, RInstance(cls))
   583
   584	    arg_info = get_args(builder, rt_args, line)
   585	    args, arg_kinds, arg_names = arg_info.args, arg_info.arg_kinds, arg_info.arg_names
   586
   587	    bitmap_args = None
   588	    if base_sig.num_bitmap_args:
   589	        args = args[: -base_sig.num_bitmap_args]
   590	        arg_kinds = arg_kinds[: -base_sig.num_bitmap_args]
   591	        arg_names = arg_names[: -base_sig.num_bitmap_args]
   592	        bitmap_args = list(builder.builder.args[-base_sig.num_bitmap_args :])
   593
   594	    # We can do a passthrough *args/**kwargs with a native call, but if the
   595	    # args need to get distributed out to arguments, we just let python handle it
   596	    if any(kind.is_star() for kind in arg_kinds) and any(
   597	        not arg.kind.is_star() for arg in target.decl.sig.args
   598	    ):
   599	        do_pycall = True
   600
   601	    if do_pycall:
   602	        if target.decl.kind == FUNC_STATICMETHOD:
   603	            # FIXME: this won't work if we can do interpreted subclasses
   604	            first = builder.builder.get_native_type(cls)
   605	            st = 0
   606	        else:
   607	            first = args[0]
   608	            st = 1
   609	        retval = builder.builder.py_method_call(
   610	            first, target.name, args[st:], line, arg_kinds[st:], arg_names[st:]
   611	        )
   612	    else:
   613	        retval = builder.builder.call(
   614	            target.decl, args, arg_kinds, arg_names, line, bitmap_args=bitmap_args
   615	        )
   616	    retval = builder.coerce(retval, base_sig.ret_type, line)
   617	    builder.add(Return(retval))
   618
   619	    arg_regs, _, blocks, ret_type, _ = builder.leave()
   620	    if base_sig.num_bitmap_args:
   621	        rt_args = rt_args[: -base_sig.num_bitmap_args]
   622	    return FuncIR(
   623	        FuncDecl(
   624	            target.name + "__" + base.name + "_glue",
   625	            cls.name,
   626	            builder.module_name,
   627	            FuncSignature(rt_args, ret_type),
   628	            target.decl.kind,
   629	        ),
   630	        arg_regs,
   631	        blocks,
   632	    )

   635	def check_native_override(
   636	    builder: IRBuilder, base_sig: FuncSignature, sub_sig: FuncSignature, line: int
   637	) -> None:
   638	    """Report an error if an override changes signature in unsupported ways.
   639
   640	    Glue methods can work around many signature changes but not all of them.
   641	    """
   642	    for base_arg, sub_arg in zip(base_sig.real_args(), sub_sig.real_args()):
   643	        if base_arg.type.error_overlap:
   644	            if not base_arg.optional and sub_arg.optional and base_sig.num_bitmap_args:
   645	                # This would change the meanings of bits in the argument defaults
   646	                # bitmap, which we don't support. We'd need to do tricky bit
   647	                # manipulations to support this generally.
   648	                builder.error(
   649	                    "An argument with type "
   650	                    + f'"{base_arg.type}" cannot be given a default value in a method override',
   651	                    line,
   652	                )
   653	        if base_arg.type.error_overlap or sub_arg.type.error_overlap:
   654	            if not is_same_type(base_arg.type, sub_arg.type):
   655	                # This would change from signaling a default via an error value to
   656	                # signaling a default via bitmap, which we don't support.
   657	                builder.error(
   658	                    "Incompatible argument type "
   659	                    + f'"{sub_arg.type}" (base class has type "{base_arg.type}")',
   660	                    line,
   661	                )

   664	def gen_glue_property(
   665	    builder: IRBuilder,
   666	    sig: FuncSignature,
   667	    target: FuncIR,
   668	    cls: ClassIR,
   669	    base: ClassIR,
   670	    line: int,
   671	    do_pygetattr: bool,
   672	) -> FuncIR:
   673	    """Generate glue methods for properties that mediate between different subclass types.
   674
   675	    Similarly to methods, properties of derived types can be covariantly subtyped. Thus,
   676	    properties also require glue. However, this only requires the return type to change.
   677	    Further, instead of a method call, an attribute get is performed.
   678
   679	    If do_pygetattr is True, then get the attribute using the Python C
   680	    API instead of a native call.
   681	    """
   682	    builder.enter()
   683
   684	    rt_arg = RuntimeArg(SELF_NAME, RInstance(cls))
   685	    self_target = builder.add_self_to_env(cls)
   686	    arg = builder.read(self_target, line)
   687	    builder.ret_types[-1] = sig.ret_type
   688	    if do_pygetattr:
   689	        retval = builder.py_get_attr(arg, target.name, line)
   690	    else:
   691	        retval = builder.add(GetAttr(arg, target.name, line))
   692	    retbox = builder.coerce(retval, sig.ret_type, line)
   693	    builder.add(Return(retbox))
   694
   695	    args, _, blocks, return_type, _ = builder.leave()
   696	    return FuncIR(
   697	        FuncDecl(
   698	            target.name + "__" + base.name + "_glue",
   699	            cls.name,
   700	            builder.module_name,
   701	            FuncSignature([rt_arg], return_type),
   702	        ),
   703	        args,
   704	        blocks,
   705	    )

   718	    if builder.fn_info.is_generator or builder.fn_info.add_nested_funcs_to_env:

   724	# This function still does not support the following imports.
   725	# import json as _json
   726	# from json import decoder
   727	# Using either _json.JSONDecoder or decoder.JSONDecoder as a type hint for a dataclass field will fail.
   728	# See issue mypyc/mypyc#1099.
   729	def load_type(builder: IRBuilder, typ: TypeInfo, unbounded_type: Type | None, line: int) -> Value:
   730	    # typ.fullname contains the module where the class object was defined. However, it is possible
   731	    # that the class object's module was not imported in the file currently being compiled. So, we
   732	    # use unbounded_type.name (if provided by caller) to load the class object through one of the
   733	    # imported modules.
   734	    # Example: for `json.JSONDecoder`, typ.fullname is `json.decoder.JSONDecoder` but the Python
   735	    # file may import `json` not `json.decoder`.
   736	    # Another corner case: The Python file being compiled imports mod1 and has a type hint
   737	    # `mod1.OuterClass.InnerClass`. But, mod1/__init__.py might import OuterClass like this:
   738	    # `from mod2.mod3 import OuterClass`. In this case, typ.fullname is
   739	    # `mod2.mod3.OuterClass.InnerClass` and `unbounded_type.name` is `mod1.OuterClass.InnerClass`.
   740	    # So, we must use unbounded_type.name to load the class object.
   741	    # See issue mypyc/mypyc#1087.
   742	    load_attr_path = (
   743	        unbounded_type.name if isinstance(unbounded_type, UnboundType) else typ.fullname
   744	    ).removesuffix(f".{typ.name}")
   745	    if typ in builder.mapper.type_to_ir:
   746	        class_ir = builder.mapper.type_to_ir[typ]
   747	        class_obj = builder.builder.get_native_type(class_ir)
   748	    elif typ.fullname in builtin_names:
   749	        builtin_addr_type, src = builtin_names[typ.fullname]
   750	        class_obj = builder.add(LoadAddress(builtin_addr_type, src, line))
   751	    # This elif-condition finds the longest import that matches the load_attr_path.
   752	    elif module_name := max(
   753	        (i for i in builder.imports if load_attr_path == i or load_attr_path.startswith(f"{i}.")),
   754	        default="",
   755	        key=len,
   756	    ):
   757	        # Load the imported module.
   758	        loaded_module = builder.load_module(module_name)
   759	        # Recursively load attributes of the imported module. These may be submodules, classes or
   760	        # any other object.
   761	        for attr in (
   762	            load_attr_path.removeprefix(f"{module_name}.").split(".")
   763	            if load_attr_path != module_name
   764	            else []
   765	        ):
   766	            loaded_module = builder.py_get_attr(loaded_module, attr, line)
   767	        class_obj = builder.builder.get_attr(
   768	            loaded_module, typ.name, object_rprimitive, line, borrow=False
   769	        )
   770	    else:
   771	        class_obj = builder.load_global_str(typ.name, line)
   772
   773	    return class_obj

   776	def load_func(builder: IRBuilder, func_name: str, fullname: str | None, line: int) -> Value:
   777	    if fullname and not fullname.startswith(builder.current_module):
   778	        # we're calling a function in a different module
   779
   780	        # We can't use load_module_attr_by_fullname here because we need to load the function using
   781	        # func_name, not the name specified by fullname (which can be different for underscore
   782	        # function)
   783	        module = fullname.rsplit(".")[0]
   784	        loaded_module = builder.load_module(module)
   785
   786	        func = builder.py_get_attr(loaded_module, func_name, line)
   787	    else:
   788	        func = builder.load_global_str(func_name, line)
   789	    return func

   792	def generate_singledispatch_dispatch_function(
   793	    builder: IRBuilder, main_singledispatch_function_name: str, fitem: FuncDef
   794	) -> None:
   795	    line = fitem.line
   796	    current_func_decl = builder.mapper.func_to_decl[fitem]
   797	    arg_info = get_args(builder, current_func_decl.sig.args, line)
   798
   799	    dispatch_func_obj = builder.self()
   800
   801	    arg_type = builder.builder.get_type_of_obj(arg_info.args[0], line)
   802	    dispatch_cache = builder.builder.get_attr(
   803	        dispatch_func_obj, "dispatch_cache", dict_rprimitive, line
   804	    )
   805	    call_find_impl, use_cache, call_func = BasicBlock(), BasicBlock(), BasicBlock()
   806	    get_result = builder.primitive_op(dict_get_method_with_none, [dispatch_cache, arg_type], line)
   807	    is_not_none = builder.translate_is_op(get_result, builder.none_object(), "is not", line)
   808	    impl_to_use = Register(object_rprimitive)
   809	    builder.add_bool_branch(is_not_none, use_cache, call_find_impl)
   810
   811	    builder.activate_block(use_cache)
   812	    builder.assign(impl_to_use, get_result, line)
   813	    builder.goto(call_func)
   814
   815	    builder.activate_block(call_find_impl)
   816	    find_impl = builder.load_module_attr_by_fullname("functools._find_impl", line)
   817	    registry = load_singledispatch_registry(builder, dispatch_func_obj, line)
   818	    uncached_impl = builder.py_call(find_impl, [arg_type, registry], line)
   819	    builder.primitive_op(dict_set_item_op, [dispatch_cache, arg_type, uncached_impl], line)
   820	    builder.assign(impl_to_use, uncached_impl, line)
   821	    builder.goto(call_func)
   822
   823	    builder.activate_block(call_func)
   824	    gen_calls_to_correct_impl(builder, impl_to_use, arg_info, fitem, line)

   827	def gen_calls_to_correct_impl(
   828	    builder: IRBuilder, impl_to_use: Value, arg_info: ArgInfo, fitem: FuncDef, line: int
   829	) -> None:
   830	    current_func_decl = builder.mapper.func_to_decl[fitem]
   831
   832	    def gen_native_func_call_and_return(fdef: FuncDef) -> None:
   833	        func_decl = builder.mapper.func_to_decl[fdef]
   834	        ret_val = builder.builder.call(
   835	            func_decl, arg_info.args, arg_info.arg_kinds, arg_info.arg_names, line
   836	        )
   837	        coerced = builder.coerce(ret_val, current_func_decl.sig.ret_type, line)
   838	        builder.add(Return(coerced))
   839
   840	    typ, src = builtin_names["builtins.int"]
   841	    int_type_obj = builder.add(LoadAddress(typ, src, line))
   842	    is_int = builder.builder.type_is_op(impl_to_use, int_type_obj, line)
   843
   844	    native_call, non_native_call = BasicBlock(), BasicBlock()
   845	    builder.add_bool_branch(is_int, native_call, non_native_call)
   846	    builder.activate_block(native_call)
   847
   848	    passed_id = builder.add(Unbox(impl_to_use, int_rprimitive, line))
   849
   850	    native_ids = get_native_impl_ids(builder, fitem)
   851	    for impl, i in native_ids.items():
   852	        call_impl, next_impl = BasicBlock(), BasicBlock()
   853
   854	        current_id = builder.load_int(i)
   855	        cond = builder.binary_op(passed_id, current_id, "==", line)
   856	        builder.add_bool_branch(cond, call_impl, next_impl)
   857
   858	        # Call the registered implementation
   859	        builder.activate_block(call_impl)
   860
   861	        gen_native_func_call_and_return(impl)
   862	        builder.activate_block(next_impl)
   863
   864	    # We've already handled all the possible integer IDs, so we should never get here
   865	    builder.add(Unreachable())
   866
   867	    builder.activate_block(non_native_call)
   868	    ret_val = builder.py_call(
   869	        impl_to_use, arg_info.args, line, arg_info.arg_kinds, arg_info.arg_names
   870	    )
   871	    coerced = builder.coerce(ret_val, current_func_decl.sig.ret_type, line)
   872	    builder.add(Return(coerced))

   875	def gen_dispatch_func_ir(
   876	    builder: IRBuilder, fitem: FuncDef, main_func_name: str, dispatch_name: str, sig: FuncSignature
   877	) -> tuple[FuncIR, Value]:
   878	    """Create a dispatch function (a function that checks the first argument type and dispatches
   879	    to the correct implementation)
   880	    """
   881	    builder.enter(FuncInfo(fitem, dispatch_name))
   882	    setup_callable_class(builder)
   883	    builder.fn_info.callable_class.ir.attributes["registry"] = dict_rprimitive
   884	    builder.fn_info.callable_class.ir.attributes["dispatch_cache"] = dict_rprimitive
   885	    builder.fn_info.callable_class.ir.has_dict = True
   886	    builder.fn_info.callable_class.ir.needs_getseters = True
   887	    generate_singledispatch_callable_class_ctor(builder)
   888
   889	    generate_singledispatch_dispatch_function(builder, main_func_name, fitem)
   890	    args, _, blocks, _, fn_info = builder.leave()
   891	    dispatch_callable_class = add_call_to_callable_class(builder, args, blocks, sig, fn_info)
   892	    builder.functions.append(dispatch_callable_class)
   893	    add_get_to_callable_class(builder, fn_info)
   894	    add_register_method_to_callable_class(builder, fn_info)
   895	    func_reg = instantiate_callable_class(builder, fn_info)
   896	    dispatch_func_ir = generate_dispatch_glue_native_function(
   897	        builder, fitem, dispatch_callable_class.decl, dispatch_name
   898	    )
   899
   900	    return dispatch_func_ir, func_reg

   922	def generate_singledispatch_callable_class_ctor(builder: IRBuilder) -> None:
   923	    """Create an __init__ that sets registry and dispatch_cache to empty dicts"""
   924	    line = -1
   925	    class_ir = builder.fn_info.callable_class.ir
   926	    with builder.enter_method(class_ir, "__init__", bool_rprimitive):
   927	        empty_dict = builder.call_c(dict_new_op, [], line)
   928	        builder.add(SetAttr(builder.self(), "registry", empty_dict, line))
   929	        cache_dict = builder.call_c(dict_new_op, [], line)
   930	        dispatch_cache_str = builder.load_str("dispatch_cache")
   931	        # use the py_setattr_op instead of SetAttr so that it also gets added to our __dict__
   932	        builder.primitive_op(py_setattr_op, [builder.self(), dispatch_cache_str, cache_dict], line)
   933	        # the generated C code seems to expect that __init__ returns a char, so just return 1
   934	        builder.add(Return(Integer(1, bool_rprimitive, line), line))

   954	def maybe_insert_into_registry_dict(builder: IRBuilder, fitem: FuncDef) -> None:
   955	    line = fitem.line
   956	    is_singledispatch_main_func = fitem in builder.singledispatch_impls
   957	    # dict of singledispatch_func to list of register_types (fitem is the function to register)
   958	    to_register: defaultdict[FuncDef, list[TypeInfo]] = defaultdict(list)
   959	    for main_func, impls in builder.singledispatch_impls.items():
   960	        for dispatch_type, impl in impls:
   961	            if fitem == impl:
   962	                to_register[main_func].append(dispatch_type)
   963
   964	    if not to_register and not is_singledispatch_main_func:
   965	        return
   966
   967	    if is_singledispatch_main_func:
   968	        main_func_name = singledispatch_main_func_name(fitem.name)
   969	        main_func_obj = load_func(builder, main_func_name, fitem.fullname, line)
   970
   971	        loaded_object_type = builder.load_module_attr_by_fullname("builtins.object", line)
   972	        registry_dict = builder.builder.make_dict([(loaded_object_type, main_func_obj)], line)
   973
   974	        dispatch_func_obj = builder.load_global_str(fitem.name, line)
   975	        builder.primitive_op(
   976	            py_setattr_op, [dispatch_func_obj, builder.load_str("registry"), registry_dict], line
   977	        )
   978
   979	    for singledispatch_func, types in to_register.items():
   980	        # TODO: avoid recomputing the native IDs for all the functions every time we find a new
   981	        # function
   982	        native_ids = get_native_impl_ids(builder, singledispatch_func)
   983	        if fitem not in native_ids:
   984	            to_insert = load_func(builder, fitem.name, fitem.fullname, line)
   985	        else:

   223	class LowLevelIRBuilder:
   224	    """A "low-level" IR builder class.
   225
   226	    LowLevelIRBuilder provides core abstractions we use for constructing
   227	    IR as well as a number of higher-level ones (accessing attributes,
   228	    calling functions and methods, and coercing between types, for
   229	    example).
   230
   231	    The core principle of the low-level IR builder is that all of its
   232	    facilities operate solely on the mypyc IR level and not the mypy AST
   233	    level---it has *no knowledge* of mypy types or expressions.
   234
   235	    The mypyc.irbuilder.builder.IRBuilder class wraps an instance of this
   236	    class and provides additional functionality to transform mypy AST nodes
   237	    to IR.
   238	    """
   239
   240	    def __init__(self, errors: Errors | None, options: CompilerOptions) -> None:
   241	        self.errors = errors
   242	        self.options = options
   243	        self.args: list[Register] = []
   244	        self.blocks: list[BasicBlock] = []
   245	        # Stack of except handler entry blocks
   246	        self.error_handlers: list[BasicBlock | None] = [None]
   247	        # Values that we need to keep alive as long as we have borrowed
   248	        # temporaries. Use flush_keep_alives() to mark the end of the live range.
   249	        self.keep_alives: list[Value] = []
   250
   251	    def set_module(self, module_name: str, module_path: str) -> None:
   252	        """Set the name and path of the current module."""
   253	        self.module_name = module_name
   254	        self.module_path = module_path
   255
   256	    # Basic operations
   257
   258	    def add(self, op: Op) -> Value:
   259	        """Add an op."""
   260	        assert not self.blocks[-1].terminated, "Can't add to finished block"
   261	        self.blocks[-1].ops.append(op)
   262	        return op
   263
   264	    def goto(self, target: BasicBlock) -> None:
   265	        """Add goto to a basic block."""
   266	        if not self.blocks[-1].terminated:
   267	            self.add(Goto(target))
   268
   269	    def activate_block(self, block: BasicBlock) -> None:
   270	        """Add a basic block and make it the active one (target of adds)."""
   271	        if self.blocks:
   272	            assert self.blocks[-1].terminated
   273
   274	        block.error_handler = self.error_handlers[-1]
   275	        self.blocks.append(block)
   276
   277	    def goto_and_activate(self, block: BasicBlock) -> None:
   278	        """Add goto a block and make it the active block."""
   279	        self.goto(block)
   280	        self.activate_block(block)
   281
   282	    def keep_alive(self, values: list[Value], *, steal: bool = False) -> None:
   283	        self.add(KeepAlive(values, steal=steal))
   284
   285	    def push_error_handler(self, handler: BasicBlock | None) -> None:
   286	        self.error_handlers.append(handler)
   287
   288	    def pop_error_handler(self) -> BasicBlock | None:
   289	        return self.error_handlers.pop()
   290
   291	    def self(self) -> Register:
   292	        """Return reference to the 'self' argument.
   293
   294	        This only works in a method.
   295	        """
   296	        return self.args[0]
   297
   298	    def flush_keep_alives(self) -> None:
   299	        if self.keep_alives:
   300	            self.add(KeepAlive(self.keep_alives.copy()))
   301	            self.keep_alives = []
   302
   303	    def debug_print(self, toprint: str | Value) -> None:
   304	        if isinstance(toprint, str):
   305	            toprint = self.load_str(toprint)
   306	        self.primitive_op(debug_print_op, [toprint], -1)

   308	    # Type conversions
   309
   310	    def box(self, src: Value) -> Value:
   311	        if src.type.is_unboxed:
   312	            if isinstance(src, Integer) and is_tagged(src.type):
   313	                return self.add(LoadLiteral(src.value >> 1, rtype=object_rprimitive))
   314	            return self.add(Box(src))
   315	        else:
   316	            return src
   317
   318	    def unbox_or_cast(
   319	        self, src: Value, target_type: RType, line: int, *, can_borrow: bool = False
   320	    ) -> Value:
   321	        if target_type.is_unboxed:
   322	            return self.add(Unbox(src, target_type, line))
   323	        else:
   324	            if can_borrow:
   325	                self.keep_alives.append(src)
   326	            return self.add(Cast(src, target_type, line, borrow=can_borrow))
   327
   328	    def coerce(
   329	        self,
   330	        src: Value,
   331	        target_type: RType,
   332	        line: int,
   333	        force: bool = False,
   334	        *,
   335	        can_borrow: bool = False,
   336	    ) -> Value:
   337	        """Generate a coercion/cast from one type to other (only if needed).
   338
   339	        For example, int -> object boxes the source int; int -> int emits nothing;
   340	        object -> int unboxes the object. All conversions preserve object value.
   341
   342	        If force is true, always generate an op (even if it is just an assignment) so
   343	        that the result will have exactly target_type as the type.
   344
   345	        Returns the register with the converted value (may be same as src).
   346	        """
   347	        src_type = src.type
   348	        if src_type.is_unboxed and not target_type.is_unboxed:
   349	            # Unboxed -> boxed
   350	            return self.box(src)
   351	        if (src_type.is_unboxed and target_type.is_unboxed) and not is_runtime_subtype(
   352	            src_type, target_type
   353	        ):
   354	            if (
   355	                isinstance(src, Integer)
   356	                and is_short_int_rprimitive(src_type)
   357	                and is_fixed_width_rtype(target_type)
   358	            ):
   359	                value = src.numeric_value()
   360	                if not check_native_int_range(target_type, value):
   361	                    self.error(f'Value {value} is out of range for "{target_type}"', line)
   362	                return Integer(src.value >> 1, target_type)
   363	            elif is_int_rprimitive(src_type) and is_fixed_width_rtype(target_type):
   364	                return self.coerce_int_to_fixed_width(src, target_type, line)
   365	            elif is_fixed_width_rtype(src_type) and is_int_rprimitive(target_type):
   366	                return self.coerce_fixed_width_to_int(src, line)
   367	            elif is_short_int_rprimitive(src_type) and is_fixed_width_rtype(target_type):
   368	                return self.coerce_short_int_to_fixed_width(src, target_type, line)
   369	            elif (
   370	                isinstance(src_type, RPrimitive)
   371	                and isinstance(target_type, RPrimitive)
   372	                and src_type.is_native_int
   373	                and target_type.is_native_int
   374	                and src_type.size == target_type.size
   375	                and src_type.is_signed == target_type.is_signed
   376	            ):
   377	                # Equivalent types
   378	                return src
   379	            elif (is_bool_rprimitive(src_type) or is_bit_rprimitive(src_type)) and is_tagged(
   380	                target_type
   381	            ):
   382	                shifted = self.int_op(
   383	                    bool_rprimitive, src, Integer(1, bool_rprimitive), IntOp.LEFT_SHIFT
   384	                )
   385	                return self.add(Extend(shifted, target_type, signed=False))
   386	            elif (
   387	                is_bool_rprimitive(src_type) or is_bit_rprimitive(src_type)
   388	            ) and is_fixed_width_rtype(target_type):
   389	                return self.add(Extend(src, target_type, signed=False))
   390	            elif isinstance(src, Integer) and is_float_rprimitive(target_type):
   391	                if is_tagged(src_type):
   392	                    return Float(float(src.value // 2))
   393	                return Float(float(src.value))
   394	            elif is_tagged(src_type) and is_float_rprimitive(target_type):
   395	                return self.int_to_float(src, line)
   396	            elif (
   397	                isinstance(src_type, RTuple)
   398	                and isinstance(target_type, RTuple)
   399	                and len(src_type.types) == len(target_type.types)
   400	            ):
   401	                # Coerce between two tuple types by coercing each item separately
   402	                values = []
   403	                for i in range(len(src_type.types)):
   404	                    v = None
   405	                    if isinstance(src, TupleSet):
   406	                        item = src.items[i]
   407	                        # We can't reuse register values, since they can be modified.
   408	                        if not isinstance(item, Register):
   409	                            v = item
   410	                    if v is None:
   411	                        v = TupleGet(src, i)
   412	                        self.add(v)
   413	                    values.append(v)
   414	                return self.add(
   415	                    TupleSet(
   416	                        [self.coerce(v, t, line) for v, t in zip(values, target_type.types)], line
   417	                    )
   418	                )
   419	            # To go between any other unboxed types, we go through a boxed
   420	            # in-between value, for simplicity.
   421	            tmp = self.box(src)
   422	            return self.unbox_or_cast(tmp, target_type, line)
   423	        if (not src_type.is_unboxed and target_type.is_unboxed) or not is_subtype(
   424	            src_type, target_type
   425	        ):
   426	            return self.unbox_or_cast(src, target_type, line, can_borrow=can_borrow)
   427	        elif force:
   428	            tmp = Register(target_type)
   429	            self.add(Assign(tmp, src))
   430	            return tmp
   431	        return src

   433	    def coerce_int_to_fixed_width(self, src: Value, target_type: RType, line: int) -> Value:
   434	        assert is_fixed_width_rtype(target_type), target_type
   435	        assert isinstance(target_type, RPrimitive)
   436
   437	        res = Register(target_type)
   438
   439	        fast, slow, end = BasicBlock(), BasicBlock(), BasicBlock()
   440
   441	        check = self.check_tagged_short_int(src, line)
   442	        self.add(Branch(check, fast, slow, Branch.BOOL))
   443
   444	        self.activate_block(fast)
   445
   446	        size = target_type.size
   447	        if size < int_rprimitive.size:
   448	            # Add a range check when the target type is smaller than the source type
   449	            fast2, fast3 = BasicBlock(), BasicBlock()
   450	            upper_bound = 1 << (size * 8 - 1)
   451	            if not target_type.is_signed:
   452	                upper_bound *= 2
   453	            check2 = self.add(ComparisonOp(src, Integer(upper_bound, src.type), ComparisonOp.SLT))
   454	            self.add(Branch(check2, fast2, slow, Branch.BOOL))
   455	            self.activate_block(fast2)
   456	            if target_type.is_signed:
   457	                lower_bound = -upper_bound
   458	            else:
   459	                lower_bound = 0
   460	            check3 = self.add(ComparisonOp(src, Integer(lower_bound, src.type), ComparisonOp.SGE))
   461	            self.add(Branch(check3, fast3, slow, Branch.BOOL))
   462	            self.activate_block(fast3)
   463	            tmp = self.int_op(
   464	                c_pyssize_t_rprimitive,
   465	                src,
   466	                Integer(1, c_pyssize_t_rprimitive),
   467	                IntOp.RIGHT_SHIFT,
   468	                line,
   469	            )
   470	            tmp = self.add(Truncate(tmp, target_type))
   471	        else:
   472	            if size > int_rprimitive.size:
   473	                tmp = self.add(Extend(src, target_type, signed=True))
   474	            else:
   475	                tmp = src
   476	            tmp = self.int_op(target_type, tmp, Integer(1, target_type), IntOp.RIGHT_SHIFT, line)
   477
   478	        self.add(Assign(res, tmp))
   479	        self.goto(end)
   480
   481	        self.activate_block(slow)
   482	        if is_int64_rprimitive(target_type) or (
   483	            is_int32_rprimitive(target_type) and size == int_rprimitive.size
   484	        ):
   485	            # Slow path calls a library function that handles more complex logic
   486	            ptr = self.int_op(
   487	                pointer_rprimitive, src, Integer(1, pointer_rprimitive), IntOp.XOR, line
   488	            )
   489	            ptr2 = Register(c_pointer_rprimitive)
   490	            self.add(Assign(ptr2, ptr))
   491	            if is_int64_rprimitive(target_type):
   492	                conv_op = int_to_int64_op
   493	            else:
   494	                conv_op = int_to_int32_op
   495	            tmp = self.call_c(conv_op, [ptr2], line)
   496	            self.add(Assign(res, tmp))
   497	            self.add(KeepAlive([src]))
   498	            self.goto(end)
   499	        elif is_int32_rprimitive(target_type):
   500	            # Slow path just always generates an OverflowError
   501	            self.call_c(int32_overflow, [], line)
   502	            self.add(Unreachable())
   503	        elif is_int16_rprimitive(target_type):
   504	            # Slow path just always generates an OverflowError
   505	            self.call_c(int16_overflow, [], line)
   506	            self.add(Unreachable())
   507	        elif is_uint8_rprimitive(target_type):
   508	            # Slow path just always generates an OverflowError
   509	            self.call_c(uint8_overflow, [], line)
   510	            self.add(Unreachable())
   511	        else:
   512	            assert False, target_type
   513
   514	        self.activate_block(end)
   515	        return res
   516
   517	    def coerce_short_int_to_fixed_width(self, src: Value, target_type: RType, line: int) -> Value:
   518	        if is_int64_rprimitive(target_type) or (
   519	            PLATFORM_SIZE == 4 and is_int32_rprimitive(target_type)
   520	        ):
   521	            return self.int_op(target_type, src, Integer(1, target_type), IntOp.RIGHT_SHIFT, line)
   522	        # TODO: i32 on 64-bit platform
   523	        assert False, (src.type, target_type, PLATFORM_SIZE)

   525	    def coerce_fixed_width_to_int(self, src: Value, line: int) -> Value:
   526	        if (
   527	            (is_int32_rprimitive(src.type) and PLATFORM_SIZE == 8)
   528	            or is_int16_rprimitive(src.type)
   529	            or is_uint8_rprimitive(src.type)
   530	        ):
   531	            # Simple case -- just sign extend and shift.
   532	            extended = self.add(Extend(src, c_pyssize_t_rprimitive, signed=src.type.is_signed))
   533	            return self.int_op(
   534	                int_rprimitive,
   535	                extended,
   536	                Integer(1, c_pyssize_t_rprimitive),
   537	                IntOp.LEFT_SHIFT,
   538	                line,
   539	            )
   540
   541	        assert is_fixed_width_rtype(src.type)
   542	        assert isinstance(src.type, RPrimitive)
   543	        src_type = src.type
   544
   545	        res = Register(int_rprimitive)
   546
   547	        fast, fast2, slow, end = BasicBlock(), BasicBlock(), BasicBlock(), BasicBlock()
   548
   549	        c1 = self.add(ComparisonOp(src, Integer(MAX_SHORT_INT, src_type), ComparisonOp.SLE))
   550	        self.add(Branch(c1, fast, slow, Branch.BOOL))
   551
   552	        self.activate_block(fast)
   553	        c2 = self.add(ComparisonOp(src, Integer(MIN_SHORT_INT, src_type), ComparisonOp.SGE))
   554	        self.add(Branch(c2, fast2, slow, Branch.BOOL))
   555
   556	        self.activate_block(slow)
   557	        if is_int64_rprimitive(src_type):
   558	            conv_op = int64_to_int_op
   559	        elif is_int32_rprimitive(src_type):
   560	            assert PLATFORM_SIZE == 4
   561	            conv_op = ssize_t_to_int_op
   562	        else:
   563	            assert False, src_type
   564	        x = self.call_c(conv_op, [src], line)
   565	        self.add(Assign(res, x))
   566	        self.goto(end)
   567
   568	        self.activate_block(fast2)
   569	        if int_rprimitive.size < src_type.size:
   570	            tmp = self.add(Truncate(src, c_pyssize_t_rprimitive))
   571	        else:
   572	            tmp = src
   573	        s = self.int_op(int_rprimitive, tmp, Integer(1, tmp.type), IntOp.LEFT_SHIFT, line)
   574	        self.add(Assign(res, s))
   575	        self.goto(end)
   576
   577	        self.activate_block(end)
   578	        return res
   579
   580	    def coerce_nullable(self, src: Value, target_type: RType, line: int) -> Value:
   581	        """Generate a coercion from a potentially null value."""
   582	        if src.type.is_unboxed == target_type.is_unboxed and (
   583	            (target_type.is_unboxed and is_runtime_subtype(src.type, target_type))
   584	            or (not target_type.is_unboxed and is_subtype(src.type, target_type))
   585	        ):
   586	            return src
   587
   588	        target = Register(target_type)
   589
   590	        valid, invalid, out = BasicBlock(), BasicBlock(), BasicBlock()
   591	        self.add(Branch(src, invalid, valid, Branch.IS_ERROR))
   592
   593	        self.activate_block(valid)
   594	        coerced = self.coerce(src, target_type, line)
   595	        self.add(Assign(target, coerced, line))
   596	        self.goto(out)
   597
   598	        self.activate_block(invalid)
   599	        error = self.add(LoadErrorValue(target_type))
   600	        self.add(Assign(target, error, line))
   601
   602	        self.goto_and_activate(out)
   603	        return target

   648	        """Fast path for isinstance() that checks against a list of native classes."""
   649	        if not class_irs:
   650	            return self.false()
   651	        ret = self.isinstance_native(obj, class_irs[0], line)
   652	        for class_ir in class_irs[1:]:
   653
   654	            def other() -> Value:
   655	                return self.isinstance_native(obj, class_ir, line)
   656
   657	            ret = self.shortcircuit_helper("or", bool_rprimitive, lambda: ret, other, line)
   658	        return ret

   670	    def isinstance_native(self, obj: Value, class_ir: ClassIR, line: int) -> Value:
   671	        """Fast isinstance() check for a native class.
   672
   673	        If there are three or fewer concrete (non-trait) classes among the class
   674	        and all its children, use even faster type comparison checks `type(obj)
   675	        is typ`.
   676	        """
   677	        concrete = all_concrete_classes(class_ir)
   678	        if concrete is None or len(concrete) > FAST_ISINSTANCE_MAX_SUBCLASSES + 1:
   679	            return self.primitive_op(
   680	                fast_isinstance_op, [obj, self.get_native_type(class_ir)], line
   681	            )
   682	        if not concrete:
   683	            # There can't be any concrete instance that matches this.
   684	            return self.false()
   685	        type_obj = self.get_native_type(concrete[0])
   686	        ret = self.type_is_op(obj, type_obj, line)
   687	        for c in concrete[1:]:
   688
   689	            def other() -> Value:
   690	                return self.type_is_op(obj, self.get_native_type(c), line)
   691
   692	            ret = self.shortcircuit_helper("or", bool_rprimitive, lambda: ret, other, line)
   693	        return ret

   706
   707	        This is pretty complicated, and almost all of the complication here stems from
   708	        one of two things (but mostly the second):
   709	          * The handling of ARG_STAR/ARG_STAR2. We want to create as much of the args/kwargs
   710	            values in one go as we can, so we collect values until our hand is forced, and
   711	            then we emit creation of the list/tuple, and expand it from there if needed.
   712
   713	          * Support potentially nullable argument values. This has very narrow applicability,
   714	            as this will never be done by our compiled Python code, but is critically used
   715	            by gen_glue_method when generating glue methods to mediate between the function
   716	            signature of a parent class and its subclasses.
   717
   718	            For named-only arguments, this is quite simple: if it is
   719	            null, don't put it in the dict.
   720
   721	            For positional-or-named arguments, things are much more complicated.
   722	              * First, anything that was passed as a positional arg
   723	                must be forwarded along as a positional arg. It *must
   724	                not* be converted to a named arg. This is because mypy
   725	                does not enforce that positional-or-named arguments
   726	                have the same name in subclasses, and it is not
   727	                uncommon for code to have different names in
   728	                subclasses (a bunch of mypy's visitors do this, for
   729	                example!). This is arguably a bug in both mypy and code doing
   730	                this, and they ought to be using positional-only arguments, but
   731	                positional-only arguments are new and ugly.
   732
   733	              * On the flip side, we're willing to accept the
   734	                infelicity of sometimes turning an argument that was
   735	                passed by keyword into a positional argument. It's wrong,
   736	                but it's very marginal, and avoiding it would require passing
   737	                a bitmask of which arguments were named with every function call,
   738	                or something similar.
   739	                (See some discussion of this in testComplicatedArgs)
   740
   741	            Thus, our strategy for positional-or-named arguments is to
   742	            always pass them as positional, except in the one
   743	            situation where we can not, and where we can be absolutely
   744	            sure they were passed by name: when an *earlier*
   745	            positional argument was missing its value.

   750	              * x missing, y present: args=(),     kwargs={'y': y}
