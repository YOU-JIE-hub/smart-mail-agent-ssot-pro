   187	class TypedPropertyCollection(PropHolder):
   188	    """A container with properties for objects of the same kind.
   189
   190	    This makes it easy to create lists of objects. You initialize
   191	    it with a class of what it is to contain, and that is all you
   192	    can add to it.  You can assign properties to the collection
   193	    as a whole, or to a numeric index within it; if so it creates
   194	    a new child object to hold that data.
   195
   196	    So:
   197	        wedges = TypedPropertyCollection(WedgeProperties)
   198	        wedges.strokeWidth = 2                # applies to all
   199	        wedges.strokeColor = colors.red       # applies to all
   200	        wedges[3].strokeColor = colors.blue   # only to one
   201
   202	    The last line should be taken as a prescription of how to
   203	    create wedge no. 3 if one is needed; no error is raised if
   204	    there are only two data points.
   205
   206	    We try and make sensible use of tuple indices.
   207	        line[(3,x)] is backed by line[(3,)] == line[3] & line
   208	    """
   209
   210	    def __init__(self, exampleClass, **kwds):
   211	        #give it same validation rules as what it holds
   212	        self.__dict__['_value'] = exampleClass(**kwds)
   213	        self.__dict__['_children'] = {}
   214
   215	    def wKlassFactory(self,Klass):
   216	        class WKlass(Klass,CloneMixin):
   217	            def __getattr__(self,name):
   218	                try:
   219	                    return self.__class__.__bases__[0].__getattr__(self,name)
   220	                except:
   221	                    parent = self.parent
   222	                    c = parent._children
   223	                    x = self.__propholder_index__
   224	                    while x:
   225	                        if x in c:
   226	                            return getattr(c[x],name)
   227	                        x = x[:-1]
   228	                    return getattr(parent,name)
   229	            @property
   230	            def parent(self):
   231	                return self.__propholder_parent__()
   232	        return WKlass
   233
   234	    def __getitem__(self, x):
   235	        x = tuple(x) if isinstance(x,(tuple,list)) else (x,)
   236	        try:
   237	            return self._children[x]
   238	        except KeyError:
   239	            Klass = self._value.__class__
   240	            if Klass in _ItemWrapper:
   241	                WKlass = _ItemWrapper[Klass]
   242	            else:
   243	                _ItemWrapper[Klass] = WKlass = self.wKlassFactory(Klass)
   244
   245	            child = WKlass()
   246
   247	            for i in filter(lambda x,K=list(child.__dict__.keys()): x in K,list(child._attrMap.keys())):
   248	                del child.__dict__[i]
   249	            child.__dict__.update(dict(
   250	                                    __propholder_parent__ = weakref_ref(self),
   251	                                    __propholder_index__ = x[:-1])
   252	                                    )
   253
   254	            self._children[x] = child
   255	            return child
   256
   257	    def __contains__(self,key):
   258	        return (tuple(key) if isinstance(key,(tuple,list)) else (key,)) in self._children
   259
   260	    def __setitem__(self, key, value):
   261	        assert isinstance(value, self._value.__class__), (
   262	            "This collection can only hold objects of type %s" % self._value.__class__.__name__)
   263
   264	    def __len__(self):
   265	        return len(list(self._children.keys()))
   266
   267	    def getProperties(self,recur=1):
   268	        # return any children which are defined and whatever
   269	        # differs from the parent
   270	        props = {}
   271
   272	        for key, value in self._value.getProperties(recur=recur).items():
   273	            props['%s' % key] = value
   274
   275	        for idx in self._children.keys():
   276	            childProps = self._children[idx].getProperties(recur=recur)
   277	            for key, value in childProps.items():
   278	                if not hasattr(self,key) or getattr(self, key)!=value:
   279	                    newKey = '[%s].%s' % (idx if len(idx)>1 else idx[0], key)
   280	                    props[newKey] = value
   281	        return props
   282
   283	    def setVector(self,**kw):
   284	        for name, value in kw.items():
   285	            for i, v in enumerate(value):
   286	                setattr(self[i],name,v)
   287
   288	    def __getattr__(self,name):
   289	        return getattr(self._value,name)
   290
   291	    def __setattr__(self,name,value):
   292	        return setattr(self._value,name,value)
   293
   294	    def checkAttr(self, key, a, default=None):
   295	        return getattr(self[key], a, default) if key in self else default

    75	class Star(_Symbol):
    76	    """This draws a 5-pointed star.
    77
    78	        possible attributes:
    79	        'x', 'y', 'size', 'fillColor', 'strokeColor'
    80
    81	        """
    82	    _attrMap = AttrMap(BASE=_Symbol,
    83	            angle = AttrMapValue(isNumber, desc='angle in degrees'),
    84	            )
    85	    _size = 100.
    86
    87	    def __init__(self):
    88	        _Symbol.__init__(self)
    89	        self.size = 100
    90	        self.fillColor = colors.yellow
    91	        self.strokeColor = None
    92	        self.angle = 0
    93
    94	    def demo(self):
    95	        D = Drawing(200, 100)
    96	        et = Star()
    97	        et.x=50
    98	        et.y=0
    99	        D.add(et)
   100	        labelFontSize = 10
   101	        D.add(String(et.x+(et.size/2.0),(et.y-(1.2*labelFontSize)),
   102	                            et.__class__.__name__, fillColor=colors.black, textAnchor='middle',
   103	                            fontSize=labelFontSize))
   104	        return D
   105
   106	    def draw(self):
   107	        s = float(self.size)  #abbreviate as we will use this a lot
   108	        g = Group()
   109
   110	        # new algorithm from markers.StarFive
   111	        R = float(self.size)/2
   112	        r = R*sin(18*(pi/180.0))/cos(36*(pi/180.0))
   113	        P = []
   114	        angle = 90
   115	        for i in range(5):
   116	            for radius in R, r:
   117	                theta = angle*(pi/180.0)
   118	                P.append(radius*cos(theta))
   119	                P.append(radius*sin(theta))
   120	                angle = angle + 36
   121	        # star specific bits
   122	        star = Polygon(P,
   123	                    fillColor = self.fillColor,
   124	                    strokeColor = self.strokeColor,
   125	                    strokeWidth=s/50)
   126	        g.rotate(self.angle)
   127	        g.shift(self.x+self.dx,self.y+self.dy)
   128	        g.add(star)
   129
   130	        return g

   132	class Flag(_Symbol):
   133	    """This is a generic flag class that all the flags in this file use as a basis.
   134
   135	        This class basically provides edges and a tidy-up routine to hide any bits of
   136	        line that overlap the 'outside' of the flag
   137
   138	        possible attributes:
   139	        'x', 'y', 'size', 'fillColor'
   140	    """

   142	    _attrMap = AttrMap(BASE=_Symbol,
   143	            fillColor = AttrMapValue(isColor, desc='Background color'),
   144	            border = AttrMapValue(isBoolean, 'Whether a background is drawn'),
   145	            kind = AttrMapValue(validFlag, desc='Which flag'),
   146	            )

   148	    _cache = {}

   150	    def __init__(self,**kw):
   151	        _Symbol.__init__(self)
   152	        self.kind = None
   153	        self.size = 100

   155	        self.border=1
   156	        self.setProperties(kw)

   168	    def _borderDraw(self,f):
   169	        s = self.size  # abbreviate as we will use this a lot
   170	        g = Group()
   171	        g.add(f)
   172	        x, y, sW = self.x+self.dx, self.y+self.dy, self.strokeWidth/2.
   173	        g.insert(0,Rect(-sW, -sW, width=getattr(self,'_width',2*s)+3*sW, height=getattr(self,'_height',s)+2*sW,
   174	                fillColor = None, strokeColor = self.strokeColor, strokeWidth=sW*2))
   175	        g.shift(x,y)
   176	        g.scale(s/_size, s/_size)
   177	        return g

   184	            self._cache[kind] = f._explode()
   185	        return self._borderDraw(f)

    12	def frange(start, end=None, inc=None):
    13	    "A range function, that does accept float increments..."
    14
    15	    if end == None:
    16	        end = start + 0.0
    17	        start = 0.0
    18
    19	    if inc == None:
    20	        inc = 1.0
    21
    22	    L = []
    23	    end = end - inc*0.0001  #to avoid numrical problems
    24	    while 1:
    25	        next = start + len(L) * inc
    26	        if inc > 0 and next >= end:
    27	            break
    28	        elif inc < 0 and next <= end:
    29	            break
    30	        L.append(next)
    31
    32	    return L

   126	    def makeLinePosList(self, start, isX=0):
   127	        "Returns a list of positions where to place lines."
   128
   129	        w, h = self.width, self.height
   130	        if isX:
   131	            length = w

   133	            length = h

   141	                r.append(r[-1] + self.deltaSteps[i % len(self.deltaSteps)])

   425	def centroid(P):
   426	    '''compute average point of a set of points'''
   427	    cx = 0
   428	    cy = 0
   429	    for x,y in P:
   430	        cx+=x
   431	        cy+=y
   432	    n = float(len(P))
   433	    return cx/n, cy/n

   435	def rotatedEnclosingRect(P, angle, rect):
   436	    '''
   437	    given P a sequence P of x,y coordinate pairs and an angle in degrees
   438	    find the centroid of P and the axis at angle theta through it
   439	    find the extreme points of P wrt axis parallel distance and axis
   440	    orthogonal distance. Then compute the least rectangle that will still
   441	    enclose P when rotated by angle.
   442
   443	    The class R
   444	    '''
   445	    from math import pi, cos, sin
   446	    x0, y0 = centroid(P)
   447	    theta = (angle/180.)*pi
   448	    s,c=sin(theta),cos(theta)
   449	    def parallelAxisDist(xy,s=s,c=c,x0=x0,y0=y0):
   450	        x,y = xy
   451	        return (s*(y-y0)+c*(x-x0))
   452	    def orthogonalAxisDist(xy,s=s,c=c,x0=x0,y0=y0):
   453	        x,y = xy
   454	        return (c*(y-y0)+s*(x-x0))
   455	    L = list(map(parallelAxisDist,P))
   456	    L.sort()
   457	    a0, a1 = L[0], L[-1]
   458	    L = list(map(orthogonalAxisDist,P))
   459	    L.sort()
   460	    b0, b1 = L[0], L[-1]
   461	    rect.x, rect.width = a0, a1-a0
   462	    rect.y, rect.height = b0, b1-b0
   463	    g = Group(transform=(c,s,-s,c,x0,y0))
   464	    g.add(rect)
   465	    return g
