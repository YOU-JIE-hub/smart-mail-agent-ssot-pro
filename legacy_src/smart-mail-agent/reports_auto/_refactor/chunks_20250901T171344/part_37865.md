  1927	    * Escaping is optional.
  1928	    * Raises UsageError so we get nice error messages on failure.
  1929	    """

  1954	            For more information please consult: {doc_url}

  1998	    """
  1999	    Copied from warnings._getcategory, but changed so it lets exceptions (specially ImportErrors)
  2000	    propagate so we can get access to their tracebacks (#9218).
  2001	    """
  2002	    __tracebackhide__ = True
  2003	    if not category:
  2004	        return Warning
  2005
  2006	    if "." not in category:
  2007	        import builtins as m
  2008
  2009	        klass = category
  2010	    else:
  2011	        module, _, klass = category.rpartition(".")
  2012	        m = __import__(module, None, None, [klass])
  2013	    cat = getattr(m, klass)
  2014	    if not issubclass(cat, Warning):
  2015	        raise UsageError(f"{cat} is not a Warning subclass")
  2016	    return cast(type[Warning], cat)
