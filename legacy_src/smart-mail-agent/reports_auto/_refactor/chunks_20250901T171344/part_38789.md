   540	def encryptPdfInMemory(inputPDF,
   541	                  userPassword, ownerPassword=None,
   542	                  canPrint=1, canModify=1, canCopy=1, canAnnotate=1,
   543	                       strength=40):
   544	    """accepts a PDF file 'as a byte array in memory'; return encrypted one.

   546	    This is a high level convenience and does not touch the hard disk in any way.

   557	    names = list(bboxInfo.keys())

   559	    firstPageSize = bboxInfo['PageForms0'][2:]

   561	    #now make a new PDF document
