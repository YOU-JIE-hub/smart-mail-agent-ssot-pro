  1089	def definePath(pathSegs=[],isClipPath=0, dx=0, dy=0, **kw):
  1090	    O = []
  1091	    P = []
  1092	    for seg in pathSegs:
  1093	        if not isSeq(seg):
  1094	            opName = seg
  1095	            args = []
  1096	        else:
  1097	            opName = seg[0]
  1098	            args = seg[1:]
  1099	        if opName not in _PATH_OP_NAMES:
  1100	            raise ValueError('bad operator name %s' % opName)
  1101	        op = _PATH_OP_NAMES.index(opName)
  1102	        if len(args)!=_PATH_OP_ARG_COUNT[op]:
  1103	            raise ValueError('%s bad arguments %s' % (opName,str(args)))
  1104	        O.append(op)
  1105	        P.extend(list(args))
  1106	    for d,o in (dx,0), (dy,1):
  1107	        for i in range(o,len(P),2):
  1108	            P[i] = P[i]+d
  1109
  1110	    #if there's a bounding box given we constrain so our points lie in it
  1111	    #partial bbox is allowed and does something sensible
  1112	    bbox = kw.pop('bbox',None)
  1113	    if bbox:
  1114	        for j in 0,1:
  1115	            d = bbox[j],bbox[j+2]
  1116	            if d[0] is None and d[1] is None: continue
  1117	            a = P[j::2]
  1118	            a, b = min(a), max(a)
  1119	            if d[0] is not None and d[1] is not None:
  1120	                c, d = min(d), max(d)
  1121	                fac = (b-a)
  1122	                if abs(fac)>=1e-6:
  1123	                    fac = (d-c)/fac
  1124	                    for i in range(j,len(P),2):
  1125	                        P[i] = c + fac*(P[i]-a)
  1126	                else:
  1127	                    #there's no range in the bbox so fixed as average
  1128	                    for i in range(j,len(P),2):
  1129	                        P[i] = (c + d)*0.5
  1130	            else:
  1131	                #if   there's a  lower bound shift so min is lower bound
  1132	                #else there's an upper bound shift so max is upper bound
  1133	                c = d[0] - a if d[0] is not None else d[1] - b
  1134	                for i in range(j,len(P),2):
  1135	                    P[i] += c
  1136
  1137	    return Path(P,O,isClipPath,**kw)
