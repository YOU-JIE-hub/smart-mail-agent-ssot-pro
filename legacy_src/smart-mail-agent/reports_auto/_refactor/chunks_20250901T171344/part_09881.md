s':
    69	            inMetrics = 0
    70	        elif inMetrics:
    71	            chunks = line.split(';')
    72	            chunks = [chunk.strip() for chunk in chunks]
    73	            cidChunk, widthChunk, nameChunk = chunks[0:3]
    74
    75	            # character ID
    76	            l, r = cidChunk.split()
    77	            assert l == 'C', 'bad line in font file %s' % line
    78	            cid = int(r)
    79
    80	            # width
    81	            l, r = widthChunk.split()
    82	            assert l == 'WX', 'bad line in font file %s' % line
    83	            try:
    84	                width = int(r)
    85	            except ValueError:
    86	                width = float(r)
    87
    88	            # name
    89	            l, r = nameChunk.split()
    90	            assert l == 'N', 'bad line in font file %s' % line
    91	            name = r
    92
    93	            glyphLevel.append((cid, width, name))
    94
    95	    # pass 2 font info
    96	    inHeader = 0
    97	    for line in lines:
    98	        if line[0:16] == 'StartFontMetrics':
    99	            inHeader = 1
   100	        if line[0:16] == 'StartCharMetrics':
   101	            inHeader = 0
   102	        elif inHeader:
   103	            if line[0:7] == 'Comment': pass
   104	            try:
   105	                left, right = line.split(' ',1)
   106	            except:
   107	                raise ValueError("Header information error in afm %s: line='%s'" % (afmFileName, line))
   108	            try:
   109	                right = int(right)
   110	            except:
   111	                pass
   112	            topLevel[left] = right
   113
   114
   115	    return (topLevel, glyphLevel)

   270	    def getDifferences(self, otherEnc):
   271	        """
   272	        Return a compact list of the code points differing between two encodings
   273
   274	        This is in the Adobe format: list of
   275	           [[b1, name1, name2, name3],
   276	           [b2, name4]]
   277
   278	        where b1...bn is the starting code point, and the glyph names following
   279	        are assigned consecutive code points.
   280
   281	        """
   282
   283	        ranges = []
   284	        curRange = None
   285	        for i in range(len(self.vector)):
   286	            glyph = self.vector[i]
   287	            if glyph==otherEnc.vector[i]:
   288	                if curRange:
   289	                    ranges.append(curRange)
   290	                    curRange = []
   291	            else:
   292	                if curRange:
   293	                    curRange.append(glyph)
   294	                elif glyph:
   295	                    curRange = [i, glyph]
   296	        if curRange:
   297	            ranges.append(curRange)
   298	        return ranges

   301	        "Returns a PDF Object representing self"
   302	        # avoid circular imports - this cannot go at module level
   303	        from reportlab.pdfbase import pdfdoc

   305	        D = {}
   306	        baseEncodingName = self.baseEncodingName
