   555	    def __new__(cls, u, X=None):
   556	        if not X:
   557	            u = u.split(_shy)
   558	            X = []
   559	            a = X.append

   561	            for s in u:

   563	                a(x)
   564	            u = u''.join(u)

   570	def _shyUnsplit(s,ss=None):
   571	    '''rejoin two parts of an original _SHYIndexedStr or str that was split'''
   572	    u = s.rstrip(u'-')
   573	    if isinstance(s,_SHYIndexedStr):
   574	        X = s._shyIndices[:]
   575	        x = X[-1]
   576	        if ss:
   577	            if hasattr(ss,'_shyIndices'):
   578	                X.extend([_+x for _ in ss._shyIndices])
   579	            u += ss
   580	        return _SHYIndexedStr(u,X)
   581	    elif ss:
   582	        u += ss
   583	        if hasattr(ss,'_shyIndices'):
   584	            X.extend([_+x for _ in ss._shyIndices])
   585	            return _SHYIndexedStr(u,X)
   586	    return u

   588	class _SHYWord(list):
   589	    '''a fragword containing soft hyphens some of its strings are _SHYIndexedStr'''
   590	    def shyphenate(self, newWidth, maxWidth):
   591	        ww = self[0]
   592	        self._fsww = 0x7fffffff
   593	        if ww==0: return []
   594	        possible = None
   595	        exceeded = False
   596	        baseWidth = baseWidth0 = newWidth - ww
   597	        fsww = None
   598	        for i,(f,t) in enumerate(self[1:]):
   599	            sW = lambda s: stringWidth(s, f.fontName, f.fontSize)
   600	            if isinstance(t,_SHYIndexedStr):
   601	                # there's a shy in this bit
   602	                shyLen = sW(u'-')
   603	                bw = baseWidth + shyLen
   604	                for j, x in enumerate(t._shyIndices):
   605	                    left, right = t[:x], t[x:]
   606	                    leftw = bw+sW(left)
   607	                    if fsww is None: fsww = leftw
   608	                    exceeded = leftw > maxWidth
   609	                    if exceeded: break
   610	                    possible = i, j, x, leftw, left, right, shyLen
   611	                baseWidth += sW(t)
   612	            else:
   613	                baseWidth += sW(t)
   614	                exceeded = baseWidth > maxWidth
   615	            if exceeded and fsww is not None: break
   616	        self._fsww = fsww-baseWidth0 if fsww is not None else 0x7fffffff
   617	        if not possible: return []
   618	        i, j, x, leftw, left, right, shyLen = possible
   619	        i1 = i+1
   620	        f, t = self[i1] #we're splitting this subfrag
   621	        X = t._shyIndices
   622	        lefts = _SHYIndexedStr(left+u'-',X[:j+1])
   623	        L = self[:i1] + [(f,lefts)]
   624	        L[0] = leftw - baseWidth0
   625	        R = [ww-L[0]+shyLen]+([] if not right else [(f,_SHYIndexedStr(right,[_-x for _ in X[j+1:]]))]) + self[i1+1:]
   626	        return _SplitFragSHY(L), _SHYWordHS(R)

   862	def _fragWordIter(w):
   863	    for f, s in w[1:]:
   864	        if hasattr(f,'cbDefn'):
   865	            yield f, getattr(f.cbDefn,'width',0), s
   866	        elif s:
   867	            if isBytes(s):
   868	                s = s.decode('utf8')    #only encoding allowed
   869	            for c in s:
   870	                yield f, stringWidth(c,f.fontName, f.fontSize), c
   871	        else:
   872	            yield f, 0, s

   874	def _splitFragWord(w,maxWidth,maxWidths,lineno):
   875	    '''given a frag word, w, as returned by getFragWords
   876	    split it into frag words that fit in lines of length
   877	    maxWidth
   878	    maxWidths[lineno+1]
   879	    .....
   880	    maxWidths[lineno+n]
   881
   882	    return the new word list which is either
   883	    _SplitFrag....._SPlitFrag or
   884	    _SplitFrag....._SplitFragHS if the word is hanging space.
   885	    '''
   886	    R = []
   887	    maxlineno = len(maxWidths)-1
   888	    W = []
   889	    lineWidth = 0
   890	    fragText = u''
   891	    wordWidth = 0
   892	    f = w[1][0]
   893	    for g,cw,c in _fragWordIter(w):
   894	        newLineWidth = lineWidth+cw
   895	        tooLong = newLineWidth>maxWidth
   896	        if g is not f or tooLong:
   897	            f = f.clone()
   898	            if hasattr(f,'text'):
   899	                f.text = fragText
   900	            W.append((f,fragText))
   901	            if tooLong:
   902	                W = _SplitFrag([wordWidth]+W)
   903	                R.append(W)
   904	                lineno += 1
   905	                maxWidth = maxWidths[min(maxlineno,lineno)]
   906	                W = []
   907	                newLineWidth = cw
   908	                wordWidth = 0
   909	            fragText = u''
   910	            f = g
   911	        wordWidth += cw
   912	        fragText += c
   913	        lineWidth = newLineWidth
   914	    W.append((f,fragText))
   915	    W = (_SplitFragHS if isinstance(w,_HSFrag) else _SplitFragH)([wordWidth]+W)
   916
   917	    R.append(W)
   918	    return R
