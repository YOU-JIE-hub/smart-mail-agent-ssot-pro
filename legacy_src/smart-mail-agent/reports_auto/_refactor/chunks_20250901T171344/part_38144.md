asses have been read, because subclasses can override parents.
   484	    last_default = False
   485
   486	    for i, attribute in enumerate(attributes):
   487	        if not attribute.init:
   488	            continue
   489
   490	        if attribute.kw_only:
   491	            # Keyword-only attributes don't care whether they are default or not.
   492	            continue
   493
   494	        # If the issue comes from merging different classes, report it
   495	        # at the class definition point.
   496	        context = attribute.context if i >= len(super_attrs) else ctx.cls
   497
   498	        if not attribute.has_default and last_default:
   499	            ctx.api.fail("Non-default attributes not allowed after default attributes.", context)
   500	        last_default |= attribute.has_default
   501
   502	    return attributes

   510	def _detect_auto_attribs(ctx: mypy.plugin.ClassDefContext) -> bool:
   511	    """Return whether auto_attribs should be enabled or disabled.
   512
   513	    It's disabled if there are any unannotated attribs()
   514	    """
   515	    for stmt in ctx.cls.defs.body:
   516	        if isinstance(stmt, AssignmentStmt):
   517	            for lvalue in stmt.lvalues:
   518	                lvalues, rvalues = _parse_assignments(lvalue, stmt)
   519
   520	                if len(lvalues) != len(rvalues):
   521	                    # This means we have some assignment that isn't 1 to 1.
   522	                    # It can't be an attrib.
   523	                    continue
   524
   525	                for lhs, rvalue in zip(lvalues, rvalues):
   526	                    # Check if the right hand side is a call to an attribute maker.
   527	                    if (
   528	                        isinstance(rvalue, CallExpr)
   529	                        and isinstance(rvalue.callee, RefExpr)
   530	                        and rvalue.callee.fullname in attr_attrib_makers
   531	                        and not stmt.new_syntax
   532	                    ):
   533	                        # This means we have an attrib without an annotation and so
   534	                        # we can't do auto_attribs=True
   535	                        return False
   536	    return True

   539	def _attributes_from_assignment(
   540	    ctx: mypy.plugin.ClassDefContext, stmt: AssignmentStmt, auto_attribs: bool, kw_only: bool
   541	) -> Iterable[Attribute]:
   542	    """Return Attribute objects that are created by this assignment.
   543
   544	    The assignments can look like this:
   545	        x = attr.ib()
   546	        x = y = attr.ib()
   547	        x, y = attr.ib(), attr.ib()
   548	    or if auto_attribs is enabled also like this:
   549	        x: type
   550	        x: type = default_value
   551	        x: type = attr.ib(...)
   552	    """
   553	    for lvalue in stmt.lvalues:
   554	        lvalues, rvalues = _parse_assignments(lvalue, stmt)
   555
   556	        if len(lvalues) != len(rvalues):
   557	            # This means we have some assignment that isn't 1 to 1.
   558	            # It can't be an attrib.
   559	            continue
   560
   561	        for lhs, rvalue in zip(lvalues, rvalues):
   562	            # Check if the right hand side is a call to an attribute maker.
   563	            if (
   564	                isinstance(rvalue, CallExpr)
   565	                and isinstance(rvalue.callee, RefExpr)
   566	                and rvalue.callee.fullname in attr_attrib_makers
   567	            ):
   568	                attr = _attribute_from_attrib_maker(ctx, auto_attribs, kw_only, lhs, rvalue, stmt)
   569	                if attr:
   570	                    yield attr
   571	            elif auto_attribs and stmt.type and stmt.new_syntax and not is_class_var(lhs):
   572	                yield _attribute_from_auto_attrib(ctx, kw_only, lhs, rvalue, stmt)

   575	def _cleanup_decorator(stmt: Decorator, attr_map: dict[str, Attribute]) -> None:
   576	    """Handle decorators in class bodies.
   577
   578	    `x.default` will set a default value on x
   579	    `x.validator` and `x.default` will get removed to avoid throwing a type error.
   580	    """
   581	    remove_me = []
   582	    for func_decorator in stmt.decorators:
   583	        if (
   584	            isinstance(func_decorator, MemberExpr)
   585	            and isinstance(func_decorator.expr, NameExpr)
   586	            and func_decorator.expr.name in attr_map
   587	        ):
   588	            if func_decorator.name == "default":
   589	                attr_map[func_decorator.expr.name].has_default = True
   590
   591	            if func_decorator.name in ("default", "validator"):
   592	                # These are decorators on the attrib object that only exist during
   593	                # class creation time.  In order to not trigger a type error later we
   594	                # just remove them.  This might leave us with a Decorator with no
   595	                # decorators (Emperor's new clothes?)
   596	                # TODO: It would be nice to type-check these rather than remove them.
   597	                #       default should be Callable[[], T]
   598	                #       validator should be Callable[[Any, 'Attribute', T], Any]
   599	                #       where T is the type of the attribute.
   600	                remove_me.append(func_decorator)
   601	    for dec in remove_me:
   602	        stmt.decorators.remove(dec)

   605	def _attribute_from_auto_attrib(
   606	    ctx: mypy.plugin.ClassDefContext,
   607	    kw_only: bool,
   608	    lhs: NameExpr,
   609	    rvalue: Expression,
   610	    stmt: AssignmentStmt,
   611	) -> Attribute:
   612	    """Return an Attribute for a new type assignment."""
   613	    name = unmangle(lhs.name)
   614	    # `x: int` (without equal sign) assigns rvalue to TempNode(AnyType())
   615	    has_rhs = not isinstance(rvalue, TempNode)
   616	    sym = ctx.cls.info.names.get(name)
   617	    init_type = sym.type if sym else None
   618	    return Attribute(name, None, ctx.cls.info, has_rhs, True, kw_only, None, stmt, init_type)

   621	def _attribute_from_attrib_maker(
   622	    ctx: mypy.plugin.ClassDefContext,
   623	    auto_attribs: bool,
   624	    kw_only: bool,
   625	    lhs: NameExpr,
   626	    rvalue: CallExpr,
   627	    stmt: AssignmentStmt,
   628	) -> Attribute | None:
   629	    """Return an Attribute from the assignment or None if you can't make one."""
   630	    if auto_attribs and not stmt.new_syntax:
   631	        # auto_attribs requires an annotation on *every* attr.ib.
   632	        assert lhs.node is not None
   633	        ctx.api.msg.need_annotation_for_var(lhs.node, stmt)
   634	        return None
   635
   636	    if len(stmt.lvalues) > 1:
   637	        ctx.api.fail("Too many names for one attribute", stmt)
   638	        return None
   639
   640	    # This is the type that belongs in the __init__ method for this attrib.
   641	    init_type = stmt.type
   642
   643	    # Read all the arguments from the call.
   644	    init = _get_bool_argument(ctx, rvalue, "init", True)
   645	    # Note: If the class decorator says kw_only=True the attribute is ignored.
   646	    # See https://github.com/python-attrs/attrs/issues/481 for explanation.
   647	    kw_only |= _get_bool_argument(ctx, rvalue, "kw_only", False)
   648
   649	    # TODO: Check for attr.NOTHING
   650	    attr_has_default = bool(_get_argument(rvalue, "default"))
   651	    attr_has_factory = bool(_get_argument(rvalue, "factory"))
   652
   653	    if attr_has_default and attr_has_factory:
   654	        ctx.api.fail('Can\'t pass both "default" and "factory".', rvalue)
   655	    elif attr_has_factory:
   656	        attr_has_default = True
   657
   658	    # If the type isn't set through annotation but is passed through `type=` use that.
   659	    type_arg = _get_argument(rvalue, "type")
   660	    if type_arg and not init_type:
   661	        try:
   662	            un_type = expr_to_unanalyzed_type(type_arg, ctx.api.options, ctx.api.is_stub_file)
   663	        except TypeTranslationError:
   664	            ctx.api.fail("Invalid argument to type", type_arg)
   665	        else:
   666	            init_type = ctx.api.anal_type(un_type)
   667	            if init_type and isinstance(lhs.node, Var) and not lhs.node.type:
   668	                # If there is no annotation, add one.
   669	                lhs.node.type = init_type
   670	                lhs.is_inferred_def = False
   671
   672	    # Note: convert is deprecated but works the same as converter.
   673	    converter = _get_argument(rvalue, "converter")
   674	    convert = _get_argument(rvalue, "convert")
   675	    if convert and converter:
   676	        ctx.api.fail('Can\'t pass both "convert" and "converter".', rvalue)
   677	    elif convert:
   678	        ctx.api.fail("convert is deprecated, use converter", rvalue)
   679	        converter = convert
   680	    converter_info = _parse_converter(ctx, converter)
   681
   682	    # Custom alias might be defined:
   683	    alias = None
   684	    alias_expr = _get_argument(rvalue, "alias")
   685	    if alias_expr:
   686	        alias = ctx.api.parse_str_literal(alias_expr)
   687	        if alias is None:
   688	            ctx.api.fail(
   689	                '"alias" argument to attrs field must be a string literal',
   690	                rvalue,
   691	                code=LITERAL_REQ,
   692	            )
   693	    name = unmangle(lhs.name)
   694	    return Attribute(
   695	        name, alias, ctx.cls.info, attr_has_default, init, kw_only, converter_info, stmt, init_type
   696	    )

   699	def _parse_converter(
   700	    ctx: mypy.plugin.ClassDefContext, converter_expr: Expression | None
   701	) -> Converter | None:
   702	    """Return the Converter object from an Expression."""
   703	    # TODO: Support complex converters, e.g. lambdas, calls, etc.
   704	    if not converter_expr:
   705	        return None
   706	    converter_info = Converter()
   707	    if (
   708	        isinstance(converter_expr, CallExpr)
   709	        and isinstance(converter_expr.callee, RefExpr)
   710	        and converter_expr.callee.fullname in attr_optional_converters
   711	        and converter_expr.args
   712	        and converter_expr.args[0]
   713	    ):
   714	        # Special handling for attr.converters.optional(type)
   715	        # We extract the type and add make the init_args Optional in Attribute.argument
   716	        converter_expr = converter_expr.args[0]
   717	        is_attr_converters_optional = True
   718	    else:
   719	        is_attr_converters_optional = False
   720
   721	    converter_type: Type | None = None
   722	    if isinstance(converter_expr, RefExpr) and converter_expr.node:
   723	        if isinstance(converter_expr.node, FuncDef):
   724	            if converter_expr.node.type and isinstance(converter_expr.node.type, FunctionLike):
   725	                converter_type = converter_expr.node.type
   726	            else:  # The converter is an unannotated function.
   727	                converter_info.init_type = AnyType(TypeOfAny.unannotated)
   728	                return converter_info
   729	        elif isinstance(converter_expr.node, OverloadedFuncDef) and is_valid_overloaded_converter(
   730	            converter_expr.node
   731	        ):
   732	            converter_type = converter_expr.node.type
   733	        elif isinstance(converter_expr.node, TypeInfo):
   734	            converter_type = type_object_type(converter_expr.node, ctx.api.named_type)
   735	    elif (
   736	        isinstance(converter_expr, IndexExpr)
   737	        and isinstance(converter_expr.analyzed, TypeApplication)
   738	        and isinstance(converter_expr.base, RefExpr)
   739	        and isinstance(converter_expr.base.node, TypeInfo)
   740	    ):
   741	        # The converter is a generic type.
   742	        converter_type = type_object_type(converter_expr.base.node, ctx.api.named_type)
   743	        if isinstance(converter_type, CallableType):
   744	            converter_type = apply_generic_arguments(
   745	                converter_type,
   746	                converter_expr.analyzed.types,
   747	                ctx.api.msg.incompatible_typevar_value,
   748	                converter_type,
   749	            )
   750	        else:
   751	            converter_type = None
   752
   753	    if isinstance(converter_expr, LambdaExpr):
   754	        # TODO: should we send a fail if converter_expr.min_args > 1?
   755	        converter_info.init_type = AnyType(TypeOfAny.unannotated)
   756	        return converter_info
   757
   758	    if not converter_type:
   759	        # Signal that we have an unsupported converter.
   760	        ctx.api.fail(
   761	            "Unsupported converter, only named functions, types and lambdas are currently "
   762	            "supported",
   763	            converter_expr,
   764	        )
   765	        converter_info.init_type = AnyType(TypeOfAny.from_error)
   766	        return converter_info
   767
   768	    converter_type = get_proper_type(converter_type)
   769	    if isinstance(converter_type, CallableType) and converter_type.arg_types:
   770	        converter_info.init_type = converter_type.arg_types[0]
   771	        if not is_attr_converters_optional:
   772	            converter_info.ret_type = converter_type.ret_type
   773	    elif isinstance(converter_type, Overloaded):
   774	        types: list[Type] = []
   775	        for item in converter_type.items:
   776	            # Walk the overloads looking for methods that can accept one argument.
   777	            num_arg_types = len(item.arg_types)
   778	            if not num_arg_types:
   779	                continue
   780	            if num_arg_types > 1 and any(kind == ARG_POS for kind in item.arg_kinds[1:]):
   781	                continue
   782	            types.append(item.arg_types[0])
   783	        # Make a union of all the valid types.
   784	        if types:
   785	            converter_info.init_type = make_simplified_union(types)
   786
   787	    if is_attr_converters_optional and converter_info.init_type:
   788	        # If the converter was attr.converter.optional(type) then add None to
   789	        # the allowed init_type.
   790	        converter_info.init_type = UnionType.make_union([converter_info.init_type, NoneType()])
   791
   792	    return converter_info
   793

   802	def _parse_assignments(
   803	    lvalue: Expression, stmt: AssignmentStmt
   804	) -> tuple[list[NameExpr], list[Expression]]:
   805	    """Convert a possibly complex assignment expression into lists of lvalues and rvalues."""
   806	    lvalues: list[NameExpr] = []
   807	    rvalues: list[Expression] = []
   808	    if isinstance(lvalue, (TupleExpr, ListExpr)):
   809	        if all(isinstance(item, NameExpr) for item in lvalue.items):
   810	            lvalues = cast(list[NameExpr], lvalue.items)
   811	        if isinstance(stmt.rvalue, (TupleExpr, ListExpr)):
   812	            rvalues = stmt.rvalue.items
   813	    elif isinstance(lvalue, NameExpr):
   814	        lvalues = [lvalue]
   815	        rvalues = [stmt.rvalue]
   816	    return lvalues, rvalues

   819	def _add_order(ctx: mypy.plugin.ClassDefContext, adder: MethodAdder) -> None:
   820	    """Generate all the ordering methods for this class."""
   821	    bool_type = ctx.api.named_type("builtins.bool")
   822	    object_type = ctx.api.named_type("builtins.object")
   823	    # Make the types be:
   824	    #    AT = TypeVar('AT')
   825	    #    def __lt__(self: AT, other: AT) -> bool
   826	    # This way comparisons with subclasses will work correctly.
   827	    fullname = f"{ctx.cls.info.fullname}.{SELF_TVAR_NAME}"
   828	    tvd = TypeVarType(
   829	        SELF_TVAR_NAME,
   830	        fullname,
   831	        # Namespace is patched per-method below.
   832	        id=TypeVarId(-1, namespace=""),
   833	        values=[],
   834	        upper_bound=object_type,
   835	        default=AnyType(TypeOfAny.from_omitted_generics),
   836	    )
   837	    self_tvar_expr = TypeVarExpr(
   838	        SELF_TVAR_NAME, fullname, [], object_type, AnyType(TypeOfAny.from_omitted_generics)
   839	    )
   840	    ctx.cls.info.names[SELF_TVAR_NAME] = SymbolTableNode(MDEF, self_tvar_expr)
   841
   842	    for method in ["__lt__", "__le__", "__gt__", "__ge__"]:
   843	        namespace = f"{ctx.cls.info.fullname}.{method}"
   844	        tvd = tvd.copy_modified(id=TypeVarId(tvd.id.raw_id, namespace=namespace))
   845	        args = [Argument(Var("other", tvd), tvd, None, ARG_POS)]
   846	        adder.add_method(method, args, bool_type, self_type=tvd, tvd=tvd)

   849	def _make_frozen(ctx: mypy.plugin.ClassDefContext, attributes: list[Attribute]) -> None:
   850	    """Turn all the attributes into properties to simulate frozen classes."""
   851	    for attribute in attributes:
   852	        if attribute.name in ctx.cls.info.names:
   853	            # This variable belongs to this class so we can modify it.
   854	            node = ctx.cls.info.names[attribute.name].node
   855	            if not isinstance(node, Var):
   856	                # The superclass attribute was overridden with a non-variable.
   857	                # No need to do anything here, override will be verified during
   858	                # type checking.
   859	                continue
   860	            node.is_property = True
   861	        else:
   862	            # This variable belongs to a super class so create new Var so we
   863	            # can modify it.
   864	            var = Var(attribute.name, attribute.init_type)
   865	            var.info = ctx.cls.info
   866	            var._fullname = f"{ctx.cls.info.fullname}.{var.name}"
   867	            ctx.cls.info.names[var.name] = SymbolTableNode(MDEF, var)
   868	            var.is_property = True
