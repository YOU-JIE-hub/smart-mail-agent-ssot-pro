  6463	        expressions into tuples, namedtuples, and typeddicts. We repeat this narrowing
  6464	        recursively if the parent is also a "lookup expression". So for example, if we have
  6465	        the expression "foo['bar'].baz[0]", we'd potentially end up refining types for the
  6466	        expressions "foo", "foo['bar']", and "foo['bar'].baz".

  6470	        if new_types is None:
  6471	            return None
  6472	        output_map = {}
  6473	        for expr, expr_type in new_types.items():
  6474	            # The original inferred type should always be present in the output map, of course
  6475	            output_map[expr] = expr_type
  6476

  6478	            new_mapping = self.refine_parent_types(expr, expr_type)
  6479	            for parent_expr, proposed_parent_type in new_mapping.items():

  6483	                if parent_expr in new_types:

  6488	    def refine_parent_types(self, expr: Expression, expr_type: Type) -> Mapping[Expression, Type]:
  6489	        """Checks if the given expr is a 'lookup operation' into a union and iteratively refines
  6490	        the parent types based on the 'expr_type'.
  6491
  6492	        For example, if 'expr' is an expression like 'a.b.c.d', we'll potentially return refined
  6493	        types for expressions 'a', 'a.b', and 'a.b.c'.
  6494
  6495	        For more details about what a 'lookup operation' is and how we use the expr_type to refine
  6496	        the parent types of lookup_expr, see the docstring in 'propagate_up_typemap_info'.
  6497	        """
  6498	        output: dict[Expression, Type] = {}
  6499
  6500	        # Note: parent_expr and parent_type are progressively refined as we crawl up the
  6501	        # parent lookup chain.
  6502	        while True:
  6503	            # First, check if this expression is one that's attempting to
  6504	            # "lookup" some key in the parent type. If so, save the parent type
  6505	            # and create function that will try replaying the same lookup
  6506	            # operation against arbitrary types.
  6507	            if isinstance(expr, MemberExpr):
  6508	                parent_expr = self._propagate_walrus_assignments(expr.expr, output)
  6509	                parent_type = self.lookup_type_or_none(parent_expr)

  6511
  6512	                def replay_lookup(new_parent_type: ProperType) -> Type | None:
  6513	                    with self.msg.filter_errors() as w:
  6514	                        member_type = analyze_member_access(
  6515	                            name=member_name,
  6516	                            typ=new_parent_type,
  6517	                            context=parent_expr,
  6518	                            is_lvalue=False,
  6519	                            is_super=False,
  6520	                            is_operator=False,
  6521	                            original_type=new_parent_type,
  6522	                            chk=self,
  6523	                            in_literal_context=False,
  6524	                        )
  6525	                    if w.has_new_errors():
  6526	                        return None
  6527	                    else:
  6528	                        return member_type
  6529
  6530	            elif isinstance(expr, IndexExpr):
  6531	                parent_expr = self._propagate_walrus_assignments(expr.base, output)
  6532	                parent_type = self.lookup_type_or_none(parent_expr)
  6533
  6534	                self._propagate_walrus_assignments(expr.index, output)
  6535	                index_type = self.lookup_type_or_none(expr.index)
  6536	                if index_type is None:
  6537	                    return output
  6538
  6539	                str_literals = try_getting_str_literals_from_type(index_type)
  6540	                if str_literals is not None:
  6541	                    # Refactoring these two indexing replay functions is surprisingly
  6542	                    # tricky -- see https://github.com/python/mypy/pull/7917, which
  6543	                    # was blocked by https://github.com/mypyc/mypyc/issues/586
  6544	                    def replay_lookup(new_parent_type: ProperType) -> Type | None:
  6545	                        if not isinstance(new_parent_type, TypedDictType):
  6546	                            return None
  6547	                        try:
  6548	                            assert str_literals is not None
  6549	                            member_types = [new_parent_type.items[key] for key in str_literals]
  6550	                        except KeyError:
  6551	                            return None
  6552	                        return make_simplified_union(member_types)
  6553
  6554	                else:
  6555	                    int_literals = try_getting_int_literals_from_type(index_type)
  6556	                    if int_literals is not None:
  6557
  6558	                        def replay_lookup(new_parent_type: ProperType) -> Type | None:
  6559	                            if not isinstance(new_parent_type, TupleType):
  6560	                                return None
  6561	                            try:
  6562	                                assert int_literals is not None
  6563	                                member_types = [new_parent_type.items[key] for key in int_literals]
  6564	                            except IndexError:
  6565	                                return None
  6566	                            return make_simplified_union(member_types)
  6567
  6568	                    else:
  6569	                        return output
  6570	            else:
  6571	                return output
  6572
  6573	            # If we somehow didn't previously derive the parent type, abort completely
  6574	            # with what we have so far: something went wrong at an earlier stage.
  6575	            if parent_type is None:
  6576	                return output
  6577
  6578	            # We currently only try refining the parent type if it's a Union.
  6579	            # If not, there's no point in trying to refine any further parents
  6580	            # since we have no further information we can use to refine the lookup
  6581	            # chain, so we end early as an optimization.
  6582	            parent_type = get_proper_type(parent_type)
  6583	            if not isinstance(parent_type, UnionType):
  6584	                return output
  6585
  6586	            # Take each element in the parent union and replay the original lookup procedure
  6587	            # to figure out which parents are compatible.
  6588	            new_parent_types = []
  6589	            for item in flatten_nested_unions(parent_type.items):
  6590	                member_type = replay_lookup(get_proper_type(item))
  6591	                if member_type is None:
  6592	                    # We were unable to obtain the member type. So, we give up on refining this
  6593	                    # parent type entirely and abort.
  6594	                    return output
  6595
  6596	                if is_overlapping_types(member_type, expr_type):
  6597	                    new_parent_types.append(item)
  6598
  6599	            # If none of the parent types overlap (if we derived an empty union), something
  6600	            # went wrong. We should never hit this case, but deriving the uninhabited type or
  6601	            # reporting an error both seem unhelpful. So we abort.
  6602	            if not new_parent_types:
  6603	                return output
  6604
  6605	            expr = parent_expr
  6606	            expr_type = output[parent_expr] = make_simplified_union(new_parent_types)

  6626	    def refine_identity_comparison_expression(
  6627	        self,
  6628	        operands: list[Expression],
  6629	        operand_types: list[Type],
  6630	        chain_indices: list[int],
  6631	        narrowable_operand_indices: AbstractSet[int],
  6632	        is_valid_target: Callable[[ProperType], bool],
  6633	        coerce_only_in_literal_context: bool,
  6634	    ) -> tuple[TypeMap, TypeMap]:
  6635	        """Produce conditional type maps refining expressions by an identity/equality comparison.
  6636
  6637	        The 'operands' and 'operand_types' lists should be the full list of operands used
  6638	        in the overall comparison expression. The 'chain_indices' list is the list of indices
  6639	        actually used within this identity comparison chain.
  6640
  6641	        So if we have the expression:
  6642
  6643	            a <= b is c is d <= e
  6644
  6645	        ...then 'operands' and 'operand_types' would be lists of length 5 and 'chain_indices'
  6646	        would be the list [1, 2, 3].
  6647
  6648	        The 'narrowable_operand_indices' parameter is the set of all indices we are allowed
  6649	        to refine the types of: that is, all operands that will potentially be a part of
  6650	        the output TypeMaps.
  6651
  6652	        Although this function could theoretically try setting the types of the operands
  6653	        in the chains to the meet, doing that causes too many issues in real-world code.
  6654	        Instead, we use 'is_valid_target' to identify which of the given chain types
  6655	        we could plausibly use as the refined type for the expressions in the chain.
  6656
  6657	        Similarly, 'coerce_only_in_literal_context' controls whether we should try coercing
  6658	        expressions in the chain to a Literal type. Performing this coercion is sometimes
  6659	        too aggressive of a narrowing, depending on context.
  6660	        """
  6661	        should_coerce = True
  6662	        if coerce_only_in_literal_context:
  6663
  6664	            def should_coerce_inner(typ: Type) -> bool:
  6665	                typ = get_proper_type(typ)
  6666	                return is_literal_type_like(typ) or (
  6667	                    isinstance(typ, Instance) and typ.type.is_enum
  6668	                )
  6669
  6670	            should_coerce = any(should_coerce_inner(operand_types[i]) for i in chain_indices)
  6671
  6672	        target: Type | None = None
  6673	        possible_target_indices = []
  6674	        for i in chain_indices:
  6675	            expr_type = operand_types[i]
  6676	            if should_coerce:
  6677	                expr_type = coerce_to_literal(expr_type)
  6678	            if not is_valid_target(get_proper_type(expr_type)):
  6679	                continue
  6680	            if target and not is_same_type(target, expr_type):
  6681	                # We have multiple disjoint target types. So the 'if' branch
  6682	                # must be unreachable.
  6683	                return None, {}
  6684	            target = expr_type
  6685	            possible_target_indices.append(i)
  6686
  6687	        # There's nothing we can currently infer if none of the operands are valid targets,
  6688	        # so we end early and infer nothing.
  6689	        if target is None:
  6690	            return {}, {}
  6691
  6692	        # If possible, use an unassignable expression as the target.
  6693	        # We skip refining the type of the target below, so ideally we'd
  6694	        # want to pick an expression we were going to skip anyways.
  6695	        singleton_index = -1
  6696	        for i in possible_target_indices:
  6697	            if i not in narrowable_operand_indices:
  6698	                singleton_index = i
  6699
  6700	        # But if none of the possible singletons are unassignable ones, we give up
  6701	        # and arbitrarily pick the last item, mostly because other parts of the
  6702	        # type narrowing logic bias towards picking the rightmost item and it'd be
  6703	        # nice to stay consistent.
  6704	        #
  6705	        # That said, it shouldn't matter which index we pick. For example, suppose we
  6706	        # have this if statement, where 'x' and 'y' both have singleton types:
  6707	        #
  6708	        #     if x is y:
  6709	        #         reveal_type(x)
  6710	        #         reveal_type(y)
  6711	        #     else:
  6712	        #         reveal_type(x)
  6713	        #         reveal_type(y)
  6714	        #
  6715	        # At this point, 'x' and 'y' *must* have the same singleton type: we would have
  6716	        # ended early in the first for-loop in this function if they weren't.
  6717	        #
  6718	        # So, we should always get the same result in the 'if' case no matter which
  6719	        # index we pick. And while we do end up getting different results in the 'else'
  6720	        # case depending on the index (e.g. if we pick 'y', then its type stays the same
  6721	        # while 'x' is narrowed to '<uninhabited>'), this distinction is also moot: mypy
  6722	        # currently will just mark the whole branch as unreachable if either operand is
  6723	        # narrowed to <uninhabited>.
  6724	        if singleton_index == -1:
  6725	            singleton_index = possible_target_indices[-1]
  6726
  6727	        sum_type_name = None
  6728	        target = get_proper_type(target)
  6729	        if isinstance(target, LiteralType) and (
  6730	            target.is_enum_literal() or isinstance(target.value, bool)
  6731	        ):
  6732	            sum_type_name = target.fallback.type.fullname
  6733
  6734	        target_type = [TypeRange(target, is_upper_bound=False)]
  6735
  6736	        partial_type_maps = []
  6737	        for i in chain_indices:
  6738	            # If we try refining a type against itself, conditional_type_map
  6739	            # will end up assuming that the 'else' branch is unreachable. This is
  6740	            # typically not what we want: generally the user will intend for the
  6741	            # target type to be some fixed 'sentinel' value and will want to refine
  6742	            # the other exprs against this one instead.
  6743	            if i == singleton_index:
  6744	                continue
  6745
  6746	            # Naturally, we can't refine operands which are not permitted to be refined.
  6747	            if i not in narrowable_operand_indices:
  6748	                continue
  6749
  6750	            expr = operands[i]
  6751	            expr_type = coerce_to_literal(operand_types[i])
  6752
  6753	            if sum_type_name is not None:
  6754	                expr_type = try_expanding_sum_type_to_union(expr_type, sum_type_name)
  6755
  6756	            # We intentionally use 'conditional_types' directly here instead of
  6757	            # 'self.conditional_types_with_intersection': we only compute ad-hoc
  6758	            # intersections when working with pure instances.
  6759	            types = conditional_types(expr_type, target_type)
  6760	            partial_type_maps.append(conditional_types_to_typemaps(expr, *types))
  6761
  6762	        return reduce_conditional_maps(partial_type_maps)
