    52	def _replace_special(ttype, value, regex, specialttype,
    53	                     replacefunc=lambda x: x):
    54	    last = 0
    55	    for match in regex.finditer(value):
    56	        start, end = match.start(), match.end()
    57	        if start != last:
    58	            yield ttype, value[last:start]

    60	        last = end
