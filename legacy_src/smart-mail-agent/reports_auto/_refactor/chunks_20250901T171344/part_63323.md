  2552	    typ: Type,
  2553	    verbosity: int,

  2555	    fullnames: set[str] | None,

  2559	    Convert a type to a relatively short string suitable for error messages.
  2560
  2561	    Args:

  2563	      fullnames: a set of names that should be printed in full

  2565
  2566	    def format(typ: Type) -> str:

  2572	    def format_union_items(types: Sequence[Type]) -> list[str]:

  2618	            return base_str

  2621	            return "<typing special form>"
  2622	        if verbosity >= 2 or (fullnames and itype.type.fullname in fullnames):
  2623	            base_str = itype.type.fullname
  2624	        else:
  2625	            base_str = itype.type.name
  2626	        if not itype.args:
  2627	            if itype.type.has_type_var_tuple_type and len(itype.type.type_vars) == 1:
  2628	                return base_str + "[()]"

  2630	            return base_str
  2631	        elif itype.type.fullname == "builtins.tuple":
  2632	            item_type_str = format(itype.args[0])
  2633	            return f"tuple[{item_type_str}, ...]"
  2634	        else:
  2635	            # There are type arguments. Convert the arguments to strings.
  2636	            return f"{base_str}[{format_list(itype.args)}]"
  2637	    elif isinstance(typ, UnpackType):
  2638	        if options.use_star_unpack():
  2639	            return f"*{format(typ.type)}"
  2640	        return f"Unpack[{format(typ.type)}]"
  2641	    elif isinstance(typ, TypeVarType):
  2642	        # This is similar to non-generic instance types.
  2643	        fullname = scoped_type_var_name(typ)
  2644	        if verbosity >= 2 or (fullnames and fullname in fullnames):
  2645	            return fullname
  2646	        return typ.name
  2647	    elif isinstance(typ, TypeVarTupleType):
  2648	        # This is similar to non-generic instance types.
  2649	        fullname = scoped_type_var_name(typ)
  2650	        if verbosity >= 2 or (fullnames and fullname in fullnames):
  2651	            return fullname
  2652	        return typ.name
  2653	    elif isinstance(typ, ParamSpecType):
  2654	        # Concatenate[..., P]
  2655	        if typ.prefix.arg_types:

  2687	        if not isinstance(typ, UnionType):

  2692	        # If there's just one Literal item, retain the original ordering.
  2693	        if len(literal_items) > 1:
  2694	            literal_str = "Literal[{}]".format(

  2707	                    if options.use_or_syntax()

  2710	            else:

  2712	        else:

  2714	            print_as_optional = (
  2715	                len(typ.items) - sum(isinstance(get_proper_type(t), NoneType) for t in typ.items)
  2716	                == 1
  2717	            )
  2718	            if print_as_optional:
  2719	                rest = [t for t in typ.items if not isinstance(get_proper_type(t), NoneType)]
  2720	                return (
  2721	                    f"{format(rest[0])} | None"
  2722	                    if options.use_or_syntax()
  2723	                    else f"Optional[{format(rest[0])}]"
  2724	                )
  2725	            else:
  2726	                s = (
  2727	                    format_union(typ.items)
  2728	                    if options.use_or_syntax()
  2729	                    else f"Union[{', '.join(format_union_items(typ.items))}]"
  2730	                )
  2731	            return s
  2732	    elif isinstance(typ, NoneType):
  2733	        return "None"
  2734	    elif isinstance(typ, AnyType):
  2735	        return "Any"
  2736	    elif isinstance(typ, DeletedType):
  2737	        return "<deleted>"
  2738	    elif isinstance(typ, UninhabitedType):
  2739	        return "Never"
  2740	    elif isinstance(typ, TypeType):
  2741	        return f"type[{format(typ.item)}]"
  2742	    elif isinstance(typ, FunctionLike):

  2744	        if func.is_type_obj():

  2746	            # return type (this always works).

  2751	            elif func.type_is is not None:
  2752	                return_type = f"TypeIs[{format(func.type_is)}]"
  2753	            else:
  2754	                return_type = format(func.ret_type)
  2755	            if func.is_ellipsis_args:
  2756	                return f"Callable[..., {return_type}]"
  2757	            param_spec = func.param_spec()
  2758	            if param_spec is not None:
  2759	                return f"Callable[{format(param_spec)}, {return_type}]"

  2773	        return f"[{args}]"
  2774	    elif typ is None:

  2828	def find_type_overlaps(*types: Type) -> set[str]:
  2829	    """Return a set of fullnames that share a short name and appear in either type.
  2830
  2831	    This is used to ensure that distinct types with the same short name are printed
  2832	    with their fullname.

  2849	    for fullnames in d.values():
  2850	        if len(fullnames) > 1:
  2851	            overlaps.update(fullnames)
