text.
   336	    inst_type = map_instance_to_supertype(inst_type, super_info)
   337	    # Finally expand the type variables in type with those in the previously
   338	    # constructed type. Note that both type and inst_type may have type
   339	    # variables, but in type they are interpreted in supertype context while
   340	    # in inst_type they are interpreted in subtype context. This works even if
   341	    # the names of type variables in supertype and subtype overlap.
   342	    return expand_type_by_instance(typ, inst_type)

   367	def bind_self(
   368	    method: F,
   369	    original_type: Type | None = None,
   370	    is_classmethod: bool = False,
   371	    ignore_instances: bool = False,
   372	) -> F:
   373	    """Return a copy of `method`, with the type of its first parameter (usually
   374	    self or cls) bound to original_type.
   375
   376	    If the type of `self` is a generic type (T, or Type[T] for classmethods),
   377	    instantiate every occurrence of type with original_type in the rest of the
   378	    signature and in the return type.
   379
   380	    original_type is the type of E in the expression E.copy(). It is None in
   381	    compatibility checks. In this case we treat it as the erasure of the
   382	    declared type of self.
   383
   384	    This way we can express "the type of self". For example:
   385
   386	    T = TypeVar('T', bound='A')
   387	    class A:
   388	        def copy(self: T) -> T: ...
   389
   390	    class B(A): pass
   391
   392	    b = B().copy()  # type: B
   393
   394	    """
   395	    if isinstance(method, Overloaded):
   396	        items = []
   397	        original_type = get_proper_type(original_type)
   398	        for c in method.items:
   399	            if isinstance(original_type, Instance):
   400	                # Filter based on whether declared self type can match actual object type.
   401	                # For example, if self has type C[int] and method is accessed on a C[str] value,
   402	                # omit this item. This is best effort since bind_self can be called in many
   403	                # contexts, and doing complete validation might trigger infinite recursion.
   404	                #
   405	                # Note that overload item filtering normally happens elsewhere. This is needed
   406	                # at least during constraint inference.
   407	                keep = is_valid_self_type_best_effort(c, original_type)
   408	            else:
   409	                keep = True
   410	            if keep:
   411	                items.append(bind_self(c, original_type, is_classmethod, ignore_instances))
   412	        if len(items) == 0:
   413	            # If no item matches, returning all items helps avoid some spurious errors
   414	            items = [
   415	                bind_self(c, original_type, is_classmethod, ignore_instances) for c in method.items
   416	            ]
   417	        return cast(F, Overloaded(items))
   418	    assert isinstance(method, CallableType)
   419	    func: CallableType = method
   420	    if not func.arg_types:
   421	        # Invalid method, return something.
   422	        return method
   423	    if func.arg_kinds[0] in (ARG_STAR, ARG_STAR2):
   424	        # The signature is of the form 'def foo(*args, ...)'.
   425	        # In this case we shouldn't drop the first arg,
   426	        # since func will be absorbed by the *args.
   427	        # TODO: infer bounds on the type of *args?
   428
   429	        # In the case of **kwargs we should probably emit an error, but
   430	        # for now we simply skip it, to avoid crashes down the line.
   431	        return method
   432	    self_param_type = get_proper_type(func.arg_types[0])
   433
   434	    variables: Sequence[TypeVarLikeType]
   435	    # Having a def __call__(self: Callable[...], ...) can cause infinite recursion. Although
   436	    # this special-casing looks not very principled, there is nothing meaningful we can infer
   437	    # from such definition, since it is inherently indefinitely recursive.
   438	    allow_callable = func.name is None or not func.name.startswith("__call__ of")
   439	    if func.variables and supported_self_type(
   440	        self_param_type, allow_callable=allow_callable, allow_instances=not ignore_instances
   441	    ):
   442	        from mypy.infer import infer_type_arguments
   443
   444	        if original_type is None:
   445	            # TODO: type check method override (see #7861).
   446	            original_type = erase_to_bound(self_param_type)
   447	        original_type = get_proper_type(original_type)
   448
   449	        # Find which of method type variables appear in the type of "self".
   450	        self_ids = {tv.id for tv in get_all_type_vars(self_param_type)}
   451	        self_vars = [tv for tv in func.variables if tv.id in self_ids]
   452
   453	        # Solve for these type arguments using the actual class or instance type.
   454	        typeargs = infer_type_arguments(
   455	            self_vars, self_param_type, original_type, is_supertype=True
   456	        )
   457	        if (
   458	            is_classmethod
   459	            and any(isinstance(get_proper_type(t), UninhabitedType) for t in typeargs)
   460	            and isinstance(original_type, (Instance, TypeVarType, TupleType))
   461	        ):
   462	            # In case we call a classmethod through an instance x, fallback to type(x).
   463	            typeargs = infer_type_arguments(
   464	                self_vars, self_param_type, TypeType(original_type), is_supertype=True
   465	            )
   466
   467	        # Update the method signature with the solutions found.
   468	        # Technically, some constraints might be unsolvable, make them Never.
   469	        to_apply = [t if t is not None else UninhabitedType() for t in typeargs]
   470	        func = expand_type(func, {tv.id: arg for tv, arg in zip(self_vars, to_apply)})
   471	        variables = [v for v in func.variables if v not in self_vars]
   472	    else:
   473	        variables = func.variables
   474
   475	    original_type = get_proper_type(original_type)
   476	    if isinstance(original_type, CallableType) and original_type.is_type_obj():
   477	        original_type = TypeType.make_normalized(original_type.ret_type)
   478	    res = func.copy_modified(
   479	        arg_types=func.arg_types[1:],
   480	        arg_kinds=func.arg_kinds[1:],
   481	        arg_names=func.arg_names[1:],
   482	        variables=variables,
   483	        is_bound=True,
   484	    )
   485	    return cast(F, res)

   488	def is_valid_self_type_best_effort(c: CallableType, self_type: Instance) -> bool:
   489	    """Quickly check if self_type might match the self in a callable.
   490
   491	    Avoid performing any complex type operations. This is performance-critical.
   492
   493	    Default to returning True if we don't know (or it would be too expensive).
   494	    """
   495	    if (
   496	        self_type.args
   497	        and c.arg_types
   498	        and isinstance((arg_type := get_proper_type(c.arg_types[0])), Instance)
   499	        and c.arg_kinds[0] in (ARG_POS, ARG_OPT)
   500	        and arg_type.args
   501	        and self_type.type.fullname != "functools._SingleDispatchCallable"
   502	    ):
   503	        if self_type.type is not arg_type.type:
   504	            # We can't map to supertype, since it could trigger expensive checks for
   505	            # protocol types, so we consevatively assume this is fine.
   506	            return True
   507
   508	        # Fast path: no explicit annotation on self
   509	        if all(
   510	            (
   511	                type(arg) is TypeVarType
   512	                and type(arg.upper_bound) is Instance
   513	                and arg.upper_bound.type.fullname == "builtins.object"
   514	            )
   515	            for arg in arg_type.args
   516	        ):
   517	            return True
   518
   519	        from mypy.meet import is_overlapping_types
   520
   521	        return is_overlapping_types(self_type, c.arg_types[0])
   522	    return True

   526	    # TODO: use value restrictions to produce a union?
   527	    t = get_proper_type(t)

   536	def callable_corresponding_argument(
   537	    typ: NormalizedCallableType | Parameters, model: FormalArgument
   538	) -> FormalArgument | None:
   539	    """Return the argument a function that corresponds to `model`"""
   540
   541	    by_name = typ.argument_by_name(model.name)
   542	    by_pos = typ.argument_by_position(model.pos)
   543	    if by_name is None and by_pos is None:
   544	        return None
   545	    if by_name is not None and by_pos is not None:
   546	        if by_name == by_pos:
   547	            return by_name
   548	        # If we're dealing with an optional pos-only and an optional
   549	        # name-only arg, merge them.  This is the case for all functions
   550	        # taking both *args and **args, or a pair of functions like so:
   551
   552	        # def right(a: int = ...) -> None: ...
   553	        # def left(__a: int = ..., *, a: int = ...) -> None: ...
   554	        from mypy.subtypes import is_equivalent
   555
   556	        if (
   557	            not (by_name.required or by_pos.required)
   558	            and by_pos.name is None
   559	            and by_name.pos is None
   560	            and is_equivalent(by_name.typ, by_pos.typ)
   561	        ):
   562	            return FormalArgument(by_name.name, by_pos.pos, by_name.typ, False)
   563	    return by_name if by_name is not None else by_pos

   583	def make_simplified_union(
   584	    items: Sequence[Type],
   585	    line: int = -1,
   586	    column: int = -1,
   587	    *,
   588	    keep_erased: bool = False,
   589	    contract_literals: bool = True,
   590	    handle_recursive: bool = True,
   591	) -> ProperType:
   592	    """Build union type with redundant union items removed.
   593
   594	    If only a single item remains, this may return a non-union type.
   595
   596	    Examples:
   597
   598	    * [int, str] -> Union[int, str]
   599	    * [int, object] -> object
   600	    * [int, int] -> int
   601	    * [int, Any] -> Union[int, Any] (Any types are not simplified away!)
   602	    * [Any, Any] -> Any
   603	    * [int, Union[bytes, str]] -> Union[int, bytes, str]
   604
   605	    Note: This must NOT be used during semantic analysis, since TypeInfos may not
   606	          be fully initialized.
   607
   608	    The keep_erased flag is used for type inference against union types
   609	    containing type variables. If set to True, keep all ErasedType items.
   610
   611	    The contract_literals flag indicates whether we need to contract literal types
   612	    back into a sum type. Set it to False when called by try_expanding_sum_type_
   613	    to_union().
   614	    """
   615	    # Step 1: expand all nested unions
   616	    items = flatten_nested_unions(items, handle_recursive=handle_recursive)
   617
   618	    # Step 2: fast path for single item
   619	    if len(items) == 1:
   620	        return get_proper_type(items[0])
   621
   622	    # Step 3: remove redundant unions
   623	    simplified_set: Sequence[Type] = _remove_redundant_union_items(items, keep_erased)
   624
   625	    # Step 4: If more than one literal exists in the union, try to simplify
   626	    if (
   627	        contract_literals
   628	        and sum(isinstance(get_proper_type(item), LiteralType) for item in simplified_set) > 1
   629	    ):
   630	        simplified_set = try_contracting_literals_in_union(simplified_set)
   631
   632	    result = get_proper_type(UnionType.make_union(simplified_set, line, column))
   633
   634	    nitems = len(items)
   635	    if nitems > 1 and (
   636	        nitems > 2 or not (type(items[0]) is NoneType or type(items[1]) is NoneType)
   637	    ):
   638	        # Step 5: At last, we erase any (inconsistent) extra attributes on instances.
   639
   640	        # Initialize with None instead of an empty set as a micro-optimization. The set
   641	        # is needed very rarely, so we try to avoid constructing it.
   642	        extra_attrs_set: set[ExtraAttrs] | None = None
   643	        for item in items:
   644	            instance = try_getting_instance_fallback(item)
   645	            if instance and instance.extra_attrs:
   646	                if extra_attrs_set is None:
   647	                    extra_attrs_set = {instance.extra_attrs}
   648	                else:
   649	                    extra_attrs_set.add(instance.extra_attrs)
   650
   651	        if extra_attrs_set is not None and len(extra_attrs_set) > 1:
   652	            fallback = try_getting_instance_fallback(result)
   653	            if fallback:
   654	                fallback.extra_attrs = None
   655
   656	    return result
   657

   659	def _remove_redundant_union_items(items: list[Type], keep_erased: bool) -> list[Type]:
   660	    from mypy.subtypes import is_proper_subtype
   661
   662	    # The first pass through this loop, we check if later items are subtypes of earlier items.
   663	    # The second pass through this loop, we check if earlier items are subtypes of later items
   664	    # (by reversing the remaining items)
   665	    for _direction in range(2):
   666	        new_items: list[Type] = []
   667	        # seen is a map from a type to its index in new_items
   668	        seen: dict[ProperType, int] = {}
   669	        unduplicated_literal_fallbacks: set[Instance] | None = None
   670	        for ti in items:
   671	            proper_ti = get_proper_type(ti)
   672
   673	            # UninhabitedType is always redundant
   674	            if isinstance(proper_ti, UninhabitedType):
   675	                continue
   676
   677	            duplicate_index = -1
   678	            # Quickly check if we've seen this type
   679	            if proper_ti in seen:
   680	                duplicate_index = seen[proper_ti]
   681	            elif (
   682	                isinstance(proper_ti, LiteralType)
   683	                and unduplicated_literal_fallbacks is not None
   684	                and proper_ti.fallback in unduplicated_literal_fallbacks
   685	            ):
   686	                # This is an optimisation for unions with many LiteralType
   687	                # We've already checked for exact duplicates. This means that any super type of
   688	                # the LiteralType must be a super type of its fallback. If we've gone through
   689	                # the expensive loop below and found no super type for a previous LiteralType
   690	                # with the same fallback, we can skip doing that work again and just add the type
   691	                # to new_items
   692	                pass
   693	            else:
   694	                # If not, check if we've seen a supertype of this type
   695	                for j, tj in enumerate(new_items):
   696	                    tj = get_proper_type(tj)
   697	                    # If tj is an Instance with a last_known_value, do not remove proper_ti
   698	                    # (unless it's an instance with the same last_known_value)
   699	                    if (
   700	                        isinstance(tj, Instance)
   701	                        and tj.last_known_value is not None
   702	                        and not (
   703	                            isinstance(proper_ti, Instance)
   704	                            and tj.last_known_value == proper_ti.last_known_value
   705	                        )
   706	                    ):
   707	                        continue
   708
   709	                    if is_proper_subtype(
   710	                        ti, tj, keep_erased_types=keep_erased, ignore_promotions=True
   711	                    ):
   712	                        duplicate_index = j
   713	                        break
   714	            if duplicate_index != -1:
   715	                # If deleted subtypes had more general truthiness, use that
   716	                orig_item = new_items[duplicate_index]
   717	                if not orig_item.can_be_true and ti.can_be_true:
   718	                    new_items[duplicate_index] = true_or_false(orig_item)
   719	                elif not orig_item.can_be_false and ti.can_be_false:
   720	                    new_items[duplicate_index] = true_or_false(orig_item)
   721	            else:
   722	                # We have a non-duplicate item, add it to new_items
   723	                seen[proper_ti] = len(new_items)
   724	                new_items.append(ti)
   725	                if isinstance(proper_ti, LiteralType):
   726	                    if unduplicated_literal_fallbacks is None:
   727	                        unduplicated_literal_fallbacks = set()
   728	                    unduplicated_literal_fallbacks.add(proper_ti.fallback)
   729
   730	        items = new_items
   731	        if len(items) <= 1:
   732	            break
   733	        items.reverse()
   734
   735	    return items

   754	def true_only(t: Type) -> ProperType:
   755	    """
   756	    Restricted version of t with only True-ish values
   757	    """
   758	    t = get_proper_type(t)
   759
   760	    if not t.can_be_true:
   761	        # All values of t are False-ish, so there are no true values in it
   762	        return UninhabitedType(line=t.line, column=t.column)
   763	    elif not t.can_be_false:
   764	        # All values of t are already True-ish, so true_only is idempotent in this case
   765	        return t
   766	    elif isinstance(t, UnionType):
   767	        # The true version of a union type
