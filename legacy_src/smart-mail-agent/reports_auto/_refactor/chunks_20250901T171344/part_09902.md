  1085	def _hmodel(s0,s1,h0,h1):
  1086	    # calculate the parameters in the model
  1087	    # h = a/s**2 + b/s
  1088	    a11 = 1./s0**2
  1089	    a12 = 1./s0
  1090	    a21 = 1./s1**2
  1091	    a22 = 1./s1
  1092	    det = a11*a22-a12*a21
  1093	    b11 = a22/det
  1094	    b12 = -a12/det
  1095	    b21 = -a21/det
  1096	    b22 = a11/det
  1097	    a = b11*h0+b12*h1
  1098	    b = b21*h0+b22*h1
  1099	    return a,b

  1101	def _qsolve(h,ab):
  1102	    '''solve the model v = a/s**2 + b/s for an s which gives us v==h'''
  1103	    a,b = ab
  1104	    if abs(a)<=_FUZZ:
  1105	        return b/h
  1106	    t = 0.5*b/a
  1107	    from math import sqrt
  1108	    f = -h/a
  1109	    r = t*t-f
  1110	    if r<0: return None
  1111	    r = sqrt(r)
  1112	    if t>=0:
  1113	        s1 = -t - r
  1114	    else:
  1115	        s1 = -t + r
  1116	    s2 = f/s1
  1117	    return max(1./s1, 1.
