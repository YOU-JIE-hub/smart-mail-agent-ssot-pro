   290	    def visit_call_expr(self, node: CallExpr) -> str:
   291	        # Call expressions are not usually types, but we also treat `X = TypeVar(...)` as a
   292	        # type alias that has to be preserved (even if TypeVar is not the same as an alias)
   293	        callee = node.callee.accept(self)
   294	        args = []
   295	        for name, arg, kind in zip(node.arg_names, node.args, node.arg_kinds):
   296	            if kind == ARG_POS:
   297	                args.append(arg.accept(self))
   298	            elif kind == ARG_STAR:
   299	                args.append("*" + arg.accept(self))
   300	            elif kind == ARG_STAR2:
   301	                args.append("**" + arg.accept(self))
   302	            elif kind == ARG_NAMED:
   303	                args.append(f"{name}={arg.accept(self)}")
   304	            else:
   305	                raise ValueError(f"Unknown argument kind {kind} in call")
   306	        return f"{callee}({', '.join(args)})"

   516	class ASTStubGenerator(BaseStubGenerator, mypy.traverser.TraverserVisitor):
   517	    """Generate stub text from a mypy AST."""
   518
   519	    def __init__(
   520	        self,
   521	        _all_: list[str] | None = None,
   522	        include_private: bool = False,
   523	        analyzed: bool = False,
   524	        export_less: bool = False,
   525	        include_docstrings: bool = False,
   526	    ) -> None:
   527	        super().__init__(_all_, include_private, export_less, include_docstrings)
   528	        self._decorators: list[str] = []
   529	        # Stack of defined variables (per scope).
   530	        self._vars: list[list[str]] = [[]]
   531	        # What was generated previously in the stub file.
   532	        self._state = EMPTY
   533	        self._class_stack: list[ClassDef] = []
   534	        # Was the tree semantically analysed before?
   535	        self.analyzed = analyzed
   536	        # Short names of methods defined in the body of the current class
   537	        self.method_names: set[str] = set()
   538	        self.processing_enum = False
   539	        self.processing_dataclass = False
   540	        self.dataclass_field_specifier: tuple[str, ...] = ()

   546	    def visit_mypy_file(self, o: MypyFile) -> None:
   547	        self.module_name = o.fullname  # Current module being processed
   548	        self.path = o.path
   549	        self.set_defined_names(find_defined_names(o))
   550	        self.referenced_names = find_referenced_names(o)

   554	    def visit_overloaded_func_def(self, o: OverloadedFuncDef) -> None:
   555	        """@property with setters and getters, @overload chain and some others."""
   556	        overload_chain = False
   557	        for item in o.items:
   558	            if not isinstance(item, Decorator):
   559	                continue
   560	            if self.is_private_name(item.func.name, item.func.fullname):
   561	                continue
   562
   563	            self.process_decorator(item)
   564	            if not overload_chain:
   565	                self.visit_func_def(item.func)
   566	                if item.func.is_overload:
   567	                    overload_chain = True
   568	            elif item.func.is_overload:
   569	                self.visit_func_def(item.func)
   570	            else:
   571	                # skip the overload implementation and clear the decorator we just processed
   572	                self.clear_decorators()

   574	    def get_default_function_sig(self, func_def: FuncDef, ctx: FunctionContext) -> FunctionSig:
   575	        args = self._get_func_args(func_def, ctx)
   576	        retname = self._get_func_return(func_def, ctx)
   577	        type_args = self.format_type_args(func_def)
   578	        return FunctionSig(func_def.name, args, retname, type_args)

   580	    def _get_func_args(self, o: FuncDef, ctx: FunctionContext) -> list[ArgSig]:
   581	        args: list[ArgSig] = []
   582
   583	        # Ignore pos-only status of magic methods whose args names are elided by mypy at parse
   584	        actually_pos_only_args = o.name not in MAGIC_METHODS_POS_ARGS_ONLY
   585	        pos_only_marker_position = 0  # Where to insert "/", if any
   586	        for i, arg_ in enumerate(o.arguments):
   587	            var = arg_.variable
   588	            kind = arg_.kind
   589	            name = var.name
   590	            annotated_type = (
   591	                o.unanalyzed_type.arg_types[i]
   592	                if isinstance(o.unanalyzed_type, CallableType)
   593	                else None
   594	            )
   595	            # I think the name check is incorrect: there are libraries which
   596	            # name their 0th argument other than self/cls
   597	            is_self_arg = i == 0 and name == "self"
   598	            is_cls_arg = i == 0 and name == "cls"
   599	            typename: str | None = None
   600	            if annotated_type and not is_self_arg and not is_cls_arg:
   601	                # Luckily, an argument explicitly annotated with "Any" has
   602	                # type "UnboundType" and will not match.
   603	                if not isinstance(get_proper_type(annotated_type), AnyType):
   604	                    typename = self.print_annotation(annotated_type)
   605
   606	            if actually_pos_only_args and arg_.pos_only:
   607	                pos_only_marker_position += 1
   608
   609	            if kind.is_named() and not any(arg.name.startswith("*") for arg in args):
   610	                args.append(ArgSig("*"))
   611
   612	            default = "..."
   613	            if arg_.initializer:
   614	                if not typename:
   615	                    typename = self.get_str_type_of_node(arg_.initializer, can_be_incomplete=False)
   616	                potential_default, valid = self.get_str_default_of_node(arg_.initializer)
   617	                if valid and len(potential_default) <= 200:
   618	                    default = potential_default
   619	            elif kind == ARG_STAR:
   620	                name = f"*{name}"
   621	            elif kind == ARG_STAR2:
   622	                name = f"**{name}"
   623
   624	            args.append(
   625	                ArgSig(name, typename, default=bool(arg_.initializer), default_value=default)
   626	            )
   627	        if pos_only_marker_position:
   628	            args.insert(pos_only_marker_position, ArgSig("/"))
   629
   630	        if ctx.class_info is not None and all(
   631	            arg.type is None and arg.default is False for arg in args
   632	        ):
   633	            new_args = infer_method_arg_types(
   634	                ctx.name, ctx.class_info.self_var, [arg.name for arg in args]
   635	            )
   636
   637	            if ctx.name == "__exit__":
   638	                self.import_tracker.add_import("types")
   639	                self.import_tracker.require_name("types")
   640
   641	            if new_args is not None:
   642	                args = new_args
   643
   644	        return args

   646	    def _get_func_return(self, o: FuncDef, ctx: FunctionContext) -> str | None:
   647	        if o.name != "__init__" and isinstance(o.unanalyzed_type, CallableType):
   648	            if isinstance(get_proper_type(o.unanalyzed_type.ret_type), AnyType):
   649	                # Luckily, a return type explicitly annotated with "Any" has
   650	                # type "UnboundType" and will enter the else branch.
   651	                return None  # implicit Any
   652	            else:
   653	                return self.print_annotation(o.unanalyzed_type.ret_type)
   654	        if o.abstract_status == IS_ABSTRACT or o.name in METHODS_WITH_RETURN_VALUE:
   655	            # Always assume abstract methods return Any unless explicitly annotated. Also
   656	            # some dunder methods should not have a None return type.
   657	            return None  # implicit Any
   658	        retname = infer_method_ret_type(o.name)
   659	        if retname is not None:
   660	            return retname
   661	        if has_yield_expression(o) or has_yield_from_expression(o):
   662	            generator_name = self.add_name("collections.abc.Generator")
   663	            yield_name = "None"
   664	            send_name: str | None = None
   665	            return_name: str | None = None
   666	            if has_yield_from_expression(o):
   667	                yield_name = send_name = self.add_name("_typeshed.Incomplete")
   668	            else:
   669	                for expr, in_assignment in all_yield_expressions(o):
   670	                    if expr.expr is not None and not is_none_expr(expr.expr):
   671	                        yield_name = self.add_name("_typeshed.Incomplete")
   672	                    if in_assignment:
   673	                        send_name = self.add_name("_typeshed.Incomplete")
   674	            if has_return_statement(o):
   675	                return_name = self.add_name("_typeshed.Incomplete")
   676	            if return_name is not None:
   677	                if send_name is None:
   678	                    send_name = "None"
   679	                return f"{generator_name}[{yield_name}, {send_name}, {return_name}]"
   680	            elif send_name is not None:
   681	                return f"{generator_name}[{yield_name}, {send_name}]"
   682	            else:
   683	                return f"{generator_name}[{yield_name}]"
   684	        if not has_return_statement(o) and o.abstract_status == NOT_ABSTRACT:
   685	            return "None"
   686	        return None

   689	        if not node.body.body:
   690	            return None
   691	        expr = node.body.body[0]
   692	        if isinstance(expr, ExpressionStmt) and isinstance(expr.expr, StrExpr):
   69
