   381	def _crc11(value):
   382	    '''
   383	    >>> usps = [USPS_4State('01234567094987654321',x).binary for x in ('','01234','012345678','01234567891')]
   384	    >>> print(' '.join(nhex(x) for x in usps))
   385	    0x1122103b5c2004b1 0xd138a87bab5cf3804b1 0x202bdc097711204d21804b1 0x16907b2a24abc16a2e5c004b1
   386	    >>> print(' '.join(nhex(_crc11(x)) for x in usps))
   387	    0x51 0x65 0x606 0x751
   388	    '''
   389	    hexbytes = nhex(int(value))[2:]
   390	    hexbytes = '0'*(26-len(hexbytes))+hexbytes
   391	    gp = 0x0F35
   392	    fcs = 0x07FF
   393	    data = int(hexbytes[:2],16)<<5
   394	    for b in range(2,8):
   395	        if (fcs ^ data)&0x400:
   396	            fcs = (fcs<<1)^gp
   397	        else:
   398	            fcs = fcs<<1
   399	        fcs &= 0x7ff
   400	        data <<= 1
   401
   402	    for x in range(2,2*13,2):
   403	        data = int(hexbytes[x:x+2],16)<<3
   404	        for b in range(8):
   405	            if (fcs ^ data)&0x400:
   406	                fcs = (fcs<<1)^gp
   407	            else:
   408	                fcs = fcs<<1
   409	            fcs &= 0x7ff
   410	            data <<= 1
   411	    return fcs

   414	    '''reverse unsigned 13 bit number
   415	    >>> print(_ru13(7936), _ru13(31), _ru13(47), _ru13(7808))
   416	    31 7936 7808 47
   417	    '''
   418	    r = 0
   419	    for x in range(13):
   420	        r <<= 1
   421	        r |= i & 1
   422	        i >>= 1

   426	    '''create and return table of 13 bit values with N bits on
   427	    >>> T = _initNof13Table(5,1287)
   428	    >>> print(' '.join('T[%d]=%d' % (i, T[i]) for i in (0,1,2,3,4,1271,1272,1284,1285,1286)))

   430	    '''
   431	    T = lenT*[None]
   432	    l = 0
   433	    u = lenT-1
   434	    for c in range(8192):
   435	        bc = 0
   436	        for b in range(13):
   437	            bc += (c&(1<<b))!=0
   438	        if bc!=N: continue
   439	        r = _ru13(c)
   440	        if r<c: continue    #we already looked at this pair
   441	        if r==c:
   442	            T[u] = c

   444	        else:
   445	            T[l] = c
   446	            l += 1
   447	            T[l] = r
   448	            l += 1
   449	    assert l==(u+1), 'u+1(%d)!=l(%d) for %d of 13 table' % (u+1,l,N)
   450	    return T
