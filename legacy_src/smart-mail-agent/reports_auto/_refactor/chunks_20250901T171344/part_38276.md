   501	    """Split line of text into words (but not within quoted groups)."""
   502	    next_word = ""
   503	    res: list[str] = []
   504	    allow_break = True

   508	            next_word = ""
   509	            continue

   513	    res.append(next_word)

   526	def soft_wrap(msg: str, max_len: int, first_offset: int, num_indent: int = 0) -> str:
   527	    """Wrap a long error message into few lines.
   528
   529	    Breaks will only happen between words, and never inside a quoted group
   530	    (to avoid breaking types such as "Union[int, str]"). The 'first_offset' is
   531	    the width before the start of first line.
   532
   533	    Pad every next line with 'num_indent' spaces. Every line will be at most 'max_len'
   534	    characters, except if it is a single word or quoted group.
   535
   536	    For example:
   537	               first_offset
   538	        ------------------------
   539	        path/to/file: error: 58: Some very long error message
   540	            that needs to be split in separate lines.
   541	            "Long[Type, Names]" are never split.
   542	        ^^^^--------------------------------------------------
   543	        num_indent           max_len
   544	    """
   545	    words = split_words(msg)
   546	    next_line = words.pop(0)
   547	    lines: list[str] = []
   548	    while words:
   549	        next_word = words.pop(0)
   550	        max_line_len = max_len - num_indent if lines else max_len - first_offset
   551	        # Add 1 to account for space between words.
   552	        if len(next_line) + len(next_word) + 1 <= max_line_len:
   553	            next_line += " " + next_word
   554	        else:
   555	            lines.append(next_line)
   556	            next_line = next_word
   557	    lines.append(next_line)
   558	    padding = "\n" + " " * num_indent
   559	    return padding.join(lines)

   761	    def colorize(self, error: str) -> str:
   762	        """Colorize an output line by highlighting the status and error code."""
   763	        if ": error:" in error:
   764	            loc, msg = error.split("error:", maxsplit=1)
   765	            if self.hide_error_codes:

   775	            return (
   776	                loc
   777	                + self.style("error:", "red", bold=True)
   778	                + self.highlight_quote_groups(msg)
   779	                + self.style(code, "yellow")

   782	            loc, msg = error.split("note:", maxsplit=1)
   783	            formatted = self.highlight_quote_groups(self.underline_link(msg))

   787	            if "^" not in error:
   788	                return self.style(error, "none", dim=True)
