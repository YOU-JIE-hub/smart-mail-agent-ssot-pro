    34	def get_target_type(
    35	    tvar: TypeVarLikeType,
    36	    type: Type,
    37	    callable: CallableType,
    38	    report_incompatible_typevar_value: Callable[[CallableType, Type, str, Context], None],
    39	    context: Context,
    40	    skip_unsatisfied: bool,
    41	) -> Type | None:
    42	    p_type = get_proper_type(type)
    43	    if isinstance(p_type, UninhabitedType) and tvar.has_default():
    44	        return tvar.default
    45	    if isinstance(tvar, ParamSpecType):
    46	        return type
    47	    if isinstance(tvar, TypeVarTupleType):
    48	        return type
    49	    assert isinstance(tvar, TypeVarType)

    60	        for value in values:
    61	            if mypy.subtypes.is_subtype(type, value):
    62	                matching.append(value)
    63	        if matching:
    64	            best = matching[0]
    65	            # If there are more than one matching value, we select the narrowest
    66	            for match in matching[1:]:
    67	                if mypy.subtypes.is_subtype(match, best):
    68	                    best = match
    69	            return best
    70	        if skip_unsatisfied:
    71	            return None
    72	        report_incompatible_typevar_value(callable, type, tvar.name, context)

    77	            # so we need to erase them to avoid false positives. This is safe because we do

    79	            upper_bound = erase_typevars(upper_bound)
    80	        if not mypy.subtypes.is_subtype(type, upper_bound):
    81	            if skip_unsatisfied:
    82	                return None
    83	            report_incompatible_typevar_value(callable, type, tvar.name, context)

   113	        if type is None:

   140	        callable = expand_type(callable, id_to_type)

   183	def apply_poly(tp: CallableType, poly_tvars: Sequence[TypeVarLikeType]) -> CallableType | None:
   184	    """Make free type variables generic in the type if possible.
   185
   186	    This will translate the type `tp` while trying to create valid bindings for
   187	    type variables `poly_tvars` while traversing the type. This follows the same rules
   188	    as we do during semantic analysis phase, examples:
   189	      * Callable[Callable[[T], T], T] -> def [T] (def (T) -> T) -> T
   190	      * Callable[[], Callable[[T], T]] -> def () -> def [T] (T -> T)
   191	      * List[T] -> None (not possible)

   194	        return tp.copy_modified(
   195	            arg_types=[t.accept(PolyTranslator(poly_tvars)) for t in tp.arg_types],
   196	            ret_type=tp.ret_type.accept(PolyTranslator(poly_tvars)),
   197	            variables=[],
   198	        )
   199	    except PolyTranslationError:
   200	        return None

   236	                    found_vars.append(normalized)

    40	    argument type with the given index.

    42	    nformals = len(formal_kinds)

    44	    ambiguous_actual_kwargs: list[int] = []

    46	    for ai, actual_kind in enumerate(actual_kinds):
    47	        if actual_kind == nodes.ARG_POS:

    49	                if not formal_kinds[fi].is_star():
    50	                    formal_to_actual[fi].append(ai)
    51	                    fi += 1
    52	                elif formal_kinds[fi] == nodes.ARG_STAR:
    53	                    formal_to_actual[fi].append(ai)
