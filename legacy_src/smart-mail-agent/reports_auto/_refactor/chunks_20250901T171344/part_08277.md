ndition = self.binary_op(post_star_len, iter_list_len, "<=", line)
   840
   841	            error_block, ok_block = BasicBlock(), BasicBlock()
   842	            self.add(Branch(condition, ok_block, error_block, Branch.BOOL))
   843
   844	            self.activate_block(error_block)
   845	            self.add(
   846	                RaiseStandardError(
   847	                    RaiseStandardError.VALUE_ERROR, "not enough values to unpack", line
   848	                )
   849	            )
   850	            self.add(Unreachable())
   851
   852	            self.activate_block(ok_block)
   853
   854	            for litem in reversed(post_star_vals):
   855	                ritem = self.primitive_op(list_pop_last, [iter_list], line)
   856	                self.assign(litem, ritem, line)
   857
   858	            # Assign the starred value
   859	            self.assign(target.items[target.star_idx], iter_list, line)
   860
   861	        # There is no starred value, so check if there are extra values in rhs that
   862	        # have not been assigned.
   863	        else:
   864	            extra = self.call_c(next_op, [iterator], line)
   865	            error_block, ok_block = BasicBlock(), BasicBlock()
   866	            self.add(Branch(extra, ok_block, error_block, Branch.IS_ERROR))
   867
   868	            self.activate_block(error_block)
   869	            self.add(
   870	                RaiseStandardError(
   871	                    RaiseStandardError.VALUE_ERROR, "too many values to unpack", line
   872	                )
   873	            )
   874	            self.add(Unreachable())
   875
   876	            self.activate_block(ok_block)

  1050	    def get_final_ref(self, expr: MemberExpr) -> tuple[str, Var, bool] | None:
  1051	        """Check if `expr` is a final attribute.
  1052
  1053	        This needs to be done differently for class and module attributes to
  1054	        correctly determine fully qualified name. Return a tuple that consists of
  1055	        the qualified name, the corresponding Var node, and a flag indicating whether
  1056	        the final name was defined in a compiled module. Return None if `expr` does not
  1057	        refer to a final attribute.
  1058	        """
  1059	        final_var = None
  1060	        if isinstance(expr.expr, RefExpr) and isinstance(expr.expr.node, TypeInfo):
  1061	            # a class attribute
  1062	            sym = expr.expr.node.get(expr.name)
  1063	            if sym and isinstance(sym.node, Var):
  1064	                # Enum attribute are treated as final since they are added to the global cache
  1065	                expr_fullname = expr.expr.node.bases[0].type.fullname
  1066	                is_final = sym.node.is_final or expr_fullname == "enum.Enum"
  1067	                if is_final:
  1068	                    final_var = sym.node
  1069	                    fullname = f"{sym.node.info.fullname}.{final_var.name}"
  1070	                    native = self.is_native_module(expr.expr.node.module_name)
  1071	        elif self.is_module_member_expr(expr):
  1072	            # a module attribute
  1073	            if isinstance(expr.node, Var) and expr.node.is_final:
  1074	                final_var = expr.node
  1075	                fullname = expr.node.fullname
  1076	                native = self.is_native_ref_expr(expr)
  1077	        if final_var is not None:
  1078	            return fullname, final_var, native
  1079	        return None

  1105	        self, expr: CallExpr, callee: RefExpr, arg_values: list[Value]
  1106	    ) -> Value:

  1109	            fullname = get_call_target_fullname(callee)

  1111	            target = self.builder.matching_primitive_op(

  1113	            )
  1114	            if target:

  1116
  1117	        # Standard native call if signature and fullname are good and all arguments are positional
  1118	        # or named.

  1125	        # some other reason)

  1127	            isinstance(callee_node, Decorator)

  1133	            callee_node is not None
