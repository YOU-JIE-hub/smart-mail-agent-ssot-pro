    95	def _buildColorFunction(colors, positions):
    96	    from reportlab.pdfbase.pdfdoc import PDFExponentialFunction, PDFStitchingFunction
    97	    if positions is not None and len(positions) != len(colors):
    98	        raise ValueError("need to have the same number of colors and positions")
    99	    # simplified functions for edge cases
   100	    if len(colors) == 1:
   101	        # for completeness
   102	        return PDFExponentialFunction(N=1, C0=colors[0], C1=colors[0])
   103	    if len(colors) == 2:
   104	        if positions is None or (positions[0] == 0 and positions[1] == 1):
   105	            return PDFExponentialFunction(N=1, C0=colors[0], C1=colors[1])
   106	    # equally distribute if positions not specified
   107	    if positions is None:
   108	        nc = len(colors)
   109	        positions = [float(x)/(nc-1) for x in range(nc)]
   110	    else:
   111	        # sort positions and colors in increasing order
   112	        poscolors = list(zip(positions, colors))
   113	        poscolors.sort(key=lambda x: x[0])
   114	        # add endpoint positions if not already present
   115	        if poscolors[0][0] != 0:
   116	            poscolors.insert(0, (0.0, poscolors[0][1]))
   117	        if poscolors[-1][0] != 1:
   118	            poscolors.append((1.0, poscolors[-1][1]))
   119	        positions, colors = list(zip(*poscolors)) # unzip
   120	    # build stitching function
   121	    functions = []
   122	    bounds = [pos for pos in positions[1:-1]]
   123	    encode = []
   124	    lastcolor = colors[0]
   125	    for color in colors[1:]:
   126	        functions.append(PDFExponentialFunction(N=1, C0=lastcolor, C1=color))
   127	        lastcolor = color
   128	        encode.append(0.0)
   129	        encode.append(1.0)
