  1603	def find_module_paths_using_imports(
  1604	    modules: list[str], packages: list[str], verbose: bool, quiet: bool
  1605	) -> tuple[list[StubSource], list[StubSource]]:
  1606	    """Find path and runtime value of __all__ (if possible) for modules and packages.
  1607
  1608	    This function uses runtime Python imports to get the information.
  1609	    """
  1610	    with ModuleInspect() as inspect:
  1611	        py_modules: list[StubSource] = []
  1612	        c_modules: list[StubSource] = []
  1613	        found = list(walk_packages(inspect, packages, verbose))
  1614	        modules = modules + found
  1615	        modules = [
  1616	            mod for mod in modules if not is_non_library_module(mod)
  1617	        ]  # We don't want to run any tests or scripts
  1618	        for mod in modules:
  1619	            try:
  1620	                result = find_module_path_and_all_py3(inspect, mod, verbose)
  1621	            except CantImport as e:
  1622	                tb = traceback.format_exc()
  1623	                if verbose:
  1624	                    sys.stderr.write(tb)
  1625	                if not quiet:
  1626	                    report_missing(mod, e.message, tb)
  1627	                continue

  1629	                c_modules.append(StubSource(mod))
  1630	            else:
  1631	                path, runtime_all = result
  1632	                py_modules.append(StubSource(mod, path, runtime_all))
  1633	        return py_modules, c_modules

  1637	    """Does module look like a test module or a script?"""
  1638	    if module.endswith(
  1639	        (
  1640	            ".tests",
  1641	            ".test",
  1642	            ".testing",
  1643	            "_tests",
  1644	            "_test_suite",
  1645	            "test_util",
  1646	            "test_utils",
  1647	            "test_base",
  1648	            ".__main__",
  1649	            ".conftest",  # Used by pytest
  1650	            ".setup",  # Typically an install script
  1651	        )
  1652	    ):
  1653	        return True
  1654	    if module.split(".")[-1].startswith("test_"):
  1655	        return True
  1656	    if (
  1657	        ".tests." in module
  1658	        or ".test." in module
  1659	        or ".testing." in module

  1667	    for pkg in VENDOR_PACKAGES:

  1677	def find_module_paths_using_search(
  1678	    modules: list[str], packages: list[str], search_path: list[str], pyversion: tuple[int, int]
  1679	) -> list[StubSource]:
  1680	    """Find sources for modules and packages requested.
  1681
  1682	    This function just looks for source files at the file system level.
  1683	    This is used if user passes --no-import, and will not find C modules.
  1684	    Exit if some of the modules or packages can't be found.
  1685	    """
  1686	    result: list[StubSource] = []
  1687	    typeshed_path = default_lib_path(mypy.build.default_data_dir(), pyversion, None)
  1688	    search_paths = SearchPaths((".",) + tuple(search_path), (), (), tuple(typeshed_path))
  1689	    cache = FindModuleCache(search_paths, fscache=None, options=None)
  1690	    for module in modules:
  1691	        m_result = cache.find_module(module)
  1692	        if isinstance(m_result, ModuleNotFoundReason):
  1693	            fail_missing(module, m_result)
  1694	            module_path = None
  1695	        else:
  1696	            module_path = m_result
  1697	        result.append(StubSource(module, module_path))
  1698	    for package in packages:
  1699	        p_result = cache.find_modules_recursive(package)
  1700	        if p_result:
  1701	            fail_missing(package, ModuleNotFoundReason.NOT_FOUND)
  1702	        sources = [StubSource(m.module, m.path) for m in p_result]
  1703	        result.extend(sources)
  1704
  1705	    result = [m for m in result if not is_non_library_module(m.module)]
  1706
  1707	    return result

  1732	def parse_source_file(mod: StubSource, mypy_options: MypyOptions) -> None:
  1733	    """Parse a source file.
  1734
  1735	    On success, store AST in the corresponding attribute of the stub source.
  1736	    If there are syntax errors, print them and exit.
  1737	    """
  1738	    assert mod.path is not None, "Not found module was not skipped"
  1739	    with open(mod.path, "rb") as f:
  1740	        data = f.read()
  1741	    source = mypy.util.decode_python_encoding(data)
  1742	    errors = Errors(mypy_options)
  1743	    mod.ast = mypy.parse.parse(
  1744	        source, fnam=mod.path, module=mod.module, errors=errors, options=mypy_options
  1745	    )
  1746	    mod.ast._fullname = mod.module
  1747	    if errors.is_blockers():
  1748	        # Syntax error!
  1749	        for m in errors.new_messages():
  1750	            sys.stderr.write(f"{m}\n")
  1751	        sys.exit(1)

  1754	def generate_asts_for_modules(
  1755	    py_modules: list[StubSource], parse_only: bool, mypy_options: MypyOptions, verbose: bool
  1756	) -> None:
  1757	    """Use mypy to parse (and optionally analyze) source files."""
  1758	    if not py_modules:
  1759	        return  # Nothing to do here, but there may be C modules
  1760	    if verbose:
  1761	        print(f"Processing {len(py_modules)} files...")
  1762	    if parse_only:
  1763	        for mod in py_modules:
  1764	            parse_source_file(mod, mypy_options)
  1765	        return
  1766	    # Perform full semantic analysis of the source set.
  1767	    try:
  1768	        res = build([module.source for module in py_modules], mypy_options)
  1769	    except CompileError as e:
  1770	        raise SystemExit(f"Critical error during semantic analysis: {e}") from e
  1771
  1772	    for mod in py_modules:
  1773	        mod.ast = res.graph[mod.module].tree
  1774	        # Use statically inferred __all__ if there is no runtime one.
  1775	        if mod.runtime_all is None:
  1776	            mod.runtime_all = res.manager.semantic_analyzer.export_map[mod.module]

  1829	def generate_stubs(options: Options) -> None:
  1830	    """Main entry point for the program."""
  1831	    mypy_opts = mypy_options(options)
  1832	    py_modules, pyc_modules, c_modules = collect_build_targets(options, mypy_opts)
  1833	    all_modules = py_modules + pyc_modules + c_modules
  1834	    all_module_names = sorted(m.module for m in all_modules)
  1835	    # Use parsed sources to generate stubs for Python modules.
  1836	    generate_asts_for_modules(py_modules, options.parse_only, mypy_opts, options.verbose)
  1837	    files = []
  1838	    for mod in py_modules + pyc_modules:
  1839	        assert mod.path is not None, "Not found module was not skipped"
  1840	        target = mod.module.replace(".", "/")
  1841	        if os.path.basename(mod.path) in ["__init__.py", "__init__.pyc"]:
  1842	            target += "/__init__.pyi"
  1843	        else:
  1844	            target += ".pyi"
  1845	        target = os.path.join(options.output_dir, target)
  1846	        files.append(target)
  1847	        with generate_guarded(mod.module, target, options.ignore_errors, options.verbose):
  1848	            generate_stub_for_py_module(
  1849	                mod,
  1850	                target,

  1852	                inspect=options.inspect or mod in pyc_modules,
  1853	                include_private=options.include_private,
  1854	                export_less=options.export_less,
  1855	                include_docstrings=options.include_docstrings,
  1856	                doc_dir=options.doc_dir,
  1857	                all_modules=all_module_names,
  1858	            )
  1859
  1860	    # Separately analyse C modules using different logic.
  1861	    for mod in c_modules:
  1862	        if any(py_mod.module.startswith(mod.module + ".") for py_mod in all_modules):
  1863	            target = mod.module.replace(".", "/") + "/__init__.pyi"

  1865	            target = mod.module.replace(".", "/") + ".pyi"
  1866	        target = os.path.join(options.output_dir, target)
  1867	        files.append(target)

  1869	            generate_stub_for_c_module(
  1870	                mod.module,
  1871	                target,
  1872	                known_modules=all_module_names,
  1873	                doc_dir=options.doc_dir,

  1875	                export_less=options.export_less,
  1876	                include_docstrings=options.include_docstrings,
  1877	            )
  1878	    num_modules = len(all_modules)
  1879	    if not options.quiet and num_modules > 0:
  1880	        print("Processed %d modules" % num_modules)
