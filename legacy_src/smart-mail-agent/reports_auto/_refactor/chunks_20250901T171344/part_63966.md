  1053	def getArcPoints(centerx, centery, radius, startangledegrees, endangledegrees, yradius=None, degreedelta=None, reverse=None):
  1054	    if yradius is None: yradius = radius
  1055	    points = []
  1056	    degreestoradians = pi/180.0
  1057	    startangle = startangledegrees*degreestoradians
  1058	    endangle = endangledegrees*degreestoradians
  1059	    while endangle<startangle:
  1060	        endangle = endangle+2*pi
  1061	    angle = float(endangle - startangle)
  1062	    a = points.append
  1063	    if angle>.001:
  1064	        degreedelta = min(angle,degreedelta or 1.)
  1065	        radiansdelta = degreedelta*degreestoradians
  1066	        n = max(int(angle/radiansdelta+0.5),1)
  1067	        radiansdelta = angle/n
  1068	        n += 1
  1069	    else:
  1070	        n = 1
  1071	        radiansdelta = 0
  1072
  1073	    for angle in range(n):
  1074	        angle = startangle+angle*radiansdelta
  1075	        a((centerx+radius*cos(angle),centery+yradius*sin(angle)))
  1076
  1077	    if reverse: points.reverse()
  1078	    return points

  1091	    P = []

  1102	        if len(args)!=_PATH_OP_ARG_COUNT[op]:
  1103	            raise ValueError('%s bad arguments %s' % (opName,str(args)))
  1104	        O.append(op)
  1105	        P.extend(list(args))

  1107	        for i in range(o,len(P),2):

  1113	    if bbox:
  1114	        for j in 0,1:
