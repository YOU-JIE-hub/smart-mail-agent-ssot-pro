  2180	    def clean_up_bases_and_infer_type_variables(
  2181	        self, defn: ClassDef, base_type_exprs: list[Expression], context: Context
  2182	    ) -> tuple[list[Expression], list[TypeVarLikeType], bool]:
  2183	        """Remove extra base classes such as Generic and infer type vars.
  2184
  2185	        For example, consider this class:
  2186
  2187	          class Foo(Bar, Generic[T]): ...
  2188
  2189	        Now we will remove Generic[T] from bases of Foo and infer that the
  2190	        type variable 'T' is a type argument of Foo.
  2191
  2192	        Note that this is performed *before* semantic analysis.
  2193
  2194	        Returns (remaining base expressions, inferred type variables, is protocol).
  2195	        """
  2196	        removed: list[int] = []
  2197	        declared_tvars: TypeVarLikeList = []
  2198	        is_protocol = False
  2199	        if defn.type_args is not None:
  2200	            for p in defn.type_args:
  2201	                node = self.lookup(p.name, context)
  2202	                assert node is not None
  2203	                assert isinstance(node.node, TypeVarLikeExpr)
  2204	                declared_tvars.append((p.name, node.node))
  2205
  2206	        for i, base_expr in enumerate(base_type_exprs):
  2207	            if isinstance(base_expr, StarExpr):
  2208	                base_expr.valid = True
  2209	            self.analyze_type_expr(base_expr)
  2210
  2211	            try:
  2212	                base = self.expr_to_unanalyzed_type(base_expr)
  2213	            except TypeTranslationError:
  2214	                # This error will be caught later.
  2215	                continue
  2216	            result = self.analyze_class_typevar_declaration(base)
  2217	            if result is not None:
  2218	                tvars = result[0]
  2219	                is_protocol |= result[1]
  2220	                if declared_tvars:
  2221	                    if defn.type_args:
  2222	                        if is_protocol:
  2223	                            self.fail('No arguments expected for "Protocol" base class', context)
  2224	                        else:
  2225	                            self.fail("Generic[...] base class is redundant", context)
  2226	                    else:
  2227	                        self.fail(
  2228	                            "Only single Generic[...] or Protocol[...] can be in bases", context
  2229	                        )
  2230	                removed.append(i)
  2231	                declared_tvars.extend(tvars)
  2232	            if isinstance(base, UnboundType):
  2233	                sym = self.lookup_qualified(base.name, base)
  2234	                if sym is not None and sym.node is not None:
  2235	                    if sym.node.fullname in PROTOCOL_NAMES and i not in removed:
  2236	                        # also remove bare 'Protocol' bases
  2237	                        removed.append(i)
  2238	                        is_protocol = True
  2239
  2240	        all_tvars = self.get_all_bases_tvars(base_type_exprs, removed)
  2241	        if declared_tvars:
  2242	            if len(remove_dups(declared_tvars)) < len(declared_tvars) and not defn.type_args:
  2243	                self.fail("Duplicate type variables in Generic[...] or Protocol[...]", context)
  2244	            declared_tvars = remove_dups(declared_tvars)
  2245	            if not set(all_tvars).issubset(set(declared_tvars)):
  2246	                if defn.type_args:
  2247	                    undeclared = sorted(set(all_tvars) - set(declared_tvars))
  2248	                    self.msg.type_parameters_should_be_declared(
  2249	                        [tv[0] for tv in undeclared], context
  2250	                    )
  2251	                else:
  2252	                    self.fail(
  2253	                        "If Generic[...] or Protocol[...] is present"
  2254	                        " it should list all type variables",
  2255	                        context,
  2256	                    )
  2257	                # In case of error, Generic tvars will go first
  2258	                declared_tvars = remove_dups(declared_tvars + all_tvars)
  2259	        else:
  2260	            declared_tvars = all_tvars
  2261	        for i in reversed(removed):
  2262	            # We need to actually remove the base class expressions like Generic[T],
  2263	            # mostly because otherwise they will create spurious dependencies in fine
  2264	            # grained incremental mode.
  2265	            defn.removed_base_type_exprs.append(defn.base_type_exprs[i])
  2266	            del base_type_exprs[i]
  2267	        tvar_defs = self.tvar_defs_from_tvars(declared_tvars, context)
  2268	        return base_type_exprs, tvar_defs, is_protocol

  2270	    def analyze_class_typevar_declaration(self, base: Type) -> tuple[TypeVarLikeList, bool] | None:
  2271	        """Analyze type variables declared using Generic[...] or Protocol[...].
  2272
  2273	        Args:
  2274	            base: Non-analyzed base class

  2276	        Return None if the base class does not declare type variables. Otherwise,
