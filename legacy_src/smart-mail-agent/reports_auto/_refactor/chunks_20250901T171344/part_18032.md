   658	            if isinstance(e := eg.exceptions[0], BaseExceptionGroup):

   660	            return e

   666	        ):

   669	        lines = format_exception_only(self.type, self.value)
   670	        text = "".join(lines)
   671	        text = text.rstrip()
   672	        if tryshort:
   673	            if text.startswith(self._striptext):
   674	                text = text[len(self._striptext) :]

   768	    def match(self, regexp: str | re.Pattern[str]) -> Literal[True]:
   769	        """Check whether the regular expression `regexp` matches the string
   770	        representation of the exception using :func:`python:re.search`.
   771
   772	        If it matches `True` is returned, otherwise an `AssertionError` is raised.
   773	        """
   774	        __tracebackhide__ = True
   775	        value = stringify_exception(self.value)
   776	        msg = f"Regex pattern did not match.\n Regex: {regexp!r}\n Input: {value!r}"
   777	        if regexp == value:
   778	            msg += "\n Did you mean to `re.escape()` the regex?"

   783	    def _group_contains(
   784	        self,
   785	        exc_group: BaseExceptionGroup[BaseException],
   786	        expected_exception: EXCEPTION_OR_MORE,
   787	        match: str | re.Pattern[str] | None,
   788	        target_depth: int | None = None,
   789	        current_depth: int = 1,
   790	    ) -> bool:
   791	        """Return `True` if a `BaseExceptionGroup` contains a matching exception."""
   792	        if (target_depth is not None) and (current_depth > target_depth):
   793	            # already descended past the target depth
   794	            return False
   795	        for exc in exc_group.exceptions:
   796	            if isinstance(exc, BaseExceptionGroup):
   797	                if self._group_contains(
   798	                    exc, expected_exception, match, target_depth, current_depth + 1
   799	                ):
   800	                    return True
   801	            if (target_depth is not None) and (current_depth != target_depth):
   802	                # not at the target depth, no match
   803	                continue
   804	            if not isinstance(exc, expected_exception):
   805	                continue
   806	            if match is not None:
   807	                value = stringify_exception(exc)
   808	                if not re.search(match, value):
   809	                    continue
   810	            return True
   811	        return False

   813	    def group_contains(
   814	        self,
   815	        expected_exception: EXCEPTION_OR_MORE,
   816	        *,
   817	        match: str | re.Pattern[str] | None = None,
   818	        depth: int | None = None,
   819	    ) -> bool:
   820	        """Check whether a captured exception group contains a matching exception.
   821
   822	        :param Type[BaseException] | Tuple[Type[BaseException]] expected_exception:
   823	            The expected exception type, or a tuple if one of multiple possible
   824	            exception types are expected.
   825
   826	        :param str | re.Pattern[str] | None match:
   827	            If specified, a string containing a regular expression,
   828	            or a regular expression object, that is tested against the string
   829	            representation of the exception and its `PEP-678 <https://peps.python.org/pep-0678/>` `__notes__`
   830	            using :func:`re.search`.
   831
   832	            To match a literal string that may contain :ref:`special characters
   833	            <re-syntax>`, the pattern can first be escaped with :func:`re.escape`.
   834
   835	        :param Optional[int] depth:
   836	            If `None`, will search for a matching exception at any nesting depth.
   837	            If >= 1, will only match an exception if it's at the specified depth (depth = 1 being
   838	            the exceptions contained within the topmost exception group).
   839
   840	        .. versionadded:: 8.0
   841
   842	        .. warning::
   843	           This helper makes it easy to check for the presence of specific exceptions,
   844	           but it is very bad for checking that the group does *not* contain
   845	           *any other exceptions*.
   846	           You should instead consider using :class:`pytest.RaisesGroup`
   847
   848	        """
   849	        msg = "Captured exception is not an instance of `BaseExceptionGroup`"
   850	        assert isinstance(self.value, BaseExceptionGroup), msg
   851	        msg = "`depth` must be >= 1 if specified"
   852	        assert (depth is None) or (depth >= 1), msg
   853	        return self._group_contains(self.value, expected_exception, match, depth)

   973	    def get_highlight_arrows_for_line(
   974	        self,
   975	        line: str,
   976	        raw_line: str,
   977	        lineno: int | None,
   978	        end_lineno: int | None,
   979	        colno: int | None,
   980	        end_colno: int | None,
   981	    ) -> list[str]:
   982	        """Return characters highlighting a source line.
   983
   984	        Example with colno and end_colno pointing to the bar expression:
   985	                   "foo() + bar()"
   986	        returns    "        ^^^^^"
   987	        """
   988	        if lineno != end_lineno:
   989	            # Don't handle expressions that span multiple lines.
   990	            return []
   991	        if colno is None or end_colno is None:
   992	            # Can't do anything without column information.
   993	            return []
   994
   995	        num_stripped_chars = len(raw_line) - len(line)
   996
   997	        start_char_offset = _byte_offset_to_character_offset(raw_line, colno)
   998	        end_char_offset = _byte_offset_to_character_offset(raw_line, end_colno)
   999	        num_carets = end_char_offset - start_char_offset
  1000	        # If the highlight would span the whole line, it is redundant, don't
  1001	        # show it.
  1002	        if num_carets >= len(line.strip()):
  1003	            return []
  1004
  1005	        highlights = "    "
  1006	        highlights += " " * (start_char_offset - num_stripped_chars + 1)
  1007	        highlights += "^" * num_carets
  1008	        return [highlights]
