   198	                            ) from None

   203	                                meta_params
   204	                                if isinstance(meta_params, tuple)
   205	                                else (meta_params,),
   206	                            )
   207
   208	                    else:
   209	                        meta_params = ()

   211	                    append_span(
   212	                        _Span(
