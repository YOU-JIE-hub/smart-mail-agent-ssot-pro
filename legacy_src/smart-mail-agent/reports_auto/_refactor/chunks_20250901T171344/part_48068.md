    50	    left: list[str], right: list[str]
    51	) -> tuple[list[tuple[int, int]], list[tuple[int, int]]]:
    52	    seq = difflib.SequenceMatcher(None, left, right)
    53	    # note last triple is a dummy, so don't need to worry
    54	    blocks = seq.get_matching_blocks()
    55
    56	    i = 0
    57	    j = 0
    58	    left_ranges = []
    59	    right_ranges = []
    60	    for block in blocks:
    61	        # mismatched range
    62	        left_ranges.append((i, block.a))
    63	        right_ranges.append((j, block.b))

    65	        i = block.a + block.size
    66	        j = block.b + block.size

    68	        # matched range

    70	        right_ranges.append((block.b, j))

   278	def retry_on_error(func: Callable[[], Any], max_wait: float = 1.0) -> None:
   279	    """Retry callback with exponential backoff when it raises OSError.

   281	    If the function still generates an error after max_wait seconds, propagate

   288	    wait_time = 0.01
