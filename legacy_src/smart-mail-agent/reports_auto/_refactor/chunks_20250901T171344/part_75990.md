def top_confusions(conf_path, k=6):
    rows=[]

def macro_f1(y_true, y_pred, labels):
    out=[]
    for lab in labels:
        tp=fp=fn=0
        for yt,yp in zip(y_true,y_pred):
            tp += (yt==lab and yp==lab)
            fp += (yt!=lab and yp==lab)
            fn += (yt==lab and yp!=lab)

        R = tp/(tp+fn) if (tp+fn)>0 else 0.0
