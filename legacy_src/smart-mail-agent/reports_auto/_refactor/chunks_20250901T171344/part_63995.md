   101	class Sequencer:
   102	    """Something to make it easy to number paragraphs, sections,
   103	    images and anything else.  The features include registering
   104	    new string formats for sequences, and 'chains' whereby
   105	    some counters are reset when their parents.

   107	    'counters', which are created on request:
   108	    Usage::

   110	        >>> seq = layout.Sequencer()

   113	        >>> seq.next('Bullets')

   115	        >>> seq.next('Bullets')
   116	        3
   117	        >>> seq.reset('Bullets')
   118	        >>> seq.next('Bullets')

   120	        >>> seq.next('Figures')
   121	        1
   122	        >>>

   127	        self._formatters = {}

   130	    def _reset(self):
   131	        self._counters.clear()
   132	        self._formatters.clear()

   134	            # the formats it knows initially
   135	            '1':_format_123,
   136	            'A':_format_ABC,

   138	            'I':_format_I,
   139	            'i':_format_i,

   142	        self.__dict__.clear()
   143	        self.__dict__.update(d)
   144	        self._defaultCounter = None

   147	        """Creates one if not present"""
   148	        try:
   149	            return self._counters[counter]

   151	            cnt = _Counter()
   152	            self._counters[counter] = cnt
   153	            return cnt

   155	    def _this(self, counter=None):
   156	        """Retrieves counter value but does not increment. For
   157	        new counters, sets base value to 1."""
   158	        if not counter:
   159	            counter = self._defaultCounter
   160	        return self._getCounter(counter)._this()
   161
   162	    def __next__(self):
   163	        """Retrieves the numeric value for the given counter, then
   164	        increments it by one.  New counters start at one."""
   165	        return next(self._getCounter(self._defaultCounter))
   166
   167	    def next(self,counter=None):
   168	        if not counter:
   169	            return next(self)
   170	        else:
   171	            dc = self._defaultCounter
   172	            try:
   173	                self._defaultCounter = counter
   174	                return next(self)
   175	            finally:
   176	                self._defaultCounter = dc
   177

   180	            counter = self._defaultCounter

   185	        increments it by one.  New counters start at one."""
   186	        if not counter:
   187	            counter = self._defaultCounter
   188	        return self._getCounter(counter).nextf()

   191	        """Changes the key used for the default"""
   192	        self._defaultCounter = default

   194	    def registerFormat(self, format, func):
   195	        """Registers a new formatting function.  The funtion
   196	        must take a number as argument and return a string;
   197	        fmt is a short menmonic string used to access it."""
   198	        self._formatters[format] = func

    59	        assert 'name' not in self.defaults, "Class Defaults may not contain a 'name' attribute"

    66	        self.parent = parent
