  1551	        ``pytest_configure`` (or similar stages).

  1562	        with warnings.catch_warnings(record=True) as records:

  1631	    def _getini(self, name: str):
  1632	        try:
  1633	            description, type, default = self._parser._inidict[name]
  1634	        except KeyError as e:
  1635	            raise ValueError(f"unknown configuration value: {name!r}") from e
  1636	        override_value = self._get_override_ini_value(name)
  1637	        if override_value is None:
  1638	            try:
  1639	                value = self.inicfg[name]
  1640	            except KeyError:
  1641	                return default
  1642	        else:
  1643	            value = override_value
  1644	        # Coerce the values based on types.
  1645	        #
  1646	        # Note: some coercions are only required if we are reading from .ini files, because
  1647	        # the file format doesn't contain type information, but when reading from toml we will
  1648	        # get either str or list of str values (see _parse_ini_config_from_pyproject_toml).
  1649	        # For example:
  1650	        #
  1651	        #   ini:
  1652	        #     a_line_list = "tests acceptance"
  1653	        #   in this case, we need to split the string to obtain a list of strings.
  1654	        #
  1655	        #   toml:
  1656	        #     a_line_list = ["tests", "acceptance"]
  1657	        #   in this case, we already have a list ready to use.
  1658	        #
  1659	        if type == "paths":
  1660	            dp = (
  1661	                self.inipath.parent
  1662	                if self.inipath is not None
  1663	                else self.invocation_params.dir
  1664	            )
  1665	            input_values = shlex.split(value) if isinstance(value, str) else value
  1666	            return [dp / x for x in input_values]
  1667	        elif type == "args":
  1668	            return shlex.split(value) if isinstance(value, str) else value
  1669	        elif type == "linelist":
  1670	            if isinstance(value, str):
  1671	                return [t for t in map(lambda x: x.strip(), value.split("\n")) if t]
  1672	            else:
  1673	                return value
  1674	        elif type == "bool":
  1675	            return _strtobool(str(value).strip())
  1676	        elif type == "string":
  1677	            return value
  1678	        elif type == "int":
  1679	            if not isinstance(value, str):
  1680	                raise TypeError(
  1681	                    f"Expected an int string for option {name} of type integer, but got: {value!r}"
  1682	                ) from None
  1683	            return int(value)
  1684	        elif type == "float":
  1685	            if not isinstance(value, str):
  1686	                raise TypeError(
  1687	                    f"Expected a float string for option {name} of type float, but got: {value!r}"
  1688	                ) from None
  1689	            return float(value)
  1690	        elif type is None:
  1691	            return value
  1692	        else:
  1693	            return self._getini_unknown_type(name, type, value)
