   828	        '''return a base class drawing which ensures all the contents fits'''
   829	        C = self.getContents()

   837	            drawing.transform = (1,0,0,1,lpad-xL,bpad-yL)

   842	            drawing.height = yH-yL+tpad+bpad

   940	def _renderPath(path,drawFuncs,countOnly=False,forceClose=False):
   941	    """Helper function for renderers."""
   942	    # this could be a method of Path...
   943	    points = path.points
   944	    i = 0
   945	    hadClosePath = 0
   946	    hadMoveTo = 0
   947	    active = not countOnly
   948	    for op in path.operators:
   949	        if op == _MOVETO:
   950	            if forceClose:
   951	                if hadMoveTo and pop!=_CLOSEPATH:
   952	                    hadClosePath += 1
   953	                    if active:
   954	                        drawFuncs[_CLOSEPATH]()
   955	            hadMoveTo += 1
   956	        nArgs = _PATH_OP_ARG_COUNT[op]
   957	        j = i + nArgs
   958	        drawFuncs[op](*points[i:j])
   959	        i = j
   960	        if op == _CLOSEPATH:
   961	            hadClosePath += 1
   962	        pop = op
   963	    if forceClose and hadMoveTo and pop!=_CLOSEPATH:
   964	        hadClosePath += 1
   965	        if active:
   966	            drawFuncs[_CLOSEPATH]()
   967	    return hadMoveTo == hadClosePath
