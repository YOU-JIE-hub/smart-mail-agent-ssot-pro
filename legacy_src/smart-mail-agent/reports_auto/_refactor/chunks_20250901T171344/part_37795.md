  3268

  3282	      change how the data is read::

  3289	        im = Image.fromarray(a, mode="RGB")

  3298	    shape = arr["shape"]
  3299	    ndim = len(shape)
  3300	    strides = arr.get("strides", None)
  3301	    if mode is None:
  3302	        try:

  3320	    else:
