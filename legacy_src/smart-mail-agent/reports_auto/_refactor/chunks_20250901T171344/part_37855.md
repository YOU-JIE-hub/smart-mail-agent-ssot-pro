   105	    if truncated_explanation[-1]:

   107	        truncated_explanation[-1] = truncated_explanation[-1] + "..."

   124	    iterated_char_count = 0
   125	    for iterated_index, input_line in enumerate(input_lines):
   126	        if iterated_char_count + len(input_line) > max_chars:
