   118	    def visit_symbol_table(self, symtab: SymbolTable, table_fullname: str) -> None:
   119	        # Copy the items because we may mutate symtab.
   120	        for key in list(symtab):
   121	            value = symtab[key]
   122	            cross_ref = value.cross_ref
   123	            if cross_ref is not None:  # Fix up cross-reference.
   124	                value.cross_ref = None
   125	                if cross_ref in self.modules:
   126	                    value.node = self.modules[cross_ref]
   127	                else:
   128	                    stnode = lookup_fully_qualified(
   129	                        cross_ref, self.modules, raise_on_missing=not self.allow_missing
   130	                    )
   131	                    if stnode is not None:
   132	                        if stnode is value:
   133	                            # The node seems to refer to itself, which can mean that
   134	                            # the target is a deleted submodule of the current module,
   135	                            # and thus lookup falls back to the symbol table of the parent
   136	                            # package. Here's how this may happen:
   137	                            #
   138	                            #   pkg/__init__.py:
   139	                            #     from pkg import sub

   142	                            # appears to refer to itself. Replace the entry with a

   144	                            # as it would stop dependency propagation.
   145	                            value.node = Var(key + "@deleted")

   147	                            assert stnode.node is not None, (table_fullname + "." + key, cross_ref)
   148	                            value.node = stnode.node
   149	                    elif not self.allow_missing:
   150	                        assert False, f"Could not find cross-ref {cross_ref}"
   151	                    else:

   373	        return node

   377	            allow_missing

     1	"""Interface for accessing the file system with automatic caching.
     2
     3	The idea is to cache the results of any file system state reads during
     4	a single transaction. This has two main benefits:
     5
     6	* This avoids redundant syscalls, as we won't perform the same OS
     7	  operations multiple times.
     8
     9	* This makes it easier to reason about concurrent FS updates, as different
    10	  operations targeting the same paths can't report different state during
    11	  a transaction.

    14
    15	Properties maintained by the API:
    16

    18	  to the reported mtime of the file, even if mtime is queried after reading
    19	  a file.
    20
    21	* Repeating an operation produces the same result as the first one during
    22	  a transaction.
    23
    24	* Call flush() to start a new transaction (flush the caches).
    25
    26	The API is a bit limited. It's easy to add new cached operations, however.
    27	Y
