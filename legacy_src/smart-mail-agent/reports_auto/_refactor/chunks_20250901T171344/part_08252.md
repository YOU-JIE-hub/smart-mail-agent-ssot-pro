   356	def construct_groups(
   357	    sources: list[BuildSource],
   358	    separate: bool | list[tuple[list[str], str | None]],
   359	    use_shared_lib: bool,
   360	    group_name_override: str | None,
   361	) -> emitmodule.Groups:
   362	    """Compute Groups given the input source list and separate configs.
   363
   364	    separate is the user-specified configuration for how to assign
   365	    modules to compilation groups (see mypycify docstring for details).
   366
   367	    This takes that and expands it into our internal representation of
   368	    group configuration, documented in mypyc.emitmodule's definition
   369	    of Group.
   370	    """
   371
   372	    if separate is True:
   373	        groups: emitmodule.Groups = [([source], None) for source in sources]
   374	    elif isinstance(separate, list):
   375	        groups = []
   376	        used_sources = set()
   377	        for files, name in separate:
   378	            group_sources = [src for src in sources if src.path in files]
   379	            groups.append((group_sources, name))
   380	            used_sources.update(group_sources)
   381	        unused_sources = [src for src in sources if src not in used_sources]
   382	        if unused_sources:
   383	            groups.extend([([source], None) for source in unused_sources])
   384	    else:
   385	        groups = [(sources, None)]
   386
   387	    # Generate missing names
   388	    for i, (group, name) in enumerate(groups):
   389	        if use_shared_lib and not name:
   390	            if group_name_override is not None:
   391	                name = group_name_override
   392	            else:
   393	                name = group_name([source.module for source in group])
   394	        groups[i] = (group, name)
   395
   396	    return groups

   416	    paths: list[str],

   499	                            (These modules must still be passed to paths.)

   502
   503	        opt_level: The optimization level, as a string. Defaults to '3' (meaning '-O3').
   504	        debug_level: The debug level, as a string. Defaults to '1' (meaning '-g1').
   505	        strip_asserts: Should asserts be stripped from the generated code.
   506
   507	        multi_file: Should each Python module be compiled into its own C source file.
   508	                    This can reduce compile time and memory requirements at the likely
   509	                    cost of runtime performance of compiled code. Defaults to false.
   510	        separate: Should compiled modules be placed in separate extension modules.
   511	                  If False, all modules are placed in a single shared library.
   512	                  If True, every module is placed in its own library.
   513	                  Otherwise separate should be a list of
   514	                  (file name list, optional shared library name) pairs specifying
   515	                  groups of files that should be placed in the same shared library
   516	                  (while all other modules will be placed in its own library).
   517
   518	                  Each group can be compiled independently, which can
   519	                  speed up compilation, but calls between groups can
   520	                  be slower than calls within a group and can't be
   521	                  inlined.
   522	        target_dir: The directory to write C output files. Defaults to 'build'.
   523	        include_runtime_files: If not None, whether the mypyc runtime library
   524	                               should be directly #include'd instead of linked
   525	                               separately in order to reduce compiler invocations.
   526	                               Defaults to False in multi_file mode, True otherwise.
   527	        strict_dunder_typing: If True, force dunder methods to have the return type
   528	                              of the method strictly, which can lead to more
   529	                              optimization opportunities. Defaults to False.
   530	        group_name: If set, override the default group name derived from
   531	                    the hash of module names. This is used for the names of the
   532	                    output C files and the shared library. This is only supported
   533	                    if there is a single group. [Experimental]
   534	    """
   535
   536	    # Figure out our configuration
   537	    compiler_options = CompilerOptions(
   538	        strip_asserts=strip_asserts,
   539	        multi_file=multi_file,
   540	        verbose=verbose,
   541	        separate=separate is not False,
   542	        target_dir=target_dir,
   543	        include_runtime_files=include_runtime_files,
   544	        strict_dunder_typing=strict_dunder_typing,
   545	        group_name=group_name,
   546	    )
   547
   548	    # Generate all the actual important C code
   549	    groups, group_cfilenames = mypyc_build(
   550	        paths,
   551	        only_compile_paths=only_compile_paths,
   552	        compiler_options=compiler_options,
   553	        separate=separate,
   554	        skip_cgen_input=skip_cgen_input,

   558	    setup_mypycify_vars()

   561	    # compiler is being used. typeshed is missing some attributes on the
   562	    # compiler object so we give it type Any
   563	    compiler: Any = ccompiler.new_compiler()
   564	    sysconfig.customize_compiler(compiler)
   565
   566	    build_dir = compiler_options.target_dir
   567

   569	    if compiler.compiler_type == "unix":
   570	        cflags += [
   571	            f"-O{opt_level}",
   572	            f"-g{debug_level}",
   573	            "-Werror",
   574	            "-Wno-unused-function",
   575	            "-Wno-unused-label",
   576	            "-Wno-unreachable-code",
   577	            "-Wno-unused-variable",
   578	            "-Wno-unused-command-line-argument",
   579	            "-Wno-unknown-warning-option",
   580	            "-Wno-unused-but-set-variable",
   581	            "-Wno-ignored-optimization-argument",
   582	            # Disables C Preprocessor (cpp) warnings
   583	            # See https://github.com/mypyc/mypyc/issues/956
   584	            "-Wno-cpp",
   585	        ]
   586	    elif compiler.compiler_type == "msvc":
   587	        # msvc doesn't have levels, '/O2' is full and '/Od' is disable
   588	        if opt_level == "0":
   589	            opt_level = "d"
   590	        elif opt_level in ("1", "2", "3"):
   591	            opt_level = "2"
   592	        if debug_level == "0":
   593	            debug_level = "NONE"
   594	        elif debug_level == "1":
   595	            debug_level = "FASTLINK"
   596	        elif debug_level in ("2", "3"):
   597	            debug_level = "FULL"
   598	        cflags += [
   599	            f"/O{opt_level}",
   600	            f"/DEBUG:{debug_level}",
   601	            "/wd4102",  # unreferenced label
   602	            "/wd4101",  # unreferenced local variable
   603	            "/wd4146",  # negating unsigned int

   613	    # compiler invocations.
   614	    shared_cfilenames = []

   626	                build_using_shared_lib(

   628	                    lib_name,
   629	                    cfilenames + shared_cfilenames,
   630	                    deps,
   631	                    build_dir,
   632	                    cflags,
   633	                )
   634	            )
   635	        else:
   636	            extensions.extend(
   637	                build_single_module(group_sources, cfilenames + shared_cfilenames, cflags)
   638	            )
   639
   640	    return extensions

     1	"""Encode valid C string literals from Python strings.
     2
     3	If a character is not allowed in C string literals, it is either emitted
     4	as a simple escape sequence (e.g. '\\n'), or an octal escape sequence
     5	with exactly three digits ('\\oXXX'). Question marks are escaped to
     6	prevent trigraphs in the string literal from being interpreted. Note
     7	that '\\?' is an invalid escape sequence in Python.
     8
     9	Consider the string literal "AB\\xCDEF". As one would expect, Python
    10	parses it as ['A', 'B', 0xCD, 'E', 'F']. However, the C standard
    11	specifies that all hexadecimal digits immediately following '\\x' will
    12	be interpreted as part of the escape sequence. Therefore, it is
    13	unexpectedly parsed as ['A', 'B', 0xCDEF].
    14
    15	Emitting ("AB\\xCD" "EF") would avoid this behaviour. However, we opt
    16	for simplicity and use octal escape sequences instead. They do not
    17	suffer from the same issue as they are defined to parse at most three
    18	octal digits.
    19	"""
    20
    21	from __future__ import annotations
    22
    23	import string
    24	from typing import Final
    25
    26	CHAR_MAP: Final = [f"\\{i:03o}" for i in range(256)]
    27
    28	# It is safe to use string.printable as it always uses the C locale.
    29	for c in string.printable:
    30	    CHAR_MAP[ord(c)] = c
    31
    32	# These assignments must come last because we prioritize simple escape
    33	# sequences over any other representation.
    34	for c in ("'", '"', "\\", "a", "b", "f", "n", "r", "t", "v"):
    35	    escaped = f"\\{c}"
    36	    decoded = escaped.encode("ascii").decode("unicode_escape")
    37	    CHAR_MAP[ord(decoded)] = escaped
    38
    39	# This escape sequence is invalid in Python.
    40	CHAR_MAP[ord("?")] = r"\?"
