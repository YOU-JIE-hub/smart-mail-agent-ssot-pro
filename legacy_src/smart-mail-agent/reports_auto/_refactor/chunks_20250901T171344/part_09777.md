   119	    def _encode_c40_char(self, char):
   120	        o = ord(char)
   121	        encoded = []
   122
   123	        if o == 32 or (o >= 48 and o <= 57) or (o >= 65 and o <= 90):
   124	            # Stay in set 0
   125	            if o == 32:
   126	                encoded.append(o - 29)
   127	            elif o >= 48 and o <= 57:
   128	                encoded.append(o - 44)
   129	            else:
   130	                encoded.append(o - 51)
   131	        elif o >= 0 and o <= 31:
   132	            encoded.append(0) # Shift to set 1
   133	            encoded.append(o)
   134	        elif (o >= 33 and o <= 64) or (o >= 91 and o <= 95):
   135	            encoded.append(1) # Shift to set 2
   136	            if o >= 33 and o <= 64:
   137	                encoded.append(o - 33)
   138	            else:
   139	                encoded.append(o - 69)
   140	        elif o >= 96 and o <= 127:
   141	            encoded.append(2) # Shift to set 3
   142	            encoded.append(o - 96)
   143	        elif o >= 128 and o <= 255:
   144	            # Extended ASCII
   145	            encoded.append(1) # Shift to set 2
   146	            encoded.append(30) # Upper shift / hibit
   147	            encoded += self._encode_c40_char(chr(o - 128))
   148	        else:
   149	            raise Exception('Cannot encode %s (%s)' % (char, o))
   150
   151	        return encoded

   153	    def _encode_c40(self, value):
   154	        encoded = []
   155
   156	        for c in value:
   157	            encoded += self._encode_c40_char(c)
   158
   159	        while len(encoded) % 3:
   160	            encoded.append(0) # Fake padding that makes chunking in the next step easier
   161
   162	        codewords = []
   163	        codewords.append(230) # Switch to C40 encoding
   164
   165	        for i in range(0, len(encoded), 3):
   166	            chunk = encoded[i:i+3]
   167	            total = chunk[0] * 1600 + chunk[1] * 40 + chunk[2] + 1
   168	            codewords.append(total // 256)
   169	            codewords.append(total % 256)
   170
   171	        codewords.append(254) # End of data
   172
   173	        if len(codewords) > self.cw_data:
   174	            raise Exception('Too much data to fit into a data matrix of this size')
   175
   176	        if len(codewords) < self.cw_data:
   177	            # Real padding
   178	            codewords.append(129) # Start padding
   179	            while len(codewords) < self.cw_data:
   180	                r = ((149 * (len(codewords) + 1)) % 253) + 1
   181	                codewords.append((129 + r) % 254)
   182
   183	        return codewords

   194	    def _get_reed_solomon_code(self, data, num_code_words):
   195	        """
   196	        This method is basically verbatim from "huBarcode" which is BSD licensed
   197	        https://github.com/hudora/huBarcode/blob/master/hubarcode/datamatrix/reedsolomon.py
   198	        """
   199	        cw_factors = FACTORS[num_code_words]
   200	        code_words = [0] * num_code_words
   201
   202	        for data_word in data:
   203	            tmp = self._gfsum(data_word, code_words[-1])
   204	            for j in range(num_code_words - 1, -1, -1):
   205	                code_words[j] = self._gfproduct(tmp, cw_factors[j])
   206	                if j > 0:
   207	                    code_words[j] = self._gfsum(code_words[j - 1], code_words[j])
   208
   209	        code_words.reverse()
   210	        return code_words

   286	    def _create_matrix(self, data):
   287	        """
   288	        This method is heavily influenced by "huBarcode" which is BSD licensed
   289	        https://github.com/hudora/huBarcode/blob/master/hubarcode/datamatrix/placement.py
   290	        """
   291	        rows = self.row_usable_modules
   292	        cols = self.col_usable_modules
   293
   294	        self._matrix = self._create_empty_matrix(rows, cols)
   295
   296	        row = 4
   297	        col = 0
   298
   299	        while True:
   300	            if row == rows and col == 0:
   301	                self._place_bit_corner_1(data)
   302	            elif row == (rows - 2) and col == 0 and (cols % 4):
   303	                self._place_bit_corner_2(data)
   304	            elif row == (rows - 2) and col == 0 and (cols % 8 == 4):
   305	                self._place_bit_corner_3(data)
   306	            elif row == (rows + 4) and col == 2 and (cols % 8 == 0):
   307	                self._place_bit_corner_4(data)
   308
   309	            while True:
   310	                if row < rows and col >= 0 and self._matrix[row][col] is None:
   311	                    self._place_bit_standard(data, row, col)
   312
   313	                row -= 2
   314	                col += 2
   315
   316	                if row < 0 or col >= cols:
   317	                    break
   318
   319	            row += 1
   320	            col += 3
   321
   322	            while True:
   323	                if row >= 0 and col < cols and self._matrix[row][col] is None:
   324	                    self._place_bit_standard(data, row, col)
   325
   326	                row += 2
   327	                col -= 2
   328
   329	                if row >= rows or col < 0:
   330	                    break
   331
   332	            row += 3
   333	            col += 1
   334
   335	            if row >= rows and col >= cols:
   336	                break
   337
   338	        for row in self._matrix:
   339	            for i in range(0, cols):
   340	                if row[i] is None:
   341	                    row[i] = 0
   342
   343	        return self._matrix

   345	    def _create_data_regions(self, matrix):
   346	        regions = []
   347	        col_offset = 0
   348	        row_offset = 0
   349
   350	        rows = int(self.row_usable_modules / self.row_regions)
   351	        cols = int(self.col_usable_modules / self.col_regions)
   352
   353	        while col_offset < self.row_regions:
   354	            while row_offset < self.col_regions:
   355	                r_offset = col_offset * rows
   356	                c_offset = row_offset * cols
   357	                region = matrix[r_offset:rows+r_offset]
   358	                for i in range(0, len(region)):
   359	                    region[i] = region[i][c_offset:cols+c_offset]
   360	                regions.append(region)
   361	                row_offset += 1
   362	            row_offset = 0
   363	            col_offset += 1
   364
   365	        return regions

   367	    def _create_empty_matrix(self, row, col):
   368	        matrix = []
   369	        for i in range(0, row):
   370	            matrix.append([None] * col)
   371	        return matrix

   373	    def _wrap_data_regions_with_finders(self, regions):
   374	        wrapped = []
   375
   376	        for region in regions:
   377	            matrix = self._create_empty_matrix(
   378	                int(self.col_modules / self.col_regions),
   379	                int(self.row_modules / self.row_regions)
   380	            )
   381
   382	            for i, rows in enumerate(region):
   383	                for j, data in enumerate(rows):
   384	                    matrix[i+1][j+1] = data
   385
   386	            for i, row in enumerate(matrix):
   387	                if i == 0:
   388	                    for j, col in enumerate(row):
   389	                        row[j] = (j + 1) % 2
   390	                elif i + 1 == len(matrix):
   391	                    for j, col in enumerate(row):
   392	                        row[j] = 1
   393	                else:
   394	                    row[0] = 1
   395	                    row[-1] = i % 2
   396
   397	            wrapped.append(matrix)
   398
   399	        return wrapped

   401	    def _merge_data_regions(self, regions):
   402	        merged = []
   403
   404	        for i in range(0, len(regions), self.row_regions):
   405	            chunk = regions[i:i+self.row_regions]
   406	            j = 0
   407	            while j < len(chunk[0]):
   408	                merged_row = []
   409	                for row in chunk:
   410	                    merged_row += row[j]
   411	                merged.append(merged_row)
   412	                j += 1
   413
   414	        return merged

   416	    def encode(self):
   417	        if hasattr(self, 'encoded'):
   418	            return self.encoded
   419
   420	        encoded = self._encode_c40(self.validated)
   421	        encoded += self._get_reed_solomon_code(encoded, self.cw_ecc)

   423	        matrix = self._create_matrix(encoded)

   425	        wrapped = self._wrap_data_regions_with_finders(data_regions)
   426	        self.encoded = self._merge_data_regions(wrapped)

   428	        self.encoded.reverse() # Helpful since PDFs start at bottom left corner

   430	        return self.encoded
