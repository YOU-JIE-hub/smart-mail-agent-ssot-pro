    15	def bezierArc(x1,y1, x2,y2, startAng=0, extent=90):
    16	    """bezierArc(x1,y1, x2,y2, startAng=0, extent=90) --> List of Bezier
    17	curve control points.
    18
    19	(x1, y1) and (x2, y2) are the corners of the enclosing rectangle.  The
    20	coordinate system has coordinates that increase to the right and down.
    21	Angles, measured in degress, start with 0 to the right (the positive X
    22	axis) and increase counter-clockwise.  The arc extends from startAng
    23	to startAng+extent.  I.e. startAng=0 and extent=180 yields an openside-down
    24	semi-circle.
    25
    26	The resulting coordinates are of the form (x1,y1, x2,y2, x3,y3, x4,y4)
    27	such that the curve goes from (x1, y1) to (x4, y4) with (x2, y2) and
    28	(x3, y3) as their respective Bezier control points."""
    29
    30	    x1,y1, x2,y2 = min(x1,x2), max(y1,y2), max(x1,x2), min(y1,y2)
    31
    32	    if abs(extent) <= 90:
    33	        arcList = [startAng]
    34	        fragAngle = float(extent)
    35	        Nfrag = 1
    36	    else:
    37	        arcList = []
    38	        Nfrag = int(ceil(abs(extent)/90.))
    39	        fragAngle = float(extent) / Nfrag
    40
    41	    x_cen = (x1+x2)/2.
    42	    y_cen = (y1+y2)/2.
    43	    rx = (x2-x1)/2.
    44	    ry = (y2-y1)/2.
    45	    halfAng = fragAngle * pi / 360.
    46	    kappa = abs(4. / 3. * (1. - cos(halfAng)) / sin(halfAng))
    47
    48	    if fragAngle < 0:
    49	        sign = -1
    50	    else:
    51	        sign = 1
    52
    53	    pointList = []
    54
    55	    for i in range(Nfrag):
    56	        theta0 = (startAng + i*fragAngle) * pi / 180.
    57	        theta1 = (startAng + (i+1)*fragAngle) *pi / 180.
    58	        if fragAngle > 0:
    59	            pointList.append((x_cen + rx * cos(theta0),
    60	                              y_cen - ry * sin(theta0),
    61	                              x_cen + rx * (cos(theta0) - kappa * sin(theta0)),
    62	                              y_cen - ry * (sin(theta0) + kappa * cos(theta0)),
    63	                              x_cen + rx * (cos(theta1) + kappa * sin(theta1)),
    64	                              y_cen - ry * (sin(theta1) - kappa * cos(theta1)),
    65	                              x_cen + rx * cos(theta1),
    66	                              y_cen - ry * sin(theta1)))
    67	        else:
    68	            pointList.append((x_cen + rx * cos(theta0),
    69	                              y_cen - ry * sin(theta0),

    71	                              y_cen - ry * (sin(theta0) - kappa * cos(theta0)),
