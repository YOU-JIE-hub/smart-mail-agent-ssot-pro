  3613	class TypeAlias(SymbolNode):
  3614	    """
  3615	    A symbol node representing a type alias.
  3616
  3617	    Type alias is a static concept, in contrast to variables with types
  3618	    like Type[...]. Namely:
  3619	        * type aliases
  3620	            - can be used in type context (annotations)
  3621	            - cannot be re-assigned
  3622	        * variables with type Type[...]
  3623	            - cannot be used in type context
  3624	            - but can be re-assigned
  3625
  3626	    An alias can be defined only by an assignment to a name (not any other lvalues).
  3627
  3628	    Such assignment defines an alias by default. To define a variable,
  3629	    an explicit Type[...] annotation is required. As an exception,
  3630	    at non-global scope non-subscripted rvalue creates a variable even without
  3631	    an annotation. This exception exists to accommodate the common use case of
  3632	    class-valued attributes. See SemanticAnalyzerPass2.check_and_set_up_type_alias
  3633	    for details.
  3634
  3635	    Aliases can be generic. We use bound type variables for generic aliases, similar
  3636	    to classes. Essentially, type aliases work as macros that expand textually.
  3637	    The definition and expansion rules are following:
  3638
  3639	        1. An alias targeting a generic class without explicit variables act as
  3640	        the given class (this doesn't apply to TypedDict, Tuple and Callable, which
  3641	        are not proper classes but special type constructors):
  3642
  3643	            A = List
  3644	            AA = List[Any]
  3645
  3646	            x: A  # same as List[Any]
  3647	            x: A[int]  # same as List[int]
  3648
  3649	            x: AA  # same as List[Any]
  3650	            x: AA[int]  # Error!
  3651
  3652	            C = Callable  # Same as Callable[..., Any]
  3653	            T = Tuple  # Same as Tuple[Any, ...]
  3654
  3655	        2. An alias using explicit type variables in its rvalue expects
  3656	        replacements (type arguments) for these variables. If missing, they
  3657	        are treated as Any, like for other generics:
  3658
  3659	            B = List[Tuple[T, T]]
  3660
  3661	            x: B  # same as List[Tuple[Any, Any]]
  3662	            x: B[int]  # same as List[Tuple[int, int]]
  3663
  3664	            def f(x: B[T]) -> T: ...  # without T, Any would be used here
  3665
  3666	        3. An alias can be defined using another aliases. In the definition
  3667	        rvalue the Any substitution doesn't happen for top level unsubscripted
  3668	        generic classes:
  3669
  3670	            A = List
  3671	            B = A  # here A is expanded to List, _not_ List[Any],
  3672	                   # to match the Python runtime behaviour
  3673	            x: B[int]  # same as List[int]
  3674	            C = List[A]  # this expands to List[List[Any]]
  3675
  3676	            AA = List[T]
  3677	            D = AA  # here AA expands to List[Any]
  3678	            x: D[int]  # Error!
  3679
  3680	    Note: the fact that we support aliases like `A = List` means that the target
  3681	    type will be initially an instance type with wrong number of type arguments.
  3682	    Such instances are all fixed either during or after main semantic analysis passes.
  3683	    We therefore store the difference between `List` and `List[Any]` rvalues (targets)
  3684	    using the `no_args` flag.
  3685
  3686	    Meaning of other fields:
  3687
  3688	    target: The target type. For generic aliases contains bound type variables
  3689	        as nested types (currently TypeVar and ParamSpec are supported).
  3690	    _fullname: Qualified name of this type alias. This is used in particular
  3691	        to track fine grained dependencies from aliases.
  3692	    alias_tvars: Type variables used to define this alias.
  3693	    normalized: Used to distinguish between `A = List`, and `A = list`. Both
  3694	        are internally stored using `builtins.list` (because `typing.List` is
  3695	        itself an alias), while the second cannot be subscripted because of
  3696	        Python runtime limitation.
  3697	    line and column: Line and column on the original alias definition.
  3698	    eager: If True, immediately expand alias when referred to (useful for aliases
  3699	        within functions that can't be looked up from the symbol table)
  3700	    """
  3701
  3702	    __slots__ = (
  3703	        "target",
  3704	        "_fullname",
  3705	        "alias_tvars",
  3706	        "no_args",
  3707	        "normalized",
  3708	        "_is_recursive",
  3709	        "eager",
  3710	        "tvar_tuple_index",
  3711	        "python_3_12_type_alias",
  3712	    )
  3713
  3714	    __match_args__ = ("name", "target", "alias_tvars", "no_args")
  3715
  3716	    def __init__(
  3717	        self,
  3718	        target: mypy.types.Type,
  3719	        fullname: str,
  3720	        line: int,
  3721	        column: int,
  3722	        *,
  3723	        alias_tvars: list[mypy.types.TypeVarLikeType] | None = None,
  3724	        no_args: bool = False,
  3725	        normalized: bool = False,
  3726	        eager: bool = False,
  3727	        python_3_12_type_alias: bool = False,
  3728	    ) -> None:
  3729	        self._fullname = fullname
  3730	        self.target = target
  3731	        if alias_tvars is None:
  3732	            alias_tvars = []
  3733	        self.alias_tvars = alias_tvars
  3734	        self.no_args = no_args
  3735	        self.normalized = normalized

  3738	        self._is_recursive: bool | None = None
  3739	        self.eager = eager
  3740	        self.python_3_12_type_alias = python_3_12_type_alias
  3741	        self.tvar_tuple_index = None

  3743	            if isinstance(t, mypy.types.TypeVarTupleType):

  3842	class PlaceholderNode(SymbolNode):
  3843	    """Temporary symbol node that will later become a real SymbolNode.
  3844
  3845	    These are only present during semantic analysis when using the new
  3846	    semantic analyzer. These are created if some essential dependencies
  3847	    of a definition are not yet complete.
  3848
  3849	    A typical use is for names imported from a module which is still
  3850	    incomplete (within an import cycle):
  3851
  3852	      from m import f  # Initially may create PlaceholderNode
  3853
  3854	    This is particularly important if the imported shadows a name from
  3855	    an enclosing scope or builtins:
  3856
  3857	      from m import int  # Placeholder avoids mixups with builtins.int
  3858
  3859	    Another case where this is useful is when there is another definition
  3860	    or assignment:
  3861
  3862	      from m import f
  3863	      def f() -> None: ...
  3864
  3865	    In the above example, the presence of PlaceholderNode allows us to
  3866	    handle the second definition as a redefinition.
  3867
  3868	    They are also used to create PlaceholderType instances for types
  3869	    that refer to incomplete types. Example:
  3870
  3871	      class C(Sequence[C]): ...
  3872
  3873	    We create a PlaceholderNode (with becomes_typeinfo=True) for C so
  3874	    that the type C in Sequence[C] can be bound.
  3875
  3876	    Attributes:
  3877
  3878	      fullname: Full name of the PlaceholderNode.
  3879	      node: AST node that contains the definition that caused this to
  3880	          be created. This is useful for tracking order of incomplete definitions
  3881	          and for debugging.
  3882	      becomes_typeinfo: If True, this refers something that could later
  3883	          become a TypeInfo. It can't be used with type variables, in
  3884	          particular, as this would cause issues with class type variable
  3885	          detection.
  3886
  3887	    The long-term purpose of placeholder nodes/types is to evolve into
  3888	    something that can support general recursive types.
  3889	    """
  3890
  3891	    __slots__ = ("_fullname", "node", "becomes_typeinfo")
  3892
  3893	    def __init__(
  3894	        self, fullname: str, node: Node, line: int, *, becomes_typeinfo: bool = False
  3895	    ) -> None:
  3896	        self._fullname = fullname
  3897	        self.node = node
  3898	        self.becomes_typeinfo = becomes_typeinfo
  3899	        self.line = line
  3900
  3901	    @property
  3902	    def name(self) -> str:
  3903	        return self._fullname.split(".")[-1]
  3904
  3905	    @property
  3906	    def fullname(self) -> str:
  3907	        return self._fullname
  3908
  3909	    def serialize(self) -> JsonDict:
  3910	        assert False, "PlaceholderNode can't be serialized"
  3911
  3912	    def accept(self, visitor: NodeVisitor[T]) -> T:
  3913	        return visitor.visit_placeholder_node(self)

  3916	class SymbolTableNode:
  3917	    """Description of a name binding in a symbol table.
  3918
  3919	    These are only used as values in module (global), function (local)
  3920	    and class symbol tables (see SymbolTable). The name that is bound is
  3921	    the key in SymbolTable.
  3922
  3923	    Symbol tables don't contain direct references to AST nodes primarily
  3924	    because there can be multiple symbol table references to a single
  3925	    AST node (due to imports and aliases), and different references can
  3926	    behave differently. This class describes the unique properties of
  3927	    each reference.
  3928
  3929	    The most fundamental attribute is 'node', which is the AST node that
  3930	    the name refers to.
  3931
  3932	    The kind is usually one of LDEF, GDEF or MDEF, depending on the scope
  3933	    of the definition. These three kinds can usually be used
  3934	    interchangeably and the difference between local, global and class
  3935	    scopes is mostly descriptive, with no semantic significance.
  3936	    However, some tools that consume mypy ASTs may care about these so
  3937	    they should be correct.
  3938

  3940	        node: AST node of definition. Among others, this can be one of
  3941	            FuncDef, Var, TypeInfo, TypeVarExpr or MypyFile -- or None
  3942	            for cross_ref that hasn't been fixed up yet.
  3943	        kind: Kind of node. Possible values:
  3944	               - LDEF: local definition
  3945	               - GDEF: global (module-level) definition
  3946	               - MDEF: class member definition
  3947	               - UNBOUND_IMPORTED: temporary kind for imported names (we
  3948	                 don't know the final kind yet)
  3949	        module_public: If False, this name won't be imported via
  3950	            'from <module> import *'. This has no effect on names within
  3951	            classes.
  3952	        module_hidden: If True, the name will be never exported (needed for
  3953	            stub files)
