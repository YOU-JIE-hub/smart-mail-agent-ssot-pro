   208	        )

   210	            return PatternType(narrowed_type, UnionType.make_union([narrowed_type, rest_type]), {})
   211	        return PatternType(narrowed_type, rest_type, {})

   228	    def visit_sequence_pattern(self, o: SequencePattern) -> PatternType:
   229	        #
   230	        # check for existence of a starred pattern
   231	        #
   232	        current_type = get_proper_type(self.type_context[-1])
   233	        if not self.can_match_sequence(current_type):
   234	            return self.early_non_match()
   235	        star_positions = [i for i, p in enumerate(o.patterns) if isinstance(p, StarredPattern)]
   236	        star_position: int | None = None
   237	        if len(star_positions) == 1:
   238	            star_position = star_positions[0]
   239	        elif len(star_positions) >= 2:
   240	            assert False, "Parser should prevent multiple starred patterns"
   241	        required_patterns = len(o.patterns)
   242	        if star_position is not None:
   243	            required_patterns -= 1
   244
   245	        #
   246	        # get inner types of original type
   247	        #
   248	        unpack_index = None
   249	        if isinstance(current_type, TupleType):
   250	            inner_types = current_type.items
   251	            unpack_index = find_unpack_in_list(inner_types)
   252	            if unpack_index is None:
   253	                size_diff = len(inner_types) - required_patterns
   254	                if size_diff < 0:
   255	                    return self.early_non_match()
   256	                elif size_diff > 0 and star_position is None:
   257	                    return self.early_non_match()
   258	            else:
   259	                normalized_inner_types = []
   260	                for it in inner_types:
   261	                    # Unfortunately, it is not possible to "split" the TypeVarTuple
   262	                    # into individual items, so we just use its upper bound for the whole
   263	                    # analysis instead.
   264	                    if isinstance(it, UnpackType) and isinstance(it.type, TypeVarTupleType):
   265	                        it = UnpackType(it.type.upper_bound)
   266	                    normalized_inner_types.append(it)
   267	                inner_types = normalized_inner_types
   268	                current_type = current_type.copy_modified(items=normalized_inner_types)
   269	                if len(inner_types) - 1 > required_patterns and star_position is None:
   270	                    return self.early_non_match()
   271	        else:
   272	            inner_type = self.get_sequence_type(current_type, o)
   273	            if inner_type is None:
   274	                inner_type = self.chk.named_type("builtins.object")
   275	            inner_types = [inner_type] * len(o.patterns)
   276
   277	        #
   278	        # match inner patterns
   279	        #
   280	        contracted_new_inner_types: list[Type] = []
   281	        contracted_rest_inner_types: list[Type] = []
   282	        captures: dict[Expression, Type] = {}
   283
   284	        contracted_inner_types = self.contract_starred_pattern_types(
   285	            inner_types, star_position, required_patterns
   286	        )
   287	        for p, t in zip(o.patterns, contracted_inner_types):
   288	            pattern_type = self.accept(p, t)
   289	            typ, rest, type_map = pattern_type
   290	            contracted_new_inner_types.append(typ)
   291	            contracted_rest_inner_types.append(rest)
   292	            self.update_type_map(captures, type_map)
   293
   294	        new_inner_types = self.expand_starred_pattern_types(
   295	            contracted_new_inner_types, star_position, len(inner_types), unpack_index is not None
   296	        )
   297	        rest_inner_types = self.expand_starred_pattern_types(
   298	            contracted_rest_inner_types, star_position, len(inner_types), unpack_index is not None
   299	        )
   300
   301	        #
   302	        # Calculate new type
   303	        #
   304	        new_type: Type
   305	        rest_type: Type = current_type
   306	        if isinstance(current_type, TupleType) and unpack_index is None:
   307	            narrowed_inner_types = []
   308	            inner_rest_types = []
   309	            for inner_type, new_inner_type in zip(inner_types, new_inner_types):
   310	                (narrowed_inner_type, inner_rest_type) = (
   311	                    self.chk.conditional_types_with_intersection(
   312	                        inner_type, [get_type_range(new_inner_type)], o, default=inner_type
   313	                    )
   314	                )
   315	                narrowed_inner_types.append(narrowed_inner_type)
   316	                inner_rest_types.append(inner_rest_type)
   317	            if all(not is_uninhabited(typ) for typ in narrowed_inner_types):
   318	                new_type = TupleType(narrowed_inner_types, current_type.partial_fallback)
   319	            else:
   320	                new_type = UninhabitedType()
   321
   322	            if all(is_uninhabited(typ) for typ in inner_rest_types):
   323	                # All subpatterns always match, so we can apply negative narrowing
   324	                rest_type = TupleType(rest_inner_types, current_type.partial_fallback)
   325	            elif sum(not is_uninhabited(typ) for typ in inner_rest_types) == 1:
   326	                # Exactly one subpattern may conditionally match, the rest always match.
   327	                # We can apply negative narrowing to this one position.
   328	                rest_type = TupleType(
   329	                    [
   330	                        curr if is_uninhabited(rest) else rest
   331	                        for curr, rest in zip(inner_types, inner_rest_types)
   332	                    ],
   333	                    current_type.partial_fallback,
   334	                )
   335	        elif isinstance(current_type, TupleType):
   336	            # For variadic tuples it is too tricky to match individual items like for fixed
   337	            # tuples, so we instead try to narrow the entire type.
   338	            # TODO: use more precise narrowing when possible (e.g. for identical shapes).
   339	            new_tuple_type = TupleType(new_inner_types, current_type.partial_fallback)
   340	            new_type, rest_type = self.chk.conditional_types_with_intersection(
   341	                new_tuple_type, [get_type_range(current_type)], o, default=new_tuple_type
   342	            )
   343	        else:
   344	            new_inner_type = UninhabitedType()
   345	            for typ in new_inner_types:
   346	                new_inner_type = join_types(new_inner_type, typ)
   347	            if isinstance(current_type, TypeVarType):
   348	                new_bound = self.narrow_sequence_child(current_type.upper_bound, new_inner_type, o)
   349	                new_type = current_type.copy_modified(upper_bound=new_bound)
   350	            else:
   351	                new_type = self.narrow_sequence_child(current_type, new_inner_type, o)
   352	        return PatternType(new_type, rest_type, captures)
   353
   354	    def get_sequence_type(self, t: Type, context: Context) -> Type | None:
   355	        t = get_proper_type(t)
   356	        if isinstance(t, AnyType):
   357	            return AnyType(TypeOfAny.from_another_any, t)
   358	        if isinstance(t, UnionType):
   359	            items = [self.get_sequence_type(item, context) for item in t.items]
   360	            not_none_items = [item for item in items if item is not None]
   361	            if not_none_items:
   362	                return make_simplified_union(not_none_items)
   363	            else:
   364	                return None
   365
   366	        if self.chk.type_is_iterable(t) and isinstance(t, (Instance, TupleType)):
   367	            if isinstance(t, TupleType):
   368	                t = tuple_fallback(t)
   369	            return self.chk.iterable_item_type(t, context)
   370	        else:
   371	            return None

   373	    def contract_starred_pattern_types(
   374	        self, types: list[Type], star_pos: int | None, num_patterns: int
   375	    ) -> list[Type]:
   376	        """
   377	        Contracts a list of types in a sequence pattern depending on the position of a starred
   378	        capture pattern.
   379
   380	        For example if the sequence pattern [a, *b, c] is matched against types [bool, int, str,
   381	        bytes] the contracted types are [bool, Union[int, str], bytes].
   382
   383	        If star_pos in None the types are returned unchanged.
   384	        """
   385	        unpack_index = find_unpack_in_list(types)
   386	        if unpack_index is not None:
   387	            # Variadic tuples require "re-shaping" to match the requested pattern.
   388	            unpack = types[unpack_index]
   389	            assert isinstance(unpack, UnpackType)
   390	            unpacked = get_proper_type(unpack.type)
   391	            # This should be guaranteed by the normalization in the caller.
   392	            assert isinstance(unpacked, Instance) and unpacked.type.fullname == "builtins.tuple"
   393	            if star_pos is None:
   394	                missing = num_patterns - len(types) + 1
   395	                new_types = types[:unpack_index]
   396	                new_types += [unpacked.args[0]] * missing
   397	                new_types += types[unpack_index + 1 :]
   398	                return new_types
   399	            prefix, middle, suffix = split_with_prefix_and_suffix(
   400	                tuple([UnpackType(unpacked) if isinstance(t, UnpackType) else t for t in types]),
   401	                star_pos,
   402	                num_patterns - star_pos,
   403	            )
   404	            new_middle = []
   405	            for m in middle:
   406	                # The existing code expects the star item type, rather than the type of
   407	                # the whole tuple "slice".
   408	                if isinstance(m, UnpackType):
   409	                    new_middle.append(unpacked.args[0])
   410	                else:
   411	                    new_middle.append(m)
   412	            return list(prefix) + [make_simplified_union(new_middle)] + list(suffix)
   413	        else:
   414	            if star_pos is None:
   415	                return types
   416	            new_types = types[:star_pos]
   417	            star_length = len(types) - num_patterns
   418	            new_types.append(make_simplified_union(types[star_pos : star_pos + star_length]))
   419	            new_types += types[star_pos + star_length :]
   420	            return new_types
   421
   422	    def expand_starred_pattern_types(
   423	        self, types: list[Type], star_pos: int | None, num_types: int, original_unpack: bool
   424	    ) -> list[Type]:
   425	        """Undoes the contraction done by contract_starred_pattern_types.
   426
   427	        For example if the sequence pattern is [a, *b, c] and types [bool, int, str] are extended
   428	        to length 4 the result is [bool, int, int, str].
   429	        """
   430	        if star_pos is None:
   431	            return types
   432	        if original_unpack:
   433	            # In the case where original tuple type has an unpack item, it is not practical
   434	            # to coerce pattern type back to the original shape (and may not even be possible),
   435	            # so we only restore the type of the star item.
   436	            res = []
   437	            for i, t in enumerate(types):
   438	                if i != star_pos:
   439	                    res.append(t)
   440	                else:
   441	                    res.append(UnpackType(self.chk.named_generic_type("builtins.tuple", [t])))
   442	            return res
   443	        new_types = types[:star_pos]
   444	        star_length = num_types - len(types) + 1
   445	        new_types += [types[star_pos]] * star_length
   446	        new_types += types[star_pos + 1 :]
   447
   448	        return new_types

   457	            new_type = outer_type
   458	        return new_type

   467	    def visit_mapping_pattern(self, o: MappingPattern) -> PatternType:
   468	        current_type = get_proper_type(self.type_context[-1])
   469	        can_match = True
   470	        captures: dict[Expression, Type] = {}
   471	        for key, value in zip(o.keys, o.values):
   472	            inner_type = self.get_mapping_item_type(o, current_type, key)
   473	            if inner_type is None:
   474	                can_match = False
   475	                inner_type = self.chk.named_type("builtins.object")
   476	            pattern_type = self.accept(value, inner_type)
   477	            if is_uninhabited(pattern_type.type):
   478	                can_match = False
   479	            else:
   480	                self.update_type_map(captures, pattern_type.captures)
   481
   482	        if o.rest is not None:
   483	            mapping = self.chk.named_type("typing.Mapping")
   484	            if is_subtype(current_type, mapping) and isinstance(current_type, Instance):
   485	                mapping_inst = map_instance_to_supertype(current_type, mapping.type)
   486	                dict_typeinfo = self.chk.lookup_typeinfo("builtins.dict")
   487	                rest_type = Instance(dict_typeinfo, mapping_inst.args)
   488	            else:
   489	                object_type = self.chk.named_type("builtins.object")
   490	                rest_type = self.chk.named_generic_type(
   491	                    "builtins.dict", [object_type, object_type]
   492	                )
   493
   494	            captures[o.rest] = rest_type
   495
   496	        if can_match:
   497	            # We can't narrow the type here, as Mapping key is invariant.
   498	            new_type = self.type_context[-1]
   499	        else:
   500	            new_type = UninhabitedType()
   501	        return PatternType(new_type, current_type, captures)

   512	                has_local_errors = local_errors.has_new_errors()

   514	            # mapping
