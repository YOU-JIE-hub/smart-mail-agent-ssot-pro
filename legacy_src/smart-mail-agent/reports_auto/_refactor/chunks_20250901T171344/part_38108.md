merate(t.items) if i != t_unpack_index]
   521	                s_other = [a for i, a in enumerate(s.items) if i != s_unpack_index]
   522	                other_joined = join_type_list(s_other + t_other)
   523	                mid_joined = join_types(mid_joined, other_joined)
   524	                return [UnpackType(s_unpacked.copy_modified(args=[mid_joined]))]
   525	            # TODO: are there other case we can handle (e.g. both prefix/suffix are shorter)?
   526	            return None
   527	        if s_unpack_index is not None:
   528	            variadic = s
   529	            unpack_index = s_unpack_index
   530	            fixed = t
   531	        else:
   532	            assert t_unpack_index is not None
   533	            variadic = t
   534	            unpack_index = t_unpack_index
   535	            fixed = s
   536	        # Case where one tuple has variadic item and the other one doesn't. The join will
   537	        # be variadic, since fixed tuple is a subtype of variadic, but not vice versa.
   538	        unpack = variadic.items[unpack_index]
   539	        assert isinstance(unpack, UnpackType)
   540	        unpacked = get_proper_type(unpack.type)
   541	        if not isinstance(unpacked, Instance):
   542	            return None
   543	        if fixed.length() < variadic.length() - 1:
   544	            # There are no non-trivial types that are supertype of both.
   545	            return None
   546	        prefix_len = unpack_index
   547	        suffix_len = variadic.length() - prefix_len - 1
   548	        prefix, middle, suffix = split_with_prefix_and_suffix(
   549	            tuple(fixed.items), prefix_len, suffix_len
   550	        )
   551	        items = []
   552	        for fi, vi in zip(prefix, variadic.items[:prefix_len]):
   553	            items.append(join_types(fi, vi))
   554	        mid_joined = join_type_list(list(middle))
   555	        mid_joined = join_types(mid_joined, unpacked.args[0])
   556	        items.append(UnpackType(unpacked.copy_modified(args=[mid_joined])))
   557	        if suffix_len:
   558	            for fi, vi in zip(suffix, variadic.items[-suffix_len:]):
   559	                items.append(join_types(fi, vi))
   560	        return items

   562	    def visit_tuple_type(self, t: TupleType) -> ProperType:
   563	        # When given two fixed-length tuples:
   564	        # * If they have the same length, join their subtypes item-wise:
   565	        #   Tuple[int, bool] + Tuple[bool, bool] becomes Tuple[int, bool]
   566	        # * If lengths do not match, return a variadic tuple:
   567	        #   Tuple[bool, int] + Tuple[bool] becomes Tuple[int, ...]
   568	        #
   569	        # Otherwise, `t` is a fixed-length tuple but `self.s` is NOT:
   570	        # * Joining with a variadic tuple returns variadic tuple:
   571	        #   Tuple[int, bool] + Tuple[bool, ...] becomes Tuple[int, ...]
   572	        # * Joining with any Sequence also returns a Sequence:
   573	        #   Tuple[int, bool] + List[bool] becomes Sequence[int]
   574	        if isinstance(self.s, TupleType):
   575	            if self.instance_joiner is None:
   576	                self.instance_joiner = InstanceJoiner()
   577	            fallback = self.instance_joiner.join_instances(
   578	                mypy.typeops.tuple_fallback(self.s), mypy.typeops.tuple_fallback(t)
   579	            )
   580	            assert isinstance(fallback, Instance)
   581	            items = self.join_tuples(self.s, t)
   582	            if items is not None:
   583	                if len(items) == 1 and isinstance(item := items[0], UnpackType):
   584	                    if isinstance(unpacked := get_proper_type(item.type), Instance):
   585	                        # Avoid double-wrapping tuple[*tuple[X, ...]]
   586	                        return unpacked
   587	                return TupleType(items, fallback)
   588	            else:
   589	                # TODO: should this be a default fallback behaviour like for meet?
   590	                if is_proper_subtype(self.s, t):
   591	                    return t
   592	                if is_proper_subtype(t, self.s):
   593	                    return self.s
   594	                return fallback
   595	        else:
   596	            return join_types(self.s, mypy.typeops.tuple_fallback(t))

   598	    def visit_typeddict_type(self, t: TypedDictType) -> ProperType:
   599	        if isinstance(self.s, TypedDictType):
   600	            items = {
   601	                item_name: s_item_type
   602	                for (item_name, s_item_type, t_item_type) in self.s.zip(t)
   603	                if (
   604	                    is_equivalent(s_item_type, t_item_type)
   605	                    and (item_name in t.required_keys) == (item_name in self.s.required_keys)
   606	                )
   607	            }
   608	            fallback = self.s.create_anonymous_fallback()
   609	            all_keys = set(items.keys())
   610	            # We need to filter by items.keys() since some required keys present in both t and
   611	            # self.s might be missing from the join if the types are incompatible.
   612	            required_keys = all_keys & t.required_keys & self.s.required_keys
   613	            # If one type has a key as readonly, we mark it as readonly for both:
   614	            readonly_keys = (t.readonly_keys | t.readonly_keys) & all_keys
   615	            return TypedDictType(items, required_keys, readonly_keys, fallback)
   616	        elif isinstance(self.s, Instance):
   617	            return join_types(self.s, t.fallback)
   618	        else:
   619	            return self.default(self.s)

   669	def is_better(t: Type, s: Type) -> bool:
   670	    # Given two possible results from join_instances_via_supertype(),
   671	    # indicate whether t is the better one.
   672	    t = get_proper_type(t)
   673	    s = get_proper_type(s)
   674
   675	    if isinstance(t, Instance):
   676	        if not isinstance(s, Instance):
   677	            return True
   678	        if t.type.is_protocol != s.type.is_protocol:
   679	            if t.type.fullname != "builtins.object" and s.type.fullname != "builtins.object":
   680	                # mro of protocol is not really relevant
   681	                return not t.type.is_protocol
   682	        # Use len(mro) as a proxy for the better choice.
   683	        if len(t.type.mro) > len(s.type.mro):
   684	            return True
   685	    return False

   726	def match_generic_callables(t: CallableType, s: CallableType) -> tuple[CallableType, CallableType]:
   727	    # The case where we combine/join/meet similar callables, situation where both are generic
   728	    # requires special care. A more principled solution may involve unify_generic_callable(),
   729	    # but it would have two problems:
   730	    #   * This adds risk of infinite recursion: e.g. join -> unification -> solver -> join
   731	    #   * Using unification is an incorrect thing for meets, as it "widens" the types
   732	    # Finally, this effectively falls back to an old behaviour before namespaces were added to
   733	    # type variables, and it worked relatively well.
   734	    max_len = max(len(t.variables), len(s.variables))
   735	    min_len = min(len(t.variables), len(s.variables))
   736	    if min_len == 0:
   737	        return t, s
   738	    new_ids = [TypeVarId.new(meta_level=0) for _ in range(max_len)]
   739	    # Note: this relies on variables being in order they appear in function definition.
   740	    return update_callable_ids(t, new_ids), update_callable_ids(s, new_ids)

   818	def combine_arg_names(
   819	    t: CallableType | Parameters, s: CallableType | Parameters
   820	) -> list[str | None]:
   821	    """Produces a list of argument names compatible with both callables.
   822
   823	    For example, suppose 't' and 's' have the following signatures:
   824
   825	    - t: (a: int, b: str, X: str) -> None
   826	    - s: (a: int, b: str, Y: str) -> None
   827
   828	    This function would return ["a", "b", None]. This information
   829	    is then used above to compute the join of t and s, which results
   830	    in a signature of (a: int, b: str, str) -> None.
   831
   832	    Note that the third argument's name is omitted and 't' and 's'
   833	    are both valid subtypes of this inferred signature.
   834
   835	    Precondition: is_similar_types(t, s) is true.
   836	    """
   837	    num_args = len(t.arg_types)
   838	    new_names = []
   839	    for i in range(num_args):
   840	        t_name = t.arg_names[i]
   841	        s_name = s.arg_names[i]
   842	        if t_name == s_name or t.arg_kinds[i].is_named() or s.arg_kinds[i].is_named():
   843	            new_names.append(t_name)
   844	        else:
   845	            new_names.append(None)
   846	    return new_names
