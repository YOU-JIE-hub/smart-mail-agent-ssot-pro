   183	                source_path = files.canonical_filename(source_path)

   185	                rel_name = filename[len(source_path)+1:]
   186	                break

   188	            rel_name = fr.relative_filename().replace("\\", "/")

   193	        package_name = dirname.replace("/", ".")

   202	        appendChild(xclass, xlines)

   204	        xclass.setAttribute("name", os.path.relpath(rel_name, dirname))
   205	        xclass.setAttribute("filename", rel_name.replace("\\", "/"))

   208	        branch_stats = analysis.branch_stats()
   209	        missing_branch_arcs = analysis.missing_branch_arcs()

   211	        # For each statement, create an XML "line" element.

   242	            class_br_hits = 0
