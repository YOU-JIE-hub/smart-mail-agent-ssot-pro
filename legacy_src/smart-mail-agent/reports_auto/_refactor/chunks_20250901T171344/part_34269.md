   621	    left_split, right_split = [], []
   622
   623	    # Get the release segment of our versions
   624	    left_split.append(list(itertools.takewhile(lambda x: x.isdigit(), left)))
   625	    right_split.append(list(itertools.takewhile(lambda x: x.isdigit(), right)))

   629	    right_split.append(right[len(right_split[0]) :])

   633	    right_split.insert(1, ["0"] * max(0, len(left_split[0]) - len(right_split[0])))

   657	        self._specs = frozenset(parsed)

   660	        # we accept prereleases or not.

   796	        # releases, and which will filter out LegacyVersion in general.
   797	        else:

   799	            found_prereleases: List[VersionTypeVar] = []

   801	            item: UnparsedVersion

   812	                if isinstance(parsed_version, LegacyVersion):
   813	                    continue

   820	                else:
   821	                    filtered.append(item)
   822
   823	            # If we've found no items except for pre-releases, then we'll go
   824	            # ahead and use the pre-releases
   825	            if not filtered and found_prereleases and prereleases is None:
   826	                return found_prereleases
   827
