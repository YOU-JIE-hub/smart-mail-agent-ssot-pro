  1059	            self_callable = name == "__call__" and isinstance(selfarg, CallableType)
  1060	            if self_callable or subtypes.is_subtype(
  1061	                dispatched_arg_type,
  1062	                # This level of erasure matches the one in checker.check_func_def(),
  1063	                # better keep these two checks consistent.
  1064	                erase_typevars(erase_to_bound(selfarg)),
  1065	                # This is to work around the fact that erased ParamSpec and TypeVarTuple
  1066	                # callables are not always compatible with non-erased ones both ways.
  1067	                always_covariant=any(
  1068	                    not isinstance(tv, TypeVarType) for tv in get_all_type_vars(selfarg)
  1069	                ),
  1070	                ignore_pos_arg_names=True,
  1071	            ):
  1072	                new_items.append(item)
  1073	            elif isinstance(selfarg, ParamSpecType):
  1074	                # TODO: This is not always right. What's the most reasonable thing to do here?
  1075	                new_items.append(item)
  1076	            elif isinstance(selfarg, TypeVarTupleType):
  1077	                raise NotImplementedError
  1078	    if not new_items:
  1079	        # Choose first item for the message (it may be not very helpful for overloads).
  1080	        msg.incompatible_self_argument(
  1081	            name, dispatched_arg_type, items[0], is_classmethod, context
  1082	        )
  1083	        return functype
  1084	    if len(new_items) == 1:
  1085	        return new_items[0]
  1086	    return Overloaded(new_items)

  1089	def analyze_class_attribute_access(
  1090	    itype: Instance,
  1091	    name: str,
  1092	    mx: MemberContext,
  1093	    *,
  1094	    mcs_fallback: Instance,
  1095	    override_info: TypeInfo | None = None,
  1096	    original_vars: Sequence[TypeVarLikeType] | None = None,
  1097	) -> Type | None:
  1098	    """Analyze access to an attribute on a class object.
  1099
  1100	    itype is the return type of the class object callable, original_type is the type
  1101	    of E in the expression E.var, original_vars are type variables of the class callable
  1102	    (for generic classes).
  1103	    """
  1104	    info = itype.type
  1105	    if override_info:
  1106	        info = override_info
  1107
  1108	    fullname = f"{info.fullname}.{name}"
  1109	    hook = mx.chk.plugin.get_class_attribute_hook(fullname)
  1110
  1111	    node = info.get(name)
  1112	    if not node:
  1113	        if itype.extra_attrs and name in itype.extra_attrs.attrs:
  1114	            # For modules use direct symbol table lookup.
  1115	            if not itype.extra_attrs.mod_name:
  1116	                return itype.extra_attrs.attrs[name]
  1117	        if info.fallback_to_any or info.meta_fallback_to_any:
  1118	            return apply_class_attr_hook(mx, hook, AnyType(TypeOfAny.special_form))
  1119	        return None
  1120
  1121	    if (
  1122	        isinstance(node.node, Var)
  1123	        and not node.node.is_classvar
  1124	        and not hook
  1125	        and mcs_fallback.type.get(name)
  1126	    ):
  1127	        # If the same attribute is declared on the metaclass and the class but with different types,
  1128	        # and the attribute on the class is not a ClassVar,
  1129	        # the type of the attribute on the metaclass should take priority
  1130	        # over the type of the attribute on the class,
  1131	        # when the attribute is being accessed from the class object itself.
  1132	        #
  1133	        # Return `None` here to signify that the name should be looked up
  1134	        # on the class object itself rather than the instance.
  1135	        return None
  1136
  1137	    mx.chk.warn_deprecated(node.node, mx.context)
  1138
  1139	    is_decorated = isinstance(node.node, Decorator)
  1140	    is_method = is_decorated or isinstance(node.node, FuncBase)
  1141	    if mx.is_lvalue and not mx.suppress_errors:
  1142	        if is_method:
  1143	            mx.msg.cant_assign_to_method(mx.context)
  1144	        if isinstance(node.node, TypeInfo):
  1145	            mx.fail(message_registry.CANNOT_ASSIGN_TO_TYPE)
  1146
  1147	    # Refuse class attribute access if slot defined
  1148	    if info.slots and name in info.slots:
  1149	        mx.fail(message_registry.CLASS_VAR_CONFLICTS_SLOTS.format(name))
  1150
  1151	    # If a final attribute was declared on `self` in `__init__`, then it
  1152	    # can't be accessed on the class object.
  1153	    if node.implicit and isinstance(node.node, Var) and node.node.is_final:
  1154	        mx.fail(message_registry.CANNOT_ACCESS_FINAL_INSTANCE_ATTR.format(node.node.name))
  1155
  1156	    # An assignment to final attribute on class object is also always an error,
  1157	    # independently of types.
  1158	    if mx.is_lvalue and not mx.chk.get_final_context():
  1159	        check_final_member(name, info, mx.msg, mx.context)
  1160
  1161	    if info.is_enum and not (mx.is_lvalue or is_decorated or is_method):
  1162	        enum_class_attribute_type = analyze_enum_class_attribute_access(itype, name, mx)
  1163	        if enum_class_attribute_type:
  1164	            return apply_class_attr_hook(mx, hook, enum_class_attribute_type)
  1165
  1166	    t = node.type
  1167	    if t:
  1168	        if isinstance(t, PartialType):
  1169	            symnode = node.node
  1170	            assert isinstance(symnode, Var)
  1171	            return apply_class_attr_hook(
  1172	                mx, hook, mx.chk.handle_partial_var_type(t, mx.is_lvalue, symnode, mx.context)
  1173	            )
  1174
  1175	        # Find the class where method/variable was defined.
  1176	        if isinstance(node.node, Decorator):
  1177	            super_info: TypeInfo | None = node.node.var.info
  1178	        elif isinstance(node.node, (Var, SYMBOL_FUNCBASE_TYPES)):
  1179	            super_info = node.node.info
  1180	        else:
  1181	            super_info = None
  1182
  1183	        # Map the type to how it would look as a defining class. For example:
  1184	        #     class C(Generic[T]): ...
  1185	        #     class D(C[Tuple[T, S]]): ...
  1186	        #     D[int, str].method()
  1187	        # Here itype is D[int, str], isuper is C[Tuple[int, str]].
  1188	        if not super_info:
  1189	            isuper = None
  1190	        else:
  1191	            isuper = map_instance_to_supertype(itype, super_info)
  1192
  1193	        if isinstance(node.node, Var):
  1194	            assert isuper is not None
  1195	            # Check if original variable type has type variables. For example:
  1196	            #     class C(Generic[T]):
  1197	            #         x: T
  1198	            #     C.x  # Error, ambiguous access
  1199	            #     C[int].x  # Also an error, since C[int] is same as C at runtime
  1200	            # Exception is Self type wrapped in ClassVar, that is safe.
  1201	            def_vars = set(node.node.info.defn.type_vars)
  1202	            if not node.node.is_classvar and node.node.info.self_type:
  1203	                def_vars.add(node.node.info.self_type)
  1204	            # TODO: should we include ParamSpec etc. here (i.e. use get_all_type_vars)?
  1205	            typ_vars = set(get_type_vars(t))
  1206	            if def_vars & typ_vars:
  1207	                # Exception: access on Type[...], including first argument of class methods is OK.
  1208	                if not isinstance(get_proper_type(mx.original_type), TypeType) or node.implicit:
  1209	                    if node.node.is_classvar:
  1210	                        message = message_registry.GENERIC_CLASS_VAR_ACCESS
  1211	                    else:
  1212	                        message = message_registry.GENERIC_INSTANCE_VAR_CLASS_ACCESS
  1213	                    mx.fail(message)
  1214	            t = expand_self_type_if_needed(t, mx, node.node, itype, is_class=True)
  1215	            # Erase non-mapped variables, but keep mapped ones, even if there is an error.
  1216	            # In the above example this means that we infer following types:
  1217	            #     C.x -> Any
  1218	            #     C[int].x -> int
  1219	            t = erase_typevars(expand_type_by_instance(t, isuper), {tv.id for tv in def_vars})
  1220
  1221	        is_classmethod = (is_decorated and cast(Decorator, node.node).func.is_class) or (
  1222	            isinstance(node.node, SYMBOL_FUNCBASE_TYPES) and node.node.is_class
  1223	        )
  1224	        t = get_proper_type(t)
  1225	        is_trivial_self = False
  1226	        if isinstance(node.node, Decorator):
  1227	            # Use fast path if there are trivial decorators like @classmethod or @property
  1228	            is_trivial_self = node.node.func.is_trivial_self and not node.node.decorators
  1229	        elif isinstance(node.node, (FuncDef, OverloadedFuncDef)):
  1230	            is_trivial_self = node.node.is_trivial_self
  1231	        if isinstance(t, FunctionLike) and is_classmethod and not is_trivial_self:
  1232	            t = check_self_arg(t, mx.self_type, False, mx.context, name, mx.msg)
  1233	        result = add_class_tvars(
  1234	            t,
  1235	            isuper,
  1236	            is_classmethod,
  1237	            mx,
  1238	            original_vars=original_vars,
  1239	            is_trivial_self=is_trivial_self,
  1240	        )
  1241	        # __set__ is not called on class objects.
  1242	        if not mx.is_lvalue:
  1243	            result = analyze_descriptor_access(result, mx)
  1244
  1245	        return apply_class_attr_hook(mx, hook, result)
  1246	    elif isinstance(node.node, Var):
  1247	        mx.not_ready_callback(name, mx.context)
  1248	        return AnyType(TypeOfAny.special_form)
  1249
  1250	    if isinstance(node.node, (TypeInfo, TypeAlias, MypyFile, TypeVarLikeExpr)):
  1251	        # TODO: should we apply class plugin here (similar to instance access)?
  1252	        return mx.chk.expr_checker.analyze_static_reference(node.node, mx.context, mx.is_lvalue)
  1253
  1254	    if is_decorated:
  1255	        assert isinstance(node.node, Decorator)
  1256	        if node.node.type:
  1257	            return apply_class_attr_hook(mx, hook, node.node.type)
  1258	        else:
  1259	            mx.not_ready_callback(name, mx.context)
  1260	            return AnyType(TypeOfAny.from_error)
  1261	    else:
  1262	        assert isinstance(node.node, SYMBOL_FUNCBASE_TYPES)
  1263	        typ = function_type(node.node, mx.named_type("builtins.function"))
  1264	        # Note: if we are accessing class method on class object, the cls argument is bound.
  1265	        # Annotated and/or explicit class methods go through other code paths above, for
  1266	        # unannotated implicit class methods we do this here.
  1267	        if node.node.is_class:
  1268	            typ = bind_self_fast(typ)
  1269	        return apply_class_attr_hook(mx, hook, typ)

  1348	def add_class_tvars(
  1349	    t: ProperType,
  1350	    isuper: Instance | None,
  1351	    is_classmethod: bool,
  1352	    mx: MemberContext,
  1353	    original_vars: Sequence[TypeVarLikeType] | None = None,
  1354	    is_trivial_self: bool = False,
  1355	) -> Type:
  1356	    """Instantiate type variables during analyze_class_attribute_access,
  1357	    e.g T and Q in the following:
  1358
  1359	    class A(Generic[T]):
  1360	        @classmethod
  1361	        def foo(cls: Type[Q]) -> Tuple[T, Q]: ...
  1362
  1363	    class B(A[str]): pass
  1364	    B.foo()
  1365
  1366	    Args:
  1367	        t: Declared type of the method (or property)
  1368	        isuper: Current instance mapped to the superclass where method was defined, this
  1369	            is usually done by map_instance_to_supertype()
  1370	        is_classmethod: True if this method is decorated with @classmethod
  1371	        original_vars: Type variables of the class callable on which the method was accessed
  1372	        is_trivial_self: if True, we can use fast path for bind_self().
  1373	    Returns:
  1374	        Expanded method type with added type variables (when needed).
  1375	    """
  1376	    # TODO: verify consistency between Q and T
  1377
  1378	    # We add class type variables if the class method is accessed on class object
  1379	    # without applied type arguments, this matches the behavior of __init__().
  1380	    # For example (continuing the example in docstring):
  1381	    #     A       # The type of callable is def [T] () -> A[T], _not_ def () -> A[Any]
  1382	    #     A[int]  # The type of callable is def () -> A[int]
  1383	    # and
  1384	    #     A.foo       # The type is generic def [T] () -> Tuple[T, A[T]]
  1385	    #     A[int].foo  # The type is non-generic def () -> Tuple[int, A[int]]
  1386	    #
  1387	    # This behaviour is useful for defining alternative constructors for generic classes.
  1388	    # To achieve such behaviour, we add the class type variables that are still free
  1389	    # (i.e. appear in the return type of the class object on which the method was accessed).
  1390	    if isinstance(t, CallableType):
  1391	        tvars = original_vars if original_vars is not None else []
  1392	        if not mx.preserve_type_var_ids:
  1393	            t = freshen_all_functions_type_vars(t)
  1394	        if is_classmethod:
  1395	            if is_trivial_self:
  1396	                t = bind_self_fast(t, mx.self_type)
  1397	            else:
  1398	                t = bind_self(t, mx.self_type, is_classmethod=True)
  1399	        if isuper is not None:
  1400	            t = expand_type_by_instance(t, isuper)
  1401	        freeze_all_type_vars(t)
  1402	        return t.copy_modified(variables=list(tvars) + list(t.variables))
  1403	    elif isinstance(t, Overloaded):
  1404	        return Overloaded(
  1405	            [
  1406	                cast(
  1407	                    CallableType,
  1408	                    add_class_tvars(item, isuper, is_classmethod, mx, original_vars=original_vars),
  1409	                )
  1410	                for item in t.items
  1411	            ]
  1412	        )
  1413	    if isuper is not None:
  1414	        t = expand_type_by_instance(t, isuper)
  1415	    return t
  1416

  1444	def bind_self_fast(method: F, original_type: Type | None = None) -> F:
  1445	    """Return a copy of `method`, with the type of its first parameter (usually
  1446	    self or cls) bound to original_type.
  1447
  1448	    This is a faster version of mypy.typeops.bind_self() that can be used for methods
  1449	    with trivial self/cls annotations.
  1450	    """
  1451	    if isinstance(method, Overloaded):
  1452	        items = [bind_self_fast(c, original_type) for c in method.items]
  1453	        return cast(F, Overloaded(items))
  1454	    assert isinstance(method, CallableType)
  1455	    func: CallableType = method
  1456	    if not func.arg_types:
  1457	        # Invalid method, return something.
  1458	        return method
  1459	    if func.arg_kinds[0] in (ARG_STAR, ARG_STAR2):
  1460	        # See typeops.py for details.
  1461	        return method
  1462	    original_type = get_proper_type(original_type)
  1463	    if isinstance(original_type, CallableType) and original_type.is_type_obj():
  1464	        original_type = TypeType.make_normalized(original_type.ret_type)
  1465	    res = func.copy_modified(
  1466	        arg_types=func.arg_types[1:],
  1467	        arg_kinds=func.arg_kinds[1:],
  1468	        arg_names=func.arg_names[1:],
  1469	        is_bound=True,
  1470	    )
  1471	    return cast(
