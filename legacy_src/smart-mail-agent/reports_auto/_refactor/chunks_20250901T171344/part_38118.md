   231	    def report(
   232	        self,
   233	        msg: str,
   234	        context: Context | None,
   235	        severity: str,
   236	        *,
   237	        code: ErrorCode | None = None,
   238	        file: str | None = None,
   239	        origin: Context | None = None,
   240	        offset: int = 0,
   241	        secondary_context: Context | None = None,
   242	        parent_error: ErrorInfo | None = None,
   243	    ) -> ErrorInfo:
   244	        """Report an error or note (unless disabled).
   245
   246	        Note that context controls where error is reported, while origin controls
   247	        where # type: ignore comments have effect.
   248	        """
   249
   250	        def span_from_context(ctx: Context) -> Iterable[int]:
   251	            """This determines where a type: ignore for a given context has effect.
   252
   253	            Current logic is a bit tricky, to keep as much backwards compatibility as
   254	            possible. We may reconsider this to always be a single line (or otherwise
   255	            simplify it) when we drop Python 3.7.
   256
   257	            TODO: address this in follow up PR
   258	            """
   259	            if isinstance(ctx, (ClassDef, FuncDef)):
   260	                return range(ctx.line, ctx.line + 1)
   261	            elif not isinstance(ctx, Expression):
   262	                return [ctx.line]
   263	            else:
   264	                return range(ctx.line, (ctx.end_line or ctx.line) + 1)
   265
   266	        origin_span: Iterable[int] | None
   267	        if origin is not None:
   268	            origin_span = span_from_context(origin)
   269	        elif context is not None:
   270	            origin_span = span_from_context(context)
   271	        else:
   272	            origin_span = None
   273
   274	        if secondary_context is not None:
   275	            assert origin_span is not None
   276	            origin_span = itertools.chain(origin_span, span_from_context(secondary_context))
   277
   278	        return self.errors.report(
   279	            context.line if context else -1,
   280	            context.column if context else -1,
   281	            msg,
   282	            severity=severity,
   283	            file=file,
   284	            offset=offset,
   285	            origin_span=origin_span,
   286	            end_line=context.end_line if context else -1,
   287	            end_column=context.end_column if context else -1,
   288	            code=code,
   289	            parent_error=parent_error,
   290	        )

   361	    def has_no_attr(
   362	        self,
   363	        original_type: Type,
   364	        typ: Type,
   365	        member: str,
   366	        context: Context,
   367	        module_symbol_table: SymbolTable | None = None,
   368	    ) -> ErrorCode | None:
   369	        """Report a missing or non-accessible member.
   370
   371	        original_type is the top-level type on which the error occurred.
   372	        typ is the actual type that is missing the member. These can be
   373	        different, e.g., in a union, original_type will be the union and typ
   374	        will be the specific item in the union that does not have the member
   375	        attribute.
   376
   377	        'module_symbol_table' is passed to this function if the type for which we
   378	        are trying to get a member was originally a module. The SymbolTable allows
   379	        us to look up and suggests attributes of the module since they are not
   380	        directly available on original_type
   381
   382	        If member corresponds to an operator, use the corresponding operator
   383	        name in the messages. Return the error code that was produced, if any.
   384	        """
   385	        original_type = get_proper_type(original_type)
   386	        typ = get_proper_type(typ)
   387
   388	        if isinstance(original_type, Instance) and original_type.type.has_readable_member(member):
   389	            self.fail(f'Member "{member}" is not assignable', context)
   390	            return None
   391	        elif member == "__contains__":
   392	            self.fail(
   393	                f"Unsupported right operand type for in ({format_type(original_type, self.options)})",
   394	                context,
   395	                code=codes.OPERATOR,
   396	            )
   397	            return codes.OPERATOR
   398	        elif member in op_methods.values():
   399	            # Access to a binary operator member (e.g. _add). This case does
   400	            # not handle indexing operations.
   401	            for op, method in op_methods.items():
   402	                if method == member:
   403	                    self.unsupported_left_operand(op, original_type, context)
   404	                    return codes.OPERATOR
   405	        elif member == "__neg__":
   406	            self.fail(
   407	                f"Unsupported operand type for unary - ({format_type(original_type, self.options)})",
   408	                context,
   409	                code=codes.OPERATOR,
   410	            )
   411	            return codes.OPERATOR
   412	        elif member == "__pos__":
   413	            self.fail(
   414	                f"Unsupported operand type for unary + ({format_type(original_type, self.options)})",
   415	                context,
   416	                code=codes.OPERATOR,
   417	            )
   418	            return codes.OPERATOR
   419	        elif member == "__invert__":
   420	            self.fail(
   421	                f"Unsupported operand type for ~ ({format_type(original_type, self.options)})",
   422	                context,
   423	                code=codes.OPERATOR,
   424	            )
   425	            return codes.OPERATOR
   426	        elif member == "__getitem__":
   427	            # Indexed get.
   428	            # TODO: Fix this consistently in format_type
   429	            if isinstance(original_type, FunctionLike) and original_type.is_type_obj():
   430	                self.fail(
   431	                    "The type {} is not generic and not indexable".format(
   432	                        format_type(original_type, self.options)
   433	                    ),
   434	                    context,
   435	                )
   436	                return None
   437	            else:
   438	                self.fail(
   439	                    f"Value of type {format_type(original_type, self.options)} is not indexable",
   440	                    context,
   441	                    code=codes.INDEX,
   442	                )
   443	                return codes.INDEX
   444	        elif member == "__setitem__":
   445	            # Indexed set.
   446	            self.fail(
   447	                "Unsupported target for indexed assignment ({})".format(
   448	                    format_type(original_type, self.options)
   449	                ),
   450	                context,
   451	                code=codes.INDEX,
   452	            )
   453	            return codes.INDEX
   454	        elif member == "__call__":
   455	            if isinstance(original_type, Instance) and (
   456	                original_type.type.fullname == "builtins.function"
   457	            ):
   458	                # "'function' not callable" is a confusing error message.
   459	                # Explain that the problem is that the type of the function is not known.
   460	                self.fail("Cannot call function of unknown type", context, code=codes.OPERATOR)
   461	                return codes.OPERATOR
   462	            else:
   463	                self.fail(
   464	                    message_registry.NOT_CALLABLE.format(format_type(original_type, self.options)),
   465	                    context,
   466	                    code=codes.OPERATOR,
   467	                )
   468	                return codes.OPERATOR
   469	        else:
   470	            # The non-special case: a missing ordinary attribute.
   471	            extra = ""
   472	            if member == "__iter__":
   473	                extra = " (not iterable)"
   474	            elif member == "__aiter__":
   475	                extra = " (not async iterable)"
   476	            if not self.are_type_names_disabled():
   477	                failed = False
   478	                if isinstance(original_type, Instance) and original_type.type.names:
   479	                    if (
   480	                        module_symbol_table is not None
   481	                        and member in module_symbol_table
   482	                        and not module_symbol_table[member].module_public
   483	                    ):
   484	                        self.fail(
   485	                            f"{format_type(original_type, self.options, module_names=True)} does not "
   486	                            f'explicitly export attribute "{member}"',
   487	                            context,
   488	                            code=codes.ATTR_DEFINED,
   489	                        )
   490	                        failed = True
   491	                    else:
   492	                        alternatives = set(original_type.type.names.keys())
   493	                        if module_symbol_table is not None:
   494	                            alternatives |= {
   495	                                k for k, v in module_symbol_table.items() if v.module_public
   496	                            }
   497	                        # Rare but possible, see e.g. testNewAnalyzerCyclicDefinitionCrossModule
   498	                        alternatives.discard(member)
   499
   500	                        matches = [m for m in COMMON_MISTAKES.get(member, []) if m in alternatives]
   501	                        matches.extend(best_matches(member, alternatives, n=3))
   502	                        if member == "__aiter__" and matches == ["__iter__"]:
   503	                            matches = []  # Avoid misleading suggestion
   504	                        if matches:
   505	                            self.fail(
   506	                                '{} has no attribute "{}"; maybe {}?{}'.format(
   507	                                    format_type(original_type, self.options),
   508	                                    member,
   509	                                    pretty_seq(matches, "or"),
   510	                                    extra,
   511	                                ),
   512	                                context,
   513	                                code=codes.ATTR_DEFINED,
   514	                            )
   515	                            failed = True
   516	                if not failed:
   517	                    self.fail(
   518	                        '{} has no attribute "{}"{}'.format(
   519	                            format_type(original_type, self.options), member, extra
   520	                        ),
   521	                        context,
   522	                        code=codes.ATTR_DEFINED,
   523	                    )
   524	                return codes.ATTR_DEFINED
   525	            elif isinstance(original_type, UnionType):
   526	                # The checker passes "object" in lieu of "None" for attribute
   527	                # checks, so we manually convert it back.
   528	                typ_format, orig_type_format = format_type_distinctly(
   529	                    typ, original_type, options=self.options
   530	                )
   531	                if typ_format == '"object"' and any(
   532	                    type(item) == NoneType for item in original_type.items
   533	                ):
   534	                    typ_format = '"None"'
   535	                self.fail(
   536	                    'Item {} of {} has no attribute "{}"{}'.format(
   537	                        typ_format, orig_type_format, member, extra
   538	                    ),
   539	                    context,
   540	                    code=codes.UNION_ATTR,
   541	                )
   542	                return codes.UNION_ATTR
   543	            elif isinstance(original_type, TypeVarType):
   544	                bound = get_proper_type(original_type.upper_bound)
   545	                if isinstance(bound, UnionType):
   546	                    typ_fmt, bound_fmt = format_type_distinctly(typ, bound, options=self.options)
   547	                    original_type_fmt = format_type(original_type, self.options)
   548	                    self.fail(
   549	                        "Item {} of the upper bound {} of type variable {} has no "
   550	                        'attribute "{}"{}'.format(
   551	                            typ_fmt, bound_fmt, original_type_fmt, member, extra
   552	                        ),
   553	                        context,
   554	                        code=codes.UNION_ATTR,
   555	                    )
   556	                    return codes.UNION_ATTR
   557	            else:
   558	                self.fail(
   559	                    '{} has no attribute "{}"{}'.format(
   560	                        format_type(original_type, self.options), member, extra
   561	                    ),
   562	                    context,
   563	                    code=codes.ATTR_DEFINED,
   564	                )
   565	                return codes.ATTR_DEFINED
   566	        return None

   569	        self,
   570	        op: str,
   571	        left_type: Any,
   572	        right_type: Any,
