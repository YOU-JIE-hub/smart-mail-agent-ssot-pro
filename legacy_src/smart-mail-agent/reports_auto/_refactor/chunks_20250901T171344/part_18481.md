  7093	        self, escape_comprehensions: bool = False, type_param: bool = False
  7094	    ) -> SymbolTable:

  7111	                            # The caller of the comprehension is in the global space.

  7116	                                names_candidate is not None

  7223	    def in_checked_function(self) -> bool:
  7224	        """Should we type-check the current function?
  7225
  7226	        - Yes if --check-untyped-defs is set.
  7227	        - Yes outside functions.
  7228	        - Yes in annotated functions.
  7229	        - No otherwise.
  7230	        """
  7231	        if self.options.check_untyped_defs or not self.function_stack:
  7232	            return True
  7233
  7234	        current_index = len(self.function_stack) - 1
  7235	        while current_index >= 0:
  7236	            current_func = self.function_stack[current_index]
  7237	            if not isinstance(current_func, LambdaExpr):
  7238	                return not current_func.is_dynamic()
  7239
  7240	            # Special case, `lambda` inherits the "checked" state from its parent.
  7241	            # Because `lambda` itself cannot be annotated.
  7242	            # `lambdas` can be deeply nested, so we try to find at least one other parent.
  7243	            current_index -= 1
  7244
  7245	        # This means that we only have a stack of `lambda` functions,
  7246	        # no regular functions.
  7247	        return True
