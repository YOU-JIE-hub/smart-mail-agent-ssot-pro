  1861	        """Is the given attribute a method with a None-compatible return type?
  1862

  1884	    def analyze_type_type_callee(self, item: ProperType, context: Context) -> Type:
  1885	        """Analyze the callee X in X(...) where X is Type[item].
  1886
  1887	        Return a Y that we can pass to check_call(Y, ...).
  1888	        """
  1889	        if isinstance(item, AnyType):
  1890	            return AnyType(TypeOfAny.from_another_any, source_any=item)
  1891	        if isinstance(item, Instance):
  1892	            res = type_object_type(item.type, self.named_type)
  1893	            if isinstance(res, CallableType):
  1894	                res = res.copy_modified(from_type_type=True)
  1895	            expanded = expand_type_by_instance(res, item)
  1896	            if isinstance(expanded, CallableType):
  1897	                # Callee of the form Type[...] should never be generic, only
  1898	                # proper class objects can be.
  1899	                expanded = expanded.copy_modified(variables=[])
  1900	            return expanded
  1901	        if isinstance(item, UnionType):
  1902	            return UnionType(
  1903	                [
  1904	                    self.analyze_type_type_callee(get_proper_type(tp), context)
  1905	                    for tp in item.relevant_items()
  1906	                ],
  1907	                item.line,
  1908	            )
  1909	        if isinstance(item, TypeVarType):
  1910	            # Pretend we're calling the typevar's upper bound,
  1911	            # i.e. its constructor (a poor approximation for reality,
  1912	            # but better than AnyType...), but replace the return type
  1913	            # with typevar.
  1914	            callee = self.analyze_type_type_callee(get_proper_type(item.upper_bound), context)
  1915	            callee = get_proper_type(callee)
  1916	            if isinstance(callee, CallableType):
  1917	                callee = callee.copy_modified(ret_type=item)
  1918	            elif isinstance(callee, Overloaded):
  1919	                callee = Overloaded([c.copy_modified(ret_type=item) for c in callee.items])
  1920	            return callee
  1921	        # We support Type of namedtuples but not of tuples in general
  1922	        if isinstance(item, TupleType) and tuple_fallback(item).type.fullname != "builtins.tuple":
  1923	            return self.analyze_type_type_callee(tuple_fallback(item), context)
  1924	        if isinstance(item, TypedDictType):
  1925	            return self.typeddict_callable_from_context(item)
  1926
  1927	        self.msg.unsupported_type_type(item, context)
  1928	        return AnyType(TypeOfAny.from_error)

  1962	    def infer_arg_types_in_context(
  1963	        self,
  1964	        callee: CallableType,
  1965	        args: list[Expression],
  1966	        arg_kinds: list[ArgKind],
  1967	        formal_to_actual: list[list[int]],
  1968	    ) -> list[Type]:
  1969	        """Infer argument expression types using a callable type as context.
  1970
  1971	        For example, if callee argument 2 has type List[int], infer the
  1972	        argument expression with List[int] type context.
  1973
  1974	        Returns the inferred types of *actual arguments*.
  1975	        """
  1976	        res: list[Type | None] = [None] * len(args)
  1977
  1978	        for i, actuals in enumerate(formal_to_actual):
  1979	            for ai in actuals:
  1980	                if not arg_kinds[ai].is_star():
  1981	                    arg_type = callee.arg_types[i]
  1982	                    # When the outer context for a function call is known to be recursive,
  1983	                    # we solve type constraints inferred from arguments using unions instead
  1984	                    # of joins. This is a bit arbitrary, but in practice it works for most
  1985	                    # cases. A cleaner alternative would be to switch to single bin type
  1986	                    # inference, but this is a lot of work.
  1987	                    old = self.infer_more_unions_for_recursive_type(arg_type)
  1988	                    res[ai] = self.accept(args[ai], arg_type)
  1989	                    # We need to manually restore union inference state, ugh.
  1990	                    type_state.infer_unions = old
  1991
  1992	        # Fill in the rest of the argument types.
  1993	        for i, t in enumerate(res):
  1994	            if not t:
  1995	                res[i] = self.accept(args[i])
  1996	        assert all(tp is not None for tp in res)
  1997	        return cast(list[Type], res)

  1999	    def infer_function_type_arguments_using_context(
  2000	        self, callable: CallableType, error_context: Context
  2001	    ) -> CallableType:
  2002	        """Unify callable return type to type context to infer type vars.
  2003
  2004	        For example, if the return type is set[t] where 't' is a type variable
  2005	        of callable, and if the context is set[int], return callable modified
  2006	        by substituting 't' with 'int'.
  2007	        """
  2008	        ctx = self.type_context[-1]
  2009	        if not ctx:
  2010	            return callable
  2011	        # The return type may have references to type metavariables that
  2012	        # we are inferring right now. We must consider them as indeterminate
  2013	        # and they are not potential results; thus we replace them with the
  2014	        # special ErasedType type. On the other hand, class type variables are
  2015	        # valid results.
  2016	        erased_ctx = replace_meta_vars(ctx, ErasedType())
  2017	        ret_type = callable.ret_type
  2018	        if is_overlapping_none(ret_type) and is_overlapping_none(ctx):
  2019	            # If both the context and the return type are optional, unwrap the optional,
  2020	            # since in 99% cases this is what a user expects. In other words, we replace
  2021	            #     Optional[T] <: Optional[int]
  2022	            # with
  2023	            #     T <: int
  2024	            # while the former would infer T <: Optional[int].
  2025	            ret_type = remove_optional(ret_type)
  2026	            erased_ctx = remove_optional(erased_ctx)
  2027	            #
  2028	            # TODO: Instead of this hack and the one below, we need to use outer and
  2029	            # inner contexts at the same time. This is however not easy because of two
  2030	            # reasons:
  2031	            #   * We need to support constraints like [1 <: 2, 2 <: X], i.e. with variables
  2032	            #     on both sides. (This is not too hard.)
  2033	            #   * We need to update all the inference "infrastructure", so that all
  2034	            #     variables in an expression are inferred at the same time.
  2035	            #     (And this is hard, also we need to be careful with lambdas that require
  2036	            #     two passes.)
  2037	        proper_ret = get_proper_type(ret_type)
  2038	        if (
  2039	            isinstance(proper_ret, TypeVarType)
  2040	            or isinstance(proper_ret, UnionType)
  2041	            and all(isinstance(get_proper_type(u), TypeVarType) for u in proper_ret.items)
  2042	        ):
  2043	            # Another special case: the return type is a type variable. If it's unrestricted,
  2044	            # we could infer a too general type for the type variable if we use context,
  2045	            # and this could result in confusing and spurious type errors elsewhere.
  2046	            #
  2047	            # So we give up and just use function arguments for type inference, with just two
  2048	            # exceptions:
  2049	            #
  2050	            # 1. If the context is a generic instance type, actually use it as context, as
  2051	            #    this *seems* to usually be the reasonable thing to do.
  2052	            #
  2053	            #    See also github issues #462 and #360.
  2054	            #
  2055	            # 2. If the context is some literal type, we want to "propagate" that information
  2056	            #    down so that we infer a more precise type for literal expressions. For example,
  2057	            #    the expression `3` normally has an inferred type of `builtins.int`: but if it's
  2058	            #    in a literal context like below, we want it to infer `Literal[3]` instead.
  2059	            #
  2060	            #        def expects_literal(x: Literal[3]) -> None: pass
  2061	            #        def identity(x: T) -> T: return x
  2062	            #
  2063	            #        expects_literal(identity(3))  # Should type-check
  2064	            # TODO: we may want to add similar exception if all arguments are lambdas, since
  2065	            # in this case external context is almost everything we have.
  2066	            if not is_generic_instance(ctx) and not is_literal_type_like(ctx):
  2067	                return callable.copy_modified()
  2068	        args = infer_type_arguments(
  2069	            callable.variables, ret_type, erased_ctx, skip_unsatisfied=True
  2070	        )
  2071	        # Only substitute non-Uninhabited and non-erased types.
  2072	        new_args: list[Type | None] = []
  2073	        for arg in args:
  2074	            if has_uninhabited_component(arg) or has_erased_component(arg):
  2075	                new_args.append(None)
  2076	            else:
  2077	                new_args.append(arg)
  2078	        # Don't show errors after we have only used the outer context for inference.
  2079	        # We will use argument context to infer more variables.
  2080	        return self.apply_generic_arguments(
  2081	            callable, new_args, error_context, skip_unsatisfied=True
  2082	        )
