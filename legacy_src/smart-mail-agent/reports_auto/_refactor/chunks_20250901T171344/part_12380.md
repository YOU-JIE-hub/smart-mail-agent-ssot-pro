  1003	        # For a fixture found in a Collector's object (e.g. a `Module`s module,
  1004	        # a `Class`'s class), the baseid is the Collector's nodeid.
  1005	        #
  1006	        # For a fixture found in a conftest plugin, the baseid is the conftest's
  1007	        # directory path relative to the rootdir.
  1008	        #
  1009	        # For other plugins, the baseid is the empty string (always matches).
  1010	        self.baseid: Final = baseid or ""
  1011	        # Whether the fixture was found from a node or a conftest in the
  1012	        # collection tree. Will be false for fixtures defined in non-conftest
  1013	        # plugins.
  1014	        self.has_location: Final = baseid is not None
  1015	        # The fixture factory function.
  1016	        self.func: Final = func
  1017	        # The name by which the fixture may be requested.
  1018	        self.argname: Final = argname
  1019	        if scope is None:
  1020	            scope = Scope.Function
  1021	        elif callable(scope):
  1022	            scope = _eval_scope_callable(scope, argname, config)
  1023	        if isinstance(scope, str):
  1024	            scope = Scope.from_user(
  1025	                scope, descr=f"Fixture '{func.__name__}'", where=baseid
  1026	            )
  1027	        self._scope: Final = scope
  1028	        # If the fixture is directly parametrized, the parameter values.
  1029	        self.params: Final = params
  1030	        # If the fixture is directly parametrized, a tuple of explicit IDs to
  1031	        # assign to the parameter values, or a callable to generate an ID given
  1032	        # a parameter value.
  1033	        self.ids: Final = ids
  1034	        # The names requested by the fixtures.
  1035	        self.argnames: Final = getfuncargnames(func, name=argname)
  1036	        # If the fixture was executed, the current value of the fixture.
  1037	        # Can change if the fixture is executed with different parameters.
  1038	        self.cached_result: _FixtureCachedResult[FixtureValue] | None = None
  1039	        self._finalizers: Final[list[Callable[[], object]]] = []
  1040
  1041	        # only used to emit a deprecationwarning, can be removed in pytest9
  1042	        self._autouse = _autouse
  1043
  1044	    @property
  1045	    def scope(self) -> _ScopeName:
  1046	        """Scope string, one of "function", "class", "module", "package", "session"."""
  1047	        return self._scope.value
  1048
  1049	    def addfinalizer(self, finalizer: Callable[[], object]) -> None:
  1050	        self._finalizers.append(finalizer)
  1051
  1052	    def finish(self, request: SubRequest) -> None:
  1053	        exceptions: list[BaseException] = []
  1054	        while self._finalizers:
  1055	            fin = self._finalizers.pop()
  1056	            try:
  1057	                fin()
  1058	            except BaseException as e:
  1059	                exceptions.append(e)
  1060	        node = request.node
  1061	        node.ihook.pytest_fixture_post_finalizer(fixturedef=self, request=request)
  1062	        # Even if finalization fails, we invalidate the cached fixture
  1063	        # value and remove all finalizers because they may be bound methods
  1064	        # which will keep instances alive.
  1065	        self.cached_result = None
  1066	        self._finalizers.clear()
  1067	        if len(exceptions) == 1:
  1068	            raise exceptions[0]
  1069	        elif len(exceptions) > 1:
  1070	            msg = f'errors while tearing down fixture "{self.argname}" of {node}'
  1071	            raise BaseExceptionGroup(msg, exceptions[::-1])
  1072
  1073	    def execute(self, request: SubRequest) -> FixtureValue:
  1074	        """Return the value of this fixture, executing it if not cached."""
  1075	        # Ensure that the dependent fixtures requested by this fixture are loaded.
  1076	        # This needs to be done before checking if we have a cached value, since
  1077	        # if a dependent fixture has their cache invalidated, e.g. due to
  1078	        # parametrization, they finalize themselves and fixtures depending on it
  1079	        # (which will likely include this fixture) setting `self.cached_result = None`.
  1080	        # See #4871
  1081	        requested_fixtures_that_should_finalize_us = []
  1082	        for argname in self.argnames:
  1083	            fixturedef = request._get_active_fixturedef(argname)
  1084	            # Saves requested fixtures in a list so we later can add our finalizer
  1085	            # to them, ensuring that if a requested fixture gets torn down we get torn
  1086	            # down first. This is generally handled by SetupState, but still currently
  1087	            # needed when this fixture is not parametrized but depends on a parametrized
  1088	            # fixture.
  1089	            if not isinstance(fixturedef, PseudoFixtureDef):
  1090	                requested_fixtures_that_should_finalize_us.append(fixturedef)
  1091
  1092	        # Check for (and return) cached value/exception.
  1093	        if self.cached_result is not None:
  1094	            request_cache_key = self.cache_key(request)
  1095	            cache_key = self.cached_result[1]
  1096	            try:
  1097	                # Attempt to make a normal == check: this might fail for objects
  1098	                # which do not implement the standard comparison (like numpy arrays -- #6497).
  1099	                cache_hit = bool(request_cache_key == cache_key)
  1100	            except (ValueError, RuntimeError):
  1101	                # If the comparison raises, use 'is' as fallback.
  1102	                cache_hit = request_cache_key is cache_key
  1103
  1104	            if cache_hit:
  1105	                if self.cached_result[2] is not None:
  1106	                    exc, exc_tb = self.cached_result[2]
  1107	                    raise exc.with_traceback(exc_tb)
  1108	                else:
  1109	                    result = self.cached_result[0]
  1110	                    return result
  1111	            # We have a previous but differently parametrized fixture instance
  1112	            # so we need to tear it down before creating a new one.
  1113	            self.finish(request)
  1114	            assert self.cached_result is None
  1115
  1116	        # Add finalizer to requested fixtures we saved previously.
  1117	        # We make sure to do this after checking for cached value to avoid
  1118	        # adding our finalizer multiple times. (#12135)
  1119	        finalizer = functools.partial(self.finish, request=request)
  1120	        for parent_fixture in requested_fixtures_that_should_finalize_us:
  1121	            parent_fixture.addfinalizer(finalizer)
  1122
  1123	        ihook = request.node.ihook
  1124	        try:
  1125	            # Setup the fixture, run the code in it, and cache the value
  1126	            # in self.cached_result
  1127	            result = ihook.pytest_fixture_setup(fixturedef=self, request=request)
  1128	        finally:
  1129	            # schedule our finalizer, even if the setup failed
  1130	            request.node.addfinalizer(finalizer)
  1131
  1132	        return result
  1133
  1134	    def cache_key(self, request: SubRequest) -> object:
  1135	        return getattr(request, "param", None)
  1136
  1137	    def __repr__(self) -> str:
  1138	        return f"<FixtureDef argname={self.argname!r} scope={self.scope!r} baseid={self.baseid!r}>"

  1141	def resolve_fixture_function(
  1142	    fixturedef: FixtureDef[FixtureValue], request: FixtureRequest
  1143	) -> _FixtureFunc[FixtureValue]:
  1144	    """Get the actual callable that can be called to obtain the fixture
  1145	    value."""
  1146	    fixturefunc = fixturedef.func
  1147	    # The fixture function needs to be bound to the actual
  1148	    # request.instance so that code working with "fixturedef" behaves
  1149	    # as expected.
  1150	    instance = request.instance
  1151	    if instance is not None:
  1152	        # Handle the case where fixture is defined not in a test class, but some other class
  1153	        # (for example a plugin class with a fixture), see #2270.
  1154	        if hasattr(fixturefunc, "__self__") and not isinstance(
  1155	            instance,
  1156	            fixturefunc.__self__.__class__,
  1157	        ):
  1158	            return fixturefunc
  1159	        fixturefunc = getimfunc(fixturedef.func)
  1160	        if fixturefunc != fixturedef.func:
  1161	            fixturefunc = fixturefunc.__get__(instance)
  1162	    return fixturefunc

  1165	def pytest_fixture_setup(
  1166	    fixturedef: FixtureDef[FixtureValue], request: SubRequest
  1167	) -> FixtureValue:
  1168	    """Execution of fixture setup."""
  1169	    kwargs = {}
  1170	    for argname in fixturedef.argnames:
  1171	        kwargs[argname] = request.getfixturevalue(argname)
  1172
  1173	    fixturefunc = resolve_fixture_function(fixturedef, request)
  1174	    my_cache_key = fixturedef.cache_key(request)
  1175
  1176	    if inspect.isasyncgenfunction(fixturefunc) or inspect.iscoroutinefunction(
  1177	        fixturefunc
  1178	    ):
  1179	        auto_str = " with autouse=True" if fixturedef._autouse else ""
  1180
  1181	        warnings.warn(
  1182	            PytestRemovedIn9Warning(
  1183	                f"{request.node.name!r} requested an async fixture "
  1184	                f"{request.fixturename!r}{auto_str}, with no plugin or hook that "
  1185	                "handled it. This is usually an error, as pytest does not natively "
  1186	                "support it. "
  1187	                "This will turn into an error in pytest 9.\n"
  1188	                "See: https://docs.pytest.org/en/stable/deprecations.html#sync-test-depending-on-async-fixture"
  1189	            ),
  1190	            # no stacklevel will point at users code, so we just point here
  1191	            stacklevel=1,
  1192	        )
  1193
  1194	    try:
  1195	        result = call_fixture_func(fixturefunc, request, kwargs)
  1196	    except TEST_OUTCOME as e:
  1197	        if isinstance(e, skip.Exception):
  1198	            # The test requested a fixture which caused a skip.
  1199	            # Don't show the fixture as the skip location, as then the user
  1200	            # wouldn't know which test skipped.
  1201	            e._use_item_location = True
  1202	        fixturedef.cached_result = (None, my_cache_key, (e, e.__traceback__))
  1203	        raise
  1204	    fixturedef.cached_result = (result, my_cache_key, None)
  1205	    return result

  1322	def fixture(
  1323	    fixture_function: FixtureFunction | None = None,
  1324	    *,
  1325	    scope: _ScopeName | Callable[[str, Config], _ScopeName] = "function",
  1326	    params: Iterable[object] | None = None,
  1327	    autouse: bool = False,
  1328	    ids: Sequence[object | None] | Callable[[Any], object | None] | None = None,
  1329	    name: str | None = None,
  1330	) -> FixtureFunctionMarker | FixtureFunctionDefinition:
  1331	    """Decorator to mark a fixture factory function.
  1332
  1333	    This decorator can be used, with or without parameters, to define a
  1334	    fixture function.
  1335
  1336	    The name of the fixture function can later be referenced to cause its
  1337	    invocation ahead of running tests: test modules or classes can use the
  1338	    ``pytest.mark.usefixtures(fixturename)`` marker.
  1339
  1340	    Test functions can directly use fixture names as input arguments in which
  1341	    case the fixture instance returned from the fixture function will be
  1342	    injected.
  1343
  1344	    Fixtures can provide their values to test functions using ``return`` or
  1345	    ``yield`` statements. When using ``yield`` the code block after the
  1346	    ``yield`` statement is executed as teardown code regardless of the test
  1347	    outcome, and must yield exactly once.
  1348
  1349	    :param scope:
  1350	        The scope for which this fixture is shared; one of ``"function"``
  1351	        (default), ``"class"``, ``"module"``, ``"package"`` or ``"session"``.
  1352
  1353	        This parameter may also be a callable which receives ``(fixture_name, config)``
  1354	        as parameters, and must return a ``str`` with one of the values mentioned above.
  1355
  1356	        See :ref:`dynamic scope` in the docs for more information.
  1357
  1358	    :param params:
  1359	        An optional list of parameters which will cause multiple invocations
  1360	        of the fixture function and all of the tests using it. The current
  1361	        parameter is available in ``request.param``.
  1362
  1363	    :param autouse:
  1364	        If True, the fixture func is activated for all tests that can see it.
  1365	        If False (the default), an explicit reference is needed to activate
  1366	        the fixture.
  1367
  1368	    :param ids:
  1369	        Sequence of ids each corresponding to the params so that they are
  1370	        part of the test id. If no ids are provided they will be generated
  1371	        automatically from the params.
  1372
  1373	    :param name:
  1374	        The name of the fixture. This defaults to the name of the decorated
  1375	        function. If a fixture is used in the same module in which it is
  1376	        defined, the function name of the fixture will be shadowed by the
  1377	        function arg that requests the fixture; one way to resolve this is to
  1378	        name the decorated function ``fixture_<fixturename>`` and then use
  1379	        ``@pytest.fixture(name='<fixturename>')``.
  1380	    """
  1381	    fixture_marker = FixtureFunctionMarker(
  1382	        scope=scope,
  1383	        params=tuple(params) if params is not None else None,
  1384	        autouse=autouse,
  1385	        ids=None if ids is None else ids if callable(ids) else tuple(ids),
  1386	        name=name,
  1387	        _ispytest=True,
  1388	    )
  1389
  1390	    # Direct decoration.
  1391	    if fixture_function:
  1392	        return fixture_marker(fixture_function)
  1393
  1394	    return fixture_marker
