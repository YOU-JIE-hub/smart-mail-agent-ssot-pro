   215	            serious = True

   217	        options.warn_unused_configs
   218	        and options.unused_configs
   219	        and not options.incremental
   220	        and not options.non_interactive
   221	    ):
   222	        print(
   223	            "Warning: unused section(s) in {}: {}".format(
   224	                options.config_file,

   228	                        glob
   229	                        for glob in options.per_module_options.keys()

   231	                    ],
   232	                ),
   233	            ),
   234	            file=stderr,
   235	        )
   236	    maybe_write_junit_xml(time.time() - t0, serious, messages, messages_by_file, options)

   289	def python_executable_prefix(v: str) -> list[str]:
   290	    if sys.platform == "win32":
   291	        # on Windows, all Python executables are named `python`. To handle this, there
   292	        # is the `py` launcher, which can be passed a version e.g. `py -3.8`, and it will
   293	        # execute an installed Python 3.8 interpreter. See also:
   294	        # https://docs.python.org/3/using/windows.html#python-launcher-for-windows
   295	        return ["py", f"-{v}"]
   296	    else:
   297	        return [f"python{v}"]

   300	def _python_executable_from_version(python_version: tuple[int, int]) -> str:
   301	    if sys.version_info[:2] == python_version:
   302	        return sys.executable
   303	    str_ver = ".".join(map(str, python_version))
   304	    try:
   305	        sys_exe = (
   306	            subprocess.check_output(
   307	                python_executable_prefix(str_ver) + ["-c", "import sys; print(sys.executable)"],
   308	                stderr=subprocess.STDOUT,
   309	            )
   310	            .decode()
   311	            .strip()
   312	        )
   313	        return sys_exe
   314	    except (subprocess.CalledProcessError, FileNotFoundError) as e:
   315	        raise PythonExecutableInferenceError(
   316	            "failed to find a Python executable matching version {},"
   317	            " perhaps try --python-executable, or --no-site-packages?".format(python_version)
   318	        ) from e

   373	class CapturableArgumentParser(argparse.ArgumentParser):
   374	    """Override ArgumentParser methods that use sys.stdout/sys.stderr directly.
   375
   376	    This is needed because hijacking sys.std* is not thread-safe,
   377	    yet output must be captured to properly support mypy.api.run.
   378	    """
   379
   380	    def __init__(self, *args: Any, **kwargs: Any) -> None:
   381	        self.stdout = kwargs.pop("stdout", sys.stdout)
   382	        self.stderr = kwargs.pop("stderr", sys.stderr)
   383	        super().__init__(*args, **kwargs)
   384
   385	    # =====================
   386	    # Help-printing methods
   387	    # =====================
   388	    def print_usage(self, file: SupportsWrite[str] | None = None) -> None:
   389	        if file is None:
   390	            file = self.stdout
   391	        self._print_message(self.format_usage(), file)
   392
   393	    def print_help(self, file: SupportsWrite[str] | None = None) -> None:
   394	        if file is None:
   395	            file = self.stdout
   396	        self._print_message(self.format_help(), file)
   397
   398	    def _print_message(self, message: str, file: SupportsWrite[str] | None = None) -> None:
   399	        if message:
   400	            if file is None:
   401	                file = self.stderr
   402	            file.write(message)
   403
   404	    # ===============
   405	    # Exiting methods
   406	    # ===============
   407	    def exit(self, status: int = 0, message: str | None = None) -> NoReturn:
   408	        if message:
   409	            self._print_message(message, self.stderr)
   410	        sys.exit(status)
   411
   412	    def error(self, message: str) -> NoReturn:
   413	        """error(message: string)
   414
   415	        Prints a usage message incorporating the message to stderr and
   416	        exits.
   417
   418	        If you override this in a subclass, it should not return -- it
   419	        should either exit or raise an exception.
   420	        """
   421	        self.print_usage(self.stderr)
   422	        args = {"prog": self.prog, "message": message}
   423	        self.exit(2, gettext("%(prog)s: error: %(message)s\n") % args)

   426	class CapturableVersionAction(argparse.Action):
   427	    """Supplement CapturableArgumentParser to handle --version.
   428
   429	    This is nearly identical to argparse._VersionAction except,
   430	    like CapturableArgumentParser, it allows output to be captured.
   431
   432	    Another notable difference is that version is mandatory.
   433	    This allows removing a line in __call__ that falls back to parser.version
   434	    (which does not appear to exist).
   435	    """
   436
   437	    def __init__(
   438	        self,
   439	        option_strings: Sequence[str],
   440	        version: str,
   441	        dest: str = argparse.SUPPRESS,
   442	        default: str = argparse.SUPPRESS,

   444	        stdout: IO[str] | None = None,

   446	        super().__init__(
   447	            option_strings=option_strings, dest=dest, default=default, nargs=0, help=help
   448	        )
   449	        self.version = version
   450	        self.stdout = stdout or sys.stdout
   451
   452	    def __call__(
   453	        self,
   454	        parser: argparse.ArgumentParser,
   455	        namespace: argparse.Namespace,
   456	        values: str | Sequence[Any] | None,
   457	        option_string: str | None = None,
   458	    ) -> NoReturn:
   459	        formatter = parser._get_formatter()
   460	        formatter.add_text(self.version)
   461	        parser._print_message(formatter.format_help(), self.stdout)
   462	        parser.exit()
