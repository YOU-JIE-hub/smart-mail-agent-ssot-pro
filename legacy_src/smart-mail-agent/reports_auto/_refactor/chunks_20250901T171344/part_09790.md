   362	        clbo = getattr(self,'categoryLabelBarOrder','auto')

   364	        clbo = clbo=='first'
   365	        style = cA.style
   366	        bars = self.bars
   367	        lineCount = sum((int(bars.checkAttr(_,'isLine',False)) for _ in range(seriesCount)))
   368	        seriesMLineCount = seriesCount - lineCount
   369	        if style=='mixed':
   370	            ss = self._seriesOrder
   371	            barsPerGroup = len(ss) - lineCount
   372	            wB = barsPerGroup*barWidth
   373	            wS = (barsPerGroup-1)*barSpacing
   374	            if barsPerGroup>1:

   376	                bGapS = barSpacing
   377	            else:

   410	        elif useAbsolute==7:    #all fixed

   412	            _cscale = cA._scale
   413	        elif useAbsolute==1: #case 1 barWidth is fixed
   414	            fB = 1.0
   415	            fG = fS = (aW-wB)/(wG+wS)
   416	        elif useAbsolute==2: #groupspacing is fixed
   417	            fG=1.0
   418	            fB = fS = (aW-wG)/(wB+wS)
   419	        elif useAbsolute==3: #groupspacing & barwidth are fixed
   420	            fB = fG = 1.0
   421	            fS = (aW-wG-wB)/wS if wS else 0
   422	        elif useAbsolute==4: #barspacing is fixed
   423	            fS=1.0
   424	            fG = fB = (aW-wS)/(wG+wB)
   425	        elif useAbsolute==5: #barspacing & barWidth are fixed
   426	            fS = fB = 1.0
   427	            fG = (aW-wB-wS)/wG
   428	        elif useAbsolute==6: #barspacing & groupspacing are fixed

   432	        self._normFactorG = fG
   433	        self._normFactorS = fS

   437	        vScale = vA.scale
   438	        vm, vM = vA._valueMin, vA._valueMax
   439	        if vm <= 0 <= vM:

   449	        bGap = bGapB*fB+bGapS*fS

   461	        reversePlotOrder = self.reversePlotOrder

   549	    def _labelXY(self,label,x,y,width,height):
   550	        'Compute x, y for a label'
   551	        nudge = label.nudge
   552	        bt = getattr(label,'boxTarget','normal')
   553	        anti = bt=='anti'
   554	        if anti: nudge = -nudge
   555	        pm = value = height
   556	        if anti: value = 0
   557	        a = x + 0.5*width
   558	        nudge = (height>=0 and 1 or -1)*nudge
   559	        if bt=='mid':
   560	            b = y+height*0.5
   561	        elif bt=='hi':
   562	            if value>=0:
   563	                b = y + value + nudge
   564	            else:
   565	                b = y - nudge
   566	                pm = -pm
   567	        elif bt=='lo':
   568	            if value<=0:
   569	                b = y + value + nudge
   570	            else:
   571	                b = y - nudge
   572	                pm = -pm
   573	        else:
   574	            b = y + value + nudge
   575	        label._pmv = pm #the plus minus val
   576	        return a,b,pm
