  1478	        if not required_plugins:
  1479	            return

  1544	    def issue_config_time_warning(self, warning: Warning, stacklevel: int) -> None:
  1545	        """Issue and handle a warning during the "configure" stage.
  1546
  1547	        During ``pytest_configure`` we can't capture warnings using the ``catch_warnings_for_item``
  1548	        function because it is not possible to have hook wrappers around ``pytest_configure``.
  1549
  1550	        This function is mainly intended for plugins that need to issue warnings during
  1551	        ``pytest_configure`` (or similar stages).
  1552
  1553	        :param warning: The warning instance.
  1554	        :param stacklevel: stacklevel forwarded to warnings.warn.
  1555	        """
  1556	        if self.pluginmanager.is_blocked("warnings"):
  1557	            return
  1558
  1559	        cmdline_filters = self.known_args_namespace.pythonwarnings or []
  1560	        config_filters = self.getini("filterwarnings")
  1561
  1562	        with warnings.catch_warnings(record=True) as records:
  1563	            warnings.simplefilter("always", type(warning))
  1564	            apply_warning_filters(config_filters, cmdline_filters)
  1565	            warnings.warn(warning, stacklevel=stacklevel)
  1566
  1567	        if records:
  1568	            frame = sys._getframe(stacklevel - 1)
  1569	            location = frame.f_code.co_filename, frame.f_lineno, frame.f_code.co_name
  1570	            self.hook.pytest_warning_recorded.call_historic(
  1571	                kwargs=dict(
  1572	                    warning_message=records[0],
  1573	                    when="config",
  1574	                    nodeid="",
  1575	                    location=location,
  1576	                )
  1577	            )

  1587	    def getini(self, name: str) -> Any:

  1589

  1591	        :ref:`ini file <configfiles>`, then the ``default`` value provided while
  1592	        registering the configuration through
  1593	        :func:`parser.addini <pytest.Parser.addini>` will be returned.

  1596

  1602	        ``paths``, ``pathlist``, ``args`` and ``linelist`` : empty list ``[]``

  1605	        ``int`` : ``0``

  1609	        while registering the configuration through
  1610	        :func:`parser.addini <pytest.Parser.addini>`, then the configuration

  1612

  1617	        try:
  1618	            return self._inicache[name]
  1619	        except KeyError:
