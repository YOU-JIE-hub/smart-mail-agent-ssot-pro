     os.unlink(self.status_file)
   267	            try:
   268	                server.cleanup()  # try to remove the socket dir on Linux
   269	            except OSError:
   270	                pass
   271	            exc_info = sys.exc_info()
   272	            if exc_info[0] and exc_info[0] is not SystemExit:
   273	                traceback.print_exception(*exc_info)

   275	    def run_command(self, command: str, data: dict[str, object]) -> dict[str, object]:
   276	        """Run a specific command from the registry."""
   277	        key = "cmd_" + command
   278	        method = getattr(self.__class__, key, None)
   279	        if method is None:
   280	            return {"error": f"Unrecognized command '{command}'"}
   281	        else:
   282	            if command not in {"check", "recheck", "run"}:
   283	                # Only the above commands use some error formatting.
   284	                del data["is_tty"]
   285	                del data["terminal_width"]
   286	            ret = method(self, **data)
   287	            assert isinstance(ret, dict)
   288	            return ret

   304	    def cmd_stop(self) -> dict[str, object]:
   305	        """Stop daemon."""
   306	        # We need to remove the status file *before* we complete the
   307	        # RPC. Otherwise a race condition exists where a subsequent
   308	        # command can see a status file from a dying server and think
   309	        # it is a live one.
   310	        os.unlink(self.status_file)
   311	        return {}

   313	    def cmd_run(
   314	        self,
   315	        version: str,
   316	        args: Sequence[str],
   317	        export_types: bool,
   318	        is_tty: bool,
   319	        terminal_width: int,
   320	    ) -> dict[str, object]:
   321	        """Check a list of files, triggering a restart if needed."""
   322	        stderr = io.StringIO()
   323	        stdout = io.StringIO()
   324	        try:
   325	            # Process options can exit on improper arguments, so we need to catch that and
   326	            # capture stderr so the client can report it
   327	            with redirect_stderr(stderr):
   328	                with redirect_stdout(stdout):
   329	                    sources, options = mypy.main.process_options(
   330	                        ["-i"] + list(args),
   331	                        require_targets=True,
   332	                        server_options=True,
   333	                        fscache=self.fscache,
   334	                        program="mypy-daemon",
   335	                        header=argparse.SUPPRESS,
   336	                    )
   337	            # Signal that we need to restart if the options have changed
   338	            if not options.compare_stable(self.options_snapshot):
   339	                return {"restart": "configuration changed"}
   340	            if __version__ != version:
   341	                return {"restart": "mypy version changed"}
   342	            if self.fine_grained_manager:
   343	                manager = self.fine_grained_manager.manager
   344	                start_plugins_snapshot = manager.plugins_snapshot
   345	                _, current_plugins_snapshot = mypy.build.load_plugins(
   346	                    options, manager.errors, sys.stdout, extra_plugins=()
   347	                )
   348	                if current_plugins_snapshot != start_plugins_snapshot:
   349	                    return {"restart": "plugins changed"}
   350	        except InvalidSourceList as err:
   351	            return {"out": "", "err": str(err), "status": 2}
   352	        except SystemExit as e:
   353	            return {"out": stdout.getvalue(), "err": stderr.getvalue(), "status": e.code}
   354	        return self.check(sources, export_types, is_tty, terminal_width)

   356	    def cmd_check(
   357	        self, files: Sequence[str], export_types: bool, is_tty: bool, terminal_width: int
   358	    ) -> dict[str, object]:
   359	        """Check a list of files."""
   360	        try:
   361	            sources = create_source_list(files, self.options, self.fscache)
   362	        except InvalidSourceList as err:
   363	            return {"out": "", "err": str(err), "status": 2}
   364	        return self.check(sources, export_types, is_tty, terminal_width)

   366	    def cmd_recheck(
   367	        self,
   368	        is_tty: bool,
   369	        terminal_width: int,
   370	        export_types: bool,
   371	        remove: list[str] | None = None,
   372	        update: list[str] | None = None,
   373	    ) -> dict[str, object]:
   374	        """Check the same list of files we checked most recently.
   375
   376	        If remove/update is given, they modify the previous list;
   377	        if all are None, stat() is called for each file in the previous list.
   378	        """
   379	        t0 = time.time()
   380	        if not self.fine_grained_manager:
   381	            return {"error": "Command 'recheck' is only valid after a 'check' command"}
   382	        sources = self.previous_sources
   383	        if remove:
   384	            removals = set(remove)
   385	            sources = [s for s in sources if s.path and s.path not in removals]
   386	        if update:
   387	            # Sort list of file updates by extension, so *.pyi files are first.
   388	            update.sort(key=lambda f: os.path.splitext(f)[1], reverse=True)
   389
   390	            known = {s.path for s in sources if s.path}
   391	            added = [p for p in update if p not in known]
   392	            try:
   393	                added_sources = create_source_list(added, self.options, self.fscache)
   394	            except InvalidSourceList as err:
   395	                return {"out": "", "err": str(err), "status": 2}
   396	            sources = sources + added_sources  # Make a copy!
   397	        t1 = time.time()
   398	        manager = self.fine_grained_manager.manager
   399	        manager.log(f"fine-grained increment: cmd_recheck: {t1 - t0:.3f}s")
   400	        old_export_types = self.options.export_types
   401	        self.options.export_types = self.options.export_types or export_types
   402	        if not self.following_imports():
   403	            messages = self.fine_grained_increment(
   404	                sources, remove, update, explicit_export_types=export_types
   405	            )
   406	        else:
   407	            assert remove is None and update is None
   408	            messages = self.fine_grained_increment_follow_imports(
   409	                sources, explicit_export_types=export_types
   410	            )
   411	        res = self.increment_output(messages, sources, is_tty, terminal_width)
   412	        self.flush_caches()
   413	        self.update_stats(res)
   414	        self.options.export_types = old_export_types
   415	        return res

   417	    def check(
   418	        self, sources: list[BuildSource], export_types: bool, is_tty: bool, terminal_width: int
   419	    ) -> dict[str, Any]:
   420	        """Check using fine-grained incremental mode.
   421
   422	        If is_tty is True format the output nicely with colors and summary line
   423	        (unless disabled in self.options). Also pass the terminal_width to formatter.
   424	        """
   425	        old_export_types = self.options.export_types
   426	        self.options.export_types = self.options.export_types or export_types
   427	        if not self.fine_grained_manager:
   428	            res = self.initialize_fine_grained(sources, is_tty, terminal_width)
   429	        else:
   430	            if not self.following_imports():
   431	                messages = self.fine_grained_increment(sources, explicit_export_types=export_types)
   432	            else:
   433	                messages = self.fine_grained_increment_follow_imports(
   434	                    sources, explicit_export_types=export_types
   435	                )
   436	            res = self.increment_output(messages, sources, is_tty, terminal_width)
   437	        self.flush_caches()
   438	        self.update_stats(res)
   439	        self.options.export_types = old_export_types
   440	        return res

   459	    def initialize_fine_grained(
   460	        self, sources: list[BuildSource], is_tty: bool, terminal_width: int
   461	    ) -> dict[str, Any]:
   462	        self.fswatcher = FileSystemWatcher(self.fscache)
   463	        t0 = time.time()
   464	        self.update_sources(sources)
   465	        t1 = time.time()
   466	        try:
   467	            result = mypy.build.build(sources=sources, options=self.options, fscache=self.fscache)
   468	        except mypy.errors.CompileError as e:
   469	            output = "".join(s + "\n" for s in e.messages)
   470	            if e.use_stdout:
   471	                out, err = output, ""
   472	            else:
   473	                out, err = "", output
   474	            return {"out": out, "err": err, "status": 2}
   475	        messages = result.errors
   476	        self.fine_grained_manager = FineGrainedBuildManager(result)
   477
   478	        original_sources_len = len(sources)
   479	        if self.following_imports():
   480	            sources = find_all_sources_in_build(self.fine_grained_manager.graph, sources)
   481	            self.update_sources(sources)
   482
   483	        self.previous_sources = sources
   484
   485	        # If we are using the fine-grained cache, build hasn't actually done
   486	        # the typechecking on the updated files yet.
   487	        # Run a fine-grained update starting from the cached data
   488	        if result.used_cache:
   489	            t2 = time.time()
   490	            # Pull times and hashes out of the saved_cache and stick them into
   491	            # the fswatcher, so we pick up the changes.
   492	            for state in self.fine_grained_manager.graph.values():
   493	                meta = state.meta
   494	                if meta is None:
   495	                    continue
   496	                assert state.path is not None
   497	                self.fswatcher.set_file_data(
   498	                    state.path,
   499	                    FileData(st_mtime=float(meta.mtime), st_size=meta.size, hash=meta.hash),
   500	                )
   501
   502	            changed, removed = self.find_changed(sources)
   503	            changed += self.find_added_suppressed(
   504	                self.fine_grained_manager.graph,
   505	                set(),
   506	                self.fine_grained_manager.manager.search_paths,
   507	            )
   508
   509	            # Find anything that has had its dependency list change
   510	            for state in self.fine_grained_manager.graph.values():
   511	                if not state.is_fresh():
   512	                    assert state.path is not None
   513	                    changed.append((state.id, state.path))
   514
   515	            t3 = time.time()
   516	            # Run an update
   517	            messages = self.fine_grained_manager.update(changed, removed)
   518
   519	            if self.following_imports():
   520	                # We need to do another update to any new files found by following imports.
   521	                messages = self.fine_grained_increment_follow_imports(sources)
   522
   523	            t4 = time.time()
   524	            self.fine_grained_manager.manager.add_stats(
   525	                update_sources_time=t1 - t0,
   526	                build_time=t2 - t1,
   527	                find_changes_time=t3 - t2,
   528	                fg_update_time=t4 - t3,
   529	                files_changed=len(removed) + len(changed),
   530	            )
   531
   532	        else:
   533	            # Stores the initial state of sources as a side effect.
   534	            self.fswatcher.find_changed()
   535
   536	        if MEM_PROFILE:
   537	            from mypy.memprofile import print_memory_profile
   538
   539	            print_memory_profile(run_gc=False)
   540
   541	        __, n_notes, __ = count_stats(messages)
   542	        status = 1 if messages and n_notes < len(messages) else 0
   543	        # We use explicit sources length to match the logic in non-incremental mode.
   544	        messages = self.pretty_messages(messages, original_sources_len, is_tty, terminal_width)
   545	        return {"out": "".join(s + "\n" for s in messages), "err": "", "status": status}

   547	    def fine_grained_increment(
   548	        self,
   549	        sources: list[BuildSource],
   550	        remove: list[str] | None = None,
   551	        update: list[str] | None = None,
   552	        explicit_export_types: bool = False,
   553	    ) -> list[str]:
   554	        """Perform a fine-grained type checking increment.
   555
   556	        If remove and update are None, determine changed paths by using
   557	        fswatcher. Otherwise, assume that only these files have changes.
   558
   559	        Args:
   560	            sources: sources passed on the command line
   561	            remove: paths of files that have been removed
   562	            update: paths of files that have been changed or created
   563	            explicit_export_types: --export-type was passed in a check command
   564	              (as opposite to being set in dmypy start)
   565	        """
   566	        assert self.fine_grained_manager is not None
   567	        manager = self.fine_grained_manager.manager
   568
   569	        t0 = time.time()
   570	        if remove is None and update is None:
   571	            # Use the fswatcher to determine which files were changed
   572	            # (updated or added) or removed.
   573	            self.update_sources(sources)
   574	            changed, removed = self.find_changed(sources)
   575	        else:
   576	            # Use the remove/update lists to update fswatcher.
   577	            # This avoids calling stat() for unchanged files.
   578	            changed, removed = self.update_changed(sources, remove or [], update or [])
   579	        if explicit_export_types:
   580	            # If --export-types is given, we need to force full re-checking of all
   581	            # explicitly passed files, since we need to visit each expression.
   582	            add_all_sources_to_changed(sources, changed)
   583	        changed += self.find_added_suppressed(
   584	            self.fine_grained_manager.graph, set(), manager.search_paths
   585	        )
   586	        manager.search_paths = compute_search_paths(sources, manager.options, manager.data_dir)
   587	        t1 = time.time()
   588	        manager.log(f"fine-grained increment: find_changed: {t1 - t0:.3f}s")
   589	        messages = self.fine_grained_manager.update(changed, removed)
   590	        t2 = time.time()
   591	        manager.log(f"fine-grained increment: update: {t2 - t1:.3f}s")
   592	        manager.add_stats(
   593	            find_changes_time=t1 - t0,
   594	            fg_update_time=t2 - t1,
   595	            files_changed=len(removed) + len(changed),
   596	        )
   597
   598	        self.previous_sources = sources
   599	        return messages
   600
   601	    def fine_grained_increment_follow_imports(
   602	        self, sources: list[BuildSource], explicit_export_types: bool = False
   603	    ) -> list[str]:
   604	        """Like fine_grained_increment, but follow imports."""
   605	        t0 = time.time()
   606
   607	        # TODO: Support file events
   608
   609	        assert self.fine_grained_manager is not None
   610	        fine_grained_manager = self.fine_grained_manager
   611	        graph = fine_grained_manager.graph
   612	        manager = fine_grained_manager.manager
   613
   614	        orig_modules = list(graph.keys())
   615
   616	        self.update_sources(sources)
   617	        changed_paths = self.fswatcher.find_changed()
   618	        manager.search_paths = compute_search_paths(sources, manager.options, manager.data_dir)
   619
   620	        t1 = time.time()
   621	        manager.log(f"fine-grained increment: find_changed: {t1 - t0:.3f}s")
   622
   623	        # Track all modules encountered so far. New entries for all dependencies
   624	        # are added below by other module finding methods below. All dependencies
   625	        # in graph but not in `seen` are considered deleted at the end of this method.
   626	        seen = {source.module for source in sources}
   627
   628	        # Find changed modules reachable from roots (or in roots) already in graph.
   629	        changed, new_files = self.find_reachable_changed_modules(
   630	            sources, graph, seen, changed_paths
   631	        )
   632	        # Same as in fine_grained_increment().
   633	        self.add_explicitly_new(sources, changed)
   634	        if explicit_export_types:
   635	            # Same as in fine_grained_increment().
   636	            add_all_sources_to_changed(sources, changed)
   637	        sources.extend(new_files)
   638
   639	        # Process changes directly reachable from roots.
   640	        messages = fine_grained_manager.update(changed, [], followed=True)
   641
   642	        # Follow deps from changed modules (still within graph).
   643	        worklist = changed.copy()
   644	        while worklist:
   645	            module = worklist.pop()
   646	            if module[0] not in graph:
   647	                continue
   648	            sources2 = self.direct_imports(module, graph)
   649	            # Filter anything already seen before. This prevents
   650	            # infinite looping if there are any self edges. (Self
   651	            # edges are maybe a bug, but...)
   652	            sources2 = [source for source in sources2 if source.module not in seen]
   653	            changed, new_files = self.find_reachable_changed_modules(
   654	                sources2, graph, seen, changed_paths
   655	            )
   656	            self.update_sources(new_files)
   657	            messages = fine_grained_manager.update(changed, [], followed=True)
   658	            worklist.extend(changed)
   659
   660	        t2 = time.time()
   661
   662	        def refresh_file(module: str, path: str) -> list[str]:
   663	            return fine_grained_manager.update([(module, path)], [], followed=True)
   664
   665	        for module_id, state in list(graph.items()):
   666	            new_messages = refresh_suppressed_submodules(
   667	                module_id, state.path, fine_grained_manager.deps, graph, self.fscache, refresh_file
   668	            )
   669	            if new_messages is not None:
   670	                messages = new_messages
   671
   672	        t3 = time.time()
   673
   674	        # There may be new files that became available, currently treated as
   675	        # suppressed imports. Process them.
   676	        while True:
   677	            new_unsuppressed = self.find_added_suppressed(
