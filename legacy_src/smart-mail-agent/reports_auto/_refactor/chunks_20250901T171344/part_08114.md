   134	            if kind.is_required() and not actuals:

   137	                break

   139	                args = [call.args[i] for i in actuals]

   324	def _add_method_by_spec(
   325	    api: SemanticAnalyzerPluginInterface | CheckerPluginInterface,
   326	    info: TypeInfo,
   327	    name: str,
   328	    spec: MethodSpec,
   329	    *,
   330	    is_classmethod: bool,
   331	    is_staticmethod: bool,
   332	) -> tuple[FuncDef | Decorator, SymbolTableNode]:
   333	    args, return_type, self_type, tvar_defs = spec
   334
   335	    assert not (
   336	        is_classmethod is True and is_staticmethod is True
   337	    ), "Can't add a new method that's both staticmethod and classmethod."
   338
   339	    if isinstance(api, SemanticAnalyzerPluginInterface):
   340	        function_type = api.named_type("builtins.function")
   341	    else:
   342	        function_type = api.named_generic_type("builtins.function", [])
   343
   344	    if is_classmethod:
   345	        self_type = self_type or TypeType(fill_typevars(info))
   346	        first = [Argument(Var("_cls"), self_type, None, ARG_POS, True)]
   347	    elif is_staticmethod:
   348	        first = []
   349	    else:
   350	        self_type = self_type or fill_typevars(info)
   351	        first = [Argument(Var("self"), self_type, None, ARG_POS)]
   352	    args = first + args
   353
   354	    arg_types, arg_names, arg_kinds = [], [], []
   355	    for arg in args:
   356	        assert arg.type_annotation, "All arguments must be fully typed."
   357	        arg_types.append(arg.type_annotation)
   358	        arg_names.append(arg.variable.name)
   359	        arg_kinds.append(arg.kind)
   360
   361	    signature = CallableType(arg_types, arg_kinds, arg_names, return_type, function_type)
   362	    if tvar_defs:
   363	        signature.variables = tvar_defs
   364
   365	    func = FuncDef(name, args, Block([PassStmt()]))
   366	    func.info = info
   367	    func.type = set_callable_name(signature, func)
   368	    func.is_class = is_classmethod
   369	    func.is_static = is_staticmethod
   370	    func._fullname = info.fullname + "." + name
   371	    func.line = info.line
   372
   373	    # Add decorator for is_staticmethod. It's unnecessary for is_classmethod.
   374	    if is_staticmethod:
   375	        func.is_decorated = True
   376	        v = Var(name, func.type)
   377	        v.info = info
   378	        v._fullname = func._fullname
   379	        v.is_staticmethod = True
   380	        dec = Decorator(func, [], v)
   381	        dec.line = info.line
   382	        sym = SymbolTableNode(MDEF, dec)
   383	        sym.plugin_generated = True
   384	        return dec, sym
   385
   386	    sym = SymbolTableNode(MDEF, func)
   387	    sym.plugin_generated = True
   388	    return func, sym

    43	def _autoconvertible_to_cdata(tp: Type, api: mypy.plugin.CheckerPluginInterface) -> Type:
    44	    """Get a type that is compatible with all types that can be implicitly converted to the given
    45	    CData type.
    46
    47	    Examples:
    48	    * c_int -> Union[c_int, int]
    49	    * c_char_p -> Union[c_char_p, bytes, int, NoneType]
    50	    * MyStructure -> MyStructure
    51	    """
    52	    allowed_types = []
    53	    # If tp is a union, we allow all types that are convertible to at least one of the union
    54	    # items. This is not quite correct - strictly speaking, only types convertible to *all* of the
    55	    # union items should be allowed. This may be worth changing in the future, but the more
    56	    # correct algorithm could be too strict to be useful.
    57	    for t in flatten_nested_unions([tp]):
    58	        t = get_proper_type(t)
    59	        # Every type can be converted from itself (obviously).
    60	        allowed_types.append(t)
    61	        if isinstance(t, Instance):
    62	            unboxed = _find_simplecdata_base_arg(t, api)
    63	            if unboxed is not None:
    64	                # If _SimpleCData appears in tp's (direct or indirect) bases, its type argument
    65	                # specifies the type's "unboxed" version, which can always be converted back to
    66	                # the original "boxed" type.
    67	                allowed_types.append(unboxed)
    68
    69	                if t.type.has_base("ctypes._PointerLike"):
    70	                    # Pointer-like _SimpleCData subclasses can also be converted from
    71	                    # an int or None.
    72	                    allowed_types.append(api.named_generic_type("builtins.int", []))
    73	                    allowed_types.append(NoneType())
    74
    75	    return make_simplified_union(allowed_types)
