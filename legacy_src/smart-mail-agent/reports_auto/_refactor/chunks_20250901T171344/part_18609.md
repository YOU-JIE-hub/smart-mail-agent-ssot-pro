  1855	# references.
  1856	#  1. Create an empty ClassIR for each class in an SCC.
  1857	#  2. Deserialize all of the functions, which can contain references

  1864	# 'cross_refs'. We don't follow that approach here because it seems to be more
  1865	# code for not a lot of gain since it is easy in mypyc to identify all the objects
  1866	# we might need to reference.

  1869	# names to ClassIRs and func IDs to FuncIRs.

  1872	# deserialization function.

   300	    def format(self, fmt: str, *args: Any) -> str:
   301	        """Helper for formatting strings.
   302
   303	        These format sequences are supported in fmt:
   304
   305	          %s: arbitrary object converted to string using str()
   306	          %r: name of IR value/register
   307	          %d: int
   308	          %f: float
   309	          %l: BasicBlock (formatted as label 'Ln')
   310	          %t: RType
   311	        """
   312	        result = []
   313	        i = 0
   314	        arglist = list(args)
   315	        while i < len(fmt):
   316	            n = fmt.find("%", i)
   317	            if n < 0:
   318	                n = len(fmt)
   319	            result.append(fmt[i:n])
   320	            if n < len(fmt):
   321	                typespec = fmt[n + 1]
   322	                arg = arglist.pop(0)
   323	                if typespec == "r":
   324	                    # Register/value
   325	                    assert isinstance(arg, Value)
   326	                    if isinstance(arg, Integer):
   327	                        result.append(str(arg.value))
   328	                    elif isinstance(arg, Float):
   329	                        result.append(repr(arg.value))
   330	                    else:
   331	                        result.append(self.names[arg])
   332	                elif typespec == "d":
   333	                    # Integer
   334	                    result.append("%d" % arg)
   335	                elif typespec == "f":
   336	                    # Float
   337	                    result.append("%f" % arg)
   338	                elif typespec == "l":
   339	                    # Basic block (label)
   340	                    assert isinstance(arg, BasicBlock)
   341	                    result.append("L%s" % arg.label)
   342	                elif typespec == "t":
   343	                    # RType
   344	                    assert isinstance(arg, RType)
   345	                    result.append(arg.name)
   346	                elif typespec == "s":
   347	                    # String
   348	                    result.append(str(arg))
   349	                else:
   350	                    raise ValueError(f"Invalid format sequence %{typespec}")
   351	                i = n + 2
   352	            else:
   353	                i = n
   354	        return "".join(result)

   372	def format_blocks(

   374	    names: dict[Value, str],
   375	    source_to_error: dict[ErrorSource, list[str]],
   376	) -> list[str]:
   377	    """Format a list of IR basic blocks into a human-readable form."""
   378	    # First label all of the blocks
   379	    for i, block in enumerate(blocks):
   380	        block.label = i
   381
   382	    handler_map: dict[BasicBlock, list[BasicBlock]] = {}
   383	    for b in blocks:
   384	        if b.error_handler:
   385	            handler_map.setdefault(b.error_handler, []).append(b)
   386
   387	    visitor = IRPrettyPrintVisitor(names)
   388
   389	    lines = []
   390	    for i, block in enumerate(blocks):
   391	        handler_msg = ""
   392	        if block in handler_map:
   393	            labels = sorted("L%d" % b.label for b in handler_map[block])
   394	            handler_msg = " (handler for {})".format(", ".join(labels))

   396	        lines.append("L%d:%s" % (block.label, handler_msg))
   397	        if block in source_to_error:
   398	            for error in source_to_error[block]:
   399	                lines.append(f"  ERR: {error}")

   401	        if (
   402	            isinstance(ops[-1], Goto)
   403	            and i + 1 < len(blocks)
   404	            and ops[-1].label == blocks[i + 1]
   405	            and not source_to_error.get(ops[-1], [])
   406	        ):
   407	            # Hide the last goto if it just goes to the next basic block,
   408	            # and there are no assocatiated errors with the op.
   409	            ops = ops[:-1]
   410	        for op in ops:
   411	            line = "    " + op.accept(visitor)
   412	            lines.append(line)
   413	            if op in source_to_error:
   414	                for error in source_to_error[op]:
   415	                    lines.append(f"  ERR: {error}")

   418
