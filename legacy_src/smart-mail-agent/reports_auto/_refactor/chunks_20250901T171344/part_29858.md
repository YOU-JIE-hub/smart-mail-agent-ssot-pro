    16	    result: set[BasicBlock] = set()
    17	    worklist = [entry_point]
    18	    while worklist:

    62	    Node 0 is always assumed to be the entry point. There must be a

    82	def get_cfg(blocks: list[BasicBlock], *, use_yields: bool = False) -> CFG:
    83	    """Calculate basic block control-flow graph.
    84
    85	    If use_yields is set, then we treat returns inserted by yields as gotos
    86	    instead of exits.
    87	    """
    88	    succ_map = {}
    89	    pred_map: dict[BasicBlock, list[BasicBlock]] = {}
    90	    exits = set()
    91	    for block in blocks:
    92	        assert not any(
    93	            isinstance(op, ControlOp) for op in block.ops[:-1]
    94	        ), "Control-flow ops must be at the end of blocks"
    95
    96	        if use_yields and isinstance(block.terminator, Return) and block.terminator.yield_target:
    97	            succ = [block.terminator.yield_target]
    98	        else:
    99	            succ = list(block.terminator.targets())
   100	        if not succ:
   101	            exits.add(block)
   102
   103	        # Errors can occur anywhere inside a block, which means that
   104	        # we can't assume that the entire block has executed before
   105	        # jumping to the error handler. In our CFG construction, we
   106	        # model this as saying that a block can jump to its error
   107	        # handler or the error handlers of any of its normal
   108	        # successors (to represent an error before that next block
   109	        # completes). This works well for analyses like "must
   110	        # defined", where it implies that registers assigned in a
   111	        # block may be undefined in its error handler, but is in
   112	        # general not a precise representation of reality; any
   113	        # analyses that require more fidelity must wait until after
   114	        # exception insertion.
   115	        for error_point in [block] + succ:
   116	            if error_point.error_handler:
   117	                succ.append(error_point.error_handler)
   118
   119	        succ_map[block] = succ
   120	        pred_map[block] = []
   121	    for prev, nxt in succ_map.items():
   122	        for label in nxt:
   123	            pred_map[label].append(prev)
   124	    return CFG(succ_map, pred_map, exits)

   134	    """Cleanup the control flow graph.
   135
   136	    This eliminates obviously dead basic blocks and eliminates blocks that contain
   137	    nothing but a single jump.

   139	    There is a lot more that could be done.
   140	    """
