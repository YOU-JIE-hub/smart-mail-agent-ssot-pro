  --------------------------+
   201	    +-------+
   202	    """
   203
   204	    # TODO: do something about abstract methods.
   205
   206	    func_reg: Value | None = None
   207
   208	    # We treat lambdas as always being nested because we always generate
   209	    # a class for lambdas, no matter where they are. (It would probably also
   210	    # work to special case toplevel lambdas and generate a non-class function.)
   211	    is_nested = fitem in builder.nested_fitems or isinstance(fitem, LambdaExpr)
   212	    contains_nested = fitem in builder.encapsulating_funcs.keys()
   213	    is_decorated = fitem in builder.fdefs_to_decorators
   214	    is_singledispatch = fitem in builder.singledispatch_impls
   215	    in_non_ext = False
   216	    add_nested_funcs_to_env = has_nested_func_self_reference(builder, fitem)
   217	    class_name = None
   218	    if cdef:
   219	        ir = builder.mapper.type_to_ir[cdef.info]
   220	        in_non_ext = not ir.is_ext_class
   221	        class_name = cdef.name
   222
   223	    if is_singledispatch:
   224	        func_name = singledispatch_main_func_name(name)
   225	    else:
   226	        func_name = name
   227
   228	    fn_info = FuncInfo(
   229	        fitem=fitem,
   230	        name=func_name,
   231	        class_name=class_name,
   232	        namespace=gen_func_ns(builder),
   233	        is_nested=is_nested,
   234	        contains_nested=contains_nested,
   235	        is_decorated=is_decorated,
   236	        in_non_ext=in_non_ext,
   237	        add_nested_funcs_to_env=add_nested_funcs_to_env,
   238	    )
   239	    is_generator = fn_info.is_generator
   240	    builder.enter(fn_info, ret_type=sig.ret_type)
   241
   242	    # Functions that contain nested functions need an environment class to store variables that
   243	    # are free in their nested functions. Generator functions need an environment class to
   244	    # store a variable denoting the next instruction to be executed when the __next__ function
   245	    # is called, along with all the variables inside the function itself.
   246	    if contains_nested or (
   247	        is_generator and not builder.fn_info.can_merge_generator_and_env_classes()
   248	    ):
   249	        setup_env_class(builder)
   250
   251	    if is_nested or in_non_ext:
   252	        setup_callable_class(builder)
   253
   254	    if is_generator:
   255	        # First generate a function that just constructs and returns a generator object.
   256	        func_ir, func_reg = gen_generator_func(
   257	            builder,
   258	            lambda args, blocks, fn_info: gen_func_ir(
   259	                builder, args, blocks, sig, fn_info, cdef, is_singledispatch
   260	            ),
   261	        )
   262
   263	        # Re-enter the FuncItem and visit the body of the function this time.
   264	        gen_generator_func_body(builder, fn_info, sig, func_reg)
   265	    else:
   266	        func_ir, func_reg = gen_func_body(builder, sig, cdef, is_singledispatch)
   267
   268	    if is_singledispatch:
   269	        # add the generated main singledispatch function
   270	        builder.functions.append(func_ir)
   271	        # create the dispatch function
   272	        assert isinstance(fitem, FuncDef)
   273	        return gen_dispatch_func_ir(builder, fitem, fn_info.name, name, sig)
   274
   275	    return func_ir, func_reg

   278	def gen_func_body(
   279	    builder: IRBuilder, sig: FuncSignature, cdef: ClassDef | None, is_singledispatch: bool
   280	) -> tuple[FuncIR, Value | None]:
   281	    load_env_registers(builder)
   282	    gen_arg_defaults(builder)
   283	    if builder.fn_info.contains_nested:
   284	        finalize_env_class(builder)
   285	    add_vars_to_env(builder)
   286	    builder.accept(builder.fn_info.fitem.body)
   287	    builder.maybe_add_implicit_return()
   288
   289	    # Hang on to the local symbol table for a while, since we use it
   290	    # to calculate argument defaults below.
   291	    symtable = builder.symtables[-1]
   292
   293	    args, _, blocks, ret_type, fn_info = builder.leave()
   294
   295	    func_ir, func_reg = gen_func_ir(builder, args, blocks, sig, fn_info, cdef, is_singledispatch)
   296
   297	    # Evaluate argument defaults in the surrounding scope, since we
   298	    # calculate them *once* when the function definition is evaluated.
   299	    calculate_arg_defaults(builder, fn_info, func_reg, symtable)
   300	    return func_ir, func_reg

   303	def has_nested_func_self_reference(builder: IRBuilder, fitem: FuncItem) -> bool:
   304	    """Does a nested function contain a self-reference in its body?
   305
   306	    If a nested function only has references in the surrounding function,
   307	    we don't need to add it to the environment.
   308	    """
   309	    if any(isinstance(sym, FuncBase) for sym in builder.free_variables.get(fitem, set())):
   310	        return True
   311	    return any(
   312	        has_nested_func_self_reference(builder, nested)
   313	        for nested in builder.encapsulating_funcs.get(fitem, [])
   314	    )

   317	def gen_func_ir(
   318	    builder: IRBuilder,
   319	    args: list[Register],
   320	    blocks: list[BasicBlock],
   321	    sig: FuncSignature,
   322	    fn_info: FuncInfo,
   323	    cdef: ClassDef | None,
   324	    is_singledispatch_main_func: bool = False,
   325	) -> tuple[FuncIR, Value | None]:
   326	    """Generate the FuncIR for a function.
   327
   328	    This takes the basic blocks and function info of a particular
   329	    function and returns the IR. If the function is nested,
   330	    also returns the register containing the instance of the
   331	    corresponding callable class.
   332	    """
   333	    func_reg: Value | None = None
   334	    if fn_info.is_nested or fn_info.in_non_ext:
   335	        func_ir = add_call_to_callable_class(builder, args, blocks, sig, fn_info)
   336	        add_get_to_callable_class(builder, fn_info)
   337	        func_reg = instantiate_callable_class(builder, fn_info)
   338	    else:
   339	        assert isinstance(fn_info.fitem, FuncDef)
   340	        func_decl = builder.mapper.func_to_decl[fn_info.fitem]
   341	        if fn_info.is_decorated or is_singledispatch_main_func:
   342	            class_name = None if cdef is None else cdef.name
   343	            func_decl = FuncDecl(
   344	                fn_info.name,
   345	                class_name,
   346	                builder.module_name,
   347	                sig,
   348	                func_decl.kind,
   349	                func_decl.is_prop_getter,
   350	                func_decl.is_prop_setter,
   351	            )
   352	            func_ir = FuncIR(
   353	                func_decl, args, blocks, fn_info.fitem.line, traceback_name=fn_info.fitem.name
   354	            )
   355	        else:
   356	            func_ir = FuncIR(
   357	                func_decl, args, blocks, fn_info.fitem.line, traceback_name=fn_info.fitem.name
   358	            )
   359	    return (func_ir, func_reg)

   362	def handle_ext_method(builder: IRBuilder, cdef: ClassDef, fdef: FuncDef) -> None:
   363	    # Perform the function of visit_method for methods inside extension classes.
   364	    name = fdef.name
   365	    class_ir = builder.mapper.type_to_ir[cdef.info]
   366	    sig = builder.mapper.fdef_to_sig(fdef, builder.options.strict_dunders_typing)
   367	    func_ir, func_reg = gen_func_item(builder, fdef, name, sig, cdef)
   368	    builder.functions.append(func_ir)
   369
   370	    if is_decorated(builder, fdef):
   371	        # Obtain the function name in order to construct the name of the helper function.
   372	        _, _, name = fdef.fullname.rpartition(".")
   373	        # Read the PyTypeObject representing the class, get the callable object
   374	        # representing the non-decorated method
   375	        typ = builder.load_native_type_object(cdef.fullname)
   376	        orig_func = builder.py_get_attr(typ, name, fdef.line)
   377
   378	        # Decorate the non-decorated method
   379	        decorated_func = load_decorated_func(builder, fdef, orig_func)
   380
   381	        # Set the callable object representing the decorated method as an attribute of the
   382	        # extension class.
   383	        builder.primitive_op(
   384	            py_setattr_op, [typ, builder.load_str(name), decorated_func], fdef.line
   385	        )
   386
   387	    if fdef.is_property:
   388	        # If there is a property setter, it will be processed after the getter,
   389	        # We populate the optional setter field with none for now.
   390	        assert name not in class_ir.properties
   391	        class_ir.properties[name] = (func_ir, None)
   392
   393	    elif fdef in builder.prop_setters:
   394	        # The respective property getter must have been processed already
   395	        assert name in class_ir.properties
   396	        getter_ir, _ = class_ir.properties[name]
   397	        class_ir.properties[name] = (getter_ir, func_ir)
   398
   399	    class_ir.methods[func_ir.decl.name] = func_ir
   400
   401	    # If this overrides a parent class method with a different type, we need
   402	    # to generate a glue method to mediate between them.
   403	    for base in class_ir.mro[1:]:
   404	        if (
   405	            name in base.method_decls
   406	            and name != "__init__"
   407	            and not is_same_method_signature(
   408	                class_ir.method_decls[name].sig, base.method_decls[name].sig
   409	            )
   410	        ):
   411	            # TODO: Support contravariant subtyping in the input argument for
   412	            # property setters. Need to make a special glue method for handling this,
   413	            # similar to gen_glue_property.
   414
   415	            f = gen_glue(builder, base.method_decls[name].sig, func_ir, class_ir, base, fdef)
   416	            class_ir.glue_methods[(base, name)] = f
   417	            builder.functions.append(f)
   418
   419	    # If the class allows interpreted children, create glue
   420	    # methods that dispatch via the Python API. These will go in a
   421	    # "shadow vtable" that will be assigned to interpreted
   422	    # children.
   423	    if class_ir.allow_interpreted_subclasses:
   424	        f = gen_glue(builder, func_ir.sig, func_ir, class_ir, class_ir, fdef, do_py_ops=True)
   425	        class_ir.glue_methods[(class_ir, name)] = f
   426	        builder.functions.append(f)

   429	def handle_non_ext_method(
   430	    builder: IRBuilder, non_ext: NonExtClassInfo, cdef: ClassDef, fdef: FuncDef
   431	) -> None:
   432	    # Perform the function of visit_method for methods inside non-extension classes.
   433	    name = fdef.name
   434	    sig = builder.mapper.fdef_to_sig(fdef, builder.options.strict_dunders_typing)
   435	    func_ir, func_reg = gen_func_item(builder, fdef, name, sig, cdef)
   436	    assert func_reg is not None
   437	    builder.functions.append(func_ir)
   438
   439	    if is_decorated(builder, fdef):
   440	        # The undecorated method is a generated callable class
   441	        orig_func = func_reg
   442	        func_reg = load_decorated_func(builder, fdef, orig_func)
   443
   444	    # TODO: Support property setters in non-extension classes
   445	    if fdef.is_property:
   446	        prop = builder.load_module_attr_by_fullname("builtins.property", fdef.line)
   447	        func_reg = builder.py_call(prop, [func_reg], fdef.line)
   448
   449	    elif builder.mapper.func_to_decl[fdef].kind == FUNC_CLASSMETHOD:
   450	        cls_meth = builder.load_module_attr_by_fullname("builtins.classmethod", fdef.line)
   451	        func_reg = builder.py_call(cls_meth, [func_reg], fdef.line)
   452
   453	    elif builder.mapper.func_to_decl[fdef].kind == FUNC_STATICMETHOD:
   454	        stat_meth = builder.load_module_attr_by_fullname("builtins.staticmethod", fdef.line)
   455	        func_reg = builder.py_call(stat_meth, [func_reg], fdef.line)
   456
   457	    builder.add_to_non_ext_dict(non_ext, name, func_reg, fdef.line)

   469	def load_decorated_func(builder: IRBuilder, fdef: FuncDef, orig_func_reg: Value) -> Value:
   470	    """Apply decorators to a function.
   471
   472	    Given a decorated FuncDef and an instance of the callable class
   473	    representing that FuncDef, apply the corresponding decorator
   474	    functions on that decorated FuncDef and return the decorated
   475	    function.
   476	    """
   477	    if not is_decorated(builder, fdef):
   478	        # If there are no decorators associated with the function, then just return the
   479	        # original function.
   480	        return orig_func_reg
   481
   482	    decorators = builder.fdefs_to_decorators[fdef]
   483	    func_reg = orig_func_reg
   484	    for d in reversed(decorators):
   485	        decorator = d.accept(builder.visitor)
   486	        assert isinstance(decorator, Value)
   487	        func_reg = builder.py_call(decorator, [func_reg], func_reg.line)
   488	    return func_reg

   495	def gen_glue(
   496	    builder: IRBuilder,
   497	    base_sig: FuncSignature,
   498	    target: FuncIR,
   499	    cls: ClassIR,
   500	    base: ClassIR,
   501	    fdef: FuncItem,
   502	    *,
   503	    do_py_ops: bool = False,
   504	) -> FuncIR:
   505	    """Generate glue methods that mediate between different method types in subclasses.
   506
   507	    Works on both properties and methods. See gen_glue_methods below
   508	    for more details.
   509
   510	    If do_py_ops is True, then the glue methods should use generic
   511	    C API operations instead of direct calls, to enable generating
   512	    "shadow" glue methods that work with interpreted subclasses.
   513	    """
   514	    if fdef.is_property:
   515	        return gen_glue_property(builder, base_sig, target, cls, base, fdef.line, do_py_ops)
   516	    else:
   517	        return gen_glue_method(builder, base_sig, target, cls, base, fdef.line, do_py_ops)

   526	def get_args(builder: IRBuilder, rt_args: Sequence[RuntimeArg], line: int) -> ArgInfo:
   527	    # The environment operates on Vars, so we make some up
   528	    fake_vars = [(Var(arg.name), arg.type) for arg in rt_args]
   529	    args = [
   530	        builder.read(builder.add_local_reg(var, type, is_arg=True), line)
   531	        for var, type in fake_vars
   532	    ]
   533	    arg_names = [
   534	        arg.name if arg.kind.is_named() or (arg.kind.is_optional() and not arg.pos_only) else None
   535	        for arg in rt_args
   536	    ]
   537	    arg_kinds = [arg.kind for arg in rt_args]
   538	    return ArgInfo(args, arg_names, arg_kinds)

   541	def gen_glue_method(
   542	    builder: IRBuilder,
   543	    base_sig: FuncSignature,
   544	    target: FuncIR,
   545	    cls: ClassIR,
   546	    base: ClassIR,
   547	    line: int,
   548	    do_pycall: bool,
   549	) -> FuncIR:
   550	    """Generate glue methods that mediate between different method types in subclasses.
   551
   552	    For example, if we have:
   553
   554	    class A:
   555	        def f(builder: IRBuilder, x: int) -> object: ...
   556
   557	    then it is totally permissible to have a subclass
   558
   559	    class B(A):
   560	        def f(builder: IRBuilder, x: object) -> int: ...
   561
   562	    since '(object) -> int' is a subtype of '(int) -> object' by the usual
   563	    contra/co-variant function subtyping rules.
   564
   565	    The trickiness here is that int and object have different
   566	    runtime representations in mypyc, so A.f and B.f have
   567	    different signatures at the native C level. To deal with this,
   568	    we need to generate glue methods that mediate between the
   569	    different versions by coercing the arguments and return
   570	    values.
   571
   572	    If do_pycall is True, then make the call using the C API
   573	    instead of a native call.
   574	    """
   575	    check_native_override(builder, base_sig, target.decl.sig, line)
   576
   577	    builder.enter()
   578	    builder.ret_types[-1] = base_sig.ret_type
   579
   580	    rt_args = list(base_sig.args)
   581	    if target.decl.kind == FUNC_NORMAL:
   582	        rt_args[0] = RuntimeArg(base_sig.args[0].name, RInstance(cls))
   583
   584	    arg_info = get_args(builder, rt_args, line)
   585	    args, arg_kinds, arg_names = arg_info.args, arg_info.arg_kinds, arg_info.arg_names
   586
   587	    bitmap_args = None
   588	    if base_sig.num_bitmap_args:
   589	        args = args[: -base_sig.num_bitmap_args]
   590	        arg_kinds = arg_kinds[: -base_sig.num_bitmap_args]
   591	        arg_names = arg_names[: -base_sig.num_bitmap_args]
   592	        bitmap_args = list(builder.builder.args[-base_sig.num_bitmap_args :])
   593
   594	    # We can do a passthrough *args/**kwargs with a native call, but if the
   595	    # args need to get distributed out to arguments, we just let python handle it
   596	    if any(kind.is_star() for kind in arg_kinds) and any(
   597	        not arg.kind.is_star() for arg in target.decl.sig.args
   598	    ):
   599	        do_pycall = True
   600
   601	    if do_pycall:
   602	        if target.decl.kind == FUNC_STATICMETHOD:
   603	            # FIXME: this won't work if we can do interpreted subclasses
   604	            first = builder.builder.get_native_type(cls)
   605	            st = 0
   606	        else:
   607	            first = args[0]
   608	            st = 1
   609	        retval = builder.builder.py_method_call(
   610	            first, target.name, args[st:], line, arg_kinds[st:], arg_names[st:]
   611	        )
   612	    else:
   613	        retval = builder.builder.call(
   614	            target.decl, args, arg_kinds, arg_names, line, bitmap_args=bitmap_args
   615	        )
   616	    retval = builder.coerce(retval, base_sig.ret_type, line)
   617	    builder.add(Return(retval))
   618
   619	    arg_regs, _, blocks, ret_type, _ = builder.leave()
   620	    if base_sig.num_bitmap_args:
   621	        rt_args = rt_args[: -base_sig.num_bitmap_args]
   622	    return FuncIR(
   623	        FuncDecl(
   624	            target.name + "__" + base.name + "_glue",
   625	            cls.name,
   626	            builder.module_name,
   627	            FuncSignature(rt_args, ret_type),
   628	            target.decl.kind,
   629	        ),
   630	        arg_regs,
   631	        blocks,
   632	    )

   635	def check_native_override(
   636	    builder: IRBuilder, base_sig: FuncSignature, sub_sig: FuncSignature, line: int
   637	) -> None:
   638	    """Report an error if an override changes signature in unsupported ways.
   639
   640	    Glue methods can work around many signature changes but not all of them.
   641	    """
   642	    for base_arg, sub_arg in zip(base_sig.real_args(), sub_sig.real_args()):
   643	        if base_arg.type.error_overlap:
   644	            if not base_arg.optional and sub_arg.optional and base_sig.num_bitmap_args:
   645	                # This would change the meanings of bits in the argument defaults
   646	                # bitmap, which we don't support. We'd need to do tricky bit
   647	                # manipulations to support this generally.
   648	                builder.error(
   649	                    "An argument with type "
   650	                    + f'"{base_arg.type}" cannot be given a default value in a method override',
   651	                    line,
   652	                )
   653	        if base_arg.type.error_overlap or sub_arg.type.error_overlap:
