   676	    def normalizeData(self,keepData=False):
   677	        data = list(map(abs,self.data))
   678	        s = self._sum = float(sum(data))
   679	        f = 360./s if s!=0 else 1

   685	    def makeAngles(self):
   686	        wr = getattr(self,'wedgeRecord',None)
   687	        if self.sideLabels:
   688	            startAngle = theta0(self.data, self.direction)
   689	            self.slices.label_visible = 1
   690	        else:
   691	            startAngle = self.startAngle % 360
   692	        whichWay = self.direction == "clockwise" and -1 or 1
   693	        D = [a for a in enumerate(self.normalizeData(keepData=wr))]
   694	        if self.orderMode=='alternate' and not self.sideLabels:
   695	            W = [a for a in D if abs(a[1])>=1e-5]
   696	            W.sort(key=_arcCF)
   697	            T = [[],[]]
   698	            i = 0
   699	            while W:
   700	                if i<2:
   701	                    a = W.pop(0)
   702	                else:
   703	                    a = W.pop(-1)
   704	                T[i%2].append(a)
   705	                i += 1
   706	                i %= 4
   707	            T[1].reverse()
   708	            D = T[0]+T[1] + [a for a in D if abs(a[1])<1e-5]
   709	        A = []
   710	        a = A.append
   711	        for i, angle in D:
   712	            endAngle = (startAngle + (angle * whichWay))
   713	            if abs(angle)>=_ANGLELO:
   714	                if startAngle >= endAngle:
   715	                    aa = endAngle,startAngle
   716	                else:
   717	                    aa = startAngle,endAngle
   718	            else:
   719	                aa = startAngle, None
   720	            if wr:
   721	                aa = (AngleData(aa[0],angle._data),aa[1])
   722	            startAngle = endAngle
   723	            a((i,aa))
   724	        return A
