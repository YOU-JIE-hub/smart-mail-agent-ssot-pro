  1381	        stderr.

  1410
  1411	        cmdargs = tuple(os.fspath(arg) for arg in cmdargs)
  1412	        p1 = self.path.joinpath("stdout")
  1413	        p2 = self.path.joinpath("stderr")
  1414	        print("running:", *cmdargs)
  1415	        print("     in:", Path.cwd())
  1416
  1417	        with p1.open("w", encoding="utf8") as f1, p2.open("w", encoding="utf8") as f2:
  1418	            instant = timing.Instant()
  1419	            popen = self.popen(
  1420	                cmdargs,
  1421	                stdin=stdin,
  1422	                stdout=f1,
  1423	                stderr=f2,
  1424	                close_fds=(sys.platform != "win32"),
  1425	            )
  1426	            if popen.stdin is not None:

  1430	                __tracebackhide__ = True
  1431
  1432	                timeout_message = f"{timeout} second timeout expired running: {cmdargs}"
  1433
  1434	                popen.kill()

  1437
