    74	    column = 0  # -1 if unknown

    77	    end_line = 0  # -1 if unknown
    78
    79	    # The end column number related to this error with file.
    80	    end_column = 0  # -1 if unknown

   161	class ErrorWatcher:
   162	    """Context manager that can be used to keep track of new errors recorded
   163	    around a given operation.
   164
   165	    Errors maintain a stack of such watchers. The handler is called starting
   166	    at the top of the stack, and is propagated down the stack unless filtered
   167	    out by one of the ErrorWatcher instances.
   168	    """
   169
   170	    # public attribute for the special treatment of `reveal_type` by
   171	    # `MessageBuilder.reveal_type`:
   172	    filter_revealed_type: bool
   173
   174	    def __init__(
   175	        self,
   176	        errors: Errors,
   177	        *,
   178	        filter_errors: bool | Callable[[str, ErrorInfo], bool] = False,
   179	        save_filtered_errors: bool = False,
   180	        filter_deprecated: bool = False,
   181	        filter_revealed_type: bool = False,
   182	    ) -> None:
   183	        self.errors = errors
   184	        self._has_new_errors = False
   185	        self._filter = filter_errors
   186	        self._filter_deprecated = filter_deprecated
   187	        self.filter_revealed_type = filter_revealed_type
   188	        self._filtered: list[ErrorInfo] | None = [] if save_filtered_errors else None
   189
   190	    def __enter__(self) -> Self:
   191	        self.errors._watchers.append(self)
   192	        return self
   193
   194	    def __exit__(self, exc_type: object, exc_val: object, exc_tb: object) -> Literal[False]:
   195	        last = self.errors._watchers.pop()
   196	        assert last == self
   197	        return False
   198
   199	    def on_error(self, file: str, info: ErrorInfo) -> bool:
   200	        """Handler called when a new error is recorded.
   201
   202	        The default implementation just sets the has_new_errors flag
   203
   204	        Return True to filter out the error, preventing it from being seen by other
   205	        ErrorWatcher further down the stack and from being recorded by Errors
   206	        """
   207	        if info.code == codes.DEPRECATED:
   208	            # Deprecated is not a type error, so it is handled on opt-in basis here.
   209	            return self._filter_deprecated
   210
   211	        self._has_new_errors = True
   212	        if isinstance(self._filter, bool):
   213	            should_filter = self._filter
   214	        elif callable(self._filter):
   215	            should_filter = self._filter(file, info)
   216	        else:
   217	            raise AssertionError(f"invalid error filter: {type(self._filter)}")
   218	        if should_filter and self._filtered is not None:
   219	            self._filtered.append(info)
   220
   221	        return should_filter
   222
   223	    def has_new_errors(self) -> bool:
   224	        return self._has_new_errors
   225
   226	    def filtered_errors(self) -> list[ErrorInfo]:
   227	        assert self._filtered is not None
   228	        return self._filtered

   231	class IterationDependentErrors:
   232	    """An `IterationDependentErrors` instance serves to collect the `unreachable`,
   233	    `redundant-expr`, and `redundant-casts` errors, as well as the revealed types,
   234	    handled by the individual `IterationErrorWatcher` instances sequentially applied to
   235	    the same code section."""
   236
   237	    # One set of `unreachable`, `redundant-expr`, and `redundant-casts` errors per
   238	    # iteration step.  Meaning of the tuple items: ErrorCode, message, line, column,
   239	    # end_line, end_column.
   240	    uselessness_errors: list[set[tuple[ErrorCode, str, int, int, int, int]]]
   241
   242	    # One set of unreachable line numbers per iteration step.  Not only the lines where
   243	    # the error report occurs but really all unreachable lines.
   244	    unreachable_lines: list[set[int]]
   245
   246	    # One list of revealed types for each `reveal_type` statement.  Each created list
   247	    # can grow during the iteration.  Meaning of the tuple items: line, column,
   248	    # end_line, end_column:
   249	    revealed_types: dict[tuple[int, int, int | None, int | None], list[Type]]
   250
   251	    def __init__(self) -> None:
   252	        self.uselessness_errors = []
   253	        self.unreachable_lines = []
   254	        self.revealed_types = defaultdict(list)
   255
   256	    def yield_uselessness_error_infos(self) -> Iterator[tuple[str, Context, ErrorCode]]:
   257	        """Report only those `unreachable`, `redundant-expr`, and `redundant-casts`
   258	        errors that could not be ruled out in any iteration step."""
   259
   260	        persistent_uselessness_errors = set()
   261	        for candidate in set(chain(*self.uselessness_errors)):
   262	            if all(
   263	                (candidate in errors) or (candidate[2] in lines)
   264	                for errors, lines in zip(self.uselessness_errors, self.unreachable_lines)
   265	            ):
   266	                persistent_uselessness_errors.add(candidate)
   267	        for error_info in persistent_uselessness_errors:
   268	            context = Context(line=error_info[2], column=error_info[3])
   269	            context.end_line = error_info[4]
   270	            context.end_column = error_info[5]
   271	            yield error_info[1], context, error_info[0]
   272
   273	    def yield_revealed_type_infos(self) -> Iterator[tuple[list[Type], Context]]:
   274	        """Yield all types revealed in at least one iteration step."""
   275
   276	        for note_info, types in self.revealed_types.items():
   277	            context = Context(line=note_info[0], column=note_info[1])
   278	            context.end_line = note_info[2]
   279	            context.end_column = note_info[3]
   280	            yield types, context

   283	class IterationErrorWatcher(ErrorWatcher):
   284	    """Error watcher that filters and separately collects `unreachable` errors,
   285	    `redundant-expr` and `redundant-casts` errors, and revealed types when analysing
   286	    code sections iteratively to help avoid making too-hasty reports."""
   287
   288	    iteration_dependent_errors: IterationDependentErrors
   289
   290	    def __init__(
   291	        self,
   292	        errors: Errors,
   293	        iteration_dependent_errors: IterationDependentErrors,
   294	        *,
   295	        filter_errors: bool | Callable[[str, ErrorInfo], bool] = False,
   296	        save_filtered_errors: bool = False,
   297	        filter_deprecated: bool = False,
   298	    ) -> None:
   299	        super().__init__(
   300	            errors,
   301	            filter_errors=filter_errors,
   302	            save_filtered_errors=save_filtered_errors,
   303	            filter_deprecated=filter_deprecated,
   304	        )
   305	        self.iteration_dependent_errors = iteration_dependent_errors
   306	        iteration_dependent_errors.uselessness_errors.append(set())
   307	        iteration_dependent_errors.unreachable_lines.append(set())
   308
   309	    def on_error(self, file: str, info: ErrorInfo) -> bool:
   310	        """Filter out the "iteration-dependent" errors and notes and store their
   311	        information to handle them after iteration is completed."""
   312
   313	        iter_errors = self.iteration_dependent_errors
   314
   315	        if info.code in (codes.UNREACHABLE, codes.REDUNDANT_EXPR, codes.REDUNDANT_CAST):
   316	            iter_errors.uselessness_errors[-1].add(
   317	                (info.code, info.message, info.line, info.column, info.end_line, info.end_column)
   318	            )
   319	            if info.code == codes.UNREACHABLE:
   320	                iter_errors.unreachable_lines[-1].update(range(info.line, info.end_line + 1))
   321	            return True
   322
   323	        return super().on_error(file, info)
