def smart_truncate(text: str, max_chars: int = 1000) -> str:
    if text is None:
        return ""
    s = str(text)
    if len(s) <= max_chars:
        return s
    # 保留結尾 "...\\n"
    keep = max(0, max_chars - 4)
    return s[:keep] + "...\n"
