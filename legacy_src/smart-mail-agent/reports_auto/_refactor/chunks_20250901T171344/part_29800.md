typ in call.arg_types[i - is_method]:
   352	                    # Collect all the types except for implicit anys
   353	                    if not is_implicit_any(typ):
   354	                        all_arg_types.append(typ)
   355	            all_use_types = []
   356	            for typ in uses[i]:
   357	                # Collect all the types except for implicit anys
   358	                if not is_implicit_any(typ):
   359	                    all_use_types.append(typ)
   360	            # Add in any default argument types
   361	            default = defaults[i]
   362	            if default:
   363	                all_arg_types.append(default)
   364	                if all_use_types:
   365	                    all_use_types.append(default)
   366
   367	            arg_types = []
   368
   369	            if all_arg_types and all(
   370	                isinstance(get_proper_type(tp), NoneType) for tp in all_arg_types
   371	            ):
   372	                arg_types.append(
   373	                    UnionType.make_union([all_arg_types[0], AnyType(TypeOfAny.explicit)])
   374	                )
   375	            elif all_arg_types:
   376	                arg_types.extend(generate_type_combinations(all_arg_types))
   377	            else:
   378	                arg_types.append(AnyType(TypeOfAny.explicit))
   379
   380	            if all_use_types:
   381	                # This is a meet because the type needs to be compatible with all the uses
   382	                arg_types.append(meet_type_list(all_use_types))
   383
   384	            types.append(arg_types)
   385	        return types

   393	    def get_guesses(
   394	        self,
   395	        is_method: bool,
   396	        base: CallableType,
   397	        defaults: list[Type | None],
   398	        callsites: list[Callsite],
   399	        uses: list[list[Type]],
   400	    ) -> list[CallableType]:
   401	        """Compute a list of guesses for a function's type.
   402
   403	        This focuses just on the argument types, and doesn't change the provided return type.
   404	        """
   405	        options = self.get_args(is_method, base, defaults, callsites, uses)
   406
   407	        # Take the first `max_guesses` guesses.
   408	        product = itertools.islice(itertools.product(*options), 0, self.max_guesses)
   409	        return [refine_callable(base, base.copy_modified(arg_types=list(x))) for x in product]

   411	    def get_callsites(self, func: FuncDef) -> tuple[list[Callsite], list[str]]:
   412	        """Find all call sites of a function."""
   413	        new_type = self.get_starting_type(func)
   414
   415	        collector_plugin = SuggestionPlugin(func.fullname)
   416
   417	        self.plugin._plugins.insert(0, collector_plugin)
   418	        try:
   419	            errors = self.try_type(func, new_type)

   421	            self.plugin._plugins.pop(0)

   423	        return collector_plugin.mystery_hits, errors

   438	    def find_best(self, func: FuncDef, guesses: list[CallableType]) -> tuple[CallableType, int]:
   439	        """From a list of possible function types, find the best one.
   440
   441	        For best, we want the fewest errors, then the best "score" from score_callable.
   442	        """
   443	        if not guesses:
   444	            raise SuggestionFailure("No guesses that match criteria!")
   445	        errors = {guess: self.try_type(func, guess) for guess in guesses}
   446	        best = min(guesses, key=lambda s: (count_errors(errors[s]), self.score_callable(s)))
   447	        return best, count_errors(errors[best])

   449	    def get_guesses_from_parent(self, node: FuncDef) -> list[CallableType]:
   450	        """Try to get a guess of a method type from a parent class."""
   451	        if not node.info:
   452	            return []
   453
   454	        for parent in node.info.mro[1:]:
   455	            pnode = parent.names.get(node.name)
   456	            if pnode and isinstance(pnode.node, (FuncDef, Decorator)):
   457	                typ = get_proper_type(pnode.node.type)
   458	                # FIXME: Doesn't work right with generic types
   459	                if isinstance(typ, CallableType) and len(typ.arg_types) == len(node.arguments):
   460	                    # Return the first thing we find, since it probably doesn't make sense
   461	                    # to grab things further up in the chain if an earlier parent has it.
   462	                    return [typ]
   463
   464	        return []

   466	    def get_suggestion(self, mod: str, node: FuncDef) -> PyAnnotateSignature:
   467	        """Compute a suggestion for a function.
   468
   469	        Return the type and whether the first argument should be ignored.
   470	        """
   471	        graph = self.graph
   472	        callsites, orig_errors = self.get_callsites(node)
   473	        uses = get_arg_uses(self.manager.all_types, node)
   474
   475	        if self.no_errors and orig_errors:
   476	            raise SuggestionFailure("Function does not typecheck.")
   477
   478	        is_method = bool(node.info) and not node.is_static
   479
   480	        with state.strict_optional_set(graph[mod].options.strict_optional):
   481	            guesses = self.get_guesses(
   482	                is_method,
   483	                self.get_starting_type(node),
   484	                self.get_default_arg_types(node),
   485	                callsites,
   486	                uses,
   487	            )
   488	        guesses += self.get_guesses_from_parent(node)
   489	        guesses = self.filter_options(guesses, is_method, ignore_return=True)
   490	        best, _ = self.find_best(node, guesses)
   491
   492	        # Now try to find the return type!
   493	        self.try_type(node, best)
   494	        returns = get_return_types(self.manager.all_types, node)
   495	        with state.strict_optional_set(graph[mod].options.strict_optional):
   496	            if returns:
   497	                ret_types = generate_type_combinations(returns)
   498	            else:
   499	                ret_types = [NoneType()]
   500
   501	        guesses = [best.copy_modified(ret_type=refine_type(best.ret_type, t)) for t in ret_types]
   502	        guesses = self.filter_options(guesses, is_method, ignore_return=False)
   503	        best, errors = self.find_best(node, guesses)
   504
   505	        if self.no_errors and errors:
   506	            raise SuggestionFailure("No annotation without errors")
   507
   508	        return self.pyannotate_signature(mod, is_method, best)

   511	        self,

   513	        arg_names: list[list[str | None]],

   530	    def find_node(self, key: str) -> tuple[str, str, FuncDef]:
   531	        """From a target name, return module/target names and the func def.
   532
   533	        The 'key' argument can be in one of two formats:
   534	        * As the function full name, e.g., package.module.Cls.method
   535	        * As the function location as file and line separated by column,
   536	          e.g., path/to/file.py:42
   537	        """
   538	        # TODO: Also return OverloadedFuncDef -- currently these are ignored.
   539	        node: SymbolNode | None = None
   540	        if ":" in key:
   541	            # A colon might be part of a drive name on Windows (like `C:/foo/bar`)
   542	            # and is also used as a delimiter between file path and lineno.
   543	            # If a colon is there for any of those reasons, it must be a file+line
   544	            # reference.
   545	            platform_key_count = 2 if sys.platform == "win32" else 1
   546	            if key.count(":") > platform_key_count:
   547	                raise SuggestionFailure(
   548	                    "Malformed location for function: {}. Must be either"
   549	                    " package.module.Class.method or path/to/file.py:line".format(key)
   550	                )
   551	            file, line = key.rsplit(":", 1)
   552	            if not line.isdigit():
   553	                raise SuggestionFailure(f"Line number must be a number. Got {line}")
   554	            line_number = int(line)
   555	            modname, node = self.find_node_by_file_and_line(file, line_number)
   556	            tail = node.fullname[len(modname) + 1 :]  # add one to account for '.'
   557	        else:
   558	            target = split_target(self.fgmanager.graph, key)
   559	            if not target:
   560	                raise SuggestionFailure(f"Cannot find module for {key}")
   561	            modname, tail = target
   562	            node = self.find_node_by_module_and_name(modname, tail)
   563
   564	        if isinstance(node, Decorator):
   565	            node = self.extract_from_decorator(node)
   566	            if not node:
   567	                raise SuggestionFailure(f"Object {key} is a decorator we can't handle")
   568
   569	        if not isinstance(node, FuncDef):
   570	            raise SuggestionFailure(f"Object {key} is not a function")
   571
   572	        return modname, tail, node

   574	    def find_node_by_module_and_name(self, modname: str, tail: str) -> SymbolNode | None:
   575	        """Find symbol node by module id and qualified name.
   576
   577	        Raise SuggestionFailure if can't find one.
   578	        """
   579	        tree = self.ensure_loaded(self.fgmanager.graph[modname])
   580
   581	        # N.B. This is reimplemented from update's lookup_target
   582	        # basically just to produce better error messages.
   583
   584	        names: SymbolTable = tree.names
   585
   586	        # Look through any classes
   587	        components = tail.split(".")
   588	        for i, component in enumerate(components[:-1]):
   589	            if component not in names:
   590	                raise SuggestionFailure(
   591	                    "Unknown class {}.{}".format(modname, ".".join(components[: i + 1]))
   592	                )
   593	            node: SymbolNode | None = names[component].node
   594	            if not isinstance(node, TypeInfo):
   595	                raise SuggestionFailure(
   596	                    "Object {}.{} is not a class".format(modname, ".".join(components[: i + 1]))
   597	                )
   598	            names = node.names
   599
   600	        # Look for the actual function/method
   601	        funcname = components[-1]
   602	        if funcname not in names:
   603	            key = modname + "." + tail
   604	            raise SuggestionFailure(
   605	                "Unknown {} {}".format("method" if len(components) > 1 else "function", key)
   606	            )
   607	        return names[funcname].node

   609	    def find_node_by_file_and_line(self, file: str, line: int) -> tuple[str, SymbolNode]:
   610	        """Find symbol node by path to file and line number.
   611
   612	        Find the first function declared *before or on* the line number.
   613
   614	        Return module id and the node found. Raise SuggestionFailure if can't find one.
   615	        """
   616	        if not any(file.endswith(ext) for ext in PYTHON_EXTENSIONS):
   617	            raise SuggestionFailure("Source file is not a Python file")
   618	        try:
   619	            modname, _ = self.finder.crawl_up(os.path.normpath(file))
   620	        except InvalidSourceList as e:
   621	            raise SuggestionFailure("Invalid source file name: " + file) from e
   622	        if modname not in self.graph:
   623	            raise SuggestionFailure("Unknown module: " + modname)
   624	        # We must be sure about any edits in this file as this might affect the line numbers.
   625	        tree = self.ensure_loaded(self.fgmanager.graph[modname], force=True)
   626	        node: SymbolNode | None = None
   627	        closest_line: int | None = None
   628	        # TODO: Handle nested functions.
   629	        for _, sym, _ in tree.local_definitions():
   630	            if isinstance(sym.node, (FuncDef, Decorator)):
   631	                sym_line = sym.node.line
   632	            # TODO: add support for OverloadedFuncDef.
   633	            else:
   634	                continue
   635
   636	            # We want the closest function above the specified line
   637	            if sym_line <= line and (closest_line is None or sym_line > closest_line):
   638	                closest_line = sym_line
   639	                node = sym.node
   640	        if not node:
   641	            raise SuggestionFailure(f"Cannot find a function at line {line}")
   642	        return modname, node

   644	    def extract_from_decorator(self, node: Decorator) -> FuncDef | None:
   645	        for dec in node.decorators:
   646	            typ = None
   647	            if isinstance(dec, RefExpr) and isinstance(dec.node, (Var, FuncDef)):
   648	                typ = get_proper_type(dec.node.type)
   649	            elif (
   650	                isinstance(dec, CallExpr)
   651	                and isinstance(dec.callee, RefExpr)
   652	                and isinstance(dec.callee.node, (Decorator, FuncDef, Var))
   653	                and isinstance((call_tp := get_proper_type(dec.callee.node.type)), CallableType)
   654	            ):
   655	                typ = get_proper_type(call_tp.ret_type)
   656
   657	            if isinstance(typ, Instance):
   658	                call_method = typ.type.get_method("__call__")
   659	                if isinstance(call_method, FuncDef) and isinstance(call_method.type, FunctionLike):
   660	                    typ = bind_self(call_method.type, None)
   661
   662	            if not isinstance(typ, FunctionLike):
   663	                return None
   664	            for ct in typ.items:
   665	                if not (
   666	                    len(ct.arg_types) == 1
   667	                    and isinstance(ct.arg_types[0], TypeVarType)
   668	                    and ct.arg_types[0] == ct.ret_type
   669	                ):
   670	                    return None
   671
   672	        return node.func

   674	    def try_type(self, func: FuncDef, typ: ProperType) -> list[str]:
   675	        """Recheck a function while assuming it has type typ.
   676
   677	        Return all error messages.
   678	        """
   679	        old = func.unanalyzed_type
   680	        # During reprocessing, unanalyzed_type gets copied to type (by aststrip).
   681	        # We set type to None to ensure that the type always changes during
   682	        # reprocessing.
   683	        func.type = None
   684	        func.unanalyzed_type = typ
   685	        try:
   686	            res = self.fgmanager.trigger(func.fullname)
   687	            # if res:
   688	            #     print('===', typ)
   689	            #     print('\n'.join(res))
   690	            return res
   691	        finally:
   692	            func.unanalyzed_type = old

   751	    def score_type(self, t: Type, arg_pos: bool) -> int:
   752	        """Generate a score for a type that we use to pick which type to use.
   753
   754	        Lower is better, prefer non-union/non-any types. Don't penalize optionals.
   755	        """
   756	        t = get_proper_type(t)
   757	        if isinstance(t, AnyType):
   758	            return 20
   759	        if arg_pos and isinstance(t, NoneType):
   760	            return 20
   761	        if isinstance(t, UnionType):
   762	            if any(isinstance(get_proper_type(x), AnyType) for x in t.items):
   763	                return 20
   764	            if any(has_any_type(x) for x in t.items):
   765	                return 15
   766	            if not is_overlapping_none(t):
   767	                return 10
   768	        if isinstance(t, CallableType) and (has_any_type(t) or is_tricky_callable(t)):
   769	            return 10
   770	        return 0
