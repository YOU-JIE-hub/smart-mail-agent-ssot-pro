    88	    s = list(str)
    89	    s.reverse()

    92	    i = 0; v = 1; c = 0
    93	    while i < len(s):
    94	        c = c + v * _patterns[s[i]][1]
    95	        i = i + 1; v = v + 1
    96	        if v > 20:
    97	            v = 1
    98	    s.insert(0, _charsbyval[c % 47])

   100	    # compute 'K' checksum
   101	    i = 0; v = 1; c = 0
   102	    while i < len(s):
   103	        c = c + v * _patterns[s[i]][1]
   104	        i = i + 1; v = v + 1
   105	        if v > 15:
   106	            v = 1
   107	    s.insert(0, _charsbyval[c % 47])
