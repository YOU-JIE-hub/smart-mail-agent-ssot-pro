    mx.context,
   807	            object_type=descriptor_type,
   808	            callable_name=callable_name,
   809	        )
   810
   811	    # And now we in fact type check the call, to show errors related to wrong arguments
   812	    # count, etc., replacing the type context for non-overloaded setters only.
   813	    inferred_dunder_set_type = get_proper_type(inferred_dunder_set_type)
   814	    if isinstance(inferred_dunder_set_type, CallableType):
   815	        type_context = TempNode(AnyType(TypeOfAny.special_form), context=mx.context)
   816	    mx.chk.expr_checker.check_call(
   817	        dunder_set_type,
   818	        [TempNode(instance_type, context=mx.context), type_context],
   819	        [ARG_POS, ARG_POS],
   820	        mx.context,
   821	        object_type=descriptor_type,
   822	        callable_name=callable_name,
   823	    )
   824
   825	    # Search for possible deprecations:
   826	    mx.chk.check_deprecated(dunder_set, mx.context)
   827	    mx.chk.warn_deprecated_overload_item(
   828	        dunder_set, mx.context, target=inferred_dunder_set_type, selftype=descriptor_type
   829	    )
   830
   831	    # In the following cases, a message already will have been recorded in check_call.
   832	    if (not isinstance(inferred_dunder_set_type, CallableType)) or (
   833	        len(inferred_dunder_set_type.arg_types) < 2
   834	    ):
   835	        return AnyType(TypeOfAny.from_error)
   836	    return inferred_dunder_set_type.arg_types[1]

   839	def is_instance_var(var: Var) -> bool:
   840	    """Return if var is an instance variable according to PEP 526."""
   841	    return (
   842	        # check the type_info node is the var (not a decorated function, etc.)
   843	        var.name in var.info.names
   844	        and var.info.names[var.name].node is var
   845	        and not var.is_classvar
   846	        # variables without annotations are treated as classvar
   847	        and not var.is_inferred
   848	    )

   851	def analyze_var(
   852	    name: str,
   853	    var: Var,
   854	    itype: Instance,
   855	    mx: MemberContext,
   856	    *,
   857	    implicit: bool = False,
   858	    is_trivial_self: bool = False,
   859	) -> Type:
   860	    """Analyze access to an attribute via a Var node.
   861
   862	    This is conceptually part of analyze_member_access and the arguments are similar.
   863	    itype is the instance type in which attribute should be looked up
   864	    original_type is the type of E in the expression E.var
   865	    if implicit is True, the original Var was created as an assignment to self
   866	    if is_trivial_self is True, we can use fast path for bind_self().
   867	    """
   868	    # Found a member variable.
   869	    original_itype = itype
   870	    itype = map_instance_to_supertype(itype, var.info)
   871	    if var.is_settable_property and mx.is_lvalue:
   872	        typ: Type | None = var.setter_type
   873	        if typ is None and var.is_ready:
   874	            # Existing synthetic properties may not set setter type. Fall back to getter.
   875	            typ = var.type
   876	    else:
   877	        typ = var.type
   878	    if typ:
   879	        if isinstance(typ, PartialType):
   880	            return mx.chk.handle_partial_var_type(typ, mx.is_lvalue, var, mx.context)
   881	        if mx.is_lvalue and not mx.suppress_errors:
   882	            if var.is_property and not var.is_settable_property:
   883	                mx.msg.read_only_property(name, itype.type, mx.context)
   884	            if var.is_classvar:
   885	                mx.msg.cant_assign_to_classvar(name, mx.context)
   886	        # This is the most common case for variables, so start with this.
   887	        result = expand_without_binding(typ, var, itype, original_itype, mx)
   888
   889	        # A non-None value indicates that we should actually bind self for this variable.
   890	        call_type: ProperType | None = None
   891	        if var.is_initialized_in_class and (not is_instance_var(var) or mx.is_operator):
   892	            typ = get_proper_type(typ)
   893	            if isinstance(typ, FunctionLike) and not typ.is_type_obj():
   894	                call_type = typ
   895	            elif var.is_property:
   896	                deco_mx = mx.copy_modified(original_type=typ, self_type=typ, is_lvalue=False)
   897	                call_type = get_proper_type(_analyze_member_access("__call__", typ, deco_mx))
   898	            else:
   899	                call_type = typ
   900
   901	        # Bound variables with callable types are treated like methods
   902	        # (these are usually method aliases like __rmul__ = __mul__).
   903	        if isinstance(call_type, FunctionLike) and not call_type.is_type_obj():
   904	            if mx.is_lvalue and not var.is_property and not mx.suppress_errors:
   905	                mx.msg.cant_assign_to_method(mx.context)
   906
   907	        # Bind the self type for each callable component (when needed).
   908	        if call_type and not var.is_staticmethod:
   909	            bound_items = []
   910	            for ct in call_type.items if isinstance(call_type, UnionType) else [call_type]:
   911	                p_ct = get_proper_type(ct)
   912	                if isinstance(p_ct, FunctionLike) and (not p_ct.bound() or var.is_property):
   913	                    item = expand_and_bind_callable(p_ct, var, itype, name, mx, is_trivial_self)
   914	                else:
   915	                    item = expand_without_binding(ct, var, itype, original_itype, mx)
   916	                bound_items.append(item)
   917	            result = UnionType.make_union(bound_items)
   918	    else:
   919	        if not var.is_ready and not mx.no_deferral:
   920	            mx.not_ready_callback(var.name, mx.context)
   921	        # Implicit 'Any' type.
   922	        result = AnyType(TypeOfAny.special_form)
   923	    fullname = f"{var.info.fullname}.{name}"
   924	    hook = mx.chk.plugin.get_attribute_hook(fullname)
   925	    if result and not (implicit or var.info.is_protocol and is_instance_var(var)):
   926	        result = analyze_descriptor_access(result, mx)
   927	    if hook:
   928	        result = hook(
   929	            AttributeContext(
   930	                get_proper_type(mx.original_type), result, mx.is_lvalue, mx.context, mx.chk
   931	            )
   932	        )
   933	    return result

   936	def expand_without_binding(
   937	    typ: Type, var: Var, itype: Instance, original_itype: Instance, mx: MemberContext
   938	) -> Type:
   939	    if not mx.preserve_type_var_ids:
   940	        typ = freshen_all_functions_type_vars(typ)
   941	    typ = expand_self_type_if_needed(typ, mx, var, original_itype)
   942	    expanded = expand_type_by_instance(typ, itype)
   943	    freeze_all_type_vars(expanded)
   944	    return expanded

   947	def expand_and_bind_callable(
   948	    functype: FunctionLike,
   949	    var: Var,
   950	    itype: Instance,
   951	    name: str,
   952	    mx: MemberContext,
   953	    is_trivial_self: bool,
   954	) -> Type:
   955	    if not mx.preserve_type_var_ids:
   956	        functype = freshen_all_functions_type_vars(functype)
   957	    typ = get_proper_type(expand_self_type(var, functype, mx.original_type))
   958	    assert isinstance(typ, FunctionLike)
   959	    if is_trivial_self:
   960	        typ = bind_self_fast(typ, mx.self_type)
   961	    else:
   962	        typ = check_self_arg(typ, mx.self_type, var.is_classmethod, mx.context, name, mx.msg)
   963	        typ = bind_self(typ, mx.self_type, var.is_classmethod)
   964	    expanded = expand_type_by_instance(typ, itype)
   965	    freeze_all_type_vars(expanded)
   966	    if not var.is_property:
   967	        return expanded
   968	    # TODO: a decorated property can result in Overloaded here.
   969	    assert isinstance(expanded, CallableType)
   970	    if var.is_settable_property and mx.is_lvalue and var.setter_type is not None:
   971	        # TODO: use check_call() to infer better type, same as for __set__().
   972	        if not expanded.arg_types:
   973	            # This can happen when accessing invalid property from its own body,
   974	            # error will be reported elsewhere.
   975	            return AnyType(TypeOfAny.from_error)
   976	        return expanded.arg_types[0]
   977	    else:
   978	        return expanded.ret_type

   981	def expand_self_type_if_needed(
   982	    t: Type, mx: MemberContext, var: Var, itype: Instance, is_class: bool = False
   983	) -> Type:
   984	    """Expand special Self type in a backwards compatible manner.
   985
   986	    This should ensure that mixing old-style and new-style self-types work
   987	    seamlessly. Also, re-bind new style self-types in subclasses if needed.
   988	    """
   989	    original = get_proper_type(mx.self_type)
   990	    if not (mx.is_self or mx.is_super):
   991	        repl = mx.self_type
   992	        if is_class:
   993	            if isinstance(original, TypeType):
   994	                repl = original.item
   995	            elif isinstance(original, CallableType):
   996	                # Problematic access errors should have been already reported.
   997	                repl = erase_typevars(original.ret_type)
   998	            else:
   999	                repl = itype
  1000	        return expand_self_type(var, t, repl)
  1001	    elif supported_self_type(
  1002	        # Support compatibility with plain old style T -> T and Type[T] -> T only.
  1003	        get_proper_type(mx.self_type),
  1004	        allow_instances=False,
  1005	        allow_callable=False,
  1006	    ):
  1007	        repl = mx.self_type
  1008	        if is_class and isinstance(original, TypeType):
  1009	            repl = original.item
  1010	        return expand_self_type(var, t, repl)
  1011	    elif (
  1012	        mx.is_self
  1013	        and itype.type != var.info
  1014	        # If an attribute with Self-type was defined in a supertype, we need to
  1015	        # rebind the Self type variable to Self type variable of current class...
  1016	        and itype.type.self_type is not None
  1017	        # ...unless `self` has an explicit non-trivial annotation.
  1018	        and itype == mx.chk.scope.active_self_type()
  1019	    ):
  1020	        return expand_self_type(var, t, itype.type.self_type)
  1021	    else:
  1022	        return t

  1025	def check_self_arg(
  1026	    functype: FunctionLike,
  1027	    dispatched_arg_type: Type,
  1028	    is_classmethod: bool,
  1029	    context: Context,
  1030	    name: str,
  1031	    msg: MessageBuilder,
  1032	) -> FunctionLike:
  1033	    """Check that an instance has a valid type for a method with annotated 'self'.
  1034
  1035	    For example if the method is defined as:
  1036	        class A:
  1037	            def f(self: S) -> T: ...
  1038	    then for 'x.f' we check that type(x) <: S. If the method is overloaded, we select
  1039	    only overloads items that satisfy this requirement. If there are no matching
  1040	    overloads, an error is generated.
  1041	    """
  1042	    items = functype.items
  1043	    if not items:
  1044	        return functype
  1045	    new_items = []
  1046	    if is_classmethod:
  1047	        dispatched_arg_type = TypeType.make_normalized(dispatched_arg_type)
  1048
  1049	    for item in items:
  1050	        if not item.arg_types or item.arg_kinds[0] not in (ARG_POS, ARG_STAR):
  1051	            # No positional first (self) argument (*args is okay).
  1052	            msg.no_formal_self(name, item, context)
  1053	            # This is pretty bad, so just return the original signature if
  1054	            # there is at least one such error.
  1055	            return functype
  1056	        else:
  1057	            selfarg = get_proper_type(item.arg_types[0])
  1058	            # This matches similar special-casing in bind_self(), see more details there.
  1059	            self_callable = name == "__call__" and isinstance(selfarg, CallableType)
  1060	            if self_callable or subtypes.is_subtype(
  1061	                dispatched_arg_type,
  1062	                # This level of erasure matches the one in checker.check_func_def(),
  1063	                # better keep these two checks consistent.
  1064	                erase_typevars(erase_to_bound(selfarg)),
  1065	                # This is to work around the fact that erased ParamSpec and TypeVarTuple
  1066	                # callables are not always compatible with non-erased ones both ways.
  1067	                always_covariant=any(
  1068	                    not isinstance(tv, TypeVarType) for tv in get_all_type_vars(selfarg)
  1069	                ),
  1070	                ignore_pos_arg_names=True,
  1071	            ):
  1072	                new_items.append(item)
  1073	            elif isinstance(selfarg, ParamSpecType):
  1074	                # TODO: This is not always right. What's the most reasonable thing to do here?
  1075	                new_items.append(item)
  1076	            elif isinstance(selfarg, TypeVarTupleType):
  1077	                raise NotImplementedError
  1078	    if not new_items:
  1079	        # Choose first item for the message (it may be not very helpful for overloads).
  1080	        msg.incompatible_self_argument(
  1081	            name, dispatched_arg_type, items[0], is_classmethod, context
  1082	        )
  1083	        return functype
  1084	    if len(new_items) == 1:
  1085	        return new_items[0]
  1086	    return Overloaded(new_items)

  1089	def analyze_class_attribute_access(
  1090	    itype: Instance,
  1091	    name: str,
  1092	    mx: MemberContext,
  1093	    *,
  1094	    mcs_fallback: Instance,
  1095	    override_info: TypeInfo | None = None,
  1096	    original_vars: Sequence[TypeVarLikeType] | None = None,
  1097	) -> Type | None:
  1098	    """Analyze access to an attribute on a class object.
  1099
  1100	    itype is the return type of the class object callable, original_type is the type
  1101	    of E in the expression E.var, original_vars are type variables of the class callable
  1102	    (for generic classes).
  1103	    """
  1104	    info = itype.type
  1105	    if override_info:
  1106	        info = override_info
  1107
  1108	    fullname = f"{info.fullname}.{name}"
  1109	    hook = mx.chk.plugin.get_class_attribute_hook(fullname)
  1110
  1111	    node = info.get(name)
  1112	    if not node:
  1113	        if itype.extra_attrs and name in itype.extra_attrs.attrs:
  1114	            # For modules use direct symbol table lookup.
  1115	            if not itype.extra_attrs.mod_name:
  1116	                return itype.extra_attrs.attrs[name]
  1117	        if info.fallback_to_any or info.meta_fallback_to_any:
  1118	            return apply_class_attr_hook(mx, hook, AnyType(TypeOfAny.special_form))
  1119	        return None
  1120
  1121	    if (
  1122	        isinstance(node.node, Var)
  1123	        and not node.node.is_classvar
  1124	        and not hook
  1125	        and mcs_fallback.type.get(name)
  1126	    ):
  1127	        # If the same attribute is declared on the metaclass and the class but with different types,
  1128	        # and the attribute on the class is not a ClassVar,
  1129	        # the type of the attribute on the metaclass should take priority
  1130	        # over the type of the attribute on the class,
  1131	        # when the attribute is being accessed from the class object itself.
  1132	        #
  1133	        # Return `None` here to signify that the name should be looked up
  1134	        # on the class object itself rather than the instance.
  1135	        return None
  1136
  1137	    mx.chk.warn_deprecated(node.node, mx.context)
  1138
  1139	    is_decorated = isinstance(node.node, Decorator)
  1140	    is_method = is_decorated or isinstance(node.node, FuncBase)
  1141	    if mx.is_lvalue and not mx.suppress_errors:
  1142	        if is_method:
  1143	            mx.msg.cant_assign_to_method(mx.context)
  1144	        if isinstance(node.node, TypeInfo):
  1145	            mx.fail(message_registry.CANNOT_ASSIGN_TO_TYPE)
  1146
  1147	    # Refuse class attribute access if slot defined
  1148	    if info.slots and name in info.slots:
  1149	        mx.fail(message_registry.CLASS_VAR_CONFLICTS_SLOTS.format(name))
  1150
  1151	    # If a final attribute was declared on `self` in `__init__`, then it
  1152	    # can't be accessed on the class object.
  1153	    if node.implicit and isinstance(node.node, Var) and node.node.is_final:
  1154	        mx.fail(message_registry.CANNOT_ACCESS_FINAL_INSTANCE_ATTR.format(node.node.name))
  1155
  1156	    # An assignment to final attribute on class object is also always an error,
  1157	    # independently of types.
  1158	    if mx.is_lvalue and not mx.chk.get_final_context():
  1159	        check_final_member(name, info, mx.msg, mx.context)
  1160
  1161	    if info.is_enum and not (mx.is_lvalue or is_decorated or is_method):
  1162	        enum_class_attribute_type = analyze_enum_class_attribute_access(itype, name, mx)
  1163	        if enum_class_attribute_type:
  1164	            return apply_class_attr_hook(mx, hook, enum_class_attribute_type)
  1165
  1166	    t = node.type
  1167	    if t:
  1168	        if isinstance(t, PartialType):
  1169	            symnode = node.node
  1170	            assert isinstance(symnode, Var)
  1171	            return apply_class_attr_hook(
  1172	                mx, hook, mx.chk.handle_partial_var_type(t, mx.is_lvalue, symnode, mx.context)
  1173	            )
  1174
  1175	        # Find the class where method/variable was defined.
  1176	        if isinstance(node.node, Decorator):
  1177	            super_info: TypeInfo | None = node.node.var.info
  1178	        elif isinstance(node.node, (Var, SYMBOL_FUNCBASE_TYPES)):
  1179	            super_info = node.node.info
  1180	        else:
  1181	            super_info = None
  1182
  1183	        # Map the type to how it would look as a defining class. For example:
  1184	        #     class C(Generic[T]): ...
  1185	        #     class D(C[Tuple[T, S]]): ...
  1186	        #     D[int, str].method()
  1187	        # Here itype is D[int, str], isuper is C[Tuple[int, str]].
  1188	        if not super_info:
  1189	            isuper = None
  1190	        else:
  1191	            isuper = map_instance_to_supertype(itype, super_info)
  1192
  1193	        if isinstance(node.node, Var):
  1194	            assert isuper is not None
  1195	            # Check if original variable type has type variables. For example:
  1196	            #     class C(Generic[T]):
  1197	            #         x: T
  1198	            #     C.x  # Error, ambiguous access
  1199	            #     C[int].x  # Also an error, since C[int] is same as C at runtime
  1200	            # Exception is Self type wrapped in ClassVar, that is safe.
  1201	            def_vars = set(node.node.info.defn.type_vars)
  1202	            if not node.node.is_classvar and node.node.info.self_type:
  1203	                def_vars.add(node.node.info.self_type)
  1204	            # TODO: should we include ParamSpec etc. here (i.e. use get_all_type_vars)?
  1205	            typ_vars = set(get_type_vars(t))
  1206	            if def_vars & typ_vars:
  1207	                # Exception: access on Type[...], including first argument of class methods is OK.
  1208	                if not isinstance(get_proper_type(mx.original_type), TypeType) or node.implicit:
  1209	                    if node.node.is_classvar:
  1210	                        message = message_registry.GENERIC_CLASS_VAR_ACCESS
  1211	                    else:
  1212	                        message = message_registry.GENERIC_INSTANCE_VAR_CLASS_ACCESS
  1213	                    mx.fail(message)
  1214	            t = expand_self_type_if_needed(t, mx, node.node, itype, is_class=True)
  1215	            # Erase non-mapped variables, but keep mapped ones, even if there is an error.
  1216	            # In the above example this means that we infer following types:
  1217	            #     C.x -> Any
  1218	            #     C[int].x -> int
  1219	            t = erase_typevars(expand_type_by_instance(t, isuper), {tv.id for tv in def_vars})
  1220
  1221	        is_cla
