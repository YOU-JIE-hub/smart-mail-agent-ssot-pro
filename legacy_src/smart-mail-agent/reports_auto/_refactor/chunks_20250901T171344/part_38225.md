   278	def retry_on_error(func: Callable[[], Any], max_wait: float = 1.0) -> None:
   279	    """Retry callback with exponential backoff when it raises OSError.
   280
   281	    If the function still generates an error after max_wait seconds, propagate
   282	    the exception.
   283
   284	    This can be effective against random file system operation failures on
   285	    Windows.
   286	    """
   287	    t0 = time.time()
   288	    wait_time = 0.01
   289	    while True:
   290	        try:
   291	            func()
   292	            return
   293	        except OSError:
   294	            wait_time = min(wait_time * 2, t0 + max_wait - time.time())
   295	            if wait_time <= 0.01:
   296	                # Done enough waiting, the error seems persistent.
   297	                raise
   298	            time.sleep(wait_time)

   374	def write_and_fudge_mtime(content: str, target_path: str) -> None:
   375	    # In some systems, mtime has a resolution of 1 second which can
   376	    # cause annoying-to-debug issues when a file has the same size
   377	    # after a change. We manually set the mtime to circumvent this.
   378	    # Note that we increment the old file's mtime, which guarantees a
   379	    # different value, rather than incrementing the mtime after the
   380	    # copy, which could leave the mtime unchanged if the old file had
   381	    # a similarly fudged mtime.
   382	    new_time = None
   383	    if os.path.isfile(target_path):
   384	        new_time = os.stat(target_path).st_mtime + 1
   385
   386	    dir = os.path.dirname(target_path)
   387	    os.makedirs(dir, exist_ok=True)
   388	    with open(target_path, "w", encoding="utf-8") as target:
   389	        target.write(content)
   390
   391	    if new_time:
   392	        os.utime(target_path, times=(new_time, new_time))
