    29	def validate_report(arg):
    30	    file_choices = ['annotate', 'html', 'xml', 'json', 'lcov']
    31	    term_choices = ['term', 'term-missing']
    32	    term_modifier_choices = ['skip-covered']
    33	    all_choices = term_choices + file_choices
    34	    values = arg.split(':', 1)
    35	    report_type = values[0]
    36	    if report_type not in [*all_choices, '']:
    37	        msg = f'invalid choice: "{arg}" (choose from "{all_choices}")'
    38	        raise argparse.ArgumentTypeError(msg)

    40	    if report_type == 'lcov' and coverage.version_info <= (6, 3):
    41	        raise argparse.ArgumentTypeError('LCOV output is only supported with coverage.py >= 6.3')

    43	    if len(values) == 1:
    44	        return report_type, None
    45
    46	    report_modifier = values[1]
    47	    if report_type in term_choices and report_modifier in term_modifier_choices:
    48	        return report_type, report_modifier

    50	    if report_type not in file_choices:
    51	        msg = f'output specifier not supported for: "{arg}" (choose from "{file_choices}")'
    52	        raise argparse.ArgumentTypeError(msg)
    53
    54	    return values
