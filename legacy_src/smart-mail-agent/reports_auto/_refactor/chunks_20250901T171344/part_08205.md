   826	class StubgenPythonSuite(DataSuite):
   827	    """Data-driven end-to-end test cases that generate stub files.
   828
   829	    You can use these magic test case name suffixes:
   830
   831	    *_semanal
   832	        Run semantic analysis (slow as this uses real stubs -- only use
   833	        when necessary)
   834	    *_import
   835	        Import module and perform runtime introspection (in the current
   836	        process!)
   837
   838	    You can use these magic comments:
   839
   840	    # flags: --some-stubgen-option ...
   841	        Specify custom stubgen options
   842
   843	    # modules: module1 module2 ...
   844	        Specify which modules to output (by default only 'main')
   845	    """
   846
   847	    required_out_section = True
   848	    base_path = "."
   849	    files = ["stubgen.test"]
   850
   851	    @unittest.skipIf(sys.platform == "win32", "clean up fails on Windows")
   852	    def run_case(self, testcase: DataDrivenTestCase) -> None:
   853	        with local_sys_path_set():
   854	            self.run_case_inner(testcase)
   855
   856	    def run_case_inner(self, testcase: DataDrivenTestCase) -> None:
   857	        extra = []  # Extra command-line args
   858	        mods = []  # Module names to process
   859	        source = "\n".join(testcase.input)
   860	        for file, content in testcase.files + [("./main.py", source)]:
   861	            # Strip ./ prefix and .py suffix.
   862	            mod = file[2:-3].replace("/", ".")
   863	            if mod.endswith(".__init__"):
   864	                mod, _, _ = mod.rpartition(".")
   865	            mods.append(mod)
   866	            if "-p " not in source:
   867	                extra.extend(["-m", mod])
   868	            with open(file, "w") as f:
   869	                f.write(content)
   870
   871	        options = self.parse_flags(source, extra)
   872	        if sys.version_info < options.pyversion:
   873	            pytest.skip()
   874	        modules = self.parse_modules(source)
   875	        out_dir = "out"
   876	        try:
   877	            try:
   878	                if testcase.name.endswith("_inspect"):
   879	                    options.inspect = True
   880	                else:
   881	                    if not testcase.name.endswith("_import"):
   882	                        options.no_import = True
   883	                    if not testcase.name.endswith("_semanal"):
   884	                        options.parse_only = True
   885
   886	                generate_stubs(options)
   887	                a: list[str] = []
   888	                for module in modules:
   889	                    fnam = module_to_path(out_dir, module)
   890	                    self.add_file(fnam, a, header=
