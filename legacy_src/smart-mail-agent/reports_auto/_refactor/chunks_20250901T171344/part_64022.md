  1452	        if toplevel:

  1464	        lastelt = firstref = lastref = None
  1465	        destinationnamestotitles = self.destinationnamestotitles

  1467	        for index in range(nelts):
  1468	            eltobj = OutlineEntryObject()
  1469	            eltobj.Parent = Parent
  1470	            eltname = "%s.%s" % (levelname, index)
  1471	            eltref = document.Reference(eltobj, eltname)
  1472	            #document.add(eltname, eltobj)
  1473	            if lastelt is not None:
  1474	                lastelt.Next = eltref
  1475	                eltobj.Prev = lastref

  1477	                firstref = eltref
  1478	            lastref = eltref
  1479	            lastelt = eltobj # advance eltobj
  1480	            lastref = eltref
  1481	            elt = destinationtree[index]

  1483	                # simple leaf {name: dest}
  1484	                leafdict = elt
  1485	            elif isinstance(elt,tuple):

  1487	                try:
  1488	                    (leafdict, subsections) = elt
  1489	                except:
  1490	                    raise ValueError("destination tree elt tuple should have two elts, got %s" % len(elt))
  1491	                eltobj.Count = count(subsections, closedict)
  1492	                (eltobj.First, eltobj.Last) = self.maketree(document, subsections, eltref)
  1493	            else:
  1494	                raise ValueError("destination tree elt should be dict or tuple, got %s" % type(elt))
  1495	            try:
  1496	                [(Title, Dest)] = list(leafdict.items())
  1497	            except:
  1498	                raise ValueError("bad outline leaf dictionary, should have one entry "+bytestr(elt))
  1499	            eltobj.Title = destinationnamestotitles[Title]
  1500	            eltobj.Dest = Dest
  1501	            if isinstance(elt,tuple) and Dest in closedict:

  1507	    """utility for outline: recursively count leaves in a tuple/list tree"""
  1508	    if isinstance(tree,tuple):
  1509	        # leaf with subsections XXXX should clean up this structural usage
  1510	        (leafdict, subsections) = tree
  1511	        [(Title, Dest)] = list(leafdict.items())
  1512	        if closedict and Dest in closedict:
  1513	            return 1 # closed tree element
  1514	    if isSeq(tree):
  1515	        #return reduce(add, map(count, tree))
  1516	        counts = []
  1517	        for e in tree:
  1518	            counts.append(count(e, closedict))
  1519	        return sum(counts)  #used to be: return reduce(add, counts)
  1520	    return 1
