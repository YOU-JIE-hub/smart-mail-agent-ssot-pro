   954	    def meet_tuples(self, s: TupleType, t: TupleType) -> list[Type] | None:
   955	        """Meet two tuple types while handling variadic entries.
   956
   957	        This is surprisingly tricky, and we don't handle some tricky corner cases.
   958	        Most of the trickiness comes from the variadic tuple items like *tuple[X, ...]
   959	        since they can have arbitrary partial overlaps (while *Ts can't be split). This
   960	        function is roughly a mirror of join_tuples() w.r.t. to the fact that fixed
   961	        tuples are subtypes of variadic ones but not vice versa.
   962	        """
   963	        s_unpack_index = find_unpack_in_list(s.items)
   964	        t_unpack_index = find_unpack_in_list(t.items)
   965	        if s_unpack_index is None and t_unpack_index is None:
   966	            if s.length() == t.length():
   967	                items: list[Type] = []
   968	                for i in range(t.length()):
   969	                    items.append(self.meet(t.items[i], s.items[i]))
   970	                return items
   971	            return None
   972	        if s_unpack_index is not None and t_unpack_index is not None:
   973	            # The only simple case we can handle if both tuples are variadic
   974	            # is when their structure fully matches. Other cases are tricky because
   975	            # a variadic item is effectively a union of tuples of all length, thus
   976	            # potentially causing overlap between a suffix in `s` and a prefix
   977	            # in `t` (see how this is handled in is_subtype() for details).
   978	            # TODO: handle more cases (like when both prefix/suffix are shorter in s or t).
   979	            if s.length() == t.length() and s_unpack_index == t_unpack_index:
   980	                unpack_index = s_unpack_index
   981	                s_unpack = s.items[unpack_index]
   982	                assert isinstance(s_unpack, UnpackType)
   983	                s_unpacked = get_proper_type(s_unpack.type)
   984	                t_unpack = t.items[unpack_index]
   985	                assert isinstance(t_unpack, UnpackType)
   986	                t_unpacked = get_proper_type(t_unpack.type)
   987	                if not (isinstance(s_unpacked, Instance) and isinstance(t_unpacked, Instance)):
   988	                    return None
   989	                meet = self.meet(s_unpacked, t_unpacked)
   990	                if not isinstance(meet, Instance):
   991	                    return None
   992	                m_prefix: list[Type] = []
   993	                for si, ti in zip(s.items[:unpack_index], t.items[:unpack_index]):
   994	                    m_prefix.append(meet_types(si, ti))
   995	                m_suffix: list[Type] = []
   996	                for si, ti in zip(s.items[unpack_index + 1 :], t.items[unpack_index + 1 :]):
   997	                    m_suffix.append(meet_types(si, ti))
   998	                return m_prefix + [UnpackType(meet)] + m_suffix
   999	            return None
  1000	        if s_unpack_index is not None:
  1001	            variadic = s
  1002	            unpack_index = s_unpack_index
  1003	            fixed = t
  1004	        else:
  1005	            assert t_unpack_index is not None
  1006	            variadic = t
  1007	            unpack_index = t_unpack_index
  1008	            fixed = s
  1009	        # If one tuple is variadic one, and the other one is fixed, the meet will be fixed.
  1010	        unpack = variadic.items[unpack_index]
  1011	        assert isinstance(unpack, UnpackType)
  1012	        unpacked = get_proper_type(unpack.type)
  1013	        if not isinstance(unpacked, Instance):
  1014	            return None
  1015	        if fixed.length() < variadic.length() - 1:
  1016	            return None
  1017	        prefix_len = unpack_index
  1018	        suffix_len = variadic.length() - prefix_len - 1
  1019	        prefix, middle, suffix = split_with_prefix_and_suffix(
  1020	            tuple(fixed.items), prefix_len, suffix_len
  1021	        )
  1022	        items = []
  1023	        for fi, vi in zip(prefix, variadic.items[:prefix_len]):
  1024	            items.append(self.meet(fi, vi))
  1025	        for mi in middle:
  1026	            items.append(self.meet(mi, unpacked.args[0]))
  1027	        if suffix_len:
  1028	            for fi, vi in zip(suffix, variadic.items[-suffix_len:]):
  1029	                items.append(self.meet(fi, vi))
  1030	        return items
