   296	def parse_config_file(
   297	    options: Options,
   298	    set_strict_flags: Callable[[], None],
   299	    filename: str | None,
   300	    stdout: TextIO | None = None,
   301	    stderr: TextIO | None = None,
   302	) -> None:
   303	    """Parse a config file into an Options object.
   304
   305	    Errors are written to stderr but are not fatal.
   306
   307	    If filename is None, fall back to default config files.
   308	    """
   309	    stdout = stdout or sys.stdout
   310	    stderr = stderr or sys.stderr
   311
   312	    ret = (
   313	        _parse_individual_file(filename, stderr)
   314	        if filename is not None
   315	        else _find_config_file(stderr)
   316	    )
   317	    if ret is None:
   318	        return
   319	    parser, config_types, file_read = ret
   320
   321	    options.config_file = file_read
   322	    os.environ["MYPY_CONFIG_FILE_DIR"] = os.path.dirname(os.path.abspath(file_read))
   323
   324	    if "mypy" not in parser:
   325	        if filename or os.path.basename(file_read) not in defaults.SHARED_CONFIG_NAMES:
   326	            print(f"{file_read}: No [mypy] section in config file", file=stderr)
   327	    else:
   328	        section = parser["mypy"]
   329	        prefix = f"{file_read}: [mypy]: "
   330	        updates, report_dirs = parse_section(
   331	            prefix, options, set_strict_flags, section, config_types, stderr
   332	        )
   333	        for k, v in updates.items():
   334	            setattr(options, k, v)
   335	        options.report_dirs.update(report_dirs)
   336
   337	    for name, section in parser.items():
   338	        if name.startswith("mypy-"):
   339	            prefix = get_prefix(file_read, name)
   340	            updates, report_dirs = parse_section(
   341	                prefix, options, set_strict_flags, section, config_types, stderr
   342	            )
   343	            if report_dirs:
   344	                print(
   345	                    prefix,
   346	                    "Per-module sections should not specify reports ({})".format(
   347	                        ", ".join(s + "_report" for s in sorted(report_dirs))
   348	                    ),
   349	                    file=stderr,
   350	                )
   351	            if set(updates) - PER_MODULE_OPTIONS:
   352	                print(
   353	                    prefix,
   354	                    "Per-module sections should only specify per-module flags ({})".format(
   355	                        ", ".join(sorted(set(updates) - PER_MODULE_OPTIONS))
   356	                    ),
   357	                    file=stderr,
   358	                )
   359	                updates = {k: v for k, v in updates.items() if k in PER_MODULE_OPTIONS}
   360
   361	            globs = name[5:]
   362	            for glob in globs.split(","):
   363	                # For backwards compatibility, replace (back)slashes with dots.
   364	                glob = glob.replace(os.sep, ".")
   365	                if os.altsep:
   366	                    glob = glob.replace(os.altsep, ".")
   367
   368	                if any(c in glob for c in "?[]!") or any(
   369	                    "*" in x and x != "*" for x in glob.split(".")
   370	                ):
   371	                    print(
   372	                        prefix,
   373	                        "Patterns must be fully-qualified module names, optionally "
   374	                        "with '*' in some components (e.g spam.*.eggs.*)",
   375	                        file=stderr,
   376	                    )
   377	                else:
   378	                    options.per_module_options[glob] = updates

   394	def destructure_overrides(toml_data: dict[str, Any]) -> dict[str, Any]:
   395	    """Take the new [[tool.mypy.overrides]] section array in the pyproject.toml file,
   396	    and convert it back to a flatter structure that the existing config_parser can handle.
   397
   398	    E.g. the following pyproject.toml file:
   399
   400	        [[tool.mypy.overrides]]
   401	        module = [
   402	            "a.b",
   403	            "b.*"
   404	        ]
   405	        disallow_untyped_defs = true
   406
   407	        [[tool.mypy.overrides]]
   408	        module = 'c'
   409	        disallow_untyped_defs = false
   410
   411	    Would map to the following config dict that it would have gotten from parsing an equivalent
   412	    ini file:
   413
   414	        {
   415	            "mypy-a.b": {
   416	                disallow_untyped_defs = true,
   417	            },
   418	            "mypy-b.*": {
   419	                disallow_untyped_defs = true,
   420	            },
   421	            "mypy-c": {
   422	                disallow_untyped_defs: false,
   423	            },
   424	        }
   425	    """
   426	    if "overrides" not in toml_data["mypy"]:
   427	        return toml_data
   428
   429	    if not isinstance(toml_data["mypy"]["overrides"], list):
   430	        raise ConfigTOMLValueError(
   431	            "tool.mypy.overrides sections must be an array. Please make "
   432	            "sure you are using double brackets like so: [[tool.mypy.overrides]]"
   433	        )
   434
   435	    result = toml_data.copy()
   436	    for override in result["mypy"]["overrides"]:
   437	        if "module" not in override:
   438	            raise ConfigTOMLValueError(
   439	                "toml config file contains a [[tool.mypy.overrides]] "
   440	                "section, but no module to override was specified."
   441	            )
   442
   443	        if isinstance(override["module"], str):
   444	            modules = [override["module"]]
   445	        elif isinstance(override["module"], list):
   446	            modules = override["module"]
   447	        else:
   448	            raise ConfigTOMLValueError(
   449	                "toml config file contains a [[tool.mypy.overrides]] "
   450	                "section with a module value that is not a string or a list of "
   451	                "strings"
   452	            )
   453
   454	        for module in modules:
   455	            module_overrides = override.copy()
   456	            del module_overrides["module"]
   457	            old_config_name = f"mypy-{module}"
   458	            if old_config_name not in result:
   459	                result[old_config_name] = module_overrides
   460	            else:
   461	                for new_key, new_value in module_overrides.items():
   462	                    if (
   463	                        new_key in result[old_config_name]
   464	                        and result[old_config_name][new_key] != new_value
   465	                    ):
   466	                        raise ConfigTOMLValueError(
   467	                            "toml config file contains "
   468	                            "[[tool.mypy.overrides]] sections with conflicting "
   469	                            f"values. Module '{module}' has two different values for '{new_key}'"
   470	                        )
   471	                    result[old_config_name][new_key] = new_value
   472
   473	    del result["mypy"]["overrides"]
   474	    return result

   477	def parse_section(
   478	    prefix: str,
   479	    template: Options,
   480	    set_strict_flags: Callable[[], None],
   481	    section: Mapping[str, Any],
   482	    config_types: dict[str, Any],
   483	    stderr: TextIO = sys.stderr,
   484	) -> tuple[dict[str, object], dict[str, str]]:
   485	    """Parse one section of a config file.
   486
   487	    Returns a dict of option values encountered, and a dict of report directories.
   488	    """
   489	    results: dict[str, object] = {}
   490	    report_dirs: dict[str, str] = {}
   491
   492	    # Because these fields exist on Options, without proactive checking, we would accept them
   493	    # and crash later
   494	    invalid_options = {
   495	        "enabled_error_codes": "enable_error_code",
   496	        "disabled_error_codes": "disable_error_code",
   497	    }
   498
   499	    for key in section:
   500	        invert = False
   501	        options_key = key
   502	        if key in config_types:
   503	            ct = config_types[key]
   504	        elif key in invalid_options:
   505	            print(
   506	                f"{prefix}Unrecognized option: {key} = {section[key]}"
   507	                f" (did you mean {invalid_options[key]}?)",
   508	                file=stderr,
   509	            )
   510	            continue
   511	        else:
   512	            dv = getattr(template, key, None)
   513	            if dv is None:
   514	                if key.endswith("_report"):
   515	                    report_type = key[:-7].replace("_", "-")
   516	                    if report_type in defaults.REPORTER_NAMES:
   517	                        report_dirs[report_type] = str(section[key])
   518	                    else:
   519	                        print(f"{prefix}Unrecognized report type: {key}", file=stderr)
   520	                    continue
   521	                if key.startswith("x_"):
   522	                    pass  # Don't complain about `x_blah` flags
   523	                elif key.startswith("no_") and hasattr(template, key[3:]):
   524	                    options_key = key[3:]
   525	                    invert = True
   526	                elif key.startswith("allow") and hasattr(template, "dis" + key):
   527	                    options_key = "dis" + key
   528	                    invert = True
   529	                elif key.startswith("disallow") and hasattr(template, key[3:]):
   530	                    options_key = key[3:]
   531	                    invert = True
   532	                elif key.startswith("show_") and hasattr(template, "hide_" + key[5:]):
   533	                    options_key = "hide_" + key[5:]
   534	                    invert = True
   535	                elif key == "strict":
   536	                    pass  # Special handling below
   537	                else:
   538	                    print(f"{prefix}Unrecognized option: {key} = {section[key]}", file=stderr)
   539	                if invert:
   540	                    dv = getattr(template, options_key, None)
   541	                else:
   542	                    continue
   543	            ct = type(dv)
   544	        v: Any = None
   545	        try:
   546	            if ct is bool:
   547	                if isinstance(section, dict):
   548	                    v = convert_to_boolean(section.get(key))
   549	                else:
   550	                    v = section.getboolean(key)  # type: ignore[attr-defined]  # Until better stub
   551	                if invert:
   552	                    v = not v
   553	            elif callable(ct):
   554	                if invert:
   555	                    print(f"{prefix}Can not invert non-boolean key {options_key}", file=stderr)
   556	                    continue
   557	                try:
   558	                    v = ct(section.get(key))
   559	                except VersionTypeError as err_version:
   560	                    print(f"{prefix}{key}: {err_version}", file=stderr)
   561	                    v = err_version.fallback
   562	                except argparse.ArgumentTypeError as err:
   563	                    print(f"{prefix}{key}: {err}", file=stderr)
   564	                    continue
   565	            else:
   566	                print(f"{prefix}Don't know what type {key} should have", file=stderr)
   567	                continue
   568	        except ValueError as err:
   569	            print(f"{prefix}{key}: {err}", file=stderr)
   570	            continue
   571	        if key == "strict":
   572	            if v:
   573	                set_strict_flags()
   574	            continue
   575	        results[options_key] = v
   576
   577	    # These two flags act as per-module overrides, so store the empty defaults.
   578	    if "disable_error_code" not in results:
   579	        results["disable_error_code"] = []
   580	    if "enable_error_code" not in results:
   581	        results["enable_error_code"] = []
   582
   583	    return results, report_dirs

   597	def split_directive(s: str) -> tuple[list[str], list[str]]:
   598	    """Split s on commas, except during quoted sections.
   599
   600	    Returns the parts and a list of error messages."""
   601	    parts = []
   602	    cur: list[str] = []
   603	    errors = []
   604	    i = 0
   605	    while i < len(s):
   606	        if s[i] == ",":
   607	            parts.append("".join(cur).strip())
   608	            cur = []
   609	        elif s[i] == '"':
   610	            i += 1
   611	            while i < len(s) and s[i] != '"':
   612	                cur.append(s[i])
   613	                i += 1
   614	            if i == len(s):
   615	                errors.append("Unterminated quote in configuration comment")
   616	                cur.clear()
   617	        else:
   618	            cur.append(s[i])
   619	        i += 1
   620	    if cur:
   621	        parts.append("".join(cur).strip())
   622
   623	    return parts, errors

   627	    """Rewrite the mypy comment syntax into ini file syntax."""

   645	def parse_mypy_comments(
   646	    args: list[tuple[int, str]], template: Options
   647	) -> tuple[dict[str, object], list[tuple[int, str]]]:
   648	    """Parse a collection of inline mypy: configuration comments.
   649
   650	    Returns a dictionary of options to be applied and a list of error messages
   651	    generated.
   652	    """
   653
   654	    errors: list[tuple[int, str]] = []
   655	    sections = {}
   656
   657	    for lineno, line in args:
   658	        # In order to easily match the behavior for bools, we abuse configparser.
   659	        # Oddly, the only way to get the SectionProxy object with the getboolean
   660	        # method is to create a config parser.
   661	        parser = configparser.RawConfigParser()
   662	        options, parse_errors = mypy_comments_to_config_map(line, template)
   663
   664	        if "python_version" in options:
   665	            errors.append((lineno, "python_version not supported in inline configuration"))
   666	            del options["python_version"]
   667
   668	        parser["dummy"] = options
   669	        errors.extend((lineno, x) for x in parse_errors)
   670
   671	        stderr = StringIO()
   672	        strict_found = False
   673
   674	        def set_strict_flags() -> None:
   675	            nonlocal strict_found
   676	            strict_found = True
   677
   678	        new_sections, reports = parse_section(
   679	            "", template, set_strict_flags, parser["dummy"], ini_config_types, stderr=stderr
   680	        )
   681	        errors.extend((lineno, x) for x in stderr.getvalue().strip().split("\n") if x)
   682	        if reports:
   683	            errors.append((lineno, "Reports not supported in inline configuration"))
   684	        if strict_found:
   685	            errors.append(
   686	                (
   687	                    lineno,
   688	                    'Setting "strict" not supported in inline configuration: specify it in '
   689	                    "a configuration file instead, or set individual inline flags "
   690	                    '(see "mypy -h" for the list of flags enabled in strict mode)',
   691	                )
   692	            )
   693
   694	        sections.update(new_sections)
   695
   696	    return sections, errors
