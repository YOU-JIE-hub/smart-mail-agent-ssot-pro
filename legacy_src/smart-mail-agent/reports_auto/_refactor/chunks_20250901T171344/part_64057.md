   271	    x0 = tx._x0
   272	    autoLeading = xs.autoLeading
   273	    leading = xs.leading

   276	    if dal:
   277	        if autoLeading=='max':

   279	            descent = max(_16*leading,-line.descent)

   281	            ascent = line.ascent
   282	            descent = -line.descent
   283	        leading = ascent+descent

   285	        tx.setLeading(leading)
   286	    if dal:
   287	        olb = tx._olb
   288	        if olb is not None:
   289	            xcy = olb-ascent
   290	            if tx._oleading!=leading:
   291	                cur_y += leading - tx._oleading
   292	            if abs(xcy-cur_y)>1e-8:
   293	                cur_y = xcy
   294	                tx.setTextOrigin(x0,cur_y)
   295	                xs.cur_y = cur_y
   296	        tx._olb = cur_y - descent
   297	        tx._oleading = leading
   298	    ws = getattr(tx,'_wordSpace',0)
   299	    nSpaces = 0
   300	    words = line.words
   301	    AL = []
   302	    LL = []
   303	    us_lines = xs.us_lines
   304	    links = xs.links
   305	    for i, f in enumerate(words):
   306	        if hasattr(f,'cbDefn'):
   307	            cbDefn = f.cbDefn
   308	            kind = cbDefn.kind
   309	            if kind=='img':

   322	            else:

   326	                else:
   327	                    func = getattr(tx._canvas,name,None)
   328	                    if not func:
   329	                        raise AttributeError("Missing %s callback attribute '%s'" % (kind,name))
   330	                    tx._canvas._curr_tx_info=dict(tx=tx,cur_x=cur_x,cur_y=cur_y,leading=leading,xs=tx.XtraState)
   331	                    try:
   332	                        func(tx._canvas,kind,getattr(cbDefn,'label',None))
   333	                    finally:
   334	                        del tx._canvas._curr_tx_info
   335	            if f is words[-1]:
   336	                if not tx._fontname:
   337	                    tx.setFont(xs.style.fontName,xs.style.fontSize)
   338	                tx._textOut('',1)
   339	        else:
   340	            cur_x_s = cur_x + nSpaces*ws
   341	            end_x = cur_x_s
   342	            fontSize = f.fontSize
   343	            textColor = f.textColor
   344	            rise = f.rise
   345	            if i > 0:
   346	                end_x = cur_x_s - (0 if preformatted else _trailingSpaceLength(words[i-1].text, tx))
   347	            if (tx._fontname,tx._fontsize)!=(f.fontName,fontSize):

   349	            if xs.textColor!=textColor:

   360	                nL = NS - S #new lines
