   630	def read_coverage_config(
   631	    config_file: bool | str,
   632	    warn: Callable[[str], None],
   633	    **kwargs: TConfigValueIn,
   634	) -> CoverageConfig:
   635	    """Read the coverage.py configuration.
   636
   637	    Arguments:
   638	        config_file: a boolean or string, see the `Coverage` class for the
   639	            tricky details.
   640	        warn: a function to issue warnings.
   641	        all others: keyword arguments from the `Coverage` class, used for
   642	            setting values in the configuration.
   643
   644	    Returns:
   645	        config:
   646	            config is a CoverageConfig object read from the appropriate
   647	            configuration file.
   648
   649	    """
   650	    # Build the configuration from a number of sources:
   651	    # 1) defaults:
   652	    config = CoverageConfig()
   653
   654	    # 2) from a file:
   655	    if config_file:
   656	        files_to_try = config_files_to_try(config_file)
   657
   658	        for fname, our_file, specified_file in files_to_try:
   659	            config_read = config.from_file(fname, warn, our_file=our_file)
   660	            if config_read:
   661	                break
   662	            if specified_file:
   663	                raise ConfigError(f"Couldn't read {fname!r} as a config file")
   664
   665	    # 3) from environment variables:
   666	    env_data_file = os.getenv("COVERAGE_FILE")
   667	    if env_data_file:
   668	        config.data_file = env_data_file
   669
   670	    # $set_env.py: COVERAGE_DEBUG - Debug options: https://coverage.rtfd.io/cmd.html#debug
   671	    debugs = os.getenv("COVERAGE_DEBUG")
   672	    if debugs:
   673	        config.debug.extend(d.strip() for d in debugs.split(","))
   674
   675	    # Read the COVERAGE_CORE environment variable for backward compatibility,
   676	    # and because we use it in the test suite to pick a specific core.
   677	    env_core = os.getenv("COVERAGE_CORE")
   678	    if env_core:
   679	        config.core = env_core
   680
   681	    # 4) from constructor arguments:
   682	    config.from_args(**kwargs)
   683
   684	    # 5) for our benchmark, force settings using a secret environment variable:
   685	    force_file = os.getenv("COVERAGE_FORCE_CONFIG")
   686	    if force_file:
   687	        config.from_file(force_file, warn, our_file=True)
   688
   689	    # Once all the config has been collected, there's a little post-processing
   690	    # to do.
   691	    config.post_process()
   692
   693	    return config
