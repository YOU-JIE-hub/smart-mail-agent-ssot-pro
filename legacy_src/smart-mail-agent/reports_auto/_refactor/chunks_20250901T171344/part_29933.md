     6	to update reference count opcodes.

   106	def transform_block(
   107	    block: BasicBlock,
   108	    pre_live: AnalysisDict[Value],
   109	    post_live: AnalysisDict[Value],
   110	    pre_borrow: AnalysisDict[Value],
   111	    post_must_defined: AnalysisDict[Value],
   112	) -> None:
   113	    old_ops = block.ops
   114	    ops: list[Op] = []
   115	    for i, op in enumerate(old_ops):
   116	        key = (block, i)
   117
   118	        assert op not in pre_live[key]
   119	        dest = op.dest if isinstance(op, Assign) else op
   120	        stolen = op.stolen()
   121
   122	        # Incref any references that are being stolen that stay live, were borrowed,
   123	        # or are stolen more than once by this operation.
   124	        for j, src in enumerate(stolen):
   125	            if src in post_live[key] or src in pre_borrow[key] or src in stolen[:j]:
   126	                maybe_append_inc_ref(ops, src)
   127	                # For assignments to registers that were already live,
   128	                # decref the old value.
   129	                if dest not in pre_borrow[key] and dest in pre_live[key]:
   130	                    assert isinstance(op, Assign)
   131	                    maybe_append_dec_ref(ops, dest, post_must_defined, key)
   132
   133	        # Strip KeepAlive. Its only purpose is to help with this transform.
   134	        if not isinstance(op, KeepAlive):
   135	            ops.append(op)
   136
   137	        # Control ops don't have any space to insert ops after them, so
   138	        # their inc/decrefs get inserted by insert_branch_inc_and_decrefs.
   139	        if isinstance(op, ControlOp):
   140	            continue
   141
   142	        for src in op.unique_sources():
   143	            # Decrement source that won't be live afterwards.
   144	            if src not in post_live[key] and src not in pre_borrow[key] and src not in stolen:
   145	                maybe_append_dec_ref(ops, src, post_must_defined, key)
   146	        # Decrement the destination if it is dead after the op and
   147	        # wasn't a borrowed RegisterOp
   148	        if (
   149	            not dest.is_void
   150	            and dest not in post_live[key]
   151	            and not (isinstance(op, RegisterOp) and dest.is_borrowed)
   152	        ):
   153	            maybe_append_dec_ref(ops, dest, post_must_defined, key)
   154	    block.ops = ops

   157	def insert_branch_inc_and_decrefs(
   158	    block: BasicBlock,
   159	    cache: BlockCache,
   160	    blocks: list[BasicBlock],
   161	    pre_live: AnalysisDict[Value],
   162	    pre_borrow: AnalysisDict[Value],
   163	    post_borrow: AnalysisDict[Value],
   164	    post_must_defined: AnalysisDict[Value],
   165	    ordering: dict[Value, int],
   166	) -> None:
   167	    """Insert inc_refs and/or dec_refs after a branch/goto.
   168
   169	    Add dec_refs for registers that become dead after a branch.
   170	    Add inc_refs for registers that become unborrowed after a branch or goto.
   171
   172	    Branches are special as the true and false targets may have a different
   173	    live and borrowed register sets. Add new blocks before the true/false target
   174	    blocks that tweak reference counts.
   175
   176	    Example where we need to add an inc_ref:
   177
   178	      def f(a: int) -> None
   179	          if a:
   180	              a = 1
   181	          return a  # a is borrowed if condition is false and unborrowed if true
   182	    """
   183	    prev_key = (block, len(block.ops) - 1)
   184	    source_live_regs = pre_live[prev_key]
   185	    source_borrowed = post_borrow[prev_key]
   186	    source_defined = post_must_defined[prev_key]
   187
   188	    term = block.terminator
   189	    for i, target in enumerate(term.targets()):
   190	        # HAX: After we've checked against an error value the value we must not touch the
   191	        #      refcount since it will be a null pointer. The correct way to do this would be
   192	        #      to perform data flow analysis on whether a value can be null (or is always
   193	        #      null).
   194	        omitted: Iterable[Value]
   195	        if isinstance(term, Branch) and term.op == Branch.IS_ERROR and i == 0:
   196	            omitted = (term.value,)
   197	        else:
   198	            omitted = ()
   199
   200	        decs = after_branch_decrefs(
   201	            target, pre_live, source_defined, source_borrowed, source_live_regs, ordering, omitted
   202	        )
   203	        incs = after_branch_increfs(target, pre_live, pre_borrow, source_borrowed, ordering)
   204	        term.set_target(i, add_block(decs, incs, cache, blocks, target))

   207	def after_branch_decrefs(
   208	    label: BasicBlock,
   209	    pre_live: AnalysisDict[Value],
   210	    source_defined: set[Value],
   211	    source_borrowed: set[Value],
   212	    source_live_regs: set[Value],
   213	    ordering: dict[Value, int],
   214	    omitted: Iterable[Value],
   215	) -> tuple[tuple[Value, bool], ...]:
   216	    target_pre_live = pre_live[label, 0]
   217	    decref = source_live_regs - target_pre_live - source_borrowed
   218	    if decref:
   219	        return tuple(
   220	            (reg, is_maybe_undefined(source_defined, reg))
   221	            for reg in sorted(decref, key=lambda r: ordering[r])
   222	            if reg.type.is_refcounted and reg not in omitted
   223	        )
   224	    return ()

   227	def after_branch_increfs(
   228	    label: BasicBlock,
   229	    pre_live: AnalysisDict[Value],

   233	) -> tuple[Value, ...]:
   234	    target_pre_live = pre_live[label, 0]
