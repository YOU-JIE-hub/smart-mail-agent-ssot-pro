  2594	def find_module_and_diagnose(
  2595	    manager: BuildManager,
  2596	    id: str,
  2597	    options: Options,
  2598	    caller_state: State | None = None,
  2599	    caller_line: int = 0,
  2600	    ancestor_for: State | None = None,
  2601	    root_source: bool = False,
  2602	    skip_diagnose: bool = False,
  2603	) -> tuple[str, str]:
  2604	    """Find a module by name, respecting follow_imports and producing diagnostics.
  2605
  2606	    If the module is not found, then the ModuleNotFound exception is raised.
  2607
  2608	    Args:
  2609	      id: module to find
  2610	      options: the options for the module being loaded
  2611	      caller_state: the state of the importing module, if applicable
  2612	      caller_line: the line number of the import
  2613	      ancestor_for: the child module this is an ancestor of, if applicable
  2614	      root_source: whether this source was specified on the command line
  2615	      skip_diagnose: skip any error diagnosis and reporting (but ModuleNotFound is
  2616	          still raised if the module is missing)
  2617
  2618	    The specified value of follow_imports for a module can be overridden
  2619	    if the module is specified on the command line or if it is a stub,
  2620	    so we compute and return the "effective" follow_imports of the module.
  2621
  2622	    Returns a tuple containing (file path, target's effective follow_imports setting)
  2623	    """
  2624	    result = find_module_with_reason(id, manager)
  2625	    if isinstance(result, str):
  2626	        # For non-stubs, look at options.follow_imports:
  2627	        # - normal (default) -> fully analyze
  2628	        # - silent -> analyze but silence errors
  2629	        # - skip -> don't analyze, make the type Any
  2630	        follow_imports = options.follow_imports
  2631	        if (
  2632	            root_source  # Honor top-level modules
  2633	            or (
  2634	                result.endswith(".pyi")  # Stubs are always normal
  2635	                and not options.follow_imports_for_stubs  # except when they aren't
  2636	            )
  2637	            or id in CORE_BUILTIN_MODULES  # core is always normal
  2638	        ):
  2639	            follow_imports = "normal"
  2640	        if skip_diagnose:
  2641	            pass
  2642	        elif follow_imports == "silent":
  2643	            # Still import it, but silence non-blocker errors.
  2644	            manager.log(f"Silencing {result} ({id})")
  2645	        elif follow_imports == "skip" or follow_imports == "error":
  2646	            # In 'error' mode, produce special error messages.
  2647	            if id not in manager.missing_modules:
  2648	                manager.log(f"Skipping {result} ({id})")
  2649	            if follow_imports == "error":
  2650	                if ancestor_for:
  2651	                    skipping_ancestor(manager, id, result, ancestor_for)
  2652	                else:
  2653	                    skipping_module(manager, caller_line, caller_state, id, result)
  2654	            raise ModuleNotFound
  2655	        if is_silent_import_module(manager, result) and not root_source:
  2656	            follow_imports = "silent"
  2657	        return (result, follow_imports)
  2658	    else:
  2659	        # Could not find a module.  Typically the reason is a
  2660	        # misspelled module name, missing stub, module not in
  2661	        # search path or the module has not been installed.
  2662
  2663	        ignore_missing_imports = options.ignore_missing_imports
  2664
  2665	        # Don't honor a global (not per-module) ignore_missing_imports
  2666	        # setting for modules that used to have bundled stubs, as
  2667	        # otherwise updating mypy can silently result in new false
  2668	        # negatives. (Unless there are stubs but they are incomplete.)
  2669	        global_ignore_missing_imports = manager.options.ignore_missing_imports
  2670	        if (
  2671	            is_module_from_legacy_bundled_package(id)
  2672	            and global_ignore_missing_imports
  2673	            and not options.ignore_missing_imports_per_module
  2674	            and result is ModuleNotFoundReason.APPROVED_STUBS_NOT_INSTALLED
  2675	        ):
  2676	            ignore_missing_imports = False
  2677
  2678	        if skip_diagnose:
  2679	            raise ModuleNotFound
  2680	        if caller_state:
  2681	            if not (ignore_missing_imports or in_partial_package(id, manager)):
  2682	                module_not_found(manager, caller_line, caller_state, id, result)
  2683	            raise ModuleNotFound
  2684	        elif root_source:
  2685	            # If we can't find a root source it's always fatal.
  2686	            # TODO: This might hide non-fatal errors from
  2687	            # root sources processed earlier.
  2688	            raise CompileError([f"mypy: can't find module '{id}'"])
  2689	        else:
  2690	            raise ModuleNotFound

  2736	    """Check if a missing module can potentially be a part of a package.
  2737
  2738	    This checks if there is any existing parent __init__.pyi stub that
  2739	    defines a module-level __getattr__ (a.k.a. partial stub package).

  2742	        parent, _ = id.rsplit(".", 1)
  2743	        if parent in manager.modules:
  2744	            parent_mod: MypyFile | None = manager.modules[parent]
  2745	        else:
  2746	            # Parent is not in build, try quickly if we can find it.
  2747	            try:
  2748	                parent_st = State(
  2749	                    id=parent, path=None, source=None, manager=manager, temporary=True
  2750	                )
  2751	            except (ModuleNotFound, CompileError):
  2752	                parent_mod = None
  2753	            else:
  2754	                parent_mod = parent_st.tree
  2755	        if parent_mod is not None:
  2756	            # Bail out soon, complete subpackage found
