   516	                    if report_type in defaults.REPORTER_NAMES:
   517	                        report_dirs[report_type] = str(section[key])

   520	                    continue
   521	                if key.startswith("x_"):

   523	                elif key.startswith("no_") and hasattr(template, key[3:]):
   524	                    options_key = key[3:]

   526	                elif key.startswith("allow") and hasattr(template, "dis" + key):
   527	                    options_key = "dis" + key
   528	                    invert = True
   529	                elif key.startswith("disallow") and hasattr(template, key[3:]):
   530	                    options_key = key[3:]
   531	                    invert = True
   532	                elif key.startswith("show_") and hasattr(template, "hide_" + key[5:]):
   533	                    options_key = "hide_" + key[5:]
   534	                    invert = True
   535	                elif key == "strict":
   536	                    pass  # Special handling below
   537	                else:
   538	                    print(f"{prefix}Unrecognized option: {key} = {section[key]}", file=stderr)
   539	                if invert:
   540	                    dv = getattr(template, options_key, None)
   541	                else:
   542	                    continue

   544	        v: Any = None

   547	                if isinstance(section, dict):
   548	                    v = convert_to_boolean(section.get(key))
   549	                else:
   550	                    v = section.getboolean(key)  # type: ignore[attr-defined]  # Until better stub

   561	                    v = err_version.fallback
   562	                except argparse.ArgumentTypeError as err:
   563	                    print(f"{prefix}{key}: {err}", file=stderr)
   564	                    continue
   565	            else:
   566	                print(f"{prefix}Don't know what type {key} should have", file=stderr)
   567	                continue
   568	        except ValueError as err:
   569	            print(f"{prefix}{key}: {err}", file=stderr)
   570	            continue

    27	def constant_fold_expr(expr: Expression, cur_mod_id: str) -> ConstantValue | None:
    28	    """Return the constant value of an expression for supported operations.
    29
    30	    Among other things, support int arithmetic and string
    31	    concatenation. For example, the expression 3 + 5 has the constant
    32	    value 8.

    35	    current module (cur_mod_id). Binding to references is best effort
    36	    -- we don't bind references to other modules. Mypyc trusts these
    37	    to be correct in compiled modules, so that it can replace a
    38	    constant expression (or a reference to one) with the statically
    39	    computed value. We don't want to infer constant values based on
    40	    stubs, in particular, as these might not match the implementation
    41	    (due to version skew, for example).

    43	    Return None if unsuccessful.
    44	    """
    45	    if isinstance(expr, IntExpr):
    46	        return expr.value
    47	    if isinstance(expr, StrExpr):
    48	        return expr.value
    49	    if isinstance(expr, FloatExpr):
    50	        return expr.value
    51	    if isinstance(expr, ComplexExpr):
    52	        return expr.value
    53	    elif isinstance(expr, NameExpr):
    54	        if expr.name == "True":
    55	            return True
    56	        elif expr.name == "False":
