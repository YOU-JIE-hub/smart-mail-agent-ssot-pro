   114	class ConftestImportFailure(Exception):
   115	    def __init__(
   116	        self,
   117	        path: pathlib.Path,
   118	        *,
   119	        cause: Exception,
   120	    ) -> None:
   121	        self.path = path
   122	        self.cause = cause
   123
   124	    def __str__(self) -> str:
   125	        return f"{type(self.cause).__name__}: {self.cause} (from {self.path})"

   128	def filter_traceback_for_conftest_import_failure(
   129	    entry: _pytest._code.TracebackEntry,
   130	) -> bool:
   131	    """Filter tracebacks entries which point to pytest internals or importlib.
   132
   133	    Make a special case for importlib because we use it to import test modules and conftest files
   134	    in _pytest.pathlib.import_path.
   135	    """
   136	    return filter_traceback(entry) and "importlib" not in str(entry.path).split(os.sep)

   139	def main(
   140	    args: list[str] | os.PathLike[str] | None = None,
   141	    plugins: Sequence[str | _PluggyPlugin] | None = None,
   142	) -> int | ExitCode:
   143	    """Perform an in-process test run.
   144
   145	    :param args:
   146	        List of command line arguments. If `None` or not given, defaults to reading
   147	        arguments directly from the process command line (:data:`sys.argv`).
   148	    :param plugins: List of plugin objects to be auto-registered during initialization.
   149
   150	    :returns: An exit code.
   151	    """
   152	    old_pytest_version = os.environ.get("PYTEST_VERSION")
   153	    try:
   154	        os.environ["PYTEST_VERSION"] = __version__
   155	        try:
   156	            config = _prepareconfig(args, plugins)
   157	        except ConftestImportFailure as e:
   158	            exc_info = ExceptionInfo.from_exception(e.cause)
   159	            tw = TerminalWriter(sys.stderr)
   160	            tw.line(f"ImportError while loading conftest '{e.path}'.", red=True)
   161	            exc_info.traceback = exc_info.traceback.filter(
   162	                filter_traceback_for_conftest_import_failure
   163	            )
   164	            exc_repr = (
   165	                exc_info.getrepr(style="short", chain=False)
   166	                if exc_info.traceback
   167	                else exc_info.exconly()
   168	            )
   169	            formatted_tb = str(exc_repr)
   170	            for line in formatted_tb.splitlines():
   171	                tw.line(line.rstrip(), red=True)
   172	            return ExitCode.USAGE_ERROR
   173	        else:
   174	            try:
   175	                ret: ExitCode | int = config.hook.pytest_cmdline_main(config=config)
   176	                try:
   177	                    return ExitCode(ret)
   178	                except ValueError:
   179	                    return ret
   180	            finally:
   181	                config._ensure_unconfigure()
   182	    except UsageError as e:
   183	        tw = TerminalWriter(sys.stderr)
   184	        for msg in e.args:
   185	            tw.line(f"ERROR: {msg}\n", red=True)
   186	        return ExitCode.USAGE_ERROR
   187	    finally:
   188	        if old_pytest_version is None:
   189	            os.environ.pop("PYTEST_VERSION", None)
   190	        else:
   191	            os.environ["PYTEST_VERSION"] = old_pytest_version

   194	def console_main() -> int:
   195	    """The CLI entry point of pytest.
   196
   197	    This function is not meant for programmable use; use `main()` instead.
   198	    """
   199	    # https://docs.python.org/3/library/signal.html#note-on-sigpipe
   200	    try:
   201	        code = main()
   202	        sys.stdout.flush()
   203	        return code
   204	    except BrokenPipeError:
   205	        # Python flushes standard streams on exit; redirect remaining output
   206	        # to devnull to avoid another BrokenPipeError at shutdown
   207	        devnull = os.open(os.devnull, os.O_WRONLY)
   208	        os.dup2(devnull, sys.stdout.fileno())
   209	        return 1  # Python exits with error code 1 on EPIPE

   283	def get_config(
   284	    args: list[str] | None = None,
   285	    plugins: Sequence[str | _PluggyPlugin] | None = None,
   286	) -> Config:
   287	    # subsequent calls to main will create a fresh instance
   288	    pluginmanager = PytestPluginManager()
   289	    config = Config(
   290	        pluginmanager,
   291	        invocation_params=Config.InvocationParams(
   292	            args=args or (),
   293	            plugins=plugins,
   294	            dir=pathlib.Path.cwd(),
   295	        ),
   296	    )
   297
   298	    if args is not None:
   299	        # Handle any "-p no:plugin" args.
   300	        pluginmanager.consider_preparse(args, exclude_only=True)
   301
   302	    for spec in default_plugins:
   303	        pluginmanager.import_plugin(spec)
   304
   305	    return config

   319	def _prepareconfig(
   320	    args: list[str] | os.PathLike[str] | None = None,
   321	    plugins: Sequence[str | _PluggyPlugin] | None = None,
   322	) -> Config:
   323	    if args is None:
   324	        args = sys.argv[1:]
   325	    elif isinstance(args, os.PathLike):
   326	        args = [os.fspath(args)]
   327	    elif not isinstance(args, list):
   328	        msg = (  # type:ignore[unreachable]
   329	            "`args` parameter expected to be a list of strings, got: {!r} (type: {})"
   330	        )
   331	        raise TypeError(msg.format(args, type(args)))
   332
   333	    config = get_config(args, plugins)
   334	    pluginmanager = config.pluginmanager
   335	    try:
   336	        if plugins:
   337	            for plugin in plugins:
   338	                if isinstance(plugin, str):
   339	                    pluginmanager.consider_pluginarg(plugin)
   340	                else:
   341	                    pluginmanager.register(plugin)
   342	        config = pluginmanager.hook.pytest_cmdline_parse(
   343	            pluginmanager=pluginmanager, args=args
   344	        )
   345	        return config
   346	    except BaseException:
   347	        config._ensure_unconfigure()
   348	        raise

   359	def _get_legacy_hook_marks(
   360	    method: Any,
   361	    hook_type: str,
   362	    opt_names: tuple[str, ...],
   363	) -> dict[str, bool]:
   364	    if TYPE_CHECKING:
   365	        # abuse typeguard from importlib to avoid massive method type union that's lacking an alias
   366	        assert inspect.isroutine(method)
   367	    known_marks: set[str] = {m.name for m in getattr(method, "pytestmark", [])}
   368	    must_warn: list[str] = []
   369	    opts: dict[str, bool] = {}
   370	    for opt_name in opt_names:
   371	        opt_attr = getattr(method, opt_name, AttributeError)
   372	        if opt_attr is not AttributeError:
   373	            must_warn.append(f"{opt_name}={opt_attr}")
   374	            opts[opt_name] = True
   375	        elif opt_name in known_marks:
   376	            must_warn.append(f"{opt_name}=True")
   377	            opts[opt_name] = True
   378	        else:
   379	            opts[opt_name] = False
   380	    if must_warn:
   381	        hook_opts = ", ".join(must_warn)
   382	        message = _pytest.deprecated.HOOK_LEGACY_MARKING.format(
   383	            type=hook_type,
   384	            fullname=method.__qualname__,
   385	            hook_opts=hook_opts,
   386	        )
   387	        warn_explicit_for(cast(FunctionType, method), message)
   388	    return opts

   391	@final
   392	class PytestPluginManager(PluginManager):
   393	    """A :py:class:`pluggy.PluginManager <pluggy.PluginManager>` with
   394	    additional pytest-specific functionality:
   395
   396	    * Loading plugins from the command line, ``PYTEST_PLUGINS`` env variable and
   397	      ``pytest_plugins`` global variables found in plugins being loaded.
   398	    * ``conftest.py`` loading during start-up.
   399	    """
   400
   401	    def __init__(self) -> None:
   402	        from _pytest.assertion import DummyRewriteHook
   403	        from _pytest.assertion import RewriteHook
   404
   405	        super().__init__("pytest")
   406
   407	        # -- State related to local conftest plugins.
   408	        # All loaded conftest modules.
   409	        self._conftest_plugins: set[types.ModuleType] = set()
   410	        # All conftest modules applicable for a directory.
   411	        # This includes the directory's own conftest modules as well
   412	        # as those of its parent directories.
   413	        self._dirpath2confmods: dict[pathlib.Path, list[types.ModuleType]] = {}
   414	        # Cutoff directory above which conftests are no longer discovered.
   415	        self._confcutdir: pathlib.Path | None = None
   416	        # If set, conftest loading is skipped.
   417	        self._noconftest = False
   418
   419	        # _getconftestmodules()'s call to _get_directory() causes a stat
   420	        # storm when it's called potentially thousands of times in a test
   421	        # session (#9478), often with the same path, so cache it.
   422	        self._get_directory = lru_cache(256)(_get_directory)
   423
   424	        # plugins that were explicitly skipped with pytest.skip
   425	        # list of (module name, skip reason)
   426	        # previously we would issue a warning when a plugin was skipped, but
   427	        # since we refactored warnings as first citizens of Config, they are
   428	        # just stored here to be used later.
   429	        self.skipped_plugins: list[tuple[str, str]] = []
   430
   431	        self.add_hookspecs(_pytest.hookspec)
   432	        self.register(self)
   433	        if os.environ.get("PYTEST_DEBUG"):
   434	            err: IO[str] = sys.stderr
   435	            encoding: str = getattr(err, "encoding", "utf8")
   436	            try:
   437	                err = open(
   438	                    os.dup(err.fileno()),
   439	                    mode=err.mode,
   440	                    buffering=1,
   441	                    encoding=encoding,
   442	                )
   443	            except Exception:
   444	                pass
   445	            self.trace.root.setwriter(err.write)
   446	            self.enable_tracing()
   447
   448	        # Config._consider_importhook will set a real object if required.
   449	        self.rewrite_hook: RewriteHook = DummyRewriteHook()
   450	        # Used to know when we are importing conftests after the pytest_configure stage.
   451	        self._configured = False

   453	    def parse_hookimpl_opts(
   454	        self, plugin: _PluggyPlugin, name: str
   455	    ) -> HookimplOpts | None:
   456	        """:meta private:"""
   457	        # pytest hooks are always prefixed with "pytest_",
   458	        # so we avoid accessing possibly non-readable attributes
   459	        # (see issue #1073).
   460	        if not name.startswith("pytest_"):
   461	            return None
   462	        # Ignore names which cannot be hooks.
   463	        if name == "pytest_plugins":
   464	            return None
   465
   466	        opts = super().parse_hookimpl_opts(plugin, name)
   467	        if opts is not None:
   468	            return opts
   469
   470	        method = getattr(plugin, name)
   471	        # Consider only actual functions for hooks (#3775).
   472	        if not inspect.isroutine(method):
   473	            return None
   474	        # Collect unmarked hooks as long as they have the `pytest_' prefix.
   475	        legacy = _get_legacy_hook_marks(
   476	            method, "impl", ("tryfirst", "trylast", "optionalhook", "hookwrapper")
   477	        )
   478	        return cast(HookimplOpts, legacy)
   479
   480	    def parse_hookspec_opts(self, module_or_class, name: str) -> HookspecOpts | None:
   481	        """:meta private:"""
   482	        opts = super().parse_hookspec_opts(module_or_class, name)
   483	        if opts is None:
   484	            method = getattr(module_or_class, name)
   485	            if name.startswith("pytest_"):
   486	                legacy = _get_legacy_hook_marks(
   487	                    method, "spec", ("firstresult", "historic")
   488	                )
   489	                opts = cast(HookspecOpts, legacy)
   490	        return opts
   491
   492	    def register(self, plugin: _PluggyPlugin, name: str | None = None) -> str | None:
   493	        if name in _pytest.deprecated.DEPRECATED_EXTERNAL_PLUGINS:
   494	            warnings.warn(
   495	                PytestConfigWarning(
   496	                    "{} plugin has been merged into the core, "
   497	                    "please remove it from your requirements.".format(
   498	                        name.replace("_", "-")
   499	                    )
   500	                )
   501	            )
   502	            return None
   503	        plugin_name = super().register(plugin, name)
   504	        if plugin_name is not None:
   505	            self.hook.pytest_plugin_registered.call_historic(
   506	                kwargs=dict(
   507	                    plugin=plugin,
   508	                    plugin_name=plugin_name,
   509	                    manager=self,
   510	                )
   511	            )
   512
   513	            if isinstance(plugin, types.ModuleType):
   514	                self.consider_module(plugin)
   515	        return plugin_name
   516
   517	    def getplugin(self, name: str):
   518	        # Support deprecated naming because plugins (xdist e.g.) use it.
   519	        plugin: _PluggyPlugin | None = self.get_plugin(name)
   520	        return plugin
   521
   522	    def hasplugin(self, name: str) -> bool:
   523	        """Return whether a plugin with the given name is registered."""
   524	        return bool(self.get_plugin(name))

   526	    def pytest_configure(self, config: Config) -> None:
   527	        """:meta private:"""
   528	        # XXX now that the pluginmanager exposes hookimpl(tryfirst...)
   529	        # we should remove tryfirst/trylast as markers.
   530	        config.addinivalue_line(
   531	            "markers",
   532	            "tryfirst: mark a hook implementation function such that the "
   533	            "plugin machinery will try to call it first/as early as possible. "
   534	            "DEPRECATED, use @pytest.hookimpl(tryfirst=True) instead.",
   535	        )
   536	        config.addinivalue_line(
   537	            "markers",
   538	            "trylast: mark a hook implementation function such that the "
   539	            "plugin machinery will try to call it last/as late as possible. "
   540	            "DEPRECATED, use @pytest.hookimpl(trylast=True) instead.",
   541	        )
   542	        self._configured = True
