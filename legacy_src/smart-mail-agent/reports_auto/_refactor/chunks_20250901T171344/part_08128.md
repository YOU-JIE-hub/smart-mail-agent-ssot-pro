   893	    def visit_func_def(self, defn: FuncDef) -> None:
   894	        self.statement = defn
   895
   896	        # Visit default values because they may contain assignment expressions.
   897	        for arg in defn.arguments:
   898	            if arg.initializer:
   899	                arg.initializer.accept(self)

   907	        # We don't add module top-level functions to symbol tables
   908	        # when we analyze their bodies in the second phase on analysis,

   910	        # get always added, since they aren't separate targets.
   911	        if not self.recurse_into_functions or len(self.function_stack) > 0:

   913	                self.add_function_to_symbol_table(defn)
   914
   915	        if not self.recurse_into_functions:
   916	            return
   917
   918	        with self.scope.function_scope(defn):
   919	            with self.inside_except_star_block_set(value=False):
   920	                self.analyze_func_def(defn)

   927	    def analyze_func_def(self, defn: FuncDef) -> None:
   928	        if self.push_type_args(defn.type_args, defn) is None:
   929	            self.defer(defn)
   930	            return
   931
   932	        self.function_stack.append(defn)
   933
   934	        if defn.type:
   935	            assert isinstance(defn.type, CallableType)
   936	            has_self_type = self.update_function_type_variables(defn.type, defn)
   937	        else:
   938	            has_self_type = False
   939
   940	        self.function_stack.pop()
   941
   942	        if self.is_class_scope():
   943	            # Method definition
   944	            assert self.type is not None
   945	            defn.info = self.type
   946	            if defn.type is not None and defn.name in ("__init__", "__init_subclass__"):
   947	                assert isinstance(defn.type, CallableType)
   948	                if isinstance(get_proper_type(defn.type.ret_type), AnyType):
   949	                    defn.type = defn.type.copy_modified(ret_type=NoneType())
   950	            self.prepare_method_signature(defn, self.type, has_self_type)
   951
   952	        # Analyze function signature
   953	        fullname = self.function_fullname(defn.fullname)
   954	        with self.tvar_scope_frame(self.tvar_scope.method_frame(fullname)):
   955	            if defn.type:
   956	                self.check_classvar_in_signature(defn.type)
   957	                assert isinstance(defn.type, CallableType)
   958	                # Signature must be analyzed in the surrounding scope so that
   959	                # class-level imported names and type variables are in scope.
   960	                analyzer = self.type_analyzer()
   961	                tag = self.track_incomplete_refs()
   962	                result = analyzer.visit_callable_type(defn.type, nested=False, namespace=fullname)
   963	                # Don't store not ready types (including placeholders).
   964	                if self.found_incomplete_ref(tag) or has_placeholder(result):
   965	                    self.defer(defn)
   966	                    self.pop_type_args(defn.type_args)
   967	                    return
   968	                assert isinstance(result, ProperType)
   969	                if isinstance(result, CallableType):
   970	                    # type guards need to have a positional argument, to spec
   971	                    skip_self = self.is_class_scope() and not defn.is_static
   972	                    if result.type_guard and ARG_POS not in result.arg_kinds[skip_self:]:
   973	                        self.fail(
   974	                            "TypeGuard functions must have a positional argument",
   975	                            result,
   976	                            code=codes.VALID_TYPE,
   977	                        )
   978	                        # in this case, we just kind of just ... remove the type guard.
   979	                        result = result.copy_modified(type_guard=None)
   980	                    if result.type_is and ARG_POS not in result.arg_kinds[skip_self:]:
   981	                        self.fail(
   982	                            '"TypeIs" functions must have a positional argument',
   983	                            result,
   984	                            code=codes.VALID_TYPE,
   985	                        )
   986	                        result = result.copy_modified(type_is=None)
   987
   988	                    result = self.remove_unpack_kwargs(defn, result)
   989	                    if has_self_type and self.type is not None:
   990	                        info = self.type
   991	                        if info.self_type is not None:
   992	                            result.variables = [info.self_type] + list(result.variables)
   993	                defn.type = result
   994	                self.add_type_alias_deps(analyzer.aliases_used)
   995	                self.check_function_signature(defn)
   996	                if isinstance(defn, FuncDef):
   997	                    assert isinstance(defn.type, CallableType)
   998	                    defn.type = set_callable_name(defn.type, defn)
   999
  1000	        self.analyze_arg_initializers(defn)
  1001	        self.analyze_function_body(defn)
  1002
  1003	        if self.is_class_scope():
  1004	            assert self.type is not None
  1005	            # Mark protocol methods with empty bodies as implicitly abstract.
  1006	            # This makes explicit protocol subclassing type-safe.
  1007	            if (
  1008	                self.type.is_protocol
  1009	                and not self.is_stub_file  # Bodies in stub files are always empty.
  1010	                and (not isinstance(self.scope.function, OverloadedFuncDef) or defn.is_property)
  1011	                and defn.abstract_status != IS_ABSTRACT
  1012	                and is_trivial_body(defn.body)
  1013	            ):
  1014	                defn.abstract_status = IMPLICITLY_ABSTRACT
  1015	            if (
  1016	                is_trivial_body(defn.body)
  1017	                and not self.is_stub_file
  1018	                and defn.abstract_status != NOT_ABSTRACT
  1019	            ):
  1020	                defn.is_trivial_body = True
  1021
  1022	        if (
  1023	            defn.is_coroutine
  1024	            and isinstance(defn.type, CallableType)
  1025	            and self.wrapped_coro_return_types.get(defn) != defn.type
  1026	        ):
  1027	            if defn.is_async_generator:
  1028	                # Async generator types are handled elsewhere
  1029	                pass
  1030	            else:
  1031	                # A coroutine defined as `async def foo(...) -> T: ...`
  1032	                # has external return type `Coroutine[Any, Any, T]`.
  1033	                any_type = AnyType(TypeOfAny.special_form)
  1034	                ret_type = self.named_type_or_none(
  1035	                    "typing.Coroutine", [any_type, any_type, defn.type.ret_type]
  1036	                )
  1037	                assert ret_type is not None, "Internal error: typing.Coroutine not found"
  1038	                defn.type = defn.type.copy_modified(ret_type=ret_type)
  1039	                self.wrapped_coro_return_types[defn] = defn.type
  1040
  1041	        self.pop_type_args(defn.type_args)

  1043	    def remove_unpack_kwargs(self, defn: FuncDef, typ: CallableType) -> CallableType:
  1044	        if not typ.arg_kinds or typ.arg_kinds[-1] is not ArgKind.ARG_STAR2:
  1045	            return typ
  1046	        last_type = typ.arg_types[-1]
  1047	        if not isinstance(last_type, UnpackType):
  1048	            return typ
  1049	        last_type = get_proper_type(last_type.type)
  1050	        if not isinstance(last_type, TypedDictType):
  1051	            self.fail("Unpack item in ** argument must be a TypedDict", last_type)
  1052	            new_arg_types = typ.arg_types[:-1] + [AnyType(TypeOfAny.from_error)]
  1053	            return typ.copy_modified(arg_types=new_arg_types)
  1054	        overlap = set(typ.arg_names) & set(last_type.items)
  1055	        # It is OK for TypedDict to have a key named 'kwargs'.
  1056	        overlap.discard(typ.arg_names[-1])
  1057	        if overlap:
  1058	            overlapped = ", ".join([f'"{name}"' for name in overlap])
  1059	            self.fail(f"Overlap between argument names and ** TypedDict items: {overlapped}", defn)
  1060	            new_arg_types = typ.arg_types[:-1] + [AnyType(TypeOfAny.from_error)]
  1061	            return typ.copy_modified(arg_types=new_arg_types)
  1062	        # OK, everything looks right now, mark the callable type as using unpack.
  1063	        new_arg_types = typ.arg_types[:-1] + [last_type]
  1064	        return typ.copy_modified(arg_types=new_arg_types, unpack_kwargs=True)

  1066	    def prepare_method_signature(self, func: FuncDef, info: TypeInfo, has_self_type: bool) -> None:
  1067	        """Check basic signature validity and tweak annotation of self/cls argument."""
  1068	        # Only non-static methods are special, as well as __new__.
  1069	        functype = func.type
  1070	        if func.name == "__new__":
  1071	            func.is_static = True
  1072	        if not func.is_static or func.name == "__new__":
  1073	            if func.name in ["__init_subclass__", "__class_getitem__"]:
  1074	                func.is_class = True
  1075	            if not func.arguments:
  1076	                self.fail(
  1077	                    'Method must have at least one argument. Did you forget the "self" argument?',
  1078	                    func,
  1079	                )
  1080	            elif isinstance(functype, CallableType):
  1081	                self_type = get_proper_type(functype.arg_types[0])
  1082	                if isinstance(self_type, AnyType):
  1083	                    if has_self_type:
  1084	                        assert self.type is not None and self.type.self_type is not None
  1085	                        leading_type: Type = self.type.self_type
  1086	                    else:
  1087	                        func.is_trivial_self = True
  1088	                        leading_type = fill_typevars(info)
  1089	                    if func.is_class or func.name == "__new__":
  1090	                        leading_type = self.class_type(leading_type)
  1091	                    func.type = replace_implicit_first_type(functype, leading_type)
  1092	                elif has_self_type and isinstance(func.unanalyzed_type, CallableType):
  1093	                    if not isinstance(get_proper_type(func.unanalyzed_type.arg_types[0]), AnyType):
  1094	                        if self.is_expected_self_type(
  1095	                            self_type, func.is_class or func.name == "__new__"
  1096	                        ):
  1097	                            # This error is off by default, since it is explicitly allowed
  1098	                            # by the PEP 673.
  1099	                            self.fail(
  1100	                                'Redundant "Self" annotation for the first method argument',
  1101	                                func,
  1102	                                code=codes.REDUNDANT_SELF_TYPE,
  1103	                            )
  1104	                        else:
  1105	                            self.fail(
  1106	                                "Method cannot have explicit self annotation and Self type", func
  1107	                            )
  1108	        elif has_self_type:
  1109	            self.fail("Static methods cannot use Self type", func)

  1111	    def is_expected_self_type(self, typ: Type, is_classmethod: bool) -> bool:
  1112	        """Does this (analyzed or not) type represent the expected Self type for a method?"""
  1113	        assert self.type is not None
  1114	        typ = get_proper_type(typ)
  1115	        if is_classmethod:
  1116	            if isinstance(typ, TypeType):
  1117	                return self.is_expected_self_type(typ.item, is_classmethod=False)
  1118	            if isinstance(typ, UnboundType):
  1119	                sym = self.lookup_qualified(typ.name, typ, suppress_errors=True)
  1120	                if sym is not None and sym.fullname in TYPE_NAMES and typ.args:
  1121	                    return self.is_expected_self_type(typ.args[0], is_classmethod=False)
  1122	            return False
  1123	        if isinstance(typ, TypeVarType):
  1124	            return typ == self.type.self_type
  1125	        if isinstance(typ, UnboundType):
  1126	            sym = self.lookup_qualified(typ.name, typ, suppress_errors=True)
  1127	            return sym is not None and sym.fullname in SELF_TYPE_NAMES
  1128	        return False

  1130	    def set_original_def(self, previous: Node | None, new: FuncDef | Decorator) -> bool:
  1131	        """If 'new' conditionally redefine 'previous', set 'previous' as original
  1132
  1133	        We reject straight redefinitions of functions, as they are usually
  1134	        a programming error. For example:
  1135
  1136	          def f(): ...
  1137	          def f(): ...  # Error: 'f' redefined
  1138	        """
  1139	        if isinstance(new, Decorator):
  1140	            new = new.func
  1141	        if (
  1142	            isinstance(previous, (FuncDef, Decorator))
  1143	            and unnamed_function(new.name)
  1144	            and unnamed_function(previous.name)
  1145	        ):
  1146	            return True
  1147	        if isinstance(previous, (FuncDef, Var, Decorator)) and new.is_conditional:
  1148	            new.original_def = previous
  1149	            return True
  1150	        else:
  1151	            return False

  1153	    def update_function_type_variables(self, fun_type: CallableType, defn: FuncItem) -> bool:

  1155

  1177	        """Setup a (shared) Self type variable for current class.
  1178
  1179	        We intentionally don't add it to the class symbol table,
  1180	        so it can be accessed only by mypy and will not cause
  1181	        clashes with user defined names.

  1185	        if info.self_type is not None:

  1187	                # Similar to regular (user defined) type variables.
  1188	                self.process_placeholder(
  1189	                    None,
  1190	                    "Self upper bound",
  1191	                    info,
  1192	                    force_progress=info.self_type.upper_bound != fill_typevars(info),
  1193	                )
  1194	            else:
  1195
