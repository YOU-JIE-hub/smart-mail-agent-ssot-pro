  1011	        """Returns a boolean indicating whether the field defined in `stmt` is a required field."""
  1012	        expr = stmt.rvalue
  1013	        if isinstance(expr, TempNode):
  1014	            # TempNode means annotation-only, so has no default
  1015	            return False
  1016	        if isinstance(expr, CallExpr) and isinstance(expr.callee, RefExpr) and expr.callee.fullname == FIELD_FULLNAME:
  1017	            # The "default value" is a call to `Field`; at this point, the field has a default if and only if:

  1019	            # * there is a keyword argument named "default" that is not `...`

  1023	                if name is None or name == 'default':
  1024	                    return arg.__class__ is not EllipsisExpr
  1025	                if name == 'default_factory':
  1026	                    return not (isinstance(arg, NameExpr) and arg.fullname == 'builtins.None')

  1272	def add_method(
  1273	    api: SemanticAnalyzerPluginInterface | CheckerPluginInterface,
  1274	    cls: ClassDef,
  1275	    name: str,
  1276	    args: list[Argument],
  1277	    return_type: Type,
  1278	    self_type: Type | None = None,
  1279	    tvar_def: TypeVarType | None = None,
  1280	    is_classmethod: bool = False,
  1281	) -> None:
  1282	    """Very closely related to `mypy.plugins.common.add_method_to_class`, with a few pydantic-specific changes."""
  1283	    info = cls.info
  1284
  1285	    # First remove any previously generated methods with the same name
  1286	    # to avoid clashes and problems in the semantic analyzer.
  1287	    if name in info.names:
  1288	        sym = info.names[name]
  1289	        if sym.plugin_generated and isinstance(sym.node, FuncDef):
  1290	            cls.defs.body.remove(sym.node)  # pragma: no cover
  1291
  1292	    if isinstance(api, SemanticAnalyzerPluginInterface):
  1293	        function_type = api.named_type('builtins.function')
  1294	    else:
  1295	        function_type = api.named_generic_type('builtins.function', [])
  1296
  1297	    if is_classmethod:
  1298	        self_type = self_type or TypeType(fill_typevars(info))
  1299	        first = [Argument(Var('_cls'), self_type, None, ARG_POS, True)]
  1300	    else:
  1301	        self_type = self_type or fill_typevars(info)
  1302	        # `self` is positional *ONLY* here, but this can't be expressed
  1303	        # fully in the mypy internal API. ARG_POS is the closest we can get.
  1304	        # Using ARG_POS will, however, give mypy errors if a `self` field
  1305	        # is present on a model:
  1306	        #
  1307	        #     Name "self" already defined (possibly by an import)  [no-redef]
  1308	        #
  1309	        # As a workaround, we give this argument a name that will
  1310	        # never conflict. By its positional nature, this name will not
  1311	        # be used or exposed to users.
  1312	        first = [Argument(Var('__pydantic_self__'), self_type, None, ARG_POS)]
  1313	    args = first + args
  1314
  1315	    arg_types, arg_names, arg_kinds = [], [], []
  1316	    for arg in args:
  1317	        assert arg.type_annotation, 'All arguments must be fully typed.'
  1318	        arg_types.append(arg.type_annotation)
  1319	        arg_names.append(arg.variable.name)
  1320	        arg_kinds.append(arg.kind)
  1321
  1322	    signature = CallableType(arg_types, arg_kinds, arg_names, return_type, function_type)
  1323	    if tvar_def:
  1324	        signature.variables = [tvar_def]
  1325
  1326	    func = FuncDef(name, args, Block([PassStmt()]))
  1327	    func.info = info
  1328	    func.type = set_callable_name(signature, func)
  1329	    func.is_class = is_classmethod
  1330	    func._fullname = info.fullname + '.' + name
  1331	    func.line = info.line
  1332
  1333	    # NOTE: we would like the plugin generated node to dominate, but we still
  1334	    # need to keep any existing definitions so they get semantically analyzed.
  1335	    if name in info.names:
  1336	        # Get a nice unique name instead.
  1337	        r_name = get_unique_redefinition_name(name, info.names)
  1338	        info.names[r_name] = info.names[name]
  1339
  1340	    # Add decorator for is_classmethod
  1341	    # The dataclasses plugin claims this is unnecessary for classmethods, but not including it results in a
  1342	    # signature incompatible with the superclass, which causes mypy errors to occur for every subclass of BaseModel.
  1343	    if is_classmethod:
  1344	        func.is_decorated = True
  1345	        v = Var(name, func.type)
  1346	        v.info = info
  1347	        v._fullname = func._fullname
  1348	        v.is_classmethod = True
  1349	        dec = Decorator(func, [NameExpr('classmethod')], v)
  1350	        dec.line = info.line
  1351	        sym = SymbolTableNode(MDEF, dec)
  1352	    else:
  1353	        sym = SymbolTableNode(MDEF, func)
  1354	    sym.plugin_generated = True
  1355	    info.names[name] = sym
  1356
  1357	    info.defn.defs.body.append(func)
