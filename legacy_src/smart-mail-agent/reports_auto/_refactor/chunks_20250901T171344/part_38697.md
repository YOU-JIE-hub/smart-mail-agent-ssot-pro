    21	    return ((r"(?i)" if ignore_case else r"")
    22	        + r"^(##)( *)"  # groups 1, 2

    33	        bygroups(
    34	            Keyword.Declaration,
    35	            Text.Whitespace,
    36	            inner_token,
    37	            Text.Whitespace,
    38	            Punctuation,
