  2447	        ok = True  # False if we've found any error
  2448

  2450	            if (
  2451	                i not in all_actuals
  2452	                and
  2453	                # We accept the other iterables than tuple (including Any)
  2454	                # as star arguments because they could be empty, resulting no arguments.
  2455	                (kind != nodes.ARG_STAR or is_non_empty_tuple(actual_types[i]))

  2458	                # dictionaries and we can't tell it from their types
  2459	                kind != nodes.ARG_STAR2

  2462	                ok = False
  2463	                if kind != nodes.ARG_NAMED:
  2464	                    self.msg.too_many_arguments(callee, context)
  2465	                else:
  2466	                    assert actual_names, "Internal error: named kinds without names given"
  2467	                    act_name = actual_names[i]
  2468	                    assert act_name is not None
  2469	                    act_type = actual_types[i]
  2470	                    self.msg.unexpected_keyword_argument(callee, act_name, act_type, context)
  2471	                    is_unexpected_arg_error = True
  2472	            elif (
  2473	                kind == nodes.ARG_STAR and nodes.ARG_STAR not in callee.arg_kinds
  2474	            ) or kind == nodes.ARG_STAR2:

  2476	                if isinstance(actual_type, (TupleType, TypedDictType)):
  2477	                    if all_actuals.get(i, 0) < len(actual_type.items):

  2479	                        if kind != nodes.ARG_STAR2 or not isinstance(actual_type, TypedDictType):

  2482	                            self.msg.too_many_arguments_from_typed_dict(
  2483	                                callee, actual_type, context

  2487	                # *args/**kwargs can be applied even if the function takes a fixed
  2488	                # number of positional arguments. This may succeed at runtime.
  2489
  2490	        return ok, is_unexpected_arg_error

  2506	    def check_argument_types(
  2507	        self,
  2508	        arg_types: list[Type],
  2509	        arg_kinds: list[ArgKind],
  2510	        args: list[Expression],
  2511	        callee: CallableType,
  2512	        formal_to_actual: list[list[int]],
  2513	        context: Context,
  2514	        check_arg: ArgChecker | None = None,
  2515	        object_type: Type | None = None,
  2516	    ) -> None:
  2517	        """Check argument types against a callable type.
  2518
  2519	        Report errors if the argument types are not compatible.
  2520
  2521	        The check_call docstring describes some of the arguments.
  2522	        """
  2523	        check_arg = check_arg or self.check_arg
  2524	        # Keep track of consumed tuple *arg items.
  2525	        mapper = ArgTypeExpander(self.argument_infer_context())
  2526
  2527	        for arg_type, arg_kind in zip(arg_types, arg_kinds):
  2528	            arg_type = get_proper_type(arg_type)
  2529	            if arg_kind == nodes.ARG_STAR and not self.is_valid_var_arg(arg_type):
  2530	                self.msg.invalid_var_arg(arg_type, context)
  2531	            if arg_kind == nodes.ARG_STAR2 and not self.is_valid_keyword_var_arg(arg_type):
  2532	                is_mapping = is_subtype(
  2533	                    arg_type, self.chk.named_type("_typeshed.SupportsKeysAndGetItem")
  2534	                )
  2535	                self.msg.invalid_keyword_var_arg(arg_type, is_mapping, context)
  2536
  2537	        for i, actuals in enumerate(formal_to_actual):
  2538	            orig_callee_arg_type = get_proper_type(callee.arg_types[i])
  2539
  2540	            # Checking the case that we have more than one item but the first argument
  2541	            # is an unpack, so this would be something like:
  2542	            # [Tuple[Unpack[Ts]], int]
  2543	            #
  2544	            # In this case we have to check everything together, we do this by re-unifying
  2545	            # the suffices to the tuple, e.g. a single actual like
  2546	            # Tuple[Unpack[Ts], int]
  2547	            expanded_tuple = False
  2548	            actual_kinds = [arg_kinds[a] for a in actuals]
  2549	            if len(actuals) > 1:
  2550	                p_actual_type = get_proper_type(arg_types[actuals[0]])
  2551	                if (
  2552	                    isinstance(p_actual_type, TupleType)
  2553	                    and len(p_actual_type.items) == 1
  2554	                    and isinstance(p_actual_type.items[0], UnpackType)
  2555	                    and actual_kinds == [nodes.ARG_STAR] + [nodes.ARG_POS] * (len(actuals) - 1)
  2556	                ):
  2557	                    actual_types = [p_actual_type.items[0]] + [arg_types[a] for a in actuals[1:]]
  2558	                    if isinstance(orig_callee_arg_type, UnpackType):
  2559	                        p_callee_type = get_proper_type(orig_callee_arg_type.type)
  2560	                        if isinstance(p_callee_type, TupleType):
  2561	                            assert p_callee_type.items
  2562	                            callee_arg_types = p_callee_type.items
  2563	                            callee_arg_kinds = [nodes.ARG_STAR] + [nodes.ARG_POS] * (
  2564	                                len(p_callee_type.items) - 1
  2565	                            )
  2566	                            expanded_tuple = True
  2567
  2568	            if not expanded_tuple:
  2569	                actual_types = [arg_types[a] for a in actuals]
  2570	                if isinstance(orig_callee_arg_type, UnpackType):
  2571	                    unpacked_type = get_proper_type(orig_callee_arg_type.type)
  2572	                    if isinstance(unpacked_type, TupleType):
  2573	                        inner_unpack_index = find_unpack_in_list(unpacked_type.items)
  2574	                        if inner_unpack_index is None:
  2575	                            callee_arg_types = unpacked_type.items
  2576	                            callee_arg_kinds = [ARG_POS] * len(actuals)
  2577	                        else:
  2578	                            inner_unpack = unpacked_type.items[inner_unpack_index]
  2579	                            assert isinstance(inner_unpack, UnpackType)
  2580	                            inner_unpacked_type = get_proper_type(inner_unpack.type)
  2581	                            if isinstance(inner_unpacked_type, TypeVarTupleType):
  2582	                                # This branch mimics the expanded_tuple case above but for
  2583	                                # the case where caller passed a single * unpacked tuple argument.
  2584	                                callee_arg_types = unpacked_type.items
  2585	                                callee_arg_kinds = [
  2586	                                    ARG_POS if i != inner_unpack_index else ARG_STAR
  2587	                                    for i in range(len(unpacked_type.items))
  2588	                                ]
  2589	                            else:
  2590	                                # We assume heterogeneous tuples are desugared earlier.
  2591	                                assert isinstance(inner_unpacked_type, Instance)
  2592	                                assert inner_unpacked_type.type.fullname == "builtins.tuple"
  2593	                                callee_arg_types = (
  2594	                                    unpacked_type.items[:inner_unpack_index]
  2595	                                    + [inner_unpacked_type.args[0]]
  2596	                                    * (len(actuals) - len(unpacked_type.items) + 1)
  2597	                                    + unpacked_type.items[inner_unpack_index + 1 :]
  2598	                                )
  2599	                                callee_arg_kinds = [ARG_POS] * len(actuals)
  2600	                    elif isinstance(unpacked_type, TypeVarTupleType):
  2601	                        callee_arg_types = [orig_callee_arg_type]
  2602	                        callee_arg_kinds = [ARG_STAR]
  2603	                    else:
  2604	                        assert isinstance(unpacked_type, Instance)
  2605	                        assert unpacked_type.type.fullname == "builtins.tuple"
  2606	                        callee_arg_types = [unpacked_type.args[0]] * len(actuals)
  2607	                        callee_arg_kinds = [ARG_POS] * len(actuals)
  2608	                else:
  2609	                    callee_arg_types = [orig_callee_arg_type] * len(actuals)
  2610	                    callee_arg_kinds = [callee.arg_kinds[i]] * len(actuals)
  2611
  2612	            assert len(actual_types) == len(actuals) == len(actual_kinds)
  2613
  2614	            if len(callee_arg_types) != len(actual_types):
  2615	                if len(actual_types) > len(callee_arg_types):
  2616	                    self.chk.msg.too_many_arguments(callee, context)
  2617	                else:
  2618	                    self.chk.msg.too_few_arguments(callee, context, None)
  2619	                continue
  2620
  2621	            assert len(callee_arg_types) == len(actual_types)
  2622	            assert len(callee_arg_types) == len(callee_arg_kinds)
  2623	            for actual, actual_type, actual_kind, callee_arg_type, callee_arg_kind in zip(
  2624	                actuals, actual_types, actual_kinds, callee_arg_types, callee_arg_kinds
  2625	            ):
  2626	                # Check that a *arg is valid as varargs.
  2627	                expanded_actual = mapper.expand_actual_type(
  2628	                    actual_type,
  2629	                    actual_kind,
  2630	                    callee.arg_names[i],
  2631	                    callee_arg_kind,
  2632	                    allow_unpack=isinstance(callee_arg_type, UnpackType),
  2633	                )
  2634	                check_arg(
  2635	                    expanded_actual,
  2636	                    actual_type,
  2637	                    actual_kind,
  2638	                    callee_arg_type,
  2639	                    actual + 1,
  2640	                    i + 1,
  2641	                    callee,
  2642	                    object_type,
  2643	                    args[actual],
  2644	                    context,
  2645	                )

  2689	    def check_overload_call(
  2690	        self,
  2691	        callee: Overloaded,
  2692	        args: list[Expression],
  2693	        arg_kinds: list[ArgKind],
  2694	        arg_names: Sequence[str | None] | None,
  2695	        callable_name: str | None,
  2696	        object_type: Type | None,
  2697	        context: Context,
  2698	    ) -> tuple[Type, Type]:
  2699	        """Checks a call to an overloaded function."""
  2700	        # Normalize unpacked kwargs before checking the call.
  2701	        callee = callee.with_unpacked_kwargs()
  2702	        arg_types = self.infer_arg_types_in_empty_context(args)
  2703	        # Step 1: Filter call targets to remove ones where the argument counts don't match
  2704	        plausible_targets = self.plausible_overload_call_targets(
  2705	            arg_types, arg_kinds, arg_names, callee
  2706	        )
  2707
  2708	        # Step 2: If the arguments contain a union, we try performing union math first,
  2709	        #         instead of picking the first matching overload.
  2710	        #         This is because picking the first overload often ends up being too greedy:
  2711	        #         for example, when we have a fallback alternative that accepts an unrestricted
  2712	        #         typevar. See https://github.com/python/mypy/issues/4063 for related discussion.
  2713	        erased_targets: list[CallableType] | None = None
  2714	        unioned_result: tuple[Type, Type] | None = None
  2715
  2716	        # Determine whether we need to encourage union math. This should be generally safe,
  2717	        # as union math infers better results in the vast majority of cases, but it is very
  2718	        # computationally intensive.
  2719	        none_type_var_overlap = self.possible_none_type_var_overlap(arg_types, plausible_targets)
  2720	        union_interrupted = False  # did we try all union combinations?
  2721	        if any(self.real_union(arg) for arg in arg_types):
  2722	            try:
  2723	                with self.msg.filter_errors():
  2724	                    unioned_return = self.union_overload_result(
  2725	                        plausible_targets,
  2726	                        args,
  2727	                        arg_types,
  2728	                        arg_kinds,
  2729	                        arg_names,
  2730	                        callable_name,
  2731	                        object_type,
  2732	                        none_type_var_overlap,
  2733	                        context,
  2734	                    )
  2735	            except TooManyUnions:
  2736	                union_interrupted = True
  2737	            else:
  2738	                # Record if we succeeded. Next we need to see if maybe normal procedure
  2739	                # gives a narrower type.
  2740	                if unioned_return:
  2741	                    returns, inferred_types = zip(*unioned_return)
  2742	                    # Note that we use `combine_function_signatures` instead of just returning
  2743	                    # a union of inferred callables because for example a call
  2744	                    # Union[int -> int, str -> str](Union[int, str]) is invalid and
  2745	                    # we don't want to introduce internal inconsistencies.
  2746	                    unioned_result = (
  2747	                        make_simplified_union(list(returns), context.line, context.column),
  2748	                        self.combine_function_signatures(get_proper_types(inferred_types)),
  2749	                    )
  2750
  2751	        # Step 3: We try checking each branch one-by-one.
  2752	        inferred_result = self.infer_overload_return_type(
  2753	            plausible_targets,
  2754	            args,
  2755	            arg_types,
  2756	            arg_kinds,
  2757	            arg_names,
  2758	            callable_name,
  2759	            object_type,
  2760	            context,
  2761	        )
  2762	        # If any of checks succeed, stop early.
  2763	        if inferred_result is not None and unioned_result is not None:
  2764	            # Both unioned and direct checks succeeded, choose the more precise type.
  2765	            if (
  2766	                is_subtype(inferred_result[0], unioned_result[0])
  2767	                and not isinstance(get_proper_type(inferred_result[0]), AnyType)
  2768	                and not none_type_var_overlap
  2769	            ):
  2770	                return inferred_result
  2771	            return unioned_result
  2772	        elif unioned_result is not None:
  2773	            return unioned_result
  2774	        elif inferred_result is not None:
  2775	            return inferred_result
  2776
  2777	        # Step 4: Failure. At this point, we know there is no match. We fall back to trying
  2778	        #         to find a somewhat plausible overload target using the erased types
  2779	        #         so we can produce a nice error message.
  2780	        #
  2781	        #         For example, suppose the user passes a value of type 'List[str]' into an
  2782	        #         overload with signatures f(x: int) -> int and f(x: List[int]) -> List[int].
  2783	        #
  2784	        #         Neither alternative matches, but we can guess the user probably wants the
  2785	        #         second one.
  2786	        erased_targets = self.overload_erased_call_targets(
  2787	            plausible_targets, arg_types, arg_kinds, arg_names, args, context
  2788	        )
  2789
  2790	        # Step 5: We try and infer a second-best alternative if possible. If not, fall back
  2791	        #         to using 'Any'.
  2792	        if len(erased_targets) > 0:
  2793	            # Pick the first plausible erased target as the fallback
  2794	            # TODO: Adjust the error message here to make it clear there was no match.
  2795	            #       In order to do this, we need to find a clean way of associating
  2796	            #       a note with whatever error message 'self.check_call' will generate.
  2797	            #       In particular, the note's line and column numbers need to be the same
  2798	            #       as the error's.
  2799	            target: Type = erased_targets[0]
  2800	        else:
  2801	            # There was no plausible match: give up
  2802	            target = AnyType(TypeOfAny.from_error)
  2803	            if not is_operator_method(callable_name):
  2804	                code = None
  2805	            else:
  2806	                code = codes.OPERATOR
  2807	            self.msg.no_variant_matches_arguments(callee, arg_types, context, code=code)
  2808
  2809	        result = self.check_call(
  2810	            target,
  2811	            args,
  2812	            arg_kind
