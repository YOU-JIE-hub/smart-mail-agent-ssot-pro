   393	    def identity(self, maxLen=30):
   394	        '''Identify our selves as well as possible'''
   395	        if self.ident: return self.ident
   396	        vx = None
   397	        nr = getattr(self,'_nrows','unknown')
   398	        nc = getattr(self,'_ncols','unknown')
   399	        cv = getattr(self,'_cellvalues',None)
   400	        rh = getattr(self, '_rowHeights', None)
   401	        if cv and 'unknown' not in (nr,nc):
   402	            b = 0
   403	            for i in range(nr):
   404	                for j in range(nc):
   405	                    v = cv[i][j]
   406	                    if isinstance(v,(list,tuple,Flowable)):
   407	                        if not isinstance(v,(tuple,list)): v = (v,)
   408	                        r = ''
   409	                        for vij in v:
   410	                            r = vij.identity(maxLen)
   411	                            if r and r[-4:]!='>...':
   412	                                break
   413	                        if r and r[-4:]!='>...':
   414	                            ix, jx, vx, b = i, j, r, 1
   415	                    else:
   416	                        v = v is None and '' or str(v)
   417	                        ix, jx, vx = i, j, v
   418	                        b = (vx and isinstance(v,strTypes)) and 1 or 0
   419	                        if maxLen: vx = vx[:maxLen]

   555	    def _elementWidth(self,v,s):
   556	        if isinstance(v,(list,tuple)):
   557	            w = 0
   558	            for e in v:
   559	                ew = self._elementWidth(e,s)
   560	                if ew is None: return None
   561	                w = max(w,ew)
   562	            return w
   563	        elif isinstance(v,Flowable):
   564	            if v._fixedWidth:
   565	                if hasattr(v, 'width') and isinstance(v.width,(int,float)): return v.width
   566	                if hasattr(v, 'drawWidth') and isinstance(v.drawWidth,(int,float)): return v.drawWidth
   567	            if hasattr(v,'__styledWrap__'): #very experimental
   568	                try:
   569	                    return getattr(v,'__styledWrap__')(s)[0]
   570	                except:
   571	                    pass
   572	        # Even if something is fixedWidth, the attribute to check is not
   573	        # necessarily consistent (cf. Image.drawWidth).  Therefore, we'll
   574	        # be extra-careful and fall through to this code if necessary.

   579	            except AttributeError:
   580	                pass

   583	        else:

   586	            except:
   587	                return 0

   589	        fontSize = s.fontsize
   590	        return max([stringWidth(x,fontName,fontSize) for x in v])
