ues from ** items.
   860	                    # TODO: for duplicate keys, type-check all values.
   861	                    result[literal_value] = [item_arg]
   862	                    always_present_keys.add(literal_value)
   863	            else:
   864	                last_star_found = item_arg
   865	                if not self.validate_star_typeddict_item(
   866	                    item_arg, callee, result, always_present_keys
   867	                ):
   868	                    return None
   869	        if self.chk.options.extra_checks and last_star_found is not None:
   870	            absent_keys = []
   871	            for key in callee.items:
   872	                if key not in callee.required_keys and key not in result:
   873	                    absent_keys.append(key)
   874	            if absent_keys:
   875	                # Having an optional key not explicitly declared by a ** unpacked
   876	                # TypedDict is unsafe, it may be an (incompatible) subtype at runtime.
   877	                # TODO: catch the cases where a declared key is overridden by a subsequent
   878	                # ** item without it (and not again overridden with complete ** item).
   879	                self.msg.non_required_keys_absent_with_star(absent_keys, last_star_found)
   880	        return result, always_present_keys

   882	    def validate_star_typeddict_item(
   883	        self,
   884	        item_arg: Expression,
   885	        callee: TypedDictType,
   886	        result: dict[str, list[Expression]],
   887	        always_present_keys: set[str],
   888	    ) -> bool:
   889	        """Update keys/expressions from a ** expression in TypedDict constructor.
   890
   891	        Note `result` and `always_present_keys` are updated in place. Return true if the
   892	        expression `item_arg` may valid in `callee` TypedDict context.
   893	        """
   894	        inferred = get_proper_type(self.accept(item_arg, type_context=callee))
   895	        possible_tds = []
   896	        if isinstance(inferred, TypedDictType):
   897	            possible_tds = [inferred]
   898	        elif isinstance(inferred, UnionType):
   899	            for item in get_proper_types(inferred.relevant_items()):
   900	                if isinstance(item, TypedDictType):
   901	                    possible_tds.append(item)
   902	                elif not self.valid_unpack_fallback_item(item):
   903	                    self.msg.unsupported_target_for_star_typeddict(item, item_arg)
   904	                    return False
   905	        elif not self.valid_unpack_fallback_item(inferred):
   906	            self.msg.unsupported_target_for_star_typeddict(inferred, item_arg)
   907	            return False
   908	        all_keys: set[str] = set()
   909	        for td in possible_tds:
   910	            all_keys |= td.items.keys()
   911	        for key in all_keys:
   912	            arg = TempNode(
   913	                UnionType.make_union([td.items[key] for td in possible_tds if key in td.items])
   914	            )
   915	            arg.set_line(item_arg)
   916	            if all(key in td.required_keys for td in possible_tds):
   917	                always_present_keys.add(key)
   918	                # Always present keys override previously found values. This is done
   919	                # to support use cases like `Config({**defaults, **overrides})`, where
   920	                # some `overrides` types are narrower that types in `defaults`, and
   921	                # former are too wide for `Config`.
   922	                if result[key]:
   923	                    first = result[key][0]
   924	                    if not isinstance(first, TempNode):
   925	                        # We must always preserve any non-synthetic values, so that
   926	                        # we will accept them even if they are shadowed.
   927	                        result[key] = [first, arg]
   928	                    else:
   929	                        result[key] = [arg]
   930	                else:
   931	                    result[key] = [arg]
   932	            else:
   933	                # If this key is not required at least in some item of a union
   934	                # it may not shadow previous item, so we need to type check both.
   935	                result[key].append(arg)
   936	        return True

   938	    def valid_unpack_fallback_item(self, typ: ProperType) -> bool:
   939	        if isinstance(typ, AnyType):
   940	            return True
   941	        if not isinstance(typ, Instance) or not typ.type.has_base("typing.Mapping"):
   942	            return False
   943	        mapped = map_instance_to_supertype(typ, self.chk.lookup_typeinfo("typing.Mapping"))
   944	        return all(isinstance(a, AnyType) for a in get_proper_types(mapped.args))

  1009	    def check_typeddict_call_with_kwargs(
  1010	        self,
  1011	        callee: TypedDictType,
  1012	        kwargs: dict[str, list[Expression]],
  1013	        context: Context,
  1014	        orig_callee: Type | None,
  1015	        always_present_keys: set[str],
  1016	    ) -> Type:
  1017	        actual_keys = kwargs.keys()
  1018	        if callee.to_be_mutated:
  1019	            assigned_readonly_keys = actual_keys & callee.readonly_keys
  1020	            if assigned_readonly_keys:
  1021	                self.msg.readonly_keys_mutated(assigned_readonly_keys, context=context)
  1022	        if not (
  1023	            callee.required_keys <= always_present_keys and actual_keys <= callee.items.keys()
  1024	        ):
  1025	            if not (actual_keys <= callee.items.keys()):
  1026	                self.msg.unexpected_typeddict_keys(
  1027	                    callee,
  1028	                    expected_keys=[
  1029	                        key
  1030	                        for key in callee.items.keys()
  1031	                        if key in callee.required_keys or key in actual_keys
  1032	                    ],
  1033	                    actual_keys=list(actual_keys),
  1034	                    context=context,
  1035	                )
  1036	            if not (callee.required_keys <= always_present_keys):
  1037	                self.msg.unexpected_typeddict_keys(
  1038	                    callee,
  1039	                    expected_keys=[
  1040	                        key for key in callee.items.keys() if key in callee.required_keys
  1041	                    ],
  1042	                    actual_keys=[
  1043	                        key for key in always_present_keys if key in callee.required_keys
  1044	                    ],
  1045	                    context=context,
  1046	                )
  1047	            if callee.required_keys > actual_keys:
  1048	                # found_set is a sub-set of the required_keys
  1049	                # This means we're missing some keys and as such, we can't
  1050	                # properly type the object
  1051	                return AnyType(TypeOfAny.from_error)
  1052
  1053	        orig_callee = get_proper_type(orig_callee)
  1054	        if isinstance(orig_callee, CallableType):
  1055	            infer_callee = orig_callee
  1056	        else:
  1057	            # Try reconstructing from type context.
  1058	            if callee.fallback.type.special_alias is not None:
  1059	                infer_callee = self.typeddict_callable(callee.fallback.type)
  1060	            else:
  1061	                # Likely a TypedDict type generated by a plugin.
  1062	                infer_callee = self.typeddict_callable_from_context(callee)
  1063
  1064	        # We don't show any errors, just infer types in a generic TypedDict type,
  1065	        # a custom error message will be given below, if there are errors.
  1066	        with self.msg.filter_errors(), self.chk.local_type_map():
  1067	            orig_ret_type, _ = self.check_callable_call(
  1068	                infer_callee,
  1069	                # We use first expression for each key to infer type variables of a generic
  1070	                # TypedDict. This is a bit arbitrary, but in most cases will work better than
  1071	                # trying to infer a union or a join.
  1072	                [args[0] for args in kwargs.values()],
  1073	                [ArgKind.ARG_NAMED] * len(kwargs),
  1074	                context,
  1075	                list(kwargs.keys()),
  1076	                None,
  1077	                None,
  1078	                None,
  1079	            )
  1080
  1081	        ret_type = get_proper_type(orig_ret_type)
  1082	        if not isinstance(ret_type, TypedDictType):
  1083	            # If something went really wrong, type-check call with original type,
  1084	            # this may give a better error message.
  1085	            ret_type = callee
  1086
  1087	        for item_name, item_expected_type in ret_type.items.items():
  1088	            if item_name in kwargs:
  1089	                item_values = kwargs[item_name]
  1090	                for item_value in item_values:
  1091	                    self.chk.check_simple_assignment(
  1092	                        lvalue_type=item_expected_type,
  1093	                        rvalue=item_value,
  1094	                        context=item_value,
  1095	                        msg=ErrorMessage(
  1096	                            message_registry.INCOMPATIBLE_TYPES.value, code=codes.TYPEDDICT_ITEM
  1097	                        ),
  1098	                        lvalue_name=f'TypedDict item "{item_name}"',
  1099	                        rvalue_name="expression",
  1100	                    )
  1101
  1102	        return orig_ret_type

  1138	    def try_infer_partial_type(self, e: CallExpr) -> None:
  1139	        """Try to make partial type precise from a call."""
  1140	        if not isinstance(e.callee, MemberExpr):
  1141	            return
  1142	        callee = e.callee
  1143	        if isinstance(callee.expr, RefExpr):
  1144	            # Call a method with a RefExpr callee, such as 'x.method(...)'.
  1145	            ret = self.get_partial_var(callee.expr)
  1146	            if ret is None:
  1147	                return
  1148	            var, partial_types = ret
  1149	            typ = self.try_infer_partial_value_type_from_call(e, callee.name, var)
  1150	            # Var may be deleted from partial_types in try_infer_partial_value_type_from_call
  1151	            if typ is not None and var in partial_types:
  1152	                self.chk.replace_partial_type(var, typ, partial_types)
  1153	        elif isinstance(callee.expr, IndexExpr) and isinstance(callee.expr.base, RefExpr):
  1154	            # Call 'x[y].method(...)'; may infer type of 'x' if it's a partial defaultdict.
  1155	            if callee.expr.analyzed is not None:
  1156	                return  # A special form
  1157	            base = callee.expr.base
  1158	            index = callee.expr.index
  1159	            ret = self.get_partial_var(base)
  1160	            if ret is None:
  1161	                return
  1162	            var, partial_types = ret
  1163	            partial_type = get_partial_instance_type(var.type)
  1164	            if partial_type is None or partial_type.value_type is None:
  1165	                return
  1166	            value_type = self.try_infer_partial_value_type_from_call(e, callee.name, var)
  1167	            if value_type is not None:
  1168	                # Infer key type.
  1169	                key_type = self.accept(index)
  1170	                if mypy.checker.is_valid_inferred_type(key_type, self.chk.options):
  1171	                    # Store inferred partial type.
  1172	                    assert partial_type.type is not None
  1173	                    typename = partial_type.type.fullname
  1174	                    new_type = self.chk.named_generic_type(typename, [key_type, value_type])
  1175	                    self.chk.replace_partial_type(var, new_type, partial_types)

  1188	    def try_infer_partial_value_type_from_call(
  1189	        self, e: CallExpr, methodname: str, var: Var
  1190	    ) -> Instance | None:
  1191	        """Try to make partial type precise from a call such as 'x.append(y)'."""
  1192	        if self.chk.current_node_deferred:
  1193	            return None
  1194	        partial_type = get_partial_instance_type(var.type)
  1195	        if partial_type is None:
  1196	            return None
  1197	        if partial_type.value_type:
  1198	            typename = partial_type.value_type.type.fullname
  1199	        else:
  1200	            assert partial_type.type is not None
  1201	            typename = partial_type.type.fullname
  1202	        # Sometimes we can infer a full type for a partial List, Dict or Set type.
  1203	        # TODO: Don't infer argument expression twice.
  1204	        if (
  1205	            typename in self.item_args
  1206	            and methodname in self.item_args[typename]
  1207	            and e.arg_kinds == [ARG_POS]
  1208	        ):
  1209	            item_type = self.accept(e.args[0])
  1210	            if mypy.checker.is_valid_inferred_type(item_type, self.chk.options):
  1211	                return self.chk.named_generic_type(typename, [item_type])
  1212	        elif (
  1213	            typename in self.container_args
  1214	            and methodname in self.container_args[typename]
  1215	            and e.arg_kinds == [ARG_POS]
  1216	        ):
  1217	            arg_type = get_proper_type(self.accept(e.args[0]))
  1218	            if isinstance(arg_type, Instance):
  1219	                arg_typename = arg_type.type.fullname
  1220	                if arg_typename in self.container_args[typename][methodname]:
  1221	                    if all(
  1222	                        mypy.checker.is_valid_inferred_type(item_type, self.chk.options)
  1223	                        for item_type in arg_type.args
  1224	                    ):
  1225	                        return self.chk.named_generic_type(typename, list(arg_type.args))
  1226	            elif isinstance(arg_type, AnyType):
  1227	                return self.chk.named_type(typename)
  1228
  1229	        return None

  1244

  1256	        formal_arg_exprs: list[list[Expression]] = [[] for _
