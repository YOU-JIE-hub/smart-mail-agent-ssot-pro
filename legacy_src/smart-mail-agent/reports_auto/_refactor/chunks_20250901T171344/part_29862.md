   122	def get_mypy_config(
   123	    mypy_options: list[str],
   124	    only_compile_paths: Iterable[str] | None,
   125	    compiler_options: CompilerOptions,
   126	    fscache: FileSystemCache | None,
   127	) -> tuple[list[BuildSource], list[BuildSource], Options]:
   128	    """Construct mypy BuildSources and Options from file and options lists"""
   129	    all_sources, options = process_options(mypy_options, fscache=fscache)
   130	    if only_compile_paths is not None:
   131	        paths_set = set(only_compile_paths)
   132	        mypyc_sources = [s for s in all_sources if s.path in paths_set]
   133	    else:
   134	        mypyc_sources = all_sources
   135
   136	    if compiler_options.separate:
   137	        mypyc_sources = [
   138	            src for src in mypyc_sources if src.path and not src.path.endswith("__init__.py")
   139	        ]
   140
   141	    if not mypyc_sources:
   142	        return mypyc_sources, all_sources, options
   143
   144	    # Override whatever python_version is inferred from the .ini file,
   145	    # and set the python_version to be the currently used version.
   146	    options.python_version = sys.version_info[:2]
   147
   148	    if options.python_version[0] == 2:
   149	        fail("Python 2 not supported")
   150	    if not options.strict_optional:
   151	        fail("Disabling strict optional checking not supported")
   152	    options.show_traceback = True
   153	    # Needed to get types for all AST nodes
   154	    options.export_types = True
   155	    # We use mypy incremental mode when doing separate/incremental mypyc compilation
   156	    options.incremental = compiler_options.separate
   157	    options.preserve_asts = True
   158
   159	    for source in mypyc_sources:
   160	        options.per_module_options.setdefault(source.module, {})["mypyc"] = True
   161
   162	    return mypyc_sources, all_sources, options

   165	def generate_c_extension_shim(
   166	    full_module_name: str, module_name: str, dir_name: str, group_name: str
   167	) -> str:
   168	    """Create a C extension shim with a passthrough PyInit function.
   169
   170	    Arguments:
   171	        full_module_name: the dotted full module name
   172	        module_name: the final component of the module name
   173	        dir_name: the directory to place source code
   174	        group_name: the name of the group
   175	    """
   176	    cname = "%s.c" % full_module_name.replace(".", os.sep)
   177	    cpath = os.path.join(dir_name, cname)
   178
   179	    # We load the C extension shim template from a file.
   180	    # (So that the file could be reused as a bazel template also.)
   181	    with open(os.path.join(include_dir(), "module_shim.tmpl")) as f:
   182	        shim_template = f.read()
   183
   184	    write_file(
   185	        cpath,
   186	        shim_template.format(
   187	            modname=module_name,
   188	            libname=shared_lib_name(group_name),
   189	            full_modname=exported_name(full_module_name),
   190	        ),
   191	    )
   192
   193	    return cpath

   196	def group_name(modules: list[str]) -> str:
   197	    """Produce a probably unique name for a group from a list of module names."""
   198	    if len(modules) == 1:
   199	        return modules[0]
   200
   201	    h = hashlib.sha1()
   202	    h.update(",".join(modules).encode())
   203	    return h.hexdigest()[:20]

   211	def generate_c(
   212	    sources: list[BuildSource],
   213	    options: Options,
   214	    groups: emitmodule.Groups,
   215	    fscache: FileSystemCache,
   216	    compiler_options: CompilerOptions,
   217	) -> tuple[list[list[tuple[str, str]]], str]:
   218	    """Drive the actual core compilation step.
   219
   220	    The groups argument describes how modules are assigned to C
   221	    extension modules. See the comments on the Groups type in
   222	    mypyc.emitmodule for details.
   223
   224	    Returns the C source code and (for debugging) the pretty printed IR.
   225	    """
   226	    t0 = time.time()
   227
   228	    try:
   229	        result = emitmodule.parse_and_typecheck(
   230	            sources, options, compiler_options, groups, fscache
   231	        )
   232	    except CompileError as e:
   233	        emit_messages(options, e.messages, time.time() - t0, serious=(not e.use_stdout))
   234	        sys.exit(1)
   235
   236	    t1 = time.time()
   237	    if result.errors:
   238	        emit_messages(options, result.errors, t1 - t0)
   239	        sys.exit(1)
   240
   241	    if compiler_options.verbose:
   242	        print(f"Parsed and typechecked in {t1 - t0:.3f}s")
   243
   244	    errors = Errors(options)
   245	    modules, ctext, mapper = emitmodule.compile_modules_to_c(
   246	        result, compiler_options=compiler_options, errors=errors, groups=groups
   247	    )
   248	    t2 = time.time()
   249	    emit_messages(options, errors.new_messages(), t2 - t1)
   250	    if errors.num_errors:
   251	        # No need to stop the build if only warnings were emitted.
   252	        sys.exit(1)
   253
   254	    if compiler_options.verbose:
   255	        print(f"Compiled to C in {t2 - t1:.3f}s")
   256
   257	    if options.mypyc_annotation_file:
   258	        generate_annotated_html(options.mypyc_annotation_file, result, modules, mapper)
   259
   260	    return ctext, "\n".join(format_modules(modules))

   263	def build_using_shared_lib(
   264	    sources: list[BuildSource],
   265	    group_name: str,
   266	    cfiles: list[str],
   267	    deps: list[str],
   268	    build_dir: str,
   269	    extra_compile_args: list[str],
   270	) -> list[Extension]:
   271	    """Produce the list of extension modules when a shared library is needed.
   272
   273	    This creates one shared library extension module that all of the
   274	    others import and then one shim extension module for each
   275	    module in the build, that simply calls an initialization function
   276	    in the shared library.
   277
   278	    The shared library (which lib_name is the name of) is a python
   279	    extension module that exports the real initialization functions in
   280	    Capsules stored in module attributes.
   281	    """
   282	    extensions = [
   283	        get_extension()(
   284	            shared_lib_name(group_name),
   285	            sources=cfiles,
   286	            include_dirs=[include_dir(), build_dir],
   287	            depends=deps,
   288	            extra_compile_args=extra_compile_args,
   289	        )
   290	    ]
   291
   292	    for source in sources:
   293	        module_name = source.module.split(".")[-1]
   294	        shim_file = generate_c_extension_shim(source.module, module_name, build_dir, group_name)
   295
   296	        # We include the __init__ in the "module name" we stick in the Extension,
   297	        # since this seems to be needed for it to end up in the right place.
   298	        full_module_name = source.module
   299	        assert source.path
   300	        if os.path.split(source.path)[1] == "__init__.py":
   301	            full_module_name += ".__init__"
   302	        extensions.append(
   303	            get_extension()(
   304	                full_module_name, sources=[shim_file], extra_compile_args=extra_compile_args
   305	            )
   306	        )
   307
   308	    return extensions

   328	def write_file(path: str, contents: str) -> None:
   329	    """Write data into a file.
   330
   331	    If the file already exists and has the same contents we
   332	    want to write, skip writing so as to preserve the mtime
   333	    and avoid triggering recompilation.
   334	    """
   335	    # We encode it ourselves and open the files as binary to avoid windows
   336	    # newline translation
   337	    encoded_contents = contents.encode("utf-8")
   338	    try:
   339	        with open(path, "rb") as f:
   340	            old_contents: bytes | None = f.read()
   341	    except OSError:
   342	        old_contents = None
   343	    if old_contents != encoded_contents:
   344	        os.makedirs(os.path.dirname(path), exist_ok=True)
   345	        with open(path, "wb") as g:
   346	            g.write(encoded_contents)
   347
   348	        # Fudge the mtime forward because otherwise when two builds happen close
   349	        # together (like in a test) setuptools might not realize the source is newer
   350	        # than the new artifact.
   351	        # XXX: This is bad though.
   352	        new_mtime = os.stat(path).st_mtime + 1
   353	        os.utime(path, times=(new_mtime, new_mtime))

   356	def construct_groups(
   357	    sources: list[BuildSource],
   358	    separate: bool | list[tuple[list[str], str | None]],
   359	    use_shared_lib: bool,
   360	    group_name_override: str | None,
   361	) -> emitmodule.Groups:
   362	    """Compute Groups given the input source list and separate configs.
   363
   364	    separate is the user-specified configuration for how to assign
   365	    modules to compilation groups (see mypycify docstring for details).
   366
   367	    This takes that and expands it into our internal representation of
   368	    group configuration, documented in mypyc.emitmodule's definition
   369	    of Group.
   370	    """
   371
   372	    if separate is True:
   373	        groups: emitmodule.Groups = [([source], None) for source in sources]
   374	    elif isinstance(separate, list):
   375	        groups = []
   376	        used_sources = set()
   377	        for files, name in separate:
   378	            group_sources = [src for src in sources if src.path in files]
   379	            groups.append((group_sources, name))
   380	            used_sources.update(group_sources)
   381	        unused_sources = [src for src in sources if src not in used_sources]
   382	        if unused_sources:
   383	            groups.extend([([source], None) for source in unused_sources])
   384	    else:
   385	        groups = [(sources, None)]
   386
   387	    # Generate missing names
   388	    for i, (group, name) in enumerate(groups):
   389	        if use_shared_lib and not name:
   390	            if group_name_override is not None:
   391	                name = group_name_override
   392	            else:
   393	                name = group_name([source.module for source in group])
   394	        groups[i] = (group, name)
   395
   396	    return groups

   399	def get_header_deps(cfiles: list[tuple[str, str]]) -> list[str]:
   400	    """Find all the headers used by a group of cfiles.
   401
   402	    We do this by just regexping the source, which is a bit simpler than
   403	    properly plumbing the data through.
   404
   405	    Arguments:
   406	        cfiles: A list of (file name, file contents) pairs.
   407	    """
   408	    headers: set[str] = set()
   409	    for _, contents in cfiles:
   410	        headers.update(re.findall(r'#include "(.*)"', contents))
   411
   412	    return sorted(headers)

   415	def mypyc_build(
   416	    paths: list[str],
   417	    compiler_options: CompilerOptions,
   418	    *,
   419	    separate: bool | list[tuple[list[str], str | None]] = False,
   420	    only_compile_paths: Iterable[str] | None = None,
   421	    skip_cgen_input: Any | None = None,
   422	    always_use_shared_lib: bool = False,
   423	) -> tuple[emitmodule.Groups, list[tuple[list[str], list[str]]]]:
   424	    """Do the front and middle end of mypyc building, producing and writing out C source."""
   425	    fscache = FileSystemCache()
   426	    mypyc_sources, all_sources, options = get_mypy_config(
   427	        paths, only_compile_paths, compiler_options, fscache
   428	    )
   429
   430	    # We generate a shared lib if there are multiple modules or if any
   431	    # of the modules are in package. (Because I didn't want to fuss
   432	    # around with making the single module code handle packages.)
   433	    use_shared_lib = (
   434	        len(mypyc_sources) > 1
   435	        or any("." in x.module for x in mypyc_sources)
   436	        or always_use_shared_lib
   437	    )
   438
   439	    groups = construct_groups(mypyc_sources, separate, use_shared_lib, compiler_options.group_name)
   440
   441	    if compiler_options.group_name is not None:
   442	        assert len(groups) == 1, "If using custom group_name, only one group is expected"
   443
   444	    # We let the test harness just pass in the c file contents instead
   445	    # so that it can do a corner-cutting version without full stubs.
   446	    if not skip_cgen_input:
   447	        group_cfiles, ops_text = generate_c(
   448	            all_sources, options, groups, fscache, compiler_options=compiler_options
   449	        )
   450	        # TODO: unique names?
   451	        write_file(os.path.join(compiler_options.target_dir, "ops.txt"), ops_text)
   452	    else:
   453	        group_cfiles = skip_cgen_input
   454
   455	    # Write out the generated C and collect the files for each group
   456	    # Should this be here??
   457	    group_cfilenames: list[tuple[list[str], list[str]]] = []
   458	    for cfiles in group_cfiles:
   459	        cfilenames = []
   460	        for cfile, ctext in cfiles:
   461	            cfile = os.path.join(compiler_options.target_dir, cfile)
   462	            if not options.mypyc_skip_c_generation:
   463	                write_file(cfile, ctext)
   464	            if os.path.splitext(cfile)[1] == ".c":
   465	                cfilenames.append(cfile)
   466
   467	        deps = [os.path.join(compiler_options.target_dir, dep) for dep in get_header_deps(cfiles)]
   468	        group_cfilenames.append((cfilenames, deps))
   469
   470	    return groups, group_cfilenames
