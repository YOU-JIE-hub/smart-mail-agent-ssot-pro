  2284	    def get_arg_infer_passes(
  2285	        self,
  2286	        callee: CallableType,
  2287	        args: list[Expression],
  2288	        arg_types: list[Type],
  2289	        formal_to_actual: list[list[int]],
  2290	        num_actuals: int,
  2291	    ) -> list[int]:
  2292	        """Return pass numbers for args for two-pass argument type inference.
  2293
  2294	        For each actual, the pass number is either 1 (first pass) or 2 (second
  2295	        pass).
  2296
  2297	        Two-pass argument type inference primarily lets us infer types of
  2298	        lambdas more effectively.
  2299	        """
  2300	        res = [1] * num_actuals
  2301	        for i, arg in enumerate(callee.arg_types):
  2302	            skip_param_spec = False
  2303	            p_formal = get_proper_type(callee.arg_types[i])
  2304	            if isinstance(p_formal, CallableType) and p_formal.param_spec():
  2305	                for j in formal_to_actual[i]:
  2306	                    p_actual = get_proper_type(arg_types[j])
  2307	                    # This is an exception from the usual logic where we put generic Callable
  2308	                    # arguments in the second pass. If we have a non-generic actual, it is
  2309	                    # likely to infer good constraints, for example if we have:
  2310	                    #   def run(Callable[P, None], *args: P.args, **kwargs: P.kwargs) -> None: ...
  2311	                    #   def test(x: int, y: int) -> int: ...
  2312	                    #   run(test, 1, 2)
  2313	                    # we will use `test` for inference, since it will allow to infer also
  2314	                    # argument *names* for P <: [x: int, y: int].
  2315	                    if isinstance(p_actual, Instance):
  2316	                        call_method = find_member("__call__", p_actual, p_actual, is_operator=True)
  2317	                        if call_method is not None:
  2318	                            p_actual = get_proper_type(call_method)
  2319	                    if (
  2320	                        isinstance(p_actual, CallableType)
  2321	                        and not p_actual.variables
  2322	                        and not isinstance(args[j], LambdaExpr)
  2323	                    ):
  2324	                        skip_param_spec = True
  2325	                        break
  2326	            if not skip_param_spec and arg.accept(ArgInferSecondPassQuery()):
  2327	                for j in formal_to_actual[i]:
  2328	                    res[j] = 2
  2329	        return res

  2431	    def check_for_extra_actual_arguments(
  2432	        self,
  2433	        callee: CallableType,

  2435	        actual_kinds: list[ArgKind],
  2436	        actual_names: Sequence[str | None] | None,

  2438	        context: Context,

  2440	        """Check for extra actual arguments.
  2441
  2442	        Return tuple (was everything ok,

  2444	        """
  2445
  2446	        is_unexpected_arg_error = False  # Keep track of errors to avoid duplicate errors
  2447	        ok = True  # False if we've found any error
  2448
  2449	        for i, kind in enumerate(actual_kinds):
  2450	            if (
  2451	                i not in all_actuals

  2455	                (kind != nodes.ARG_STAR or is_non_empty_tuple(actual_types[i]))

  2458	                # dictionaries and we can't tell it from their types
  2459	                kind != nodes.ARG_STAR2

  2469	                    act_type = actual_types[i]

  2471	                    is_unexpected_arg_error = True
