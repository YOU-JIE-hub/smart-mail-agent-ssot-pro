   840	    def handle_breakBefore(self, flowables):
   841	        '''preprocessing step to allow pageBreakBefore and frameBreakBefore attributes'''
   842	        first = flowables[0]
   843	        # if we insert a page break before, we'll process that, see it again,
   844	        # and go in an infinite loop.  So we need to set a flag on the object
   845	        # saying 'skip me'.  This should be unset on the next pass
   846	        if hasattr(first, '_skipMeNextTime'):
   847	            delattr(first, '_skipMeNextTime')
   848	            return
   849	        # this could all be made much quicker by putting the attributes
   850	        # in to the flowables with a defult value of 0
   851	        if hasattr(first,'pageBreakBefore') and first.pageBreakBefore == 1:
   852	            first._skipMeNextTime = 1
   853	            first.insert(0, PageBreak())
   854	            return
   855	        if hasattr(first,'style') and hasattr(first.style, 'pageBreakBefore') and first.style.pageBreakBefore == 1:
   856	            first._skipMeNextTime = 1
   857	            flowables.insert(0, PageBreak())
   858	            return
   859	        if hasattr(first,'frameBreakBefore') and first.frameBreakBefore == 1:
   860	            first._skipMeNextTime = 1
   861	            flowables.insert(0, FrameBreak())
   862	            return
   863	        if hasattr(first,'style') and hasattr(first.style, 'frameBreakBefore') and first.style.frameBreakBefore == 1:
   864	            first._skipMeNextTime = 1
   865	            flowables.insert(0, FrameBreak())
   866	            return

   869	        "implements keepWithNext"

   897	    def handle_flowable(self,flowables):
   898	        '''try to handle one flowable from the front of list flowables.'''
   899
   900	        #allow document a chance to look at, modify or ignore
   901	        #the object(s) about to be processed
   902	        self.filterFlowables(flowables)
   903
   904	        f = flowables[0]
   905	        if f:
   906	            self.handle_breakBefore(flowables)
   907	            self.handle_keepWithNext(flowables)
   908	            f = flowables[0]
   909	        del flowables[0]
   910	        if f is None:
   911	            return
   912
   913	        if isinstance(f,PageBreak):
   914	            npt = f.nextTemplate
   915	            if npt and not self._samePT(npt):
   916	                npt=NextPageTemplate(npt)
   917	                npt.apply(self)
   918	                self.afterFlowable(npt)
   919	            if isinstance(f,SlowPageBreak):
   920	                self.handle_pageBreak(slow=1)
   921	            else:
   922	                self.handle_pageBreak()
   923	            self.afterFlowable(f)
   924	        elif isinstance(f,ActionFlowable):
   925	            f.apply(self)
   926	            self.afterFlowable(f)
   927	        else:
   928	            frame = self.frame
   929	            canv = self.canv
   930	            #try to fit it then draw it
   931	            if frame.add(f, canv, trySplit=self.allowSplitting):
   932	                if not isinstance(f,FrameActionFlowable):
   933	                    self._curPageFlowableCount += 1
   934	                    self.afterFlowable(f)
   935	                _addGeneratedContent(flowables,frame)
   936	            else:
   937	                if self.allowSplitting:
   938	                    # see if this is a splittable thing
   939	                    S = frame.split(f,canv)
   940	                    n = len(S)
   941	                else:
   942	                    n = 0
   943	                if n:
   944	                    if not isinstance(S[0],(PageBreak,SlowPageBreak,ActionFlowable,DDIndenter)):
   945	                        if not frame.add(S[0], canv, trySplit=0):
   946	                            ident = "Splitting error(n==%d) on page %d in\n%s\nS[0]=%s" % (n,self.page,self._fIdent(f,60,frame),self._fIdent(S[0],60,frame))
   947	                            #leave to keep apart from the raise
   948	                            raise LayoutError(ident)
   949	                        self._curPageFlowableCount += 1
   950	                        self.afterFlowable(S[0])
   951	                        flowables[0:0] = S[1:]  # put rest of splitted flowables back on the list
   952	                        _addGeneratedContent(flowables,frame)
   953	                    else:
   954	                        flowables[0:0] = S  # put split flowables back on the list
   955	                else:
   956	                    if hasattr(f,'_postponed'):
   957	                        ident = "Flowable %s%s too large on page %d in frame %r%s of template %r" % \
   958	                                (self._fIdent(f,60,frame),_fSizeString(f),self.page, self.frame.id,
   959	                                        self.frame._aSpaceString(), self.pageTemplate.id)
   960	                        #leave to keep apart from the raise
   961	                        raise LayoutError(ident)
   962	                    # this ought to be cleared when they are finally drawn!
   963	                    f._postponed = 1
   964	                    mbe = getattr(self,'_multiBuildEdits',None)
   965	                    if mbe:
   966	                        mbe((delattr,f,'_postponed'))
   967	                    flowables.insert(0,f)           # put the flowable back
   968	                    self.handle_frameEnd()

  1046	    def build(self, flowables, filename=None, canvasmaker=canvas.Canvas):
  1047	        """Build the document from a list of flowables.
  1048	           If the filename argument is provided then that filename is used
  1049	           rather than the one provided upon initialization.
  1050	           If the canvasmaker argument is provided then it will be used
  1051	           instead of the default.  For example a slideshow might use
  1052	           an alternate canvas which places 6 slides on a page (by
  1053	           doing translations, scalings and redefining the page break
  1054	           operations).
  1055	        """
  1056	        #assert filter(lambda x: not isinstance(x,Flowable), flowables)==[], "flowables argument error"
  1057	        flowableCount = len(flowables)
  1058	        if self._onProgress:
  1059	            self._onProgress('STARTED',0)
  1060	            self._onProgress('SIZE_EST', len(flowables))
  1061	        self._startBuild(filename,canvasmaker)
  1062
  1063	        #pagecatcher can drag in information from embedded PDFs and we want ours
  1064	        #to take priority, so cache and reapply our own info dictionary after the build.
  1065	        canv = self.canv
  1066	        self._savedInfo = canv._doc.info
  1067	        handled = 0
  1068
  1069	        try:
  1070	            canv._doctemplate = self
  1071	            while len(flowables):
  1072	                if self._hanging and self._hanging[-1] is PageBegin and isinstance(flowables[0],PageBreakIfNotEmpty):
  1073	                    npt = flowables[0].nextTemplate
  1074	                    if npt and not self._samePT(npt):
  1075	                        npt=NextPageTemplate(npt)
  1076	                        npt.apply(self)
  1077	                        self._setPageTemplate()
  1078	                    del flowables[0]
  1079	                self.clean_hanging()
  1080	                try:
  1081	                    first = flowables[0]
  1082	                    self.handle_flowable(flowables)
  1083	                    handled += 1
  1084	                except:
  1085	                    #if it has trace info, add it to the traceback message.
  1086	                    if hasattr(first, '_traceInfo') and first._traceInfo:
  1087	                        exc = sys.exc_info()[1]
  1088	                        args = list(exc.args)
  1089	                        tr = first._traceInfo
  1090	                        args[0] += '\n(srcFile %s, line %d char %d to line %d char %d)' % (
  1091	                            tr.srcFile,
  1092	                            tr.startLineNo,
  1093	                            tr.startLinePos,
  1094	                            tr.endLineNo,
  1095	                            tr.endLinePos
  1096	                            )
  1097	                        exc.args = tuple(args)
  1098	                    raise
  1099	                if self._onProgress:
  1100	                    self._onProgress('PROGRESS',flowableCount - len(flowables))
  1101	        finally:
  1102	            del canv._doctemplate
  1103
  1104
  1105	        #reapply pagecatcher info
  1106	        canv._doc.info = self._savedInfo
  1107
  1108	        self._endBuild()
  1109	        if self._onProgress:
  1110	            self._onProgress('FINISHED',0)

  1139	    def multiBuild(self, story,
  1140	                   maxPasses = 10,
  1141	                   **buildKwds
  1142	                   ):
  1143	        """Makes multiple passes until all indexing flowables
  1144	        are happy.
  1145
  1146	        Returns number of passes"""
  1147	        self._indexingFlowables = []
  1148	        #scan the story and keep a copy
  1149	        for thing in story:
  1150	            if thing.isIndexing():
  1151	                self._indexingFlowables.append(thing)
  1152
  1153	        #better fix for filename is a 'file' problem
  1154	        self._doSave = 0
  1155	        passes = 0
  1156	        mbe = []
  1157	        self._multiBuildEdits = mbe.append
  1158	        while 1:
  1159	            passes += 1
  1160	            if self._onProgress:
  1161	                self._onProgress('PASS', passes)
  1162	            if verbose: sys.stdout.write('building pass '+str(passes) + '...')
  1163
  1164	            for fl in self._indexingFlowables:
  1165	                fl.beforeBuild()
  1166
  1167	            # work with a copy of the story, since it is consumed
  1168	            tempStory = story[:]
  1169	            self.build(tempStory, **buildKwds)
  1170	            #self.notify('debug',None)
  1171
  1172	            for fl in self._indexingFlowables:
  1173	                fl.afterBuild()
  1174
  1175	            happy = self._allSatisfied()
  1176
  1177	            if happy:
  1178	                self._doSave = 0
  1179	                self.canv.save()
  1180	                break
  1181	            if passes > maxPasses:
  1182	                raise IndexError("Index entries not resolved after %d passes" % maxPasses)
  1183
  1184	            #work through any edits
  1185	            while mbe:
  1186	                e = mbe.pop(0)
  1187	                e[0](*e[1:])
  1188
  1189	        del self._multiBuildEdits
  1190	        if verbose: print('saved')
  1191	        return passes
