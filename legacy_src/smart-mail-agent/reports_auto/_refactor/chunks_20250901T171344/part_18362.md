  1188	    def try_infer_partial_value_type_from_call(
  1189	        self, e: CallExpr, methodname: str, var: Var
  1190	    ) -> Instance | None:
  1191	        """Try to make partial type precise from a call such as 'x.append(y)'."""
  1192	        if self.chk.current_node_deferred:
  1193	            return None
  1194	        partial_type = get_partial_instance_type(var.type)
  1195	        if partial_type is None:
  1196	            return None
  1197	        if partial_type.value_type:
  1198	            typename = partial_type.value_type.type.fullname
  1199	        else:
  1200	            assert partial_type.type is not None
  1201	            typename = partial_type.type.fullname
  1202	        # Sometimes we can infer a full type for a partial List, Dict or Set type.
  1203	        # TODO: Don't infer argument expression twice.
  1204	        if (
  1205	            typename in self.item_args
  1206	            and methodname in self.item_args[typename]
  1207	            and e.arg_kinds == [ARG_POS]
  1208	        ):
  1209	            item_type = self.accept(e.args[0])
  1210	            if mypy.checker.is_valid_inferred_type(item_type, self.chk.options):
  1211	                return self.chk.named_generic_type(typename, [item_type])
  1212	        elif (
  1213	            typename in self.container_args
  1214	            and methodname in self.container_args[typename]
  1215	            and e.arg_kinds == [ARG_POS]
  1216	        ):
  1217	            arg_type = get_proper_type(self.accept(e.args[0]))
  1218	            if isinstance(arg_type, Instance):
  1219	                arg_typename = arg_type.type.fullname
  1220	                if arg_typename in self.container_args[typename][methodname]:
  1221	                    if all(
  1222	                        mypy.checker.is_valid_inferred_type(item_type, self.chk.options)
  1223	                        for item_type in arg_type.args
  1224	                    ):
  1225	                        return self.chk.named_generic_type(typename, list(arg_type.args))
  1226	            elif isinstance(arg_type, AnyType):
  1227	                return self.chk.named_type(typename)
  1228
  1229	        return None

  1231	    def apply_function_plugin(
  1232	        self,
  1233	        callee: CallableType,
  1234	        arg_kinds: list[ArgKind],
  1235	        arg_types: list[Type],
  1236	        arg_names: Sequence[str | None] | None,
  1237	        formal_to_actual: list[list[int]],
  1238	        args: list[Expression],
  1239	        fullname: str,
  1240	        object_type: Type | None,
  1241	        context: Context,
  1242	    ) -> Type:
  1243	        """Use special case logic to infer the return type of a specific named function/method.
  1244
  1245	        Caller must ensure that a plugin hook exists. There are two different cases:
  1246
  1247	        - If object_type is None, the caller must ensure that a function hook exists
  1248	          for fullname.
  1249	        - If object_type is not None, the caller must ensure that a method hook exists
  1250	          for fullname.
  1251
  1252	        Return the inferred return type.
  1253	        """
  1254	        num_formals = len(callee.arg_types)
  1255	        formal_arg_types: list[list[Type]] = [[] for _ in range(num_formals)]
  1256	        formal_arg_exprs: list[list[Expression]] = [[] for _ in range(num_formals)]
  1257	        formal_arg_names: list[list[str | None]] = [[] for _ in range(num_formals)]
  1258	        formal_arg_kinds: list[list[ArgKind]] = [[] for _ in range(num_formals)]
  1259	        for formal, actuals in enumerate(formal_to_actual):
  1260	            for actual in actuals:
  1261	                formal_arg_types[formal].append(arg_types[actual])
  1262	                formal_arg_exprs[formal].append(args[actual])
  1263	                if arg_names:
  1264	                    formal_arg_names[formal].append(arg_names[actual])
  1265	                else:
  1266	                    formal_arg_names[formal].append(None)
  1267	                formal_arg_kinds[formal].append(arg_kinds[actual])
  1268
  1269	        if object_type is None:
  1270	            # Apply function plugin
  1271	            callback = self.plugin.get_function_hook(fullname)
  1272	            assert callback is not None  # Assume that caller ensures this
  1273	            return callback(
  1274	                FunctionContext(
  1275	                    arg_types=formal_arg_types,
  1276	                    arg_kinds=formal_arg_kinds,
  1277	                    callee_arg_names=callee.arg_names,
  1278	                    arg_names=formal_arg_names,
  1279	                    default_return_type=callee.ret_type,
  1280	                    args=formal_arg_exprs,
  1281	                    context=context,
  1282	                    api=self.chk,
  1283	                )
  1284	            )
  1285	        else:
  1286	            # Apply method plugin
  1287	            method_callback = self.plugin.get_method_hook(fullname)
  1288	            assert method_callback is not None  # Assume that caller ensures this
  1289	            object_type = get_proper_type(object_type)
  1290	            return method_callback(
  1291	                MethodContext(
  1292	                    type=object_type,
  1293	                    arg_types=formal_arg_types,
  1294	                    arg_kinds=formal_arg_kinds,
  1295	                    callee_arg_names=callee.arg_names,
  1296	                    arg_names=formal_arg_names,
  1297	                    default_return_type=callee.ret_type,
  1298	                    args=formal_arg_exprs,
  1299	                    context=context,
  1300	                    api=self.chk,
  1301	                )
  1302	            )

  1438	    def handle_decorator_overload_call(
  1439	        self, callee_type: CallableType, overloaded: Overloaded, ctx: Context
  1440	    ) -> tuple[Type, Type] | None:

  1442

  1444	        overload. This function should be only used if callee_type takes and returns a Callable.
  1445	        """
  1446	        result = []
  1447	        inferred_args = []
  1448	        for item in overloaded.items:
  1449	            arg = TempNode(typ=item)

  1451	                item_result, inferred_arg = self.check_call(callee_type, [arg], [ARG_POS], ctx)
  1452	            if err.has_new_errors():
  1453	                # This overload doesn't match.
  1454	                continue
  1455	            p_item_result = get_proper_type(item_result)
  1456	            if not isinstance(p_item_result, CallableType):
  1457	                continue
  1458	            p_inferred_arg = get_proper_type(inferred_arg)
  1459	            if not isinstance(p_inferred_arg, CallableType):
  1460	                continue
  1461	            inferred_args.append(p_inferred_arg)
  1462	            result.append(p_item_result)

  1464	            # None of the overload matched (or overload was initially malformed).
  1465	            return None
  1466	        return Overloaded(result), Overloaded(inferred_args)
