  1405	class Instance(ProperType):
  1406	    """An instance type of form C[T1, ..., Tn].
  1407
  1408	    The list of type variables may be empty.
  1409
  1410	    Several types have fallbacks to `Instance`, because in Python everything is an object
  1411	    and this concept is impossible to express without intersection types. We therefore use
  1412	    fallbacks for all "non-special" (like UninhabitedType, ErasedType etc) types.
  1413	    """
  1414
  1415	    __slots__ = ("type", "args", "invalid", "type_ref", "last_known_value", "_hash", "extra_attrs")
  1416
  1417	    def __init__(
  1418	        self,
  1419	        typ: mypy.nodes.TypeInfo,
  1420	        args: Sequence[Type],
  1421	        line: int = -1,
  1422	        column: int = -1,
  1423	        *,
  1424	        last_known_value: LiteralType | None = None,
  1425	        extra_attrs: ExtraAttrs | None = None,
  1426	    ) -> None:
  1427	        super().__init__(line, column)
  1428	        self.type = typ
  1429	        self.args = tuple(args)
  1430	        self.type_ref: str | None = None
  1431
  1432	        # True if recovered after incorrect number of type arguments error
  1433	        self.invalid = False
  1434
  1435	        # This field keeps track of the underlying Literal[...] value associated with
  1436	        # this instance, if one is known.
  1437	        #
  1438	        # This field is set whenever possible within expressions, but is erased upon
  1439	        # variable assignment (see erasetype.remove_instance_last_known_values) unless
  1440	        # the variable is declared to be final.
  1441	        #
  1442	        # For example, consider the following program:
  1443	        #
  1444	        #     a = 1
  1445	        #     b: Final[int] = 2
  1446	        #     c: Final = 3
  1447	        #     print(a + b + c + 4)
  1448	        #
  1449	        # The 'Instance' objects associated with the expressions '1', '2', '3', and '4' will
  1450	        # have last_known_values of type Literal[1], Literal[2], Literal[3], and Literal[4]
  1451	        # respectively. However, the Instance object assigned to 'a' and 'b' will have their
  1452	        # last_known_value erased: variable 'a' is mutable; variable 'b' was declared to be
  1453	        # specifically an int.
  1454	        #
  1455	        # Or more broadly, this field lets this Instance "remember" its original declaration
  1456	        # when applicable. We want this behavior because we want implicit Final declarations
  1457	        # to act pretty much identically with constants: we should be able to replace any
  1458	        # places where we use some Final variable with the original value and get the same
  1459	        # type-checking behavior. For example, we want this program:
  1460	        #
  1461	        #    def expects_literal(x: Literal[3]) -> None: pass
  1462	        #    var: Final = 3
  1463	        #    expects_literal(var)
  1464	        #
  1465	        # ...to type-check in the exact same way as if we had written the program like this:
  1466	        #
  1467	        #    def expects_literal(x: Literal[3]) -> None: pass
  1468	        #    expects_literal(3)
  1469	        #
  1470	        # In order to make this work (especially with literal types), we need var's type
  1471	        # (an Instance) to remember the "original" value.
  1472	        #
  1473	        # Preserving this value within expressions is useful for similar reasons.
  1474	        #
  1475	        # Currently most of mypy will ignore this field and will continue to treat this type like
  1476	        # a regular Instance. We end up using this field only when we are explicitly within a
  1477	        # Literal context.
  1478	        self.last_known_value = last_known_value
  1479
  1480	        # Cached hash value
  1481	        self._hash = -1
  1482
  1483	        # Additional attributes defined per instance of this type. For example modules
  1484	        # have different attributes per instance of types.ModuleType.
  1485	        self.extra_attrs = extra_attrs
  1486
  1487	    def accept(self, visitor: TypeVisitor[T]) -> T:
  1488	        return visitor.visit_instance(self)
  1489
  1490	    def __hash__(self) -> int:
  1491	        if self._hash == -1:
  1492	            self._hash = hash((self.type, self.args, self.last_known_value, self.extra_attrs))
  1493	        return self._hash
  1494
  1495	    def __eq__(self, other: object) -> bool:
  1496	        if not isinstance(other, Instance):
  1497	            return NotImplemented
  1498	        return (
  1499	            self.type == other.type
  1500	            and self.args == other.args
  1501	            and self.last_known_value == other.last_known_value
  1502	            and self.extra_attrs == other.extra_attrs
  1503	        )
  1504
  1505	    def serialize(self) -> JsonDict | str:
  1506	        assert self.type is not None
  1507	        type_ref = self.type.fullname
  1508	        if not self.args and not self.last_known_value:
  1509	            return type_ref
  1510	        data: JsonDict = {
  1511	            ".class": "Instance",
  1512	            "type_ref": type_ref,
  1513	            "args": [arg.serialize() for arg in self.args],
  1514	        }
  1515	        if self.last_known_value is not None:
  1516	            data["last_known_value"] = self.last_known_value.serialize()
  1517	        data["extra_attrs"] = self.extra_attrs.serialize() if self.extra_attrs else None
  1518	        return data
  1519
  1520	    @classmethod
  1521	    def deserialize(cls, data: JsonDict | str) -> Instance:
  1522	        if isinstance(data, str):
  1523	            inst = Instance(NOT_READY, [])
  1524	            inst.type_ref = data
  1525	            return inst
  1526	        assert data[".class"] == "Instance"
  1527	        args: list[Type] = []
  1528	        if "args" in data:
  1529	            args_list = data["args"]
  1530	            assert isinstance(args_list, list)
  1531	            args = [deserialize_type(arg) for arg in args_list]
  1532	        inst = Instance(NOT_READY, args)
  1533	        inst.type_ref = data["type_ref"]  # Will be fixed up by fixup.py later.
  1534	        if "last_known_value" in data:
  1535	            inst.last_known_value = LiteralType.deserialize(data["last_known_value"])
  1536	        if data.get("extra_attrs") is not None:
  1537	            inst.extra_attrs = ExtraAttrs.deserialize(data["extra_attrs"])
  1538	        return inst
  1539
  1540	    def copy_modified(
  1541	        self,
  1542	        *,
  1543	        args: Bogus[list[Type]] = _dummy,
  1544	        last_known_value: Bogus[LiteralType | None] = _dummy,
  1545	    ) -> Instance:
  1546	        new = Instance(
  1547	            typ=self.type,
  1548	            args=args if args is not _dummy else self.args,
  1549	            line=self.line,
  1550	            column=self.column,
  1551	            last_known_value=(
  1552	                last_known_value if last_known_value is not _dummy else self.last_known_value
  1553	            ),
  1554	            extra_attrs=self.extra_attrs,
  1555	        )
  1556	        # We intentionally don't copy the extra_attrs here, so they will be erased.
  1557	        new.can_be_true = self.can_be_true
  1558	        new.can_be_false = self.can_be_false
  1559	        return new
  1560
  1561	    def copy_with_extra_attr(self, name: str, typ: Type) -> Instance:
  1562	        if self.extra_attrs:
  1563	            existing_attrs = self.extra_attrs.copy()
  1564	        else:
  1565	            existing_attrs = ExtraAttrs({}, set(), None)
  1566	        existing_attrs.attrs[name] = typ
  1567	        new = self.copy_modified()
  1568	        new.extra_attrs = existing_attrs
  1569	        return new

  1626

  1629	    so that it behaves contravariantly, in particular [x: int] <: [int].
  1630	    """
  1631
  1632	    __slots__ = (
  1633	        "arg_types",
  1634	        "arg_kinds",
  1635	        "arg_names",
  1636	        "min_args",
  1637	        "is_ellipsis_args",
  1638	        # TODO: variables don't really belong here, but they are used to allow hacky support
  1639	        # for forall . Foo[[x: T], T] by capturing generic callable with ParamSpec, see #15909
  1640	        "variables",
  1641	        "imprecise_arg_kinds",
  1642	    )
  1643
  1644	    def __init__(
  1645	        self,
  1646	        arg_types: Sequence[Type],
  1647	        arg_kinds: list[ArgKind],
  1648	        arg_names: Sequence[str | None],
  1649	        *,
  1650	        variables: Sequence[TypeVarLikeType] | None = None,
  1651	        is_ellipsis_args: bool = False,
  1652	        imprecise_arg_kinds: bool = False,

  1654	        column: int = -1,

  1656	        super().__init__(line, column)
  1657	        self.arg_types = list(arg_types)
  1658	        self.arg_kinds = arg_kinds
  1659	        self.arg_names = list(arg_names)
  1660	        assert len(arg_types) == len(arg_kinds) == len(arg_names)
  1661	        assert not any(isinstance(t, Parameters) for t in arg_types)
  1662	        self.min_args = arg_kinds.count(ARG_POS)
  1663	        self.is_ellipsis_args = is_ellipsis_args
  1664	        self.variables = variables or []
  1665	        self.imprecise_arg_kinds = imprecise_arg_kinds

  1668	        self,
  1669	        arg_types: Bogus[Sequence[Type]] = _dummy,
  1670	        arg_kinds: Bogus[list[ArgKind]] = _dummy,
  1671	        arg_names: Bogus[Sequence[str | None]] = _dummy,

  1687	                if imprecise_arg_kinds is not _dummy

  1693	    def var_arg(self) -> FormalArgument | None:
  1694	        """The formal argument for *args."""
  1695	        for position, (type, kind) in enumerate(zip(self.arg_types, self.arg_kinds)):
  1696	            if kind == ARG_STAR:

  1700	    def kw_arg(self) -> FormalArgument | None:
  1701	        """The formal argument for **kwargs."""
  1702	        for position, (type, kind) in enumerate(zip(self.arg_types, self.arg_kinds)):
  1703	            if kind == ARG_STAR2:
  1704	                return FormalArgument(None, position, type, False)

  1707	    def formal_arguments(self, include_star_args: bool = False) -> list[FormalArgument]:
  1708	        """Yields the formal arguments corresponding to this callable, ignoring *arg and **kwargs.
  1709
  1710	        To handle *args and **kwargs, use the 'callable.var_args' and 'callable.kw_args' fields,
  1711	        if they are not None.
  1712
  1713	        If you really want to include star args in the yielded output, set the
  1714	        'include_star_args' parameter to 'True'."""
  1715	        args = []
  1716	        done_with_positional = False
  1717	        for i in range(len(self.arg_types)):
  1718	            kind = self.arg_kinds[i]
  1719	            if kind.is_named() or kind.is_star():
  1720	                done_with_positional = True
  1721	            if not include_star_args and kind.is_star():
  1722	                continue
  1723
  1724	            required = kind.is_required()
  1725	            pos = None if done_with_positional else i
  1726	            arg = FormalArgument(self.arg_names[i], pos, self.arg_types[i], required)
  1727	            args.append(arg)
  1728	        return args

  1730	    def argument_by_name(self, name: str | None) -> FormalArgument | None:
  1731	        if name is None:
  1732	            return None
  1733	        seen_star = False
  1734	        for i, (arg_name, kind, typ) in enumerate(
  1735	            zip(self.arg_names, self.arg_kinds, self.arg_types)
  1736	        ):
  1737	            # No more positional arguments after these.
  1738	            if kind.is_named() or kind.is_star():
  1739	                seen_star = True
  1740	            if kind.is_star():
  1741	                continue
  1742	            if arg_name == name:
  1743	                position = None if seen_star else i
  1744	                return FormalArgument(name, position, typ, kind.is_required())
  1745	        return self.try_synthesizing_arg_from_kwarg(name)

  1747	    def argument_by_position(self, position: int | None) -> FormalArgument | None:

  1749	            return None
  1750	        if position >= len(self.arg_names):
  1751	            return self.try_synthesizing_arg_from_vararg(position)
  1752	        name, kind, typ = (
  1753	            self.arg_names[position],
  1754	            self.arg_kinds[position],
  1755	            self.arg_types[position],
  1756	        )
  1757	        if kind.is_positional():
  1758	            return FormalArgument(name, position, typ, kind == ARG_POS)
  1759	        else:
  1760	            return self.try_synthesizing_arg_from_vararg(position)
