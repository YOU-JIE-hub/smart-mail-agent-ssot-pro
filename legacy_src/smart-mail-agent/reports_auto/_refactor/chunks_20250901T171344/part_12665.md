  3435	                if base_type:

  3446	                        # base classes are also incompatible

  3450	                            lvalue_node.name, base, lvalue, setter_type=True

  3453	                        # the getter type is ready.

  3458	                            )
  3459	                            return

  3574	        self, node: Var, base: TypeInfo, base_node: Node | None
  3575	    ) -> bool:
  3576	        """Check if an assignment overrides a final attribute in a base class.
  3577
  3578	        This only checks situations where either a node in base class is not a variable
  3579	        but a final method, or where override is explicitly declared as final.
  3580	        In these cases we give a more detailed error message. In addition, we check that
  3581	        a final variable doesn't override writeable attribute, which is not safe.

  3591	            # if we are overriding a final method with variable.

  3593	            # in `check_final()`.
  3594	            self.msg.cant_override_final(node.name, base.name, node)

  3690	    def check_assignment_to_slots(self, lvalue: Lvalue) -> None:
  3691	        if not isinstance(lvalue, MemberExpr):
  3692	            return
  3693
  3694	        inst = get_proper_type(self.expr_checker.accept(lvalue.expr))
  3695	        if not isinstance(inst, Instance):
  3696	            return
  3697	        if inst.type.slots is None:
  3698	            return  # Slots do not exist, we can allow any assignment
  3699	        if lvalue.name in inst.type.slots:
  3700	            return  # We are assigning to an existing slot
  3701	        for base_info in inst.type.mro[:-1]:
  3702	            if base_info.names.get("__setattr__") is not None:
  3703	                # When type has `__setattr__` defined,
  3704	                # we can assign any dynamic value.
  3705	                # We exclude object, because it always has `__setattr__`.
  3706	                return
  3707
  3708	        definition = inst.type.get(lvalue.name)
  3709	        if definition is None:
  3710	            # We don't want to duplicate
  3711	            # `"SomeType" has no attribute "some_attr"`
  3712	            # error twice.
  3713	            return
  3714	        if self.is_assignable_slot(lvalue, definition.type):
  3715	            return

  3721	    def is_assignable_slot(self, lvalue: Lvalue, typ: Type | None) -> bool:
  3722	        if getattr(lvalue, "node", None):
  3723	            return False  # This is a definition
  3724
  3725	        typ = get_proper_type(typ)
  3726	        if typ is None or isinstance(typ, AnyType):
  3727	            return True  # Any can be literally anything, like `@property`
  3728	        if isinstance(typ, Instance):
  3729	            # When working with instances, we need to know if they contain
  3730	            # `__set__` special method. Like `@property` does.
  3731	            # This makes assigning to properties possible,
  3732	            # even without extra slot spec.
  3733	            return typ.type.get("__set__") is not None
  3734	        if isinstance(typ, FunctionLike):
  3735	            return True  # Can be a property, or some other magic
  3736	        if isinstance(typ, UnionType):
  3737	            return all(self.is_assignable_slot(lvalue, u) for u in typ.items)
  3738	        return False

  3740	    def flatten_rvalues(self, rvalues: list[Expression]) -> list[Expression]:
  3741	        """Flatten expression list by expanding those * items that have tuple type.
  3742
  3743	        For each regular type item in the tuple type use a TempNode(), for an Unpack
  3744	        item use a corresponding StarExpr(TempNode()).

  3746	        new_rvalues = []
  3747	        for rv in rvalues:
  3748	            if not isinstance(rv, StarExpr):
  3749	                new_rvalues.append(rv)
  3750	                continue

  3752	            if not isinstance(typ, TupleType):
  3753	                new_rvalues.append(rv)

  3764	                            isinstance(unpacked, Instance)
