   550	def send_jspi_request(
   551	    request: EmscriptenRequest, streaming: bool
   552	) -> EmscriptenResponse:
   553	    """
   554	    Send a request using WebAssembly JavaScript Promise Integration
   555	    to wrap the asynchronous JavaScript fetch api (experimental).
   556
   557	    :param request:
   558	        Request to send
   559
   560	    :param streaming:
   561	        Whether to stream the response
   562
   563	    :return: The response object
   564	    :rtype: EmscriptenResponse
   565	    """
   566	    timeout = request.timeout
   567	    js_abort_controller = js.AbortController.new()
   568	    headers = {k: v for k, v in request.headers.items() if k not in HEADERS_TO_IGNORE}
   569	    req_body = request.body
   570	    fetch_data = {
   571	        "headers": headers,
   572	        "body": to_js(req_body),
   573	        "method": request.method,
   574	        "signal": js_abort_controller.signal,
   575	    }

   577	    # so urllib3 can set `redirect: manual` to control redirects itself.
   578	    # https://stackoverflow.com/a/78524615
   579	    if _is_node_js():
   580	        fetch_data["redirect"] = "manual"
   581	    # Call JavaScript fetch (async api, returns a promise)
   582	    fetcher_promise_js = js.fetch(request.url, _obj_from_dict(fetch_data))
   583	    # Now suspend WebAssembly until we resolve that promise
   584	    # or time out.
   585	    response_js = _run_sync_with_timeout(
   586	        fetcher_promise_js,
   587	        timeout,
   588	        js_abort_controller,
   589	        request=request,
   590	        response=None,
   591	    )
   592	    headers = {}
   593	    header_iter = response_js.headers.entries()
   594	    while True:
   595	        iter_value_js = header_iter.next()
   596	        if getattr(iter_value_js, "done", False):
   597	            break
   598	        else:
   599	            headers[str(iter_value_js.value[0])] = str(iter_value_js.value[1])
   600	    status_code = response_js.status
   601	    body: bytes | io.RawIOBase = b""
   602
   603	    response = EmscriptenResponse(
   604	        status_code=status_code, headers=headers, body=b"", request=request
   605	    )
   606	    if streaming:
   607	        # get via inputstream
   608	        if response_js.body is not None:
   609	            # get a reader from the fetch response
   610	            body_stream_js = response_js.body.getReader()
   611	            body = _JSPIReadStream(
   612	                body_stream_js, timeout, request, response, js_abort_controller
   613	            )
   614	    else:
   615	        # get directly via arraybuffer
   616	        # n.b. this is another async JavaScript call.
   617	        body = _run_sync_with_timeout(
   618	            response_js.arrayBuffer(),
   619	            timeout,
   620	            js_abort_controller,
   621	            request=request,
   622	            response=response,
   623	        ).to_py()
   624	    response.body = body
   625	    return response

   628	def _run_sync_with_timeout(
   629	    promise: Any,
   630	    timeout: float,
   631	    js_abort_controller: Any,
   632	    request: EmscriptenRequest | None,
   633	    response: EmscriptenResponse | None,
   634	) -> Any:
   635	    """
   636	    Await a JavaScript promise synchronously with a timeout which is implemented
   637	    via the AbortController
   638
   639	    :param promise:
   640	        Javascript promise to await
   641
   642	    :param timeout:
   643	        Timeout in seconds
   644
   645	    :param js_abort_controller:
   646	        A JavaScript AbortController object, used on timeout
   647
   648	    :param request:
   649	        The request being handled
   650
   651	    :param response:
   652	        The response being handled (if it exists yet)
   653
   654	    :raises _TimeoutError: If the request times out
   655	    :raises _RequestError: If the request raises a JavaScript exception
   656
   657	    :return: The result of awaiting the promise.
   658	    """
   659	    timer_id = None
   660	    if timeout > 0:
   661	        timer_id = js.setTimeout(
   662	            js_abort_controller.abort.bind(js_abort_controller), int(timeout * 1000)
   663	        )
   664	    try:
   665	        from pyodide.ffi import run_sync
   666
   667	        # run_sync here uses WebAssembly JavaScript Promise Integration to
   668	        # suspend python until the JavaScript promise resolves.
   669	        return run_sync(promise)
   670	    except JsException as err:
   671	        if err.name == "AbortError":
   672	            raise _TimeoutError(
   673	                message="Request timed out", request=request, response=response
   674	            )
   675	        else:
   676	            raise _RequestError(message=err.message, request=request, response=response)
   677	    finally:
   678	        if timer_id is not None:
   679	            js.clearTimeout(timer_id)
