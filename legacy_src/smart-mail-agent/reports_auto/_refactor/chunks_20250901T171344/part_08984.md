ndex=None):
   585	        """Obtain a distribution suitable for fulfilling `requirement`
   586
   587	        `requirement` must be a ``pkg_resources.Requirement`` instance.
   588	        If necessary, or if the `force_scan` flag is set, the requirement is
   589	        searched for in the (online) package index as well as the locally
   590	        installed packages.  If a distribution matching `requirement` is found,
   591	        the returned distribution's ``location`` is the value you would have
   592	        gotten from calling the ``download()`` method with the matching
   593	        distribution's URL or filename.  If no matching distribution is found,
   594	        ``None`` is returned.
   595
   596	        If the `source` flag is set, only source distributions and source
   597	        checkout links will be considered.  Unless the `develop_ok` flag is
   598	        set, development and system eggs (i.e., those using the ``.egg-info``
   599	        format) will be ignored.
   600	        """
   601	        # process a Requirement
   602	        self.info("Searching for %s", requirement)
   603	        skipped = {}
   604	        dist = None
   605
   606	        def find(req, env=None):
   607	            if env is None:
   608	                env = self
   609	            # Find a matching distribution; may be called more than once
   610
   611	            for dist in env[req.key]:
   612
   613	                if dist.precedence == DEVELOP_DIST and not develop_ok:
   614	                    if dist not in skipped:
   615	                        self.warn(
   616	                            "Skipping development or system egg: %s", dist,
   617	                        )
   618	                        skipped[dist] = 1
   619	                    continue
   620
   621	                test = (
   622	                    dist in req
   623	                    and (dist.precedence <= SOURCE_DIST or not source)
   624	                )
   625	                if test:
   626	                    loc = self.download(dist.location, tmpdir)
   627	                    dist.download_location = loc
   628	                    if os.path.exists(dist.download_location):
   629	                        return dist
   630
   631	        if force_scan:
   632	            self.prescan()
   633	            self.find_packages(requirement)
   634	            dist = find(requirement)
   635
   636	        if not dist and local_index is not None:
   637	            dist = find(requirement, local_index)
   638
   639	        if dist is None:
   640	            if self.to_scan is not None:
   641	                self.prescan()
   642	            dist = find(requirement)
   643
   644	        if dist is None and not force_scan:
   645	            self.find_packages(requirement)
   646	            dist = find(requirement)
   647
   648	        if dist is None:
   649	            self.warn(
   650	                "No local packages or working download links found for %s%s",
   651	                (source and "a source distribution of " or ""),
   652	                requirement,
   653	            )
   654	        else:
   655	            self.info("Best match: %s", dist)
   656	            return dist.clone(location=dist.download_location)

   714	    def _download_to(self, url, filename):
   715	        self.info("Downloading %s", url)
   716	        # Download the file
   717	        fp = None
   718	        try:
   719	            checker = HashChecker.from_url(url)
   720	            fp = self.open_url(url)
   721	            if isinstance(fp, urllib.error.HTTPError):
   722	                raise DistutilsError(
   723	                    "Can't download %s: %s %s" % (url, fp.code, fp.msg)
   724	                )
   725	            headers = fp.info()
   726	            blocknum = 0
   727	            bs = self.dl_blocksize
   728	            size = -1
   729	            if "content-length" in headers:
   730	                # Some servers return multiple Content-Length headers :(
   731	                sizes = headers.get_all('Content-Length')
   732	                size = max(map(int, sizes))
   733	                self.reporthook(url, filename, blocknum, bs, size)
   734	            with open(filename, 'wb') as tfp:
   735	                while True:
   736	                    block = fp.read(bs)
   737	                    if block:
   738	                        checker.feed(block)
   739	                        tfp.write(block)
   740	                        blocknum += 1
   741	                        self.reporthook(url, filename, blocknum, bs, size)
   742	                    else:
   743	                        break
   744	                self.check_hash(checker, filename, tfp)
   745	            return headers
   746	        finally:
   747	            if fp:
   748	                fp.close()

   832	    @staticmethod
   833	    def _resolve_vcs(url):
   834	        """
   835	        >>> rvcs = PackageIndex._resolve_vcs
   836	        >>> rvcs('git+http://foo/bar')
   837	        'git'
   838	        >>> rvcs('hg+https://foo/bar')
   839	        'hg'
   840	        >>> rvcs('git:myhost')
   841	        'git'
   842	        >>> rvcs('hg:myhost')
   843	        >>> rvcs('http://foo/bar')
   844	        """
   845	        scheme = urllib.parse.urlsplit(url).scheme
   846	        pre, sep, post = scheme.partition('+')

   848	        allowed = set(['svn', 'git'] + ['hg'] * bool(sep))
   849	        return next(iter({pre} & allowed), None)

   851	    def _download_vcs(self, url, spec_filename):
   852	        vcs = self._resolve_vcs(url)
   853	        if not vcs:
   854	            return
   855	        if vcs == 'svn':
   856	            return self._download_svn(url, spec_filename)
   857
   858	        filename, _, _ = spec_filename.partition('#')
   859	        url, rev = self._vcs_split_rev_from_url(url)

   861	        self.info(f"Doing {vcs} clone from {url} to {filename}")
   862	        subprocess.check_call([vcs, 'clone', '--quiet', url, filename])
   863
   864	        co_commands = dict(
   865	            git=[vcs, '-C', filename, 'checkout', '--quiet', rev],
   866	            hg=[vcs, '--cwd', filename, 'up', '-C', '-r', rev, '-q'],
   867	        )
   868	        if rev is not None:
   869	            self.info(f"Checking out {rev}")
   870	            subprocess.check_call(co_commands[vcs])

   872	        return filename

   932	    def _vcs_split_rev_from_url(url):
   933	        """
   934	        Given a possible VCS URL, return a clean URL and resolved revision if any.
   935
   936	        >>> vsrfu = PackageIndex._vcs_split_rev_from_url
   937	        >>> vsrfu('git+https://github.com/pypa/setuptools@v69.0.0#egg-info=setuptools')

   939	        >>> vsrfu('git+https://github.com/pypa/setuptools#egg-info=setuptools')
   940	        ('https://github.com/pypa/setuptools', None)
   941	        >>> vsrfu('http://foo/bar')
   942	        ('http://foo/bar', None)
   943	        """
   944	        parts = urllib.parse.urlsplit(url)
   945
   946	        clean_scheme = parts.scheme.split('+', 1)[-1]
   947
   948	        # Some fragment identification fails
   949	        no_fragment_path, _, _ = parts.path.partition('#')
   950
   951	        pre, sep, post = no_fragment_path.rpartition('@')
   952	        clean_path, rev = (pre, post) if sep else (post, None)
   953
   954	        resolved = parts._replace(
   955	            scheme=clean_scheme,
   956	            path=clean_path,
   957	            # discard the fragment
   958	            fragment='',
   959	        ).geturl()
   960
   961	        return resolved, rev

  1010	def _encode_auth(auth):
  1011	    """
  1012	    Encode auth from a URL suitable for an HTTP header.
  1013	    >>> str(_encode_auth('username%3Apassword'))
  1014	    'dXNlcm5hbWU6cGFzc3dvcmQ='
  1015
  1016	    Long auth strings should not cause a newline to be inserted.
  1017	    >>> long_auth = 'username:' + 'password'*10

  1019	    False
  1020	    """
  1021	    auth_s = urllib.parse.unquote(auth)
  1022	    # convert to bytes
  1023	    auth_bytes = auth_s.encode()
  1024	    encoded_bytes = base64.b64encode(auth_bytes)
  1025	    # convert back to a string
  1026	    encoded = encoded_bytes.decode()
  1027	    # strip the trailing carriage return
  1028	    return encoded.replace('\n', '')
