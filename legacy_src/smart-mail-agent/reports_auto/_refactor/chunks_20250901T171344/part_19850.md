def _escape_pdf_text(s: str) -> str:
    # 先把非 ASCII 轉成 \uXXXX / \xNN 形式（皆為 ASCII 字元）
    ascii_safe = s.encode("ascii", "backslashreplace").decode("ascii")

    return "".join(ch for ch in ascii_safe if 32 <= ord(ch) <= 126)
