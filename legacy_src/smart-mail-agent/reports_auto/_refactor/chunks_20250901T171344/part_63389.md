))
   324	        if info.typeddict_type:
   325	            self.add_type_dependencies(info.typeddict_type, target=make_trigger(target))
   326	        if info.declared_metaclass:
   327	            self.add_type_dependencies(info.declared_metaclass, target=make_trigger(target))
   328	        if info.is_protocol:
   329	            for base_info in info.mro[:-1]:
   330	                # We add dependencies from whole MRO to cover explicit subprotocols.
   331	                # For example:
   332	                #
   333	                #     class Super(Protocol):
   334	                #         x: int
   335	                #     class Sub(Super, Protocol):
   336	                #         y: int
   337	                #
   338	                # In this example we add <Super[wildcard]> -> <Sub>, to invalidate Sub if
   339	                # a new member is added to Super.
   340	                self.add_dependency(
   341	                    make_wildcard_trigger(base_info.fullname), target=make_trigger(target)
   342	                )
   343	                # More protocol dependencies are collected in type_state._snapshot_protocol_deps
   344	                # after a full run or update is finished.
   345
   346	        self.add_type_alias_deps(self.scope.current_target())
   347	        for name, node in info.names.items():
   348	            if isinstance(node.node, Var):
   349	                # Recheck Liskov if needed, self definitions are checked in the defining method
   350	                if node.node.is_initialized_in_class and has_user_bases(info):
   351	                    self.add_dependency(make_trigger(info.fullname + "." + name))
   352	                for base_info in non_trivial_bases(info):
   353	                    # If the type of an attribute changes in a base class, we make references
   354	                    # to the attribute in the subclass stale.
   355	                    self.add_dependency(
   356	                        make_trigger(base_info.fullname + "." + name),
   357	                        target=make_trigger(info.fullname + "." + name),
   358	                    )
   359	        for base_info in non_trivial_bases(info):
   360	            for name, node in base_info.names.items():
   361	                if self.use_logical_deps():
   362	                    # Skip logical dependency if an attribute is not overridden. For example,
   363	                    # in case of:
   364	                    #     class Base:
   365	                    #         x = 1
   366	                    #         y = 2
   367	                    #     class Sub(Base):
   368	                    #         x = 3
   369	                    # we skip <Base.y> -> <Child.y>, because even if `y` is unannotated it
   370	                    # doesn't affect precision of Liskov checking.
   371	                    if name not in info.names:
   372	                        continue
   373	                    # __init__ and __new__ can be overridden with different signatures, so no
   374	                    # logical dependency.
   375	                    if name in ("__init__", "__new__"):
   376	                        continue
   377	                self.add_dependency(
   378	                    make_trigger(base_info.fullname + "." + name),
   379	                    target=make_trigger(info.fullname + "." + name),
   380	                )
   381	            if not self.use_logical_deps():
   382	                # These dependencies are only useful for propagating changes --
   383	                # they aren't logical dependencies since __init__ and __new__ can be
   384	                # overridden with a different signature.
   385	                self.add_dependency(
   386	                    make_trigger(base_info.fullname + ".__init__"),
   387	                    target=make_trigger(info.fullname + ".__init__"),
   388	                )
   389	                self.add_dependency(
   390	                    make_trigger(base_info.fullname + ".__new__"),
   391	                    target=make_trigger(info.fullname + ".__new__"),
   392	                )
   393	                # If the set of abstract attributes change, this may invalidate class
   394	                # instantiation, or change the generated error message, since Python checks
   395	                # class abstract status when creating an instance.
   396	                self.add_dependency(
   397	                    make_trigger(base_info.fullname + ".(abstract)"),
   398	                    target=make_trigger(info.fullname + ".__init__"),
   399	                )
   400	                # If the base class abstract attributes change, subclass abstract
   401	                # attributes need to be recalculated.
   402	                self.add_dependency(make_trigger(base_info.fullname + ".(abstract)"))
   403

   431	    def visit_assignment_stmt(self, o: AssignmentStmt) -> None:
   432	        rvalue = o.rvalue
   433	        if isinstance(rvalue, CallExpr) and isinstance(rvalue.analyzed, TypeVarExpr):
   434	            analyzed = rvalue.analyzed
   435	            self.add_type_dependencies(
   436	                analyzed.upper_bound, target=make_trigger(analyzed.fullname)
   437	            )
   438	            for val in analyzed.values:
   439	                self.add_type_dependencies(val, target=make_trigger(analyzed.fullname))
   440	            # We need to re-analyze the definition if bound or value is deleted.
   441	            super().visit_call_expr(rvalue)
   442	        elif isinstance(rvalue, CallExpr) and isinstance(rvalue.analyzed, NamedTupleExpr):
   443	            # Depend on types of named tuple items.
   444	            info = rvalue.analyzed.info
   445	            prefix = f"{self.scope.current_full_target()}.{info.name}"
   446	            for name, symnode in info.names.items():
   447	                if not name.startswith("_") and isinstance(symnode.node, Var):
   448	                    typ = symnode.node.type
   449	                    if typ:
   450	                        self.add_type_dependencies(typ)
   451	                        self.add_type_dependencies(typ, target=make_trigger(prefix))
   452	                        attr_target = make_trigger(f"{prefix}.{name}")
   453	                        self.add_type_dependencies(typ, target=attr_target)
   454	        elif isinstance(rvalue, CallExpr) and isinstance(rvalue.analyzed, TypedDictExpr):
   455	            # Depend on the underlying typeddict type
   456	            info = rvalue.analyzed.info
   457	            assert info.typeddict_type is not None
   458	            prefix = f"{self.scope.current_full_target()}.{info.name}"
   459	            self.add_type_dependencies(info.typeddict_type, target=make_trigger(prefix))
   460	        elif isinstance(rvalue, CallExpr) and isinstance(rvalue.analyzed, EnumCallExpr):
   461	            # Enum values are currently not checked, but for future we add the deps on them
   462	            for name, symnode in rvalue.analyzed.info.names.items():
   463	                if isinstance(symnode.node, Var) and symnode.node.type:
   464	                    self.add_type_dependencies(symnode.node.type)
   465	        elif o.is_alias_def:
   466	            assert len(o.lvalues) == 1
   467	            lvalue = o.lvalues[0]
   468	            assert isinstance(lvalue, NameExpr)
   469	            typ = get_proper_type(self.type_map.get(lvalue))
   470	            if isinstance(typ, FunctionLike) and typ.is_type_obj():
   471	                class_name = typ.type_object().fullname
   472	                self.add_dependency(make_trigger(class_name + ".__init__"))
   473	                self.add_dependency(make_trigger(class_name + ".__new__"))
   474	            if isinstance(rvalue, IndexExpr) and isinstance(rvalue.analyzed, TypeAliasExpr):
   475	                self.add_type_dependencies(rvalue.analyzed.node.target)
   476	            elif typ:
   477	                self.add_type_dependencies(typ)
   478	        else:
   479	            # Normal assignment
   480	            super().visit_assignment_stmt(o)
   481	            for lvalue in o.lvalues:
   482	                self.process_lvalue(lvalue)
   483	            items = o.lvalues + [rvalue]
   484	            for i in range(len(items) - 1):
   485	                lvalue = items[i]
   486	                rvalue = items[i + 1]
   487	                if isinstance(lvalue, TupleExpr):
   488	                    self.add_attribute_dependency_for_expr(rvalue, "__iter__")
   489	            if o.type:
   490	                self.add_type_dependencies(o.type)
   491	        if self.use_logical_deps() and o.unanalyzed_type is None:
   492	            # Special case: for definitions without an explicit type like this:
   493	            #     x = func(...)
   494	            # we add a logical dependency <func> -> <x>, because if `func` is not annotated,
   495	            # then it will make all points of use of `x` unchecked.
   496	            if (
   497	                isinstance(rvalue, CallExpr)
   498	                and isinstance(rvalue.callee, RefExpr)
   499	                and rvalue.callee.fullname
   500	            ):
   501	                fname: str | None = None
   502	                if isinstance(rvalue.callee.node, TypeInfo):
   503	                    # use actual __init__ as a dependency source
   504	                    init = rvalue.callee.node.get("__init__")
   505	                    if init and isinstance(init.node, SYMBOL_FUNCBASE_TYPES):
   506	                        fname = init.node.fullname
   507	                else:
   508	                    fname = rvalue.callee.fullname
   509	                if not fname:
   510	                    return
   511	                for lv in o.lvalues:
   512	                    if isinstance(lv, RefExpr) and lv.fullname and lv.is_new_def:
   513	                        if lv.kind == LDEF:
   514	                            return  # local definitions don't generate logical deps
   515	                        self.add_dependency(make_trigger(fname), make_trigger(lv.fullname))
   516
   517	    def process_lvalue(self, lvalue: Expression) -> None:
   518	        """Generate additional dependencies for an lvalue."""
   519	        if isinstance(lvalue, IndexExpr):
   520	            self.add_operator_method_dependency(lvalue.base, "__setitem__")
   521	        elif isinstance(lvalue, NameExpr):
   522	            if lvalue.kind in (MDEF, GDEF):
   523	                # Assignment to an attribute in the class body, or direct assignment to a
   524	                # global variable.
   525	                lvalue_type = self.get_non_partial_lvalue_type(lvalue)
   526	                type_triggers = self.get_type_triggers(lvalue_type)
   527	                attr_trigger = make_trigger(f"{self.scope.current_full_target()}.{lvalue.name}")
   528	                for type_trigger in type_triggers:
   529	                    self.add_dependency(type_trigger, attr_trigger)
   530	        elif isinstance(lvalue, MemberExpr):
   531	            if self.is_self_member_ref(lvalue) and lvalue.is_new_def:
   532	                node = lvalue.node
   533	                if isinstance(node, Var):
   534	                    info = node.info
   535	                    if info and has_user_bases(info):
   536	                        # Recheck Liskov for self definitions
   537	                        self.add_dependency(make_trigger(info.fullname + "." + lvalue.name))
   538	            if lvalue.kind is None:
   539	                # Reference to a non-module attribute
   540	                if lvalue.expr not in self.type_map:
   541	                    # Unreachable assignment -> not checked so no dependencies to generate.
   542	                    return
   543	                object_type = self.type_map[lvalue.expr]
   544	                lvalue_type = self.get_non_partial_lvalue_type(lvalue)
   545	                type_triggers = self.get_type_triggers(lvalue_type)
   546	                for attr_trigger in self.attribute_triggers(object_type, lvalue.name):
   547	                    for type_trigger in type_triggers:
   548	                        self.add_dependency(type_trigger, attr_trigger)
   549	        elif isinstance(lvalue, TupleExpr):
   550	            for item in lvalue.items:
   551	                self.process_lvalue(item)
   552	        elif isinstance(lvalue, StarExpr):
   553	            self.process_lvalue(lvalue.expr)
   554
   555	    def is_self_member_ref(self, memberexpr: MemberExpr) -> bool:
   556	        """Does memberexpr to refer to an attribute of self?"""
   557	        if not isinstance(memberexpr.expr, NameExpr):
   558	            return False
   559	        node = memberexpr.expr.node
   560	        return isinstance(node, Var) and node.is_self
   561
   562	    def get_non_partial_lvalue_type(self, lvalue: RefExpr) -> Type:
   563	        if lvalue not in self.type_map:
   564	            # Likely a block considered unreachable during type checking.
   565	            return UninhabitedType()
   566	        lvalue_type = get_proper_type(self.type_map[lvalue])
   567	        if isinstance(lvalue_type, PartialType):
   568	            if isinstance(lvalue.node, Var):
   569	                if lvalue.node.type:
   570	                    lvalue_type = get_proper_type(lvalue.node.type)
   571	                else:
   572	                    lvalue_type = UninhabitedType()
   573	            else:
   574	                # Probably a secondary, non-definition assignment that doesn't
   575	                # result in a non-partial type. We won't be able to infer any
   576	                # dependencies from this so just return something. (The first,
   577	                # definition assignment with a partial type is handled
   578	                # differently, in the semantic analyzer.)
   579	                assert not lvalue.is_new_def
   580	                return UninhabitedType()
   581	        return lvalue_type

   592	    def visit_for_stmt(self, o: ForStmt) -> None:
   593	        super().visit_for_stmt(o)
   594	        if not o.is_async:
   595	            # __getitem__ is only used if __iter__ is missing but for simplicity we
   596	            # just always depend on both.
   597	            self.add_attribute_dependency_for_expr(o.expr, "__iter__")
   598	            self.add_attribute_dependency_for_expr(o.expr, "__getitem__")
   599	            if o.inferred_iterator_type:
   600	                self.add_attribute_dependency(o.inferred_iterator_type, "__next__")
   601	        else:
   602	            self.add_attribute_dependency_for_expr(o.expr, "__aiter__")
   603	            if o.inferred_iterator_type:
   604	                self.add_attribute_dependency(o.inferred_iterator_type, "__anext__")
   605
   606	        self.process_lvalue(o.index)
   607	        if isinstance(o.index, TupleExpr):
   608	            # Process multiple assignment to index variables.
   609	            item_type = o.inferred_item_type
   610	            if item_type:
   611	                # This is similar to above.
   612	                self.add_attribute_dependency(item_type, "__iter__")
   613	                self.add_attribute_dependency(item_type, "__getitem__")
   614	        if o.index_type:
   615	            self.add_type_dependencies(o.index_type)

   636	    def process_global_ref_expr(self, o: RefExpr) -> None:
   637	        if o.fullname:
   638	            self.add_dependency(make_trigger(o.fullname))
   639
   640	        # If this is a reference to a type, generate a dependency to its
   641	        # constructor.

   643	        #       class attribute references, etc., if performance is a problem.
   644	        typ = get_proper_type(self.type_map.get(o))
   645	        if isinstance(typ, FunctionLike) and typ.is_type_obj():
   646	            class_name = typ.type_object().fullname
   647	            self.add_dependency(make_trigger(class_name + ".__init__"))
   648	            self.add_dependency(make_trigger(class_name + ".__new__"))

   650	    def visit_name_expr(self, o: NameExpr) -> None:
   651	        if o.kind == LDEF:
   652	            # We don't track dependencies to local variables, since they
   653	            # aren't externally visible.
   654	            return
   655	        if o.kind == MDEF:
   656	            # Direct reference to member is only possible in the scope that
   657	            # defined the name, so no dependency is required.

   661	    def visit_member_expr(self, e: MemberExpr) -> None:
   662	        if isinstance(e.expr, RefExpr) and isinstance(e.expr.node, TypeInfo):
   663	            # Special case class attribute so that we don't depend on "__init__".
   664	            self.add_dependency(make_trigger(e.expr.node.fullname))
   665	        else:
   666	            super().visit_member_expr(e)

   668	            # Reference to a module attribute
   669	            self.process_global_ref_expr(e)
   670	        else:
   671	            # Reference to a non-module (or missing) attribute
   672	            if e.expr not in self.type_map:
   673	                # No type available -- this happens for unreachable code. Since it's unreachable,
   674	                # it wasn't type checked an
