   210	    def calcPositions(self):
   211	        """Works out where they go.
   212
   213	        Sets an attribute _positions which is a list of
   214	        lists of (x, y) matching the data.
   215	        """
   216
   217	        self._seriesCount = len(self.data)
   218	        self._rowLength = max(list(map(len,self.data)))
   219
   220	        if self.useAbsolute:
   221	            # Dimensions are absolute.
   222	            normFactor = 1.0
   223	        else:
   224	            # Dimensions are normalized to fit.
   225	            normWidth = self.groupSpacing
   226	            availWidth = self.categoryAxis.scale(0)[1]
   227	            normFactor = availWidth / normWidth
   228	        self._normFactor = normFactor
   229	        self._yzero = yzero = self.valueAxis.scale(0)
   230	        self._hngs = hngs = 0.5 * self.groupSpacing * normFactor
   231
   232	        pairs = set()
   233	        P = [].append
   234	        cscale = self.categoryAxis.scale
   235	        vscale = self.valueAxis.scale
   236	        data = self.data
   237	        n = len(data)
   238	        for rowNo,row in enumerate(data):
   239	            if isinstance(row, FillPairedData):
   240	                other = row.other
   241	                if 0<=other<n:
   242	                    if other==rowNo:
   243	                        raise ValueError('data row %r may not be paired with itself' % rowNo)
   244	                    t = (rowNo,other)
   245	                    pairs.add((min(t),max(t)))
   246	                else:
   247	                    raise ValueError('data row %r is paired with invalid data row %r' % (rowNo, other))
   248	            line = [].append
   249	            for colNo,datum in enumerate(row):
   250	                if datum is not None:
   251	                    groupX, groupWidth = cscale(colNo)
   252	                    x = groupX + hngs
   253	                    y = yzero
   254	                    height = vscale(datum) - y
   255	                    line((x, y+height))
   256	            P(line.__self__)
   257	        P = P.__self__
   258
   259	        #if there are some paired lines we ensure only one is created
   260	        for rowNo, other in pairs:
   261	            P[rowNo] = FillPairedData(P[rowNo],other)
   262	        self._pairInFills = len(pairs)
   263	        self._positions = P

   265	    def _innerDrawLabel(self, rowNo, colNo, x, y):
   266	        "Draw a label for a given item in the list."
   267
   268	        labelFmt = self.lineLabelFormat
   269	        labelValue = self.data[rowNo][colNo]
   270
   271	        if labelFmt is None:
   272	            labelText = None
   273	        elif type(labelFmt) is str:
   274	            if labelFmt == 'values':
   275	                try:
   276	                    labelText = self.lineLabelArray[rowNo][colNo]
   277	                except:
   278	                    labelText = None
   279	            else:
   280	                labelText = labelFmt % labelValue
   281	        elif hasattr(labelFmt,'__call__'):
   282	            labelText = labelFmt(labelValue)
   283	        else:
   284	            raise ValueError("Unknown formatter type %s, expected string or function"%labelFmt)
   285
   286	        if labelText:
   287	            label = self.lineLabels[(rowNo, colNo)]
   288	            if not label.visible: return
   289	            # Make sure labels are some distance off the data point.
   290	            if y > 0:
   291	                label.setOrigin(x, y + self.lineLabelNudge)
   292	            else:
   293	                label.setOrigin(x, y - self.lineLabelNudge)
   294	            label.setText(labelText)
   295	        else:
   296	            label = None
   297	        return label

   304	    def makeLines(self):
   305	        g = Group()
   306
   307	        labelFmt = self.lineLabelFormat
   308	        P = self._positions
   309	        if self.reversePlotOrder: P.reverse()
   310	        lines = self.lines
   311	        styleCount = len(lines)
   312	        _inFill = self.inFill
   313	        if (_inFill or self._pairInFills or
   314	                [rowNo for rowNo in range(len(P))
   315	                        if getattr(lines[rowNo%styleCount],'inFill',False)]
   316	                ):
   317	            inFillY = self.categoryAxis._y
   318	            inFillX0 = self.valueAxis._x
   319	            inFillX1 = inFillX0 + self.categoryAxis._length
   320	            inFillG = getattr(self,'_inFillG',g)
   321	        yzero = self._yzero
   322
   323	        # Iterate over data rows.
   324	        for rowNo, row in enumerate(reversed(P) if self.reversePlotOrder else P):
   325	            styleIdx = rowNo % styleCount
   326	            rowStyle = lines[styleIdx]
   327	            strokeColor = rowStyle.strokeColor
   328	            fillColor = getattr(rowStyle,'fillColor',strokeColor)
   329	            inFill = getattr(rowStyle,'inFill',_inFill)
   330	            dash = getattr(rowStyle, 'strokeDashArray', None)
   331	            lineStyle = getattr(rowStyle,'lineStyle',None)
   332
   333	            if hasattr(rowStyle, 'strokeWidth'):
   334	                strokeWidth = rowStyle.strokeWidth
   335	            elif hasattr(lines, 'strokeWidth'):
   336	                strokeWidth = lines.strokeWidth
   337	            else:
   338	                strokeWidth = None
   339
   340	            # Iterate over data columns.
   341	            if lineStyle=='bar':
   342	                barWidth = getattr(rowStyle,'barWidth',Percentage(50))
   343	                if isinstance(barWidth,Percentage):
   344	                    hbw = self._hngs*barWidth*0.01
   345	                else:
   346	                    hbw = barWidth*0.5
   347	                for x, y in row:
   348	                    g.add(Rect(x-hbw,min(y,yzero),2*hbw,abs(y-yzero),strokeWidth=strokeWidth,strokeColor=strokeColor,fillColor=fillColor))
   349	            elif self.joinedLines or lineStyle=='joinedLine':
   350	                points = flatten(row)
   351	                if inFill or isinstance(row,FillPairedData):
   352	                    filler = getattr(rowStyle, 'filler', None)
   353	                    if isinstance(row,FillPairedData):
   354	                        fpoints = points + flatten(reversed(P[row.other]))
   355	                    else:
   356	                        fpoints = [inFillX0,inFillY] + points + [inFillX1,inFillY]
   357	                    if filler:
   358	                        filler.fill(self,inFillG,rowNo,fillColor,fpoints)
   359	                    else:
   360	                        inFillG.add(Polygon(fpoints,fillColor=fillColor,strokeColor=strokeColor if strokeColor==fillColor else None,strokeWidth=strokeWidth or 0.1))
   361	                if not inFill or inFill==2 or strokeColor!=fillColor:
   362	                    line = PolyLine(points,strokeColor=strokeColor,strokeLineCap=0,strokeLineJoin=1)
   363	                    if strokeWidth:
   364	                        line.strokeWidth = strokeWidth
   365	                    if dash:
   366	                        line.strokeDashArray = dash
   367	                    g.add(line)
   368
   369	            if hasattr(rowStyle, 'symbol'):
   370	                uSymbol = rowStyle.symbol
   371	            elif hasattr(lines, 'symbol'):
   372	                uSymbol = lines.symbol
   373	            else:
   374	                uSymbol = None
   375
   376	            if uSymbol:
   377	                for colNo,(x,y) in enumerate(row):
   378	                    symbol = uSymbol2Symbol(tpcGetItem(uSymbol,colNo),x,y,rowStyle.strokeColor)
   379	                    if symbol: g.add(symbol)
   380
   381	            # Draw item labels.
   382	            for colNo, (x, y) in enumerate(row):
   383	                self.drawLabel(g, rowNo, colNo, x, y)
   384
   385	        return g

   393	        if cA: cA.joinAxis = vA

   402	            y = xAxisCrossesAt

   407	        self.calcPositions()

   411	        if self.inFill:
   412	            self._inFillG = Group()
   413	            g.add(self._inFillG)
   414
   415	        g.add(cA)
   416	        g.add(vA)
   417	        cAdgl = getattr(cA,'drawGridLast',False)
   418	        vAdgl = getattr(vA,'drawGridLast',False)

   420	        if not vAdgl: vA.makeGrid(g,parent=self,dim=cA.getGridDims)

   423	        if vAdgl: vA.makeGrid(g,parent=self,dim=cA.getGridDims)
   424	        for a in getattr(self,'annotations',()): g.add(a(self,cA.scale,vA.scale))
