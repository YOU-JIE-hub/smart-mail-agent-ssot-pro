  1534	    def analyze_callable_type(self, t: UnboundType) -> Type:
  1535	        fallback = self.named_type("builtins.function")
  1536	        if len(t.args) == 0:
  1537	            # Callable (bare). Treat as Callable[..., Any].
  1538	            any_type = self.get_omitted_any(t)
  1539	            ret = callable_with_ellipsis(any_type, any_type, fallback)
  1540	        elif len(t.args) == 2:
  1541	            callable_args = t.args[0]
  1542	            ret_type = t.args[1]
  1543	            if isinstance(callable_args, TypeList):
  1544	                # Callable[[ARG, ...], RET] (ordinary callable type)
  1545	                analyzed_args = self.analyze_callable_args(callable_args)
  1546	                if analyzed_args is None:
  1547	                    return AnyType(TypeOfAny.from_error)
  1548	                args, kinds, names = analyzed_args
  1549	                ret = CallableType(args, kinds, names, ret_type=ret_type, fallback=fallback)
  1550	            elif isinstance(callable_args, EllipsisType):
  1551	                # Callable[..., RET] (with literal ellipsis; accept arbitrary arguments)
  1552	                ret = callable_with_ellipsis(
  1553	                    AnyType(TypeOfAny.explicit), ret_type=ret_type, fallback=fallback
  1554	                )
  1555	            else:
  1556	                # Callable[P, RET] (where P is ParamSpec)
  1557	                with self.tvar_scope_frame(namespace=""):
  1558	                    # Temporarily bind ParamSpecs to allow code like this:
  1559	                    #     my_fun: Callable[Q, Foo[Q]]
  1560	                    # We usually do this later in visit_callable_type(), but the analysis
  1561	                    # below happens at very early stage.
  1562	                    variables = []
  1563	                    for name, tvar_expr in self.find_type_var_likes(callable_args):
  1564	                        variables.append(self.tvar_scope.bind_new(name, tvar_expr))
  1565	                    maybe_ret = self.analyze_callable_args_for_paramspec(
  1566	                        callable_args, ret_type, fallback
  1567	                    ) or self.analyze_callable_args_for_concatenate(
  1568	                        callable_args, ret_type, fallback
  1569	                    )
  1570	                    if isinstance(maybe_ret, CallableType):
  1571	                        maybe_ret = maybe_ret.copy_modified(variables=variables)
  1572	                if maybe_ret is None:
  1573	                    # Callable[?, RET] (where ? is something invalid)
  1574	                    self.fail(
  1575	                        "The first argument to Callable must be a "
  1576	                        'list of types, parameter specification, or "..."',
  1577	                        t,
  1578	                        code=codes.VALID_TYPE,
  1579	                    )
  1580	                    self.note(
  1581	                        "See https://mypy.readthedocs.io/en/stable/kinds_of_types.html#callable-types-and-lambdas",
  1582	                        t,
  1583	                    )
  1584	                    return AnyType(TypeOfAny.from_error)
  1585	                elif isinstance(maybe_ret, AnyType):
  1586	                    return maybe_ret
  1587	                ret = maybe_ret
  1588	        else:
  1589	            if self.options.disallow_any_generics:
  1590	                self.fail('Please use "Callable[[<parameters>], <return type>]"', t)
  1591	            else:
  1592	                self.fail('Please use "Callable[[<parameters>], <return type>]" or "Callable"', t)
  1593	            return AnyType(TypeOfAny.from_error)
  1594	        assert isinstance(ret, CallableType)
  1595	        return ret.accept(self)

  1604	    def analyze_callable_args(
  1605	        self, arglist: TypeList
  1606	    ) -> tuple[list[Type], list[ArgKind], list[str | None]] | None:
  1607	        args: list[Type] = []
  1608	        kinds: list[ArgKind] = []
  1609	        names: list[str | None] = []
  1610	        seen_unpack = False
  1611	        unpack_types: list[Type] = []
  1612	        invalid_unpacks: list[Type] = []
  1613	        second_unpack_last = False
  1614	        for i, arg in enumerate(arglist.items):
  1615	            if isinstance(arg, CallableArgument):
  1616	                args.append(arg.typ)
  1617	                names.append(arg.name)
  1618	                if arg.constructor is None:
  1619	                    return None
  1620	                found = self.lookup_qualified(arg.constructor, arg)
  1621	                if found is None:
  1622	                    # Looking it up already put an error message in
  1623	                    return None
  1624	                elif found.fullname not in ARG_KINDS_BY_CONSTRUCTOR:
  1625	                    self.fail(f'Invalid argument constructor "{found.fullname}"', arg)
  1626	                    return None
  1627	                else:
  1628	                    assert found.fullname is not None
  1629	                    kind = ARG_KINDS_BY_CONSTRUCTOR[found.fullname]
  1630	                    kinds.append(kind)
  1631	                    if arg.name is not None and kind.is_star():
  1632	                        self.fail(f"{arg.constructor} arguments should not have names", arg)
  1633	                        return None
  1634	            elif (
  1635	                isinstance(arg, UnboundType)
  1636	                and self.refers_to_full_names(arg, UNPACK_TYPE_NAMES)
  1637	                or isinstance(arg, UnpackType)
  1638	            ):
  1639	                if seen_unpack:
  1640	                    # Multiple unpacks, preserve them, so we can give an error later.
  1641	                    if i == len(arglist.items) - 1 and not invalid_unpacks:
  1642	                        # Special case: if there are just two unpacks, and the second one appears
  1643	                        # as last type argument, it can be still valid, if the second unpacked type
  1644	                        # is a TypedDict. This should be checked by the caller.
  1645	                        second_unpack_last = True
  1646	                    invalid_unpacks.append(arg)
  1647	                    continue
  1648	                seen_unpack = True
  1649	                unpack_types.append(arg)
  1650	            else:
  1651	                if seen_unpack:
  1652	                    unpack_types.append(arg)
  1653	                else:
  1654	                    args.append(arg)
  1655	                    kinds.append(ARG_POS)
  1656	                    names.append(None)
  1657	        if seen_unpack:
  1658	            if len(unpack_types) == 1:
  1659	                args.append(unpack_types[0])
  1660	            else:
  1661	                first = unpack_types[0]
  1662	                if isinstance(first, UnpackType):
  1663	                    # UnpackType doesn't have its own line/column numbers,
  1664	                    # so use the unpacked type for error messages.
  1665	                    first = first.type
  1666	                args.append(
  1667	                    UnpackType(self.tuple_type(unpack_types, line=first.line, column=first.column))
  1668	                )
  1669	            kinds.append(ARG_STAR)
  1670	            names.append(None)
  1671	        for arg in invalid_unpacks:
  1672	            args.append(arg)
  1673	            kinds.append(ARG_STAR2 if second_unpack_last else ARG_STAR)
  1674	            names.append(None)
  1675	        # Note that arglist below is only used for error context.
  1676	        check_arg_names(names, [arglist] * len(args), self.fail, "Callable")
  1677	        check_arg_kinds(kinds, [arglist] * len(args), self.fail)
  1678	        return args, kinds, names
