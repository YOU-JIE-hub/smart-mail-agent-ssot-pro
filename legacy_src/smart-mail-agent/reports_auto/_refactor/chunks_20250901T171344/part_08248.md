    59	class CFG:
    60	    """Control-flow graph.
    61
    62	    Node 0 is always assumed to be the entry point. There must be a
    63	    non-empty set of exits.
    64	    """
    65
    66	    def __init__(
    67	        self,
    68	        succ: dict[BasicBlock, list[BasicBlock]],
    69	        pred: dict[BasicBlock, list[BasicBlock]],
    70	        exits: set[BasicBlock],
    71	    ) -> None:
    72	        assert exits
    73	        self.succ = succ
    74	        self.pred = pred
    75	        self.exits = exits
    76
    77	    def __str__(self) -> str:
    78	        exits = sorted(self.exits, key=lambda e: int(e.label))
    79	        return f"exits: {exits}\nsucc: {self.succ}\npred: {self.pred}"
    80

    82	def get_cfg(blocks: list[BasicBlock], *, use_yields: bool = False) -> CFG:
    83	    """Calculate basic block control-flow graph.
    84
    85	    If use_yields is set, then we treat returns inserted by yields as gotos
    86	    instead of exits.
    87	    """
    88	    succ_map = {}
    89	    pred_map: dict[BasicBlock, list[BasicBlock]] = {}
    90	    exits = set()
    91	    for block in blocks:
    92	        assert not any(
    93	            isinstance(op, ControlOp) for op in block.ops[:-1]
    94	        ), "Control-flow ops must be at the end of blocks"
    95
    96	        if use_yields and isinstance(block.terminator, Return) and block.terminator.yield_target:
    97	            succ = [block.terminator.yield_target]
    98	        else:
    99	            succ = list(block.terminator.targets())
   100	        if not succ:
   101	            exits.add(block)
   102
   103	        # Errors can occur anywhere inside a block, which means that
   104	        # we can't assume that the entire block has executed before
   105	        # jumping to the error handler. In our CFG construction, we
   106	        # model this as saying that a block can jump to its error
   107	        # handler or the error handlers of any of its normal
   108	        # successors (to represent an error before that next block
   109	        # completes). This works well for analyses like "must
   110	        # defined", where it implies that registers assigned in a
   111	        # block may be undefined in its error handler, but is in
   112	        # general not a precise representation of reality; any
   113	        # analyses that require more fidelity must wait until after
   114	        # exception insertion.
   115	        for error_point in [block] + succ:
   116	            if error_point.error_handler:
   117	                succ.append(error_point.error_handler)
   118
   119	        succ_map[block] = succ
   120	        pred_map[block] = []
   121	    for prev, nxt in succ_map.items():
   122	        for label in nxt:
   123	            pred_map[label].append(prev)
   124	    return CFG(succ_map, pred_map, exits)
   125

   127	def get_real_target(label: BasicBlock) -> BasicBlock:
   128	    if len(label.ops) == 1 and isinstance(label.ops[-1], Goto):
   129	        label = label.ops[-1].label
   130	    return label

   133	def cleanup_cfg(blocks: list[BasicBlock]) -> None:
   134	    """Cleanup the control flow graph.
   135
   136	    This eliminates obviously dead basic blocks and eliminates blocks that contain
   137	    nothing but a single jump.
   138
   139	    There is a lot more that could be done.
   140	    """
   141	    changed = True
   142	    while changed:
   143	        # First collapse any jumps to basic block that only contain a goto
   144	        for block in blocks:
   145	            for i, tgt in enumerate(block.terminator.targets()):
   146	                block.terminator.set_target(i, get_real_target(tgt))
   147
   148	        # Then delete any blocks that have no predecessors
   149	        changed = False
   150	        cfg = get_cfg(blocks)
   151	        orig_blocks = blocks.copy()
   152	        blocks.clear()
   153	        for i, block in enumerate(orig_blocks):
   154	            if i == 0 or cfg.pred[block]:
   155	                blocks.append(block)
   156	            else:
   157	                changed = True

   285	class DefinedVisitor(BaseAnalysisVisitor[Value]):
   286	    """Visitor for finding defined registers.
   287
   288	    Note that this only deals with registers and not temporaries, on
   289	    the assumption that we never access temporaries when they might be
   290	    undefined.
   291
   292	    If strict_errors is True, then we regard any use of LoadErrorValue
   293	    as making a register undefined. Otherwise we only do if
   294	    `undefines` is set on the error value.
   295
   296	    This lets us only consider the things we care about during
   297	    uninitialized variable checking while capturing all possibly
   298	    undefined things for refcounting.
   299	    """
   300
   301	    def __init__(self, strict_errors: bool = False) -> None:
   302	        self.strict_errors = strict_errors
   303
   304	    def visit_branch(self, op: Branch) -> GenAndKill[Value]:
   305	        return set(), set()
   306
   307	    def visit_return(self, op: Return) -> GenAndKill[Value]:
   308	        return set(), set()
   309
   310	    def visit_unreachable(self, op: Unreachable) -> GenAndKill[Value]:
   311	        return set(), set()
   312
   313	    def visit_register_op(self, op: RegisterOp) -> GenAndKill[Value]:
   314	        return set(), set()
   315
   316	    def visit_assign(self, op: Assign) -> GenAndKill[Value]:
   317	        # Loading an error value may undefine the register.
   318	        if isinstance(op.src, LoadErrorValue) and (op.src.undefines or self.strict_errors):
   319	            return set(), {op.dest}
   320	        else:
   321	            return {op.dest}, set()
   322
   323	    def visit_assign_multi(self, op: AssignMulti) -> GenAndKill[Value]:
   324	        # Array registers are special and we don't track the definedness of them.
   325	        return set(), set()
   326
   327	    def visit_set_mem(self, op: SetMem) -> GenAndKill[Value]:
   328	        return set(), set()

   509	def run_analysis(
   510	    blocks: list[BasicBlock],
   511	    cfg: CFG,
   512	    gen_and_kill: OpVisitor[GenAndKill[T]],
   513	    initial: set[T],
   514	    kind: int,
   515	    backward: bool,
   516	    universe: set[T] | None = None,
   517	) -> AnalysisResult[T]:
   518	    """Run a general set-based data flow analysis.
   519
   520	    Args:
   521	        blocks: All basic blocks
   522	        cfg: Control-flow graph for the code
   523	        gen_and_kill: Implementation of gen and kill functions for each op
   524	        initial: Value of analysis for the entry points (for a forward analysis) or the
   525	            exit points (for a backward analysis)
   526	        kind: MUST_ANALYSIS or MAYBE_ANALYSIS
   527	        backward: If False, the analysis is a forward analysis; it's backward otherwise
   528	        universe: For a must analysis, the set of all possible values. This is the starting
   529	            value for the work list algorithm, which will narrow this down until reaching a
   530	            fixed point. For a maybe analysis the iteration always starts from an empty set
   531	            and this argument is ignored.
   532
   533	    Return analysis results: (before, after)
   534	    """
   535	    block_gen = {}
   536	    block_kill = {}
   537
   538	    # Calculate kill and gen sets for entire basic blocks.
   539	    for block in blocks:
   540	        gen: set[T] = set()
   541	        kill: set[T] = set()
   542	        ops = block.ops
   543	        if backward:
   544	            ops = list(reversed(ops))
   545	        for op in ops:
   546	            opgen, opkill = op.accept(gen_and_kill)
   547	            gen = (gen - opkill) | opgen
   548	            kill = (kill - opgen) | opkill
   549	        block_gen[block] = gen
   550	        block_kill[block] = kill
   551
   552	    # Set up initial state for worklist algorithm.
   553	    worklist = list(blocks)
   554	    if not backward:
   555	        worklist.reverse()  # Reverse for a small performance improvement
   556	    workset = set(worklist)
   557	    before: dict[BasicBlock, set[T]] = {}
   558	    after: dict[BasicBlock, set[T]] = {}
   559	    for block in blocks:
   560	        if kind == MAYBE_ANALYSIS:
   561	            before[block] = set()
   562	            after[block] = set()
   563	        else:
   564	            assert universe is not None, "Universe must be defined for a must analysis"
   565	            before[block] = set(universe)
   566	            after[block] = set(universe)
   567
   568	    if backward:
   569	        pred_map = cfg.succ
   570	        succ_map = cfg.pred
   571	    else:
   572
