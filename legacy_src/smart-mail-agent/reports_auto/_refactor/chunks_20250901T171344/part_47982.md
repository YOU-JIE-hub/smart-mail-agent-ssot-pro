   522	    def build_per_module_cache(self) -> None:
   523	        self._per_module_cache = {}
   524
   525	        # Config precedence is as follows:
   526	        #  1. Concrete section names: foo.bar.baz
   527	        #  2. "Unstructured" glob patterns: foo.*.baz, in the order
   528	        #     they appear in the file (last wins)
   529	        #  3. "Well-structured" wildcard patterns: foo.bar.*, in specificity order.
   530
   531	        # Since structured configs inherit from structured configs above them in the hierarchy,
   532	        # we need to process per-module configs in a careful order.
   533	        # We have to process foo.* before foo.bar.* before foo.bar,
   534	        # and we need to apply *.bar to foo.bar but not to foo.bar.*.
   535	        # To do this, process all well-structured glob configs before non-glob configs and
   536	        # exploit the fact that foo.* sorts earlier ASCIIbetically (unicodebetically?)
   537	        # than foo.bar.*.
   538	        # (A section being "processed last" results in its config "winning".)
   539	        # Unstructured glob configs are stored and are all checked for each module.
   540	        unstructured_glob_keys = [k for k in self.per_module_options.keys() if "*" in k[:-1]]
   541	        structured_keys = [k for k in self.per_module_options.keys() if "*" not in k[:-1]]
   542	        wildcards = sorted(k for k in structured_keys if k.endswith(".*"))
   543	        concrete = [k for k in structured_keys if not k.endswith(".*")]
   544
   545	        for glob in unstructured_glob_keys:
   546	            self._glob_options.append((glob, self.compile_glob(glob)))
   547
   548	        # We (for ease of implementation) treat unstructured glob
   549	        # sections as used if any real modules use them or if any
   550	        # concrete config sections use them. This means we need to
   551	        # track which get used while constructing.
   552	        self.unused_configs = set(unstructured_glob_keys)
   553
   554	        for key in wildcards + concrete:
   555	            # Find what the options for this key would be, just based
   556	            # on inheriting from parent configs.
   557	            options = self.clone_for_module(key)
   558	            # And then update it with its per-module options.
   559	            self._per_module_cache[key] = options.apply_changes(self.per_module_options[key])
   560
   561	        # Add the more structured sections into unused configs, since
   562	        # they only count as used if actually used by a real module.
   563	        self.unused_configs.update(structured_keys)

   565	    def clone_for_module(self, module: str) -> Options:
   566	        """Create an Options object that incorporates per-module options.
   567
   568	        NOTE: Once this method is called all Options objects should be
   569	        considered read-only, else the caching might be incorrect.
   570	        """
   571	        if self._per_module_cache is None:
   572	            self.build_per_module_cache()
   573	        assert self._per_module_cache is not None
   574
   575	        # If the module just directly has a config entry, use it.
   576	        if module in self._per_module_cache:
   577	            self.unused_configs.discard(module)
   578	            return self._per_module_cache[module]
   579
   580	        # If not, search for glob paths at all the parents. So if we are looking for
   581	        # options for foo.bar.baz, we search foo.bar.baz.*, foo.bar.*, foo.*,
   582	        # in that order, looking for an entry.
   583	        # This is technically quadratic in the length of the path, but module paths
   584	        # don't actually get all that long.
   585	        options = self
   586	        path = module.split(".")
   587	        for i in range(len(path), 0, -1):
   588	            key = ".".join(path[:i] + ["*"])
   589	            if key in self._per_module_cache:
   590	                self.unused_configs.discard(key)
   591	                options = self._per_module_cache[key]
   592	                break
   593
   594	        # OK and *now* we need to look for unstructured glob matches.
   595	        # We only do this for concrete modules, not structured wildcards.
   596	        if not module.endswith(".*"):
   597	            for key, pattern in self._glob_options:
   598	                if pattern.match(module):
   599	                    self.unused_configs.discard(key)
   600	                    options = options.apply_changes(self.per_module_options[key])
   601
   602	        # We could update the cache to directly point to modules once
   603	        # they have been looked up, but in testing this made things
   604	        # slower and not faster, so we don't bother.
   605
   606	        return options

   608	    def compile_glob(self, s: str) -> Pattern[str]:
   609	        # Compile one of the glob patterns to a regex so that '.*' can
   610	        # match *zero or more* module sections. This means we compile
   611	        # '.*' into '(\..*)?'.
   612	        parts = s.split(".")
   613	        expr = re.escape(parts[0]) if parts[0] != "*" else ".*"
   614	        for part in parts[1:]:
   615	            expr += re.escape("." + part) if part != "*" else r"(\..*)?"
   616	        return re.compile(expr + "\\Z")
