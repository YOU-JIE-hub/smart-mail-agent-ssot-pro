   231	class IterationDependentErrors:
   232	    """An `IterationDependentErrors` instance serves to collect the `unreachable`,
   233	    `redundant-expr`, and `redundant-casts` errors, as well as the revealed types,
   234	    handled by the individual `IterationErrorWatcher` instances sequentially applied to
   235	    the same code section."""
   236
   237	    # One set of `unreachable`, `redundant-expr`, and `redundant-casts` errors per
   238	    # iteration step.  Meaning of the tuple items: ErrorCode, message, line, column,
   239	    # end_line, end_column.
   240	    uselessness_errors: list[set[tuple[ErrorCode, str, int, int, int, int]]]
   241
   242	    # One set of unreachable line numbers per iteration step.  Not only the lines where
   243	    # the error report occurs but really all unreachable lines.
   244	    unreachable_lines: list[set[int]]
   245
   246	    # One list of revealed types for each `reveal_type` statement.  Each created list
   247	    # can grow during the iteration.  Meaning of the tuple items: line, column,
   248	    # end_line, end_column:
   249	    revealed_types: dict[tuple[int, int, int | None, int | None], list[Type]]
   250
   251	    def __init__(self) -> None:
   252	        self.uselessness_errors = []
   253	        self.unreachable_lines = []
   254	        self.revealed_types = defaultdict(list)
   255
   256	    def yield_uselessness_error_infos(self) -> Iterator[tuple[str, Context, ErrorCode]]:
   257	        """Report only those `unreachable`, `redundant-expr`, and `redundant-casts`
   258	        errors that could not be ruled out in any iteration step."""
   259
   260	        persistent_uselessness_errors = set()
   261	        for candidate in set(chain(*self.uselessness_errors)):
   262	            if all(
   263	                (candidate in errors) or (candidate[2] in lines)
   264	                for errors, lines in zip(self.uselessness_errors, self.unreachable_lines)
   265	            ):
   266	                persistent_uselessness_errors.add(candidate)
   267	        for error_info in persistent_uselessness_errors:
   268	            context = Context(line=error_info[2], column=error_info[3])
   269	            context.end_line = error_info[4]
   270	            context.end_column = error_info[5]
   271	            yield error_info[1], context, error_info[0]
   272
   273	    def yield_revealed_type_infos(self) -> Iterator[tuple[list[Type], Context]]:
   274	        """Yield all types revealed in at least one iteration step."""
   275
   276	        for note_info, types in self.revealed_types.items():
   277	            context = Context(line=note_info[0], column=note_info[1])
   278	            context.end_line = note_info[2]
   279	            context.end_column = note_info[3]
   280	            yield types, context

   283	class IterationErrorWatcher(ErrorWatcher):
   284	    """Error watcher that filters and separately collects `unreachable` errors,
   285	    `redundant-expr` and `redundant-casts` errors, and revealed types when analysing
   286	    code sections iteratively to help avoid making too-hasty reports."""
   287
   288	    iteration_dependent_errors: IterationDependentErrors
   289
   290	    def __init__(
   291	        self,
   292	        errors: Errors,
   293	        iteration_dependent_errors: IterationDependentErrors,
   294	        *,
   295	        filter_errors: bool | Callable[[str, ErrorInfo], bool] = False,
   296	        save_filtered_errors: bool = False,
   297	        filter_deprecated: bool = False,
   298	    ) -> None:
   299	        super().__init__(
   300	            errors,
   301	            filter_errors=filter_errors,
   302	            save_filtered_errors=save_filtered_errors,
   303	            filter_deprecated=filter_deprecated,
   304	        )
   305	        self.iteration_dependent_errors = iteration_dependent_errors
   306	        iteration_dependent_errors.uselessness_errors.append(set())
   307	        iteration_dependent_errors.unreachable_lines.append(set())
   308
   309	    def on_error(self, file: str, info: ErrorInfo) -> bool:
   310	        """Filter out the "iteration-dependent" errors and notes and store their
   311	        information to handle them after iteration is completed."""
   312
   313	        iter_errors = self.iteration_dependent_errors
   314
   315	        if info.code in (codes.UNREACHABLE, codes.REDUNDANT_EXPR, codes.REDUNDANT_CAST):
   316	            iter_errors.uselessness_errors[-1].add(
   317	                (info.code, info.message, info.line, info.column, info.end_line, info.end_column)
   318	            )
   319	            if info.code == codes.UNREACHABLE:
   320	                iter_errors.unreachable_lines[-1].update(range(info.line, info.end_line + 1))
   321	            return True
   322
   323	        return super().on_error(file, info)

   487	    def report(
   488	        self,
   489	        line: int,
   490	        column: int | None,
   491	        message: str,
   492	        code: ErrorCode | None = None,
   493	        *,
   494	        blocker: bool = False,
   495	        severity: str = "error",
   496	        file: str | None = None,
   497	        only_once: bool = False,
   498	        origin_span: Iterable[int] | None = None,
   499	        offset: int = 0,
   500	        end_line: int | None = None,
   501	        end_column: int | None = None,
   502	        parent_error: ErrorInfo | None = None,
   503	    ) -> ErrorInfo:
   504	        """Report message at the given line using the current error context.
   505
   506	        Args:
   507	            line: line number of error
   508	            column: column number of error
   509	            message: message to report
   510	            code: error code (defaults to 'misc'; not shown for notes)
   511	            blocker: if True, don't continue analysis after this error
   512	            severity: 'error' or 'note'
   513	            file: if non-None, override current file as context
   514	            only_once: if True, only report this exact message once per build
   515	            origin_span: if non-None, override current context as origin
   516	                         (type: ignores have effect here)
   517	            end_line: if non-None, override current context as end
   518	            parent_error: an error this note is attached to (for notes only).
   519	        """
   520	        if self.scope:
   521	            type = self.scope.current_type_name()
   522	            if self.scope.ignored > 0:
   523	                type = None  # Omit type context if nested function
   524	            function = self.scope.current_function_name()
   525	        else:
   526	            type = None
   527	            function = None
   528
   529	        if column is None:
   530	            column = -1
   531	        if end_column is None:
   532	            if column == -1:
   533	                end_column = -1
   534	            else:
   535	                end_column = column + 1
   536
   537	        if file is None:
   538	            file = self.file
   539	        if offset:
   540	            message = " " * offset + message
   541
   542	        if origin_span is None:
   543	            origin_span = [line]
   544
   545	        if end_line is None:
   546	            end_line = line
   547
   548	        code = code or (parent_error.code if parent_error else None)
   549	        code = code or (codes.MISC if not blocker else None)
   550
   551	        info = ErrorInfo(
   552	            import_ctx=self.import_context(),
   553	            file=file,
   554	            module=self.current_module(),
   555	            typ=type,
   556	            function_or_member=function,
   557	            line=line,
   558	            column=column,
   559	            end_line=end_line,
   560	            end_column=end_column,
   561	            severity=severity,
   562	            message=message,
   563	            code=code,
   564	            blocker=blocker,
   565	            only_once=only_once,
   566	            origin=(self.file, origin_span),
   567	            target=self.current_target(),
   568	            parent_error=parent_error,
   569	        )
   570	        self.add_error_info(info)
   571	        return info

   573	    def _add_error_info(self, file: str, info: ErrorInfo) -> None:
   574	        assert file not in self.flushed_files
   575	        # process the stack of ErrorWatchers before modifying any internal state
   576	        # in case we need to filter out the error entirely
   577	        if self._filter_error(file, info):
   578	            return
   579	        if file not in self.error_info_map:
   580	            self.error_info_map[file] = []
   581	        self.error_info_map[file].append(info)
   582	        if info.blocker:
   583	            self.has_blockers.add(file)
   584	        if info.code in (IMPORT, IMPORT_UNTYPED, IMPORT_NOT_FOUND):
   585	            self.seen_import_error = True
   586
   587	    def get_watchers(self) -> Iterator[ErrorWatcher]:
   588	        """Yield the `ErrorWatcher` stack from top to bottom."""
   589	        i = len(self._watchers)
   590	        while i > 0:
   591	            i -= 1
   592	            yield self._watchers[i]
   593
   594	    def _filter_error(self, file: str, info: ErrorInfo) -> bool:
   595	        """
   596	        process ErrorWatcher stack from top to bottom,
   597	        stopping early if error needs to be filtered out
   598	        """
   599	        return any(w.on_error(file, info) for w in self.get_watchers())
   600
   601	    def add_error_info(self, info: ErrorInfo) -> None:
   602	        file, lines = info.origin
   603	        # process the stack of ErrorWatchers before modifying any internal state
   604	        # in case we need to filter out the error entirely
   605	        # NB: we need to do this both here and in _add_error_info, otherwise we
   606	        # might incorrectly update the sets of ignored or only_once messages
   607	        if self._filter_error(file, info):
   608	            return
   609	        if not info.blocker:  # Blockers cannot be ignored
   610	            if file in self.ignored_lines:
   611	                # Check each line in this context for "type: ignore" comments.
   612	                # line == end_line for most nodes, so we only loop once.
   613	                for scope_line in lines:
   614	                    if self.is_ignored_error(scope_line, info, self.ignored_lines[file]):
   615	                        err_code = info.code or codes.MISC
   616	                        if not self.is_error_code_enabled(err_code):
   617	                            # Error code is disabled - don't mark the current
   618	                            # "type: ignore" comment as used.
   619	                            return
   620	                        # Annotation requests us to ignore all errors on this line.
   621	                        self.used_ignored_lines[file][scope_line].append(err_code.code)
   622	                        return
   623	            if file in self.ignored_files:
   624	                return
   625	        if info.only_once:
   626	            if info.message in self.only_once_messages:
   627	                return
   628	            self.only_once_messages.add(info.message)
   629	        if (
   630	            self.seen_import_error
   631	            and info.code not in (IMPORT, IMPORT_UNTYPED, IMPORT_NOT_FOUND)
   632	            and self.has_many_errors()
   633	        ):
   634	            # Missing stubs can easily cause thousands of errors about
   635	            # Any types, especially when upgrading to mypy 0.900,
   636	            # which no longer bundles third-party library stubs. Avoid
   637	            # showing too many errors to make it easier to see
   638	            # import-related errors.
   639	            info.hidden = True
   640	            self.report_hidden_errors(info)
   641	        self._add_error_info(file, info)
   642	        ignored_codes = self.ignored_lines.get(file, {}).get(info.line, [])
   643	        if ignored_codes and info.code:
   644	            # Something is ignored on the line, but not this error, so maybe the error
   645	            # code is incorrect.
   646	            msg = f'Error code "{info.code.code}" not covered by "type: ignore" comment'
   647	            if info.code in original_error_codes:
   648	                # If there seems to be a "type: ignore" with a stale error
   649	                # code, report a more specific note.
   650	                old_code = original_error_codes[info.code].code
   651	                if old_code in ignored_codes:
   652	                    msg = (
   653	                        f'Error code changed to {info.code.code}; "type: ignore" comment '
   654	                        + "may be out of date"
   655	                    )
   656	            note = ErrorInfo(
   657	                import_ctx=info.import_ctx,
   658	                file=info.file,
   659	                module=info.module,
   660	                typ=info.type,
   661	                function_or_member=info.function_or_member,
   662	                line=info.line,
   663	                column=info.column,
   664	                end
