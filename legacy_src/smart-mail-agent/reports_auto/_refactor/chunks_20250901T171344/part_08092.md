  2647	    elif isinstance(typ, TypeVarTupleType):
  2648	        # This is similar to non-generic instance types.
  2649	        fullname = scoped_type_var_name(typ)
  2650	        if verbosity >= 2 or (fullnames and fullname in fullnames):
  2651	            return fullname
  2652	        return typ.name
  2653	    elif isinstance(typ, ParamSpecType):
  2654	        # Concatenate[..., P]
  2655	        if typ.prefix.arg_types:
  2656	            args = format_callable_args(
  2657	                typ.prefix.arg_types, typ.prefix.arg_kinds, typ.prefix.arg_names, format, verbosity
  2658	            )
  2659
  2660	            return f"[{args}, **{typ.name_with_suffix()}]"
  2661	        else:
  2662	            # TODO: better disambiguate ParamSpec name clashes.
  2663	            return typ.name_with_suffix()
  2664	    elif isinstance(typ, TupleType):
  2665	        # Prefer the name of the fallback class (if not tuple), as it's more informative.
  2666	        if typ.partial_fallback.type.fullname != "builtins.tuple":
  2667	            return format(typ.partial_fallback)
  2668	        type_items = format_list(typ.items) or "()"
  2669	        return f"tuple[{type_items}]"
  2670	    elif isinstance(typ, TypedDictType):
  2671	        # If the TypedDictType is named, return the name
  2672	        if not typ.is_anonymous():
  2673	            return format(typ.fallback)
  2674	        items = []
  2675	        for item_name, item_type in typ.items.items():
  2676	            modifier = ""
  2677	            if item_name not in typ.required_keys:
  2678	                modifier += "?"
  2679	            if item_name in typ.readonly_keys:
  2680	                modifier += "="
  2681	            items.append(f"{item_name!r}{modifier}: {format(item_type)}")
  2682	        return f"TypedDict({{{', '.join(items)}}})"
  2683	    elif isinstance(typ, LiteralType):
  2684	        return f"Literal[{format_literal_value(typ)}]"
  2685	    elif isinstance(typ, UnionType):
  2686	        typ = get_proper_type(ignore_last_known_values(typ))
  2687	        if not isinstance(typ, UnionType):
  2688	            return format(typ)
  2689	        literal_items, union_items = separate_union_literals(typ)
  2690
  2691	        # Coalesce multiple Literal[] members. This also changes output order.
  2692	        # If there's just one Literal item, retain the original ordering.
  2693	        if len(literal_items) > 1:
  2694	            literal_str = "Literal[{}]".format(
  2695	                ", ".join(format_literal_value(t) for t in literal_items)
  2696	            )
  2697
  2698	            if len(union_items) == 1 and isinstance(get_proper_type(union_items[0]), NoneType):
  2699	                return (
  2700	                    f"{literal_str} | None"
  2701	                    if options.use_or_syntax()
  2702	                    else f"Optional[{literal_str}]"
  2703	                )
  2704	            elif union_items:
  2705	                return (
  2706	                    f"{literal_str} | {format_union(union_items)}"
  2707	                    if options.use_or_syntax()
  2708	                    else f"Union[{', '.join(format_union_items(union_items))}, {literal_str}]"
  2709	                )
  2710	            else:
  2711	                return literal_str
  2712	        else:
  2713	            # Only print Union as Optional if the Optional wouldn't have to contain another Union
  2714	            print_as_optional = (
  2715	                len(typ.items) - sum(isinstance(get_proper_type(t), NoneType) for t in typ.items)
  2716	                == 1
  2717	            )
  2718	            if print_as_optional:
  2719	                rest = [t for t in typ.items if not isinstance(get_proper_type(t), NoneType)]
  2720	                return (
  2721	                    f"{format(rest[0])} | None"
  2722	                    if options.use_or_syntax()
  2723	                    else f"Optional[{format(rest[0])}]"
  2724	                )
  2725	            else:
  2726	                s = (
  2727	                    format_union(typ.items)
  2728	                    if options.use_or_syntax()
  2729	                    else f"Union[{', '.join(format_union_items(typ.items))}]"
  2730	                )
  2731	            return s
  2732	    elif isinstance(typ, NoneType):
  2733	        return "None"
  2734	    elif isinstance(typ, AnyType):
  2735	        return "Any"
  2736	    elif isinstance(typ, DeletedType):
  2737	        return "<deleted>"
  2738	    elif isinstance(typ, UninhabitedType):
  2739	        return "Never"
  2740	    elif isinstance(typ, TypeType):
  2741	        return f"type[{format(typ.item)}]"
  2742	    elif isinstance(typ, FunctionLike):
  2743	        func = typ
  2744	        if func.is_type_obj():
  2745	            # The type of a type object type can be derived from the
  2746	            # return type (this always works).
  2747	            return format(TypeType.make_normalized(func.items[0].ret_type))
  2748	        elif isinstance(func, CallableType):
  2749	            if func.type_guard is not None:
  2750	                return_type = f"TypeGuard[{format(func.type_guard)}]"
  2751	            elif func.type_is is not None:
  2752	                return_type = f"TypeIs[{format(func.type_is)}]"
  2753	            else:
  2754	                return_type = format(func.ret_type)
  2755	            if func.is_ellipsis_args:
  2756	                return f"Callable[..., {return_type}]"
  2757	            param_spec = func.param_spec()
  2758	            if param_spec is not None:
  2759	                return f"Callable[{format(param_spec)}, {return_type}]"
  2760	            args = format_callable_args(
  2761	                func.arg_types, func.arg_kinds, func.arg_names, format, verbosity
  2762	            )
  2763	            return f"Callable[[{args}], {return_type}]"
  2764	        else:
  2765	            # Use a simple representation for function types; proper
  2766	            # function types may result in long and difficult-to-read
  2767	            # error messages.
  2768	            return "overloaded function"
  2769	    elif isinstance(typ, UnboundType):
  2770	        return typ.accept(TypeStrVisitor(options=options))
  2771	    elif isinstance(typ, Parameters):
  2772	        args = format_callable_args(typ.arg_types, typ.arg_kinds, typ.arg_names, format, verbosity)
  2773	        return f"[{args}]"
  2774	    elif typ is None:
  2775	        raise RuntimeError("Type is None")
  2776	    else:
  2777	        # Default case; we simply have to return something meaningful here.
  2778	        return "object"

  2828	def find_type_overlaps(*types: Type) -> set[str]:
  2829	    """Return a set of fullnames that share a short name and appear in either type.
  2830
  2831	    This is used to ensure that distinct types with the same short name are printed
  2832	    with their fullname.
  2833	    """
  2834	    d: dict[str, set[str]] = {}
  2835	    for type in types:
  2836	        for t in collect_all_named_types(type):
  2837	            if isinstance(t, ProperType) and isinstance(t, Instance):
  2838	                d.setdefault(t.type.name, set()).add(t.type.fullname)
  2839	            elif isinstance(t, TypeAliasType) and t.alias:
  2840	                d.setdefault(t.alias.name, set()).add(t.alias.fullname)
  2841	            else:
  2842	                assert isinstance(t, TypeVarLikeType)
  2843	                d.setdefault(t.name, set()).add(scoped_type_var_name(t))
  2844	    for shortname in d.keys():
  2845	        if f"typing.{shortname}" in TYPES_FOR_UNIMPORTED_HINTS:
  2846	            d[shortname].add(f"typing.{shortname}")
  2847
  2848	    overlaps: set[str] = set()
  2849	    for fullnames in d.values():
  2850	        if len(fullnames) > 1:
  2851	            overlaps.update(fullnames)
  2852	    return overlaps

  2888	def format_type_distinctly(*types: Type, options: Options, bare: bool = False) -> tuple[str, ...]:
  2889	    """Jointly format types to distinct strings.
  2890
  2891	    Increase the verbosity of the type strings until they become distinct
  2892	    while also requiring that distinct types with the same short name are
  2893	    formatted distinctly.
  2894
  2895	    By default, the returned strings are created using format_type() and will be
  2896	    quoted accordingly. If ``bare`` is True, the returned strings will not
  2897	    be quoted; callers who need to do post-processing of the strings before
  2898	    quoting them (such as prepending * or **) should use this.
  2899	    """
  2900	    overlapping = find_type_overlaps(*types)
  2901
  2902	    def format_single(arg: Type) -> str:
  2903	        return format_type_inner(arg, verbosity=0, options=options, fullnames=overlapping)
  2904
  2905	    min_verbosity = 0
  2906	    # Prevent emitting weird errors like:
  2907	    # ... has incompatible type "Callable[[int], Child]"; expected "Callable[[int], Parent]"
  2908	    if len(types) == 2:
  2909	        left, right = types
  2910	        left = get_proper_type(left)
  2911	        right = get_proper_type(right)
  2912	        # If the right type has named arguments, they may be the reason for incompatibility.
  2913	        # This excludes cases when right is Callable[[Something], None] without named args,
  2914	        # because that's usually the right thing to do.
  2915	        if (
  2916	            isinstance(left, CallableType)
  2917	            and isinstance(right, CallableType)
  2918	            and any(right.arg_names)
  2919	            and is_subtype(left, right, ignore_pos_arg_names=True)
  2920	        ):
  2921	            min_verbosity = 1
  2922
  2923	    for verbosity in range(min_verbosity, 2):
  2924	        strs = [
  2925	            format_type_inner(type, verbosity=verbosity, options=options, fullnames=overlapping)
  2926	            for type in types
  2927	        ]
  2928	        if len(set(strs)) == len(strs):
  2929	            break
  2930	    if bare:
  2931	        return tuple(strs)
  2932	    else:
  2933	        return tuple(quote_type_string(s) for s in strs)

  2946	def pretty_callable(tp: CallableType, options: Options, skip_self: bool = False) -> str:
  2947	    """Return a nice easily-readable representation of a callable type.
  2948	    For example:
  2949	        def [T <: int] f(self, x: int, y: T) -> None
  2950
  2951	    If skip_self is True, print an actual callable type, as it would appear
  2952	    when bound on an instance/class, rather than how it would appear in the
  2953	    defining statement.
  2954	    """
  2955	    s = ""
  2956	    asterisk = False
  2957	    slash = False
  2958	    for i in range(len(tp.arg_types)):
  2959	        if s:
  2960	            s += ", "
  2961	        if tp.arg_kinds[i].is_named() and not asterisk:
  2962	            s += "*, "
  2963	            asterisk = True
  2964	        if tp.arg_kinds[i] == ARG_STAR:
  2965	            s += "*"
  2966	            asterisk = True
  2967	        if tp.arg_kinds[i] == ARG_STAR2:
  2968	            s += "**"
  2969	        name = tp.arg_names[i]
  2970	        if name:
  2971	            s += name + ": "
  2972	        type_str = format_type_bare(tp.arg_types[i], options)
  2973	        if tp.arg_kinds[i] == ARG_STAR2 and tp.unpack_kwargs:
  2974	            type_str = f"Unpack[{type_str}]"
  2975	        s += type_str
  2976	        if tp.arg_kinds[i].is_optional():
  2977	            s += " = ..."
  2978	        if (
  2979	            not slash
  2980	            and tp.arg_kinds[i].is_positional()
  2981	            and name is None
  2982	            and (
  2983	                i == len(tp.arg_types) - 1
  2984	                or (tp.arg_names[i + 1] is not None or not tp.arg_kinds[i + 1].is_positional())
  2985	            )
  2986	        ):
  2987	            s += ", /"
  2988	            slash = True
  2989
  2990	    # If we got a "special arg" (i.e: self, cls, etc...), prepend it to the arg list
  2991	    if (
  2992	        isinstance(tp.definition, FuncDef)
  2993	        and hasattr(tp.definition, "arguments")
  2994	        and not tp.from_concatenate
  2995	    ):
  2996	        definition_arg_names = [arg.variable.name for arg in tp.definition.arguments]
  2997	        if (
  2998	            len(definition_arg_names) > len(tp.arg_names)
  2999	            and definition_arg_names[0]
  3000	            and not skip_self
  3001	        ):
  3002	            if s:
  3003	                s = ", " + s
  3004	            s = definition_arg_names[0] + s
  3005	        s = f"{tp.definition.name}({s})"
  3006	    elif tp.name:
  3007	        first_arg = tp.def_extras.get("first_arg")
  3008	        if first_arg:
  3009	            if s:
  3010	                s = ", " + s
  3011	            s = first_arg + s
  3012	        s = f"{tp.name.split()[0]}({s})"  # skip "of Class" part
  3013	    else:
  3014	        s = f"({s})"
  3015
  3016	    s += " -> "
  3017	    if tp.type_guard is not None:
  3018	        s += f"TypeGuard[{format_type_bare(tp.type_guard, options)}]"
  3019	    elif tp.type_is is not None:
  3020	        s += f"TypeIs[{format_type_bare(tp.type_is, options)}]"
  3021	    else:
  3022	        s += format_type_bare(tp.ret_type, options)
  3023
  3024	    if tp.variables:
  3025	        tvars = []
  3026	        for tvar in tp.variables:
  3027	            if isinstance(tvar, TypeVarType):
  3028	                upper_bound = get_proper_type(tvar.upper_bound)
  3029	                if not (
  3030	                    isinstance(upper_bound, Instance)
  3031	                    and upper_bound.type.fullname == "builtins.object"
  3032	                ):
  3033	                    tvars.append(f"{tvar.name}: {format_type_bare(upper_bound, options)}")
  3034	                elif tvar.values:
  3035	                    tvars.append(

  3037	                            tvar.name,
  3038	                            ", ".join([format_type_bare(tp, options) for tp in tvar.values]),

  3040	                    )
  3041	                else:
  3042	                    tvars.append(tvar.name)
  3043	            else:
  3044	                # For other TypeVarLikeTypes, just use the repr
  3045	                tvars.append(repr(tvar))
  3046	        s = f"[{', '.join(tvars)}] {s}"
  3047	    return f"def {s}"
