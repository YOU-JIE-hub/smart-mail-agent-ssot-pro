PassStmt) or (
   155	            isinstance(stmt, ExpressionStmt) and isinstance(stmt.expr, EllipsisExpr)
   156	        ):
   157	            continue
   158	        elif isinstance(stmt, AssignmentStmt):
   159	            if len(stmt.lvalues) != 1:
   160	                builder.error("Multiple assignment in class bodies not supported", stmt.line)
   161	                continue
   162	            lvalue = stmt.lvalues[0]
   163	            if not isinstance(lvalue, NameExpr):
   164	                builder.error(
   165	                    "Only assignment to variables is supported in class bodies", stmt.line
   166	                )
   167	                continue
   168	            # We want to collect class variables in a dictionary for both real
   169	            # non-extension classes and fake dataclass ones.
   170	            cls_builder.add_attr(lvalue, stmt)
   171
   172	        elif isinstance(stmt, ExpressionStmt) and isinstance(stmt.expr, StrExpr):
   173	            # Docstring. Ignore
   174	            pass
   175	        else:
   176	            builder.error("Unsupported statement in class body", stmt.line)
   177
   178	    # Generate implicit property setters/getters
   179	    for name, decl in ir.method_decls.items():
   180	        if decl.implicit and decl.is_prop_getter:
   181	            getter_ir = gen_property_getter_ir(builder, decl, cdef, ir.is_trait)
   182	            builder.functions.append(getter_ir)
   183	            ir.methods[getter_ir.decl.name] = getter_ir
   184
   185	            setter_ir = None
   186	            setter_name = PROPSET_PREFIX + name
   187	            if setter_name in ir.method_decls:
   188	                setter_ir = gen_property_setter_ir(
   189	                    builder, ir.method_decls[setter_name], cdef, ir.is_trait
   190	                )
   191	                builder.functions.append(setter_ir)
   192	                ir.methods[setter_name] = setter_ir
   193
   194	            ir.properties[name] = (getter_ir, setter_ir)
   195	            # TODO: Generate glue method if needed?
   196	            # TODO: Do we need interpreted glue methods? Maybe not?
   197
   198	    cls_builder.finalize(ir)

   290	        # Variable declaration with no body
   291	        if isinstance(stmt.rvalue, TempNode):

   294	        if not (is_class_var(lvalue) or stmt.is_final_def):

   299	            py_setattr_op, [typ, self.builder.load_str(lvalue.name), value], stmt.line
   300	        )
   301	        if self.builder.non_function_scope() and stmt.is_final_def:
   302	            self.builder.init_final_static(lvalue, value, self.cdef.name)
   303

   305	        attrs_with_defaults, default_assignments = find_attr_initializers(
   306	            self.builder, self.cdef, self.skip_attr_default

   313	class DataClassBuilder(ExtClassBuilder):
   314	    # controls whether an __annotations__ attribute should be added to the class
   315	    # __dict__.  This is not desirable for attrs classes where auto_attribs is
   316	    # disabled, as attrs will reject it.
   317	    add_annotations_to_dict = True
   318
   319	    def __init__(self, builder: IRBuilder, cdef: ClassDef) -> None:
   320	        super().__init__(builder, cdef)
   321	        self.non_ext = self.create_non_ext_info()
   322
   323	    def create_non_ext_info(self) -> NonExtClassInfo:
   324	        """Set up a NonExtClassInfo to track dataclass attributes.
   325
   326	        In addition to setting up a normal extension class for dataclasses,
   327	        we also collect its class attributes like a non-extension class so
   328	        that we can hand them to the dataclass decorator.
   329	        """
   330	        return NonExtClassInfo(
   331	            self.builder.call_c(dict_new_op, [], self.cdef.line),
   332	            self.builder.add(TupleSet([], self.cdef.line)),
   333	            self.builder.call_c(dict_new_op, [], self.cdef.line),
   334	            self.builder.add(LoadAddress(type_object_op.type, type_object_op.src, self.cdef.line)),
   335	        )
   336
   337	    def skip_attr_default(self, name: str, stmt: AssignmentStmt) -> bool:
   338	        return stmt.type is not None
   339
   340	    def get_type_annotation(self, stmt: AssignmentStmt) -> TypeInfo | None:
   341	        # We populate __annotations__ because dataclasses uses it to determine
   342	        # which attributes to compute on.
   343	        ann_type = get_proper_type(stmt.type)
   344	        if isinstance(ann_type, Instance):
   345	            return ann_type.type
   346	        return None
   347
   348	    def add_attr(self, lvalue: NameExpr, stmt: AssignmentStmt) -> None:
   349	        add_non_ext_class_attr_ann(
   350	            self.builder, self.non_ext, lvalue, stmt, self.get_type_annotation
   351	        )
   352	        add_non_ext_class_attr(
   353	            self.builder, self.non_ext, lvalue, stmt, self.cdef, self.attrs_to_cache
   354	        )
   355	        super().add_attr(lvalue, stmt)
   356
   357	    def finalize(self, ir: ClassIR) -> None:
   358	        """Generate code to finish instantiating a dataclass.
   359
   360	        This works by replacing all of the attributes on the class
   361	        (which will be descriptors) with whatever they would be in a
   362	        non-extension class, calling dataclass, then switching them back.
   363
   364	        The resulting class is an extension class and instances of it do not
   365	        have a __dict__ (unless something else requires it).
   366	        All methods written explicitly in the source are compiled and
   367	        may be called through the vtable while the methods generated
   368	        by dataclasses are interpreted and may not be.
   369
   370	        (If we just called dataclass without doing this, it would think that all
   371	        of the descriptors for our attributes are default values and generate an
   372	        incorrect constructor. We need to do the switch so that dataclass gets the
   373	        appropriate defaults.)
   374	        """
   375	        super().finalize(ir)
   376	        assert self.type_obj
   377	        add_dunders_to_non_ext_dict(
   378	            self.builder, self.non_ext, self.cdef.line, self.add_annotations_to_dict
   379	        )
   380	        dec = self.builder.accept(
   381	            next(d for d in self.cdef.decorators if is_dataclass_decorator(d))
   382	        )
   383	        dataclass_type_val = self.builder.load_str(dataclass_type(self.cdef) or "unknown")
   384	        self.builder.call_c(
   385	            dataclass_sleight_of_hand,
   386	            [dec, self.type_obj, self.non_ext.dict, self.non_ext.anns, dataclass_type_val],
   387	            self.cdef.line,
   388	        )

   391	class AttrsClassBuilder(DataClassBuilder):
   392	    """Create IR for an attrs class where auto_attribs=False (the default).
   393
   394	    When auto_attribs is enabled, attrs classes behave similarly to dataclasses
   395	    (i.e. types are stored as annotations on the class) and are thus handled
   396	    by DataClassBuilder, but when auto_attribs is disabled the types are
   397	    provided via attr.ib(type=...)
   398	    """
   399
   400	    add_annotations_to_dict = False
   401
   402	    def skip_attr_default(self, name: str, stmt: AssignmentStmt) -> bool:
   403	        return True
   404
   405	    def get_type_annotation(self, stmt: AssignmentStmt) -> TypeInfo | None:
   406	        if isinstance(stmt.rvalue, CallExpr):
   407	            # find the type arg in `attr.ib(type=str)`
   408	            callee = stmt.rvalue.callee
   409	            if (
   410	                isinstance(callee, MemberExpr)
   411	                and callee.fullname in ["attr.ib", "attr.attr"]
   412	                and "type" in stmt.rvalue.arg_names
   413	            ):
   414	                index = stmt.rvalue.arg_names.index("type")
   415	                type_name = stmt.rvalue.args[index]
   416	                if isinstance(type_name, NameExpr) and isinstance(type_name.node, TypeInfo):
   417	                    lvalue = stmt.lvalues[0]
   418	                    assert isinstance(lvalue, NameExpr)
   419	                    return type_name.node
   420	        return None

   423	def allocate_class(builder: IRBuilder, cdef: ClassDef) -> Value:
   424	    # OK AND NOW THE FUN PART
   425	    base_exprs = cdef.base_type_exprs + cdef.removed_base_type_exprs
   426	    new_style_type_args = cdef.type_args
   427	    if new_style_type_args:
   428	        bases = [make_generic_base_class(builder, cdef.fullname, new_style_type_args, cdef.line)]
   429	    else:
   430	        bases = []
   431
   432	    if base_exprs or new_style_type_args:
   433	        bases.extend([builder.accept(x) for x in base_exprs])
   434	        tp_bases = builder.new_tuple(bases, cdef.line)
   435	    else:
   436	        tp_bases = builder.add(LoadErrorValue(object_rprimitive, is_borrowed=True))
   437	    modname = builder.load_str(builder.module_name)
   438	    template = builder.add(
   439	        LoadStatic(object_rprimitive, cdef.name + "_template", builder.module_name, NAMESPACE_TYPE)
   440	    )
   441	    # Create the class
   442	    tp = builder.call_c(pytype_from_template_op, [template, tp_bases, modname], cdef.line)
   443	    # Immediately fix up the trait vtables, before doing anything with the class.
   444	    ir = builder.mapper.type_to_ir[cdef.info]
   445	    if not ir.is_trait and not ir.builtin_base:
   446	        builder.add(
   447	            Call(
   448	                FuncDecl(
   449	                    cdef.name + "_trait_vtable_setup",
   450	                    None,
   451	                    builder.module_name,
   452	                    FuncSignature([], bool_rprimitive),
   453	                ),
   454	                [],
   455	                -1,
   456	            )
   457	        )
   458	    # Populate a '__mypyc_attrs__' field containing the list of attrs
   459	    builder.primitive_op(
   460	        py_setattr_op,
   461	        [
   462	            tp,
   463	            builder.load_str("__mypyc_attrs__"),
   464	            create_mypyc_attrs_tuple(builder, builder.mapper.type_to_ir[cdef.info], cdef.line),
   465	        ],
   466	        cdef.line,
   467	    )
   468
   469	    # Save the class
   470	    builder.add(InitStatic(tp, cdef.name, builder.module_name, NAMESPACE_TYPE))
   471
   472	    # Add it to the dict
   473	    builder.primitive_op(
   474	        dict_set_item_op, [builder.load_globals_dict(), builder.load_str(cdef.name), tp], cdef.line
   475	    )
   476
   477	    return tp

   480	def make_generic_base_class(
   481	    builder: IRBuilder, fullname: str, type_args: list[TypeParam], line: int
   482	) -> Value:
   483	    """Construct Generic[...] base class object for a new-style generic class (Python 3.12)."""
   484	    mod = builder.call_c(import_op, [builder.load_str("_typing")], line)
   485	    tvs = create_type_params(builder, mod, type_args, line)
   486	    args = []
   487	    for tv, type_param in zip(tvs, type_args):
   488	        if type_param.kind == TYPE_VAR_TUPLE_KIND:
   489	            # Evaluate *Ts for a TypeVarTuple
   490	            it = builder.primitive_op(iter_op, [tv], line)
   491	            tv = builder.call_c(next_op, [it], line)
   492	        args.append(tv)
   493
   494	    gent = builder.py_get_attr(mod, "Generic", line)
   495	    if len(args) == 1:
   496	        arg = args[0]
   497	    else:
   498	        arg = builder.new_tuple(args, line)
   499
   500	    base = builder.primitive_op(py_get_item_op, [gent, arg], line)
   501	    return base

   512	def populate_non_ext_bases(builder: IRBuilder, cdef: ClassDef) -> Value:
   513	    """Create base class tuple of a non-extension class.
   514
   515	    The tuple is passed to the metaclass constructor.
   516	    """
   517	    is_named_tuple = cdef.info.is_named_tuple
   518	    ir = builder.mapper.type_to_ir[cdef.info]
   519	    bases = []
   520	    for cls in (b.type for b in cdef.info.bases):
   521	        if cls.fullname == "builtins.object":
   522	            continue
   523	        if is_named_tuple and cls.fullname in (
   524	            "typing.Sequence",
   525	            "typing.Iterable",
   526	            "typing.Collection",
   527	            "typing.Reversible",
   528	            "typing.Container",
   529	            "typing.Sized",
   530	        ):
   531	            # HAX: Synthesized base classes added by mypy don't exist at runtime, so skip them.
   532	            #      This could break if they were added explicitly, though...
   533	            continue
   534	        # Add the current class to the base classes list of concrete subclasses
   535	        if cls in builder.mapper.type_to_ir:
   536	            base_ir = builder.mapper.type_to_ir[cls]
   537	            if base_ir.children is not None:
   538	                base_ir.children.append(ir)
   539
   540	        if cls.fullname in MAGIC_TYPED_DICT_CLASSES:
   541	            # HAX: Mypy internally represents TypedDict classes differently from what
   542	            #      should happen at runtime. Replace with something that works.
   543	            module = "typing"
   544	            name = "_TypedDict"
   545	            base = builder.get_module_attr(module, name, cdef.line)
   546	        elif is_named_tuple and cls.fullname == "builtins.tuple":
   547	            name = "_NamedTuple"
   548	            base = builder.get_module_attr("typing", name, cdef.line)
   549	        else:
   550	            cls_module = cls.fullname.rsplit(".", 1)[0]
   551	            if cls_module == builder.current_module:
   552	                base = builder.load_global_str(cls.name, cdef.line)
   553	            else:
   554	                base = builder.load_module_attr_by_fullname(cls.fullname, cdef.line)
   555	        bases.append(base)
   556	        if cls.fullname in MAGIC_TYPED_DICT_CLASSES:
   557	            # The remaining base classes are synthesized by mypy and should be ignored.
   558	            break
   559	    return builder.new_tuple(bases, cdef.line)
