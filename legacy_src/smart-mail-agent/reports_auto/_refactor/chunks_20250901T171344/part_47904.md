  2650	        original_caller_type: Type,

  2657	        context: Context,
  2658	        outer_context: Context,

  2689	    def check_overload_call(
  2690	        self,
  2691	        callee: Overloaded,
  2692	        args: list[Expression],
  2693	        arg_kinds: list[ArgKind],
  2694	        arg_names: Sequence[str | None] | None,
  2695	        callable_name: str | None,
  2696	        object_type: Type | None,
  2697	        context: Context,
  2698	    ) -> tuple[Type, Type]:
  2699	        """Checks a call to an overloaded function."""
  2700	        # Normalize unpacked kwargs before checking the call.
  2701	        callee = callee.with_unpacked_kwargs()
  2702	        arg_types = self.infer_arg_types_in_empty_context(args)
  2703	        # Step 1: Filter call targets to remove ones where the argument counts don't match
  2704	        plausible_targets = self.plausible_overload_call_targets(
  2705	            arg_types, arg_kinds, arg_names, callee
  2706	        )
  2707
  2708	        # Step 2: If the arguments contain a union, we try performing union math first,
  2709	        #         instead of picking the first matching overload.
  2710	        #         This is because picking the first overload often ends up being too greedy:
  2711	        #         for example, when we have a fallback alternative that accepts an unrestricted
  2712	        #         typevar. See https://github.com/python/mypy/issues/4063 for related discussion.
  2713	        erased_targets: list[CallableType] | None = None
  2714	        unioned_result: tuple[Type, Type] | None = None
  2715
  2716	        # Determine whether we need to encourage union math. This should be generally safe,
  2717	        # as union math infers better results in the vast majority of cases, but it is very
  2718	        # computationally intensive.
  2719	        none_type_var_overlap = self.possible_none_type_var_overlap(arg_types, plausible_targets)
  2720	        union_interrupted = False  # did we try all union combinations?
  2721	        if any(self.real_union(arg) for arg in arg_types):
  2722	            try:
  2723	                with self.msg.filter_errors():
  2724	                    unioned_return = self.union_overload_result(
  2725	                        plausible_targets,
  2726	                        args,
  2727	                        arg_types,
  2728	                        arg_kinds,
  2729	                        arg_names,
  2730	                        callable_name,
  2731	                        object_type,
  2732	                        none_type_var_overlap,
  2733	                        context,
  2734	                    )
  2735	            except TooManyUnions:
  2736	                union_interrupted = True
  2737	            else:
  2738	                # Record if we succeeded. Next we need to see if maybe normal procedure
  2739	                # gives a narrower type.
  2740	                if unioned_return:
  2741	                    returns, inferred_types = zip(*unioned_return)
  2742	                    # Note that we use `combine_function_signatures` instead of just returning
  2743	                    # a union of inferred callables because for example a call
  2744	                    # Union[int -> int, str -> str](Union[int, str]) is invalid and
  2745	                    # we don't want to introduce internal inconsistencies.
  2746	                    unioned_result = (
  2747	                        make_simplified_union(list(returns), context.line, context.column),
  2748	                        self.combine_function_signatures(get_proper_types(inferred_types)),
  2749	                    )
  2750
  2751	        # Step 3: We try checking each branch one-by-one.
  2752	        inferred_result = self.infer_overload_return_type(
  2753	            plausible_targets,
  2754	            args,
  2755	            arg_types,
  2756	            arg_kinds,
  2757	            arg_names,
  2758	            callable_name,
  2759	            object_type,
  2760	            context,
  2761	        )
  2762	        # If any of checks succeed, stop early.
  2763	        if inferred_result is not None and unioned_result is not None:
  2764	            # Both unioned and direct checks succeeded, choose the more precise type.
  2765	            if (
  2766	                is_subtype(inferred_result[0], unioned_result[0])
  2767	                and not isinstance(get_proper_type(inferred_result[0]), AnyType)
  2768	                and not none_type_var_overlap
  2769	            ):
  2770	                return inferred_result
  2771	            return unioned_result
  2772	        elif unioned_result is not None:
  2773	            return unioned_result
  2774	        elif inferred_result is not None:
  2775	            return inferred_result
  2776
  2777	        # Step 4: Failure. At this point, we know there is no match. We fall back to trying
  2778	        #         to find a somewhat plausible overload target using the erased types
  2779	        #         so we can produce a nice error message.
  2780	        #
  2781	        #         For example, suppose the user passes a value of type 'List[str]' into an
  2782	        #         overload with signatures f(x: int) -> int and f(x: List[int]) -> List[int].
  2783	        #
  2784	        #         Neither alternative matches, but we can guess the user probably wants the
  2785	        #         second one.
  2786	        erased_targets = self.overload_erased_call_targets(
  2787	            plausible_targets, arg_types, arg_kinds, arg_names, args, context
  2788	        )
  2789
  2790	        # Step 5: We try and infer a second-best alternative if possible. If not, fall back
  2791	        #         to using 'Any'.
  2792	        if len(erased_targets) > 0:
  2793	            # Pick the first plausible erased target as the fallback
  2794	            # TODO: Adjust the error message here to make it clear there was no match.
  2795	            #       In order to do this, we need to find a clean way of associating
  2796	            #       a note with whatever error message 'self.check_call' will generate.
  2797	            #       In particular, the note's line and column numbers need to be the same
  2798	            #       as the error's.
  2799	            target: Type = erased_targets[0]
  2800	        else:
  2801	            # There was no plausible match: give up
  2802	            target = AnyType(TypeOfAny.from_error)
  2803	            if not is_operator_method(callable_name):
  2804	                code = None
  2805	            else:
  2806	                code = codes.OPERATOR
  2807	            self.msg.no_variant_matches_arguments(callee, arg_types, context, code=code)
  2808
  2809	        result = self.check_call(
  2810	            target,
  2811	            args,
  2812	            arg_kinds,
  2813	            context,
  2814	            arg_names,
  2815	            callable_name=callable_name,
  2816	            object_type=object_type,
  2817	        )
  2818	        # Do not show the extra error if the union math was forced.
  2819	        if union_interrupted and not none_type_var_overlap:
  2820	            self.chk.fail(message_registry.TOO_MANY_UNION_COMBINATIONS, context)
  2821	        return result
  2822
  2823	    def plausible_overload_call_targets(
  2824	        self,
  2825	        arg_types: list[Type],
  2826	        arg_kinds: list[ArgKind],
  2827	        arg_names: Sequence[str | None] | None,
  2828	        overload: Overloaded,
  2829	    ) -> list[CallableType]:
  2830	        """Returns all overload call targets that having matching argument counts.
  2831
  2832	        If the given args contains a star-arg (*arg or **kwarg argument, except for
  2833	        ParamSpec), this method will ensure all star-arg overloads appear at the start
  2834	        of the list, instead of their usual location.
  2835
  2836	        The only exception is if the starred argument is something like a Tuple or a
  2837	        NamedTuple, which has a definitive "shape". If so, we don't move the corresponding
  2838	        alternative to the front since we can infer a more precise match using the original
  2839	        order."""
  2840
  2841	        def has_shape(typ: Type) -> bool:
  2842	            typ = get_proper_type(typ)
  2843	            return isinstance(typ, (TupleType, TypedDictType)) or (
  2844	                isinstance(typ, Instance) and typ.type.is_named_tuple
  2845	            )
  2846
  2847	        matches: list[CallableType] = []
  2848	        star_matches: list[CallableType] = []
  2849
  2850	        args_have_var_arg = False
  2851	        args_have_kw_arg = False
  2852	        for kind, typ in zip(arg_kinds, arg_types):
  2853	            if kind == ARG_STAR and not has_shape(typ):
  2854	                args_have_var_arg = True
  2855	            if kind == ARG_STAR2 and not has_shape(typ):
  2856	                args_have_kw_arg = True
  2857
  2858	        for typ in overload.items:
  2859	            formal_to_actual = map_actuals_to_formals(
  2860	                arg_kinds, arg_names, typ.arg_kinds, typ.arg_names, lambda i: arg_types[i]
  2861	            )
  2862	            with self.msg.filter_errors():
  2863	                if typ.param_spec() is not None:
  2864	                    # ParamSpec can be expanded in a lot of different ways. We may try
  2865	                    # to expand it here instead, but picking an impossible overload
  2866	                    # is safe: it will be filtered out later.
  2867	                    # Unlike other var-args signatures, ParamSpec produces essentially
  2868	                    # a fixed signature, so there's no need to push them to the top.
  2869	                    matches.append(typ)
  2870	                elif self.check_argument_count(
  2871	                    typ, arg_types, arg_kinds, arg_names, formal_to_actual, None
  2872	                ):
  2873	                    if args_have_var_arg and typ.is_var_arg:
  2874	                        star_matches.append(typ)
  2875	                    elif args_have_kw_arg and typ.is_kw_arg:
  2876	                        star_matches.append(typ)
  2877	                    else:
  2878	                        matches.append(typ)
  2879
  2880	        return star_matches + matches

  2882	    def infer_overload_return_type(
  2883	        self,
  2884	        plausible_targets: list[CallableType],
  2885	        args: list[Expression],
  2886	        arg_types: list[Type],
  2887	        arg_kinds: list[ArgKind],
  2888	        arg_names: Sequence[str | None] | None,
  2889	        callable_name: str | None,
  2890	        object_type: Type | None,
  2891	        context: Context,
  2892	    ) -> tuple[Type, Type] | None:
  2893	        """Attempts to find the first matching callable from the given list.
  2894
  2895	        If a match is found, returns a tuple containing the result type and the inferred
  2896	        callee type. (This tuple is meant to be eventually returned by check_call.)
  2897	        If multiple targets match due to ambiguous Any parameters, returns (AnyType, AnyType).
  2898	        If no targets match, returns None.
  2899
  2900	        Assumes all of the given targets have argument counts compatible with the caller.
  2901	        """
  2902
  2903	        matches: list[CallableType] = []
  2904	        return_types: list[Type] = []
  2905	        inferred_types: list[Type] = []
  2906	        args_contain_any = any(map(has_any_type, arg_types))
  2907	        type_maps: list[dict[Expression, Type]] = []
  2908
  2909	        for typ in plausible_targets:
  2910	            assert self.msg is self.chk.msg
  2911	            with self.msg.filter_errors() as w:
  2912	                with self.chk.local_type_map() as m:
  2913	                    ret_type, infer_type = self.check_call(
  2914	                        callee=typ,
  2915	                        args=args,
  2916	                        arg_kinds=arg_kinds,
  2917	                        arg_names=arg_names,
  2918	                        context=context,
  2919	                        callable_name=callable_name,
  2920	                        object_type=object_type,
  2921	                    )
  2922	            is_match = not w.has_new_errors()
  2923	            if is_match:
  2924	                # Return early if possible; otherwise record info, so we can
  2925	                # check for ambiguity due to 'Any' below.
  2926	                if not args_contain_any:
  2927	                    self.chk.store_types(m)
  2928	                    return ret_type, infer_type
  2929	                p_infer_type = get_proper_type(infer_type)
  2930	                if isinstance(p_infer_type, CallableType):
  2931	                    # Prefer inferred types if possible, this will avoid false triggers for
  2932	                    # Any-ambiguity caused by arguments with Any passed to generic overloads.
  2933	                    matches.append(p_infer_type)
  2934	                else:
  2935	                    matches.append(typ)
  2936	                return_types.append(ret_type)
  2937	                inferred_types.append(infer_type)
  2938	                type_maps.append(m)
  2939
  2940	        if not matches:
  2941	            return None
  2942	        elif any_causes_overload_ambiguity(matches, return_types, arg_types, arg_kinds, arg_names):
  2943	            # An argument of type or containing the type 'Any' caused ambiguity.
  2944	            # We try returning a precise type if we can. If not, we give up and just return 'Any'.
  2945	            if all_same_types(return_types):
  2946	                self.chk.store_types(type_maps[0])
  2947	                return return_types[0], inferred_types[0]
  2948	            elif all_same_types([erase_type(typ) for typ in return_types]):
  2949	                self.chk.store_types(type_maps[0])
  2950	                return erase_type(return_types[0]), erase_type(inferred_types[0])
  2951	            else:
  2952	                return self.check_call(
  2953	                    callee=AnyType(TypeOfAny.special_form),
  2954	                    args=args,
  2955	                    arg_kinds=arg_kinds,
  2956	                    arg_names=arg_names,
  2957	                    context=context,
  2958	                    callable_name=callable_name,
  2959	                    object_type=object_type,
  2960	                )
  2961	        else:
  2962	            # Success! No ambiguity; return the first match.
  2963	            self.chk.store_types(type_maps[0])
  2964	            return return_types[0], inferred_types[0]

  2987	    def possible_none_type_var_overlap(
  2988	        self, arg_types: list[Type], plausible_targets: list[CallableType]
  2989	    ) -> bool:
  2990	        """Heuristic to determine whether we need to try forcing union math.
  2991
  2992	        This is needed to avoid greedy type variable match in situations like this:
  2993	            @overload
  2994	            def foo(x: None) -> None: ...
  2995	            @overload
  2996	            def foo(x: T) -> list[T]: ...
  2997
  2998	            x: int | None
  2999	            foo(x)
  3000	        we want this call to infer list[int] | None, not list[int | None].
  3001	        """
  3002	        if not plausible_targets or not arg_types:
  3003	            return False
  3004	        has_optional_arg = False
  3005	        for arg_type in get_proper_types(arg_types):
  3006	            if not isinstance(arg_type, UnionType):
  3007	                continue
  3008	            for item in get_proper_types(arg_type.items):
  3009	                if isinstance(item, NoneType):
  3010	                    has_optional_arg = True
  3011	                    break
  3012	        if not has_optional_arg:
  3013	            return False
  3014
  3015	        min_prefix = min(len(c.arg_types) for c in plausible_targets)
  3016	        for i in range(min_prefix):
  3017	            if any(
  3018	                isinstance(get_proper_type(c.arg_types[i]), NoneType) for c in plausible_targets
  3019	            ) and any(
  3020	                isinstance(get_proper_type(c.arg_types[i]), TypeVarType) for c in plausible_targets
  3021	            ):
  3022	                return True
  3023	        return False

  3025	    def union_overload_result(
  3026	        self,
  3027	        plausible_targets: list[CallableType],
  3028	        args: list[Expression],
  3029	        arg_types: list[Type],
  3030	        arg_kinds: list[ArgKind],
  3031	        arg_names: Sequence[str | None] | None,
  3032	        callable_name: str | None,
  3033	        object_type: Type | None,
  3034	        none_type_var_overlap: bool,
  3035	        context: Context,
  3036	        level: int = 0,
  3037	    ) -> list[tuple[Type, Type]] | None:
  3038	        """Accepts a list of overload signatures and attempts to match calls by destructuring
  3039	        the first union.
  3040
  3041	        Return a list of (<return type>, <inferred variant type>) if call succeeds for every
  3042	        item of the desctructured union. Returns None if there is no match.
  3043	        """
  3044	        # Step 1: If we are already too deep, then stop immediately. Otherwise mypy might
  3045	        # hang for long time because of a weird overload call. The caller will get
  3046	        # the exception and generate an appropriate note message, if needed.
  3047	        if level >= MAX_UNIONS:
  3048	            raise TooManyUnions
  3049
  3050	        # Step 2: Find position of the first union in arguments. Return the normal inferred
  3051
