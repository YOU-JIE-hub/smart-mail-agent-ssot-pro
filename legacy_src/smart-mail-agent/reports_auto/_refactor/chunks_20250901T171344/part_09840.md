    25		format = _format
    26
    27		S = {}
    28		chunks = []
    29		pos = 0
    30		n = 0
    31
    32		while 1:
    33			pc = format.find("%", pos)
    34			if pc < 0: break
    35			nextchar = format[pc+1]
    36
    37			if nextchar == "(":
    38				chunks.append(format[pos:pc])

    40				while level:
    41					match, pos = _matchorfail(format, pos)
    42					tstart, tend = match.regs[3]
    43					token = format[tstart:tend]
