   652	    def handle_pageEnd(self):
   653	        ''' show the current page
   654	            check the next page template
   655	            hang a page begin
   656	        '''
   657	        self._removeVars(('page','frame'))
   658	        self._leftExtraIndent = self.frame._leftExtraIndent
   659	        self._rightExtraIndent = self.frame._rightExtraIndent
   660	        self._frameBGs = self.frame._frameBGs
   661	        #detect infinite loops...
   662	        if self._curPageFlowableCount == 0:
   663	            self._emptyPages += 1
   664	        else:
   665	            self._emptyPages = 0
   666	        if self._emptyPages >= self._emptyPagesAllowed:
   667	            if 1:
   668	                ident = "More than %d pages generated without content - halting layout.  Likely that a flowable is too large for any frame." % self._emptyPagesAllowed
   669	                #leave to keep apart from the raise
   670	                raise LayoutError(ident)
   671	            else:
   672	                pass    #attempt to restore to good state
   673	        else:
   674	            if self._onProgress:
   675	                self._onProgress('PAGE', self.canv.getPageNumber())
   676	            self.pageTemplate.afterDrawPage(self.canv, self)
   677	            self.pageTemplate.onPageEnd(self.canv, self)
   678	            self.afterPage()
   679	            if self._debug: logger.debug("ending page %d" % self.page)
   680	            self.canv.setPageRotation(getattr(self.pageTemplate,'rotation',self.rotation))
   681	            self.canv.showPage()
   682	            self._setPageTemplate()
   683	            if self._emptyPages==0:
   684	                pass    #store good state here
   685	        self._hanging.append(PageBegin)

   735	    def handle_nextPageTemplate(self,pt):
   736	        '''On endPage change to the page template with name or index pt'''
   737	        if isinstance(pt,strTypes):
   738	            if hasattr(self, '_nextPageTemplateCycle'): del self._nextPageTemplateCycle
   739	            for t in self.pageTemplates:
   740	                if t.id == pt:
   741	                    self._nextPageTemplateIndex = self.pageTemplates.index(t)
   742	                    return
   743	            raise ValueError("can't find template('%s')"%pt)
   744	        elif isinstance(pt,int):
   745	            if hasattr(self, '_nextPageTemplateCycle'): del self._nextPageTemplateCycle
   746	            self._nextPageTemplateIndex = pt
   747	        elif isSeq(pt):
   748	            #used for alternating left/right pages
   749	            #collect the refs to the template objects, complain if any are bad
   750	            c = PTCycle()
   751	            for ptn in pt:
   752	                found = 0
   753	                if ptn=='*':    #special case name used to short circuit the iteration
   754	                    c._restart = len(c)
   755	                    continue
   756	                for t in self.pageTemplates:
   757	                    if t.id == ptn:
   758	                        c.append(t)
   759	                        found = 1
   760	                if not found:
   761	                    raise ValueError("Cannot find page template called %s" % ptn)
   762	            if not c:
   763	                raise ValueError("No valid page templates in cycle")
   764	            elif c._restart>len(c):
   765	                raise ValueError("Invalid cycle restart position")
   766
   767	            #ensure we start on the first one
   768	            self._nextPageTemplateCycle = c
   769	        else:
   770	            raise TypeError("argument pt should be string or integer or list")

   840	    def handle_breakBefore(self, flowables):
   841	        '''preprocessing step to allow pageBreakBefore and frameBreakBefore attributes'''
   842	        first = flowables[0]
   843	        # if we insert a page break before, we'll process that, see it again,
   844	        # and go in an infinite loop.  So we need to set a flag on the object
   845	        # saying 'skip me'.  This should be unset on the next pass
   846	        if hasattr(first, '_skipMeNextTime'):
   847	            delattr(first, '_skipMeNextTime')
   848	            return
   849	        # this could all be made much quicker by putting the attributes
   850	        # in to the flowables with a defult value of 0
   851	        if hasattr(first,'pageBreakBefore') and first.pageBreakBefore == 1:
   852	            first._skipMeNextTime = 1
   853	            first.insert(0, PageBreak())
   854	            return
   855	        if hasattr(first,'style') and hasattr(first.style, 'pageBreakBefore') and first.style.pageBreakBefore == 1:
   856	            first._skipMeNextTime = 1
   857	            flowables.insert(0, PageBreak())
   858	            return
   859	        if hasattr(first,'frameBreakBefore') and first.frameBreakBefore == 1:
   860	            first._skipMeNextTime = 1
   861	            flowables.insert(0, FrameBreak())
   862	            return
   863	        if hasattr(first,'style') and hasattr(first.style, 'frameBreakBefore') and first.style.frameBreakBefore == 1:
   864	            first._skipMeNextTime = 1
   865	            flowables.insert(0, FrameBreak())
   866	            return
