   234	    """Enable calling a nested function (with a callable class) recursively.
   235
   236	    Adds the instance of the callable class representing the given
   237	    FuncDef to a register in the environment so that the function can
   238	    be called recursively. Note that this needs to be done only for
   239	    nested functions.
   240	    """
   241	    # First, set the attribute of the environment class so that GetAttr can be called on it.
   242	    prev_env = builder.fn_infos[-2].env_class
   243	    prev_env.attributes[fdef.name] = builder.type_to_rtype(fdef.type)

   246	        # If we are dealing with a generator class, then we need to first get the register
   247	        # holding the current environment class, and load the previous environment class from
   248	        # there.
   249	        prev_env_reg = builder.add(GetAttr(base.curr_env_reg, ENV_ATTR_NAME, -1))
   250	    else:
   251	        prev_env_reg = base.prev_env_reg

   254	    # current environment.
   255	    val = builder.add(GetAttr(prev_env_reg, fdef.name, -1))
   256	    target = builder.add_local_reg(fdef, object_rprimitive)
