    39	def split_blocks_at_uninits(
    40	    blocks: list[BasicBlock], pre_must_defined: AnalysisDict[Value]
    41	) -> list[BasicBlock]:
    42	    new_blocks: list[BasicBlock] = []
    43
    44	    init_registers = []
    45	    init_registers_set = set()
    46	    bitmap_registers: list[Register] = []  # Init status bitmaps
    47	    bitmap_backed: list[Register] = []  # These use bitmaps to track init status
    48
    49	    # First split blocks on ops that may raise.
    50	    for block in blocks:
    51	        ops = block.ops
    52	        block.ops = []
    53	        cur_block = block
    54	        new_blocks.append(cur_block)
    55
    56	        for i, op in enumerate(ops):
    57	            defined = pre_must_defined[block, i]
    58	            for src in op.unique_sources():
    59	                # If a register operand is not guaranteed to be
    60	                # initialized is an operand to something other than a
    61	                # check that it is defined, insert a check.
    62
    63	                # Note that for register operand in a LoadAddress op,
    64	                # we should be able to use it without initialization
    65	                # as we may need to use its address to update itself
    66	                if (
    67	                    isinstance(src, Register)
    68	                    and src not in defined
    69	                    and not (isinstance(op, Branch) and op.op == Branch.IS_ERROR)
    70	                    and not isinstance(op, LoadAddress)
    71	                ):
    72	                    if src not in init_registers_set:
    73	                        init_registers.append(src)
    74	                        init_registers_set.add(src)
    75
    76	                    # XXX: if src.name is empty, it should be a
    77	                    # temp... and it should be OK??
    78	                    if not src.name:
    79	                        continue
    80
    81	                    new_block, error_block = BasicBlock(), BasicBlock()
    82	                    new_block.error_handler = error_block.error_handler = cur_block.error_handler
    83	                    new_blocks += [error_block, new_block]
    84
    85	                    if not src.type.error_overlap:
    86	                        cur_block.ops.append(
    87	                            Branch(
    88	                                src,
    89	                                true_label=error_block,
    90	                                false_label=new_block,
    91	                                op=Branch.IS_ERROR,
    92	                                line=op.line,
    93	                            )
    94	                        )
    95	                    else:
    96	                        # We need to use bitmap for this one.
    97	                        check_for_uninit_using_bitmap(
    98	                            cur_block.ops,
    99	                            src,
   100	                            bitmap_registers,
   101	                            bitmap_backed,
   102	                            error_block,
   103	                            new_block,
   104	                            op.line,
   105	                        )
   106
   107	                    raise_std = RaiseStandardError(
   108	                        RaiseStandardError.UNBOUND_LOCAL_ERROR,
   109	                        f'local variable "{src.name}" referenced before assignment',
   110	                        op.line,
   111	                    )
   112	                    error_block.ops.append(raise_std)
   113	                    error_block.ops.append(Unreachable())
   114	                    cur_block = new_block
   115	            cur_block.ops.append(op)
   116
   117	    if bitmap_backed:
   118	        update_register_assignments_to_set_bitmap(new_blocks, bitmap_registers, bitmap_backed)
   119
   120	    if init_registers:
   121	        new_ops: list[Op] = []
   122	        for reg in init_registers:
   123	            err = LoadErrorValue(reg.type, undefines=True)
   124	            new_ops.append(err)
   125	            new_ops.append(Assign(reg, err))
   126	        for reg in bitmap_registers:
   127	            new_ops.append(Assign(reg, Integer(0, bitmap_rprimitive)))
   128	        new_blocks[0].ops[0:0] = new_ops
   129
   130	    return new_blocks

   133	def check_for_uninit_using_bitmap(
   134	    ops: list[Op],
   135	    src: Register,
   136	    bitmap_registers: list[Register],
   137	    bitmap_backed: list[Register],
   138	    error_block: BasicBlock,
   139	    ok_block: BasicBlock,
   140	    line: int,
   141	) -> None:
   142	    """Check if src is defined using a bitmap.
   143
   144	    Modifies ops, bitmap_registers and bitmap_backed.
   145	    """
   146	    if src not in bitmap_backed:
   147	        # Set up a new bitmap backed register.
   148	        bitmap_backed.append(src)
   149	        n = (len(bitmap_backed) - 1) // BITMAP_BITS
   150	        if len(bitmap_registers) <= n:
   151	            bitmap_registers.append(Register(bitmap_rprimitive, f"__locals_bitmap{n}"))
   152
   153	    index = bitmap_backed.index(src)
   154	    masked = IntOp(
   155	        bitmap_rprimitive,
   156	        bitmap_registers[index // BITMAP_BITS],
   157	        Integer(1 << (index & (BITMAP_BITS - 1)), bitmap_rprimitive),
   158	        IntOp.AND,
   159	        line,
   160	    )
   161	    ops.append(masked)
   162	    chk = ComparisonOp(masked, Integer(0, bitmap_rprimitive), ComparisonOp.EQ)
   163	    ops.append(chk)
   164	    ops.append(Branch(chk, error_block, ok_block, Branch.BOOL))

   167	def update_register_assignments_to_set_bitmap(
   168	    blocks: list[BasicBlock], bitmap_registers: list[Register], bitmap_backed: list[Register]
   169	) -> None:
   170	    """Update some assignments to registers to also set a bit in a bitmap.
   171
   172	    The bitmaps are used to track if a local variable has been assigned to.
   173
   174	    Modifies blocks.
   175	    """
   176	    for block in blocks:
   177	        if any(isinstance(op, Assign) and op.dest in bitmap_backed for op in block.ops):
   178	            new_ops: list[Op] = []
   179	            for op in block.ops:
   180	                if isinstance(op, Assign) and op.dest in bitmap_backed:
   181	                    index = bitmap_backed.index(op.dest)
   182	                    new_ops.append(op)
   183	                    reg = bitmap_registers[index // BITMAP_BITS]
   184	                    new = IntOp(
   185	                        bitmap_rprimitive,
   186	                        reg,
   187	                        Integer(1 << (index & (BITMAP_BITS - 1)), bitmap_rprimitive),
   188	                        IntOp.OR,
   189	                        op.line,
   190	                    )
   191	                    new_ops.append(new)
   192	                    new_ops.append(Assign(reg, new))
   193	                else:
   194	                    new_ops.append(op)
   195	            block.ops = new_ops

    68	def to_utf8(text):
    69	    """Convert given text to UTF-8 encoding (as far as possible)."""
    70	    if not text or is_PY3:
    71	        return text
    72
    73	    try:           # unicode or pure ascii

    75	    except UnicodeDecodeError:
    76	        try:       # successful UTF-8 decode means it's pretty sure UTF-8
    77	            text.decode("utf8")
    78	            return text

    81	                return text.decode("cp1252").encode("utf8")

    83	                pass
