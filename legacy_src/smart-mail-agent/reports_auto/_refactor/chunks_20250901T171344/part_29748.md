  4616	        )

  4641	        for t in values + [upper_bound, default]:

  4643	                t, self.options, self.is_typeshed_stub_file, self.msg, context=s
  4644	            )

  4647	        # possible type, so set the upper bound to Any to prevent that

  4650	            upper_bound = AnyType(TypeOfAny.implementation_artifact)

  4652	        # Yes, it's a valid type variable definition! Add it to the symbol table.
  4653	        if not call.analyzed:
  4654	            type_var = TypeVarExpr(

  4660	        else:

  4662	            updated = (
  4663	                values != call.analyzed.values
  4664	                or upper_bound != call.analyzed.upper_bound
  4665	                or default != call.analyzed.default
  4666	            )
  4667	            call.analyzed.upper_bound = upper_bound
  4668	            call.analyzed.values = values
  4669	            call.analyzed.default = default
  4670	        if any(has_placeholder(v) for v in values):
  4671	            self.process_placeholder(None, "TypeVar values", s, force_progress=updated)
  4672	        elif has_placeholder(upper_bound):
  4673	            self.process_placeholder(None, "TypeVar upper bound", s, force_progress=updated)
  4674	        elif has_placeholder(default):
  4675	            self.process_placeholder(None, "TypeVar default", s, force_progress=updated)
  4676
  4677	        self.add_symbol(name, call.analyzed, s)

  4751	    def process_typevar_parameters(
  4752	        self,
  4753	        args: list[Expression],
  4754	        names: list[str | None],
  4755	        kinds: list[ArgKind],
  4756	        num_values: int,
  4757	        context: Context,
  4758	    ) -> tuple[int, Type, Type] | None:
  4759	        has_values = num_values > 0
  4760	        covariant = False
  4761	        contravariant = False
  4762	        upper_bound: Type = self.object_type()
  4763	        default: Type = AnyType(TypeOfAny.from_omitted_generics)
  4764	        for param_value, param_name, param_kind in zip(args, names, kinds):
  4765	            if not param_kind.is_named():
  4766	                self.fail(message_registry.TYPEVAR_UNEXPECTED_ARGUMENT, context)
  4767	                return None
  4768	            if param_name == "covariant":
  4769	                if isinstance(param_value, NameExpr) and param_value.name in ("True", "False"):
  4770	                    covariant = param_value.name == "True"
  4771	                else:
  4772	                    self.fail(message_registry.TYPEVAR_VARIANCE_DEF.format("covariant"), context)
  4773	                    return None
  4774	            elif param_name == "contravariant":
  4775	                if isinstance(param_value, NameExpr) and param_value.name in ("True", "False"):
  4776	                    contravariant = param_value.name == "True"
  4777	                else:
  4778	                    self.fail(
  4779	                        message_registry.TYPEVAR_VARIANCE_DEF.format("contravariant"), context
  4780	                    )
  4781	                    return None
  4782	            elif param_name == "bound":
  4783	                if has_values:
  4784	                    self.fail("TypeVar cannot have both values and an upper bound", context)
  4785	                    return None
  4786	                tv_arg = self.get_typevarlike_argument("TypeVar", param_name, param_value, context)
  4787	                if tv_arg is None:
  4788	                    return None
  4789	                upper_bound = tv_arg
  4790	            elif param_name == "default":
  4791	                tv_arg = self.get_typevarlike_argument(
  4792	                    "TypeVar", param_name, param_value, context, allow_unbound_tvars=True
  4793	                )
  4794	                default = tv_arg or AnyType(TypeOfAny.from_error)
  4795	            elif param_name == "values":
  4796	                # Probably using obsolete syntax with values=(...). Explain the current syntax.
  4797	                self.fail('TypeVar "values" argument not supported', context)
  4798	                self.fail(
  4799	                    "Use TypeVar('T', t, ...) instead of TypeVar('T', values=(t, ...))", context
  4800	                )
  4801	                return None
  4802	            else:
  4803	                self.fail(
  4804	                    f'{message_registry.TYPEVAR_UNEXPECTED_ARGUMENT}: "{param_name}"', context
  4805	                )
  4806	                return None
  4807
  4808	        if covariant and contravariant:
  4809	            self.fail("TypeVar cannot be both covariant and contravariant", context)
  4810	            return None
  4811	        elif num_values == 1:
  4812	            self.fail(message_registry.TYPE_VAR_TOO_FEW_CONSTRAINED_TYPES, context)
  4813	            return None
  4814	        elif covariant:
  4815	            variance = COVARIANT
  4816	        elif contravariant:
  4817	            variance = CONTRAVARIANT
  4818	        else:
  4819	            variance = INVARIANT
  4820	        return variance, upper_bound, default
