option_name, value)
   543
   544	    def load(self) -> None:
   545	        """Load previously-collected coverage data from the data file."""
   546	        self._init()
   547	        if self._collector is not None:
   548	            self._collector.reset()
   549	        should_skip = self.config.parallel and not os.path.exists(self.config.data_file)
   550	        if not should_skip:
   551	            self._init_data(suffix=None)
   552	        self._post_init()
   553	        if not should_skip:
   554	            assert self._data is not None
   555	            self._data.read()

   557	    def _init_for_start(self) -> None:
   558	        """Initialization for start()"""
   559	        # Construct the collector.
   560	        concurrency: list[str] = self.config.concurrency or []
   561	        if "multiprocessing" in concurrency:
   562	            if self.config.config_file is None:
   563	                raise ConfigError("multiprocessing requires a configuration file")
   564	            patch_multiprocessing(rcfile=self.config.config_file)
   565
   566	        dycon = self.config.dynamic_context
   567	        if not dycon or dycon == "none":
   568	            context_switchers = []
   569	        elif dycon == "test_function":
   570	            context_switchers = [should_start_context_test_function]
   571	        else:
   572	            raise ConfigError(f"Don't understand dynamic_context setting: {dycon!r}")
   573
   574	        context_switchers.extend(
   575	            plugin.dynamic_context for plugin in self._plugins.context_switchers
   576	        )
   577
   578	        should_start_context = combine_context_switchers(context_switchers)
   579
   580	        self._core = Core(
   581	            warn=self._warn,
   582	            config=self.config,
   583	            dynamic_contexts=(should_start_context is not None),
   584	            metacov=self._metacov,
   585	        )
   586	        self._collector = Collector(
   587	            core=self._core,
   588	            should_trace=self._should_trace,
   589	            check_include=self._check_include_omit_etc,
   590	            should_start_context=should_start_context,
   591	            file_mapper=self._file_mapper,
   592	            branch=self.config.branch,
   593	            warn=self._warn,
   594	            concurrency=concurrency,
   595	        )
   596
   597	        suffix = self._data_suffix_specified
   598	        if suffix:
   599	            if not isinstance(suffix, str):
   600	                # if data_suffix=True, use .machinename.pid.random
   601	                suffix = True
   602	        elif self.config.parallel:
   603	            if suffix is None:
   604	                suffix = True
   605	            elif not isinstance(suffix, str):
   606	                suffix = bool(suffix)
   607	        else:
   608	            suffix = None
   609
   610	        self._init_data(suffix)
   611
   612	        assert self._data is not None
   613	        self._collector.use_data(self._data, self.config.context)
   614
   615	        # Early warning if we aren't going to be able to support plugins.
   616	        if self._plugins.file_tracers and not self._core.supports_plugins:
   617	            self._warn(
   618	                "Plugin file tracers ({}) aren't supported with {}".format(
   619	                    ", ".join(
   620	                        plugin._coverage_plugin_name
   621	                            for plugin in self._plugins.file_tracers
   622	                    ),
   623	                    self._collector.tracer_name(),
   624	                ),
   625	            )
   626	            for plugin in self._plugins.file_tracers:
   627	                plugin._coverage_enabled = False
   628
   629	        # Create the file classifying substructure.
   630	        self._inorout = InOrOut(
   631	            config=self.config,
   632	            warn=self._warn,
   633	            debug=(self._debug if self._debug.should("trace") else None),
   634	            include_namespace_packages=self.config.include_namespace_packages,
   635	        )
   636	        self._inorout.plugins = self._plugins
   637	        self._inorout.disp_class = self._core.file_disposition_class
   638
   639	        # It's useful to write debug info after initing for start.
   640	        self._should_write_debug = True
   641
   642	        # Register our clean-up handlers.
   643	        atexit.register(self._atexit)
   644	        if self.config.sigterm:
   645	            is_main = (threading.current_thread() == threading.main_thread())
   646	            if is_main and not env.WINDOWS:
   647	                # The Python docs seem to imply that SIGTERM works uniformly even
   648	                # on Windows, but that's not my experience, and this agrees:
   649	                # https://stackoverflow.com/questions/35772001/x/35792192#35792192
   650	                self._old_sigterm = signal.signal(      # type: ignore[assignment]
   651	                    signal.SIGTERM, self._on_sigterm,
   652	                )
   653
   654	    def _init_data(self, suffix: str | bool | None) -> None:
   655	        """Create a data file if we don't have one yet."""
   656	        if self._data is None:
   657	            # Create the data file.  We do this at construction time so that the
   658	            # data file will be written into the directory where the process
   659	            # started rather than wherever the process eventually chdir'd to.
   660	            ensure_dir_for_file(self.config.data_file)
   661	            self._data = CoverageData(
   662	                basename=self.config.data_file,
   663	                suffix=suffix,
   664	                warn=self._warn,
   665	                debug=self._debug,
   666	                no_disk=self._no_disk,
   667	            )
   668	            self._data_to_close.append(self._data)

   670	    def start(self) -> None:
   671	        """Start measuring code coverage.
   672
   673	        Coverage measurement is only collected in functions called after
   674	        :meth:`start` is invoked.  Statements in the same scope as
   675	        :meth:`start` won't be measured.
   676
   677	        Once you invoke :meth:`start`, you must also call :meth:`stop`
   678	        eventually, or your process might not shut down cleanly.
   679
   680	        The :meth:`collect` method is a context manager to handle both
   681	        starting and stopping collection.
   682
   683	        """
   684	        self._init()
   685	        if not self._inited_for_start:
   686	            self._inited_for_start = True
   687	            self._init_for_start()
   688	        self._post_init()
   689
   690	        assert self._collector is not None
   691	        assert self._inorout is not None
   692
   693	        # Issue warnings for possible problems.
   694	        self._inorout.warn_conflicting_settings()
   695
   696	        # See if we think some code that would eventually be measured has
   697	        # already been imported.
   698	        if self._warn_preimported_source:
   699	            self._inorout.warn_already_imported_files()
   700
   701	        if self._auto_load:
   702	            self.load()
   703
   704	        apply_patches(self, self.config, self._debug, make_pth_file=self._make_pth_file)
   705
   706	        self._collector.start()
   707	        self._started = True
   708	        self._instances.append(self)
   709
   710	    def stop(self) -> None:
   711	        """Stop measuring code coverage."""
   712	        if self._instances:
   713	            if self._instances[-1] is self:
   714	                self._instances.pop()
   715	        if self._started:
   716	            assert self._collector is not None
   717	            self._collector.stop()
   718	        self._started = False
   719
   720	    @contextlib.contextmanager
   721	    def collect(self) -> Iterator[None]:
   722	        """A context manager to start/stop coverage measurement collection.
   723
   724	        .. versionadded:: 7.3
   725
   726	        """
   727	        self.start()
   728	        try:
   729	            yield
   730	        finally:
   731	            self.stop()     # pragma: nested

   733	    def _atexit(self, event: str = "atexit") -> None:
   734	        """Clean up on process shutdown."""
   735	        if self._debug.should("process"):
   736	            self._debug.write(f"{event}: pid: {os.getpid()}, instance: {self!r}")
   737	        if self._started:
   738	            self.stop()
   739	        if self._auto_save or event == "sigterm":
   740	            self.save()
   741	        for d in self._data_to_close:
   742	            d.close(force=True)
   743
   744	    def _on_sigterm(self, signum_unused: int, frame_unused: FrameType | None) -> None:
   745	        """A handler for signal.SIGTERM."""
   746	        self._atexit("sigterm")
   747	        # Statements after here won't be seen by metacov because we just wrote
   748	        # the data, and are about to kill the process.
   749	        signal.signal(signal.SIGTERM, self._old_sigterm)    # pragma: not covered
   750	        os.kill(os.getpid(), signal.SIGTERM)                # pragma: not covered

   797	    def exclude(self, regex: str, which: str = "exclude") -> None:
   798	        """Exclude source lines from execution consideration.
   799
   800	        A number of lists of regular expressions are maintained.  Each list
   801	        selects lines that are treated differently during reporting.
   802
   803	        `which` determines which list is modified.  The "exclude" list selects
   804	        lines that are not considered executable at all.  The "partial" list
   805	        indicates lines with branches that are not taken.
   806
   807	        `regex` is a regular expression.  The regex is added to the specified
   808	        list.  If any of the regexes in the list is found in a line, the line
   809	        is marked for special treatment during reporting.
   810
   811	        """
   812	        self._init()
   813	        excl_list = getattr(self.config, which + "_list")
   814	        excl_list.append(regex)
   815	        self._exclude_regex_stale()
   816
   817	    def _exclude_regex_stale(self) -> None:
   818	        """Drop all the compiled exclusion regexes, a list was modified."""
   819	        self._exclude_re.clear()

   821	    def _exclude_regex(self, which: str) -> str:
   822	        """Return a regex string for the given exclusion list."""
   823	        if which not in self._exclude_re:
   824	            excl_list = getattr(self.config, which + "_list")
   825	            self._exclude_re[which] = join_regex(excl_list)
   826	        return self._exclude_re[which]

   829	        """Return a list of excluded regex strings.

   855	    def combine(
   856	        self,
   857	        data_paths: Iterable[str] | None = None,
   858	        strict: bool = False,
   859	        keep: bool = False,
   860	    ) -> None:
   861	        """Combine together a number of similarly-named coverage data files.
   862
   863	        All coverage data files whose name starts with `data_file` (from the
   864	        coverage() constructor) will be read, and combined together into the
   865	        current measurements.
   866
   867	        `data_paths` is a list of files or directories from which data should
   868	        be combined. If no list is passed, then the data files from the
   869	        directory indicated by the current data file (probably the current
   870	        directory) will be combined.
   871
   872	        If `strict` is true, then it is an error to attempt to combine when
   873	        there are no data files to combine.
   874
   875	        If `keep` is true, then original input data files won't be deleted.
   876
   877	        .. versionadded:: 4.0
   878	            The `data_paths` parameter.
   879
   880	        .. versionadded:: 4.3
   881	            The `strict` parameter.
   882
   883	        .. versionadded: 5.5
   884	            The `keep` parameter.

   886	        self._init()
   887	        self._init_data(suffix=None)
   888	        self._post_init()
   889	        self.get_data()
   890
   891	        assert self._data is not None
   892	        combine_parallel_data(
   893	            self._data,
   894	            aliases=self._make_aliases(),
   895	            data_paths=data_paths,
   896	            strict=strict,
   897	            keep=keep,
   898	            message=self._message,
   899	        )

   901	    def get_data(self) -> CoverageData:
   902	        """Get the collected data.
   903
   904	        Also warn about various problems collecting data.
   905
   906	        Returns a :class:`coverage.CoverageData`, the collected coverage data.
   907
   908	        .. versionadded:: 4.0
   909
   910	        """
   911	        self._init()
   912	        self._init_data(suffix=None)
   913	        self._post_init()
   914
   915	        if self._collector is not None:
   916	            for plugin in self._plugins:
   917	                if not plugin._coverage_enabled:
   918	                    self._collector.plugin_was_disabled(plugin)
   919
   920	            if self._collector.flush_data():
   921	                self._post_save_work()
   922
   923	        assert self._data is not None
   924	        return self._data

   926	    def _post_save_work(self) -> None:
   927	        """After saving data, look for warnings, post-work, etc.
   928
   929	        Warn about things that should have happened but didn't.
   930	        Look for un-executed files.
   931
   932	        """
   933	        assert self._data is not None
   934	        assert self._inorout is not None
   935
   936	        # If there are still entries in the source_pkgs_unmatched list,
   937	        # then we never encountered those packages.
   938	        if self._warn_unimported_source:
   939	            self._inorout.warn_unimported_source()
   940
   941	        # Find out if we got any data.
   942	        if not self._data and self._warn_no_data:
   943	            self._warn("No data was collected.", slug="no-data-collected")
   944
   945	        # Touch all the files that could have executed, so that we can
   946	        # mark completely un-executed files as 0% covered.
   947	        file_paths = collections.defaultdict(list)
   948	        for file_path, plugin_name in self._inorout.find_possibly_unexecuted_files():
   949	            file_path = self._file_mapper(file_path)
   950	            file_paths[plugin_name].append(file_path)
   951	        for plugin_name, paths in file_paths.items():
   952	            self._data.touch_files(paths, plugin_name)

   960	    def analysis2(
   961	        self,
   962	        morf: TMorf,
   963	    ) -> tuple[str, list[TLineNo], list[TLineNo], list[TLineNo], str]:
   964	        """Analyze a module.
   965
   966	        `morf` is a module or a file name.  It will be analyzed to determine
   967	        its coverage statistics.  The return value is a 5-tuple:
   968
   969	        * The file name for the module.
   970	        * A list of line numbers of executable statements.
   971	        * A list of line numbers of excluded statements.
   972	        * A list of line numbers of statements not run (missing from
   973	          execution).
   974	        * A readable formatted string of the missing line numbers.
   975
   976	        The analysis uses the source file itself and the current measured
   977	        coverage data.
   978
   979	        """
   980	        analysis = self._analyze(morf)
   981	        return (
   982	            analysis.filename,
   983	            sorted(analysis.statements),
   984	            sorted(analysis.excluded),
   985	            sorted(analysis.missing),
   986	            analysis.missing_formatted(),
   987	        )
