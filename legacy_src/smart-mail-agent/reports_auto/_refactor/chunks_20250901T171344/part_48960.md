     7	def moduleClasses(mod):
     8	    def P(obj, m=mod.__name__, CT=type):
     9	        return (type(obj)==CT and obj.__module__==m)
    10	    try:
    11	        return inspect.getmembers(mod, P)[0][1]
    12	    except:
    13	        return None
    14
    15	def getclass(f):
    16	    return moduleClasses(__import__(f))
    17
    18	def run(format, VERBOSE=0):
    19	    formats = format.split( ',')
    20	    for i in range(0, len(formats)):
    21	        formats[i] == formats[i].strip().lower()
    22	    allfiles = glob.glob('*.py')

    24	    for fn in allfiles:

    30	                for fmt in formats:
