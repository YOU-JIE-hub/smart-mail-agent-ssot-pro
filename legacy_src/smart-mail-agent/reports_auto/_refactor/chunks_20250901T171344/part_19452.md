  1020			"""Protect sequence unpacking of targets in a 'for loop'.
  1021

  1023			For example "for a, b in it"

  1030		def __rl_unpack_sequence__(self, it, spec, _getiter_):
  1031			"""Protect nested sequence unpacking.
  1032
  1033			Protect the unpacking of 'it' by wrapping it with '_getiter_'.
  1034			Furthermore for each child element, defined by spec,
  1035			__rl_unpack_sequence__ is called again.
  1036
  1037			Have a look at transformer.py 'gen_unpack_spec' for a more detailed
  1038			explanation.
  1039			"""
  1040			# Do the guarded unpacking of the sequence.
  1041			ret = list(self.__rl__getiter__(it))
  1042
  1043			# If the sequence is shorter then expected the interpreter will raise
  1044			# 'ValueError: need more than X value to unpack' anyway
  1045			# => No childs are unpacked => nothing to protect.
  1046			if len(ret) < spec['min_len']:
  1047				return ret
  1048
  1049			# For all child elements do the guarded unpacking again.
  1050			for (idx, child_spec) in spec['childs']:
  1051				ret[idx] = self.__rl_unpack_sequence__(ret[idx], child_spec, _getiter_)
  1052			return ret
