   516	def _shortened(word):
   517	    dpos = word.find('$')
   518	    return '|'.join(word[:dpos] + word[dpos+1:i] + r'\b'
   519	                    for i in range(len(word), dpos, -1))

   523	    return '|'.join(map(_shortened, words))
