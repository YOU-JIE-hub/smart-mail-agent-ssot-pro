    60	# [Note Literals and literal_hash]
    61	# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
    62	#
    63	# Mypy uses the term "literal" to refer to any expression built out of
    64	# the following:
    65	#
    66	# * Plain literal expressions, like `1` (integer, float, string, etc.)
    67	#
    68	# * Compound literal expressions, like `(lit1, lit2)` (list, dict,
    69	#   set, or tuple)
    70	#
    71	# * Operator expressions, like `lit1 + lit2`
    72	#
    73	# * Variable references, like `x`
    74	#
    75	# * Member references, like `lit.m`
    76	#
    77	# * Index expressions, like `lit[0]`
    78	#
    79	# A typical "literal" looks like `x[(i,j+1)].m`.
    80	#
    81	# An expression that is a literal has a `literal_hash`, with the
    82	# following properties.
    83	#
    84	# * `literal_hash` is a Key: a tuple containing basic data types and
    85	#   possibly other Keys. So it can be used as a key in a dictionary
    86	#   that will be compared by value (as opposed to the Node itself,
    87	#   which is compared by identity).
    88	#
    89	# * Two expressions have equal `literal_hash`es if and only if they
    90	#   are syntactically equal expressions. (NB: Actually, we also
    91	#   identify as equal expressions like `3` and `3.0`; is this a good
    92	#   idea?)
    93	#
    94	# * The elements of `literal_hash` that are tuples are exactly the
    95	#   subexpressions of the original expression (e.g. the base and index
    96	#   of an index expression, or the operands of an operator expression).
    97

   117	def literal(e: Expression) -> int:
   118	    """Return the literal kind for an expression."""
   119
   120	    if isinstance(e, ComparisonExpr):
   121	        return min(literal(o) for o in e.operands)
   122
   123	    elif isinstance(e, OpExpr):
   124	        return min(literal(e.left), literal(e.right))
   125
   126	    elif isinstance(e, (MemberExpr, UnaryExpr, StarExpr)):
   127	        return literal(e.expr)
   128
   129	    elif isinstance(e, AssignmentExpr):
   130	        return literal(e.target)
   131
   132	    elif isinstance(e, IndexExpr):
   133	        if literal(e.index) == LITERAL_YES:
   134	            return literal(e.base)
   135	        else:
   136	            return LITERAL_NO
   137
   138	    elif isinstance(e, NameExpr):
   139	        if isinstance(e.node, Var) and e.node.is_final and e.node.final_value is not None:
   140	            return LITERAL_YES
   141	        return LITERAL_TYPE
   142
   143	    if isinstance(e, (IntExpr, FloatExpr, ComplexExpr, StrExpr, BytesExpr)):
   144	        return LITERAL_YES
   145
   146	    if literal_hash(e):
   147	        return LITERAL_YES
   148
   149	    return LITERAL_NO
