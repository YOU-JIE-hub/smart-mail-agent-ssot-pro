   306	    def should_trace(self, filename: str, frame: FrameType | None = None) -> TFileDisposition:
   307	        """Decide whether to trace execution in `filename`, with a reason.
   308
   309	        This function is called from the trace function.  As each new file name
   310	        is encountered, this function determines whether it is traced or not.
   311
   312	        Returns a FileDisposition object.
   313
   314	        """
   315	        original_filename = filename
   316	        disp = disposition_init(self.disp_class, filename)
   317
   318	        def nope(disp: TFileDisposition, reason: str) -> TFileDisposition:
   319	            """Simple helper to make it easy to return NO."""
   320	            disp.trace = False
   321	            disp.reason = reason
   322	            return disp
   323
   324	        if original_filename.startswith("<"):
   325	            return nope(disp, "original file name is not real")
   326
   327	        if frame is not None:
   328	            # Compiled Python files have two file names: frame.f_code.co_filename is
   329	            # the file name at the time the .pyc was compiled.  The second name is
   330	            # __file__, which is where the .pyc was actually loaded from.  Since
   331	            # .pyc files can be moved after compilation (for example, by being
   332	            # installed), we look for __file__ in the frame and prefer it to the
   333	            # co_filename value.
   334	            dunder_file = frame.f_globals and frame.f_globals.get("__file__")
   335	            if dunder_file:
   336	                # Danger: __file__ can (rarely?) be of type Path.
   337	                filename = source_for_file(str(dunder_file))
   338	                if original_filename and not original_filename.startswith("<"):
   339	                    orig = os.path.basename(original_filename)
   340	                    if orig != os.path.basename(filename):
   341	                        # Files shouldn't be renamed when moved. This happens when
   342	                        # exec'ing code.  If it seems like something is wrong with
   343	                        # the frame's file name, then just use the original.
   344	                        filename = original_filename
   345
   346	        if not filename:
   347	            # Empty string is pretty useless.
   348	            return nope(disp, "empty string isn't a file name")
   349
   350	        if filename.startswith("memory:"):
   351	            return nope(disp, "memory isn't traceable")
   352
   353	        if filename.startswith("<"):
   354	            # Lots of non-file execution is represented with artificial
   355	            # file names like "<string>", "<doctest readme.txt[0]>", or
   356	            # "<exec_function>".  Don't ever trace these executions, since we
   357	            # can't do anything with the data later anyway.
   358	            return nope(disp, "file name is not real")
   359
   360	        canonical = canonical_filename(filename)
   361	        disp.canonical_filename = canonical
   362
   363	        # Try the plugins, see if they have an opinion about the file.
   364	        plugin = None
   365	        for plugin in self.plugins.file_tracers:
   366	            if not plugin._coverage_enabled:
   367	                continue
   368
   369	            try:
   370	                file_tracer = plugin.file_tracer(canonical)
   371	                if file_tracer is not None:
   372	                    file_tracer._coverage_plugin = plugin
   373	                    disp.trace = True
   374	                    disp.file_tracer = file_tracer
   375	                    if file_tracer.has_dynamic_source_filename():
   376	                        disp.has_dynamic_filename = True
   377	                    else:
   378	                        disp.source_filename = canonical_filename(
   379	                            file_tracer.source_filename(),
   380	                        )
   381	                    break
   382	            except Exception:
   383	                plugin_name = plugin._coverage_plugin_name
   384	                tb = traceback.format_exc()
   385	                self.warn(f"Disabling plug-in {plugin_name!r} due to an exception:\n{tb}")
   386	                plugin._coverage_enabled = False
   387	                continue
   388	        else:
   389	            # No plugin wanted it: it's Python.
   390	            disp.trace = True
   391	            disp.source_filename = canonical
   392
   393	        if not disp.has_dynamic_filename:
   394	            if not disp.source_filename:
   395	                raise PluginError(
   396	                    f"Plugin {plugin!r} didn't set source_filename for '{disp.original_filename}'",
   397	                )
   398	            reason = self.check_include_omit_etc(disp.source_filename, frame)
   399	            if reason:
   400	                nope(disp, reason)
   401
   402	        return disp

   404	    def check_include_omit_etc(self, filename: str, frame: FrameType | None) -> str | None:
   405	        """Check a file name against the include, omit, etc, rules.
   406
   407	        Returns a string or None.  String means, don't trace, and is the reason
   408	        why.  None means no reason found to not trace.
   409
   410	        """
   411	        modulename = name_for_module(filename, frame)
   412
   413	        # If the user specified source or include, then that's authoritative
   414	        # about the outer bound of what to measure and we don't have to apply
   415	        # any canned exclusions. If they didn't, then we have to exclude the
   416	        # stdlib and coverage.py directories.
   417	        if self.source_match or self.source_pkgs_match:
   418	            extra = ""
   419	            ok = False
   420	            if self.source_pkgs_match:
   421	                if isinstance(modulename, str) and self.source_pkgs_match.match(modulename):
   422	                    ok = True
   423	                    if modulename in self.source_pkgs_unmatched:
   424	                        self.source_pkgs_unmatched.remove(modulename)
   425	                else:
   426	                    extra = f"module {modulename!r} "
   427	            if not ok and self.source_match:
   428	                if self.source_match.match(filename):
   429	                    ok = True
   430	            if not ok:
   431	                return extra + "falls outside the --source spec"
   432	            if self.third_match.match(filename) and not self.source_in_third_match.match(filename):
   433	                return "inside --source, but is third-party"
   434	        elif self.include_match:
   435	            if not self.include_match.match(filename):
   436	                return "falls outside the --include trees"
   437	        else:
   438	            # We exclude the coverage.py code itself, since a little of it
   439	            # will be measured otherwise.
   440	            if self.cover_match.match(filename):
   441	                return "is part of coverage.py"
   442
   443	            # If we aren't supposed to trace installed code, then check if this
   444	            # is near the Python standard library and skip it if so.
   445	            if self.pylib_match and self.pylib_match.match(filename):
   446	                return "is in the stdlib"
   447
   448	            # Exclude anything in the third-party installation areas.
   449	            if self.third_match.match(filename):
   450	                return "is a third-party module"
   451
   452	        # Check the file against the omit pattern.
   453	        if self.omit_match and self.omit_match.match(filename):
   454	            return "is inside an --omit pattern"
   455
   456	        # No point tracing a file we can't later write to SQLite.
   457	        try:
   458	            filename.encode("utf-8")
   459	        except UnicodeEncodeError:
   460	            return "non-encodable filename"
   461
   462	        # No reason found to skip this file.
   463	        return None

   471	    def warn_already_imported_files(self) -> None:
   472	        """Warn if files have already been imported that we will be measuring."""
   473	        if self.include or self.source_dirs or self.source_pkgs:
   474	            warned = set()
   475	            for mod in list(sys.modules.values()):
   476	                filename = getattr(mod, "__file__", None)
   477	                if filename is None:
   478	                    continue
   479	                if filename in warned:
   480	                    continue
   481
   482	                if len(getattr(mod, "__path__", ())) > 1:
   483	                    # A namespace package, which confuses this code, so ignore it.
   484	                    continue
   485
   486	                disp = self.should_trace(filename)
   487	                if disp.has_dynamic_filename:
   488	                    # A plugin with dynamic filenames: the Python file
   489	                    # shouldn't cause a warning, since it won't be the subject
   490	                    # of tracing anyway.
   491	                    continue
   492	                if disp.trace:
   493	                    msg = f"Already imported a file that will be measured: {filename}"
   494	                    self.warn(msg, slug="already-imported")
   495	                    warned.add(filename)
   496	                elif self.debug and self.debug.should("trace"):
   497	                    self.debug.write(
   498	                        "Didn't trace already imported file {!r}: {}".format(
   499	                            disp.original_filename, disp.reason,
   500	                        ),
   501	                    )

   508	    def _warn_about_unmeasured_code(self, pkg: str) -> None:
   509	        """Warn about a package or module that we never traced.
   510
   511	        `pkg` is a string, the name of the package or module.
   512
   513	        """
   514	        mod = sys.modules.get(pkg)
   515	        if mod is None:
   516	            self.warn(f"Module {pkg} was never imported.", slug="module-not-imported")
   517	            return
   518
   519	        if module_is_namespace(mod):
   520	            # A namespace package. It's OK for this not to have been traced,
   521	            # since there is no code directly in it.
   522	            return
   523
   524	        if not module_has_file(mod):
   525	            self.warn(f"Module {pkg} has no Python source.", slug="module-not-python")
   526	            return
   527
   528	        # The module was in sys.modules, and seems like a module with code, but
   529	        # we never measured it. I guess that means it was imported before
   530	        # coverage even started.
   531	        msg = f"Module {pkg} was previously imported, but not measured"
   532	        self.warn(msg, slug="module-not-measured")
