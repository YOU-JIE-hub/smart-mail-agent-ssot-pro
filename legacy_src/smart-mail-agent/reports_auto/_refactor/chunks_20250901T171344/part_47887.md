  5326	    def visit_with_stmt(self, s: WithStmt) -> None:
  5327	        exceptions_maybe_suppressed = False
  5328	        for expr, target in zip(s.expr, s.target):
  5329	            if s.is_async:
  5330	                exit_ret_type = self.check_async_with_item(expr, target, s.unanalyzed_type is None)
  5331	            else:
  5332	                exit_ret_type = self.check_with_item(expr, target, s.unanalyzed_type is None)
  5333
  5334	            # Based on the return type, determine if this context manager 'swallows'
  5335	            # exceptions or not. We determine this using a heuristic based on the
  5336	            # return type of the __exit__ method -- see the discussion in
  5337	            # https://github.com/python/mypy/issues/7214 and the section about context managers
  5338	            # in https://github.com/python/typeshed/blob/main/CONTRIBUTING.md#conventions
  5339	            # for more details.

  5341	            exit_ret_type = get_proper_type(exit_ret_type)
  5342	            if is_literal_type(exit_ret_type, "builtins.bool", False):

  5346	                isinstance(exit_ret_type, Instance)
  5347	                and exit_ret_type.type.fullname == "builtins.bool"
  5348	                and state.strict_optional
  5349	            ):

  5351	                # could actually be an Optional[bool].
  5352	                exceptions_maybe_suppressed = True

  5355	            # Treat this 'with' block in the same way we'd treat a 'try: BODY; except: pass'
  5356	            # block. This means control flow can continue after the 'with' even if the 'with'
  5357	            # block immediately returns.
  5358	            with self.binder.frame_context(can_skip=True, try_frame=True):

  5360	        else:
