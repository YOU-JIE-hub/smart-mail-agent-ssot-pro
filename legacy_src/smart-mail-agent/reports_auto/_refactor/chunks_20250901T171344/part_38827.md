     7	    allowedTypes = strTypes if isinstance(strTypes, tuple) else (strTypes,)

     9	    for x in pattern_sequence:

    11	            if len(x)!=1:

    16	class PDFPattern(PDFObject):
    17	    __RefOnly__ = 1
    18	    def __init__(self, pattern_sequence, **keywordargs):
    19	        """
    20	        Description of a kind of PDF object using a pattern.
    21
    22	        Pattern sequence should contain strings, singletons of form [string] or
    23	        PDFPatternIf objects.
    24	        Strings are literal strings to be used in the object.
    25	        Singletons are names of keyword arguments to include.
    26	        PDFpatternIf objects allow some conditionality.
    27	        Keyword arguments can be non-instances which are substituted directly in string conversion,
    28	        or they can be object instances in which case they should be pdfdoc.* style
    29	        objects with a x.format(doc) method.
    30	        Keyword arguments may be set on initialization or subsequently using __setitem__, before format.
    31	        "constant object" instances can also be inserted in the patterns.
    32	        """
    33	        _patternSequenceCheck(pattern_sequence)
    34	        self.pattern = pattern_sequence
    35	        self.arguments = keywordargs
    36
    37	    def __setitem__(self, item, value):
    38	        self.arguments[item] = value
    39
    40	    def __getitem__(self, item):
    41	        return self.arguments[item]
    42
    43	    def eval(self,L):
    44	        arguments = self.arguments
    45	        document = self.__document
    46	        for x in L:
    47	            if isinstance(x,strTypes):
    48	                yield pdfdocEnc(x)
    49	            elif isinstance(x,PDFObject):
    50	                yield x.format(document)
    51	            elif isinstance(x,PDFPatternIf):
    52	                result = list(self.eval(x.cond))
    53	                cond = result and result[0]
    54	                for z in self.eval(x.thenPart if cond else x.elsePart):
    55	                    yield z
    56	            else:
    57	                name = x[0]
    58	                value = arguments.get(name, None)
    59	                if value is None:
    60	                    raise ValueError("%s value not defined" % ascii(name))
    61	                if isinstance(value,PDFObject):
    62	                    yield format(value,document)
    63	                elif isinstance(value,strTypes):
    64	                    yield pdfdocEnc(value)
    65	                else:
    66	                    yield pdfdocEnc(str(value))
    67
    68	    def format(self, document):
    69	        self.__document = document
    70	        try:
    71	            return b"".join(self.eval(self.pattern))
    72	        finally:
    73	            del self.__document
    74
    75	    def clone(self):
    76	        c = object.__new__(self.__class__)
    77	        c.pattern = self.pattern
    78	        c.arguments = self.arguments
    79	        return c

    82	    '''cond will be evaluated as [cond] in PDFpattern eval.

    84	    thenPart is a list to be evaluated if the cond evaulates true,
    85	    elsePart is the false sequence.
    86	    '''
    87	    def __init__(self,cond,thenPart=[],elsePart=[]):
    88	        if not isinstance(cond,list): cond = [cond]
