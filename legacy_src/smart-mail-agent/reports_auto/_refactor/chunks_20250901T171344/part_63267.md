# since these are the type variables we are ultimately trying to infer;
  5450	        # they must be considered as indeterminate. We use ErasedType since it
  5451	        # does not affect type inference results (it is for purposes like this
  5452	        # only).
  5453	        if not self.chk.options.old_type_inference:
  5454	            # With new type inference we can preserve argument types even if they
  5455	            # are generic, since new inference algorithm can handle constraints
  5456	            # like S <: T (we still erase return type since it's ultimately unknown).
  5457	            extra_vars = []
  5458	            for arg in ctx.arg_types:
  5459	                meta_vars = [tv for tv in get_all_type_vars(arg) if tv.id.is_meta_var()]
  5460	                extra_vars.extend([tv for tv in meta_vars if tv not in extra_vars])
  5461	            callable_ctx = ctx.copy_modified(
  5462	                ret_type=replace_meta_vars(ctx.ret_type, ErasedType()),
  5463	                variables=list(ctx.variables) + extra_vars,
  5464	            )
  5465	        else:
  5466	            erased_ctx = replace_meta_vars(ctx, ErasedType())
  5467	            assert isinstance(erased_ctx, ProperType) and isinstance(erased_ctx, CallableType)
  5468	            callable_ctx = erased_ctx
  5469
  5470	        # The callable_ctx may have a fallback of builtins.type if the context
  5471	        # is a constructor -- but this fallback doesn't make sense for lambdas.
  5472	        callable_ctx = callable_ctx.copy_modified(fallback=self.named_type("builtins.function"))
  5473
  5474	        if callable_ctx.type_guard is not None or callable_ctx.type_is is not None:
  5475	            # Lambda's return type cannot be treated as a `TypeGuard`,
  5476	            # because it is implicit. And `TypeGuard`s must be explicit.
  5477	            # See https://github.com/python/mypy/issues/9927
  5478	            return None, None
  5479
  5480	        arg_kinds = [arg.kind for arg in e.arguments]
  5481
  5482	        if callable_ctx.is_ellipsis_args or ctx.param_spec() is not None:
  5483	            # Fill in Any arguments to match the arguments of the lambda.
  5484	            callable_ctx = callable_ctx.copy_modified(
  5485	                is_ellipsis_args=False,
  5486	                arg_types=[AnyType(TypeOfAny.special_form)] * len(arg_kinds),
  5487	                arg_kinds=arg_kinds,
  5488	                arg_names=e.arg_names.copy(),
  5489	            )
  5490
  5491	        if ARG_STAR in arg_kinds or ARG_STAR2 in arg_kinds:
  5492	            # TODO treat this case appropriately
  5493	            return callable_ctx, None
  5494
  5495	        if callable_ctx.arg_kinds != arg_kinds:
  5496	            # Incompatible context; cannot use it to infer types.
  5497	            self.chk.fail(message_registry.CANNOT_INFER_LAMBDA_TYPE, e)
  5498	            return None, None
  5499
  5500	        # Type of lambda must have correct argument names, to prevent false
  5501	        # negatives when lambdas appear in `ParamSpec` context.
  5502	        return callable_ctx.copy_modified(arg_names=e.arg_names), callable_ctx

  5504	    def visit_super_expr(self, e: SuperExpr) -> Type:
  5505	        """Type check a super expression (non-lvalue)."""
  5506
  5507	        # We have an expression like super(T, var).member
  5508
  5509	        # First compute the types of T and var
  5510	        types = self._super_arg_types(e)
  5511	        if isinstance(types, tuple):
  5512	            type_type, instance_type = types
  5513	        else:
  5514	            return types
  5515
  5516	        # Now get the MRO
  5517	        type_info = type_info_from_type(type_type)
  5518	        if type_info is None:
  5519	            self.chk.fail(message_registry.UNSUPPORTED_ARG_1_FOR_SUPER, e)
  5520	            return AnyType(TypeOfAny.from_error)
  5521
  5522	        instance_info = type_info_from_type(instance_type)
  5523	        if instance_info is None:
  5524	            self.chk.fail(message_registry.UNSUPPORTED_ARG_2_FOR_SUPER, e)
  5525	            return AnyType(TypeOfAny.from_error)
  5526
  5527	        mro = instance_info.mro
  5528
  5529	        # The base is the first MRO entry *after* type_info that has a member
  5530	        # with the right name
  5531	        index = None
  5532	        if type_info in mro:
  5533	            index = mro.index(type_info)
  5534	        else:
  5535	            method = self.chk.scope.current_function()
  5536	            # Mypy explicitly allows supertype upper bounds (and no upper bound at all)
  5537	            # for annotating self-types. However, if such an annotation is used for
  5538	            # checking super() we will still get an error. So to be consistent, we also
  5539	            # allow such imprecise annotations for use with super(), where we fall back
  5540	            # to the current class MRO instead. This works only from inside a method.
  5541	            if method is not None and is_self_type_like(
  5542	                instance_type, is_classmethod=method.is_class
  5543	            ):
  5544	                if e.info and type_info in e.info.mro:
  5545	                    mro = e.info.mro
  5546	                    index = mro.index(type_info)
  5547	        if index is None:
  5548	            if (
  5549	                instance_info.is_protocol
  5550	                and instance_info != type_info
  5551	                and not type_info.is_protocol
  5552	            ):
  5553	                # A special case for mixins, in this case super() should point
  5554	                # directly to the host protocol, this is not safe, since the real MRO
  5555	                # is not known yet for mixin, but this feature is more like an escape hatch.
  5556	                index = -1
  5557	            else:
  5558	                self.chk.fail(message_registry.SUPER_ARG_2_NOT_INSTANCE_OF_ARG_1, e)
  5559	                return AnyType(TypeOfAny.from_error)
  5560
  5561	        if len(mro) == index + 1:
  5562	            self.chk.fail(message_registry.TARGET_CLASS_HAS_NO_BASE_CLASS, e)
  5563	            return AnyType(TypeOfAny.from_error)
  5564
  5565	        for base in mro[index + 1 :]:
  5566	            if e.name in base.names or base == mro[-1]:
  5567	                if e.info and e.info.fallback_to_any and base == mro[-1]:
  5568	                    # There's an undefined base class, and we're at the end of the
  5569	                    # chain.  That's not an error.
  5570	                    return AnyType(TypeOfAny.special_form)
  5571
  5572	                return analyze_member_access(
  5573	                    name=e.name,
  5574	                    typ=instance_type,
  5575	                    is_lvalue=False,
  5576	                    is_super=True,
  5577	                    is_operator=False,
  5578	                    original_type=instance_type,
  5579	                    override_info=base,
  5580	                    context=e,
  5581	                    chk=self.chk,
  5582	                    in_literal_context=self.is_literal_context(),
  5583	                )
  5584
  5585	        assert False, "unreachable"

  5587	    def _super_arg_types(self, e: SuperExpr) -> Type | tuple[Type, Type]:
  5588	        """
  5589	        Computes the types of the type and instance expressions in super(T, instance), or the
  5590	        implicit ones for zero-argument super() expressions.  Returns a single type for the whole
  5591	        super expression when possible (for errors, anys), otherwise the pair of computed types.
  5592	        """
  5593
  5594	        if not self.chk.in_checked_function():
  5595	            return AnyType(TypeOfAny.unannotated)
  5596	        elif len(e.call.args) == 0:
  5597	            if not e.info:
  5598	                # This has already been reported by the semantic analyzer.
  5599	                return AnyType(TypeOfAny.from_error)
  5600	            elif self.chk.scope.active_class():
  5601	                self.chk.fail(message_registry.SUPER_OUTSIDE_OF_METHOD_NOT_SUPPORTED, e)
  5602	                return AnyType(TypeOfAny.from_error)
  5603
  5604	            # Zero-argument super() is like super(<current class>, <self>)
  5605	            current_type = fill_typevars(e.info)
  5606	            type_type: ProperType = TypeType(current_type)
  5607
  5608	            # Use the type of the self argument, in case it was annotated
  5609	            method = self.chk.scope.current_function()
  5610	            assert method is not None
  5611	            if method.arguments:
  5612	                instance_type: Type = method.arguments[0].variable.type or current_type
  5613	            else:
  5614	                self.chk.fail(message_registry.SUPER_ENCLOSING_POSITIONAL_ARGS_REQUIRED, e)
  5615	                return AnyType(TypeOfAny.from_error)
  5616	        elif ARG_STAR in e.call.arg_kinds:
  5617	            self.chk.fail(message_registry.SUPER_VARARGS_NOT_SUPPORTED, e)
  5618	            return AnyType(TypeOfAny.from_error)
  5619	        elif set(e.call.arg_kinds) != {ARG_POS}:
  5620	            self.chk.fail(message_registry.SUPER_POSITIONAL_ARGS_REQUIRED, e)
  5621	            return AnyType(TypeOfAny.from_error)
  5622	        elif len(e.call.args) == 1:
  5623	            self.chk.fail(message_registry.SUPER_WITH_SINGLE_ARG_NOT_SUPPORTED, e)
  5624	            return AnyType(TypeOfAny.from_error)
  5625	        elif len(e.call.args) == 2:
  5626	            type_type = get_proper_type(self.accept(e.call.args[0]))
  5627	            instance_type = self.accept(e.call.args[1])
  5628	        else:
  5629	            self.chk.fail(message_registry.TOO_MANY_ARGS_FOR_SUPER, e)
  5630	            return AnyType(TypeOfAny.from_error)
  5631
  5632	        # Imprecisely assume that the type is the current class
  5633	        if isinstance(type_type, AnyType):
  5634	            if e.info:
  5635	                type_type = TypeType(fill_typevars(e.info))
  5636	            else:
  5637	                return AnyType(TypeOfAny.from_another_any, source_any=type_type)
  5638	        elif isinstance(type_type, TypeType):
  5639	            type_item = type_type.item
  5640	            if isinstance(type_item, AnyType):
  5641	                if e.info:
  5642	                    type_type = TypeType(fill_typevars(e.info))
  5643	                else:
  5644	                    return AnyType(TypeOfAny.from_another_any, source_any=type_item)
  5645
  5646	        if not isinstance(type_type, TypeType) and not (
  5647	            isinstance(type_type, FunctionLike) and type_type.is_type_obj()
  5648	        ):
  5649	            self.msg.first_argument_for_super_must_be_type(type_type, e)
  5650	            return AnyType(TypeOfAny.from_error)
  5651
  5652	        # Imprecisely assume that the instance is of the current class
  5653	        instance_type = get_proper_type(instance_type)
  5654	        if isinstance(instance_type, AnyType):
  5655	            if e.info:
  5656	                instance_type = fill_typevars(e.info)
  5657	            else:
  5658	                return AnyType(TypeOfAny.from_another_any, source_any=instance_type)
  5659	        elif isinstance(instance_type, TypeType):
  5660	            instance_item = instance_type.item
  5661	            if isinstance(instance_item, AnyType):
  5662	                if e.info:
  5663	                    instance_type = TypeType(fill_typevars(e.info))
  5664	                else:
  5665	                    return AnyType(TypeOfAny.from_another_any, source_any=instance_item)
  5666
  5667	        return type_type, instance_type
