) == 0 and not t.empty_tuple_index:
   624	                # Bare 'Tuple' is same as 'tuple'
   625	                any_type = self.get_omitted_any(t)
   626	                return self.named_type("builtins.tuple", [any_type], line=t.line, column=t.column)
   627	            if len(t.args) == 2 and isinstance(t.args[1], EllipsisType):
   628	                # Tuple[T, ...] (uniform, variable-length tuple)
   629	                instance = self.named_type("builtins.tuple", [self.anal_type(t.args[0])])
   630	                instance.line = t.line
   631	                return instance
   632	            return self.tuple_type(
   633	                self.anal_array(t.args, allow_unpack=True), line=t.line, column=t.column
   634	            )
   635	        elif fullname == "typing.Union":
   636	            items = self.anal_array(t.args)
   637	            return UnionType.make_union(items)
   638	        elif fullname == "typing.Optional":
   639	            if len(t.args) != 1:
   640	                self.fail(
   641	                    "Optional[...] must have exactly one type argument", t, code=codes.VALID_TYPE
   642	                )
   643	                return AnyType(TypeOfAny.from_error)
   644	            item = self.anal_type(t.args[0])
   645	            return make_optional_type(item)
   646	        elif fullname == "typing.Callable":
   647	            return self.analyze_callable_type(t)
   648	        elif fullname in TYPE_NAMES:
   649	            if len(t.args) == 0:
   650	                if fullname == "typing.Type":
   651	                    any_type = self.get_omitted_any(t)
   652	                    return TypeType(any_type, line=t.line, column=t.column)
   653	                else:
   654	                    # To prevent assignment of 'builtins.type' inferred as 'builtins.object'
   655	                    # See https://github.com/python/mypy/issues/9476 for more information
   656	                    return None
   657	            type_str = "Type[...]" if fullname == "typing.Type" else "type[...]"
   658	            if len(t.args) != 1:
   659	                self.fail(
   660	                    f"{type_str} must have exactly one type argument", t, code=codes.VALID_TYPE
   661	                )
   662	            item = self.anal_type(t.args[0])
   663	            bad_item_name = get_bad_type_type_item(item)
   664	            if bad_item_name:
   665	                self.fail(f'{type_str} can\'t contain "{bad_item_name}"', t, code=codes.VALID_TYPE)
   666	                item = AnyType(TypeOfAny.from_error)
   667	            return TypeType.make_normalized(item, line=t.line, column=t.column)
   668	        elif fullname == "typing.ClassVar":
   669	            if self.nesting_level > 0:
   670	                self.fail(
   671	                    "Invalid type: ClassVar nested inside other type", t, code=codes.VALID_TYPE
   672	                )
   673	            if self.prohibit_special_class_field_types:
   674	                self.fail(
   675	                    f"ClassVar[...] can't be used inside a {self.prohibit_special_class_field_types}",
   676	                    t,
   677	                    code=codes.VALID_TYPE,
   678	                )
   679	            if self.defining_alias:
   680	                self.fail(
   681	                    "ClassVar[...] can't be used inside a type alias", t, code=codes.VALID_TYPE
   682	                )
   683	            if len(t.args) == 0:
   684	                return AnyType(TypeOfAny.from_omitted_generics, line=t.line, column=t.column)
   685	            if len(t.args) != 1:
   686	                self.fail(
   687	                    "ClassVar[...] must have at most one type argument", t, code=codes.VALID_TYPE
   688	                )
   689	                return AnyType(TypeOfAny.from_error)
   690	            return self.anal_type(t.args[0], allow_final=self.options.python_version >= (3, 13))
   691	        elif fullname in NEVER_NAMES:
   692	            return UninhabitedType()
   693	        elif fullname in LITERAL_TYPE_NAMES:
   694	            return self.analyze_literal_type(t)
   695	        elif fullname in ANNOTATED_TYPE_NAMES:
   696	            if len(t.args) < 2:
   697	                self.fail(
   698	                    "Annotated[...] must have exactly one type argument"
   699	                    " and at least one annotation",
   700	                    t,
   701	                    code=codes.VALID_TYPE,
   702	                )
   703	                return AnyType(TypeOfAny.from_error)
   704	            return self.anal_type(
   705	                t.args[0], allow_typed_dict_special_forms=self.allow_typed_dict_special_forms
   706	            )
   707	        elif fullname in ("typing_extensions.Required", "typing.Required"):
   708	            if not self.allow_typed_dict_special_forms:
   709	                self.fail(
   710	                    "Required[] can be only used in a TypedDict definition",
   711	                    t,
   712	                    code=codes.VALID_TYPE,
   713	                )
   714	                return AnyType(TypeOfAny.from_error)
   715	            if len(t.args) != 1:
   716	                self.fail(
   717	                    "Required[] must have exactly one type argument", t, code=codes.VALID_TYPE
   718	                )
   719	                return AnyType(TypeOfAny.from_error)
   720	            return RequiredType(
   721	                self.anal_type(t.args[0], allow_typed_dict_special_forms=True), required=True
   722	            )
   723	        elif fullname in ("typing_extensions.NotRequired", "typing.NotRequired"):
   724	            if not self.allow_typed_dict_special_forms:
   725	                self.fail(
   726	                    "NotRequired[] can be only used in a TypedDict definition",
   727	                    t,
   728	                    code=codes.VALID_TYPE,
   729	                )
   730	                return AnyType(TypeOfAny.from_error)
   731	            if len(t.args) != 1:
   732	                self.fail(
   733	                    "NotRequired[] must have exactly one type argument", t, code=codes.VALID_TYPE
   734	                )
   735	                return AnyType(TypeOfAny.from_error)
   736	            return RequiredType(
   737	                self.anal_type(t.args[0], allow_typed_dict_special_forms=True), required=False
   738	            )
   739	        elif fullname in ("typing_extensions.ReadOnly", "typing.ReadOnly"):
   740	            if not self.allow_typed_dict_special_forms:
   741	                self.fail(
   742	                    "ReadOnly[] can be only used in a TypedDict definition",
   743	                    t,
   744	                    code=codes.VALID_TYPE,
   745	                )
   746	                return AnyType(TypeOfAny.from_error)
   747	            if len(t.args) != 1:
   748	                self.fail(
   749	                    '"ReadOnly[]" must have exactly one type argument', t, code=codes.VALID_TYPE
   750	                )
   751	                return AnyType(TypeOfAny.from_error)
   752	            return ReadOnlyType(self.anal_type(t.args[0], allow_typed_dict_special_forms=True))
   753	        elif (
   754	            self.anal_type_guard_arg(t, fullname) is not None
   755	            or self.anal_type_is_arg(t, fullname) is not None
   756	        ):
   757	            # In most contexts, TypeGuard[...] acts as an alias for bool (ignoring its args)
   758	            return self.named_type("builtins.bool")
   759	        elif fullname in UNPACK_TYPE_NAMES:
   760	            if len(t.args) != 1:
   761	                self.fail("Unpack[...] requires exactly one type argument", t)
   762	                return AnyType(TypeOfAny.from_error)
   763	            if not self.allow_unpack:
   764	                self.fail(message_registry.INVALID_UNPACK_POSITION, t, code=codes.VALID_TYPE)
   765	                return AnyType(TypeOfAny.from_error)
   766	            self.allow_type_var_tuple = True
   767	            result = UnpackType(self.anal_type(t.args[0]), line=t.line, column=t.column)
   768	            self.allow_type_var_tuple = False
   769	            return result
   770	        elif fullname in SELF_TYPE_NAMES:
   771	            if t.args:
   772	                self.fail("Self type cannot have type arguments", t)
   773	            if self.prohibit_self_type is not None:
   774	                self.fail(f"Self type cannot be used in {self.prohibit_self_type}", t)
   775	                return AnyType(TypeOfAny.from_error)
   776	            if self.api.type is None:
   777	                self.fail("Self type is only allowed in annotations within class definition", t)
   778	                return AnyType(TypeOfAny.from_error)
   779	            if self.api.type.has_base("builtins.type"):
   780	                self.fail("Self type cannot be used in a metaclass", t)
   781	            if self.api.type.self_type is not None:
   782	                if self.api.type.is_final:
   783	                    return fill_typevars(self.api.type)
   784	                return self.api.type.self_type.copy_modified(line=t.line, column=t.column)
   785	            # TODO: verify this is unreachable and replace with an assert?
   786	            self.fail("Unexpected Self type", t)
   787	            return AnyType(TypeOfAny.from_error)
   788	        return None

   792	        return get_omitted_any(disallow_any, self.fail, self.note, typ, self.options, fullname)

   794	    def check_and_warn_deprecated(self, info: TypeInfo, ctx: Context) -> None:
   795	        """Similar logic to `TypeChecker.check_deprecated` and `TypeChecker.warn_deprecated."""
   796
   797	        if (
   798	            (deprecated := info.deprecated)
   799	            and not self.is_typeshed_stub
   800	            and not (self.api.type and (self.api.type.fullname == info.fullname))
   801	            and not any(
   802	                info.fullname == p or info.fullname.startswith(f"{p}.")
   803	                for p in self.options.deprecated_calls_exclude
   804	            )
   805	        ):
   806	            for imp in self.cur_mod_node.imports:
   807	                if isinstance(imp, ImportFrom) and any(info.name == n[0] for n in imp.names):
   808	                    break
   809	            else:
   810	                warn = self.note if self.options.report_deprecated_as_note else self.fail
   811	                warn(deprecated, ctx, code=codes.DEPRECATED)

   813	    def analyze_type_with_type_info(
   814	        self, info: TypeInfo, args: Sequence[Type], ctx: Context, empty_tuple_index: bool
   815	    ) -> Type:
   816	        """Bind unbound type when were able to find target TypeInfo.
   817
   818	        This handles simple cases like 'int', 'modname.UserClass[str]', etc.
   819	        """
   820
   821	        self.check_and_warn_deprecated(info, ctx)
   822
   823	        if len(args) > 0 and info.fullname == "builtins.tuple":
   824	            fallback = Instance(info, [AnyType(TypeOfAny.special_form)], ctx.line)
   825	            return TupleType(self.anal_array(args, allow_unpack=True), fallback, ctx.line)
   826
   827	        # Analyze arguments and (usually) construct Instance type. The
   828	        # number of type arguments and their values are
   829	        # checked only later, since we do not always know the
   830	        # valid count at this point. Thus we may construct an
   831	        # Instance with an invalid number of type arguments.
   832	        #
   833	        # We allow ParamSpec literals based on a heuristic: it will be
   834	        # checked later anyways but the error message may be worse.
   835	        instance = Instance(
   836	            info,
   837	            self.anal_array(
   838	                args,
   839	                allow_param_spec=True,
   840	                allow_param_spec_literals=info.has_param_spec_type,
   841	                allow_unpack=True,  # Fixed length tuples can be used for non-variadic types.
   842	            ),
   843	            ctx.line,
   844	            ctx.column,
   845	        )
   846	        instance.end_line = ctx.end_line
   847	        instance.end_column = ctx.end_column
   848	        if len(info.type_vars) == 1 and info.has_param_spec_type:
   849	            instance.args = tuple(self.pack_paramspec_args(instance.args))
   850
   851	        # Check type argument count.
   852	        instance.args = tuple(flatten_nested_tuples(instance.args))
   853	        if not (self.defining_alias and self.nesting_level == 0) and not validate_instance(
   854	            instance, self.fail, empty_tuple_index
   855	        ):
   856	            fix_instance(
   857	                instance,
   858	                self.fail,
   859	                self.note,
   860	                disallow_any=self.options.disallow_any_generics and not self.is_typeshed_stub,
   861	                options=self.options,
   862	            )
   863
   864	        tup = info.tuple_type
   865	        if tup is not None:
   866	            # The class has a Tuple[...] base class so it will be
   867	            # represented as a tuple type.
   868	            if info.special_alias:
   869	                return instantiate_type_alias(
   870	                    info.special_alias,
   871	                    # TODO: should we allow NamedTuples generic in ParamSpec?
   872	                    self.anal_array(args, allow_unpack=True),
   873	                    self.fail,
   874	                    False,
   875	                    ctx,
   876	                    self.options,
   877	                    use_standard_error=True,
   878	                )
   879	            return tup.copy_modified(
   880	                items=self.anal_array(tup.items, allow_unpack=True), fallback=instance
   881	            )
   882	        td = info.typeddict_type
   883	        if td is not None:
   884	            # The class has a TypedDict[...] base class so it will be
   885	            # represented as a typeddict type.
   886	            if info.special_alias:
   887	                return instantiate_type_alias(
   888	                    info.special_alias,
   889	                    # TODO: should we allow TypedDicts generic in ParamSpec?
   890	                    self.anal_array(args, allow_unpack=True),
   891	                    self.fail,
   892	                    False,
   893	                    ctx,
   894	                    self.options,
   895	                    use_standard_error=True,
   896	                )
   897	            # Create a named TypedDictType
   898	            return td.copy_modified(
   899	                item_types=self.anal_array(list(td.items.values())), fallback=instance
   900	            )
   901
   902	        if info.fullname == "types.NoneType":
   903	            self.fail(
   904	                "NoneType should not be used as a type, please use None instead",
   905	                ctx,
   906	                code=codes.VALID_TYPE,
   907	            )
   908	            return NoneType(ctx.line, ctx.column)
   909
   910	        return instance

   914	    ) -> Type:
   915	        """Figure out what an unbound type that doesn't refer to a TypeInfo node means.
   916

   918	        """
   919	        name = sym.fullname
   920	        if name is None:
   921	            assert sym.node is not None
   922	            name = sym.node.name
   923	        # Option 1:
   924	        # Something with an Any type -- make it an alias for Any in a type
   925	        # context. This is slightly problematic as it allows using the type 'Any'
   926	        # as a base class -- however, this will fail soon at runtime so the problem
   927	        # is pretty minor.
   928	        if isinstance(sym.node, Var):
   929	            typ = get_proper_type(sym.node.type)
   930	            if isinstance(typ, AnyType):
   931	                return AnyType(
   932	                    TypeOfAny.from_unimported_type, missing_import_name=typ.missing_import_name
   933	                )
   934	            elif self.allow_type_any:
   935	                if isinstance(typ, Instance) and typ.type.fullname == "builtins.type":
   936	                    return AnyType(TypeOfAny.special_form)
   937	                if isinstance(typ, TypeType) and isinstance(typ.item, AnyType):
   938	                    return AnyType(TypeOfAny.from_another_any, source_any=typ.item)
   939	        # Option 2:
   940	        # Unbound type variable. Currently these may be still valid,
   941	        # for example when defining a generic type alias.

   943	            isinstance(sym.node, (TypeVarExpr, TypeVarTupleExpr))
   944	            and self.tvar_scope.get_binding(sym) is None

   975	                line=t.line,
   976	                column=t.column,
   977	            )
   978

   980	        # to make sure there are no remaining semanal-only types, then give up.

   988	                "stable/common_issues.html#variables-vs-type-aliases"
   989	            )
