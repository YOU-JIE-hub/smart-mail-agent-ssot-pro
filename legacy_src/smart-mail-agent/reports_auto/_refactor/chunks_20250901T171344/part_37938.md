de_everything:
   268	            return False
   269
   270	        # self.present==True matches everything except None.
   271	        if self.present is True:
   272	            return string is not None
   273
   274	        # self.present==False matches _only_ None.
   275	        if self.present is False:
   276	            return string is None
   277
   278	        # self.string does an exact string match.
   279	        if self.string is not None:
   280	            # print(f"{self.string} ?= {string}")
   281	            return self.string == string
   282
   283	        # self.pattern does a regular expression search.
   284	        if self.pattern is not None:
   285	            # print(f"{self.pattern} ?~ {string}")
   286	            if string is None:
   287	                return False
   288	            return self.pattern.search(string) is not None
   289
   290	        return None
   291
   292	    def matches_string(self, string: Optional[str]) -> bool:
   293	        _base_result = self._base_match(string)
   294	        if _base_result is not None:
   295	            # No need to invoke the test function.
   296	            return _base_result
   297	        if self.function is not None and not self.function(string):
   298	            # print(f"{self.function}({string}) == False")
   299	            return False
   300	        return True
   301
   302	    def __repr__(self) -> str:
   303	        cls = type(self).__name__
   304	        return f"<{cls} string={self.string} pattern={self.pattern} function={self.function} present={self.present}>"
   305
   306	    def __eq__(self, other: Any) -> bool:
   307	        return (
   308	            isinstance(other, MatchRule)
   309	            and self.string == other.string
   310	            and self.pattern == other.pattern
   311	            and self.function == other.function
   312	            and self.present == other.present
   313	        )

   347	    """The `ElementFilter` subclass used internally by Beautiful Soup.
   348
   349	    A `SoupStrainer` encapsulates the logic necessary to perform the
   350	    kind of matches supported by methods such as
   351	    :py:meth:`Tag.find`. `SoupStrainer` objects are primarily created
   352	    internally, but you can create one yourself and pass it in as
   353	    ``parse_only`` to the `BeautifulSoup` constructor, to parse a
   354	    subset of a large document.

   357	    constructor arguments into `MatchRule` objects. Incoming
   358	    tags/markup are matched against those rules.

   363	      restrictions on tags that use those attributes.

   490	    @classmethod
   491	    def _make_match_rules(
   492	        cls,
   493	        obj: Optional[Union[_StrainableElement, _StrainableAttribute]],
   494	        rule_class: Type[MatchRule],
   495	    ) -> Iterator[MatchRule]:
   496	        """Convert a vaguely-specific 'object' into one or more well-defined
   497	        `MatchRule` objects.
   498
   499	        :param obj: Some kind of object that corresponds to one or more
   500	           matching rules.
   501	        :param rule_class: Create instances of this `MatchRule` subclass.
   502	        """
   503	        if obj is None:
   504	            return
   505	        if isinstance(obj, (str, bytes)):
   506	            yield rule_class(string=obj)
   507	        elif isinstance(obj, bool):
   508	            yield rule_class(present=obj)
   509	        elif callable(obj):
   510	            yield rule_class(function=obj)
   511	        elif isinstance(obj, _RegularExpressionProtocol):
   512	            yield rule_class(pattern=obj)
   513	        elif hasattr(obj, "__iter__"):
   514	            if not obj:
   515	                # The attribute is being matched against the null set,
   516	                # which means it should exclude everything.
   517	                yield rule_class(exclude_everything=True)
   518	            for o in obj:
   519	                if not isinstance(o, (bytes, str)) and hasattr(o, "__iter__"):
   520	                    # This is almost certainly the user's
   521	                    # mistake. This list contains another list, which
   522	                    # opens up the possibility of infinite
   523	                    # self-reference. In the interests of avoiding
   524	                    # infinite recursion, we'll treat this as an
   525	                    # impossible match and issue a rule that excludes
   526	                    # everything, rather than looking inside.
   527	                    warnings.warn(
   528	                        f"Ignoring nested list {o} to avoid the possibility of infinite recursion.",
   529	                        stacklevel=5,
   530	                    )
   531	                    yield rule_class(exclude_everything=True)
   532	                    continue
   533	                for x in cls._make_match_rules(o, rule_class):
   534	                    yield x
   535	        else:
   536	            yield rule_class(string=str(obj))

   538	    def matches_tag(self, tag: Tag) -> bool:
   539	        """Do the rules of this `SoupStrainer` trigger a match against the
   540	        given `Tag`?
   541
   542	        If the `SoupStrainer` has any `TagNameMatchRule`, at least one
   543	        must match the `Tag` or its `Tag.name`.
   544
   545	        If there are any `AttributeValueMatchRule` for a given
   546	        attribute, at least one of them must match the attribute
   547	        value.
   548
   549	        If there are any `StringMatchRule`, at least one must match,
   550	        but a `SoupStrainer` that *only* contains `StringMatchRule`
   551	        cannot match a `Tag`, only a `NavigableString`.
   552	        """
   553	        # If there are no rules at all, let anything through.
   554	        #if self.includes_everything:
   555	        #    return True
   556
   557	        # String rules cannot not match a Tag on their own.
   558	        if not self.name_rules and not self.attribute_rules:
   559	            return False
   560
   561	        # Optimization for a very common case where the user is
   562	        # searching for a tag with one specific name, and we're
   563	        # looking at a tag with a different name.
   564	        if (
   565	            not tag.prefix
   566	            and len(self.name_rules) == 1
   567	            and self.name_rules[0].string is not None
   568	            and tag.name != self.name_rules[0].string
   569	        ):
   570	            return False
   571
   572	        # If there are name rules, at least one must match. It can
   573	        # match either the Tag object itself or the prefixed name of
   574	        # the tag.
   575	        prefixed_name = None
   576	        if tag.prefix:
   577	            prefixed_name = f"{tag.prefix}:{tag.name}"
   578	        if self.name_rules:
   579	            name_matches = False
   580	            for rule in self.name_rules:
   581	                # attrs = " ".join(
   582	                #     [f"{k}={v}" for k, v in sorted(tag.attrs.items())]
   583	                # )
   584	                # print(f"Testing <{tag.name} {attrs}>{tag.string}</{tag.name}> against {rule}")
   585
   586	                # If the rule contains a function, the function will be called
   587	                # with `tag`. It will not be called a second time with
   588	                # `prefixed_name`.
   589	                if rule.matches_tag(tag) or (
   590	                        not rule.function and prefixed_name is not None and rule.matches_string(prefixed_name)
   591	                ):
   592	                    name_matches = True
   593	                    break
   594
   595	            if not name_matches:
   596	                return False
   597
   598	        # If there are attribute rules for a given attribute, at least
   599	        # one of them must match. If there are rules for multiple
   600	        # attributes, each attribute must have at least one match.
   601	        for attr, rules in self.attribute_rules.items():
   602	            attr_value = tag.get(attr, None)
   603	            this_attr_match = self._attribute_match(attr_value, rules)
   604	            if not this_attr_match:
   605	                return False
   606
   607	        # If there are string rules, at least one must match.
   608	        if self.string_rules:
   609	            _str = tag.string
   610	            if _str is None:
   611	                return False
   612	            if not self.matches_any_string_rule(_str):
   613	                return False
   614	        return True

   616	    def _attribute_match(
   617	        self,
   618	        attr_value: Optional[_AttributeValue],
   619	        rules: Iterable[AttributeValueMatchRule],
   620	    ) -> bool:
   621	        attr_values: Sequence[Optional[str]]
   622	        if isinstance(attr_value, list):
   623	            attr_values = attr_value
   624	        else:
   625	            attr_values = [cast(str, attr_value)]
   626
   627	        def _match_attribute_value_helper(attr_values: Sequence[Optional[str]]) -> bool:
   628	            for rule in rules:
   629	                for attr_value in attr_values:
   630	                    if rule.matches_string(attr_value):
   631	                        return True
   632	            return False
   633
   634	        this_attr_match = _match_attribute_value_helper(attr_values)
   635	        if not this_attr_match and len(attr_values) > 1:
   636	            # This cast converts Optional[str] to plain str.
   637	            #
   638	            # We know if there's more than one value, there can't be
   639	            # any None in the list, because Beautiful Soup never uses
   640	            # None as a value of a multi-valued attribute, and if None
   641	            # is passed in as attr_value, it's turned into a list with
   642	            # a single element (thus len(attr_values) > 1 fails).
   643	            attr_values = cast(Sequence[str], attr_values)
   644
   645	            # Try again but treat the attribute value
   646	            # as a single string.
   647	            joined_attr_value = " ".join(attr_values)
   648	            this_attr_match = _match_attribute_value_helper([joined_attr_value])
   649	        return this_attr_match
