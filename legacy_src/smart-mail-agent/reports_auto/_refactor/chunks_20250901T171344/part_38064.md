  4097	                            self.chk.check_deprecated(item.func, context)
  4098	                return result
  4099
  4100	        # We finish invoking above operators and no early return happens. Therefore,
  4101	        # we check if either the LHS or the RHS is Instance and fallbacks to Any,
  4102	        # if so, we also return Any
  4103	        if (isinstance(left_type, Instance) and left_type.type.fallback_to_any) or (
  4104	            isinstance(right_type, Instance) and right_type.type.fallback_to_any
  4105	        ):
  4106	            any_type = AnyType(TypeOfAny.special_form)
  4107	            return any_type, any_type
  4108
  4109	        # STEP 4b:
  4110	        # Sometimes, the variants list is empty. In that case, we fall-back to attempting to
  4111	        # call the __op__ method (even though it's missing).
  4112
  4113	        if not variants:
  4114	            with self.msg.filter_errors(save_filtered_errors=True) as local_errors:
  4115	                result = self.check_method_call_by_name(
  4116	                    op_name, left_type, [right_expr], [ARG_POS], context
  4117	                )
  4118
  4119	            if local_errors.has_new_errors():
  4120	                errors.append(local_errors.filtered_errors())
  4121	                results.append(result)
  4122	            else:
  4123	                # Although we should not need this case anymore, we keep it just in case, as
  4124	                # otherwise we will get a crash if we introduce inconsistency in checkmember.py
  4125	                return result
  4126
  4127	        self.msg.add_errors(errors[0])
  4128	        if len(results) == 1:
  4129	            return results[0]
  4130	        else:
  4131	            error_any = AnyType(TypeOfAny.from_error)
  4132	            result = error_any, error_any
  4133	            return result

  4135	    def check_op(
  4136	        self,
  4137	        method: str,
  4138	        base_type: Type,
  4139	        arg: Expression,
  4140	        context: Context,
  4141	        allow_reverse: bool = False,
  4142	    ) -> tuple[Type, Type]:
  4143	        """Type check a binary operation which maps to a method call.
  4144
  4145	        Return tuple (result type, inferred operator method type).
  4146	        """
  4147
  4148	        if allow_reverse:
  4149	            left_variants = [base_type]
  4150	            base_type = get_proper_type(base_type)
  4151	            if isinstance(base_type, UnionType):
  4152	                left_variants = list(flatten_nested_unions(base_type.relevant_items()))
  4153	            right_type = self.accept(arg)
  4154
  4155	            # Step 1: We first try leaving the right arguments alone and destructure
  4156	            # just the left ones. (Mypy can sometimes perform some more precise inference
  4157	            # if we leave the right operands a union -- see testOperatorWithEmptyListAndSum.)
  4158	            all_results = []
  4159	            all_inferred = []
  4160
  4161	            with self.msg.filter_errors() as local_errors:
  4162	                for left_possible_type in left_variants:
  4163	                    result, inferred = self.check_op_reversible(
  4164	                        op_name=method,
  4165	                        left_type=left_possible_type,
  4166	                        left_expr=TempNode(left_possible_type, context=context),
  4167	                        right_type=right_type,
  4168	                        right_expr=arg,
  4169	                        context=context,
  4170	                    )
  4171	                    all_results.append(result)
  4172	                    all_inferred.append(inferred)
  4173
  4174	            if not local_errors.has_new_errors():
  4175	                results_final = make_simplified_union(all_results)
  4176	                inferred_final = make_simplified_union(all_inferred)
  4177	                return results_final, inferred_final
  4178
  4179	            # Step 2: If that fails, we try again but also destructure the right argument.
  4180	            # This is also necessary to make certain edge cases work -- see
  4181	            # testOperatorDoubleUnionInterwovenUnionAdd, for example.
  4182
  4183	            # Note: We want to pass in the original 'arg' for 'left_expr' and 'right_expr'
  4184	            # whenever possible so that plugins and similar things can introspect on the original
  4185	            # node if possible.
  4186	            #
  4187	            # We don't do the same for the base expression because it could lead to weird
  4188	            # type inference errors -- e.g. see 'testOperatorDoubleUnionSum'.
  4189	            # TODO: Can we use `type_overrides_set()` here?
  4190	            right_variants = [(right_type, arg)]
  4191	            right_type = get_proper_type(right_type)
  4192	            if isinstance(right_type, UnionType):
  4193	                right_variants = [
  4194	                    (item, TempNode(item, context=context))
  4195	                    for item in flatten_nested_unions(right_type.relevant_items())
  4196	                ]
  4197
  4198	            all_results = []
  4199	            all_inferred = []
  4200
  4201	            with self.msg.filter_errors(save_filtered_errors=True) as local_errors:
  4202	                for left_possible_type in left_variants:
  4203	                    for right_possible_type, right_expr in right_variants:
  4204	                        result, inferred = self.check_op_reversible(
  4205	                            op_name=method,
  4206	                            left_type=left_possible_type,
  4207	                            left_expr=TempNode(left_possible_type, context=context),
  4208	                            right_type=right_possible_type,
  4209	                            right_expr=right_expr,
  4210	                            context=context,
  4211	                        )
  4212	                        all_results.append(result)
  4213	                        all_inferred.append(inferred)
  4214
  4215	            if local_errors.has_new_errors():
  4216	                self.msg.add_errors(local_errors.filtered_errors())
  4217	                # Point any notes to the same location as an existing message.
  4218	                err = local_errors.filtered_errors()[-1]
  4219	                recent_context = TempNode(NoneType())
  4220	                recent_context.line = err.line
  4221	                recent_context.column = err.column
  4222	                if len(left_variants) >= 2 and len(right_variants) >= 2:
  4223	                    self.msg.warn_both_operands_are_from_unions(recent_context)
  4224	                elif len(left_variants) >= 2:
  4225	                    self.msg.warn_operand_was_from_union("Left", base_type, context=recent_context)
  4226	                elif len(right_variants) >= 2:
  4227	                    self.msg.warn_operand_was_from_union(
  4228	                        "Right", right_type, context=recent_context
  4229	                    )
  4230
  4231	            # See the comment in 'check_overload_call' for more details on why
  4232	            # we call 'combine_function_signature' instead of just unioning the inferred
  4233	            # callable types.
  4234	            results_final = make_simplified_union(all_results)
  4235	            inferred_final = self.combine_function_signatures(get_proper_types(all_inferred))
  4236	            return results_final, inferred_final
  4237	        else:
  4238	            return self.check_method_call_by_name(
  4239	                method=method,
  4240	                base_type=base_type,
  4241	                args=[arg],
  4242	                arg_kinds=[ARG_POS],
  4243	                context=context,
  4244	            )

  4246	    def check_boolean_op(self, e: OpExpr, context: Context) -> Type:
  4247	        """Type check a boolean operation ('and' or 'or')."""
  4248
  4249	        # A boolean operation can evaluate to either of the operands.
  4250
  4251	        # We use the current type context to guide the type inference of of
  4252	        # the left operand. We also use the left operand type to guide the type
  4253	        # inference of the right operand so that expressions such as
  4254	        # '[1] or []' are inferred correctly.
  4255	        ctx = self.type_context[-1]
  4256	        left_type = self.accept(e.left, ctx)
  4257	        expanded_left_type = try_expanding_sum_type_to_union(
  4258	            self.accept(e.left, ctx), "builtins.bool"
  4259	        )
  4260
  4261	        assert e.op in ("and", "or")  # Checked by visit_op_expr
  4262
  4263	        if e.right_always:
  4264	            left_map: mypy.checker.TypeMap = None
  4265	            right_map: mypy.checker.TypeMap = {}
  4266	        elif e.right_unreachable:
  4267	            left_map, right_map = {}, None
  4268	        elif e.op == "and":
  4269	            right_map, left_map = self.chk.find_isinstance_check(e.left)
  4270	        elif e.op == "or":
  4271	            left_map, right_map = self.chk.find_isinstance_check(e.left)
  4272
  4273	        # If left_map is None then we know mypy considers the left expression
  4274	        # to be redundant.
  4275	        if (
  4276	            codes.REDUNDANT_EXPR in self.chk.options.enabled_error_codes
  4277	            and left_map is None
  4278	            # don't report an error if it's intentional
  4279	            and not e.right_always
  4280	        ):
  4281	            self.msg.redundant_left_operand(e.op, e.left)
  4282
  4283	        if (
  4284	            self.chk.should_report_unreachable_issues()
  4285	            and right_map is None
  4286	            # don't report an error if it's intentional
  4287	            and not e.right_unreachable
  4288	        ):
  4289	            self.msg.unreachable_right_operand(e.op, e.right)
  4290
  4291	        right_type = self.analyze_cond_branch(right_map, e.right, expanded_left_type)
  4292
  4293	        if left_map is None and right_map is None:
  4294	            return UninhabitedType()
  4295
  4296	        if right_map is None:
  4297	            # The boolean expression is statically known to be the left value
  4298	            assert left_map is not None
  4299	            return left_type
  4300	        if left_map is None:
  4301	            # The boolean expression is statically known to be the right value
  4302	            assert right_map is not None
  4303	            return right_type
  4304
  4305	        if e.op == "and":
  4306	            restricted_left_type = false_only(expanded_left_type)
  4307	            result_is_left = not expanded_left_type.can_be_true
  4308	        elif e.op == "or":
  4309	            restricted_left_type = true_only(expanded_left_type)
  4310	            result_is_left = not expanded_left_type.can_be_false
  4311
  4312	        if isinstance(restricted_left_type, UninhabitedType):
  4313	            # The left operand can never be the result
  4314	            return right_type
  4315	        elif result_is_left:
  4316	            # The left operand is always the result
  4317	            return left_type
  4318	        else:
  4319	            return make_simplified_union([restricted_left_type, right_type])

  4384	    def visit_index_with_type(
  4385	        self,
  4386	        left_type: Type,
  4387	        e: IndexExpr,
  4388	        original_type: ProperType | None = None,
  4389	        self_type: Type | None = None,
  4390	    ) -> Type:
  4391	        """Analyze type of an index expression for a given type of base expression.
  4392
  4393	        The 'original_type' is used for error messages (currently used for union types). The
  4394	        'self_type' is to bind self in methods (see analyze_member_access for more details).
  4395	        """
  4396	        index = e.index
  4397	        self_type = self_type or left_type
  4398	        left_type = get_proper_type(left_type)
  4399
  4400	        # Visit the index, just to make sure we have a type for it available
  4401	        self.accept(index)
  4402
  4403	        if isinstance(left_type, TupleType) and any(
  4404	            isinstance(it, UnpackType) for it in left_type.items
  4405	        ):
  4406	            # Normalize variadic tuples for consistency.
  4407	            left_type = expand_type(left_type, {})
  4408

  4410	            original_type = original_type or left_type
  4411	            # Don't combine literal types, since we may need them for type narrowing.
  4412	            return make_simplified_union(
  4413	                [
  4414	                    self.visit_index_with_type(typ, e, original_type)
  4415	                    for typ in left_type.relevant_items()

  4417	                contract_literals=False,
  4418	            )
  4419	        elif isinstance(left_type, TupleType) and self.chk.in_checked_function():

  4421	            # indexed by an integer literal.

  4430	                    if item is not None:

  4432	                    else:
  4433	                        self.chk.fail(message_registry.TUPLE_INDEX_OUT_OF_RANGE, e)

  4436	                            self.chk.note(f"Variadic tuple can have length {min_len}", e)
