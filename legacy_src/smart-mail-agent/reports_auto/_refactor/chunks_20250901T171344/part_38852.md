   121	        state = self.textStateStack[-1]
   122	        self.textStateStack = self.textStateStack[:-1]
   123	        #print "pop", self.textStateStack
   124	        state = state[:] # copy for destruction

   127	        for var in self.TEXT_STATE_VARIABLES:
   128	            val = state[0]
   129	            del state[0]
   130	            setattr(self, var, val)

   254	    def fitLine(self, program, totalLength):
   255	        "fit words (and other things) onto a line"
   256	        # assuming word lengths and spaces have not been yet added
   257	        # fit words onto a line up to maxlength, adding spaces and respecting extra space
   258	        from reportlab.pdfbase.pdfmetrics import stringWidth
   259	        usedIndent = self.indent
   260	        maxLength = totalLength - usedIndent - self.rightIndent
   261	        done = 0
   262	        line = []
   263	        cursor = 0
   264	        lineIsFull = 0
   265	        currentLength = 0
   266	        maxcursor = len(program)
   267	        needspace = 0
   268	        first = 1
   269	        terminated = None
   270	        fontName = self.fontName
   271	        fontSize = self.fontSize
   272	        spacewidth = stringWidth(" ", fontName, fontSize) #self.sizeOfWord(" ")
   273	        justStrings = 1
   274	        while not done and cursor<maxcursor:
   275	            opcode = program[cursor]
   276	            #if debug: print "opcode", cursor, opcode
   277	            if isinstance(opcode,str) or hasattr(opcode,'width'):
   278	                lastneedspace = needspace
   279	                needspace = 0
   280	                if hasattr(opcode,'width'):
   281	                    justStrings = 0
   282	                    width = opcode.width(self)
   283	                    needspace = 0
   284	                else:
   285	                    saveopcode = opcode
   286	                    opcode = opcode.strip()
   287	                    if opcode:
   288	                        width = stringWidth(opcode, fontName, fontSize)
   289	                    else:
   290	                        width = 0
   291	                    if saveopcode and (width or currentLength):
   292	                        # ignore white space at margin
   293	                        needspace = (saveopcode[-1]==" ")
   294	                    else:
   295	                        needspace = 0
   296	                fullwidth = width
   297	                if lastneedspace:
   298	                    #spacewidth = stringWidth(" ", fontName, fontSize) #self.sizeOfWord(" ")
   299	                    fullwidth = width + spacewidth
   300	                newlength = currentLength+fullwidth
   301	                if newlength>maxLength and not first: # always do at least one thing
   302	                    # this word won't fit
   303	                    #if debug:
   304	                    #    print "WORD", opcode, "wont fit, width", width, "fullwidth", fullwidth
   305	                    #    print "   currentLength", currentLength, "newlength", newlength, "maxLength", maxLength
   306	                    done = 1
   307	                    lineIsFull = 1
   308	                else:
   309	                    # fit the word: add a space then the word
   310	                    if lastneedspace:
   311	                        line.append( spacewidth ) # expandable space: positive
   312	                    if opcode:
   313	                        line.append( opcode )
   314	                    if abs(width)>TOOSMALLSPACE:
   315	                        line.append( -width ) # non expanding space: negative
   316	                        currentLength = newlength
   317	                    #print line
   318	                    #stop
   319	                first = 0
   320	            elif isinstance(opcode,float):
   321	                justStrings = 0
   322	                aopcode = abs(opcode) # negative means non expanding
   323	                if aopcode>TOOSMALLSPACE:
   324	                    nextLength = currentLength+aopcode
   325	                    if nextLength>maxLength and not first: # always do at least one thing
   326	                        #if debug: print "EXPLICIT spacer won't fit", maxLength, nextLength, opcode
   327	                        done = 1
   328	                    else:
   329	                        if aopcode>TOOSMALLSPACE:
   330	                            currentLength = nextLength
   331	                            line.append(opcode)
   332	                    first = 0
   333	            elif isinstance(opcode,tuple):
   334	                justStrings = 0
   335	                indicator = opcode[0]
   336	                #line.append(opcode)
   337	                if indicator=="nextLine":
   338	                    # advance to nextLine
   339	                    #(i, endallmarks) = opcode
   340	                    line.append(opcode)
   341	                    cursor += 1 # consume this element
   342	                    terminated = done = 1
   343	                    #if debug:
   344	                    #    print "nextLine encountered"
   345	                elif indicator=="color":
   346	                    # change fill color
   347	                    oldcolor = self.fontColor
   348	                    (i, colorname) = opcode
   349	                    #print "opcode", opcode
   350	                    if isinstance(colorname,str):
   351	                        color = self.fontColor = getattr(colors, colorname)
   352	                    else:
   353	                        color = self.fontColor = colorname # assume its something sensible :)
   354	                    line.append(opcode)
   355	                elif indicator=="face":
   356	                    # change font face
   357	                    (i, fontname) = opcode
   358	                    fontName = self.fontName = fontname
   359	                    spacewidth = stringWidth(" ", fontName, fontSize) #self.sizeOfWord(" ")
   360	                    line.append(opcode)
   361	                elif indicator=="size":
   362	                    # change font size
   363	                    (i, fontsize) = opcode
   364	                    size = abs(float(fontsize))
   365	                    if isinstance(fontsize,str):
   366	                        if fontsize[:1]=="+":
   367	                            fontSize = self.fontSize = self.fontSize + size
   368	                        elif fontsize[:1]=="-":
   369	                            fontSize = self.fontSize = self.fontSize - size
   370	                        else:
   371	                            fontSize = self.fontSize = size
   372	                    else:
   373	                        fontSize = self.fontSize = size
   374	                    spacewidth = stringWidth(" ", fontName, fontSize) #self.sizeOfWord(" ")
   375	                    line.append(opcode)
   376	                elif indicator=="leading":
   377	                    # change font leading
   378	                    (i, leading) = opcode
   379	                    self.leading = leading
   380	                    line.append(opcode)
   381	                elif indicator=="indent":
   382	                    # increase the indent
   383	                    (i, increment) = opcode
   384	                    indent = self.indent = self.indent + increment
   385	                    if first:
   386	                        usedIndent = max(indent, usedIndent)
   387	                        maxLength = totalLength - usedIndent - self.rightIndent
   388	                    line.append(opcode)
   389	                elif indicator=="push":
   390	                    self.pushTextState()
   391	                    line.append(opcode)
   392	                elif indicator=="pop":
   393	                    try:
   394	                        self.popTextState()
   395	                    except:
   396	##                        print "stack fault near", cursor
   397	##                        for i in program[max(0, cursor-10):cursor+10]:
   398	##                            if i==cursor:
   399	##                                print "***>>>",
   400	##                            print i
   401	                        raise
   402	                    fontName = self.fontName
   403	                    fontSize = self.fontSize
   404	                    spacewidth = stringWidth(" ", fontName, fontSize) #self.sizeOfWord(" ")
   405	                    line.append(opcode)
   406	                elif indicator=="bullet":
   407	                    (i, bullet, indent, font, size) = opcode
   408	                    # adjust for base indent (only at format time -- only execute once)
   409	                    indent = indent + self.baseindent
   410	                    opcode = (i, bullet, indent, font, size)
   411	                    if not first:
   412	                        raise ValueError("bullet not at beginning of line")
   413	                    bulletwidth = float(stringWidth(bullet, font, size))
   414	                    spacewidth = float(stringWidth(" ", font, size))
   415	                    bulletmin = indent+spacewidth+bulletwidth
   416	                    # decrease the line size to allow bullet
   417	                    usedIndent = max(bulletmin, usedIndent)
   418	                    if first:
   419	                        maxLength = totalLength - usedIndent - self.rightIndent
   420	                    line.append(opcode)
   421	                elif indicator=="rightIndent":
   422	                    # increase the right indent
   423	                    (i, increment) = opcode
   424	                    self.rightIndent = self.rightIndent+increment
   425	                    if first:
   426	                        maxLength = totalLength - usedIndent - self.rightIndent
   427	                    line.append(opcode)
   428	                elif indicator=="rise":
   429	                    (i, rise) = opcode
   430	                    newrise = self.rise = self.rise+rise
   431	                    line.append(opcode)
   432	                elif indicator=="align":
   433	                    (i, alignment) = opcode
   434	                    #if debug:
   435	                    #    print "SETTING ALIGNMENT", alignment
   436	                    self.alignment = alignment
   437	                    line.append(opcode)
   438	                elif indicator=="lineOperation":
   439	                    (i, handler) = opcode
   440	                    line.append(opcode)
   441	                    self.lineOpHandlers = self.lineOpHandlers + [handler] # fresh copy
   442	                elif indicator=="endLineOperation":
   443	                    (i, handler) = opcode
   444	                    h = self.lineOpHandlers[:] # fresh copy
   445	                    h.remove(handler)
   446	                    self.lineOpHandlers = h
   447	                    line.append(opcode)
   448
   449	                else:
   450	                    raise ValueError("at format time don't understand indicator "+repr(indicator))
   451	            else:
   452	                raise ValueError("op must be string, float, instance, or tuple "+repr(opcode))
   453	            if not done:
   454	                cursor += 1
   455	                #first = 0
   456	##            if debug:
   457	##                if done:
   458	##                    print "DONE FLAG IS SET"
   459	##                if cursor>=maxcursor:
   460	##                    print "AT END OF PROGRAM"
   461	        if not terminated:
   462	            line.append( ("nextLine", 0) )
   463	        #print "fitline", line
   464	        return (lineIsFull, line, cursor, currentLength, usedIndent, maxLength, justStrings)
