  3728	        self,

  3739	        (and None for equality checks).

  3746	            non-overlapping, although technically None is overlap, it is most
  3747	            likely an error.

  3752	            return False
  3753

  3755	            seen_types = set()

  3757	            return False
  3758	        seen_types.add((left, right))
  3759
  3760	        left, right = get_proper_types((left, right))
  3761

  3763	        # side. User defined (or even standard library) classes can define this
  3764	        # to return True for comparisons between non-overlapping types.

  3766	            return False

  3781	            #     T = TypeVar('T', str, int)

  3788	            return False

  3790	            return False
  3791	        if isinstance(left, UnionType) and isinstance(right, UnionType):
  3792	            left = remove_optional(left)
  3793	            right = remove_optional(right)
  3794	            left, right = get_proper_types((left, right))
