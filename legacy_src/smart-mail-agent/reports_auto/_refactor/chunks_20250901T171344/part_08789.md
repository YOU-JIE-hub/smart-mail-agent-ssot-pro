   277
   278			ast.Slice nodes are only allowed within a ast.Subscript node.
   279			To use a slice as an argument of ast.Call it has to be converted.
   280			Conversion is done by calling the 'slice' function from builtins
   281			"""
   282
   283			if isinstance(slice_, ast.Index):
   284				return slice_.value
   285
   286			elif isinstance(slice_, ast.Slice):
   287				# Create a python slice object.
   288				args = []
   289
   290				if slice_.lower:
   291					args.append(slice_.lower)
   292				else:
   293					args.append(self.gen_none_node())
   294
   295				if slice_.upper:
   296					args.append(slice_.upper)
   297				else:

   316			elif isPy39:

   328			# In python3 arguments are always identifiers.
   329			# In python2 the 'Python.asdl' specifies expressions, but
   330			# the python grammer allows only identifiers or a tuple of
   331			# identifiers. If its a tuple 'tuple parameter unpacking' is used,

   335			for arg in node.args.args:
   336				self.isAllowedName(node, arg.arg)

   338			if node.args.vararg:
   339				self.isAllowedName(node, node.args.vararg.arg)
   340
   341			if node.args.kwarg:
   342				self.isAllowedName(node, node.args.kwarg.arg)

   344			for arg in node.args.kwonlyargs:
   345				self.isAllowedName(node, arg.arg)

   406		def visit_Call(self, node):
   407			"""Checks calls with '*args' and '**kwargs'.
   408
   409			Note: The following happens only if '*args' or '**kwargs' is used.
   410
   411			Transfroms 'foo(<all the possible ways of args>)' into
   412			__rl_apply__(foo, <all the possible ways for args>)
   413
   414			The thing is that '__rl_apply__' has only '*args', '**kwargs', so it gets
   415			Python to collapse all the myriad ways to call functions
   416			into one manageable from.
   417
   418			From there, '__rl_apply__()' wraps args and kws in guarded accessors,
   419			then calls the function, returning the value.
   420			"""
   421
   422			if isinstance(node.func, ast.Name):
   423				if node.func.id == 'exec':
   424					self.error(node, 'Exec calls are not allowed.')
   425				elif node.func.id == 'eval':
   426					self.error(node, 'Eval calls are not allowed.')
   427
   428			needs_wrap = False
   429
   430			for pos_arg in node.args:
   431				if isinstance(pos_arg, ast.Starred):
   432					needs_wrap = True
   433
   434			for keyword_arg in node.keywords:
   435				if keyword_arg.arg is None:
   436					needs_wrap = True
   437
   438			node = self.visit_children(node)
   439
   440			#if not needs_wrap:
   441			#	return node
   442
   443			node.args.insert(0, node.func)
   444			node.func = ast.Name('__rl_apply__', ast.Load())
   445			copy_locations(node.func, node.args[0])
   446			return node

   484
   485			'v[a]' becomes '__rl_getitem__(foo, a)'
   486			'v[:b]' becomes '__rl_getitem__(foo, slice(None, b, None))'
   487			'v[a:]' becomes '__rl_getitem__(foo, slice(a, None, None))'
   488			'v[a:b]' becomes '__rl_getitem__(foo, slice(a, b, None))'
   489			'v[a:b:c]' becomes '__rl_getitem__(foo, slice(a, b, c))'

   503					func=ast.Name('__rl_getitem__', ast.Load()),
   504					args=[node.value, self.transform_slice(node.slice)],
   505					keywords=[])
   506
   507				copy_locations(new_node, node)

   526			node = self.visit_children(node)
   527
   528			if not any(isinstance(t, ast.Tuple) for t in node.targets):
   529				return node
   530
   531			# Handle sequence unpacking.
   532			# For briefness this example omits cleanup of the temporary variables.
   533			# Check 'transform_tuple_assign' how its done.
   534			#
   535			# - Single target (with nested support)
   536			# (a, (b, (c, d))) = <exp>
   537			# is converted to
   538			# (a, t1) = __rl_getiter__(<exp>)
   539			# (b, t2) = __rl_getiter__(t1)
   540			# (c, d) = __rl_getiter__(t2)
   541			#
   542			# - Multi targets
   543			# (a, b) = (c, d) = <exp>
   544			# is converted to
   545			# (c, d) = __rl_getiter__(<exp>)
   546			# (a, b) = __rl_getiter__(<exp>)
   547			# Why is this valid ? The original bytecode for this multi targets
   548			# behaves the same way.
   549
   550			# ast.NodeTransformer works with list results.
   551			# He injects it at the rightplace of the node's parent statements.
   552			new_nodes = []
   553
   554			# python fills the right most target first.
   555			for target in reversed(node.targets):
   556				if isinstance(target, ast.Tuple):
   557					wrapper = ast.Assign(
   558						targets=[target],
   559						value=self.protect_unpack_sequence(target, node.value))
   560					new_nodes.append(wrapper)
   561				else:
   562					new_node = ast.Assign(targets=[target], value=node.value)
   563					new_nodes.append(new_node)
   564
   565			for new_node in new_nodes:
   566				copy_locations(new_node, node)
   567
   568			return new_nodes

   573			According to the language reference (and ast.c) the following nodes

   575			Name, Attribute, Subscript

   577			Note that although augmented assignment of attributes and
