   341	        builder.goto(next_block)
   342

   424	    builder: IRBuilder,
   425	    gen: GeneratorExpr,
   426	    initial_value: Callable[[], Value],
   427	    modify: Callable[[Value], Value],

   441	        builder.activate_block(false_block)

   451	    # specialized implementation is used if:
   452	    # - only one or two arguments given (if not, sum() has been given invalid arguments)
   453	    # - first argument is a Generator (there is no benefit to optimizing the performance of eg.
   454	    #   sum([1, 2, 3]), so non-Generator Iterables are not handled)

   505	@specialize_function("builtins.next")
   506	def translate_next_call(builder: IRBuilder, expr: CallExpr, callee: RefExpr) -> Value | None:
   507	    """Special case for calling next() on a generator expression, an
   508	    idiom that shows up some in mypy.
   509
   510	    For example, next(x for x in l if x.id == 12, None) will
   511	    generate code that searches l for an element where x.id == 12
   512	    and produce the first such object, or None if no such element
   513	    exists.
   514	    """
   515	    if not (
   516	        expr.arg_kinds in ([ARG_POS], [ARG_POS, ARG_POS])
   517	        and isinstance(expr.args[0], GeneratorExpr)
   518	    ):
   519	        return None
   520
   521	    gen = expr.args[0]
   522	    retval = Register(builder.node_type(expr))
   523	    default_val = builder.accept(expr.args[1]) if len(expr.args) > 1 else None
   524	    exit_block = BasicBlock()
   525
   526	    def gen_inner_stmts() -> None:
   527	        # next takes the first element of the generator, so if
   528	        # something gets produced, we are done.
   529	        builder.assign(retval, builder.accept(gen.left_expr), gen.left_expr.line)
   530	        builder.goto(exit_block)
   531
   532	    loop_params = list(zip(gen.indices, gen.sequences, gen.condlists, gen.is_async))
   533	    comprehension_helper(builder, loop_params, gen_inner_stmts, gen.line)
   534
   535	    # Now we need the case for when nothing got hit. If there was
   536	    # a default value, we produce it, and otherwise we raise
   537	    # StopIteration.
   538	    if default_val:
   539	        builder.assign(retval, default_val, gen.left_expr.line)
   540	        builder.goto(exit_block)
   541	    else:
   542	        builder.add(RaiseStandardError(RaiseStandardError.STOP_ITERATION, None, expr.line))
   543	        builder.add(Unreachable())
   544
   545	    builder.activate_block(exit_block)
   546	    return retval
