   289	    def _open_db(self) -> None:
   290	        """Open an existing db file, and read its metadata."""
   291	        self._debug_dataio("Opening data file", self._filename)
   292	        self._dbs[threading.get_ident()] = SqliteDb(self._filename, self._debug)
   293	        self._read_db()

   295	    def _read_db(self) -> None:
   296	        """Read the metadata from a database so that we are ready to use it."""
   297	        with self._dbs[threading.get_ident()] as db:
   298	            try:
   299	                row = db.execute_one("select version from coverage_schema")
   300	                assert row is not None
   301	            except Exception as exc:
   302	                if "no such table: coverage_schema" in str(exc):
   303	                    self._init_db(db)
   304	                else:
   305	                    raise DataError(
   306	                        "Data file {!r} doesn't seem to be a coverage data file: {}".format(
   307	                            self._filename, exc,
   308	                        ),
   309	                    ) from exc
   310	            else:
   311	                schema_version = row[0]
   312	                if schema_version != SCHEMA_VERSION:
   313	                    raise DataError(
   314	                        "Couldn't use data file {!r}: wrong schema: {} instead of {}".format(
   315	                            self._filename, schema_version, SCHEMA_VERSION,
   316	                        ),
   317	                    )
   318
   319	            row = db.execute_one("select value from meta where key = 'has_arcs'")
   320	            if row is not None:
   321	                self._has_arcs = bool(int(row[0]))
   322	                self._has_lines = not self._has_arcs
   323
   324	            with db.execute("select id, path from file") as cur:
   325	                for file_id, path in cur:
   326	                    self._file_map[path] = file_id

   328	    def _init_db(self, db: SqliteDb) -> None:
   329	        """Write the initial contents of the database."""
   330	        self._debug_dataio("Initing data file", self._filename)
   331	        db.executescript(SCHEMA)
   332	        db.execute_void("insert into coverage_schema (version) values (?)", (SCHEMA_VERSION,))
   333
   334	        # When writing metadata, avoid information that will needlessly change
   335	        # the hash of the data file, unless we're debugging processes.
   336	        meta_data = [
   337	            ("version", __version__),
   338	        ]
   339	        if self._debug.should("process"):
   340	            meta_data.extend([
   341	                ("sys_argv", str(getattr(sys, "argv", None))),
   342	                ("when", datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")),
   343	            ])
   344	        db.executemany_void("insert or ignore into meta (key, value) values (?, ?)", meta_data)
   345
   346	    def _connect(self) -> SqliteDb:
   347	        """Get the SqliteDb object to use."""
   348	        if threading.get_ident() not in self._dbs:
   349	            self._open_db()
   350	        return self._dbs[threading.get_ident()]

   352	    def __bool__(self) -> bool:
   353	        if (threading.get_ident() not in self._dbs and not os.path.exists(self._filename)):
   354	            return False
   355	        try:
   356	            with self._connect() as con:
   357	                with con.execute("select * from file limit 1") as cur:
   358	                    return bool(list(cur))
   359	        except CoverageException:
   360	            return False

   362	    def dumps(self) -> bytes:
   363	        """Serialize the current data to a byte string.
   364
   365	        The format of the serialized data is not documented. It is only
   366	        suitable for use with :meth:`loads` in the same version of
   367	        coverage.py.
   368
   369	        Note that this serialization is not what gets stored in coverage data
   370	        files.  This method is meant to produce bytes that can be transmitted
   371	        elsewhere and then deserialized with :meth:`loads`.
   372
   373	        Returns:
   374	            A byte string of serialized data.
   375
   376	        .. versionadded:: 5.0
   377
   378	        """
   379	        self._debug_dataio("Dumping data from data file", self._filename)
   380	        with self._connect() as con:
   381	            script = con.dump()
   382	            return b"z" + zlib.compress(script.encode("utf-8"))
   383
   384	    def loads(self, data: bytes) -> None:
   385	        """Deserialize data from :meth:`dumps`.
   386
   387	        Use with a newly-created empty :class:`CoverageData` object.  It's
   388	        undefined what happens if the object already has data in it.
   389
   390	        Note that this is not for reading data from a coverage data file.  It
   391	        is only for use on data you produced with :meth:`dumps`.
   392
   393	        Arguments:
   394	            data: A byte string of serialized data produced by :meth:`dumps`.
   395
   396	        .. versionadded:: 5.0
   397
   398	        """
   399	        self._debug_dataio("Loading data into data file", self._filename)
   400	        if data[:1] != b"z":
   401	            raise DataError(
   402	                f"Unrecognized serialization: {data[:40]!r} (head of {len(data)} bytes)",
   403	            )
   404	        script = zlib.decompress(data[1:]).decode("utf-8")
   405	        self._dbs[threading.get_ident()] = db = SqliteDb(self._filename, self._debug)
   406	        with db:
   407	            db.executescript(script)
   408	        self._read_db()
   409	        self._have_used = True

   411	    def _file_id(self, filename: str, add: bool = False) -> int | None:
   412	        """Get the file id for `filename`.
   413
   414	        If filename is not in the database yet, add it if `add` is True.
   415	        If `add` is not True, return None.
   416	        """
   417	        if filename not in self._file_map:
   418	            if add:
   419	                with self._connect() as con:
   420	                    self._file_map[filename] = con.execute_for_rowid(
   421	                        "insert or replace into file (path) values (?)",
   422	                        (filename,),
   423	                    )
   424	        return self._file_map.get(filename)

   481	    @_locked
   482	    def add_lines(self, line_data: Mapping[str, Collection[TLineNo]]) -> None:
   483	        """Add measured line data.
   484
   485	        `line_data` is a dictionary mapping file names to iterables of ints::
   486
   487	            { filename: { line1, line2, ... }, ...}
   488
   489	        """
   490	        if self._debug.should("dataop"):
   491	            self._debug.write("Adding lines: %d files, %d lines total" % (
   492	                len(line_data), sum(len(lines) for lines in line_data.values()),
   493	            ))
   494	            if self._debug.should("dataop2"):
   495	                for filename, linenos in sorted(line_data.items()):
   496	                    self._debug.write(f"  {filename}: {linenos}")
   497	        self._start_using()
   498	        self._choose_lines_or_arcs(lines=True)
   499	        if not line_data:
   500	            return
   501	        with self._connect() as con:
   502	            self._set_context_id()
   503	            for filename, linenos in line_data.items():
   504	                line_bits = nums_to_numbits(linenos)
   505	                file_id = self._file_id(filename, add=True)
   506	                query = "select numbits from line_bits where file_id = ? and context_id = ?"
   507	                with con.execute(query, (file_id, self._current_context_id)) as cur:
   508	                    existing = list(cur)
   509	                if existing:
   510	                    line_bits = numbits_union(line_bits, existing[0][0])
   511
   512	                con.execute_void(
   513	                    "insert or replace into line_bits " +
   514	                    " (file_id, context_id, numbits) values (?, ?, ?)",
   515	                    (file_id, self._current_context_id, line_bits),
   516	                )

   518	    @_locked
   519	    def add_arcs(self, arc_data: Mapping[str, Collection[TArc]]) -> None:
   520	        """Add measured arc data.
   521
   522	        `arc_data` is a dictionary mapping file names to iterables of pairs of
   523	        ints::
   524
   525	            { filename: { (l1,l2), (l1,l2), ... }, ...}
   526
   527	        """
   528	        if self._debug.should("dataop"):
   529	            self._debug.write("Adding arcs: %d files, %d arcs total" % (
   530	                len(arc_data), sum(len(arcs) for arcs in arc_data.values()),
   531	            ))
   532	            if self._debug.should("dataop2"):
   533	                for filename, arcs in sorted(arc_data.items()):
   534	                    self._debug.write(f"  {filename}: {arcs}")
   535	        self._start_using()
   536	        self._choose_lines_or_arcs(arcs=True)
   537	        if not arc_data:
   538	            return
   539	        with self._connect() as con:
   540	            self._set_context_id()
   541	            for filename, arcs in arc_data.items():
   542	                if not arcs:
   543	                    continue
   544	                file_id = self._file_id(filename, add=True)
   545	                data = [(file_id, self._current_context_id, fromno, tono) for fromno, tono in arcs]
   546	                con.executemany_void(
   547	                    "insert or ignore into arc " +
   548	                    "(file_id, context_id, fromno, tono) values (?, ?, ?, ?)",
   549	                    data,
   550	                )

   552	    def _choose_lines_or_arcs(self, lines: bool = False, arcs: bool = False) -> None:
   553	        """Force the data file to choose between lines and arcs."""
   554	        assert lines or arcs
   555	        assert not (lines and arcs)
   556	        if lines and self._has_arcs:
   557	            if self._debug.should("dataop"):
   558	                self._debug.write("Error: Can't add line measurements to existing branch data")
   559	            raise DataError("Can't add line measurements to existing branch data")
   560	        if arcs and self._has_lines:
   561	            if self._debug.should("dataop"):
   562	                self._debug.write("Error: Can't add branch measurements to existing line data")
   563	            raise DataError("Can't add branch measurements to existing line data")
   564	        if not self._has_arcs and not self._has_lines:
   565	            self._has_lines = lines
   566	            self._has_arcs = arcs
   567	            with self._connect() as con:
   568	                con.execute_void(
   569	                    "insert or ignore into meta (key, value) values (?, ?)",
   570	                    ("has_arcs", str(int(arcs))),
   571	                )

   573	    @_locked
   574	    def add_file_tracers(self, file_tracers: Mapping[str, str]) -> None:
   575	        """Add per-file plugin information.
   576
   577	        `file_tracers` is { filename: plugin_name, ... }
   578
   579	        """
   580	        if self._debug.should("dataop"):
   581	            self._debug.write("Adding file tracers: %d files" % (len(file_tracers),))
   582	        if not file_tracers:
   583	            return
   584	        self._start_using()
   585	        with self._connect() as con:
   586	            for filename, plugin_name in file_tracers.items():
   587	                file_id = self._file_id(filename, add=True)
   588	                existing_plugin = self.file_tracer(filename)
   589	                if existing_plugin:
   590	                    if existing_plugin != plugin_name:
   591	                        raise DataError(
   592	                            "Conflicting file tracer name for '{}': {!r} vs {!r}".format(
   593	                                filename, existing_plugin, plugin_name,
   594	                            ),
   595	                        )
   596	                elif plugin_name:
   597	                    con.execute_void(
   598	                        "insert into tracer (file_id, tracer) values (?, ?)",
   599	                        (file_id, plugin_name),
   600	                    )

   610	    def touch_files(self, filenames: Collection[str], plugin_name: str | None = None) -> None:
   611	        """Ensure that `filenames` appear in the data, empty if needed.
   612
   613	        `plugin_name` is the name of the plugin responsible for these files.
   614	        It is used to associate the right filereporter, etc.
   615	        """
   616	        if self._debug.should("dataop"):
   617	            self._debug.write(f"Touching {filenames!r}")
   618	        self._start_using()
   619	        with self._connect(): # Use this to get one transaction.
   620	            if not self._has_arcs and not self._has_lines:
   621	                raise DataError("Can't touch files in an empty CoverageData")
   622
   623	            for filename in filenames:
   624	                self._file_id(filename, add=True)
   625	                if plugin_name:
   626	                    # Set the tracer for this file
   627	                    self.add_file_tracers({filename: plugin_name})

   629	    def purge_files(self, filenames: Collection[str]) -> None:
   630	        """Purge any existing coverage data for the given `filenames`.
   631
   632	        .. versionadded:: 7.2
   633
   634	        """
   635	        if self._debug.should("dataop"):
   636	            self._debug.write(f"Purging data for {filenames!r}")
   637	        self._start_using()
   638	        with self._connect() as con:
   639
   640	            if self._has_lines:
   641	                sql = "delete from line_bits where file_id=?"
   642	            elif self._has_arcs:
   643	                sql = "delete from arc where file_id=?"
   644	            else:
   645	                raise DataError("Can't purge files in an empty CoverageData")
   646
   647	            for filename in filenames:
   648	                file_id = self._file_id(filename, add=False)
   649	                if file_id is None:
   650	                    continue
   651	                con.execute_void(sql, (file_id,))
