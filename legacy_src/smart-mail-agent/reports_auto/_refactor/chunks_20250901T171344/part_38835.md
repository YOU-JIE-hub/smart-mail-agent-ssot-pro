   626	    def showPage(self):
   627	        """Close the current page and possibly start on a new page."""
   628	        # ensure a space at the end of the stream - Acrobat does
   629	        # not mind, but Ghostscript dislikes 'Qendstream' even if
   630	        # the length marker finishes after 'Q'
   631
   632	        pageWidth = self._pagesize[0]
   633	        pageHeight = self._pagesize[1]
   634	        cM = self._cropMarks
   635	        code = self._code
   636	        if cM:
   637	            bw = max(0,getattr(cM,'borderWidth',36))
   638	            if bw:
   639	                markLast = getattr(cM,'markLast',1)
   640	                ml = min(bw,max(0,getattr(cM,'markLength',18)))
   641	                mw = getattr(cM,'markWidth',0.5)
   642	                mc = getattr(cM,'markColor',black)
   643	                mg = 2*bw-ml
   644	                cx0 = len(code)
   645	                if ml and mc:
   646	                    self.saveState()
   647	                    self.setStrokeColor(mc)
   648	                    self.setLineWidth(mw)
   649	                    self.lines([
   650	                        (bw,0,bw,ml),
   651	                        (pageWidth+bw,0,pageWidth+bw,ml),
   652	                        (bw,pageHeight+mg,bw,pageHeight+2*bw),
   653	                        (pageWidth+bw,pageHeight+mg,pageWidth+bw,pageHeight+2*bw),
   654	                        (0,bw,ml,bw),
   655	                        (pageWidth+mg,bw,pageWidth+2*bw,bw),
   656	                        (0,pageHeight+bw,ml,pageHeight+bw),
   657	                        (pageWidth+mg,pageHeight+bw,pageWidth+2*bw,pageHeight+bw),
   658	                        ])
   659	                    self.restoreState()
   660	                    if markLast:
   661	                        #if the marks are to be drawn after the content
   662	                        #save the code we just drew for later use
   663	                        L = code[cx0:]
   664	                        del code[cx0:]
   665	                        cx0 = len(code)
   666
   667	                bleedW = max(0,getattr(cM,'bleedWidth',0))
   668	                self.saveState()
   669	                self.translate(bw-bleedW,bw-bleedW)
   670	                if bleedW:
   671	                    #scale everything
   672	                    self.scale(1+(2.0*bleedW)/pageWidth,1+(2.0*bleedW)/pageHeight)
   673
   674	                #move our translation/expansion code to the beginning
   675	                C = code[cx0:]
   676	                del code[cx0:]
   677	                code[0:0] = C
   678	                self.restoreState()
   679	                if markLast:
   680	                    code.extend(L)
   681	                pageWidth = 2*bw + pageWidth
   682	                pageHeight = 2*bw + pageHeight
   683
   684	        code.append(' ')
   685	        page = pdfdoc.PDFPage()
   686	        page.pagewidth = pageWidth
   687	        page.pageheight = pageHeight
   688	        page.Rotate = self._pageRotation
   689	        page.hasImages = self._currentPageHasImages
   690	        page.setPageTransition(self._pageTransition)
   691	        page.setCompression(self._pageCompression)
   692	        for box in ('crop','art','bleed','trim'):

   694	            if size:
   695	                setattr(page,box.capitalize()+'Box',pdfdoc.PDFArray(size))
   696	        if self._pa
