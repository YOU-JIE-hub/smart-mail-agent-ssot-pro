  3453	    def visit_op_expr(self, e: OpExpr) -> Type:

  3455	        if e.analyzed:
  3456	            # It's actually a type expression X | Y.
  3457	            return self.accept(e.analyzed)
  3458	        if e.op == "and" or e.op == "or":
  3459	            return self.check_boolean_op(e, e)
  3460	        if e.op == "*" and isinstance(e.left, ListExpr):
  3461	            # Expressions of form [...] * e get special type inference.
  3462	            return self.check_list_multiply(e)

  3464	            if isinstance(e.left, BytesExpr):

  3472	            left_add_method = proper_left_type.partial_fallback.type.get("__add__")
  3473	            if left_add_method and left_add_method.fullname == "builtins.tuple.__add__":
  3474	                proper_right_type = get_proper_type(self.accept(e.right))
  3475	                if isinstance(proper_right_type, TupleType):
  3476	                    right_radd_method = proper_right_type.partial_fallback.type.get("__radd__")
  3477	                    if right_radd_method is None:
  3478	                        # One cannot have two variadic items in the same tuple.
  3479	                        if (
  3480	                            find_unpack_in_list(proper_left_type.items) is None
  3481	                            or find_unpack_in_list(proper_right_type.items) is None
  3482	                        ):

  3486	                    and isinstance(proper_right_type, Instance)
  3487	                    and self.chk.type_is_iterable(proper_right_type)
  3488	                ):
  3489	                    # Handle tuple[X, Y] + tuple[Z, ...] = tuple[X, Y, *tuple[Z, ...]].
  3490	                    right_radd_method = proper_right_type.type.get("__radd__")
  3491	                    if (
  3492	                        right_radd_method is None
  3493	                        and proper_left_type.partial_fallback.type.fullname == "builtins.tuple"
  3494	                        and find_unpack_in_list(proper_left_type.items) is None
  3495	                    ):
  3496	                        item_type = self.chk.iterable_item_type(proper_right_type, e)
  3497	                        mapped = self.chk.named_generic_type("builtins.tuple", [item_type])
  3498	                        return proper_left_type.copy_modified(
  3499	                            items=proper_left_type.items + [UnpackType(mapped)]
  3500	                        )
  3501
  3502	        use_reverse: UseReverse = USE_REVERSE_DEFAULT
  3503	        if e.op == "|":
  3504	            if is_named_instance(proper_left_type, "builtins.dict"):
  3505	                # This is a special case for `dict | TypedDict`.
  3506	                # 1. Find `dict | TypedDict` case
  3507	                # 2. Switch `dict.__or__` to `TypedDict.__ror__` (the same from both runtime and typing perspective)
  3508	                proper_right_type = get_proper_type(self.accept(e.right))
  3509	                if isinstance(proper_right_type, TypedDictType):
  3510	                    use_reverse = USE_REVERSE_ALWAYS
  3511	            if isinstance(proper_left_type, TypedDictType):
  3512	                # This is the reverse case: `TypedDict | dict`,
  3513	                # simply do not allow the reverse checking:
  3514	                # do not call `__dict__.__ror__`.
  3515	                proper_right_type = get_proper_type(self.accept(e.right))
  3516	                if is_named_instance(proper_right_type, "builtins.dict"):
  3517	                    use_reverse = USE_REVERSE_NEVER
  3518
  3519	        if PRECISE_TUPLE_TYPES in self.chk.options.enable_incomplete_feature:
  3520	            # Handle tuple[X, ...] + tuple[Y, Z] = tuple[*tuple[X, ...], Y, Z].
  3521	            if (
  3522	                e.op == "+"
  3523	                and isinstance(proper_left_type, Instance)
  3524	                and proper_left_type.type.fullname == "builtins.tuple"
  3525	            ):
  3526	                proper_right_type = get_proper_type(self.accept(e.right))
  3527	                if (
  3528	                    isinstance(proper_right_type, TupleType)
  3529	                    and proper_right_type.partial_fallback.type.fullname == "builtins.tuple"
  3530	                    and find_unpack_in_list(proper_right_type.items) is None
  3531	                ):

  3533	                        items=[UnpackType(proper_left_type)] + proper_right_type.items
  3534	                    )
  3535
  3536	        if e.op in operators.op_methods:
  3537	            method = operators.op_methods[e.op]
  3538	            if use_reverse is UseReverse.DEFAULT or use_reverse is UseReverse.NEVER:
  3539	                result, method_type = self.check_op(
  3540	                    method,
  3541	                    base_type=left_type,
  3542	                    arg=e.right,
  3543	                    context=e,
  3544	                    allow_reverse=use_reverse is UseReverse.DEFAULT,
  3545	                )
  3546	            elif use_reverse is UseReverse.ALWAYS:
  3547	                result, method_type = self.check_op(
  3548	                    # The reverse operator here gives better error messages:
  3549	                    operators.reverse_op_methods[method],
  3550	                    base_type=self.accept(e.right),
  3551	                    arg=e.left,
  3552	                    context=e,
  3553	                    allow_reverse=False,
  3554	                )
  3555	            else:
  3556	                assert_never(use_reverse)
  3557	            e.method_type = method_type
  3558	            return result
  3559	        else:
  3560	            raise RuntimeError(f"Unknown operator {e.op}")

  3564
  3565	        Comparison expressions are type checked consecutive-pair-wise
  3566	        That is, 'a < b > c == d' is check as 'a < b and b > c and c == d'
  3567	        """
  3568	        result: Type | None = None
  3569	        sub_result: Type
  3570
  3571	        # Check each consecutive operand pair and their operator
  3572	        for left, right, operator in zip(e.operands, e.operands[1:], e.operators):
  3573	            left_type = self.accept(left)
  3574
  3575	            if operator == "in" or operator == "not in":
  3576	                # This case covers both iterables and containers, which have different meanings.
  3577	                # For a container, the in operator calls the __contains__ method.
  3578	                # For an iterable, the in operator iterates over the iterable, and compares each item one-by-one.
  3579	                # We allow `in` for a union of containers and iterables as long as at least one of them matches the
  3580	                # type of the left operand, as the operation will simply return False if the union's container/iterator
  3581	                # type doesn't match the left operand.

  3584	                # a "Need type annotation ..." message, as it would be noise.

  3586	                if right_type is None:
  3587	                    right_type = self.accept(right)  # Validate the right operand
  3588
  3589	                right_type = get_proper_type(right_type)
  3590	                item_types: Sequence[Type] = [right_type]
  3591	                if isinstance(right_type, UnionType):
  3592	                    item_types = list(right_type.relevant_items())
  3593
  3594	                sub_result = self.bool_type()

  3622	                        pass

  3626	                        # is_valid_var_arg is True for any Iterable

  3632	                                item_type, e
  3633	                            )

  3636	                            failed_out = True
  3637	                        else:

  3639	                                [left_type],
  3640	                                [nodes.ARG_POS],

  3643	                                self.named_type("builtins.function"),

  3646	                            iterable_types.append(itertype)

  3648	                        container_types.append(cont_type)

  3689
  3690	            elif operator == "is" or operator == "is not":

  3692	                sub_result = self.bool_type()

  3694	                    # Show the most specific literal types possible
  3695	                    left_type = try_getting_literal(left_type)

  3703	            if result is None:
