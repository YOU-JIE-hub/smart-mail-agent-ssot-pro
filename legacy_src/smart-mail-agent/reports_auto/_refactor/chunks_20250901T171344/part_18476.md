  5112	    def process_module_assignment(
  5113	        self, lvals: list[Lvalue], rval: Expression, ctx: AssignmentStmt
  5114	    ) -> None:
  5115	        """Propagate module references across assignments.
  5116
  5117	        Recursively handles the simple form of iterable unpacking; doesn't
  5118	        handle advanced unpacking with *rest, dictionary unpacking, etc.
  5119
  5120	        In an expression like x = y = z, z is the rval and lvals will be [x,
  5121	        y].
  5122
  5123	        """
  5124	        if isinstance(rval, (TupleExpr, ListExpr)) and all(
  5125	            isinstance(v, TupleExpr) for v in lvals
  5126	        ):
  5127	            # rval and all lvals are either list or tuple, so we are dealing
  5128	            # with unpacking assignment like `x, y = a, b`. Mypy didn't
  5129	            # understand our all(isinstance(...)), so cast them as TupleExpr
  5130	            # so mypy knows it is safe to access their .items attribute.
  5131	            seq_lvals = cast(list[TupleExpr], lvals)
  5132	            # given an assignment like:
  5133	            #     (x, y) = (m, n) = (a, b)
  5134	            # we now have:
  5135	            #     seq_lvals = [(x, y), (m, n)]
  5136	            #     seq_rval = (a, b)
  5137	            # We now zip this into:
  5138	            #     elementwise_assignments = [(a, x, m), (b, y, n)]
  5139	            # where each elementwise assignment includes one element of rval and the
  5140	            # corresponding element of each lval. Basically we unpack
  5141	            #     (x, y) = (m, n) = (a, b)
  5142	            # into elementwise assignments
  5143	            #     x = m = a
  5144	            #     y = n = b
  5145	            # and then we recursively call this method for each of those assignments.
  5146	            # If the rval and all lvals are not all of the same length, zip will just ignore
  5147	            # extra elements, so no error will be raised here; mypy will later complain
  5148	            # about the length mismatch in type-checking.
  5149	            elementwise_assignments = zip(rval.items, *[v.items for v in seq_lvals])
  5150	            for rv, *lvs in elementwise_assignments:
  5151	                self.process_module_assignment(lvs, rv, ctx)
  5152	        elif isinstance(rval, RefExpr):
  5153	            rnode = self.lookup_type_node(rval)
  5154	            if rnode and isinstance(rnode.node, MypyFile):
  5155	                for lval in lvals:
  5156	                    if not isinstance(lval, RefExpr):
  5157	                        continue
  5158	                    # respect explicitly annotated type
  5159	                    if isinstance(lval.node, Var) and lval.node.type is not None:
  5160	                        continue
  5161
  5162	                    # We can handle these assignments to locals and to self
  5163	                    if isinstance(lval, NameExpr):
  5164	                        lnode = self.current_symbol_table().get(lval.name)
  5165	                    elif isinstance(lval, MemberExpr) and self.is_self_member_ref(lval):
  5166	                        assert self.type is not None
  5167	                        lnode = self.type.names.get(lval.name)
  5168	                    else:
  5169	                        continue
  5170
  5171	                    if lnode:
  5172	                        if isinstance(lnode.node, MypyFile) and lnode.node is not rnode.node:
  5173	                            assert isinstance(lval, (NameExpr, MemberExpr))
  5174	                            self.fail(
  5175	                                'Cannot assign multiple modules to name "{}" '
  5176	                                'without explicit "types.ModuleType" annotation'.format(lval.name),
  5177	                                ctx,
  5178	                            )
  5179	                        # never create module alias except on initial var definition
  5180	                        elif lval.is_inferred_def:
  5181	                            assert rnode.node is not None
  5182	                            lnode.node = rnode.node

  5218	    def process__slots__(self, s: AssignmentStmt) -> None:
  5219	        """
  5220	        Processing ``__slots__`` if defined in type.
  5221
  5222	        See: https://docs.python.org/3/reference/datamodel.html#slots
  5223	        """
  5224	        # Later we can support `__slots__` defined as `__slots__ = other = ('a', 'b')`
  5225	        if (
  5226	            isinstance(self.type, TypeInfo)
  5227	            and len(s.lvalues) == 1
  5228	            and isinstance(s.lvalues[0], NameExpr)
  5229	            and s.lvalues[0].name == "__slots__"
  5230	            and s.lvalues[0].kind == MDEF
  5231	        ):
  5232	            # We understand `__slots__` defined as string, tuple, list, set, and dict:
  5233	            if not isinstance(s.rvalue, (StrExpr, ListExpr, TupleExpr, SetExpr, DictExpr)):
  5234	                # For example, `__slots__` can be defined as a variable,
  5235	                # we don't support it for now.
  5236	                return
  5237
  5238	            if any(p.slots is None for p in self.type.mro[1:-1]):
  5239	                # At least one type in mro (excluding `self` and `object`)
  5240	                # does not have concrete `__slots__` defined. Ignoring.
  5241	                return
  5242
  5243	            concrete_slots = True
  5244	            rvalue: list[Expression] = []
  5245	            if isinstance(s.rvalue, StrExpr):
  5246	                rvalue.append(s.rvalue)
  5247	            elif isinstance(s.rvalue, (ListExpr, TupleExpr, SetExpr)):
  5248	                rvalue.extend(s.rvalue.items)
  5249	            else:
  5250	                # We have a special treatment of `dict` with possible `{**kwargs}` usage.
  5251	                # In this case we consider all `__slots__` to be non-concrete.
  5252	                for key, _ in s.rvalue.items:
  5253	                    if concrete_slots and key is not None:
  5254	                        rvalue.append(key)
  5255	                    else:
  5256	                        concrete_slots = False
  5257
  5258	            slots = []
  5259	            for item in rvalue:
  5260	                # Special case for `'__dict__'` value:
  5261	                # when specified it will still allow any attribute assignment.
  5262	                if isinstance(item, StrExpr) and item.value != "__dict__":
  5263	                    slots.append(item.value)
  5264	                else:
  5265	                    concrete_slots = False
  5266	            if not concrete_slots:
  5267	                # Some slot items are dynamic, we don't want any false positives,
  5268	                # so, we just pretend that this type does not have any slots at all.
  5269	                return
  5270
  5271	            # We need to copy all slots from super types:
  5272	            for super_type in self.type.mro[1:-1]:
  5273	                assert super_type.slots is not None
  5274	                slots.extend(super_type.slots)
  5275	            self.type.slots = set(slots)
