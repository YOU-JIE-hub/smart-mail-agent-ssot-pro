   374	def write_and_fudge_mtime(content: str, target_path: str) -> None:
   375	    # In some systems, mtime has a resolution of 1 second which can
   376	    # cause annoying-to-debug issues when a file has the same size
   377	    # after a change. We manually set the mtime to circumvent this.
   378	    # Note that we increment the old file's mtime, which guarantees a
   379	    # different value, rather than incrementing the mtime after the
   380	    # copy, which could leave the mtime unchanged if the old file had
   381	    # a similarly fudged mtime.
   382	    new_time = None
   383	    if os.path.isfile(target_path):
   384	        new_time = os.stat(target_path).st_mtime + 1
   385
   386	    dir = os.path.dirname(target_path)
   387	    os.makedirs(dir, exist_ok=True)
   388	    with open(target_path, "w", encoding="utf-8") as target:
   389	        target.write(content)
   390
   391	    if new_time:
   392	        os.utime(target_path, times=(new_time, new_time))

   453	def normalize_file_output(content: list[str], current_abs_path: str) -> list[str]:
   454	    """Normalize file output for comparison."""
   455	    timestamp_regex = re.compile(r"\d{10}")
   456	    result = [x.replace(current_abs_path, "$PWD") for x in content]
   457	    version = mypy.version.__version__
   458	    result = [re.sub(r"\b" + re.escape(version) + r"\b", "$VERSION", x) for x in result]
   459	    # We generate a new mypy.version when building mypy wheels that
   460	    # lacks base_version, so handle that case.
   461	    base_version = getattr(mypy.version, "base_version", version)
   462	    result = [re.sub(r"\b" + re.escape(base_version) + r"\b", "$VERSION", x) for x in result]
   463	    result = [timestamp_regex.sub("$TIMESTAMP", x) for x in result]
   464	    return result

    26	    data_suite: str,
    27	    *,
    28	    data_file_prefix: str = "check",
    29	    pytest_node_prefix: str = "mypy/test/testcheck.py::TypeCheckSuite",
    30	    extra_args: Iterable[str],
    31	    max_attempts: int,
    32	) -> PytestResult:

    34	    Runs a suite of data test cases through pytest until either tests pass
    35	    or until a maximum number of attempts (needed for incremental tests).
    36
    37	    :param data_suite: the actual "suite" i.e. the contents of a .test file
    38	    """
    39	    p_test_data = Path(test_data_prefix)

    50	        for i in range(max_attempts - 1, -1, -1):
    51	            print(f">> {cmd}")
    52	            proc = subprocess.run(extra_args, capture_output=True, check=False, cwd=p_root)
    53	            if proc.returncode == 0:
    54	                break
    55	            prefix = "NESTED PYTEST STDOUT"
    56	            for line in proc.stdout.decode().splitlines():
    57	                print(f"{prefix}: {line}")
    58	                prefix = " " * len(prefix)
    59	            prefix = "NESTED PYTEST STDERR"
    60	            for line in proc.stderr.decode().splitlines():
    61	                print(f"{prefix}: {line}")
    62	                prefix = " " * len(prefix)
    63	            print(f"Exit code {proc.returncode} ({i} attempts remaining)")

    65	        return PytestResult(
    66	            input=data_suite,
    67	            input_updated=p.read_text(),

    69	            stderr=proc.stderr.decode(),
