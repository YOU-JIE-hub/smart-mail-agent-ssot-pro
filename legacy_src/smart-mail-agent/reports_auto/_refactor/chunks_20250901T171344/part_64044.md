   227	        self._scaleFactor = None
   228	        self.background = background

   230	    def _scale(self,availWidth,availHeight):
   231	        "Rescale to fit according to the rules, but only once"
   232	        if self._scaleFactor is None or self.width>availWidth or self.height>availHeight:
   233	            w, h = Figure.wrap(self, availWidth, availHeight)
   234	            captionHeight = h - self.figureHeight
   235	            if self.scaleFactor is None:
   236	                #scale factor None means auto
   237	                self._scaleFactor = min(availWidth/self.width,(availHeight-captionHeight)/self.figureHeight)
   238	            else: #they provided a factor
   239	                self._scaleFactor = self.scaleFactor
   240	            if self._scaleFactor<1 and self.shrinkToFit:
   241	                self.width = self.width * self._scaleFactor - 0.0001
   242	                self.figureHeight = self.figureHeight * self._scaleFactor
   243	            elif self._scaleFactor>1 and self.growToFit:
   244	                self.width = self.width*self._scaleFactor - 0.0001
   245	                self.figureHeight = self.figureHeight * self._scaleFactor
   246
   247	    def wrap(self, availWidth, availHeight):
   248	        self._scale(availWidth,availHeight)
   249	        return Figure.wrap(self, availWidth, availHeight)
