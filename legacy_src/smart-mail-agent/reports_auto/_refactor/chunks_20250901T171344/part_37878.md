
   171	        if self.xml.logging in ["system-out", "out-err", "all"]:
   172	            content_all += self._prepare_content(content_out, " Captured Out ")
   173	            self._write_content(report, content_all, "system-out")
   174	            content_all = ""
   175	        if self.xml.logging in ["system-err", "out-err", "all"]:
   176	            content_all += self._prepare_content(content_err, " Captured Err ")
   177	            self._write_content(report, content_all, "system-err")
   178	            content_all = ""
   179	        if content_all:
   180	            self._write_content(report, content_all, "system-out")

   182	    def _prepare_content(self, content: str, header: str) -> str:
   183	        return "\n".join([header.center(80, "-"), content, ""])

   185	    def _write_content(self, report: TestReport, content: str, jheader: str) -> None:
   186	        tag = ET.Element(jheader)
   187	        tag.text = bin_xml_escape(content)
   188	        self.append(tag)

   190	    def append_pass(self, report: TestReport) -> None:
   191	        self.add_stats("passed")

   193	    def append_failure(self, report: TestReport) -> None:
   194	        # msg = str(report.longrepr.reprtraceback.extraline)
   195	        if hasattr(report, "wasxfail"):
   196	            self._add_simple("skipped", "xfail-marked test passes unexpectedly")
   197	        else:
   198	            assert report.longrepr is not None
   199	            reprcrash: ReprFileLocation | None = getattr(
   200	                report.longrepr, "reprcrash", None
   201	            )
   202	            if reprcrash is not None:
   203	                message = reprcrash.message
   204	            else:
   205	                message = str(report.longrepr)
   206	            message = bin_xml_escape(message)
   207	            self._add_simple("failure", message, str(report.longrepr))

   209	    def append_collect_error(self, report: TestReport) -> None:
   210	        # msg = str(report.longrepr.reprtraceback.extraline)
   211	        assert report.longrepr is not None
   212	        self._add_simple("error", "collection failure", str(report.longrepr))
   213
   214	    def append_collect_skipped(self, report: TestReport) -> None:
   215	        self._add_simple("skipped", "collection skipped", str(report.longrepr))
   216
   217	    def append_error(self, report: TestReport) -> None:
   218	        assert report.longrepr is not None
   219	        reprcrash: ReprFileLocation | None = getattr(report.longrepr, "reprcrash", None)
   220	        if reprcrash is not None:
   221	            reason = reprcrash.message
   222	        else:
   223	            reason = str(report.longrepr)
   224
   225	        if report.when == "teardown":
   226	            msg = f'failed on teardown with "{reason}"'
   227	        else:
   228	            msg = f'failed on setup with "{reason}"'
   229	        self._add_simple("error", bin_xml_escape(msg), str(report.longrepr))
   230
   231	    def append_skipped(self, report: TestReport) -> None:
   232	        if hasattr(report, "wasxfail"):
   233	            xfailreason = report.wasxfail
   234	            if xfailreason.startswith("reason: "):
   235	                xfailreason = xfailreason[8:]
   236	            xfailreason = bin_xml_escape(xfailreason)
   237	            skipped = ET.Element("skipped", type="pytest.xfail", message=xfailreason)
   238	            self.append(skipped)
   239	        else:
   240	            assert isinstance(report.longrepr, tuple)
   241	            filename, lineno, skipreason = report.longrepr
   242	            if skipreason.startswith("Skipped: "):
   243	                skipreason = skipreason[9:]
   244	            details = f"{filename}:{lineno}: {skipreason}"
   245
   246	            skipped = ET.Element(
   247	                "skipped", type="pytest.skip", message=bin_xml_escape(skipreason)
   248	            )
   249	            skipped.text = bin_xml_escape(details)
   250	            self.append(skipped)
   251	            self.write_captured_output(report)
   252
   253	    def finalize(self) -> None:
   254	        data = self.to_xml()
   255	        self.__dict__.clear()
   256	        # Type ignored because mypy doesn't like overriding a method.
   257	        # Also the return value doesn't match...
   258	        self.to_xml = lambda: data  # type: ignore[method-assign]

   531	    def pytest_runtest_logreport(self, report: TestReport) -> None:
   532	        """Handle a setup/call/teardown report, generating the appropriate
   533	        XML tags as necessary.
   534
   535	        Note: due to plugins like xdist, this hook may be called in interlaced
   536	        order with reports from other nodes. For example:
   537
   538	        Usual call order:
   539	            -> setup node1
   540	            -> call node1
   541	            -> teardown node1
   542	            -> setup node2
   543	            -> call node2
   544	            -> teardown node2
   545
   546	        Possible call order in xdist:
   547	            -> setup node1
   548	            -> call node1
   549	            -> setup node2
   550	            -> call node2
   551	            -> teardown node2
   552	            -> teardown node1
   553	        """
   554	        close_report = None
   555	        if report.passed:
   556	            if report.when == "call":  # ignore setup/teardown
   557	                reporter = self._opentestcase(report)
   558	                reporter.append_pass(report)
   559	        elif report.failed:
   560	            if report.when == "teardown":
   561	                # The following vars are needed when xdist plugin is used.
   562	                report_wid = getattr(report, "worker_id", None)
   563	                report_ii = getattr(report, "item_index", None)
   564	                close_report = next(
   565	                    (
   566	                        rep
   567	                        for rep in self.open_reports
   568	                        if (
   569	                            rep.nodeid == report.nodeid
   570	                            and getattr(rep, "item_index", None) == report_ii
   571	                            and getattr(rep, "worker_id", None) == report_wid
   572	                        )
   573	                    ),
   574	                    None,
   575	                )
   576	                if close_report:
   577	                    # We need to open new testcase in case we have failure in
   578	                    # call and error in teardown in order to follow junit
   579	                    # schema.
   580	                    self.finalize(close_report)
   581	                    self.cnt_double_fail_tests += 1
   582	            reporter = self._opentestcase(report)
   583	            if report.when == "call":
   584	                reporter.append_failure(report)
   585	                self.open_reports.append(report)
   586	                if not self.log_passing_tests:
   587	                    reporter.write_captured_output(report)
   588	            else:
   589	                reporter.append_error(report)
   590	        elif report.skipped:
   591	            reporter = self._opentestcase(report)
   592	            reporter.append_skipped(report)
   593	        self.update_testcase_duration(report)
   594	        if report.when == "teardown":
   595	            reporter = self._opentestcase(report)
   596	            reporter.write_captured_output(report)
   597
   598	            self.finalize(report)
   599	            report_wid = getattr(report, "worker_id", None)
   600	            report_ii = getattr(report, "item_index", None)
   601	            close_report = next(
   602	                (
   603	                    rep
   604	                    for rep in self.open_reports
   605	                    if (
   606	                        rep.nodeid == report.nodeid
   607	                        and getattr(rep, "item_index", None) == report_ii
   608	                        and getattr(rep, "worker_id", None) == report_wid
   609	                    )
   610	                ),
   611	                None,
   612	            )
   613	            if close_report:
   614	                self.open_reports.remove(close_report)
