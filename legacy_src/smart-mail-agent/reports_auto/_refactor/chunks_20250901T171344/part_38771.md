    78	                            props['%s%s' % (name, childKey)] = childValue

    88	        """Permits bulk setting of properties.  These may include
    89	        child objects e.g. "chart.legend.width = 200".
    90
    91	        All assignments will be validated by the object as if they
    92	        were set individually in python code.
    93
    94	        All properties of a top-level object are guaranteed to be
    95	        set before any of the children, which may be helpful to
    96	        widget designers.
    97	        """
    98
    99	        childPropDicts = {}
   100	        for name, value in propDict.items():
   101	            parts = name.split('.', 1)
   102	            if len(parts) == 1:
   103	                #simple attribute, set it now
   104	                setattr(self, name, value)
   105	            else:
   106	                (childName, remains) = parts
   107	                try:
   108	                    childPropDicts[childName][remains] = value
   109	                except KeyError:
   110	                    childPropDicts[childName] = {remains: value}
   111
   112	        # now assign to children
   113	        for childName, childPropDict in childPropDicts.items():
   114	            child = getattr(self, childName)
   115	            child.setProperties(childPropDict)

   187	class TypedPropertyCollection(PropHolder):
   188	    """A container with properties for objects of the same kind.
   189
   190	    This makes it easy to create lists of objects. You initialize
   191	    it with a class of what it is to contain, and that is all you
   192	    can add to it.  You can assign properties to the collection
   193	    as a whole, or to a numeric index within it; if so it creates
   194	    a new child object to hold that data.
   195
   196	    So:
   197	        wedges = TypedPropertyCollection(WedgeProperties)
   198	        wedges.strokeWidth = 2                # applies to all
   199	        wedges.strokeColor = colors.red       # applies to all
   200	        wedges[3].strokeColor = colors.blue   # only to one
   201
   202	    The last line should be taken as a prescription of how to
   203	    create wedge no. 3 if one is needed; no error is raised if
   204	    there are only two data points.
   205
   206	    We try and make sensible use of tuple indices.
   207	        line[(3,x)] is backed by line[(3,)] == line[3] & line
   208	    """
   209
   210	    def __init__(self, exampleClass, **kwds):
   211	        #give it same validation rules as what it holds
   212	        self.__dict__['_value'] = exampleClass(**kwds)
   213	        self.__dict__['_children'] = {}
   214
   215	    def wKlassFactory(self,Klass):
   216	        class WKlass(Klass,CloneMixin):
   217	            def __getattr__(self,name):
   218	                try:
   219	                    return self.__class__.__bases__[0].__getattr__(self,name)
   220	                except:
   221	                    parent = self.parent
   222	                    c = parent._children
   223	                    x = self.__propholder_index__
   224	                    while x:
   225	                        if x in c:
   226	                            return getattr(c[x],name)
   227	                        x = x[:-1]
   228	                    return getattr(parent,name)
   229	            @property
   230	            def parent(self):
   231	                return self.__propholder_parent__()
   232	        return WKlass
   233
   234	    def __getitem__(self, x):
   235	        x = tuple(x) if isinstance(x,(tuple,list)) else (x,)
   236	        try:
   237	            return self._children[x]
   238	        except KeyError:
   239	            Klass = self._value.__class__
   240	            if Klass in _ItemWrapper:
   241	                WKlass = _ItemWrapper[Klass]
   242	            else:
   243	                _ItemWrapper[Klass] = WKlass = self.wKlassFactory(Klass)
   244
   245	            child = WKlass()
   246
   247	            for i in filter(lambda x,K=list(child.__dict__.keys()): x in K,list(child._attrMap.keys())):
   248	                del child.__dict__[i]
   249	            child.__dict__.update(dict(
   250	                                    __propholder_parent__ = weakref_ref(self),
   251	                                    __propholder_index__ = x[:-1])
   252	                                    )
   253
   254	            self._children[x] = child
   255	            return child
   256
   257	    def __contains__(self,key):
   258	        return (tuple(key) if isinstance(key,(tuple,list)) else (key,)) in self._children
   259
   260	    def __setitem__(self, key, value):
   261	        assert isinstance(value, self._value.__class__), (
   262	            "This collection can only hold objects of type %s" % self._value.__class__.__name__)
   263
   264	    def __len__(self):
   265	        return len(list(self._children.keys()))
   266
   267	    def getProperties(self,recur=1):
   268	        # return any children which are defined and whatever
   269	        # differs from the parent
   270	        props = {}
   271
   272	        for key, value in self._value.getProperties(recur=recur).items():
   273	            props['%s' % key] = value
   274
   275	        for idx in self._children.keys():
   276	            childProps = self._children[idx].getProperties(recur=recur)
   277	            for key, value in childProps.items():
   278	                if not hasattr(self,key) or getattr(self, key)!=value:
   279	                    newKey = '[%s].%s' % (idx if len(idx)>1 else idx[0], key)
   280	                    props[newKey] = value
   281	        return props
   282
   283	    def setVector(self,**kw):
   284	        for name, value in kw.items():
   285	            for i, v in enumerate(value):
   286	                setattr(self[i],name,v)
   287
   288	    def __getattr__(self,name):
   289	        return getattr(self._value,name)
   290
   291	    def __setattr__(self,name,value):
   292	        return setattr(self._value,name,value)
   293
   294	    def checkAttr(self, key, a, default=None):
   295	        return getattr(self[key], a, default) if key in self else default
