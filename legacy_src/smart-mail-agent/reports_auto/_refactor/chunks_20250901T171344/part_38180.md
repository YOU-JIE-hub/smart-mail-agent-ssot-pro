
   138	        # This can't be a valid named tuple.
   139	        return False, None

   141	    def check_namedtuple_classdef(
   142	        self, defn: ClassDef, is_stub_file: bool
   143	    ) -> tuple[list[str], list[Type], dict[str, Expression], list[Statement]] | None:
   144	        """Parse and validate fields in named tuple class definition.
   145
   146	        Return a four tuple:
   147	          * field names
   148	          * field types
   149	          * field default values
   150	          * valid statements
   151	        or None, if any of the types are not ready.
   152	        """
   153	        if len(defn.base_type_exprs) > 1:
   154	            self.fail("NamedTuple should be a single base", defn)
   155	        items: list[str] = []
   156	        types: list[Type] = []
   157	        default_items: dict[str, Expression] = {}
   158	        statements: list[Statement] = []
   159	        for stmt in defn.defs.body:
   160	            statements.append(stmt)
   161	            if not isinstance(stmt, AssignmentStmt):
   162	                # Still allow pass or ... (for empty namedtuples).
   163	                if isinstance(stmt, PassStmt) or (
   164	                    isinstance(stmt, ExpressionStmt) and isinstance(stmt.expr, EllipsisExpr)
   165	                ):
   166	                    continue
   167	                # Also allow methods, including decorated ones.
   168	                if isinstance(stmt, (Decorator, FuncBase)):
   169	                    continue
   170	                # And docstrings.
   171	                if isinstance(stmt, ExpressionStmt) and isinstance(stmt.expr, StrExpr):
   172	                    continue
   173	                statements.pop()
   174	                defn.removed_statements.append(stmt)
   175	                self.fail(NAMEDTUP_CLASS_ERROR, stmt)
   176	            elif len(stmt.lvalues) > 1 or not isinstance(stmt.lvalues[0], NameExpr):
   177	                # An assignment, but an invalid one.
   178	                statements.pop()
   179	                defn.removed_statements.append(stmt)
   180	                self.fail(NAMEDTUP_CLASS_ERROR, stmt)
   181	            else:
   182	                # Append name and type in this case...
   183	                name = stmt.lvalues[0].name
   184	                items.append(name)
   185	                if stmt.type is None:
   186	                    types.append(AnyType(TypeOfAny.unannotated))
   187	                else:
   188	                    # We never allow recursive types at function scope. Although it is
   189	                    # possible to support this for named tuples, it is still tricky, and
   190	                    # it would be inconsistent with type aliases.
   191	                    analyzed = self.api.anal_type(
   192	                        stmt.type,
   193	                        allow_placeholder=not self.api.is_func_scope(),
   194	                        prohibit_self_type="NamedTuple item type",
   195	                        prohibit_special_class_field_types="NamedTuple",
   196	                    )
   197	                    if analyzed is None:
   198	                        # Something is incomplete. We need to defer this named tuple.
   199	                        return None
   200	                    types.append(analyzed)
   201	                # ...despite possible minor failures that allow further analysis.
   202	                if name.startswith("_"):
   203	                    self.fail(
   204	                        f"NamedTuple field name cannot start with an underscore: {name}", stmt
   205	                    )
   206	                if stmt.type is None or hasattr(stmt, "new_syntax") and not stmt.new_syntax:
   207	                    self.fail(NAMEDTUP_CLASS_ERROR, stmt)
   208	                elif isinstance(stmt.rvalue, TempNode):
   209	                    # x: int assigns rvalue to TempNode(AnyType())
   210	                    if default_items:
   211	                        self.fail(
   212	                            "Non-default NamedTuple fields cannot follow default fields", stmt
   213	                        )
   214	                else:
   215	                    default_items[name] = stmt.rvalue
   216	        if defn.keywords:
   217	            for_function = ' for "__init_subclass__" of "NamedTuple"'
   218	            for key in defn.keywords:
   219	                self.msg.unexpected_keyword_argument_for_function(for_function, key, defn)
   220	        return items, types, default_items, statements

   222	    def check_namedtuple(
   223	        self, node: Expression, var_name: str | None, is_func_scope: bool
   224	    ) -> tuple[str | None, TypeInfo | None, list[TypeVarLikeType]]:
   225	        """Check if a call defines a namedtuple.
   226
   227	        The optional var_name argument is the name of the variable to
   228	        which this is assigned, if any.
   229
   230	        Return a tuple of two items:
   231	          * Internal name of the named tuple (e.g. the name passed as an argument to namedtuple)
   232	            or None if it is not a valid named tuple
   233	          * Corresponding TypeInfo, or None if not ready.
   234
   235	        If the definition is invalid but looks like a namedtuple,
   236	        report errors but return (some) TypeInfo.
   237	        """
   238	        if not isinstance(node, CallExpr):
   239	            return None, None, []
   240	        call = node
   241	        callee = call.callee
   242	        if not isinstance(callee, RefExpr):
   243	            return None, None, []
   244	        fullname = callee.fullname
   245	        if fullname == "collections.namedtuple":
   246	            is_typed = False
   247	        elif fullname in TYPED_NAMEDTUPLE_NAMES:
   248	            is_typed = True
   249	        else:
   250	            return None, None, []
   251	        result = self.parse_namedtuple_args(call, fullname)
   252	        if result:
   253	            items, types, defaults, typename, tvar_defs, ok = result
   254	        else:
   255	            # Error. Construct dummy return value.
   256	            if var_name:
   257	                name = var_name
   258	                if is_func_scope:
   259	                    name += "@" + str(call.line)
   260	            else:
   261	                name = var_name = "namedtuple@" + str(call.line)
   262	            info = self.build_namedtuple_typeinfo(name, [], [], {}, node.line, None)
   263	            self.store_namedtuple_info(info, var_name, call, is_typed)
   264	            if name != var_name or is_func_scope:
   265	                # NOTE: we skip local namespaces since they are not serialized.
   266	                self.api.add_symbol_skip_local(name, info)
   267	            return var_name, info, []
   268	        if not ok:
   269	            # This is a valid named tuple but some types are not ready.
   270	            return typename, None, []
   271
   272	        # We use the variable name as the class name if it exists. If
   273	        # it doesn't, we use the name passed as an argument. We prefer
   274	        # the variable name because it should be unique inside a
   275	        # module, and so we don't need to disambiguate it with a line
   276	        # number.
   277	        if var_name:
   278	            name = var_name
   279	        else:
   280	            name = typename
   281
   282	        if var_name is None or is_func_scope:
   283	            # There are two special cases where need to give it a unique name derived
   284	            # from the line number:
   285	            #   * This is a base class expression, since it often matches the class name:
   286	            #         class NT(NamedTuple('NT', [...])):
   287	            #             ...
   288	            #   * This is a local (function or method level) named tuple, since
   289	            #     two methods of a class can define a named tuple with the same name,
   290	            #     and they will be stored in the same namespace (see below).
   291	            name += "@" + str(call.line)
   292	        if defaults:
   293	            default_items = {
   294	                arg_name: default for arg_name, default in zip(items[-len(defaults) :], defaults)
   295	            }
   296	        else:
   297	            default_items = {}
   298
   299	        existing_info = None
   300	        if isinstance(node.analyzed, NamedTupleExpr):
   301	            existing_info = node.analyzed.info
   302	        info = self.build_namedtuple_typeinfo(
   303	            name, items, types, default_items, node.line, existing_info
   304	        )
   305
   306	        # If var_name is not None (i.e. this is not a base class expression), we always
   307	        # store the generated TypeInfo under var_name in the current scope, so that
   308	        # other definitions can use it.
   309	        if var_name:
   310	            self.store_namedtuple_info(info, var_name, call, is_typed)
   311	        else:
   312	            call.analyzed = NamedTupleExpr(info, is_typed=is_typed)
   313	            call.analyzed.set_line(call)
   314	        # There are three cases where we need to store the generated TypeInfo
   315	        # second time (for the purpose of serialization):
   316	        #   * If there is a name mismatch like One = NamedTuple('Other', [...])
   317	        #     we also store the info under name 'Other@lineno', this is needed
   318	        #     because classes are (de)serialized using their actual fullname, not
   319	        #     the name of l.h.s.
   320	        #   * If this is a method level named tuple. It can leak from the method
   321	        #     via assignment to self attribute and therefore needs to be serialized
   322	        #     (local namespaces are not serialized).
   323	        #   * If it is a base class expression. It was not stored above, since
   324	        #     there is no var_name (but it still needs to be serialized
   325	        #     since it is in MRO of some class).
   326	        if name != var_name or is_func_scope:
   327	            # NOTE: we skip local namespaces since they are not serialized.
   328	            self.api.add_symbol_skip_local(name, info)
   329	        return typename, info, tvar_defs
   330
   331	    def store_namedtuple_info(
   332	        self, info: TypeInfo, name: str, call: CallExpr, is_typed: bool
   333	    ) -> None:
   334	        self.api.add_symbol(name, info, call)
   335	        call.analyzed = NamedTupleExpr(info, is_typed=is_typed)
   336	        call.analyzed.set_line(call)

   338	    def parse_namedtuple_args(
   339	        self, call: CallExpr, fullname: str
   340	    ) -> None | (tuple[list[str], list[Type], list[Expression], str, list[TypeVarLikeType], bool]):
   341	        """Parse a namedtuple() call into data needed to construct a type.
   342
   343	        Returns a 6-tuple:
   344	        - List of argument names
   345	        - List of argument types
   346	        - List of default values
   347	        - First argument of namedtuple
   348	        - All typevars found in the field definition
   349	        - Whether all types are ready.
   350
   351	        Return None if the definition didn't typecheck.
   352	        """
   353	        type_name = "NamedTuple" if fullname in TYPED_NAMEDTUPLE_NAMES else "namedtuple"
   354	        # TODO: Share code with check_argument_count in checkexpr.py?
   355	        args = call.args
   356	        if len(args) < 2:
   357	            self.fail(f'Too few arguments for "{type_name}()"', call)
   358	            return None
   359	        defaults: list[Expression] = []
   360	        rename = False
   361	        if len(args) > 2:
   362	            # Typed namedtuple doesn't support additional arguments.
   363	            if fullname in TYPED_NAMEDTUPLE_NAMES:
   364	                self.fail('Too many arguments for "NamedTuple()"', call)
   365	                return None
   366	            for i, arg_name in enumerate(call.arg_names[2:], 2):
   367	                if arg_name == "defaults":
   368	                    arg = args[i]
   369	                    # We don't care what the values are, as long as the argument is an iterable
   370	                    # and we can count how many defaults there are.
   371	                    if isinstance(arg, (ListExpr, TupleExpr)):
   372	                        defaults = list(arg.items)
   373	                    else:
   374	                        self.fail(
   375	                            "List or tuple literal expected as the defaults argument to "
   376	                            "{}()".format(type_name),
   377	                            arg,
   378	                        )
   379	                elif arg_name == "rename":
   380	                    arg = args[i]
   381	                    if isinstance(arg, NameExpr) and arg.name in ("True", "False"):
   382	                        rename = arg.name == "True"
   383	                    else:
   384	                        self.fail(
   385	                            f'Boolean literal expected as the "rename" argument to {type_name}()',
   386	                            arg,
   387	                            code=ARG_TYPE,
   388	                        )
   389	        if call.arg_kinds[:2] != [ARG_POS, ARG_POS]:
   390	            self.fail(f'Unexpected arguments to "{type_name}()"', call)
   391	            return None
   392	        if not isinstance(args[0], StrExpr):
   393	            self.fail(f'"{type_name}()" expects a string literal as the first argument', call)
   394	            return None
   395	        typename = args[0].value
   396	        types: list[Type] = []
   397	        tvar_defs = []
   398	        if not isinstance(args[1], (ListExpr, TupleExpr)):
   399	            if fullname == "collections.namedtuple" and isinstance(args[1], StrExpr):
   400	                str_expr = args[1]
   401	                items = str_expr.value.replace(",", " ").split()
   402	            else:
   403	                self.fail(
   404	                    'List or tuple literal expected as the second argument to "{}()"'.format(
   405	                        type_name
   406	                    ),
   407	                    call,
   408	                )
   409	                return None
   410	        else:
   411	            listexpr = args[1]
   412	            if fullname == "collections.namedtuple":
   413	                # The fields argument contains just names, with implicit Any types.
   414	                if not is_StrExpr_list(listexpr.items):
   415	                    self.fail('String literal expected as "namedtuple()" item', call)
   416	                    return None
   417	                items = [item.value for item in listexpr.items]
   418	            else:
   419	                type_exprs = [
   420	                    t.items[1]
   421	                    for t in listexpr.items
   422	                    if isinstance(t, TupleExpr) and len(t.items) == 2
   423	                ]
   424	                tvar_defs = self.api.get_and_bind_all_tvars(type_exprs)
   425	                # The fields argument contains (name, type) tuples.
   426	                result = self.parse_namedtuple_fields_with_types(listexpr.items, call)
   427	                if result is None:
   428	                    # One of the types is not ready, defer.
   429	                    return None
   430	                items, types, _, ok = result
   431	                if not ok:
   432	                    return [], [], [], typename, [], False
   433	        if not types:
   434	            types = [AnyType(TypeOfAny.unannotated) for _ in items]
   435	        processed_items = []
   436	        seen_names: set[str] = set()
   437	        for i, item in enumerate(items):
   438	            problem = self.check_namedtuple_field_name(item, seen_names)
   439	            if problem is None:
   440	                processed_items.append(item)
   441	                seen_names.add(item)
   442	            else:
   443	                if not rename:
   444	                    self.fail(f'"{type_name}()" {problem}', call)
   445	                # Even if rename=False, we pretend that it is True.
   446	                # At runtime namedtuple creation would throw an error;
   447	                # applying the rename logic means we create a more sensible
   448	                # namedtuple.
   449	                new_name = f"_{i}"
   450	                processed_items.append(new_name)
   451	                seen_names.add(new_name)
   452
   453	        if len(defaults) > len(items):
   454	            self.fail(f'Too many defaults given in call to "{type_name}()"', call)
   455	            defaults = defaults[: len(items)]
   456	        return processed_items, types, defaults, typename, tvar_defs, True

   458	    def parse_namedtuple_fields_with_types(
   459	        self, nodes: list[Expression], context: Context
   460	    ) -> tuple[list[str], list[Type], list[Expression], bool] | None:
   461	        """Parse typed named tuple fields.
   462
   463	        Return (names, types, defaults, whether types are all ready), or None if error occurred.
   464	        """
   465	        items: list[str] = []
   466	        types: list[Type] = []
   467	        for item in nodes:
   468	            if isinstance(item, TupleExpr):
   469	                if len(item.items) != 2:
   470	                    self.fail('Invalid "NamedTuple()" field definition', item)
   471	                    return None
   472	                name, type_node = item.items
   473	                if isinstance(name, StrExpr):
   474	                    items.append(name.value)
   475	                else:
   476	                    self.fail('Invalid "NamedTuple()" field name', item)
   477	                    return None
   478	                try:
   479	                    type = expr_to_unanalyzed_type(type_node, self.options, self.api.is_stub_file)
   480	                except TypeTranslationError:
   481	                    self.fail("Invalid field type", type_node)
   482	                    return None
   483	                # We never allow recursive types at function scope.
   484	                analyzed = self.api.anal_type(
   485	                    type,
   486	                    allow_placeholder=not self.api.is_func_scope(),
   487	                    prohibit_self_type="NamedTuple item type",
   488	                    prohibit_special_class_field_types="NamedTuple",
   489	                )
   490	                # Workaround #4987 and avoid introducing a bogus UnboundType
   491	                if isinstance(analyzed, UnboundType):
   492	                    analyzed = AnyType(TypeOfAny.from_error)
   493	                # These should be all known, otherwise we would defer in visit_assignment_stmt().
   494	                if analyzed is None:
   495	                    return [], [], [], False
   496	                types.append(analyzed)
   497	            else:
   498	                self.fail('Tuple expected as "NamedTuple()" field', item)
   499	                return None
   500	        return items, types, [], True

   502	    def build_namedtuple_typeinfo(
   503	        self,
   504	        name: str,
   505	        items: list[str],
   506	        types: list[Type],
   507	        default_items: Mapping[str, Expression],
   508	        line: int,
   509	        existing_info: TypeInfo | None,
   510	    ) -> TypeInfo:
   511	        strtype = self.api.named_type("builtins.str")
   512	        implicit_any = AnyType(TypeOfAny.special_form)
   513	        basetuple_type = self.api.named_type("builtins.tuple", [implicit_any])
   514	        dictype = self.api.named_type("builtins.dict", [strtype, implicit_any])
   515	        # Actual signature should return OrderedDict[str, Union[types]]
   516	        ordereddictype = self.api.named_type("builtins.dict", [strtype, implicit_any])
   517	        fallback = self.api.named_type("builtins.tuple", [implicit_any])
   518	        # Note: actual signature should accept an invariant version of Iterable[UnionType[types]].
   519	        # but it can't be expressed. 'new' and 'len' should be callable types.
