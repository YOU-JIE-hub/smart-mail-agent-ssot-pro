 851	    ) -> None:
   852	        """Adds a fields-aware `__init__` method to the class.
   853
   854	        The added `__init__` will be annotated with types vs. all `Any` depending on the plugin settings.
   855	        """
   856	        if '__init__' in self._cls.info.names and not self._cls.info.names['__init__'].plugin_generated:
   857	            return  # Don't generate an __init__ if one already exists
   858
   859	        typed = self.plugin_config.init_typed
   860	        model_strict = bool(config.strict)
   861	        use_alias = not (config.validate_by_name or config.populate_by_name) and config.validate_by_alias is not False
   862	        requires_dynamic_aliases = bool(config.has_alias_generator and not config.validate_by_name)
   863	        args = self.get_field_arguments(
   864	            fields,
   865	            typed=typed,
   866	            model_strict=model_strict,
   867	            requires_dynamic_aliases=requires_dynamic_aliases,
   868	            use_alias=use_alias,
   869	            is_settings=is_settings,
   870	            is_root_model=is_root_model,
   871	            force_typevars_invariant=True,
   872	        )
   873
   874	        if is_settings:
   875	            base_settings_node = self._api.lookup_fully_qualified(BASESETTINGS_FULLNAME).node
   876	            assert isinstance(base_settings_node, TypeInfo)
   877	            if '__init__' in base_settings_node.names:
   878	                base_settings_init_node = base_settings_node.names['__init__'].node
   879	                assert isinstance(base_settings_init_node, FuncDef)
   880	                if base_settings_init_node is not None and base_settings_init_node.type is not None:
   881	                    func_type = base_settings_init_node.type
   882	                    assert isinstance(func_type, CallableType)
   883	                    for arg_idx, arg_name in enumerate(func_type.arg_names):
   884	                        if arg_name is None or arg_name.startswith('__') or not arg_name.startswith('_'):
   885	                            continue
   886	                        analyzed_variable_type = self._api.anal_type(func_type.arg_types[arg_idx])
   887	                        if analyzed_variable_type is not None and arg_name == '_cli_settings_source':
   888	                            # _cli_settings_source is defined as CliSettingsSource[Any], and as such
   889	                            # the Any causes issues with --disallow-any-explicit. As a workaround, change
   890	                            # the Any type (as if CliSettingsSource was left unparameterized):
   891	                            analyzed_variable_type = analyzed_variable_type.accept(
   892	                                ChangeExplicitTypeOfAny(TypeOfAny.from_omitted_generics)
   893	                            )
   894	                        variable = Var(arg_name, analyzed_variable_type)
   895	                        args.append(Argument(variable, analyzed_variable_type, None, ARG_OPT))
   896
   897	        if not self.should_init_forbid_extra(fields, config):
   898	            var = Var('kwargs')
   899	            args.append(Argument(var, AnyType(TypeOfAny.explicit), None, ARG_STAR2))
   900
   901	        add_method(self._api, self._cls, '__init__', args=args, return_type=NoneType())

   974	    def get_config_update(self, name: str, arg: Expression, lax_extra: bool = False) -> ModelConfigData | None:
   975	        """Determines the config update due to a single kwarg in the ConfigDict definition.
   976
   977	        Warns if a tracked config attribute is set to a value the plugin doesn't know how to interpret (e.g., an int)
   978	        """
   979	        if name not in self.tracked_config_fields:
   980	            return None
   981	        if name == 'extra':
   982	            if isinstance(arg, StrExpr):
   983	                forbid_extra = arg.value == 'forbid'
   984	            elif isinstance(arg, MemberExpr):
   985	                forbid_extra = arg.name == 'forbid'
   986	            else:
   987	                if not lax_extra:
   988	                    # Only emit an error for other types of `arg` (e.g., `NameExpr`, `ConditionalExpr`, etc.) when
   989	                    # reading from a config class, etc. If a ConfigDict is used, then we don't want to emit an error
   990	                    # because you'll get type checking from the ConfigDict itself.
   991	                    #
   992	                    # It would be nice if we could introspect the types better otherwise, but I don't know what the API
   993	                    # is to evaluate an expr into its type and then check if that type is compatible with the expected
   994	                    # type. Note that you can still get proper type checking via: `model_config = ConfigDict(...)`, just
   995	                    # if you don't use an explicit string, the plugin won't be able to infer whether extra is forbidden.
   996	                    error_invalid_config_value(name, self._api, arg)
   997	                return None
   998	            return ModelConfigData(forbid_extra=forbid_extra)
   999	        if name == 'alias_generator':
  1000	            has_alias_generator = True
  1001	            if isinstance(arg, NameExpr) and arg.fullname == 'builtins.None':
  1002	                has_alias_generator = False
  1003	            return ModelConfigData(has_alias_generator=has_alias_generator)
  1004	        if isinstance(arg, NameExpr) and arg.fullname in ('builtins.True', 'builtins.False'):
  1005	            return ModelConfigData(**{name: arg.fullname == 'builtins.True'})
  1006	        error_invalid_config_value(name, self._api, arg)
  1007	        return None

  1009	    @staticmethod
  1010	    def get_has_default(stmt: AssignmentStmt) -> bool:
  1011	        """Returns a boolean indicating whether the field defined in `stmt` is a required field."""
  1012	        expr = stmt.rvalue
  1013	        if isinstance(expr, TempNode):
  1014	            # TempNode means annotation-only, so has no default
  1015	            return False
  1016	        if isinstance(expr, CallExpr) and isinstance(expr.callee, RefExpr) and expr.callee.fullname == FIELD_FULLNAME:
  1017	            # The "default value" is a call to `Field`; at this point, the field has a default if and only if:
  1018	            # * there is a positional argument that is not `...`
  1019	            # * there is a keyword argument named "default" that is not `...`
  1020	            # * there is a "default_factory" that is not `None`
  1021	            for arg, name in zip(expr.args, expr.arg_names):
  1022	                # If name is None, then this arg is the default because it is the only positional argument.
  1023	                if name is None or name == 'default':
  1024	                    return arg.__class__ is not EllipsisExpr
  1025	                if name == 'default_factory':
  1026	                    return not (isinstance(arg, NameExpr) and arg.fullname == 'builtins.None')
  1027	            return False
  1028	        # Has no default if the "default value" is Ellipsis (i.e., `field_name: Annotation = ...`)
  1029	        return not isinstance(expr, EllipsisExpr)

  1048	    def get_alias_info(stmt: AssignmentStmt) -> tuple[str | None, bool]:
  1049	        """Returns a pair (alias, has_dynamic_alias), extracted from the declaration of the field defined in `stmt`.
  1050
  1051	        `has_dynamic_alias` is True if and only if an alias is provided, but not as a string literal.
  1052	        If `has_dynamic_alias` is True, `alias` will be None.
  1053	        """

  1055	        if isinstance(expr, TempNode):
  1056	            # TempNode means annotation-only

  1061	        ):

  1063	            return None, False

  1066	            arg = expr.args[expr.arg_names.index('validation_alias')]
  1067	        elif 'alias' in expr.arg_names:
  1068	            arg = expr.args[expr.arg_names.index('alias')]

  1134	    def should_init_forbid_extra(self, fields: list[PydanticModelField], config: ModelConfigData) -> bool:
  1135	        """Indicates whether the generated `__init__` should get a `**kwargs` at the end of its signature.

  1138	        *unless* a required dynamic alias is present (since then we can't determine a valid signature).
  1139	        """
  1140	        if not (config.validate_by_name or config.populate_by_name):
  1141	            if self.is_dynamic_alias_present(fields, bool(config.has_alias_generator)):
  1142	                return False
  1143	        if config.forbid_extra:
  1144	            return True
