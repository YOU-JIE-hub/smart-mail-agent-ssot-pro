    39	def structToPDF(structure):
    40	    "Converts deeply nested structure to PDFdoc dictionary/array objects"
    41	    if isinstance(structure,dict):
    42	        newDict = {}
    43	        for k, v in structure.items():
    44	            newDict[k] = structToPDF(v)
    45	        return pdfdoc.PDFDictionary(newDict)
    46	    elif isSeq(structure):
    47	        newList = []
    48	        for elem in structure:
    49	            newList.append(structToPDF(elem))
    50	        return pdfdoc.PDFArray(newList)
    51	    else:
    52	        return structure

    54	class CIDEncoding(pdfmetrics.Encoding):
    55	    """Multi-byte encoding.  These are loaded from CMAP files.
    56
    57	    A CMAP file is like a mini-codec.  It defines the correspondence
    58	    between code points in the (multi-byte) input data and Character
    59	    IDs. """
    60	    # aims to do similar things to Brian Hooper's CMap class,
    61	    # but I could not get it working and had to rewrite.
    62	    # also, we should really rearrange our current encoding
    63	    # into a SingleByteEncoding since many of its methods
    64	    # should not apply here.
    65
    66	    def __init__(self, name, useCache=1):
    67	        self.name = name
    68	        self._mapFileHash = None
    69	        self._codeSpaceRanges = []
    70	        self._notDefRanges = []
    71	        self._cmap = {}
    72	        self.source = None
    73	        if not DISABLE_CMAP:
    74	            if useCache:
    75	                from reportlab.lib.utils import get_rl_tempdir
    76	                fontmapdir = get_rl_tempdir('FastCMAPS')
    77	                if os.path.isfile(fontmapdir + os.sep + name + '.fastmap'):
    78	                    self.fastLoad(fontmapdir)
    79	                    self.source = fontmapdir + os.sep + name + '.fastmap'
    80	                else:
    81	                    self.parseCMAPFile(name)
    82	                    self.source = 'CMAP: ' + name
    83	                    self.fastSave(fontmapdir)
    84	            else:
    85	                self.parseCMAPFile(name)
    86
    87	    def _hash(self, text):
    88	        hasher = md5()
    89	        hasher.update(text)
    90	        return hasher.digest()
    91
    92	    def parseCMAPFile(self, name):
    93	        """This is a tricky one as CMAP files are Postscript
    94	        ones.  Some refer to others with a 'usecmap'
    95	        command"""
    96	        #started = time.clock()
    97	        cmapfile = findCMapFile(name)
    98	        # this will CRAWL with the unicode encodings...
    99	        rawdata = open(cmapfile, 'r').read()
   100
   101	        self._mapFileHash = self._hash(rawdata)
   102	        #if it contains the token 'usecmap', parse the other
   103	        #cmap file first....
   104	        usecmap_pos = rawdata.find('usecmap')
   105	        if  usecmap_pos > -1:
   106	            #they tell us to look in another file
   107	            #for the code space ranges. The one
   108	            # to use will be the previous word.
   109	            chunk = rawdata[0:usecmap_pos]
   110	            words = chunk.split()
   111	            otherCMAPName = words[-1]
   112	            #print 'referred to another CMAP %s' % otherCMAPName
   113	            self.parseCMAPFile(otherCMAPName)
   114	            # now continue parsing this, as it may
   115	            # override some settings
   116
   117
   118	        words = rawdata.split()
   119	        while words != []:
   120	            if words[0] == 'begincodespacerange':
   121	                words = words[1:]
   122	                while words[0] != 'endcodespacerange':
   123	                    strStart, strEnd, words = words[0], words[1], words[2:]
   124	                    start = int(strStart[1:-1], 16)
   125	                    end = int(strEnd[1:-1], 16)
   126	                    self._codeSpaceRanges.append((start, end),)
   127	            elif words[0] == 'beginnotdefrange':
   128	                words = words[1:]
   129	                while words[0] != 'endnotdefrange':
   130	                    strStart, strEnd, strValue = words[0:3]
   131	                    start = int(strStart[1:-1], 16)
   132	                    end = int(strEnd[1:-1], 16)
   133	                    value = int(strValue)
   134	                    self._notDefRanges.append((start, end, value),)
   135	                    words = words[3:]
   136	            elif words[0] == 'begincidrange':
   137	                words = words[1:]
   138	                while words[0] != 'endcidrange':
   139	                    strStart, strEnd, strValue = words[0:3]
   140	                    start = int(strStart[1:-1], 16)
   141	                    end = int(strEnd[1:-1], 16)
   142	                    value = int(strValue)
   143	                    # this means that 'start' corresponds to 'value',
   144	                    # start+1 corresponds to value+1 and so on up
   145	                    # to end
   146	                    offset = 0
   147	                    while start + offset <= end:
   148	                        self._cmap[start + offset] = value + offset
   149	                        offset = offset + 1
   150	                    words = words[3:]
   151
   152	            else:
   153	                words = words[1:]
   154	        #finished = time.clock()
   155	        #print 'parsed CMAP %s in %0.4f seconds' % (self.name, finished - started)
   156
   157	    def translate(self, text):
   158	        "Convert a string into a list of CIDs"
   159	        output = []
   160	        cmap = self._cmap
   161	        lastChar = ''
   162	        for char in text:
   163	            if lastChar != '':
   164	                #print 'convert character pair "%s"' % (lastChar + char)
   165	                num = ord(lastChar) * 256 + ord(char)
   166	            else:
   167	                #print 'convert character "%s"' % char
   168	                num = ord(char)
   169	            lastChar = char
   170	            found = 0
   171	            for low, high in self._codeSpaceRanges:
   172	                if low < num < high:
   173	                    try:
   174	                        cid = cmap[num]
   175	                        #print '%d -> %d' % (num, cid)
   176	                    except KeyError:
   177	                        #not defined.  Try to find the appropriate
   178	                        # notdef character, or failing that return
   179	                        # zero
   180	                        cid = 0
   181	                        for low2, high2, notdef in self._notDefRanges:
   182	                            if low2 < num < high2:
   183	                                cid = notdef
   184	                                break
   185	                    output.append(cid)
   186	                    found = 1
   187	                    break
   188	            if found:
   189	                lastChar = ''
   190	            else:
   191	                lastChar = char
   192	        return output

   222	class CIDTypeFace(pdfmetrics.TypeFace):
   223	    """Multi-byte type face.
   224
   225	    Conceptually similar to a single byte typeface,
   226	    but the glyphs are identified by a numeric Character
   227	    ID (CID) and not a glyph name. """
   228	    def __init__(self, name):
   229	        """Initialised from one of the canned dictionaries in allowedEncodings
   230
   231	        Or rather, it will be shortly..."""
   232	        pdfmetrics.TypeFace.__init__(self, name)
   233	        self._extractDictInfo(name)
   234	    def _extractDictInfo(self, name):
   235	        try:
   236	            fontDict = CIDFontInfo[name]
   237	        except KeyError:
   238	            raise KeyError("Unable to find information on CID typeface '%s'" % name +
   239	                            "Only the following font names work:" + repr(allowedTypeFaces))
   240	        descFont = fontDict['DescendantFonts'][0]
   241	        self.ascent = descFont['FontDescriptor']['Ascent']
   242	        self.descent = descFont['FontDescriptor']['Descent']
   243	        self._defaultWidth = descFont['DW']
   244	        self._explicitWidths = self._expandWidths(descFont['W'])
   245
   246	        # should really support self.glyphWidths, self.glyphNames
   247	        # but not done yet.
   248
   249
   250	    def _expandWidths(self, compactWidthArray):
   251	        """Expands Adobe nested list structure to get a dictionary of widths.
   252
   253	        Here is an example of such a structure.::
   254
   255	            (
   256	            # starting at character ID 1, next n  characters have the widths given.
   257	            1,  (277,305,500,668,668,906,727,305,445,445,508,668,305,379,305,539),
   258	            # all Characters from ID 17 to 26 are 668 em units wide
   259	            17, 26, 668,
   260	            27, (305, 305, 668, 668, 668, 566, 871, 727, 637, 652, 699, 574, 555,
   261	                 676, 687, 242, 492, 664, 582, 789, 707, 734, 582, 734, 605, 605,
   262	                 641, 668, 727, 945, 609, 609, 574, 445, 668, 445, 668, 668, 590,
   263	                 555, 609, 547, 602, 574, 391, 609, 582, 234, 277, 539, 234, 895,
   264	                 582, 605, 602, 602, 387, 508, 441, 582, 562, 781, 531, 570, 555,
   265	                 449, 246, 449, 668),
   266	            # these must be half width katakana and the like.
   267	            231, 632, 500
   268	            )
   269
   270	        """
   271	        data = compactWidthArray[:]
   272	        widths = {}
   273	        while data:
   274	            start, data = data[0], data[1:]
   275	            if isSeq(data[0]):
   276	                items, data = data[0], data[1:]
   277	                for offset in range(len(items)):
   278	                    widths[start + offset] = items[offset]
   279	            else:
   280	                end, width, data = data[0], data[1], data[2:]
   281	                for idx in range(start, end+1):
   282	                    widths[idx] = width
   283	        return widths
   284
   285	    def getCharWidth(self, characterId):
   286	        return self._explicitWidths.get(characterId, self._defaultWidth)

   288	class CIDFont(pdfmetrics.Font):
   289	    "Represents a built-in multi-byte font"

   292	    def __init__(self, face, encoding):
   293
   294	        assert face in allowedTypeFaces, "TypeFace '%s' not supported! Use any of these instead: %s" % (face, allowedTypeFaces)
   295	        self.faceName = face
   296	        #should cache in registry...
   297	        self.face = CIDTypeFace(face)
   298
   299	        assert encoding in allowedEncodings, "Encoding '%s' not supported!  Use any of these instead: %s" % (encoding, allowedEncodings)
   300	        self.encodingName = encoding
   301	        self.encoding = CIDEncoding(encoding)
   302
   303	        #legacy hack doing quick cut and paste.
   304	        self.fontName = self.faceName + '-' + self.encodingName
   305	        self.name = self.fontName
   306
   307	        # need to know if it is vertical or horizontal
   308	        self.isVertical = (self.encodingName[-1] == 'V')
   309
   310	        #no substitutes initially
   311	        self.substitutionFonts = []
   312
   313	    def formatForPdf(self, text):
   314	        encoded = escapePDF(text)
   315	        #print 'encoded CIDFont:', encoded
   316	        return encoded
   317
   318	    def stringWidth(self, text, size, encoding=None):
   319	        """This presumes non-Unicode input.  UnicodeCIDFont wraps it for that context"""
   320	        cidlist = self.encoding.translate(text)
   321	        if self.isVertical:
   322	            #this part is "not checked!" but seems to work.
   323	            #assume each is 1000 ems high
   324	            return len(cidlist) * size
   325	        else:
   326	            w = 0
   327	            for cid in cidlist:
   328	                w = w + self.face.getCharWidth(cid)
   329	            return 0.001 * w * size

   352	class UnicodeCIDFont(CIDFont):
   353	    """Wraps up CIDFont to hide explicit encoding choice;
   354	    encodes text for output as UTF16.
   355
   356	    lang should be one of 'jpn',chs','cht','kor' for now.
   357	    if vertical is set, it will select a different widths array
   358	    and possibly glyphs for some punctuation marks.
   359
   360	    halfWidth is only for Japanese.
   361
   362
   363	    >>> dodgy = UnicodeCIDFont('nonexistent')
   364	    Traceback (most recent call last):
   365	    ...
   366	    KeyError: "don't know anything about CID font nonexistent"
   367	    >>> heisei = UnicodeCIDFont('HeiseiMin-W3')
   368	    >>> heisei.name
   369	    'HeiseiMin-W3'
   370	    >>> heisei.language
   371	    'jpn'
   372	    >>> heisei.encoding.name
   373	    'UniJIS-UCS2-H'
   374	    >>> #This is how PDF data gets encoded.
   375	    >>> print(heisei.formatForPdf('hello'))
   376	    \\000h\\000e\\000l\\000l\\000o
   377	    >>> tokyo = u'\u6771\u4AEC'
   378	    >>> print(heisei.formatForPdf(tokyo))
   379	    gqJ\\354
   380	    >>> print(heisei.stringWidth(tokyo,10))
   381	    20.0
   382	    >>> print(heisei.stringWidth('hello world',10))
   383	    45.83
   384	    """
   385
   386	    def __init__(self, face, isVertical=False, isHalfWidth=False):
   387	        #pass
   388	        try:
   389	            lang, defaultEncoding = defaultUnicodeEncodings[face]
   390	        except KeyError:
   391	            raise KeyError("don't know anything about CID font %s" % face)
   392
   393	        #we know the languages now.
   394	        self.language = lang
   395
   396	        #rebuilt encoding string.  They follow rules which work
   397	        #for the 7 fonts provided.
   398	        enc = defaultEncoding[:-1]
   399	        if isHalfWidth:
   400	            enc = enc + 'HW-'
   401	        if isVertical:
   402	            enc = enc + 'V'
   403	        else:
   404	            enc = enc + 'H'
   405
   406	        #now we can do the more general case
   407	        CIDFont.__init__(self, face, enc)
   408	        #self.encName = 'utf_16_le'
   409	        #it's simpler for unicode, just use the face name
   410	        self.name = self.fontName = face
   411	        self.vertical = isVertical
   412	        self.isHalfWidth = isHalfWidth
   413
   414	        self.unicodeWidths = widthsByUnichar[self.name]
   415

   417	    def formatForPdf(self, text):
   418	        #these ones should be encoded asUTF16 minus the BOM
   419	        from codecs import utf_16_be_encode
   420	        #print 'formatting %s: %s' % (type(text), repr(text))
   421	        if isBytes(text):
   422	            text = text.decode('utf8')
   423	        utfText = utf_16_be_encode(text)[0]
   424	        encoded = escapePDF(utfText)
   425	        #print '  encoded:',encoded
   426	        return encoded
   427	        #
   428	        #result = escapePDF(encoded)

   430	        #return result

   436	            text = text.decode('utf8')
