def move_aliases(plan_only=False):
    moves=[]
    for d in ALIAS_DIRS:
        if not d.exists(): continue
        target=MAP_DIR[d]
        for f in py_files(d):
            rel=f.relative_to(d)
            dst=target/rel

                if f.resolve()!=dst.resolve():
