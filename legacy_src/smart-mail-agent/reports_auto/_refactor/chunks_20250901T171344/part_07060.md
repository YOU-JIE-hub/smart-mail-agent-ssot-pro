  5524	    def infer_variable_types_from_type_maps(
  5525	        self, type_maps: list[TypeMap]
  5526	    ) -> dict[SymbolNode, Type]:
  5527	        # Type maps may contain variables inherited from previous code which are not
  5528	        # necessary `Var`s (e.g. a function defined earlier with the same name).
  5529	        all_captures: dict[SymbolNode, list[tuple[NameExpr, Type]]] = defaultdict(list)
  5530	        for tm in type_maps:
  5531	            if tm is not None:
  5532	                for expr, typ in tm.items():
  5533	                    if isinstance(expr, NameExpr):
  5534	                        node = expr.node
  5535	                        assert node is not None
  5536	                        all_captures[node].append((expr, typ))
  5537
  5538	        inferred_types: dict[SymbolNode, Type] = {}
  5539	        for var, captures in all_captures.items():
  5540	            already_exists = False
  5541	            types: list[Type] = []
  5542	            for expr, typ in captures:
  5543	                types.append(typ)
  5544
  5545	                previous_type, _, _ = self.check_lvalue(expr)
  5546	                if previous_type is not None:
  5547	                    already_exists = True
  5548	                    if self.check_subtype(
  5549	                        typ,
  5550	                        previous_type,
  5551	                        expr,
  5552	                        msg=message_registry.INCOMPATIBLE_TYPES_IN_CAPTURE,
  5553	                        subtype_label="pattern captures type",
  5554	                        supertype_label="variable has type",
  5555	                    ):
  5556	                        inferred_types[var] = previous_type
  5557
  5558	            if not already_exists:
  5559	                new_type = UnionType.make_union(types)
  5560	                # Infer the union type at the first occurrence
  5561	                first_occurrence, _ = captures[0]
  5562	                # If it didn't exist before ``match``, it's a Var.
  5563	                assert isinstance(var, Var)
  5564	                inferred_types[var] = new_type
  5565	                self.infer_variable_type(var, first_occurrence, new_type, first_occurrence)
