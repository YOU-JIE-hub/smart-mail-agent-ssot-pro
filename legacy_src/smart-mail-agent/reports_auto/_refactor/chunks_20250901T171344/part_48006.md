  1802	                # is equally necessary for new-style classes to break a vicious circle.
  1803	                upper_bound = PlaceholderType(None, [], context.line)
  1804	        else:
  1805	            if type_param.kind == TYPE_VAR_TUPLE_KIND:

  1811	                type_param.default,

  1821	            elif type_param.kind == TYPE_VAR_KIND:

  1823	            elif type_param.kind == PARAM_SPEC_KIND:

  1834	                    if analyzed is None:

  1853	                name=type_param.name,

  1856	                default=default,

  1879	    def analyze_class(self, defn: ClassDef) -> None:
  1880	        fullname = self.qualified_name(defn.name)
  1881	        if not defn.info and not self.is_core_builtin_class(defn):
  1882	            # Add placeholder so that self-references in base classes can be
  1883	            # resolved.  We don't want this to cause a deferral, since if there
  1884	            # are no incomplete references, we'll replace this with a TypeInfo
  1885	            # before returning.
  1886	            placeholder = PlaceholderNode(fullname, defn, defn.line, becomes_typeinfo=True)
  1887	            self.add_symbol(defn.name, placeholder, defn, can_defer=False)
  1888
  1889	        tag = self.track_incomplete_refs()
  1890
  1891	        # Restore base classes after previous iteration (things like Generic[T] might be removed).
  1892	        defn.base_type_exprs.extend(defn.removed_base_type_exprs)
  1893	        defn.removed_base_type_exprs.clear()
  1894
  1895	        self.infer_metaclass_and_bases_from_compat_helpers(defn)
  1896
  1897	        bases = defn.base_type_exprs
  1898	        bases, tvar_defs, is_protocol = self.clean_up_bases_and_infer_type_variables(
  1899	            defn, bases, context=defn
  1900	        )
  1901
  1902	        self.check_type_alias_bases(bases)
  1903
  1904	        for tvd in tvar_defs:
  1905	            if isinstance(tvd, TypeVarType) and any(
  1906	                has_placeholder(t) for t in [tvd.upper_bound] + tvd.values
  1907	            ):
  1908	                # Some type variable bounds or values are not ready, we need
  1909	                # to re-analyze this class.
  1910	                self.defer()
  1911	            if has_placeholder(tvd.default):
  1912	                # Placeholder values in TypeVarLikeTypes may get substituted in.
  1913	                # Defer current target until they are ready.
  1914	                self.mark_incomplete(defn.name, defn)
  1915	                return
  1916
  1917	        self.analyze_class_keywords(defn)
  1918	        bases_result = self.analyze_base_classes(bases)
  1919	        if bases_result is None or self.found_incomplete_ref(tag):
  1920	            # Something was incomplete. Defer current target.
  1921	            self.mark_incomplete(defn.name, defn)
  1922	            return
  1923
  1924	        base_types, base_error = bases_result
  1925	        if any(isinstance(base, PlaceholderType) for base, _ in base_types):
  1926	            # We need to know the TypeInfo of each base to construct the MRO. Placeholder types
  1927	            # are okay in nested positions, since they can't affect the MRO.
  1928	            self.mark_incomplete(defn.name, defn)
  1929	            return
  1930
  1931	        declared_metaclass, should_defer, any_meta = self.get_declared_metaclass(
  1932	            defn.name, defn.metaclass
  1933	        )
  1934	        if should_defer or self.found_incomplete_ref(tag):
  1935	            # Metaclass was not ready. Defer current target.
  1936	            self.mark_incomplete(defn.name, defn)
  1937	            return
  1938
  1939	        if self.analyze_typeddict_classdef(defn):
  1940	            if defn.info:
  1941	                self.setup_type_vars(defn, tvar_defs)
  1942	                self.setup_alias_type_vars(defn)
  1943	            return
  1944
  1945	        if self.analyze_namedtuple_classdef(defn, tvar_defs):
  1946	            return
  1947
  1948	        # Create TypeInfo for class now that base classes and the MRO can be calculated.
  1949	        self.prepare_class_def(defn)
  1950	        self.setup_type_vars(defn, tvar_defs)
  1951	        if base_error:
  1952	            defn.info.fallback_to_any = True
  1953	        if any_meta:
  1954	            defn.info.meta_fallback_to_any = True
  1955
  1956	        with self.scope.class_scope(defn.info):
  1957	            self.configure_base_classes(defn, base_types)
  1958	            defn.info.is_protocol = is_protocol
  1959	            self.recalculate_metaclass(defn, declared_metaclass)
  1960	            defn.info.runtime_protocol = False
  1961
  1962	            if defn.type_args:
  1963	                # PEP 695 type parameters are not in scope in class decorators, so
  1964	                # temporarily disable type parameter namespace.
  1965	                type_params_names = self.locals.pop()
  1966	                self.scope_stack.pop()
  1967	            for decorator in defn.decorators:
  1968	                self.analyze_class_decorator(defn, decorator)
  1969	            if defn.type_args:
  1970	                self.locals.append(type_params_names)
  1971	                self.scope_stack.append(SCOPE_ANNOTATION)
  1972
  1973	            self.analyze_class_body_common(defn)

  2180	    def clean_up_bases_and_infer_type_variables(
  2181	        self, defn: ClassDef, base_type_exprs: list[Expression], context: Context
  2182	    ) -> tuple[list[Expression], list[TypeVarLikeType], bool]:

  2184

  2186
  2187	          class Foo(Bar, Generic[T]): ...

  2189	        Now we will remove Generic[T] from bases of Foo and infer that the
  2190	        type variable 'T' is a type argument of Foo.
  2191
  2192	        Note that this is performed *before* semantic analysis.
  2193
  2194	        Returns (remaining base expressions, inferred type variables, is protocol).
  2195	        """
  2196	        removed: list[int] = []
  2197	        declared_tvars: TypeVarLikeList = []
  2198	        is_protocol = False
  2199	        if defn.type_args is not None:
  2200	            for p in defn.type_args:
  2201	                node = self.lookup(p.name, context)
  2202	                assert node is not None
  2203	                assert isinstance(node.node, TypeVarLikeExpr)
  2204
