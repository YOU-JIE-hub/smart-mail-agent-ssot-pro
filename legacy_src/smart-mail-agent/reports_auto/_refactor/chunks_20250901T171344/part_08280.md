   406	        if isinstance(stmt.rvalue, CallExpr):

   423	def allocate_class(builder: IRBuilder, cdef: ClassDef) -> Value:
   424	    # OK AND NOW THE FUN PART
   425	    base_exprs = cdef.base_type_exprs + cdef.removed_base_type_exprs
   426	    new_style_type_args = cdef.type_args
   427	    if new_style_type_args:
   428	        bases = [make_generic_base_class(builder, cdef.fullname, new_style_type_args, cdef.line)]
   429	    else:
   430	        bases = []
   431
   432	    if base_exprs or new_style_type_args:
   433	        bases.extend([builder.accept(x) for x in base_exprs])
   434	        tp_bases = builder.new_tuple(bases, cdef.line)
   435	    else:
   436	        tp_bases = builder.add(LoadErrorValue(object_rprimitive, is_borrowed=True))
   437	    modname = builder.load_str(builder.module_name)
   438	    template = builder.add(
   439	        LoadStatic(object_rprimitive, cdef.name + "_template", builder.module_name, NAMESPACE_TYPE)
   440	    )
   441	    # Create the class
   442	    tp = builder.call_c(pytype_from_template_op, [template, tp_bases, modname], cdef.line)
   443	    # Immediately fix up the trait vtables, before doing anything with the class.
   444	    ir = builder.mapper.type_to_ir[cdef.info]
   445	    if not ir.is_trait and not ir.builtin_base:
   446	        builder.add(
   447	            Call(
   448	                FuncDecl(
   449	                    cdef.name + "_trait_vtable_setup",
   450	                    None,
   451	                    builder.module_name,
   452	                    FuncSignature([], bool_rprimitive),
   453	                ),
   454	                [],
   455	                -1,
   456	            )
   457	        )
   458	    # Populate a '__mypyc_attrs__' field containing the list of attrs
   459	    builder.primitive_op(
   460	        py_setattr_op,
   461	        [
   462	            tp,
   463	            builder.load_str("__mypyc_attrs__"),
   464	            create_mypyc_attrs_tuple(builder, builder.mapper.type_to_ir[cdef.info], cdef.line),
   465	        ],
   466	        cdef.line,
   467	    )
   468
   469	    # Save the class
   470	    builder.add(InitStatic(tp, cdef.name, builder.module_name, NAMESPACE_TYPE))
   471
   472	    # Add it to the dict
   473	    builder.primitive_op(
   474	        dict_set_item_op, [builder.load_globals_dict(), builder.load_str(cdef.name), tp], cdef.line
   475	    )
   476
   477	    return tp

   480	def make_generic_base_class(
   481	    builder: IRBuilder, fullname: str, type_args: list[TypeParam], line: int
   482	) -> Value:
   483	    """Construct Generic[...] base class object for a new-style generic class (Python 3.12)."""
   484	    mod = builder.call_c(import_op, [builder.load_str("_typing")], line)
   485	    tvs = create_type_params(builder, mod, type_args, line)
   486	    args = []
   487	    for tv, type_param in zip(tvs, type_args):
   488	        if type_param.kind == TYPE_VAR_TUPLE_KIND:
   489	            # Evaluate *Ts for a TypeVarTuple
   490	            it = builder.primitive_op(iter_op, [tv], line)
   491	            tv = builder.call_c(next_op, [it], line)
   492	        args.append(tv)
   493
   494	    gent = builder.py_get_attr(mod, "Generic", line)
   495	    if len(args) == 1:
   496	        arg = args[0]
   497	    else:
   498	        arg = builder.new_tuple(args, line)
   499
   500	    base = builder.primitive_op(py_get_item_op, [gent, arg], line)
   501	    return base

   512	def populate_non_ext_bases(builder: IRBuilder, cdef: ClassDef) -> Value:
   513	    """Create base class tuple of a non-extension class.
   514
   515	    The tuple is passed to the metaclass constructor.
   516	    """
   517	    is_named_tuple = cdef.info.is_named_tuple
   518	    ir = builder.mapper.type_to_ir[cdef.info]
   519	    bases = []
   520	    for cls in (b.type for b in cdef.info.bases):
   521	        if cls.fullname == "builtins.object":
   522	            continue
   523	        if is_named_tuple and cls.fullname in (
   524	            "typing.Sequence",
   525	            "typing.Iterable",
   526	            "typing.Collection",
   527	            "typing.Reversible",
   528	            "typing.Container",
   529	            "typing.Sized",
   530	        ):
   531	            # HAX: Synthesized base classes added by mypy don't exist at runtime, so skip them.
   532	            #      This could break if they were added explicitly, though...
   533	            continue
   534	        # Add the current class to the base classes list of concrete subclasses
   535	        if cls in builder.mapper.type_to_ir:
   536	            base_ir = builder.mapper.type_to_ir[cls]
   537	            if base_ir.children is not None:
   538	                base_ir.children.append(ir)
   539
   540	        if cls.fullname in MAGIC_TYPED_DICT_CLASSES:
   541	            # HAX: Mypy internally represents TypedDict classes differently from what
   542	            #      should happen at runtime. Replace with something that works.
   543	            module = "typing"
   544	            name = "_TypedDict"
   545	            base = builder.get_module_attr(module, name, cdef.line)
   546	        elif is_named_tuple and cls.fullname == "builtins.tuple":
   547	            name = "_NamedTuple"
   548	            base = builder.get_module_attr("typing", name, cdef.line)
   549	        else:
   550	            cls_module = cls.fullname.rsplit(".", 1)[0]
   551	            if cls_module == builder.current_module:
   552	                base = builder.load_global_str(cls.name, cdef.line)
   553	            else:
   554	                base = builder.load_module_attr_by_fullname(cls.fullname, cdef.line)
   555	        bases.append(base)
   556	        if cls.fullname in MAGIC_TYPED_DICT_CLASSES:
   557	            # The remaining base classes are synthesized by mypy and should be ignored.
   558	            break
   559	    return builder.new_tuple(bases, cdef.line)

   562	def find_non_ext_metaclass(builder: IRBuilder, cdef: ClassDef, bases: Value) -> Value:
   563	    """Find the metaclass of a class from its defs and bases."""
   564	    if cdef.metaclass:
   565	        declared_metaclass = builder.accept(cdef.metaclass)
   566	    else:
   567	        if cdef.info.typeddict_type is not None:
   568	            # In Python 3.9, the metaclass for class-based TypedDict is typing._TypedDictMeta.
   569	            # We can't easily calculate it generically, so special case it.
   570	            return builder.get_module_attr("typing", "_TypedDictMeta", cdef.line)
   571	        elif cdef.info.is_named_tuple:
   572	            # In Python 3.9, the metaclass for class-based NamedTuple is typing.NamedTupleMeta.
   573	            # We can't easily calculate it generically, so special case it.
   574	            return builder.get_module_attr("typing", "NamedTupleMeta", cdef.line)
   575
   576	        declared_metaclass = builder.add(
   577	            LoadAddress(type_object_op.type, type_object_op.src, cdef.line)
   578	        )
   579
   580	    return builder.call_c(py_calc_meta_op, [declared_metaclass, bases], cdef.line)

   583	def setup_non_ext_dict(
   584	    builder: IRBuilder, cdef: ClassDef, metaclass: Value, bases: Value
   585	) -> Value:
   586	    """Initialize the class dictionary for a non-extension class.
   587
   588	    This class dictionary is passed to the metaclass constructor.
   589	    """
   590	    # Check if the metaclass defines a __prepare__ method, and if so, call it.
   591	    has_prepare = builder.primitive_op(
   592	        py_hasattr_op, [metaclass, builder.load_str("__prepare__")], cdef.line
   593	    )
   594
   595	    non_ext_dict = Register(dict_rprimitive)
   596
   597	    true_block, false_block, exit_block = BasicBlock(), BasicBlock(), BasicBlock()
   598	    builder.add_bool_branch(has_prepare, true_block, false_block)
   599
   600	    builder.activate_block(true_block)
   601	    cls_name = builder.load_str(cdef.name)
   602	    prepare_meth = builder.py_get_attr(metaclass, "__prepare__", cdef.line)
   603	    prepare_dict = builder.py_call(prepare_meth, [cls_name, bases], cdef.line)
   604	    builder.assign(non_ext_dict, prepare_dict, cdef.line)
   605	    builder.goto(exit_block)
   606
   607	    builder.activate_block(false_block)
   608	    builder.assign(non_ext_dict, builder.call_c(dict_new_op, [], cdef.line), cdef.line)
   609	    builder.goto(exit_block)
   610	    builder.activate_block(exit_block)
   611
   612	    return non_ext_dict

   615	def add_non_ext_class_attr_ann(
   616	    builder: IRBuilder,
   617	    non_ext: NonExtClassInfo,
   618	    lvalue: NameExpr,
   619	    stmt: AssignmentStmt,
   620	    get_type_info: Callable[[AssignmentStmt], TypeInfo | None] | None = None,
   621	) -> None:
   622	    """Add a class attribute to __annotations__ of a non-extension class."""
   623	    # FIXME: try to better preserve the special forms and type parameters of generics.
   624	    typ: Value | None = None
   625	    if get_type_info is not None:
   626	        type_info = get_type_info(stmt)
   627	        if type_info:
   628	            # NOTE: Using string type information is similar to using
   629	            # `from __future__ import annotations` in standard python.
   630	            # NOTE: For string types we need to use the fullname since it
   631	            # includes the module. If string type doesn't have the module,
   632	            # @dataclass will try to get the current module and fail since the
   633	            # current module is not in sys.modules.
   634	            if builder.current_module == type_info.module_name and stmt.line < type_info.line:
   635	                typ = builder.load_str(type_info.fullname)
   636	            else:
   637	                typ = load_type(builder, type_info, stmt.unanalyzed_type, stmt.line)
   638
   639	    if typ is None:
   640	        # FIXME: if get_type_info is not provided, don't fall back to stmt.type?
   641	        ann_type = get_proper_type(stmt.type)
   642	        if (
   643	            isinstance(stmt.unanalyzed_type, UnboundType)
   644	            and stmt.unanalyzed_type.original_str_expr is not None
   645	        ):
   646	            # Annotation is a forward reference, so don't attempt to load the actual
   647	            # type and load the string instead.
   648	            #
   649	            # TODO: is it possible to determine whether a non-string annotation is
   650	            # actually a forward reference due to the __annotations__ future?
   651	            typ = builder.load_str(stmt.unanalyzed_type.original_str_expr)
   652	        elif isinstance(ann_type, Instance):
   653	            typ = load_type(builder, ann_type.type, stmt.unanalyzed_type, stmt.line)
   654	        else:
   655	            typ = builder.add(LoadAddress(type_object_op.type, type_object_op.src, stmt.line))
   656
   657	    key = builder.load_str(lvalue.name)
   658	    builder.primitive_op(dict_set_item_op, [non_ext.anns, key, typ], stmt.line)

   688	def find_attr_initializers(
   689	    builder: IRBuilder, cdef: ClassDef, skip: Callable[[str, AssignmentStmt], bool] | None = None
   690	) -> tuple[set[str], list[AssignmentStmt]]:
   691	    """Find initializers of attributes in a class body.
   692
   693	    If provided, the skip arg should be a callable which will return whether
   694	    to skip generating a default for an attribute.  It will be passed the name of
   695	    the attribute and the corresponding AssignmentStmt.
   696	    """
   697	    cls = builder.mapper.type_to_ir[cdef.info]
   698	    if cls.builtin_base:
   699	        return set(), []
   700
   701	    attrs_with_defaults = set()
   702
   703	    # Pull out all assignments in classes in the mro so we can initialize them
   704	    # TODO: Support nested statements
   705	    default_assignments = []
   706	    for info in reversed(cdef.info.mro):
   707	        if info not in builder.mapper.type_to_ir:
   708	            continue
   709	        for stmt in info.defn.defs.body:
   710	            if (
   711	                isinstance(stmt, AssignmentStmt)
   712	                and isinstance(stmt.lvalues[0], NameExpr)
   713	                and not is_class_var(stmt.lvalues[0])
   714	                and not isinstance(stmt.rvalue, TempNode)
   715	            ):
   716	                name = stmt.lvalues[0].name
   717	                if name == "__slots__":
   718	                    continue
   719
   720	                if name == "__deletable__":
   721	                    check_deletable_declaration(builder, cls, stmt.line)
   722	                    continue
   723
   724	                if skip is not None and skip(name, stmt):
   725	                    continue
   726
   727	                attr_type = cls.attr_type(name)
   728
   729	                # If the attribute is initialized to None and type isn't optional,
   730	                # doesn't initialize it to anything (special case for "# type:" comments).
   731	                if isinstance(stmt.rvalue, RefExpr) and stmt.rvalue.fullname == "builtins.None":
   732	                    if (
   733	                        not is_optional_type(attr_type)
   734	                        and not is_object_rprimitive(attr_type)
   735	                        and not is_none_rprimitive(attr_type)
   736	                    ):
   737	                        continue
   738
   739	                attrs_with_defaults.add(name)
   740	                default_assignments.append(stmt)
   741
   742	    return attrs_with_defaults, default_assignments

   745	def generate_attr_defaults_init(
   746	    builder: IRBuilder, cdef: ClassDef, default_assignments: list[AssignmentStmt]
   747	) -> None:
   748	    """Generate an initialization method for default attr values (from class vars)."""
   749	    if not default_assignments:
   750	        return
   751	    cls = builder.mapper.type_to_ir[cdef.info]
   752	    if cls.builtin_base:
   753	        return
   754
   755	    with builder.enter_method(cls, "__mypyc_defaults_setup", bool_rprimitive):
   756	        self_var = builder.self()
   757	        for stmt in default_assignments:
   758	            lvalue = stmt.lvalues[0]
   759	            assert isinstance(lvalue, NameExpr)
   760	            if not stmt.is_final_def and not is_constant(stmt.rvalue):
   761	                builder.warning("Unsupported default attribute value", stmt.rvalue.line)
   762
   763	            attr_type = cls.attr_type(lvalue.name)
   764	            val = builder.coerce(builder.accept(stmt.rvalue), attr_type, stmt.line)
   765	            init = SetAttr(self_var, lvalue.name, val, -1)
   766	            init.mark_as_initializer()
   767	            builder.add(init)
   768
   769	        builder.add(Return(builder.true()))

   772	def check_deletable_declaration(builder: IRBuilder, cl: ClassIR, line: int) -> None:
   773	    for attr in cl.deletable:
   774	        if attr not in cl.attributes:
   775	            if not cl.has_attr(attr):
   776	                builder.error(f'Attribute "{attr}" not defined', line)
   777	                continue
   778	            for base in cl.mro:
   779	                if attr in base.property_types:
   780	                    builder.error(f'Cannot make property "{attr}" deletable', line)
   781	                    break
   782	            else:
   783	                _, base = cl.attr_details(attr)
   784	                builder.error(
   785	                    ('Attribute "{}" not defined in "{}" ' + '(defined in "{}")').format(
   786	                        attr, cl.name, base.name
   787	                    ),
   788	                    line,
   789	                )
