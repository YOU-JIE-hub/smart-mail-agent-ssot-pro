 def visit_erased_type(self, t: ErasedType) -> Type:
   203	        # This may happen during type inference if some function argument
   204	        # type is a generic callable, and its erased form will appear in inferred
   205	        # constraints, then solver may check subtyping between them, which will trigger
   206	        # unify_generic_callables(), this is why we can get here. Another example is
   207	        # when inferring type of lambda in generic context, the lambda body contains
   208	        # a generic method in generic class.
   209	        return t

   211	    def visit_instance(self, t: Instance) -> Type:
   212	        args = self.expand_types_with_unpack(list(t.args))
   213
   214	        if isinstance(t.type, FakeInfo):
   215	            # The type checker expands function definitions and bodies
   216	            # if they depend on constrained type variables but the body
   217	            # might contain a tuple type comment (e.g., # type: (int, float)),
   218	            # in which case 't.type' is not yet available.
   219	            #
   220	            # See: https://github.com/python/mypy/issues/16649
   221	            return t.copy_modified(args=args)
   222
   223	        if t.type.fullname == "builtins.tuple":
   224	            # Normalize Tuple[*Tuple[X, ...], ...] -> Tuple[X, ...]
   225	            arg = args[0]
   226	            if isinstance(arg, UnpackType):
   227	                unpacked = get_proper_type(arg.type)
   228	                if isinstance(unpacked, Instance):
   229	                    # TODO: this and similar asserts below may be unsafe because get_proper_type()
   230	                    # may be called during semantic analysis before all invalid types are removed.
   231	                    assert unpacked.type.fullname == "builtins.tuple"
   232	                    args = list(unpacked.args)
   233	        return t.copy_modified(args=args)

   235	    def visit_type_var(self, t: TypeVarType) -> Type:
   236	        # Normally upper bounds can't contain other type variables, the only exception is
   237	        # special type variable Self`0 <: C[T, S], where C is the class where Self is used.
   238	        if t.id.is_self():
   239	            t = t.copy_modified(upper_bound=t.upper_bound.accept(self))
   240	        repl = self.variables.get(t.id, t)
   241	        if isinstance(repl, ProperType) and isinstance(repl, Instance):
   242	            # TODO: do we really need to do this?
   243	            # If I try to remove this special-casing ~40 tests fail on reveal_type().
   244	            return repl.copy_modified(last_known_value=None)
   245	        if isinstance(repl, TypeVarType) and repl.has_default():
   246	            if (tvar_id := repl.id) in self.recursive_tvar_guard:
   247	                return self.recursive_tvar_guard[tvar_id] or repl
   248	            self.recursive_tvar_guard[tvar_id] = None
   249	            repl = repl.accept(self)
   250	            if isinstance(repl, TypeVarType):
   251	                repl.default = repl.default.accept(self)
   252	            self.recursive_tvar_guard[tvar_id] = repl
   253	        return repl

   255	    def visit_param_spec(self, t: ParamSpecType) -> Type:
   256	        # Set prefix to something empty, so we don't duplicate it below.
   257	        repl = self.variables.get(t.id, t.copy_modified(prefix=Parameters([], [], [])))
   258	        if isinstance(repl, ParamSpecType):
   259	            return repl.copy_modified(
   260	                flavor=t.flavor,
   261	                prefix=t.prefix.copy_modified(
   262	                    arg_types=self.expand_types(t.prefix.arg_types) + repl.prefix.arg_types,
   263	                    arg_kinds=t.prefix.arg_kinds + repl.prefix.arg_kinds,
   264	                    arg_names=t.prefix.arg_names + repl.prefix.arg_names,
   265	                ),
   266	            )
   267	        elif isinstance(repl, Parameters):
   268	            assert t.flavor == ParamSpecFlavor.BARE
   269	            return Parameters(
   270	                self.expand_types(t.prefix.arg_types) + repl.arg_types,
   271	                t.prefix.arg_kinds + repl.arg_kinds,
   272	                t.prefix.arg_names + repl.arg_names,
   273	                variables=[*t.prefix.variables, *repl.variables],
   274	                imprecise_arg_kinds=repl.imprecise_arg_kinds,
   275	            )
   276	        else:
   277	            # We could encode Any as trivial parameters etc., but it would be too verbose.
   278	            # TODO: assert this is a trivial type, like Any, Never, or object.
   279	            return repl

   281	    def visit_type_var_tuple(self, t: TypeVarTupleType) -> Type:
   282	        # Sometimes solver may need to expand a type variable with (a copy of) itself
   283	        # (usually together with other TypeVars, but it is hard to filter out TypeVarTuples).
   284	        repl = self.variables.get(t.id, t)
   285	        if isinstance(repl, TypeVarTupleType):
   286	            return repl
   287	        elif isinstance(repl, ProperType) and isinstance(repl, (AnyType, UninhabitedType)):
   288	            # Some failed inference scenarios will try to set all type variables to Never.
   289	            # Instead of being picky and require all the callers to wrap them,
   290	            # do this here instead.
   291	            # Note: most cases when this happens are handled in expand unpack below, but
   292	            # in rare cases (e.g. ParamSpec containing Unpack star args) it may be skipped.
   293	            return t.tuple_fallback.copy_modified(args=[repl])
   294	        raise NotImplementedError

   296	    def visit_unpack_type(self, t: UnpackType) -> Type:
   297	        # It is impossible to reasonably implement visit_unpack_type, because
   298	        # unpacking inherently expands to something more like a list of types.
   299	        #
   300	        # Relevant sections that can call unpack should call expand_unpack()
   301	        # instead.
   302	        # However, if the item is a variadic tuple, we can simply carry it over.
   303	        # In particular, if we expand A[*tuple[T, ...]] with substitutions {T: str},
   304	        # it is hard to assert this without getting proper type. Another important
   305	        # example is non-normalized types when called from semanal.py.
   306	        return UnpackType(t.type.accept(self))

   308	    def expand_unpack(self, t: UnpackType) -> list[Type]:
   309	        assert isinstance(t.type, TypeVarTupleType)
   310	        repl = get_proper_type(self.variables.get(t.type.id, t.type))
   311	        if isinstance(repl, UnpackType):
   312	            repl = get_proper_type(repl.type)
   313	        if isinstance(repl, TupleType):
   314	            return repl.items
   315	        elif (
   316	            isinstance(repl, Instance)
   317	            and repl.type.fullname == "builtins.tuple"
   318	            or isinstance(repl, TypeVarTupleType)
   319	        ):
   320	            return [UnpackType(typ=repl)]
   321	        elif isinstance(repl, (AnyType, UninhabitedType)):
   322	            # Replace *Ts = Any with *Ts = *tuple[Any, ...] and same for Never.
   323	            # These types may appear here as a result of user error or failed inference.
   324	            return [UnpackType(t.type.tuple_fallback.copy_modified(args=[repl]))]
   325	        else:
   326	            raise RuntimeError(f"Invalid type replacement to expand: {repl}")
   327
   328	    def visit_parameters(self, t: Parameters) -> Type:
   329	        return t.copy_modified(arg_types=self.expand_types(t.arg_types))

   331	    def interpolate_args_for_unpack(self, t: CallableType, var_arg: UnpackType) -> list[Type]:
   332	        star_index = t.arg_kinds.index(ARG_STAR)
   333	        prefix = self.expand_types(t.arg_types[:star_index])
   334	        suffix = self.expand_types(t.arg_types[star_index + 1 :])
   335
   336	        var_arg_type = get_proper_type(var_arg.type)
   337	        new_unpack: Type
   338	        if isinstance(var_arg_type, TupleType):
   339	            # We have something like Unpack[Tuple[Unpack[Ts], X1, X2]]
   340	            expanded_tuple = var_arg_type.accept(self)
   341	            assert isinstance(expanded_tuple, ProperType) and isinstance(expanded_tuple, TupleType)
   342	            expanded_items = expanded_tuple.items
   343	            fallback = var_arg_type.partial_fallback
   344	            new_unpack = UnpackType(TupleType(expanded_items, fallback))
   345	        elif isinstance(var_arg_type, TypeVarTupleType):
   346	            # We have plain Unpack[Ts]
   347	            fallback = var_arg_type.tuple_fallback
   348	            expanded_items = self.expand_unpack(var_arg)
   349	            new_unpack = UnpackType(TupleType(expanded_items, fallback))
   350	        # Since get_proper_type() may be called in semanal.py before callable
   351	        # normalization happens, we need to also handle non-normal cases here.
   352	        elif isinstance(var_arg_type, Instance):
   353	            # we have something like Unpack[Tuple[Any, ...]]
   354	            new_unpack = UnpackType(var_arg.type.accept(self))
   355	        else:
   356	            # We have invalid type in Unpack. This can happen when expanding aliases
   357	            # to Callable[[*Invalid], Ret]
   358	            new_unpack = AnyType(TypeOfAny.from_error, line=var_arg.line, column=var_arg.column)
   359	        return prefix + [new_unpack] + suffix

   361	    def visit_callable_type(self, t: CallableType) -> CallableType:
   362	        param_spec = t.param_spec()
   363	        if param_spec is not None:
   364	            repl = self.variables.get(param_spec.id)
   365	            # If a ParamSpec in a callable type is substituted with a
   366	            # callable type, we can't use normal substitution logic,
   367	            # since ParamSpec is actually split into two components
   368	            # *P.args and **P.kwargs in the original type. Instead, we
   369	            # must expand both of them with all the argument types,
   370	            # kinds and names in the replacement. The return type in
   371	            # the replacement is ignored.
   372	            if isinstance(repl, Parameters):
   373	                # We need to expand both the types in the prefix and the ParamSpec itself
   374	                expanded = t.copy_modified(
   375	                    arg_types=self.expand_types(t.arg_types[:-2]) + repl.arg_types,
   376	                    arg_kinds=t.arg_kinds[:-2] + repl.arg_kinds,
   377	                    arg_names=t.arg_names[:-2] + repl.arg_names,
   378	                    ret_type=t.ret_type.accept(self),
   379	                    type_guard=(t.type_guard.accept(self) if t.type_guard is not None else None),
   380	                    type_is=(t.type_is.accept(self) if t.type_is is not None else None),
   381	                    imprecise_arg_kinds=(t.imprecise_arg_kinds or repl.imprecise_arg_kinds),
   382	                    variables=[*repl.variables, *t.variables],
   383	                )
   384	                var_arg = expanded.var_arg()
   385	                if var_arg is not None and isinstance(var_arg.typ, UnpackType):
   386	                    # Sometimes we get new unpacks after expanding ParamSpec.
   387	                    expanded.normalize_trivial_unpack()
   388	                return expanded
   389	            elif isinstance(repl, ParamSpecType):
   390	                # We're substituting one ParamSpec for another; this can mean that the prefix
   391	                # changes, e.g. substitute Concatenate[int, P] in place of Q.
   392	                prefix = repl.prefix
   393	                clean_repl = repl.copy_modified(prefix=Parameters([], [], []))
   394	                return t.copy_modified(
   395	                    arg_types=self.expand_types(t.arg_types[:-2])
   396	                    + prefix.arg_types
   397	                    + [
   398	                        clean_repl.with_flavor(ParamSpecFlavor.ARGS),
   399	                        clean_repl.with_flavor(ParamSpecFlavor.KWARGS),
   400	                    ],
   401	                    arg_kinds=t.arg_kinds[:-2] + prefix.arg_kinds + t.arg_kinds[-2:],
   402	                    arg_names=t.arg_names[:-2] + prefix.arg_names + t.arg_names[-2:],
   403	                    ret_type=t.ret_type.accept(self),
   404	                    from_concatenate=t.from_concatenate or bool(repl.prefix.arg_types),
   405	                    imprecise_arg_kinds=(t.imprecise_arg_kinds or prefix.imprecise_arg_kinds),
   406	                )
   407
   408	        var_arg = t.var_arg()
   409	        needs_normalization = False
   410	        if var_arg is not None and isinstance(var_arg.typ, UnpackType):
   411	            needs_normalization = True
   412	            arg_types = self.interpolate_args_for_unpack(t, var_arg.typ)
   413	        else:
   414	            arg_types = self.expand_types(t.arg_types)
   415	        expanded = t.copy_modified(
   416	            arg_types=arg_types,
   417	            ret_type=t.ret_type.accept(self),
   418	            type_guard=(t.type_guard.accept(self) if t.type_guard is not None else None),
   419	            type_is=(t.type_is.accept(self) if t.type_is is not None else None),
   420	        )
   421	        if needs_normalization:
   422	            return expanded.with_normalized_var_args()
   423	        return expanded

   434	    def expand_types_with_unpack(self, typs: Sequence[Type]) -> list[Type]:
   435	        """Expands a list of types that has an unpack."""
   436	        items: list[Type] = []
   437	        for item in typs:
   438	            if isinstance(item, UnpackType) and isinstance(item.type, TypeVarTupleType):
   439	                items.extend(self.expand_unpack(item))
   440	            else:
   441	                items.append(item.accept(self))
   442	        return items

   444	    def visit_tuple_type(self, t: TupleType) -> Type:
   445	        items = self.expand_types_with_unpack(t.items)
   446	        if len(items) == 1:
   447	            # Normalize Tuple[*Tuple[X, ...]] -> Tuple[X, ...]
   448	            item = items[0]
   449	            if isinstance(item, UnpackType):
   450	                unpacked = get_proper_type(item.type)
   451	                if isinstance(unpacked, Instance):
   452	                    assert unpacked.type.fullname == "builtins.tuple"
   453	                    if t.partial_fallback.type.fullname != "builtins.tuple":
   454	                        # If it is a subtype (like named tuple) we need to preserve it,
   455	                        # this essentially mimics the logic in tuple_fallback().
   456	                        return t.partial_fallback.accept(self)
   457	                    return unpacked
   458	        fallback = t.partial_fallback.accept(self)
   459	        assert isinstance(fallback, ProperType) and isinstance(fallback, Instance)
   460	        return t.copy_modified(items=items, fallback=fallback)

   462	    def visit_typeddict_type(self, t: TypedDictType) -> Type:
   463	        if cached := self.get_cached(t):
   464	            return cached
   465	        fallback = t.fallback.accept(self)
   466	        assert isinstance(fallback, ProperType) and isinstance(fallback, Instance)
   467	        result = t.copy_modified(item_types=self.expand_types(t.items.values()), fallback=fallback)
   468	        self.set_cached(t, result)
   469	        return result

   475	    def visit_union_type(self, t: UnionType) -> Type:
   476	        # Use cache to avoid O(n**2) or worse expansion of types during translation
   477	        # (only for large unions, since caching adds overhead)
   478	        use_cache = len(t.items) > 3
   479	        if use_cache and (cached := self.get_cached(t)):
   480	            return cached
   481
   482	        expanded = self.expand_types(t.items)
   483	        # After substituting for type variables in t.items, some resulting types
   484	        # might be subtypes of others, however calling  make_simplified_union()
   485	        # can cause recursion, so we just remove strict duplicates.
   486	        simplified = UnionType.make_union(
   487	            remove_trivial(flatten_nested_unions(expanded)), t.line, t.column
   488	        )
   489	        # This call to get_proper_type() is unfortunate but is required to preserve
   490	        # the invariant that ProperType will stay ProperType after applying expand_type(),
   491	        # otherwise a single item union of a type alias will break it. Note this should not
   492	        # cause infinite recursion since pathological aliases like A = Union[A, B] are
   493	        # banned at the semantic analysis level.
   494	        result = get_proper_type(simplified)
   495
   496	        if use_cache:
   497	            self.set_cached(t, result)
   498	        return result

   513	        args = self.expand_types_with_unpack(t.args)

   539	def remove_trivial(types: Iterable[Type]) -> list[Type]:
   540	    """Make trivial simplifications on a list of types without calling is_subtype().
   541
   542	    This makes following simplifications:
   543	        * Remove bottom types (taking into account strict optional setting)
   544	        * Remove everything else if there is an `object`
   545	        * Remove strict duplicate types
   546	    """
   547	    removed_none = False
   548	    new_types = []
   549	    all_types = set()
   550	    for t in types:
   551	        p_t = get_proper_type(t)
   552	        if isinstance(p_t, UninhabitedType):
   553	            continue
   554	        if isinstance(p_t, NoneType) and not state.strict_optional:
   555	            removed_none = True
   556	            continue
   557	        if isinstance(p_t, Instance) and p_t.type.fullname == "builtins.object":
   558	            return [p_t]
   559	        if p_t not in all_types:
   560	            new_types.append(t)
   561	            all_types.add(p_t)
   562	    if new_types:
   563	        return
