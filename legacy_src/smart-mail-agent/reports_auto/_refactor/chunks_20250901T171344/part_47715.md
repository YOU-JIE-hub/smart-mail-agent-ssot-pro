   100	        used if a warning needs to be issued.

   203	        """Clear collected data, and prepare to collect more."""
   204	        self.data_lock = self.threading.Lock() if self.threading else None

   215	        # The .should_trace_cache attribute is a cache from file names to
   216	        # coverage.FileDisposition objects, or None.  When a file is first
   217	        # considered for tracing, a FileDisposition is obtained from

   219	        # file should be traced or not.  If it should be, a plugin with dynamic

   225	            # Alex Gaynor said:
   226	            # should_trace_cache is a strictly growing key: once a key is in

   228	            # generally constant, given sufficient context. That is to say, at

   235	            # which is optimized for such workloads.

   239	            self.should_trace_cache = __pypy__.newdict("module")
   240	        else:
   241	            self.should_trace_cache = {}

   289	    # The trace function has to be set individually on each thread before
   290	    # execution begins.  Ironically, the only support the threading module has
   291	    # for running code before the thread main is the tracing function.  So we
   292	    # install this as a trace function, and the first time it's called, it does
   293	    # the real trace installation.

   295	    # New in 3.12: threading.settrace_all_threads: https://github.com/python/cpython/pull/96681

   297	    def _installation_trace(self, frame: FrameType, event: str, arg: Any) -> TTraceFn | None:
   298	        """Called on new threads, installs the real tracer."""
   299	        # Remove ourselves as the trace function.
   300	        sys.settrace(None)
   301	        # Install the real tracer.
   302	        fn: TTraceFn | None = self._start_tracer()
   303	        # Invoke the real trace function with the current event, to be sure
   304	        # not to lose an event.
   305	        if fn:
   306	            fn = fn(frame, event, arg)
   307	        # Return the new trace function to continue tracing in this scope.
   308	        return fn

   310	    def start(self) -> None:
   311	        """Start collecting trace information."""
   312	        # We may be a new collector in a forked process.  The old process'
   313	        # collectors will be in self._collectors, but they won't be usable.
   314	        # Find them and discard them.
   315	        keep_collectors = []
   316	        for c in self._collectors:
   317	            if c.pid == self.pid:
   318	                keep_collectors.append(c)
   319	            else:
   320	                c.post_fork()
   321	        self._collectors[:] = keep_collectors
   322
   323	        if self._collectors:
   324	            self._collectors[-1].pause()
   325
   326	        self.tracers = []
   327
   328	        try:
   329	            # Install the tracer on this thread.
   330	            self._start_tracer()
   331	        except:
   332	            if self._collectors:
   333	                self._collectors[-1].resume()
   334	            raise
   335
   336	        # If _start_tracer succeeded, then we add ourselves to the global
   337	        # stack of collectors.
   338	        self._collectors.append(self)
   339
   340	        # Install our installation tracer in threading, to jump-start other
   341	        # threads.
   342	        if self.core.systrace and self.threading:
   343	            self.threading.settrace(self._installation_trace)

   345	    def stop(self) -> None:
   346	        """Stop collecting trace information."""
   347	        assert self._collectors
   348	        if self._collectors[-1] is not self:
   349	            print("self._collectors:")
   350	            for c in self._collectors:
   351	                print(f"  {c!r}\n{c.origin}")
   352	        assert self._collectors[-1] is self, (
   353	            f"Expected current collector to be {self!r}, but it's {self._collectors[-1]!r}"
   354	        )
   355
   356	        self.pause()
   357
   358	        # Remove this Collector from the stack, and resume the one underneath (if any).
   359	        self._collectors.pop()
   360	        if self._collectors:
   361	            self._collectors[-1].resume()

   426	    def mapped_file_dict(self, d: Mapping[str, T]) -> dict[str, T]:
   427	        """Return a dict like d, but with keys modified by file_mapper."""
   428	        # The call to list(items()) ensures that the GIL protects the dictionary
   429	        # iterator against concurrent modifications by tracers running
   430	        # in other threads. We try three times in case of concurrent
   431	        # access, hoping to get a clean copy.
   432	        runtime_err = None
   433	        for _ in range(3):                      # pragma: part covered
   434	            try:
   435	                items = list(d.items())
   436	            except RuntimeError as ex:          # pragma: cant happen
   437	                runtime_err = ex
   438	            else:
   439	                break
   440	        else:                                   # pragma: cant happen
   441	            assert isinstance(runtime_err, Exception)
   442	            raise runtime_err
   443
   444	        return {self.cached_mapped_file(k): v for k, v in items if v}

   450	    def flush_data(self) -> bool:
   451	        """Save the collected data to our associated `CoverageData`.
   452
   453	        Data may have also been saved along the way. This forces the
   454	        last of the data to be saved.
   455
   456	        Returns True if there was data to save, False if not.
   457	        """
   458	        if not self._activity():
   459	            return False
   460
   461	        if self.branch:
   462	            if self.core.packed_arcs:
   463	                # Unpack the line number pairs packed into integers.  See
   464	                # tracer.c:CTracer_record_pair for the C code that creates
   465	                # these packed ints.
   466	                arc_data: dict[str, list[TArc]] = {}
   467	                packed_data = cast(dict[str, set[int]], self.data)
   468
   469	                # The list() here and in the inner loop are to get a clean copy
   470	                # even as tracers are continuing to add data.
   471	                for fname, packeds in list(packed_data.items()):
   472	                    tuples = []
   473	                    for packed in list(packeds):
   474	                        l1 = packed & 0xFFFFF
   475	                        l2 = (packed & (0xFFFFF << 20)) >> 20
   476	                        if packed & (1 << 40):
   477	                            l1 *= -1
   478	                        if packed & (1 << 41):
   479	                            l2 *= -1
   480	                        tuples.append((l1, l2))
   481	                    arc_data[fname] = tuples
   482	            else:
   483	                arc_data = cast(dict[str, list[TArc]], self.data)
   484	            self.covdata.add_arcs(self.mapped_file_dict(arc_data))
   485	        else:
   486	            line_data = cast(dict[str, set[int]], self.data)
   487	            self.covdata.add_lines(self.mapped_file_dict(line_data))
   488
   489	        file_tracers = {
   490	            k: v for k, v in self.file_tracers.items()
   491	            if v not in self.disabled_plugins
   492	        }
   493	        self.covdata.add_file_tracers(self.mapped_file_dict(file_tracers))
   494
   495	        self._clear_data()
   496	        return True
