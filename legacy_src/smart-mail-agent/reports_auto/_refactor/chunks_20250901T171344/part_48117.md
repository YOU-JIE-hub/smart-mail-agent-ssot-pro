    58	def is_invalid_recursive_alias(seen_nodes: set[TypeAlias], target: Type) -> bool:
    59	    """Flag aliases like A = Union[int, A], T = tuple[int, *T] (and similar mutual aliases).
    60

    62	    """

    64	        if target.alias in seen_nodes:
    65	            return True

    67	        return is_invalid_recursive_alias(seen_nodes | {target.alias}, get_proper_type(target))
    68	    assert isinstance(target, ProperType)
    69	    if not isinstance(target, (UnionType, TupleType)):
    70	        return False
    71	    if isinstance(target, UnionType):
    72	        return any(is_invalid_recursive_alias(seen_nodes, item) for item in target.items)
    73	    for item in target.items:
    74	        if isinstance(item, UnpackType):
    75	            if is_invalid_recursive_alias(seen_nodes, item.type):

    81	    """Prohibit types like Type[Type[...]].

    83	    Such types are explicitly prohibited by PEP 484. Also, they cause problems
