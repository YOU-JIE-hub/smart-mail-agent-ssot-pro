l_control.append(ExceptNonlocalControl(builder.nonlocal_control[-1], old_exc))
   499
   500	    # Process the bodies
   501	    for type, var, handler_body in handlers:
   502	        next_block = None
   503	        if type:
   504	            type_f, type_line = type
   505	            next_block, body_block = BasicBlock(), BasicBlock()
   506	            matches = builder.call_c(exc_matches_op, [type_f()], type_line)
   507	            builder.add(Branch(matches, body_block, next_block, Branch.BOOL))
   508	            builder.activate_block(body_block)
   509	        if var:
   510	            target = builder.get_assignment_target(var)
   511	            builder.assign(target, builder.call_c(get_exc_value_op, [], var.line), var.line)
   512	        handler_body()
   513	        builder.goto(cleanup_block)
   514	        if next_block:
   515	            builder.activate_block(next_block)
   516
   517	    # Reraise the exception if needed
   518	    if next_block:
   519	        builder.call_c(reraise_exception_op, [], NO_TRACEBACK_LINE_NO)
   520	        builder.add(Unreachable())
   521
   522	    builder.nonlocal_control.pop()
   523	    builder.builder.pop_error_handler()
   524
   525	    # Cleanup for if we leave except through normal control flow:
   526	    # restore the saved exc_info information and continue propagating
   527	    # the exception if it exists.
   528	    builder.activate_block(cleanup_block)
   529	    builder.call_c(restore_exc_info_op, [builder.read(old_exc)], line)
   530	    builder.goto(exit_block)
   531
   532	    # Cleanup for if we leave except through a raised exception:
   533	    # restore the saved exc_info information and continue propagating
   534	    # the exception.
   535	    builder.activate_block(double_except_block)
   536	    builder.call_c(restore_exc_info_op, [builder.read(old_exc)], line)
   537	    builder.call_c(keep_propagating_op, [], NO_TRACEBACK_LINE_NO)
   538	    builder.add(Unreachable())
   539
   540	    # If present, compile the else body in the obvious way
   541	    if else_body:
   542	        builder.activate_block(else_block)
   543	        else_body()
   544	        builder.goto(exit_block)
   545
   546	    builder.activate_block(exit_block)

   550	    def body() -> None:
   551	        builder.accept(t.body)
   552
   553	    # Work around scoping woes
   554	    def make_handler(body: Block) -> GenFunc:
   555	        return lambda: builder.accept(body)
   556
   557	    def make_entry(type: Expression) -> tuple[ValueGenFunc, int]:
   558	        return (lambda: builder.accept(type), type.line)
   559
   560	    handlers = [
   561	        (make_entry(type) if type else None, var, make_handler(body))
   562	        for type, var, body in zip(t.types, t.vars, t.handlers)
   563	    ]
   564	    else_body = (lambda: builder.accept(t.else_body)) if t.else_body else None
   565	    transform_try_except(builder, body, handlers, else_body, t.line)

   590	    builder: IRBuilder,

   592	    return_entry: BasicBlock,
   593	    main_entry: BasicBlock,

   595	    ret_reg: Register | AssignmentTarget | None,

   600	    builder.activate_block(main_entry)
   601	    if ret_reg:

   611	    if ret_reg:
   612	        builder.assign(ret_reg, builder.add(LoadErrorValue(builder.ret_types[-1])), -1)
   613	    builder.add(Assign(old_exc, builder.call_c(error_catch_op, [], -1)))
   614	    builder.goto(finally_block)

   621	) -> tuple[BasicBlock, FinallyNonlocalControl]:

   634	def try_finally_resolve_control(
   635	    builder: IRBuilder,
   636	    cleanup_block: BasicBlock,
   637	    finally_control: FinallyNonlocalControl,
   638	    old_exc: Value,
   639	    ret_reg: Register | AssignmentTarget | None,
   640	) -> BasicBlock:
   641	    """Resolve the control flow out of a finally block.
   642
   643	    This means returning if there was a return, propagating
   644	    exceptions, break/continue (soon), or just continuing on.
   645	    """
   646	    reraise, rest = BasicBlock(), BasicBlock()
   647	    builder.add(Branch(old_exc, rest, reraise, Branch.IS_ERROR))
   648
   649	    # Reraise the exception if there was one
   650	    builder.activate_block(reraise)
   651	    builder.call_c(reraise_exception_op, [], NO_TRACEBACK_LINE_NO)
   652	    builder.add(Unreachable())
   653	    builder.builder.pop_error_handler()
   654
   655	    # If there was a return, keep returning
   656	    if ret_reg:
   657	        builder.activate_block(rest)
   658	        return_block, rest = BasicBlock(), BasicBlock()
   659	        # For spill targets in try/finally, use nullable read to avoid AttributeError
   660	        if isinstance(ret_reg, AssignmentTargetAttr) and ret_reg.attr.startswith(TEMP_ATTR_NAME):
   661	            ret_val = builder.read_nullable_attr(ret_reg.obj, ret_reg.attr, -1)
   662	        else:
   663	            ret_val = builder.read(ret_reg)
   664	        builder.add(Branch(ret_val, rest, return_block, Branch.IS_ERROR))
   665
   666	        builder.activate_block(return_block)
   667	        builder.nonlocal_control[-1].gen_return(builder, ret_val, -1)
   668
   669	    # TODO: handle break/continue
   670	    builder.activate_block(rest)
   671	    out_block = BasicBlock()
   672	    builder.goto(out_block)
   673
   674	    # If there was an exception, restore again
   675	    builder.activate_block(cleanup_block)
   676	    finally_control.gen_cleanup(builder, -1)
   677	    builder.call_c(keep_propagating_op, [], NO_TRACEBACK_LINE_NO)
   678	    builder.add(Unreachable())
   679
   680	    return out_block

   683	def transform_try_finally_stmt(
   684	    builder: IRBuilder, try_body: GenFunc, finally_body: GenFunc, line: int = -1
   685	) -> None:
   686	    """Generalized try/finally handling that takes functions to gen the bodies.
   687
   688	    The point of this is to also be able to support with."""
   689	    # Finally is a big pain, because there are so many ways that
   690	    # exits can occur. We emit 10+ basic blocks for every finally!
   691
   692	    err_handler, main_entry, return_entry, finally_block = (
   693	        BasicBlock(),
   694	        BasicBlock(),
   695	        BasicBlock(),
   696	        BasicBlock(),
   697	    )
   698
   699	    # Compile the body of the try
   700	    ret_reg = try_finally_try(builder, err_handler, return_entry, main_entry, try_body)
   701
   702	    # Set up the entry blocks for the finally statement
   703	    old_exc = try_finally_entry_blocks(
   704	        builder, err_handler, return_entry, main_entry, finally_block, ret_reg
   705	    )
   706
   707	    # Compile the body of the finally
   708	    cleanup_block, finally_control = try_finally_body(
   709	        builder, finally_block, finally_body, old_exc
   710	    )
   711
   712	    # Resolve the control flow out of the finally block
   713	    out_block = try_finally_resolve_control(
   714	        builder, cleanup_block, finally_control, old_exc, ret_reg
   715	    )
   716
   717	    builder.activate_block(out_block)

   720	def transform_try_finally_stmt_async(
   721	    builder: IRBuilder, try_body: GenFunc, finally_body: GenFunc, line: int = -1
   722	) -> None:
   723	    """Async-aware try/finally handling for when finally contains await.
   724
   725	    This version uses a modified approach that preserves exceptions across await."""
   726
   727	    # We need to handle returns properly, so we'll use TryFinallyNonlocalControl
   728	    # to track return values, similar to the regular try/finally implementation
   729
   730	    err_handler, main_entry, return_entry, finally_entry = (
   731	        BasicBlock(),
   732	        BasicBlock(),
   733	        BasicBlock(),
   734	        BasicBlock(),
   735	    )
   736
   737	    # Track if we're returning from the try block
   738	    control = TryFinallyNonlocalControl(return_entry)
   739	    builder.builder.push_error_handler(err_handler)
   740	    builder.nonlocal_control.append(control)
   741	    builder.goto_and_activate(BasicBlock())
   742	    try_body()
   743	    builder.goto(main_entry)
   744	    builder.nonlocal_control.pop()
   745	    builder.builder.pop_error_handler()
   746	    ret_reg = control.ret_reg
   747
   748	    # Normal case - no exception or return
   749	    builder.activate_block(main_entry)
   750	    builder.goto(finally_entry)
   751
   752	    # Return case
   753	    builder.activate_block(return_entry)
   754	    builder.goto(finally_entry)
   755
   756	    # Exception case - need to catch to clear the error indicator
   757	    builder.activate_block(err_handler)
   758	    # Catch the error to clear Python's error indicator
   759	    builder.call_c(error_catch_op, [], line)
   760	    # We're not going to use old_exc since it won't survive await
   761	    # The exception is now in sys.exc_info()
   762	    builder.goto(finally_entry)
   763
   764	    # Finally block
   765	    builder.activate_block(finally_entry)
   766
   767	    # Execute finally body
   768	    finally_body()
   769
   770	    # After finally, we need to handle exceptions carefully:
   771	    # 1. If finally raised a new exception, it's in the error indicator - let it propagate
   772	    # 2. If finally didn't raise, check if we need to reraise the original from sys.exc_info()
   773	    # 3. If there was a return, return that value
   774	    # 4. Otherwise, normal exit
   775
   776	    # First, check if there's a current exception in the error indicator
   777	    # (this would be from the finally block)
   778	    no_current_exc = builder.call_c(no_err_occurred_op, [], line)
   779	    finally_raised = BasicBlock()
   780	    check_original = BasicBlock()
   781	    builder.add(Branch(no_current_exc, check_original, finally_raised, Branch.BOOL))
   782
   783	    # Finally raised an exception - let it propagate naturally
   784	    builder.activate_block(finally_raised)
   785	    builder.call_c(keep_propagating_op, [], NO_TRACEBACK_LINE_NO)
   786	    builder.add(Unreachable())
   787
   788	    # No exception from finally, check if we need to handle return or original exception
   789	    builder.activate_block(check_original)
   790
   791	    # Check if we have a return value
   792	    if ret_reg:
   793	        return_block, check_old_exc = BasicBlock(), BasicBlock()
   794	        builder.add(Branch(builder.read(ret_reg), check_old_exc, return_block, Branch.IS_ERROR))
   795
   796	        builder.activate_block(return_block)
   797	        builder.nonlocal_control[-1].gen_return(builder, builder.read(ret_reg), -1)
   798
   799	        builder.activate_block(check_old_exc)
   800
   801	    # Check if we need to reraise the original exception from sys.exc_info
   802	    exc_info = builder.call_c(get_exc_info_op, [], line)
   803	    exc_type = builder.add(TupleGet(exc_info, 0, line))
   804
   805	    # Check if exc_type is None
   806	    none_obj = builder.none_object()
   807	    has_exc = builder.binary_op(exc_type, none_obj, "is not", line)
   808
   809	    reraise_block, exit_block = BasicBlock(), BasicBlock()
   810	    builder.add(Branch(has_exc, reraise_block, exit_block, Branch.BOOL))
   811
   812	    # Reraise the original exception
   813	    builder.activate_block(reraise_block)
   814	    builder.call_c(reraise_exception_op, [], NO_TRACEBACK_LINE_NO)
   815	    builder.add(Unreachable())
   816
   817	    # Normal exit
   818	    builder.activate_block(exit_block)

   823	    def __init__(self) -> None:

   832	def transform_try_stmt(builder: IRBuilder, t: TryStmt) -> None:
   833	    # Our compilation strategy for try/except/else/finally is to
   834	    # treat try/except/else and try/finally as separate language
   835	    # constructs that we compile separately. When we have a
   836	    # try/except/else/finally, we treat the try/except/else as the
   837	    # body of a try/finally block.
   838	    if t.is_star:
   839	        builder.error("Exception groups and except* cannot be compiled yet", t.line)
   840
   841	    # Check if we're in an async function with a finally block that contains await
   842	    use_async_version = False
   843	    if t.finally_body and builder.fn_info.is_coroutine:
   844	        detector = AwaitDetector()
   845	        t.finally_body.accept(detector)
   846
   847	        if detector.has_await:
   848	            # Use the async version that handles exceptions correctly
   849	            use_async_version = True
   850
   851	    if t.finally_body:
   852
   853	        def transform_try_body() -> None:
   854	            if t.handlers:
   855	                transform_try_except_stmt(builder, t)
   856	            else:
   857	                builder.accept(t.body)
   858
   859	        body = t.finally_body
   860
   861	        if use_async_version:
   862	            transform_try_finally_stmt_async(
   863	                builder, transform_try_body, lambda: builder.accept(body), t.line
   864	            )
   865	        else:
   866	            transform_try_finally_stmt(
   867	                builder, transform_try_body, lambda: builder.accept(body), t.line
   868	            )
   869	    else:
   870	        transform_try_except_stmt(builder, t)

   878	def transform_with(
   879	    builder: IRBuilder,
   880	    expr: Expression,
   881	    target: Lvalue | None,
   882	    body: GenFunc,
   883	    is_async: bool,
   884	    line: int,
   885	) -> None:
   886	    # This is basically a straight transcription of the Python code in PEP 343.
   887	    # I don't actually understand why a bunch of it is the way it is.
   888	    # We could probably optimize the case where the manager is compiled by us,
   889	    # but that is not our common case at all, so.
   890
   891	    al = "a" if is_async else ""
   892
   893	    mgr_v = builder.accept(expr)
   894	    is_native = isinstance(mgr_v.type, RInstance)
   895	    if is_native:
   896	        value = builder.add(MethodCall(mgr_v, f"__{al}enter__", args=[], line=line))
   897	        exit_ = None
   898	    else:
   899	        typ = builder.primitive_op(type_op, [mgr_v], line)
   900	        exit_ = builder.maybe_spill(builder.py_get_attr(typ, f"__{al}exit__", line))
   901	        value = builder.py_call(builder.py_get_attr(typ, f"__{al}enter__", line), [mgr_v], line)
   902
   903	    mgr = builder.maybe_spill(mgr_v)
   904	    exc = builder.maybe_spill_assignable(builder.true())
   905	    if is_async:
   906	        value = emit_await(builder, value, line)
   907
   908	    def maybe_natively_call_exit(exc_info: bool) -> Value:
   909	        if exc_info:
   910	            args = get_sys_exc_info(builder)
   911	        else:
   912	            none = builder.none_object()
   913	            args = [none, none, none]
   914
   915	        if is_native:
   916	            assert isinstance(mgr_v.type, RInstance)
   917	            exit_val = builder.gen_method_call(
   918	                builder.read(mgr),
   919	                f"__{al}exit__",
   920	                arg_values=args,
   921	                line=line,
   922	                result_type=none_rprimitive,
   923	            )
   924	        else:
   925	            assert exit_ is not None
   926	            exit_val = builder.py_call(builder.read(exit_), [builder.read(mgr)] + args, line)
   927
   928	        if is_async:
   929	            return emit_await(builder, exit_val, line)
   930	        else:
   931	            return exit_val
   932
   933	    def try_body() -> None:
   934	        if target:
   935	            builder.assign(builder.get_assignment_target(target), value, line)
   936	        body()
   937
   938	    def except_body() -> None:
   939	        builder.assign(exc, builder.false(), line)
   940	        out_block, reraise_block = BasicBlock(), BasicBlock()
   941	        builder.add_bool_branch(maybe_natively_call_exit(exc_info=True), out_block, reraise_block)
   942	        builder.activate_block(reraise_block)
   943	        builder.call_c(reraise_exception_op, [], NO_TRACEBACK_LINE_NO)
   944	        builder.add(Unreachable())
   945	        builder.activate_block(out_block)
   946
   947	    def finally_body() -> None:
   948	        out_block, exit_block = BasicBlock(), BasicBlock()
   949	        builder.add(Branch(builder.read(exc), exit_block, out_block, Branch.BOOL))
   950	        builder.activate_block(exit_block)
   951
   952	        maybe_natively_call_exit(exc_info=False)
   953	        builder.goto_and_activate(out_block)
   954
   955	    transform_try_finally_stmt(
   956	        builder,
   957	        lambda: transform_try_except(builder, try_body, [(None, None, except_body)], None, line),
   958	        finally_body,
   959	        line,
   960	    )
