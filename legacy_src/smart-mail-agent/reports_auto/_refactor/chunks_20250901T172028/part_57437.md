   370	class Color3DLUT(MultibandFilter):
   371	    """Three-dimensional color lookup table.
   372
   373	    Transforms 3-channel pixels using the values of the channels as coordinates
   374	    in the 3D lookup table and interpolating the nearest elements.
   375
   376	    This method allows you to apply almost any color transformation
   377	    in constant time by using pre-calculated decimated tables.
   378
   379	    .. versionadded:: 5.2.0
   380
   381	    :param size: Size of the table. One int or tuple of (int, int, int).
   382	                 Minimal size in any dimension is 2, maximum is 65.
   383	    :param table: Flat lookup table. A list of ``channels * size**3``
   384	                  float elements or a list of ``size**3`` channels-sized
   385	                  tuples with floats. Channels are changed first,
   386	                  then first dimension, then second, then third.
   387	                  Value 0.0 corresponds lowest value of output, 1.0 highest.
   388	    :param channels: Number of channels in the table. Could be 3 or 4.
   389	                     Default is 3.
   390	    :param target_mode: A mode for the result image. Should have not less
   391	                        than ``channels`` channels. Default is ``None``,
   392	                        which means that mode wouldn't be changed.
   393	    """
   394
   395	    name = "Color 3D LUT"
   396
   397	    def __init__(
   398	        self,
   399	        size: int | tuple[int, int, int],
   400	        table: Sequence[float] | Sequence[Sequence[int]] | NumpyArray,
   401	        channels: int = 3,
   402	        target_mode: str | None = None,
   403	        **kwargs: bool,
   404	    ) -> None:
   405	        if channels not in (3, 4):
   406	            msg = "Only 3 or 4 output channels are supported"
   407	            raise ValueError(msg)
   408	        self.size = size = self._check_size(size)
   409	        self.channels = channels
   410	        self.mode = target_mode
   411
   412	        # Hidden flag `_copy_table=False` could be used to avoid extra copying
   413	        # of the table if the table is specially made for the constructor.
   414	        copy_table = kwargs.get("_copy_table", True)
   415	        items = size[0] * size[1] * size[2]
   416	        wrong_size = False
   417
   418	        numpy: ModuleType | None = None
   419	        if hasattr(table, "shape"):
   420	            try:
   421	                import numpy
   422	            except ImportError:
   423	                pass
   424
   425	        if numpy and isinstance(table, numpy.ndarray):
   426	            numpy_table: NumpyArray = table
   427	            if copy_table:
   428	                numpy_table = numpy_table.copy()
   429
   430	            if numpy_table.shape in [
   431	                (items * channels,),
   432	                (items, channels),
   433	                (size[2], size[1], size[0], channels),
   434	            ]:
   435	                table = numpy_table.reshape(items * channels)
   436	            else:
   437	                wrong_size = True
   438
   439	        else:
   440	            if copy_table:
   441	                table = list(table)
   442
   443	            # Convert to a flat list
   444	            if table and isinstance(table[0], (list, tuple)):
   445	                raw_table = cast(Sequence[Sequence[int]], table)
   446	                flat_table: list[int] = []
   447	                for pixel in raw_table:
   448	                    if len(pixel) != channels:
   449	                        msg = (
   450	                            "The elements of the table should "
   451	                            f"have a length of {channels}."
   452	                        )
   453	                        raise ValueError(msg)
   454	                    flat_table.extend(pixel)
   455	                table = flat_table
   456
   457	        if wrong_size or len(table) != items * channels:
   458	            msg = (
   459	                "The table should have either channels * size**3 float items "
   460	                "or size**3 items of channels-sized tuples with floats. "
   461	                f"Table should be: {channels}x{size[0]}x{size[1]}x{size[2]}. "
   462	                f"Actual length: {len(table)}"
   463	            )
   464	            raise ValueError(msg)
   465	        self.table = table
   466
   467	    @staticmethod
   468	    def _check_size(size: Any) -> tuple[int, int, int]:
   469	        try:
   470	            _, _, _ = size
   471	        except ValueError as e:
   472	            msg = "Size should be either an integer or a tuple of three integers."
   473	            raise ValueError(msg) from e
   474	        except TypeError:
   475	            size = (size, size, size)
   476	        size = tuple(int(x) for x in size)
   477	        for size_1d in size:
   478	            if not 2 <= size_1d <= 65:
   479	                msg = "Size should be in [2, 65] range."
   480	                raise ValueError(msg)
   481	        return size
   482
   483	    @classmethod
   484	    def generate(
   485	        cls,
   486	        size: int | tuple[int, int, int],
   487	        callback: Callable[[float, float, float], tuple[float, ...]],
   488	        channels: int = 3,
   489	        target_mode: str | None = None,
   490	    ) -> Color3DLUT:
   491	        """Generates new LUT using provided callback.
   492
   493	        :param size: Size of the table. Passed to the constructor.
   494	        :param callback: Function with three parameters which correspond
   495	                         three color channels. Will be called ``size**3``
   496	                         times with values from 0.0 to 1.0 and should return
   497	                         a tuple with ``channels`` elements.
   498	        :param channels: The number of channels which should return callback.
   499	        :param target_mode: Passed to the constructor of the resulting
   500	                            lookup table.
   501	        """
   502	        size_1d, size_2d, size_3d = cls._check_size(size)
   503	        if channels not in (3, 4):
   504	            msg = "Only 3 or 4 output channels are supported"
   505	            raise ValueError(msg)
   506
   507	        table: list[float] = [0] * (size_1d * size_2d * size_3d * channels)
   508	        idx_out = 0
   509	        for b in range(size_3d):
   510	            for g in range(size_2d):
   511	                for r in range(size_1d):
   512	                    table[idx_out : idx_out + channels] = callback(
   513	                        r / (size_1d - 1), g / (size_2d - 1), b / (size_3d - 1)
   514	                    )
   515	                    idx_out += channels
   516
   517	        return cls(
   518	            (size_1d, size_2d, size_3d),
   519	            table,
   520	            channels=channels,
   521	            target_mode=target_mode,
   522	            _copy_table=False,
   523	        )
   524
   525	    def transform(
   526	        self,
   527	        callback: Callable[..., tuple[float, ...]],
   528	        with_normals: bool = False,
   529	        channels: int | None = None,
   530	        target_mode: str | None = None,
   531	    ) -> Color3DLUT:
   532	        """Transforms the table values using provided callback and returns
   533	        a new LUT with altered values.
   534
   535	        :param callback: A function which takes old lookup table values
   536	                         and returns a new set of values. The number
   537	                         of arguments which function should take is
   538	                         ``self.channels`` or ``3 + self.channels``
   539	                         if ``with_normals`` flag is set.
   540	                         Should return a tuple of ``self.channels`` or
   541	                         ``channels`` elements if it is set.
   542	        :param with_normals: If true, ``callback`` will be called with
   543	                             coordinates in the color cube as the first
   544	                             three arguments. Otherwise, ``callback``
   545	                             will be called only with actual color values.
   546	        :param channels: The number of channels in the resulting lookup table.
   547	        :param target_mode: Passed to the constructor of the resulting
   548	                            lookup table.
   549	        """
   550	        if channels not in (None, 3, 4):
   551	            msg = "Only 3 or 4 output channels are supported"
   552	            raise ValueError(msg)
   553	        ch_in = self.channels
   554	        ch_out = channels or ch_in
   555	        size_1d, size_2d, size_3d = self.size
   556
   557	        table: list[float] = [0] * (size_1d * size_2d * size_3d * ch_out)
   558	        idx_in = 0
   559	        idx_out = 0
   560	        for b in range(size_3d):
   561	            for g in range(size_2d):
   562	                for r in range(size_1d):
   563	                    values = self.table[idx_in : idx_in + ch_in]
   564	                    if with_normals:
   565	                        values = callback(
   566	                            r / (size_1d - 1),
   567	                            g / (size_2d - 1),
   568	                            b / (size_3d - 1),
   569	                            *values,
   570	                        )
   571	                    else:
   572	                        values = callback(*values)
   573	                    table[idx_out : idx_out + ch_out] = values
   574	                    idx_in += ch_in
   575	                    idx_out += ch_out
   576
   577	        return type(self)(
   578	            self.size,
   579	            table,
   580	            channels=ch_out,
   581	            target_mode=target_mode or self.mode,
   582	            _copy_table=False,
   583	        )

   592	            r.append(f"target_mode={self.mode}")
