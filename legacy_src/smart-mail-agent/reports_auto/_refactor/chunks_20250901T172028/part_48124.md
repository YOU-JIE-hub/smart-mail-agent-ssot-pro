   509	def run_analysis(
   510	    blocks: list[BasicBlock],
   511	    cfg: CFG,
   512	    gen_and_kill: OpVisitor[GenAndKill[T]],
   513	    initial: set[T],
   514	    kind: int,
   515	    backward: bool,
   516	    universe: set[T] | None = None,
   517	) -> AnalysisResult[T]:
   518	    """Run a general set-based data flow analysis.
   519
   520	    Args:
   521	        blocks: All basic blocks
   522	        cfg: Control-flow graph for the code
   523	        gen_and_kill: Implementation of gen and kill functions for each op
   524	        initial: Value of analysis for the entry points (for a forward analysis) or the
   525	            exit points (for a backward analysis)
   526	        kind: MUST_ANALYSIS or MAYBE_ANALYSIS
   527	        backward: If False, the analysis is a forward analysis; it's backward otherwise
   528	        universe: For a must analysis, the set of all possible values. This is the starting
   529	            value for the work list algorithm, which will narrow this down until reaching a
   530	            fixed point. For a maybe analysis the iteration always starts from an empty set
   531	            and this argument is ignored.
   532
   533	    Return analysis results: (before, after)
   534	    """
   535	    block_gen = {}
   536	    block_kill = {}
   537
   538	    # Calculate kill and gen sets for entire basic blocks.
   539	    for block in blocks:
   540	        gen: set[T] = set()
   541	        kill: set[T] = set()
   542	        ops = block.ops
   543	        if backward:
   544	            ops = list(reversed(ops))
   545	        for op in ops:
   546	            opgen, opkill = op.accept(gen_and_kill)
   547	            gen = (gen - opkill) | opgen
   548	            kill = (kill - opgen) | opkill
   549	        block_gen[block] = gen
   550	        block_kill[block] = kill
   551
   552	    # Set up initial state for worklist algorithm.
   553	    worklist = list(blocks)
   554	    if not backward:
   555	        worklist.reverse()  # Reverse for a small performance improvement
   556	    workset = set(worklist)
   557	    before: dict[BasicBlock, set[T]] = {}
   558	    after: dict[BasicBlock, set[T]] = {}
   559	    for block in blocks:
   560	        if kind == MAYBE_ANALYSIS:
   561	            before[block] = set()
   562	            after[block] = set()
   563	        else:
   564	            assert universe is not None, "Universe must be defined for a must analysis"
   565	            before[block] = set(universe)
   566	            after[block] = set(universe)
   567
   568	    if backward:
   569	        pred_map = cfg.succ
   570	        succ_map = cfg.pred
   571	    else:
   572	        pred_map = cfg.pred
   573	        succ_map = cfg.succ
   574
   575	    # Run work list algorithm to generate in and out sets for each basic block.
   576	    while worklist:
   577	        label = worklist.pop()
   578	        workset.remove(label)
   579	        if pred_map[label]:
   580	            new_before: set[T] | None = None
   581	            for pred in pred_map[label]:
   582	                if new_before is None:
   583	                    new_before = set(after[pred])
   584	                elif kind == MAYBE_ANALYSIS:
   585	                    new_before |= after[pred]
   586	                else:
   587	                    new_before &= after[pred]
   588	            assert new_before is not None
   589	        else:
   590	            new_before = set(initial)
   591	        before[label] = new_before
   592	        new_after = (new_before - block_kill[label]) | block_gen[label]
   593	        if new_after != after[label]:
   594	            for succ in succ_map[label]:
   595	                if succ not in workset:
   596	                    worklist.append(succ)
   597	                    workset.add(succ)
   598	        after[label] = new_after
   599
   600	    # Run algorithm for each basic block to generate opcode-level sets.
   601	    op_before: dict[tuple[BasicBlock, int], set[T]] = {}
   602	    op_after: dict[tuple[BasicBlock, int], set[T]] = {}
   603	    for block in blocks:
   604	        label = block
   605	        cur = before[label]
   606	        ops_enum: Iterator[tuple[int, Op]] = enumerate(block.ops)
   607	        if backward:
   608	            ops_enum = reversed(list(ops_enum))
   609	        for idx, op in ops_enum:
   610	            op_before[label, idx] = cur
   611	            opgen, opkill = op.accept(gen_and_kill)
   612	            cur = (cur - opkill) | opgen
   613	            op_after[label, idx] = cur
   614	    if backward:
   615	        op_after, op_before = op_before, op_after
   616
   617	    return AnalysisResult(op_before, op_after)

   184	def can_coerce_to(src: RType, dest: RType) -> bool:
   185	    """Check if src can be assigned to dest_rtype.
   186
   187	    Currently okay to have false positives.
   188	    """
   189	    if isinstance(dest, RUnion):
   190	        return any(can_coerce_to(src, d) for d in dest.items)
   191
   192	    if isinstance(dest, RPrimitive):
   193	        if isinstance(src, RPrimitive):
   194	            # If either src or dest is a disjoint type, then they must both be.
   195	            if src.name in disjoint_types and dest.name in disjoint_types:
   196	                return src.name == dest.name
   197	            return src.size == dest.size
   198	        if isinstance(src, RInstance):
   199	            return is_object_rprimitive(dest)
   200	        if isinstance(src, RUnion):
   201	            # IR doesn't have the ability to narrow unions based on
   202	            # control flow, so cannot be a strict all() here.
   203	            return any(can_coerce_to(s, dest) for s in src.items)
   204	        return False
   205
   206	    return True
