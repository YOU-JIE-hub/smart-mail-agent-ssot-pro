   137	def type_object_type(info: TypeInfo, named_type: Callable[[str], Instance]) -> ProperType:
   138	    """Return the type of a type object.
   139
   140	    For a generic type G with type variables T and S the type is generally of form
   141
   142	      Callable[..., G[T, S]]
   143
   144	    where ... are argument types for the __init__/__new__ method (without the self
   145	    argument). Also, the fallback type will be 'type' instead of 'function'.
   146	    """
   147
   148	    # We take the type from whichever of __init__ and __new__ is first
   149	    # in the MRO, preferring __init__ if there is a tie.
   150	    init_method = info.get("__init__")
   151	    new_method = info.get("__new__")
   152	    if not init_method or not is_valid_constructor(init_method.node):
   153	        # Must be an invalid class definition.
   154	        return AnyType(TypeOfAny.from_error)
   155	    # There *should* always be a __new__ method except the test stubs
   156	    # lack it, so just copy init_method in that situation
   157	    new_method = new_method or init_method
   158	    if not is_valid_constructor(new_method.node):
   159	        # Must be an invalid class definition.
   160	        return AnyType(TypeOfAny.from_error)
   161
   162	    # The two is_valid_constructor() checks ensure this.
   163	    assert isinstance(new_method.node, (SYMBOL_FUNCBASE_TYPES, Decorator))
   164	    assert isinstance(init_method.node, (SYMBOL_FUNCBASE_TYPES, Decorator))
   165
   166	    init_index = info.mro.index(init_method.node.info)
   167	    new_index = info.mro.index(new_method.node.info)
   168
   169	    fallback = info.metaclass_type or named_type("builtins.type")
   170	    if init_index < new_index:
   171	        method: FuncBase | Decorator = init_method.node
   172	        is_new = False
   173	    elif init_index > new_index:
   174	        method = new_method.node
   175	        is_new = True
   176	    else:
   177	        if init_method.node.info.fullname == "builtins.object":
   178	            # Both are defined by object.  But if we've got a bogus
   179	            # base class, we can't know for sure, so check for that.
   180	            if info.fallback_to_any:
   181	                # Construct a universal callable as the prototype.
   182	                any_type = AnyType(TypeOfAny.special_form)
   183	                sig = CallableType(
   184	                    arg_types=[any_type, any_type],
   185	                    arg_kinds=[ARG_STAR, ARG_STAR2],
   186	                    arg_names=["_args", "_kwds"],
   187	                    ret_type=any_type,
   188	                    is_bound=True,
   189	                    fallback=named_type("builtins.function"),
   190	                )
   191	                return class_callable(sig, info, fallback, None, is_new=False)
   192
   193	        # Otherwise prefer __init__ in a tie. It isn't clear that this
   194	        # is the right thing, but __new__ caused problems with
   195	        # typeshed (#5647).
   196	        method = init_method.node
   197	        is_new = False
   198	    # Construct callable type based on signature of __init__. Adjust
   199	    # return type and insert type arguments.
   200	    if isinstance(method, FuncBase):
   201	        t = function_type(method, fallback)
   202	    else:
   203	        assert isinstance(method.type, ProperType)
   204	        assert isinstance(method.type, FunctionLike)  # is_valid_constructor() ensures this
   205	        t = method.type
   206	    return type_object_type_from_function(t, info, method.info, fallback, is_new)

   222	def type_object_type_from_function(
   223	    signature: FunctionLike, info: TypeInfo, def_info: TypeInfo, fallback: Instance, is_new: bool
   224	) -> FunctionLike:
   225	    # We first need to record all non-trivial (explicit) self types in __init__,
   226	    # since they will not be available after we bind them. Note, we use explicit
   227	    # self-types only in the defining class, similar to __new__ (but not exactly the same,
   228	    # see comment in class_callable below). This is mostly useful for annotating library
   229	    # classes such as subprocess.Popen.
   230	    default_self = fill_typevars(info)
   231	    if not is_new and not info.is_newtype:
   232	        orig_self_types = [get_self_type(it, default_self) for it in signature.items]
   233	    else:
   234	        orig_self_types = [None] * len(signature.items)
   235
   236	    # The __init__ method might come from a generic superclass 'def_info'
   237	    # with type variables that do not map identically to the type variables of
   238	    # the class 'info' being constructed. For example:
   239	    #
   240	    #   class A(Generic[T]):
   241	    #       def __init__(self, x: T) -> None: ...
   242	    #   class B(A[List[T]]):
   243	    #      ...
   244	    #
   245	    # We need to map B's __init__ to the type (List[T]) -> None.
   246	    signature = bind_self(
   247	        signature,
   248	        original_type=default_self,
   249	        is_classmethod=is_new,
   250	        # Explicit instance self annotations have special handling in class_callable(),
   251	        # we don't need to bind any type variables in them if they are generic.
   252	        ignore_instances=True,
   253	    )
   254	    signature = cast(FunctionLike, map_type_from_supertype(signature, info, def_info))
   255
   256	    special_sig: str | None = None
   257	    if def_info.fullname == "builtins.dict":
   258	        # Special signature!
   259	        special_sig = "dict"
   260
   261	    if isinstance(signature, CallableType):
   262	        return class_callable(signature, info, fallback, special_sig, is_new, orig_self_types[0])
   263	    else:
   264	        # Overloaded __init__/__new__.
   265	        assert isinstance(signature, Overloaded)
   266	        items: list[CallableType] = []
   267	        for item, orig_self in zip(signature.items, orig_self_types):
   268	            items.append(class_callable(item, info, fallback, special_sig, is_new, orig_self))
   269	        return Overloaded(items)

   273	    init_type: CallableType,
   274	    info: TypeInfo,

   277	    is_new: bool,

   306	        ret_type=ret_type,
   307	        fallback=type_type,
   308	        name=None,
   309	        variables=variables,
   310	        special_sig=special_sig,

   316	def map_type_from_supertype(typ: Type, sub_info: TypeInfo, super_info: TypeInfo) -> Type:
   317	    """Map type variables in a type defined in a supertype context to be valid
   318	    in the subtype context. Assume that the result is unique; if more than
   319	    one type is possible, return one of the alternatives.
   320
   321	    For example, assume
   322
   323	      class D(Generic[S]): ...
   324	      class C(D[E[T]], Generic[T]): ...
   325
   326	    Now S in the context of D would be mapped to E[T] in the context of C.
   327	    """
   328	    # Create the type of self in subtype, of form t[a1, ...].
   329	    inst_type = fill_typevars(sub_info)
   330	    if isinstance(inst_type, TupleType):
   331	        inst_type = tuple_fallback(inst_type)
   332	    # Map the type of self to supertype. This gets us a description of the
   333	    # supertype type variables in terms of subtype variables, i.e. t[t1, ...]
   334	    # so that any type variables in tN are to be interpreted in subtype
   335	    # context.
   336	    inst_type = map_instance_to_supertype(inst_type, super_info)
   337	    # Finally expand the type variables in type with those in the previously
   338	    # constructed type. Note that both type and inst_type may have type
   339	    # variables, but in type they are interpreted in supertype context while
   340	    # in inst_type they are interpreted in subtype context. This works even if
   341	    # the names of type variables in supertype and subtype overlap.
   342	    return expand_type_by_instance(typ, inst_type)
