leType):
   459	            item = tuple_fallback(upper_bound)
   460	        elif isinstance(upper_bound, AnyType):
   461	            with mx.msg.filter_errors():
   462	                return _analyze_member_access(name, fallback, mx, override_info)
   463	    elif isinstance(typ.item, TupleType):
   464	        item = tuple_fallback(typ.item)
   465	    elif isinstance(typ.item, FunctionLike) and typ.item.is_type_obj():
   466	        item = typ.item.fallback
   467	    elif isinstance(typ.item, TypeType):
   468	        # Access member on metaclass object via Type[Type[C]]
   469	        if isinstance(typ.item.item, Instance):
   470	            item = typ.item.item.type.metaclass_type
   471	    ignore_messages = False
   472
   473	    if item is not None:
   474	        fallback = item.type.metaclass_type or fallback
   475
   476	    if item and not mx.is_operator:
   477	        # See comment above for why operators are skipped
   478	        result = analyze_class_attribute_access(
   479	            item, name, mx, mcs_fallback=fallback, override_info=override_info
   480	        )
   481	        if result:
   482	            if not (isinstance(get_proper_type(result), AnyType) and item.type.fallback_to_any):
   483	                return result
   484	            else:
   485	                # We don't want errors on metaclass lookup for classes with Any fallback
   486	                ignore_messages = True
   487
   488	    with mx.msg.filter_errors(filter_errors=ignore_messages):
   489	        return _analyze_member_access(name, fallback, mx, override_info)
   490

   493	    with mx.msg.disable_type_names():

   497	            item_mx = mx.copy_modified(self_type=subtype)
   498	            results.append(_analyze_member_access(name, subtype, item_mx))
   499	    return make_simplified_union(results)

   516	def analyze_member_var_access(
   517	    name: str, itype: Instance, info: TypeInfo, mx: MemberContext
   518	) -> Type:
   519	    """Analyse attribute access that does not target a method.
   520
   521	    This is logically part of analyze_member_access and the arguments are similar.
   522
   523	    original_type is the type of E in the expression E.var
   524	    """
   525	    # It was not a method. Try looking up a variable.
   526	    node = info.get(name)
   527	    v = node.node if node else None
   528
   529	    mx.chk.warn_deprecated(v, mx.context)
   530
   531	    vv = v
   532	    is_trivial_self = False
   533	    if isinstance(vv, Decorator):
   534	        # The associated Var node of a decorator contains the type.
   535	        v = vv.var
   536	        is_trivial_self = vv.func.is_trivial_self and not vv.decorators
   537	        if mx.is_super and not mx.suppress_errors:
   538	            validate_super_call(vv.func, mx)
   539	    if isinstance(v, FuncDef):
   540	        assert False, "Did not expect a function"
   541	    if isinstance(v, MypyFile):
   542	        mx.chk.module_refs.add(v.fullname)
   543
   544	    if isinstance(vv, (TypeInfo, TypeAlias, MypyFile, TypeVarLikeExpr)):
   545	        # If the associated variable is a TypeInfo synthesize a Var node for
   546	        # the purposes of type checking.  This enables us to type check things
   547	        # like accessing class attributes on an inner class. Similar we allow
   548	        # using qualified type aliases in runtime context. For example:
   549	        #     class C:
   550	        #         A = List[int]
   551	        #     x = C.A() <- this is OK
   552	        typ = mx.chk.expr_checker.analyze_static_reference(vv, mx.context, mx.is_lvalue)
   553	        v = Var(name, type=typ)
   554	        v.info = info
   555
   556	    if isinstance(v, Var):
   557	        implicit = info[name].implicit
   558
   559	        # An assignment to final attribute is always an error,
   560	        # independently of types.
   561	        if mx.is_lvalue and not mx.chk.get_final_context():
   562	            check_final_member(name, info, mx.msg, mx.context)
   563
   564	        return analyze_var(name, v, itype, mx, implicit=implicit, is_trivial_self=is_trivial_self)
   565	    elif (
   566	        not v
   567	        and name not in ["__getattr__", "__setattr__", "__getattribute__"]
   568	        and not mx.is_operator
   569	        and mx.module_symbol_table is None
   570	    ):
   571	        # Above we skip ModuleType.__getattr__ etc. if we have a
   572	        # module symbol table, since the symbol table allows precise
   573	        # checking.
   574	        if not mx.is_lvalue:
   575	            for method_name in ("__getattribute__", "__getattr__"):
   576	                method = info.get_method(method_name)
   577
   578	                # __getattribute__ is defined on builtins.object and returns Any, so without
   579	                # the guard this search will always find object.__getattribute__ and conclude
   580	                # that the attribute exists
   581	                if method and method.info.fullname != "builtins.object":
   582	                    bound_method = analyze_decorator_or_funcbase_access(
   583	                        defn=method, itype=itype, name=method_name, mx=mx
   584	                    )
   585	                    typ = map_instance_to_supertype(itype, method.info)
   586	                    getattr_type = get_proper_type(expand_type_by_instance(bound_method, typ))
   587	                    if isinstance(getattr_type, CallableType):
   588	                        result = getattr_type.ret_type
   589	                    else:
   590	                        result = getattr_type
   591
   592	                    # Call the attribute hook before returning.
   593	                    fullname = f"{method.info.fullname}.{name}"
   594	                    hook = mx.chk.plugin.get_attribute_hook(fullname)
   595	                    if hook:
   596	                        result = hook(
   597	                            AttributeContext(
   598	                                get_proper_type(mx.original_type),
   599	                                result,
   600	                                mx.is_lvalue,
   601	                                mx.context,
   602	                                mx.chk,
   603	                            )
   604	                        )
   605	                    return result
   606	        else:
   607	            setattr_meth = info.get_method("__setattr__")
   608	            if setattr_meth and setattr_meth.info.fullname != "builtins.object":
   609	                bound_type = analyze_decorator_or_funcbase_access(
   610	                    defn=setattr_meth,
   611	                    itype=itype,
   612	                    name="__setattr__",
   613	                    mx=mx.copy_modified(is_lvalue=False),
   614	                )
   615	                typ = map_instance_to_supertype(itype, setattr_meth.info)
   616	                setattr_type = get_proper_type(expand_type_by_instance(bound_type, typ))
   617	                if isinstance(setattr_type, CallableType) and len(setattr_type.arg_types) > 0:
   618	                    return setattr_type.arg_types[-1]
   619
   620	    if itype.type.fallback_to_any:
   621	        return AnyType(TypeOfAny.special_form)
   622
   623	    # Could not find the member.
   624	    if itype.extra_attrs and name in itype.extra_attrs.attrs:
   625	        # For modules use direct symbol table lookup.
   626	        if not itype.extra_attrs.mod_name:
   627	            return itype.extra_attrs.attrs[name]
   628
   629	    if mx.is_super and not mx.suppress_errors:
   630	        mx.msg.undefined_in_superclass(name, mx.context)
   631	        return AnyType(TypeOfAny.from_error)
   632	    else:
   633	        ret = report_missing_attribute(mx.original_type, itype, name, mx)
   634	        # Avoid paying double jeopardy if we can't find the member due to --no-implicit-reexport
   635	        if (
   636	            mx.module_symbol_table is not None
   637	            and name in mx.module_symbol_table
   638	            and not mx.module_symbol_table[name].module_public
   639	        ):
   640	            v = mx.module_symbol_table[name].node
   641	            e = NameExpr(name)
   642	            e.set_line(mx.context)
   643	            e.node = v
   644	            return mx.chk.expr_checker.analyze_ref_expr(e, lvalue=mx.is_lvalue)
   645	        return ret
   646

   656	def analyze_descriptor_access(descriptor_type: Type, mx: MemberContext) -> Type:
   657	    """Type check descriptor access.
   658
   659	    Arguments:
   660	        descriptor_type: The type of the descriptor attribute being accessed
   661	            (the type of ``f`` in ``a.f`` when ``f`` is a descriptor).
   662	        mx: The current member access context.
   663	    Return:
   664	        The return type of the appropriate ``__get__/__set__`` overload for the descriptor.
   665	    """
   666	    instance_type = get_proper_type(mx.self_type)
   667	    orig_descriptor_type = descriptor_type
   668	    descriptor_type = get_proper_type(descriptor_type)
   669
   670	    if isinstance(descriptor_type, UnionType):
   671	        # Map the access over union types
   672	        return make_simplified_union(
   673	            [analyze_descriptor_access(typ, mx) for typ in descriptor_type.items]
   674	        )
   675	    elif not isinstance(descriptor_type, Instance):
   676	        return orig_descriptor_type
   677
   678	    if not mx.is_lvalue and not descriptor_type.type.has_readable_member("__get__"):
   679	        return orig_descriptor_type
   680
   681	    # We do this check first to accommodate for descriptors with only __set__ method.
   682	    # If there is no __set__, we type-check that the assigned value matches
   683	    # the return type of __get__. This doesn't match the python semantics,
   684	    # (which allow you to override the descriptor with any value), but preserves
   685	    # the type of accessing the attribute (even after the override).
   686	    if mx.is_lvalue and descriptor_type.type.has_readable_member("__set__"):
   687	        return analyze_descriptor_assign(descriptor_type, mx)
   688
   689	    if mx.is_lvalue and not descriptor_type.type.has_readable_member("__get__"):
   690	        # This turned out to be not a descriptor after all.
   691	        return orig_descriptor_type
   692
   693	    dunder_get = descriptor_type.type.get_method("__get__")
   694	    if dunder_get is None:
   695	        mx.fail(
   696	            message_registry.DESCRIPTOR_GET_NOT_CALLABLE.format(
   697	                descriptor_type.str_with_options(mx.msg.options)
   698	            )
   699	        )
   700	        return AnyType(TypeOfAny.from_error)
   701
   702	    bound_method = analyze_decorator_or_funcbase_access(
   703	        defn=dunder_get,
   704	        itype=descriptor_type,
   705	        name="__get__",
   706	        mx=mx.copy_modified(self_type=descriptor_type),
   707	    )
   708
   709	    typ = map_instance_to_supertype(descriptor_type, dunder_get.info)
   710	    dunder_get_type = expand_type_by_instance(bound_method, typ)
   711
   712	    if isinstance(instance_type, FunctionLike) and instance_type.is_type_obj():
   713	        owner_type = instance_type.items[0].ret_type
   714	        instance_type = NoneType()
   715	    elif isinstance(instance_type, TypeType):
   716	        owner_type = instance_type.item
   717	        instance_type = NoneType()
   718	    else:
   719	        owner_type = instance_type
   720
   721	    callable_name = mx.chk.expr_checker.method_fullname(descriptor_type, "__get__")
   722	    dunder_get_type = mx.chk.expr_checker.transform_callee_type(
   723	        callable_name,
   724	        dunder_get_type,
   725	        [
   726	            TempNode(instance_type, context=mx.context),
   727	            TempNode(TypeType.make_normalized(owner_type), context=mx.context),
   728	        ],
   729	        [ARG_POS, ARG_POS],
   730	        mx.context,
   731	        object_type=descriptor_type,
   732	    )
   733
   734	    _, inferred_dunder_get_type = mx.chk.expr_checker.check_call(
   735	        dunder_get_type,
   736	        [
   737	            TempNode(instance_type, context=mx.context),
   738	            TempNode(TypeType.make_normalized(owner_type), context=mx.context),
   739	        ],
   740	        [ARG_POS, ARG_POS],
   741	        mx.context,
   742	        object_type=descriptor_type,
   743	        callable_name=callable_name,
   744	    )
   745
   746	    mx.chk.check_deprecated(dunder_get, mx.context)
   747	    mx.chk.warn_deprecated_overload_item(
   748	        dunder_get, mx.context, target=inferred_dunder_get_type, selftype=descriptor_type
   749	    )
   750
   751	    inferred_dunder_get_type = get_proper_type(inferred_dunder_get_type)
   752	    if isinstance(inferred_dunder_get_type, AnyType):
   753	        # check_call failed, and will have reported an error
   754	        return inferred_dunder_get_type
   755
   756	    if not isinstance(inferred_dunder_get_type, CallableType):
   757	        mx.fail(
   758	            message_registry.DESCRIPTOR_GET_NOT_CALLABLE.format(
   759	                descriptor_type.str_with_options(mx.msg.options)
   760	            )
   761	        )
   762	        return AnyType(TypeOfAny.from_error)
   763
   764	    return inferred_dunder_get_type.ret_type

   767	def analyze_descriptor_assign(descriptor_type: Instance, mx: MemberContext) -> Type:
   768	    instance_type = get_proper_type(mx.self_type)
   769	    dunder_set = descriptor_type.type.get_method("__set__")
   770	    if dunder_set is None:
   771	        mx.fail(
   772	            message_registry.DESCRIPTOR_SET_NOT_CALLABLE.format(
   773	                descriptor_type.str_with_options(mx.msg.options)
   774	            ).value
   775	        )
   776	        return AnyType(TypeOfAny.from_error)
   777
   778	    bound_method = analyze_decorator_or_funcbase_access(
   779	        defn=dunder_set,
   780	        itype=descriptor_type,
   781	        name="__set__",
   782	        mx=mx.copy_modified(is_lvalue=False, self_type=descriptor_type),
   783	    )
   784	    typ = map_instance_to_supertype(descriptor_type, dunder_set.info)
   785	    dunder_set_type = expand_type_by_instance(bound_method, typ)
   786
   787	    callable_name = mx.chk.expr_checker.method_fullname(descriptor_type, "__set__")
   788	    rvalue = mx.rvalue or TempNode(AnyType(TypeOfAny.special_form), context=mx.context)
   789	    dunder_set_type = mx.chk.expr_checker.transform_callee_type(
   790	        callable_name,
   791	        dunder_set_type,
   792	        [TempNode(instance_type, context=mx.context), rvalue],
   793	        [ARG_POS, ARG_POS],
   794	        mx.context,
   795	        object_type=descriptor_type,
   796	    )
   797
   798	    # For non-overloaded setters, the result should be type-checked like a regular assignment.
   799	    # Hence, we first only try to infer the type by using the rvalue as type context.
   800	    type_context = rvalue
   801	    with mx.msg.filter_errors():
   802	        _, inferred_dunder_set_type = mx.chk.expr_checker.check_call(
   803	            dunder_set_type,
   804	            [TempNode(instance_type, context=mx.context), type_context],
   805	            [ARG_POS, ARG_POS],
   806	            mx.context,
   807	            object_type=descriptor_type,
   808	            callable_name=callable_name,
   809	        )
   810
   811	    # And now we in fact type check the call, to show errors related to wrong arguments
   812	    # count, etc., replacing the type context for non-overloaded setters only.
   813	    inferred_dunder_set_type = get_proper_type(inferred_dunder_set_type)
   814	    if isinstance(inferred_dunder_set_type, CallableType):
   815	        type_context = TempNode(AnyType(TypeOfAny.special_form), context=mx.context)
   816	    mx.chk.expr_checker.check_call(
   817	        dunder_set_type,
   818	        [TempNode(instance_type, context=mx.context), type_context],
   819	        [ARG_POS, ARG_POS],
   820	        mx.context,
   821	        object_type=descriptor_type,
   822	        callable_name=callable_name,
   823	    )
   824
   825	    # Search for possible deprecations:
   826	    mx.chk.check_deprecated(dunder_set, mx.context)
   827	    mx.chk.warn_deprecated_overload_item(
   828	        dunder_set, mx.context, target=inferred_dunder_set_type, selftype=descriptor_type
   829	    )
   830
   831	    # In the following cases, a message already will have been recorded in check_call.
   832	    if (not isinstance(inferred_dunder_set_type, CallableType)) or (
   833	        len(inferred_dunder_set_type.arg_types) < 2
   834	    ):
   835	        return AnyType(TypeOfAny.from_error)
   836	    return inferred_dunder_set_type.arg_types[1]

   851	def analyze_var(
   852	    name: str,
   853	    var: Var,
   854	    itype: Instance,
   855	    mx: MemberContext,
   856	    *,
   857	    implicit: bool = False,
   858	    is_trivial_self: bool = False,
   859	) -> Type:
   860	    """Analyze access to an attribute via a Var node.
   861
   862	    This is conceptually part of analyze_member_access and the arguments are similar.
   863	    itype is the instance type in which attribute should be looked up
   864	    original_type is the type of E in the expression E.var
   865	    if implicit is True, the original Var was created as an assignment to self
   866	    if is_trivial_self is True, we can use fast path for bind_self().
   867	    """
   868	    # Found a member variable.
   869	    original_itype = itype
   870	    itype = map_instance_to_supertype(itype, var.info)
   871	    if var.is_settable_property and mx.is_lvalue:
   872	        typ: Type | None = var.setter_type
   873	        if typ is None and var.is_ready:
   874	            # Existing synthetic properties may not set setter type. Fall back to getter.
   875	            typ = var.type
   876	    else:
   877	        typ = var.type
   878	    if typ:
   879	        if isinstance(typ, PartialType):
   880	            return mx.chk.handle_partial_var_type(typ, mx.is_lvalue, var, mx.context)
   881	        if mx.is_lvalue and not mx.suppress_errors:
   882	            if var.is_property and not var.is_settable_property:
   883	                mx.msg.read_only_property(name, itype.type, mx.context)
   884	            if var.is_classvar:
   885	                mx.msg.cant_assign_to_classvar(name, mx.context)
   886	        # This is the most common cas
