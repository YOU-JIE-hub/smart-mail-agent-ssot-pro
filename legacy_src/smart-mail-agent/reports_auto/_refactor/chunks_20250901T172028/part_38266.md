
   334	        assert self.alias is not None
   335	        if self.alias.no_args:
   336	            # We know that no_args=True aliases like L = List must have an instance
   337	            # as their target.
   338	            assert isinstance(self.alias.target, Instance)  # type: ignore[misc]
   339	            return self.alias.target.copy_modified(args=self.args)
   340
   341	        # TODO: this logic duplicates the one in expand_type_by_instance().
   342	        if self.alias.tvar_tuple_index is None:
   343	            mapping = {v.id: s for (v, s) in zip(self.alias.alias_tvars, self.args)}
   344	        else:
   345	            prefix = self.alias.tvar_tuple_index
   346	            suffix = len(self.alias.alias_tvars) - self.alias.tvar_tuple_index - 1
   347	            start, middle, end = split_with_prefix_and_suffix(tuple(self.args), prefix, suffix)
   348	            tvar = self.alias.alias_tvars[prefix]
   349	            assert isinstance(tvar, TypeVarTupleType)
   350	            mapping = {tvar.id: TupleType(list(middle), tvar.tuple_fallback)}
   351	            for tvar, sub in zip(
   352	                self.alias.alias_tvars[:prefix] + self.alias.alias_tvars[prefix + 1 :], start + end
   353	            ):
   354	                mapping[tvar.id] = sub
   355
   356	        new_tp = self.alias.target.accept(InstantiateAliasVisitor(mapping))
   357	        new_tp.accept(LocationSetter(self.line, self.column))
   358	        new_tp.line = self.line
   359	        new_tp.column = self.column
   360	        return new_tp
   361
   362	    def _partial_expansion(self, nothing_args: bool = False) -> tuple[ProperType, bool]:
   363	        # Private method mostly for debugging and testing.
   364	        unroller = UnrollAliasVisitor(set(), {})
   365	        if nothing_args:
   366	            alias = self.copy_modified(args=[UninhabitedType()] * len(self.args))
   367	        else:
   368	            alias = self
   369	        unrolled = alias.accept(unroller)
   370	        assert isinstance(unrolled, ProperType)
   371	        return unrolled, unroller.recursed
   372
   373	    def expand_all_if_possible(self, nothing_args: bool = False) -> ProperType | None:
   374	        """Attempt a full expansion of the type alias (including nested aliases).
   375
   376	        If the expansion is not possible, i.e. the alias is (mutually-)recursive,
   377	        return None. If nothing_args is True, replace all type arguments with an
   378	        UninhabitedType() (used to detect recursively defined aliases).
   379	        """
   380	        unrolled, recursed = self._partial_expansion(nothing_args=nothing_args)
   381	        if recursed:
   382	            return None
   383	        return unrolled
   384
   385	    @property
   386	    def is_recursive(self) -> bool:
   387	        """Whether this type alias is recursive.
   388
   389	        Note this doesn't check generic alias arguments, but only if this alias
   390	        *definition* is recursive. The property value thus can be cached on the
   391	        underlying TypeAlias node. If you want to include all nested types, use
   392	        has_recursive_types() function.
   393	        """
   394	        assert self.alias is not None, "Unfixed type alias"
   395	        is_recursive = self.alias._is_recursive
   396	        if is_recursive is None:
   397	            is_recursive = self.expand_all_if_possible(nothing_args=True) is None
   398	            # We cache the value on the underlying TypeAlias node as an optimization,
   399	            # since the value is the same for all instances of the same alias.
   400	            self.alias._is_recursive = is_recursive
   401	        return is_recursive

   511	class TypeVarId:
   512	    # A type variable is uniquely identified by its raw id and meta level.
   513
   514	    # For plain variables (type parameters of generic classes and
   515	    # functions) raw ids are allocated by semantic analysis, using
   516	    # positive ids 1, 2, ... for generic class parameters and negative
   517	    # ids -1, ... for generic function type arguments. A special value 0
   518	    # is reserved for Self type variable (autogenerated). This convention
   519	    # is only used to keep type variable ids distinct when allocating
   520	    # them; the type checker makes no distinction between class and
   521	    # function type variables.
   522
   523	    # Metavariables are allocated unique ids starting from 1.
   524	    raw_id: int
   525
   526	    # Level of the variable in type inference. Currently either 0 for
   527	    # declared types, or 1 for type inference metavariables.
   528	    meta_level: int = 0
   529
   530	    # Class variable used for allocating fresh ids for metavariables.
   531	    next_raw_id: ClassVar[int] = 1
   532
   533	    # Fullname of class or function/method which declares this type
   534	    # variable (not the fullname of the TypeVar definition!), or ''
   535	    namespace: str
   536
   537	    def __init__(self, raw_id: int, meta_level: int = 0, *, namespace: str = "") -> None:
   538	        self.raw_id = raw_id
   539	        self.meta_level = meta_level
   540	        self.namespace = namespace
   541
   542	    @staticmethod
   543	    def new(meta_level: int) -> TypeVarId:
   544	        raw_id = TypeVarId.next_raw_id
   545	        TypeVarId.next_raw_id += 1
   546	        return TypeVarId(raw_id, meta_level)
   547
   548	    def __repr__(self) -> str:
   549	        return self.raw_id.__repr__()
   550
   551	    def __eq__(self, other: object) -> bool:
   552	        return (
   553	            isinstance(other, TypeVarId)
   554	            and self.raw_id == other.raw_id
   555	            and self.meta_level == other.meta_level
   556	            and self.namespace == other.namespace
   557	        )
   558
   559	    def __ne__(self, other: object) -> bool:
   560	        return not (self == other)
   561
   562	    def __hash__(self) -> int:
   563	        return hash((self.raw_id, self.meta_level, self.namespace))
   564
   565	    def is_meta_var(self) -> bool:
   566	        return self.meta_level > 0
   567
   568	    def is_self(self) -> bool:
   569	        # This is a special value indicating typing.Self variable.
   570	        return self.raw_id == 0
