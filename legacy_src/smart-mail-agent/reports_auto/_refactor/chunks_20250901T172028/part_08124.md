    54	class Reports:
    55	    def __init__(self, data_dir: str, report_dirs: dict[str, str]) -> None:
    56	        self.data_dir = data_dir
    57	        self.reporters: list[AbstractReporter] = []
    58	        self.named_reporters: dict[str, AbstractReporter] = {}
    59
    60	        for report_type, report_dir in sorted(report_dirs.items()):
    61	            self.add_report(report_type, report_dir)
    62
    63	    def add_report(self, report_type: str, report_dir: str) -> AbstractReporter:
    64	        try:
    65	            return self.named_reporters[report_type]
    66	        except KeyError:
    67	            pass
    68	        reporter_cls, needs_lxml = reporter_classes[report_type]
    69	        if needs_lxml and not LXML_INSTALLED:
    70	            print(
    71	                (
    72	                    "You must install the lxml package before you can run mypy"
    73	                    " with `--{}-report`.\n"
    74	                    "You can do this with `python3 -m pip install lxml`."
    75	                ).format(report_type),
    76	                file=sys.stderr,
    77	            )
    78	            raise ImportError
    79	        reporter = reporter_cls(self, report_dir)
    80	        self.reporters.append(reporter)
    81	        self.named_reporters[report_type] = reporter
    82	        return reporter
    83
    84	    def file(
    85	        self,
    86	        tree: MypyFile,
    87	        modules: dict[str, MypyFile],
    88	        type_map: dict[Expression, Type],
    89	        options: Options,
    90	    ) -> None:
    91	        for reporter in self.reporters:
    92	            reporter.on_file(tree, modules, type_map, options)

   161	class LineCountReporter(AbstractReporter):
   162	    def __init__(self, reports: Reports, output_dir: str) -> None:
   163	        super().__init__(reports, output_dir)
   164	        self.counts: dict[str, tuple[int, int, int, int]] = {}
   165
   166	    def on_file(
   167	        self,
   168	        tree: MypyFile,
   169	        modules: dict[str, MypyFile],
   170	        type_map: dict[Expression, Type],
   171	        options: Options,
   172	    ) -> None:
   173	        # Count physical lines.  This assumes the file's encoding is a
   174	        # superset of ASCII (or at least uses \n in its line endings).
   175	        try:
   176	            with open(tree.path, "rb") as f:
   177	                physical_lines = len(f.readlines())
   178	        except IsADirectoryError:
   179	            # can happen with namespace packages
   180	            physical_lines = 0
   181
   182	        func_counter = FuncCounterVisitor()
   183	        tree.accept(func_counter)
   184	        unannotated_funcs, annotated_funcs = func_counter.counts
   185	        total_funcs = annotated_funcs + unannotated_funcs
   186
   187	        # Don't count lines or functions as annotated if they have their errors ignored.
   188	        if options.ignore_errors:
   189	            annotated_funcs = 0
   190
   191	        imputed_annotated_lines = (
   192	            physical_lines * annotated_funcs // total_funcs if total_funcs else physical_lines
   193	        )
   194
   195	        self.counts[tree._fullname] = (
   196	            imputed_annotated_lines,
   197	            physical_lines,
   198	            annotated_funcs,
   199	            total_funcs,
   200	        )
   201
   202	    def on_finish(self) -> None:
   203	        counts: list[tuple[tuple[int, int, int, int], str]] = sorted(
   204	            ((c, p) for p, c in self.counts.items()), reverse=True
   205	        )
   206	        total_counts = tuple(sum(c[i] for c, p in counts) for i in range(4))
   207	        with open(os.path.join(self.output_dir, "linecount.txt"), "w") as f:
   208	            f.write("{:7} {:7} {:6} {:6} total\n".format(*total_counts))
   209	            for c, p in counts:
   210	                f.write(f"{c[0]:7} {c[1]:7} {c[2]:6} {c[3]:6} {p}\n")
   211

   216	class AnyExpressionsReporter(AbstractReporter):
   217	    """Report frequencies of different kinds of Any types."""
   218
   219	    def __init__(self, reports: Reports, output_dir: str) -> None:
   220	        super().__init__(reports, output_dir)
   221	        self.counts: dict[str, tuple[int, int]] = {}
   222	        self.any_types_counter: dict[str, collections.Counter[int]] = {}
   223
   224	    def on_file(
   225	        self,
   226	        tree: MypyFile,
   227	        modules: dict[str, MypyFile],
   228	        type_map: dict[Expression, Type],
   229	        options: Options,
   230	    ) -> None:
   231	        visitor = stats.StatisticsVisitor(
   232	            inferred=True,
   233	            filename=tree.fullname,
   234	            modules=modules,
   235	            typemap=type_map,
   236	            all_nodes=True,
   237	            visit_untyped_defs=False,
   238	        )
   239	        tree.accept(visitor)
   240	        self.any_types_counter[tree.fullname] = visitor.type_of_any_counter
   241	        num_unanalyzed_lines = list(visitor.line_map.values()).count(stats.TYPE_UNANALYZED)
   242	        # count each line of dead code as one expression of type "Any"
   243	        num_any = visitor.num_any_exprs + num_unanalyzed_lines
   244	        num_total = visitor.num_imprecise_exprs + visitor.num_precise_exprs + num_any
   245	        if num_total > 0:
   246	            self.counts[tree.fullname] = (num_any, num_total)
   247
   248	    def on_finish(self) -> None:
   249	        self._report_any_exprs()
   250	        self._report_types_of_anys()
   251
   252	    def _write_out_report(
   253	        self, filename: str, header: list[str], rows: list[list[str]], footer: list[str]
   254	    ) -> None:
   255	        row_len = len(header)
   256	        assert all(len(row) == row_len for row in rows + [header, footer])
   257	        min_column_distance = 3  # minimum distance between numbers in two columns
   258	        widths = [-1] * row_len
   259	        for row in rows + [header, footer]:
   260	            for i, value in enumerate(row):
   261	                widths[i] = max(widths[i], len(value))
   262	        for i, w in enumerate(widths):
   263	            # Do not add min_column_distance to the first column.
   264	            if i > 0:
   265	                widths[i] = w + min_column_distance
   266	        with open(os.path.join(self.output_dir, filename), "w") as f:
   267	            header_str = ("{:>{}}" * len(widths)).format(*itertools.chain(*zip(header, widths)))
   268	            separator = "-" * len(header_str)
   269	            f.write(header_str + "\n")
   270	            f.write(separator + "\n")
   271	            for row_values in rows:
   272	                r = ("{:>{}}" * len(widths)).format(*itertools.chain(*zip(row_values, widths)))
   273	                f.write(r + "\n")
   274	            f.write(separator + "\n")
   275	            footer_str = ("{:>{}}" * len(widths)).format(*itertools.chain(*zip(footer, widths)))
   276	            f.write(footer_str + "\n")

   278	    def _report_any_exprs(self) -> None:
   279	        total_any = sum(num_any for num_any, _ in self.counts.values())
   280	        total_expr = sum(total for _, total in self.counts.values())
   281	        total_coverage = 100.0
   282	        if total_expr > 0:
   283	            total_coverage = (float(total_expr - total_any) / float(total_expr)) * 100
   284
   285	        column_names = ["Name", "Anys", "Exprs", "Coverage"]
   286	        rows: list[list[str]] = []
   287	        for filename in sorted(self.counts):
   288	            (num_any, num_total) = self.counts[filename]
   289	            coverage = (float(num_total - num_any) / float(num_total)) * 100
   290	            coverage_str = f"{coverage:.2f}%"
   291	            rows.append([filename, str(num_any), str(num_total), coverage_str])
   292	        rows.sort(key=lambda x: x[0])
   293	        total_row = ["Total", str(total_any), str(total_expr), f"{total_coverage:.2f}%"]
   294	        self._write_out_report("any-exprs.txt", column_names, rows, total_row)

   296	    def _report_types_of_anys(self) -> None:
   297	        total_counter: collections.Counter[int] = collections.Counter()
   298	        for counter in self.any_types_counter.values():
   299	            for any_type, value in counter.items():
   300	                total_counter[any_type] += value
   301	        file_column_name = "Name"
   302	        total_row_name = "Total"
   303	        column_names = [file_column_name] + list(type_of_any_name_map.values())
   304	        rows: list[list[str]] = []
   305	        for filename, counter in self.any_types_counter.items():
   306	            rows.append([filename] + [str(counter[typ]) for typ in type_of_any_name_map])
   307	        rows.sort(key=lambda x: x[0])
   308	        total_row = [total_row_name] + [str(total_counter[typ]) for typ in type_of_any_name_map]
   309	        self._write_out_report("types-of-anys.txt", column_names, rows, total_row)

   315	class LineCoverageVisitor(TraverserVisitor):
   316	    def __init__(self, source: list[str]) -> None:
   317	        self.source = source
   318
   319	        # For each line of source, we maintain a pair of
   320	        #  * the indentation level of the surrounding function
   321	        #    (-1 if not inside a function), and
   322	        #  * whether the surrounding function is typed.
   323	        # Initially, everything is covered at indentation level -1.
   324	        self.lines_covered = [(-1, True) for l in source]
   325
   326	    # The Python AST has position information for the starts of
   327	    # elements, but not for their ends. Fortunately the
   328	    # indentation-based syntax makes it pretty easy to find where a
   329	    # block ends without doing any real parsing.
   330
   331	    # TODO: Handle line continuations (explicit and implicit) and
   332	    # multi-line string literals. (But at least line continuations
   333	    # are normally more indented than their surrounding block anyways,
   334	    # by PEP 8.)
   335
   336	    def indentation_level(self, line_number: int) -> int | None:
   337	        """Return the indentation of a line of the source (specified by
   338	        zero-indexed line number). Returns None for blank lines or comments."""
   339	        line = self.source[line_number]
   340	        indent = 0
   341	        for char in list(line):
   342	            if char == " ":
   343	                indent += 1
   344	            elif char == "\t":
   345	                indent = 8 * ((indent + 8) // 8)
   346	            elif char == "#":
   347	                # Line is a comment; ignore it
   348	                return None
   349	            elif char == "\n":
   350	                # Line is entirely whitespace; ignore it
   351	                return None
   352	            # TODO line continuation (\)
   353	            else:
   354	                # Found a non-whitespace character
   355	                return indent
   356	        # Line is entirely whitespace, and at end of file
   357	        # with no trailing newline; ignore it
   358	        return None
   359
   360	    def visit_func_def(self, defn: FuncDef) -> None:
   361	        start_line = defn.line - 1
   362	        start_indent = None
   363	        # When a function is decorated, sometimes the start line will point to
   364	        # whitespace or comments between the decorator and the function, so
   365	        # we have to look for the start.
   366	        while start_line < len(self.source):
   367	            start_indent = self.indentation_level(start_line)
   368	            if start_indent is not None:
   369	                break
   370	            start_line += 1
   371	        # If we can't find the function give up and don't annotate anything.
   372	        # Our line numbers are not reliable enough to be asserting on.
   373	        if start_indent is None:
   374	            return
   375
   376	        cur_line = start_line + 1
   377	        end_line = cur_line
   378	        # After this loop, function body will be lines [start_line, end_line)
   379	        while cur_line < len(self.source):
   380	            cur_indent = self.indentation_level(cur_line)
   381	            if cur_indent is None:
   382	                # Consume the line, but don't mark it as belonging to the function yet.
   383	                cur_line += 1
   384	            elif cur_indent > start_indent:
   385	                # A non-blank line that belongs to the function.
   386	                cur_line += 1
   387	                end_line = cur_line
   388	            else:
   389	                # We reached a line outside the function definition.
   390	                break
   391
   392	        is_typed = defn.type is not None
   393	        for line in range(start_line, end_line):
   394	            old_indent, _ = self.lines_covered[line]
   395	            # If there was an old indent level for this line, and the new
   396	            # level isn't increasing the indentation, ignore it.
   397	            # This is to be defensive against funniness in our line numbers,
   398	            # which are not always reliable.
   399	            if old_indent <= start_indent:
   400	                self.lines_covered[line] = (start_indent, is_typed)
   401
   402	        # Visit the body, in case there are nested functions
   403	        super().visit_func_def(defn)

   406	class LineCoverageReporter(AbstractReporter):
   407	    """Exact line coverage reporter.
   408
   409	    This reporter writes a JSON dictionary with one field 'lines' to
   410	    the file 'coverage.json' in the specified report directory. The
   411	    value of that field is a dictionary which associates to each
   412	    source file's absolute pathname the list of line numbers that
   413	    belong to typed functions in that file.
   414	    """
   415
   416	    def __init__(self, reports: Reports, output_dir: str) -> None:
   417	        super().__init__(reports, output_dir)
   418	        self.lines_covered: dict[str, list[int]] = {}
   419
   420	    def on_file(
   421	        self,
   422	        tree: MypyFile,
   423	        modules: dict[str, MypyFile],
   424	        type_map: dict[Expression, Type],
   425	        options: Options,
   426	    ) -> None:
   427	        with open(tree.path) as f:
   428	            tree_source = f.readlines()
   429
   430	        coverage_visitor = LineCoverageVisitor(tree_source)
   431	        tree.accept(coverage_visitor)
   432
   433	        covered_lines = []
   434	        for line_number, (_, typed) in enumerate(coverage_visitor.lines_covered):
   435	            if typed:
   436	                covered_lines.append(line_number + 1)
   437
   438	        self.lines_covered[os.path.abspath(tree.path)] = covered_lines
   439
   440	    def on_finish(self) -> None:
   441	        with open(os.path.join(self.output_dir, "coverage.json"), "w") as f:
   442	            json.dump({"lines": self.lines_covered}, f)
   443
