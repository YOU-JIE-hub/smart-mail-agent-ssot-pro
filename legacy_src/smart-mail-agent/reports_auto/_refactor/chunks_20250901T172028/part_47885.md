  4439	    def check_simple_assignment(
  4440	        self,
  4441	        lvalue_type: Type | None,
  4442	        rvalue: Expression,
  4443	        context: Context,
  4444	        msg: ErrorMessage = message_registry.INCOMPATIBLE_TYPES_IN_ASSIGNMENT,
  4445	        lvalue_name: str = "variable",
  4446	        rvalue_name: str = "expression",
  4447	        *,
  4448	        notes: list[str] | None = None,
  4449	        lvalue: Expression | None = None,
  4450	        inferred: Var | None = None,
  4451	    ) -> tuple[Type, Type | None]:
  4452	        if self.is_stub and isinstance(rvalue, EllipsisExpr):
  4453	            # '...' is always a valid initializer in a stub.
  4454	            return AnyType(TypeOfAny.special_form), lvalue_type
  4455	        else:
  4456	            always_allow_any = lvalue_type is not None and not isinstance(
  4457	                get_proper_type(lvalue_type), AnyType
  4458	            )
  4459	            if inferred is None or is_typeddict_type_context(lvalue_type):
  4460	                type_context = lvalue_type
  4461	            else:
  4462	                type_context = None
  4463	            rvalue_type = self.expr_checker.accept(
  4464	                rvalue, type_context=type_context, always_allow_any=always_allow_any
  4465	            )
  4466	            if (
  4467	                lvalue_type is not None
  4468	                and type_context is None
  4469	                and not is_valid_inferred_type(rvalue_type, self.options)
  4470	            ):
  4471	                # Inference in an empty type context didn't produce a valid type, so
  4472	                # try using lvalue type as context instead.
  4473	                rvalue_type = self.expr_checker.accept(
  4474	                    rvalue, type_context=lvalue_type, always_allow_any=always_allow_any
  4475	                )
  4476	                if not is_valid_inferred_type(rvalue_type, self.options) and inferred is not None:
  4477	                    self.msg.need_annotation_for_var(
  4478	                        inferred, context, self.options.python_version
  4479	                    )
  4480	                    rvalue_type = rvalue_type.accept(SetNothingToAny())
  4481
  4482	            if (
  4483	                isinstance(lvalue, NameExpr)
  4484	                and inferred is not None
  4485	                and inferred.type is not None
  4486	                and not inferred.is_final

  4489	                if not is_same_type(inferred.type, new_inferred):
  4490	                    # Should we widen the inferred type or the lvalue? Variables defined
  4491	                    # at module level or class bodies can't be widened in functions, or
  4492	                    # in another module.
  4493	                    if not self.refers_to_different_scope(lvalue):
  4494	                        lvalue_type = make_simplified_union([inferred.type, new_inferred])
  4495	                        if not is_same_type(lvalue_type, inferred.type) and not isinstance(
  4496	                            inferred.type, PartialType
  4497	                        ):
  4498	                            # Widen the type to the union of original and new type.
  4499	                            self.widened_vars.append(inferred.name)
  4500	                            self.set_inferred_type(inferred, lvalue, lvalue_type)
  4501	                            self.binder.put(lvalue, rvalue_type)
  4502	                            # TODO: A bit hacky, maybe add a binder method that does put and
  4503	                            #       updates declaration?
  4504	                            lit = literal_hash(lvalue)
  4505	                            if lit is not None:
  4506	                                self.binder.declarations[lit] = lvalue_type
  4507	            if (
  4508	                isinstance(get_proper_type(lvalue_type), UnionType)
  4509	                # Skip literal types, as they have special logic (for better errors).
  4510	                and not is_literal_type_like(rvalue_type)
  4511	                and not self.simple_rvalue(rvalue)
  4512	            ):
  4513	                # Try re-inferring r.h.s. in empty context, and use that if it
  4514	                # results in a narrower type. We don't do this always because this
  4515	                # may cause some perf impact, plus we want to partially preserve
  4516	                # the old behavior. This helps with various practical examples, see
  4517	                # e.g. testOptionalTypeNarrowedByGenericCall.
  4518	                with self.msg.filter_errors() as local_errors, self.local_type_map() as type_map:
  4519	                    alt_rvalue_type = self.expr_checker.accept(
  4520	                        rvalue, None, always_allow_any=always_allow_any

  4522	                if (
  4523	                    not local_errors.has_new_errors()
  4524	                    # Skip Any type, since it is special cased in binder.
  4525	                    and not isinstance(get_proper_type(alt_rvalue_type), AnyType)
  4526	                    and is_valid_inferred_type(alt_rvalue_type, self.options)
  4527	                    and is_proper_subtype(alt_rvalue_type, rvalue_type)
  4528	                ):
  4529	                    rvalue_type = alt_rvalue_type
  4530	                    self.store_types(type_map)
  4531	            if isinstance(rvalue_type, DeletedType):
  4532	                self.msg.deleted_as_rvalue(rvalue_type, context)
  4533	            if isinstance(lvalue_type, DeletedType):
  4534	                self.msg.deleted_as_lvalue(lvalue_type, context)
  4535	            elif lvalue_type:
  4536	                self.check_subtype(
  4537	                    # Preserve original aliases for error messages when possible.
  4538	                    rvalue_type,
  4539	                    lvalue_type,
  4540	                    context,
  4541	                    msg,
  4542	                    f"{rvalue_name} has type",
  4543	                    f"{lvalue_name} has type",
  4544	                    notes=notes,

  4564	        attribute_type: Type,
  4565	        rvalue: Expression,
  4566	        context: Context,
  4567	    ) -> tuple[Type, Type, bool]:
  4568	        """Type member assignment.
  4569

  4579	            instance_type, TypeType

  4582	            return rvalue_type, attribute_type, True
