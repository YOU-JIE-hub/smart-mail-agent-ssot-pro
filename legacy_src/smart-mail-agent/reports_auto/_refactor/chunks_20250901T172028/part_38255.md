   187	    massively expanding the sizes of types.

   190	    def __init__(self, cache: dict[Type, Type] | None = None) -> None:
   191	        # For deduplication of results

   273	        # Use cache to avoid O(n**2) or worse expansion of types during translation
   274	        if cached := self.get_cached(t):
   275	            return cached
   276	        items = {item_name: item_type.accept(self) for (item_name, item_type) in t.items.items()}
   277	        result = TypedDictType(
   278	            items,

   296	        # (only for large unions, since caching adds overhead)
   297	        use_cache = len(t.items) > 3
   298	        if use_cache and (cached := self.get_cached(t)):
   299	            return cached

   302	            self.translate_types(t.items),

   304	            t.column,
   305	            uses_pep604_syntax=t.uses_pep604_syntax,

   323	            assert isinstance(new, CallableType)  # type: ignore[misc]
   324	            items.append(new)
   325	        return Overloaded(items=items)

   328	        return TypeType.make_normalized(t.item.accept(self), line=t.line, column=t.column)

   333	        # because type aliases are special: some information is contained in the
