   958	def reprocess_nodes(
   959	    manager: BuildManager,
   960	    graph: dict[str, State],
   961	    module_id: str,
   962	    nodeset: set[FineGrainedDeferredNode],
   963	    deps: dict[str, set[str]],
   964	    processed_targets: list[str],
   965	) -> set[str]:
   966	    """Reprocess a set of nodes within a single module.
   967
   968	    Return fired triggers.
   969	    """
   970	    if module_id not in graph:
   971	        manager.log_fine_grained("%s not in graph (blocking errors or deleted?)" % module_id)
   972	        return set()
   973
   974	    file_node = manager.modules[module_id]
   975	    old_symbols = find_symbol_tables_recursive(file_node.fullname, file_node.names)
   976	    old_symbols = {name: names.copy() for name, names in old_symbols.items()}
   977	    old_symbols_snapshot = snapshot_symbol_table(file_node.fullname, file_node.names)
   978
   979	    def key(node: FineGrainedDeferredNode) -> int:
   980	        # Unlike modules which are sorted by name within SCC,
   981	        # nodes within the same module are sorted by line number, because
   982	        # this is how they are processed in normal mode.
   983	        return node.node.line
   984
   985	    nodes = sorted(nodeset, key=key)
   986
   987	    state = graph[module_id]
   988	    options = state.options
   989	    manager.errors.set_file_ignored_lines(
   990	        file_node.path, file_node.ignored_lines, options.ignore_errors or state.ignore_all
   991	    )
   992	    manager.errors.set_skipped_lines(file_node.path, file_node.skipped_lines)
   993
   994	    targets = set()
   995	    for node in nodes:
   996	        target = target_from_node(module_id, node.node)
   997	        if target is not None:
   998	            targets.add(target)
   999	    manager.errors.clear_errors_in_targets(file_node.path, targets)
  1000
  1001	    # If one of the nodes is the module itself, emit any errors that
  1002	    # happened before semantic analysis.
  1003	    for target in targets:
  1004	        if target == module_id:
  1005	            for info in graph[module_id].early_errors:
  1006	                manager.errors.add_error_info(info)
  1007
  1008	    # Strip semantic analysis information.
  1009	    saved_attrs: SavedAttributes = {}
  1010	    for deferred in nodes:
  1011	        processed_targets.append(deferred.node.fullname)
  1012	        strip_target(deferred.node, saved_attrs)
  1013	    semantic_analysis_for_targets(graph[module_id], nodes, graph, saved_attrs)
  1014	    # Merge symbol tables to preserve identities of AST nodes. The file node will remain
  1015	    # the same, but other nodes may have been recreated with different identities, such as
  1016	    # NamedTuples defined using assignment statements.
  1017	    new_symbols = find_symbol_tables_recursive(file_node.fullname, file_node.names)
  1018	    for name in old_symbols:
  1019	        if name in new_symbols:
  1020	            merge_asts(file_node, old_symbols[name], file_node, new_symbols[name])
  1021
  1022	    # Type check.
  1023	    checker = graph[module_id].type_checker()
  1024	    checker.reset()
  1025	    # We seem to need additional passes in fine-grained incremental mode.
  1026	    checker.pass_num = 0
  1027	    checker.last_pass = 3
  1028	    more = checker.check_second_pass(nodes)
  1029	    while more:
  1030	        more = False
  1031	        if graph[module_id].type_checker().check_second_pass():
  1032	            more = True
  1033
  1034	    if manager.options.export_types:
  1035	        manager.all_types.update(graph[module_id].type_map())
  1036
  1037	    new_symbols_snapshot = snapshot_symbol_table(file_node.fullname, file_node.names)
  1038	    # Check if any attribute types were changed and need to be propagated further.
  1039	    changed = compare_symbol_table_snapshots(
  1040	        file_node.fullname, old_symbols_snapshot, new_symbols_snapshot
  1041	    )
  1042	    new_triggered = {make_trigger(name) for name in changed}
  1043
  1044	    # Dependencies may have changed.
  1045	    update_deps(module_id, nodes, graph, deps, options)
  1046
  1047	    # Report missing imports.
  1048	    graph[module_id].verify_dependencies()
  1049
  1050	    graph[module_id].free_state()
  1051
  1052	    return new_triggered

  1094	def lookup_target(
  1095	    manager: BuildManager, target: str
  1096	) -> tuple[list[FineGrainedDeferredNode], TypeInfo | None]:
  1097	    """Look up a target by fully-qualified name.
  1098
  1099	    The first item in the return tuple is a list of deferred nodes that
  1100	    needs to be reprocessed. If the target represents a TypeInfo corresponding
  1101	    to a protocol, return it as a second item in the return tuple, otherwise None.
  1102	    """
  1103
  1104	    def not_found() -> None:
  1105	        manager.log_fine_grained(f"Can't find matching target for {target} (stale dependency?)")
  1106
  1107	    modules = manager.modules
  1108	    items = split_target(modules, target)
  1109	    if items is None:
  1110	        not_found()  # Stale dependency
  1111	        return [], None
  1112	    module, rest = items
  1113	    if rest:
  1114	        components = rest.split(".")
  1115	    else:
  1116	        components = []
  1117	    node: SymbolNode | None = modules[module]
  1118	    file: MypyFile | None = None
  1119	    active_class = None
  1120	    for c in components:
  1121	        if isinstance(node, TypeInfo):
  1122	            active_class = node
  1123	        if isinstance(node, MypyFile):
  1124	            file = node
  1125	        if not isinstance(node, (MypyFile, TypeInfo)) or c not in node.names:
  1126	            not_found()  # Stale dependency
  1127	            return [], None
  1128	        # Don't reprocess plugin generated targets. They should get
  1129	        # stripped and regenerated when the containing target is
  1130	        # reprocessed.
  1131	        if node.names[c].plugin_generated:
  1132	            return [], None
  1133	        node = node.names[c].node
  1134	    if isinstance(node, TypeInfo):
  1135	        # A ClassDef target covers the body of the class and everything defined
  1136	        # within it.  To get the body we include the entire surrounding target,
  1137	        # typically a module top-level, since we don't support processing class
  1138	        # bodies as separate entities for simplicity.
  1139	        assert file is not None
  1140	        if node.fullname != target:
  1141	            # This is a reference to a different TypeInfo, likely due to a stale dependency.
  1142	            # Processing them would spell trouble -- for example, we could be refreshing
  1143	            # a deserialized TypeInfo with missing attributes.
  1144	            not_found()
  1145	            return [], None
  1146	        result = [FineGrainedDeferredNode(file, None)]
  1147	        stale_info: TypeInfo | None = None
  1148	        if node.is_protocol:
  1149	            stale_info = node
  1150	        for name, symnode in node.names.items():
  1151	            node = symnode.node
  1152	            if isinstance(node, FuncDef):
  1153	                method, _ = lookup_target(manager, target + "." + name)
  1154	                result.extend(method)
  1155	        return result, stale_info
  1156	    if isinstance(node, Decorator):
  1157	        # Decorator targets actually refer to the function definition only.
  1158	        node = node.func
  1159	    if not isinstance(node, (FuncDef, MypyFile, OverloadedFuncDef)):
  1160	        # The target can't be refreshed. It's possible that the target was
  1161	        # changed to another type and we have a stale dependency pointing to it.
  1162	        not_found()
  1163	        return [], None
  1164	    if node.fullname != target:
  1165	        # Stale reference points to something unexpected. We shouldn't process since the
  1166	        # context will be wrong and it could be a partially initialized deserialized node.
  1167	        not_found()
  1168	        return [], None
  1169	    return [FineGrainedDeferredNode(node, active_class)], None

  1209	    module: str,

  1211	    deps: dict[str, set[str]],
  1212	    graph: Graph,
  1213	    fscache: FileSystemCache,
  1214	    refresh_file: Callable[[str, str], list[str]],
  1215	) -> list[str] | None:

  1217
  1218	    If a submodule a.b gets added, we need to mark it as suppressed
  1219	    in modules that contain "from a import b". Previously we assumed
  1220	    that 'a.b' is not a module but a regular name.

  1223
  1224	    Args:
  1225	        module: target package in which to look for submodules
  1226	        path: path of the module

  1228
  1229	    Return a list of errors from refresh_file() if it was called. If the
  1230	    return value is None, we didn't call refresh_file().
  1231	    """
  1232	    messages = None
  1233	    if path is None or not path.endswith(INIT_SUFFIXES):
  1234	        # Only packages have submodules.
  1235	        return None
  1236	    # Find any submodules present in the directory.
  1237	    pkgdir = os.path.dirname(path)
  1238	    try:
  1239	        entries = fscache.listdir(pkgdir)
  1240	    except FileNotFoundError:
  1241	        entries = []
  1242	    for fnam in entries:
  1243	        if (
  1244	            not fnam.endswith((".py", ".pyi"))
  1245	            or fnam.startswith("__init__.")
  1246	            or fnam.count(".") != 1
  1247	        ):
  1248	            continue
  1249	        shortname = fnam.split(".")[0]
  1250	        submodule = module + "." + shortname
  1251	        trigger = make_trigger(submodule)
  1252
  1253	        # We may be missing the required fine-grained deps.
  1254	        ensure_deps_loaded(module, deps, graph)
  1255
  1256	        if trigger in deps:
  1257	            for dep in deps[trigger]:
  1258	                # We can ignore <...> deps since a submodule can't trigger any.
  1259	                state = graph.get(dep)
  1260	                if not state:
  1261	                    # Maybe it's a non-top-level target. We only care about the module.
  1262	                    dep_module = module_prefix(graph, dep)
  1263	                    if dep_module is not None:
  1264	                        state = graph.get(dep_module)
  1265	                if state:
  1266	                    # Is the file may missing an AST in case it's read from cache?
  1267	                    if state.tree is None:
  1268	                        # Create AST for the file. This may produce some new errors
  1269	                        # that we need to propagate.
  1270	                        assert state.path is not None
  1271	                        messages = refresh_file(state.id, state.path)
  1272	                    tree = state.tree
  1273	                    assert tree  # Will be fine, due to refresh_file() above
  1274	                    for imp in tree.imports:
  1275	                        if isinstance(imp, ImportFrom):

  1304	    arbitrary order.  Preserve the order of files to avoid messages
  1305	    getting reshuffled continuously.  If there are messages in

  1309	    n = 0

  1311	    for msg in prev_messages:

  1313	        if fnam and fnam not in order:

  1318	    groups = []
  1319	    i = 0
  1320	    while i < len(messages):
  1321	        msg = messages[i]
  1322	        maybe_fnam = extract_possible_fnam_from_message(msg)
  1323	        group = [msg]
  1324	        if maybe_fnam in order:
  1325	            # This looks like a file name. Collect all lines related to this message.
  1326	            while (
  1327	                i + 1 < len(messages)
  1328	                and extract_possible_fnam_from_message(messages[i + 1]) not in order
  1329	                and extract_fnam_from_message(messages[i + 1]) is None
  1330	                and not messages[i + 1].startswith("mypy: ")
  1331	            ):
  1332	                i += 1
  1333	                group.append(messages[i])
  1334	        groups.append((order.get(maybe_fnam, n), group))
  1335	        i += 1
  1336
  1337	    groups = sorted(groups, key=lambda g: g[0])
  1338	    result = []
  1339	    for key, group in groups:
  1340	        result.extend(group)
