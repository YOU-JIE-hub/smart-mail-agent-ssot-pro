   433	    def snapshot(self) -> dict[str, object]:
   434	        """Produce a comparable snapshot of this Option"""
   435	        # Under mypyc, we don't have a __dict__, so we need to do worse things.
   436	        d = dict(getattr(self, "__dict__", ()))
   437	        for k in get_class_descriptors(Options):
   438	            if hasattr(self, k):
   439	                d[k] = getattr(self, k)
   440	        # Remove private attributes from snapshot
   441	        d = {k: v for k, v in d.items() if not k.startswith("_")}
   442	        return d

   447	    def process_error_codes(self, *, error_callback: Callable[[str], Any]) -> None:
   448	        # Process `--enable-error-code` and `--disable-error-code` flags
   449	        disabled_codes = set(self.disable_error_code)
   450	        enabled_codes = set(self.enable_error_code)
   451
   452	        valid_error_codes = set(error_codes.keys())
   453
   454	        invalid_codes = (enabled_codes | disabled_codes) - valid_error_codes
   455	        if invalid_codes:
   456	            error_callback(f"Invalid error code(s): {', '.join(sorted(invalid_codes))}")
   457
   458	        self.disabled_error_codes |= {error_codes[code] for code in disabled_codes}
   459	        self.enabled_error_codes |= {error_codes[code] for code in enabled_codes}
   460
   461	        # Enabling an error code always overrides disabling
   462	        self.disabled_error_codes -= self.enabled_error_codes

   464	    def process_incomplete_features(

   466	    ) -> None:

   484	    def apply_changes(self, changes: dict[str, object]) -> Options:
   485	        # Note: effects of this method *must* be idempotent.
   486	        new_options = Options()
   487	        # Under mypyc, we don't have a __dict__, so we need to do worse things.
   488	        replace_object_state(new_options, self, copy_dict=True)
   489	        for key, value in changes.items():
   490	            setattr(new_options, key, value)
   491	        if changes.get("ignore_missing_imports"):
   492	            # This is the only option for which a per-module and a global
   493	            # option sometimes beheave differently.
   494	            new_options.ignore_missing_imports_per_module = True
   495
   496	        # These two act as overrides, so apply them when cloning.
   497	        # Similar to global codes enabling overrides disabling, so we start from latter.
   498	        new_options.disabled_error_codes = self.disabled_error_codes.copy()
   499	        new_options.enabled_error_codes = self.enabled_error_codes.copy()
   500	        for code_str in new_options.disable_error_code:
   501	            code = error_codes[code_str]
   502	            new_options.disabled_error_codes.add(code)
   503	            new_options.enabled_error_codes.discard(code)
   504	        for code_str in new_options.enable_error_code:
   505	            code = error_codes[code_str]
   506	            new_options.enabled_error_codes.add(code)
   507	            new_options.disabled_error_codes.discard(code)
   508
   509	        return new_options

   511	    def compare_stable(self, other_snapshot: dict[str, object]) -> bool:
   512	        """Compare options in a way that is stable for snapshot() -> apply_changes() roundtrip.
   513
   514	        This is needed because apply_changes() has non-trivial effects for some flags, so
   515	        Options().apply_changes(options.snapshot()) may result in a (slightly) different object.
   516	        """
   517	        return (
   518	            Options().apply_changes(self.snapshot()).snapshot()
   519	            == Options().apply_changes(other_snapshot).snapshot()
   520	        )
