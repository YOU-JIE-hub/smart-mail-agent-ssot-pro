   754	        # Parse a terraform heredoc
   755	        # match: 1 = <<[-]?, 2 = name 3 = rest of line

   759	        yield match.start(2), String.Delimiter, match.group(2)  # heredoc name

   780	                ctx.pos = match.end()
   781	                break

   783	                lines.append(match)
   784	        else:
