    52	    """determine if specified year is leap year, returns Python boolean"""
    53	    if year < 1600:
    54	        if year % 4:
    55	            return 0
    56	        else:

    71	class NormalDate:
    72	    """
    73	    NormalDate is a specialized class to handle dates without
    74	    all the excess baggage (time zones, daylight savings, leap
    75	    seconds, etc.) of other date structures.  The minimalist
    76	    strategy greatly simplifies its implementation and use.
    77
    78	    Internally, NormalDate is stored as an integer with values
    79	    in a discontinuous range of -99990101 to 99991231.  The
    80	    integer value is used principally for storage and to simplify
    81	    the user interface.  Internal calculations are performed by
    82	    a scalar based on Jan 1, 1900.
    83
    84	    Valid NormalDate ranges include (-9999,1,1) B.C.E. through
    85	    (9999,12,31) C.E./A.D.

   292	    def formatMS(self,fmt):
   293	        '''format like MS date using the notation
   294	        {YY}    --> 2 digit year
   295	        {YYYY}  --> 4 digit year
   296	        {M}     --> month as digit
   297	        {MM}    --> 2 digit month
   298	        {MMM}   --> abbreviated month name
   299	        {MMMM}  --> monthname
   300	        {MMMMM} --> first character of monthname
   301	        {D}     --> day of month as digit
   302	        {DD}    --> 2 digit day of month
   303	        {DDD}   --> abrreviated weekday name
   304	        {DDDD}  --> weekday name
   305	        '''
   306	        r = fmt[:]
   307	        f = 0
   308	        while 1:
   309	            m = _fmtPat.search(r,f)
   310	            if m:
   311	                y = getattr(self,'_fmt'+(m.group()[1:-1].upper()))()
   312	                i, j = m.span()
   313	                r = (r[0:i] + y) + r[j:]
   314	                f = i + len(y)
   315	            else:
   316	                return r

   335	    def _isValidNormalDate(self, normalDate):
   336	        """checks for date validity in [-]yyyymmdd format"""
   337	        if not isinstance(normalDate,int):
   338	            return 0
   339	        if len(repr(normalDate)) > 9:
   340	            return 0
   341	        if normalDate < 0:
   342	            dateStr = "%09d" % normalDate
   343	        else:
   344	            dateStr = "%08d" % normalDate
   345	        if len(dateStr) < 8:
   346	            return 0
   347	        elif len(dateStr) == 9:
   348	            if (dateStr[0] != '-' and dateStr[0] != '+'):
   349	                return 0
   350	        year = int(dateStr[:-4])
   351	        if year < -9999 or year > 9999 or year == 0:
   352	            return 0    # note: zero (0) is not a valid year
   353	        month = int(dateStr[-4:-2])
   354	        if month < 1 or month > 12:
   355	            return 0
   356	        if isLeapYear(year):
   357	            maxDay = _daysInMonthLeapYear[month - 1]
   358	        else:
   359	            maxDay = _daysInMonthNormal[month - 1]
   360	        day = int(dateStr[-2:])
   361	        if day < 1 or day > maxDay:
   362	            return 0
   363	        if year == 1582 and month == 10 and day > 4 and day < 15:
   364	            return 0  # special case of 10 days dropped: Oct 5-14, 1582
   365	        return 1

   394	    def normalize(self, scalar):
   395	        """convert scalar to normalDate"""
   396	        if scalar < _bigBangScalar:
   397	            msg = "normalize(%d): scalar below minimum" % \
   398	                  _bigBangScalar
   399	            raise NormalDateException(msg)
   400	        if scalar > _bigCrunchScalar:
   401	            msg = "normalize(%d): scalar exceeds maximum" % \
   402	                  _bigCrunchScalar
   403	            raise NormalDateException(msg)
   404	        from math import floor
   405	        if scalar >= -115860:
   406	            year = 1600 + int(floor((scalar + 109573) / 365.2425))
   407	        elif scalar >= -693597:
   408	            year = 4 + int(floor((scalar + 692502) / 365.2425))
   409	        else:
   410	            year = -4 + int(floor((scalar + 695058) / 365.2425))
   411	        days = scalar - firstDayOfYear(year) + 1
   412	        if days <= 0:
   413	            year = year - 1
   414	            days = scalar - firstDayOfYear(year) + 1
   415	        daysInYear = 365
   416	        if isLeapYear(year):
   417	            daysInYear = daysInYear + 1
   418	        if days > daysInYear:
   419	            year = year + 1
   420	            days = scalar - firstDayOfYear(year) + 1
   421	        # add 10 days if between Oct 15, 1582 and Dec 31, 1582
   422	        if (scalar >= -115860 and scalar <= -115783):
   423	            days = days + 10
   424	        if isLeapYear(year):
   425	            daysByMonth = _daysInMonthLeapYear
   426	        else:
   427	            daysByMonth = _daysInMonthNormal
   428	        dc = 0; month = 12
   429	        for m in range(len(daysByMonth)):
   430	            dc = dc + daysByMonth[m]
   431	            if dc >= days:
   432	                month = m + 1
   433	                break
   434	        # add up the days in prior months
   435	        priorMonthDays = 0
   436	        for m in range(month - 1):
   437	            priorMonthDays = priorMonthDays + daysByMonth[m]
   438	        day = days - priorMonthDays
   439	        self.setNormalDate((year, month, day))

   461	    def scalar(self):
   462	        """days since baseline date: Jan 1, 1900"""
   463	        (year, month, day) = self.toTuple()
   464	        days = firstDayOfYear(year) + day - 1
   465	        if self.isLeapYear():
   466	            for m in range(month - 1):
   467	                days = days + _daysInMonthLeapYear[m]
   468	        else:
   469	            for m in range(month - 1):
   470	                days = days + _daysInMonthNormal[m]
   471	        if year == 1582:
   472	            if month > 10 or (month == 10 and day > 4):
   473	                days = days - 10
   474	        return days
