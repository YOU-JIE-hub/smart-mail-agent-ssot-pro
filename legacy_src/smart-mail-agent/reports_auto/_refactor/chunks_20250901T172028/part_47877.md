  1304	                                message_registry.INVALID_RETURN_TYPE_FOR_ASYNC_GENERATOR, typ

  1329	                self.return_types.append(typ.ret_type)

  1331	                with self.scope.push_function(defn):

  1338	                    # For methods and classmethods, we want the second parameter
  1339	                    if ref_type is not None and (not defn.is_static or defn.name == "__new__"):
  1340	                        arg_index = 1

  1342	                        typ.type_is, typ.arg_types[arg_index]
  1343	                    ):

  1345	                            message_registry.NARROWED_TYPE_NOT_SUBTYPE.format(

  1347	                                format_type(typ.arg_types[arg_index], self.options),
  1348	                            ),
  1349	                            item,
  1350	                        )

  1353	                for i in range(len(typ.arg_types)):
  1354	                    arg_type = typ.arg_types[i]
  1355	                    if (
  1356	                        isinstance(defn, FuncDef)
  1357	                        and ref_type is not None
  1358	                        and i == 0
  1359	                        and (not defn.is_static or defn.name == "__new__")
  1360	                        and typ.arg_kinds[0] not in [nodes.ARG_STAR, nodes.ARG_STAR2]
  1361	                    ):

  1363	                            ref_type = mypy.types.TypeType.make_normalized(ref_type)
  1364	                        if not is_same_type(arg_type, ref_type):
  1365	                            # This level of erasure matches the one in checkmember.check_self_arg(),
  1366	                            # better keep these two checks consistent.
  1367	                            erased = get_proper_type(erase_typevars(erase_to_bound(arg_type)))
  1368	                            if not is_subtype(ref_type, erased, ignore_type_params=True):
  1369	                                if (
  1370	                                    isinstance(erased, Instance)
  1371	                                    and erased.type.is_protocol
  1372	                                    or isinstance(erased, TypeType)
  1373	                                    and isinstance(erased.item, Instance)
  1374	                                    and erased.item.type.is_protocol
  1375	                                ):
  1376	                                    # We allow the explicit self-type to be not a supertype of
  1377	                                    # the current class if it is a protocol. For such cases
  1378	                                    # the consistency check will be performed at call sites.
  1379	                                    msg = None
  1380	                                elif typ.arg_names[i] in {"self", "cls"}:
  1381	                                    msg = message_registry.ERASED_SELF_TYPE_NOT_SUPERTYPE.format(
  1382	                                        erased.str_with_options(self.options),
  1383	                                        ref_type.str_with_options(self.options),
  1384	                                    )
  1385	                                else:
  1386	                                    msg = message_registry.MISSING_OR_INVALID_SELF_TYPE
  1387	                                if msg:
  1388	                                    self.fail(msg, defn)
  1389	                    elif isinstance(arg_type, TypeVarType):
  1390	                        # Refuse covariant parameter type variables
  1391	                        # TODO: check recursively for inner type variables
  1392	                        if (
  1393	                            arg_type.variance == COVARIANT
  1394	                            and defn.name not in ("__init__", "__new__", "__post_init__")
  1395	                            and not is_private(defn.name)  # private methods are not inherited
  1396	                        ):

  1399	                                ctx = typ

  1402	                    store_argument_type(item, i, typ, self.named_generic_type)

  1421	                            if new_frame is None:

  1427	                    # Add formal argument types to the binder.

  1430	                        v = arg.variable
  1431	                        if v.type is not None:
  1432	                            n = NameExpr(v.name)
  1433	                            n.node = v
  1434	                            self.binder.assign_type(n, v.type, v.type)

  1437	                    # We suppress reachability warnings for empty generator functions
  1438	                    # (return; yield) which have a "yield" that's unreachable by definition
  1439	                    # since it's only there to promote the function into a generator function.
  1440	                    #

  1443	                    # marked as being suppressed in *all* of the expansions, but we currently
  1444	                    # have no good way of doing this.
  1445	                    #

  1447	                    if _is_empty_generator_function(item) or len(expanded) >= 2:
  1448	                        self.binder.suppress_unreachable_warnings()

  1450	                unreachable = self.binder.is_unreachable()
  1451	                if new_frame is not None:
  1452	                    self.binder.pop_frame(True, 0)
  1453
  1454	            if not unreachable:
  1455	                if defn.is_generator or is_named_instance(
  1456	                    self.return_types[-1], "typing.AwaitableGenerator"
  1457	                ):
  1458	                    return_type = self.get_generator_return_type(
  1459	                        self.return_types[-1], defn.is_coroutine
  1460	                    )
  1461	                elif defn.is_coroutine:
  1462	                    return_type = self.get_coroutine_return_type(self.return_types[-1])
  1463	                else:
  1464	                    return_type = self.return_types[-1]
  1465	                return_type = get_proper_type(return_type)
  1466
  1467	                allow_empty = allow_empty or self.options.allow_empty_bodies
  1468
  1469	                show_error = (
  1470	                    not body_is_trivial
  1471	                    or
  1472	                    # Allow empty bodies for abstract methods, overloads, in tests and stubs.
  1473	                    (
  1474	                        not allow_empty
  1475	                        and not (
  1476	                            isinstance(defn, FuncDef) and defn.abstract_status != NOT_ABSTRACT
  1477	                        )
  1478	                        and not self.is_stub
  1479	                    )
  1480	                )

  1482	                # Ignore plugin generated methods, these usually don't need any bodies.

  1484	                    defn.name not in defn.info.names or defn.info.names[defn.name].plugin_generated

  1490	                if isinstance(defn, FuncDef) and defn.is_mypy_only:
  1491	                    show_error = False

  1493	                # We want to minimize the fallout from checking empty bodies
  1494	                # that was absent in many mypy versions.
  1495	                if body_is_trivial and is_subtype(NoneType(), return_type):
  1496	                    show_error = False
  1497
  1498	                may_be_abstract = (
  1499	                    body_is_trivial
  1500	                    and defn.info is not FUNC_NO_INFO
  1501	                    and defn.info.metaclass_type is not None

  1545	    def is_var_redefined_in_outer_context(self, v: Var, after_line: int) -> bool:
  1546	        """Can the variable be assigned to at module top level or outer function?
  1547
  1548	        Note that this doesn't do a full CFG analysis but uses a line number based
  1549	        heuristic that isn't correct in some (rare) cases.
  1550	        """
  1551	        if v.is_final:
  1552	            # Final vars are definitely never reassigned.
  1553	            return False
  1554
  1555	        outers = self.tscope.outer_functions()
  1556	        if not outers:
  1557	            # Top-level function -- outer context is top level, and we can't reason about
  1558	            # globals
  1559	            return True
  1560	        for outer in outers:
  1561	            if isinstance(outer, FuncDef):
  1562	                if find_last_var_assignment_line(outer.body, v) >= after_line:
  1563	                    return True
  1564	        return False

  1567	        """Fails when the return typevar is not defined in arguments."""
  1568	        if isinstance(typ.ret_type, TypeVarType) and typ.ret_type in typ.variables:
  1569	            arg_type_visitor = CollectArgTypeVarTypes()

  1586	    def check_default_args(self, item: FuncItem, body_is_trivial: bool) -> None:
  1587	        for arg in item.arguments:
  1588	            if arg.initializer is None:
  1589	                continue
  1590	            if body_is_trivial and isinstance(arg.initializer, EllipsisExpr):
  1591	                continue

  1593	            msg = "Incompatible default for "
  1594	            if name.startswith("__tuple_arg_"):

  1597	                msg += f'argument "{name}"'
  1598	            if (
  1599	                not self.options.implicit_optional
  1600	                and isinstance(arg.initializer, NameExpr)
  1601	                and arg.initializer.fullname == "builtins.None"
