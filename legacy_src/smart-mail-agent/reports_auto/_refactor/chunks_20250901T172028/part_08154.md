   108	# Snapshot representation of a symbol table node or type. The representation is
   109	# opaque -- the only supported operations are comparing for equality and
   110	# hashing (latter for type snapshots only). Snapshots can contain primitive
   111	# objects, nested tuples, lists and dictionaries and primitive objects (type
   112	# snapshots are immutable).
   113	#
   114	# For example, the snapshot of the 'int' type is ('Instance', 'builtins.int', ()).
   115
   116	# Type snapshots are strict, they must be hashable and ordered (e.g. for Unions).
   117	Primitive: _TypeAlias = Union[str, float, int, bool]  # float is for Literal[3.14] support.
   118	SnapshotItem: _TypeAlias = tuple[Union[Primitive, "SnapshotItem"], ...]
   119
   120	# Symbol snapshots can be more lenient.
   121	SymbolSnapshot: _TypeAlias = tuple[object, ...]
   122

   124	def compare_symbol_table_snapshots(
   125	    name_prefix: str, snapshot1: dict[str, SymbolSnapshot], snapshot2: dict[str, SymbolSnapshot]
   126	) -> set[str]:
   127	    """Return names that are different in two snapshots of a symbol table.
   128
   129	    Only shallow (intra-module) differences are considered. References to things defined
   130	    outside the module are compared based on the name of the target only.
   131
   132	    Recurse into class symbol tables (if the class is defined in the target module).
   133
   134	    Return a set of fully-qualified names (e.g., 'mod.func' or 'mod.Class.method').
   135	    """
   136	    # Find names only defined only in one version.
   137	    names1 = {f"{name_prefix}.{name}" for name in snapshot1}
   138	    names2 = {f"{name_prefix}.{name}" for name in snapshot2}
   139	    triggers = names1 ^ names2
   140
   141	    # Look for names defined in both versions that are different.
   142	    for name in set(snapshot1.keys()) & set(snapshot2.keys()):
   143	        item1 = snapshot1[name]
   144	        item2 = snapshot2[name]
   145	        kind1 = item1[0]
   146	        kind2 = item2[0]
   147	        item_name = f"{name_prefix}.{name}"
   148	        if kind1 != kind2:
   149	            # Different kind of node in two snapshots -> trivially different.
   150	            triggers.add(item_name)
   151	        elif kind1 == "TypeInfo":
   152	            if item1[:-1] != item2[:-1]:
   153	                # Record major difference (outside class symbol tables).
   154	                triggers.add(item_name)
   155	            # Look for differences in nested class symbol table entries.
   156	            assert isinstance(item1[-1], dict)
   157	            assert isinstance(item2[-1], dict)
   158	            triggers |= compare_symbol_table_snapshots(item_name, item1[-1], item2[-1])
   159	        else:
   160	            # Shallow node (no interesting internal structure). Just use equality.
   161	            if snapshot1[name] != snapshot2[name]:
   162	                triggers.add(item_name)
   163
   164	    return triggers

   167	def snapshot_symbol_table(name_prefix: str, table: SymbolTable) -> dict[str, SymbolSnapshot]:
   168	    """Create a snapshot description that represents the state of a symbol table.
   169
   170	    The snapshot has a representation based on nested tuples and dicts
   171	    that makes it easy and fast to find differences.
   172
   173	    Only "shallow" state is included in the snapshot -- references to
   174	    things defined in other modules are represented just by the names of
   175	    the targets.
   176	    """
   177	    result: dict[str, SymbolSnapshot] = {}
   178	    for name, symbol in table.items():
   179	        node = symbol.node
   180	        # TODO: cross_ref?
   181	        fullname = node.fullname if node else None
   182	        common = (fullname, symbol.kind, symbol.module_public)
   183	        if isinstance(node, MypyFile):
   184	            # This is a cross-reference to another module.
   185	            # If the reference is busted because the other module is missing,
   186	            # the node will be a "stale_info" TypeInfo produced by fixup,
   187	            # but that doesn't really matter to us here.
   188	            result[name] = ("Moduleref", common)
   189	        elif isinstance(node, TypeVarExpr):
   190	            result[name] = (
   191	                "TypeVar",
   192	                node.variance,
   193	                [snapshot_type(value) for value in node.values],
   194	                snapshot_type(node.upper_bound),
   195	                snapshot_type(node.default),
   196	            )
   197	        elif isinstance(node, TypeAlias):
   198	            result[name] = (
   199	                "TypeAlias",
   200	                snapshot_types(node.alias_tvars),
   201	                node.normalized,
   202	                node.no_args,
   203	                snapshot_optional_type(node.target),
   204	            )
   205	        elif isinstance(node, ParamSpecExpr):
   206	            result[name] = (
   207	                "ParamSpec",
   208	                node.variance,
   209	                snapshot_type(node.upper_bound),
   210	                snapshot_type(node.default),
   211	            )
   212	        elif isinstance(node, TypeVarTupleExpr):
   213	            result[name] = (
   214	                "TypeVarTuple",
   215	                node.variance,
   216	                snapshot_type(node.upper_bound),
   217	                snapshot_type(node.default),
   218	            )
   219	        else:
   220	            assert symbol.kind != UNBOUND_IMPORTED
   221	            if node and get_prefix(node.fullname) != name_prefix:
   222	                # This is a cross-reference to a node defined in another module.
   223	                # Include the node kind (FuncDef, Decorator, TypeInfo, ...), so that we will
   224	                # reprocess when a *new* node is created instead of merging an existing one.
   225	                result[name] = ("CrossRef", common, type(node).__name__)
   226	            else:
   227	                result[name] = snapshot_definition(node, common)
   228	    return result

   231	def snapshot_definition(node: SymbolNode | None, common: SymbolSnapshot) -> SymbolSnapshot:
   232	    """Create a snapshot description of a symbol table node.
   233
   234	    The representation is nested tuples and dicts. Only externally
   235	    visible attributes are included.
   236	    """
   237	    if isinstance(node, SYMBOL_FUNCBASE_TYPES):
   238	        # TODO: info
   239	        if node.type:
   240	            signature: tuple[object, ...] = snapshot_type(node.type)
   241	        else:
   242	            signature = snapshot_untyped_signature(node)
   243	        impl: FuncDef | None = None
   244	        if isinstance(node, FuncDef):
   245	            impl = node
   246	        elif node.impl:
   247	            impl = node.impl.func if isinstance(node.impl, Decorator) else node.impl
   248	        setter_type = None
   249	        if isinstance(node, OverloadedFuncDef) and node.items:
   250	            first_item = node.items[0]
   251	            if isinstance(first_item, Decorator) and first_item.func.is_property:
   252	                setter_type = snapshot_optional_type(first_item.var.setter_type)
   253	        is_trivial_body = impl.is_trivial_body if impl else False
   254	        dataclass_transform_spec = find_dataclass_transform_spec(node)
   255	        return (
   256	            "Func",
   257	            common,
   258	            node.is_property,
   259	            node.is_final,
   260	            node.is_class,
   261	            node.is_static,
   262	            signature,
   263	            is_trivial_body,
   264	            dataclass_transform_spec.serialize() if dataclass_transform_spec is not None else None,
   265	            node.deprecated if isinstance(node, FuncDef) else None,
   266	            setter_type,  # multi-part properties are stored as OverloadedFuncDef
   267	        )
   268	    elif isinstance(node, Var):
   269	        return ("Var", common, snapshot_optional_type(node.type), node.is_final)
   270	    elif isinstance(node, Decorator):
   271	        # Note that decorated methods are represented by Decorator instances in
   272	        # a symbol table since we need to preserve information about the
   273	        # decorated function (whether it's a class function, for
   274	        # example). Top-level decorated functions, however, are represented by
   275	        # the corresponding Var node, since that happens to provide enough
   276	        # context.
   277	        return (
   278	            "Decorator",
   279	            node.is_overload,
   280	            snapshot_optional_type(node.var.type),
   281	            snapshot_definition(node.func, common),
   282	        )
   283	    elif isinstance(node, TypeInfo):
   284	        dataclass_transform_spec = node.dataclass_transform_spec
   285	        if dataclass_transform_spec is None:
   286	            dataclass_transform_spec = find_dataclass_transform_spec(node)
   287
   288	        attrs = (
   289	            node.is_abstract,
   290	            node.is_enum,
   291	            node.is_protocol,
   292	            node.fallback_to_any,
   293	            node.meta_fallback_to_any,
   294	            node.is_named_tuple,
   295	            node.is_newtype,
   296	            # We need this to e.g. trigger metaclass calculation in subclasses.
   297	            snapshot_optional_type(node.metaclass_type),
   298	            snapshot_optional_type(node.tuple_type),
   299	            snapshot_optional_type(node.typeddict_type),
   300	            [base.fullname for base in node.mro],
   301	            # Note that the structure of type variables is a part of the external interface,
   302	            # since creating instances might fail, for example:
   303	            #     T = TypeVar('T', bound=int)
   304	            #     class C(Generic[T]):
   305	            #         ...
   306	            #     x: C[str] <- this is invalid, and needs to be re-checked if `T` changes.
   307	            # An alternative would be to create both deps: <...> -> C, and <...> -> <C>,
   308	            # but this currently seems a bit ad hoc.
   309	            tuple(snapshot_type(tdef) for tdef in node.defn.type_vars),
   310	            [snapshot_type(base) for base in node.bases],
   311	            [snapshot_type(p) for p in node._promote],
   312	            dataclass_transform_spec.serialize() if dataclass_transform_spec is not None else None,
   313	            node.deprecated,
   314	        )
   315	        prefix = node.fullname
   316	        symbol_table = snapshot_symbol_table(prefix, node.names)
   317	        # Special dependency for abstract attribute handling.
   318	        symbol_table["(abstract)"] = ("Abstract", tuple(sorted(node.abstract_attributes)))
   319	        return ("TypeInfo", common, attrs, symbol_table)
   320	    else:
   321	        # Other node types are handled elsewhere.
   322	        assert False, type(node)

   326	    """Create a snapshot representation of a type using nested tuples."""
   327	    return typ.accept(SnapshotTypeVisitor())

   330	def snapshot_optional_type(typ: Type | None) -> SnapshotItem:
   331	    if typ:
   332	        return snapshot_type(typ)
   333	    else:
   334	        return ("<not set>",)

   352	class SnapshotTypeVisitor(TypeVisitor[SnapshotItem]):
   353	    """Creates a read-only, self-contained snapshot of a type object.
   354
   355	    Properties of a snapshot:
   356
   357	    - Contains (nested) tuples and other immutable primitive objects only.
   358	    - References to AST nodes are replaced with full names of targets.
   359	    - Has no references to mutable or non-primitive objects.
   360	    - Two snapshots represent the same object if and only if they are
   361	      equal.
   362	    - Results must be sortable. It's important that tuples have
   363	      consistent types and can't arbitrarily mix str and None values,
   364	      for example, since they can't be compared.
   365	    """
   366
   367	    def visit_unbound_type(self, typ: UnboundType) -> SnapshotItem:
   368	        return (
   369	            "UnboundType",
   370	            typ.name,
   371	            typ.optional,
   372	            typ.empty_tuple_index,
   373	            snapshot_types(typ.args),
   374	        )
   375
   376	    def visit_any(self, typ: AnyType) -> SnapshotItem:
   377	        return snapshot_simple_type(typ)
   378
   379	    def visit_none_type(self, typ: NoneType) -> SnapshotItem:
   380	        return snapshot_simple_type(typ)
   381
   382	    def visit_uninhabited_type(self, typ: UninhabitedType) -> SnapshotItem:
   383	        return snapshot_simple_type(typ)
   384
   385	    def visit_erased_type(self, typ: ErasedType) -> SnapshotItem:
   386	        return snapshot_simple_type(typ)
   387
   388	    def visit_deleted_type(self, typ: DeletedType) -> SnapshotItem:
   389	        return snapshot_simple_type(typ)
   390
   391	    def visit_instance(self, typ: Instance) -> SnapshotItem:
   392	        extra_attrs: SnapshotItem
   393	        if typ.extra_attrs:
   394	            extra_attrs = (
   395	                tuple(sorted((k, v.accept(self)) for k, v in typ.extra_attrs.attrs.items())),
   396	                tuple(typ.extra_attrs.immutable),
   397	            )
   398	        else:
   399	            extra_attrs = ()
   400	        return (
   401	            "Instance",
   402	            encode_optional_str(typ.type.fullname),
   403	            snapshot_types(typ.args),
   404	            ("None",) if typ.last_known_value is None else snapshot_type(typ.last_known_value),
   405	            extra_attrs,
   406	        )
   407
   408	    def visit_type_var(self, typ: TypeVarType) -> SnapshotItem:
   409	        return (
   410	            "TypeVar",
   411	            typ.name,
   412	            typ.fullname,
   413	            typ.id.raw_id,
   414	            typ.id.meta_level,
   415	            snapshot_types(typ.values),
   416	            snapshot_type(typ.upper_bound),
   417	            snapshot_type(typ.default),
   418	            typ.variance,
   419	        )
   420
   421	    def visit_param_spec(self, typ: ParamSpecType) -> SnapshotItem:
   422	        return (
   423	            "ParamSpec",
   424	            typ.id.raw_id,
   425	            typ.id.meta_level,
   426	            typ.flavor,
   427	            snapshot_type(typ.upper_bound),
   428	            snapshot_type(typ.default),
   429	        )
   430
   431	    def visit_type_var_tuple(self, typ: TypeVarTupleType) -> SnapshotItem:
   432	        return (
   433	            "TypeVarTupleType",
   434	            typ.id.raw_id,
   435	            typ.id.meta_level,
   436	            snapshot_type(typ.upper_bound),
   437	            snapshot_type(typ.default),
   438	        )
   439
   440	    def visit_unpack_type(self, typ: UnpackType) -> SnapshotItem:
   441	        return ("UnpackType", snapshot_type(typ.type))
   442
   443	    def visit_parameters(self, typ: Parameters) -> SnapshotItem:
   444	        return (
   445	            "Parameters",
   446	            snapshot_types(typ.arg_types),
   447	            tuple(encode_optional_str(name) for name in typ.arg_names),
   448	            tuple(k.value for k in typ.arg_kinds),
   449	        )
   450
   451	    def visit_callable_type(self, typ: CallableType) -> SnapshotItem:
   452	        if typ.is_generic():
   453	            typ = self.normalize_callable_variables(typ)
   454	        return (
   455	            "CallableType",
   456	            snapshot_types(typ.arg_types),
   457	            snapshot_type(typ.ret_type),
   458	            tuple(encode_optional_str(name) for name in typ.arg_names),
   459	            tuple(k.value for k in typ.arg_kinds),
   460	            typ.is_type_obj(),
   461	            typ.is_ellipsis_args,
   462	            snapshot_types(typ.variables),
   463	            typ.is_bound,
   464	        )
   465
   466	    def normalize_callable_variables(self, typ: CallableType) -> CallableType:
   467	        """Normalize all type variable ids to run from -1 to -len(variables)."""
   468	        tvs = []
   469	        tvmap: dict[TypeVarId, Type] = {}
   470	        for i, v in enumerate(typ.variables):
   471	            tid = TypeVarId(-1 - i)
   472	            if isinstance(v, TypeVarType):
   473	                tv: TypeVarLikeType = v.copy_modified(id=tid)
   474	            elif isinstance(v, TypeVarTupleType):
   475	                tv = v.copy_modified(id=tid)
   476	            else:
   477	                assert isinstance(v, ParamSpecType)
   478	                tv = v.copy_modified(id=tid)
   479	            tvs.append(tv)
   480	            tvmap[v.id] = tv
   481	        with state.strict_optional_set(True):
   482	            return expand_type(typ, tvmap).copy_modified(variables=tvs)
   483
   484	    def visit_tuple_type(self, typ: TupleType) -> SnapshotItem:
   485	        return ("TupleType", snapshot_types(typ.items))
   486
   487	    def visit_typeddict_type(self, typ: TypedDictType) -> SnapshotItem:
   488	        items = tuple((key, snapshot_type(item_type)) for key, item_type in typ.items.items())
   489	        required = tuple(sorted(typ.required_keys))
   490	        readonly = tuple(sorted(typ.readonly_keys))
   491	        return ("TypedDictType", items, required, readonly)
   492
   493	    def visit_literal_type(self, typ: LiteralType) -> SnapshotItem:
   494	        return ("LiteralType", snapshot_type(typ.fallback), typ.value)
   495
   496	    def visit_union_type(self, typ: UnionType) -> SnapshotItem:
   497	        # Sort and remove duplicates so that we can use equality to test for
   498	        # equivalent union type snapshots.
   499	        items = {snapshot_type(item) for item in typ.items}
   500	        normalized = tuple(sorted(items))
   501	        return ("UnionType", normalized)
   502
   503	    def visit_overloaded(self, typ: Overloaded) -> SnapshotItem:
   504	        return ("Overloaded", snapshot_types(typ.items))
   505
   506	    def visit_partial_type(self, typ: PartialType) -> SnapshotItem:
   507	        # A partial type is not fully defined, so the result is indeterminate. We shouldn't
   508	        # get here.
   509	        raise RuntimeError
   510
   511	    def visit_type_type(self, typ: TypeType) -> SnapshotItem:
   512	        return ("TypeType", snapshot_type(typ.item))
   513
   514	    def visit_type_alias_type(self, typ: TypeAliasType) -> SnapshotItem:
   515	        assert typ.alias is not None
   516	        return ("TypeAliasType", typ.alias.fullname, snapshot_types(typ.args))

   519	def snapshot_untyped_signature(func: OverloadedFuncDef | FuncItem) -> SymbolSnapshot:
   520	    """Create a snapshot of the signature of a function that has no explicit signature.
   521
   522	    If the arguments to a function without signature change, it must be
   523	    considered as different. We have this special casing since we don't store
   524	    the implicit signature anywhere, and we'd rather not construct new
   525	    Callable objects in this module (the idea is to only read properties of
   526	    the AST here).
   527	    """
   528	    if isinstance(func, FuncItem):
   529	        return (tuple(func.arg_names), tuple(func.arg_kinds))
   530	    else:
   531	        result: list[SymbolSnapshot] = []
   532	        for item in func.items:
   533	            if isinstance(item
