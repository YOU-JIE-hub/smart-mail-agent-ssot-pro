   341	    def __init__(self):
   342	        "Initializes the generator."
   343	        self.tables = {}

   345	    def add(self, tag, data):
   346	        "Adds a table to the TTF file."
   347	        if tag == 'head':
   348	            data = splice(data, 8, b'\0\0\0\0')
   349	        self.tables[tag] = data
   350
   351	    def makeStream(self):
   352	        "Finishes the generation and returns the TTF file as a string"
   353	        stm = BytesIO()
   354	        write = stm.write
   355
   356	        tables = self.tables
   357	        numTables = len(tables)
   358	        searchRange = 1
   359	        entrySelector = 0
   360	        while searchRange * 2 <= numTables:
   361	            searchRange = searchRange * 2
   362	            entrySelector = entrySelector + 1
   363	        searchRange = searchRange * 16
   364	        rangeShift = numTables * 16 - searchRange
   365
   366	        # Header
   367	        write(pack(">lHHHH", 0x00010000, numTables, searchRange,
   368	                                 entrySelector, rangeShift))
   369
   370	        # Table directory
   371	        offset = 12 + numTables * 16
   372	        wStr = lambda x:write(bytes(tag,'latin1'))
   373	        tables_items = list(sorted(tables.items()))
   374	        for tag, data in tables_items:
   375	            if tag == 'head':
   376	                head_start = offset
   377	            checksum = calcChecksum(data)
   378	            wStr(tag)
   379	            write(pack(">LLL", checksum, offset, len(data)))
   380	            paddedLength = (len(data)+3)&~3
   381	            offset = offset + paddedLength
   382
   383	        # Table data
   384	        for tag, data in tables_items:
   385	            data += b"\0\0\0"
   386	            write(data[:len(data)&~3])
   387
   388	        checksum = calcChecksum(stm.getvalue())
   389	        checksum = add32(0xB1B0AFBA, -checksum)
   390	        stm.seek(head_start + 8)
   391	        write(pack('>L', checksum))
   392
   393	        return stm.getvalue()

   398	class TTFNameBytes(bytes):
   399	    '''class used to return named strings'''
   400	    def __new__(cls,b,enc='utf8'):
   401	        try:
   402	            ustr = b.decode(enc)
   403	        except:
   404	            ustr = b.decode('latin1')
   405	        self = bytes.__new__(cls,ustr.encode('utf8'))
   406	        self.ustr = ustr
   407	        return self

   409	class TTFontFile(TTFontParser):
   410	    "TTF file parser and generator"
   411	    _agfnc = 0
   412	    _agfnm = {}
   413
   414	    def __init__(self, file, charInfo=1, validate=0,subfontIndex=0):
   415	        """Loads and parses a TrueType font file.
   416
   417	        file can be a filename or a file object.  If validate is set to a false
   418	        values, skips checksum validation.  This can save time, especially if
   419	        the font is large.  See TTFontFile.extractInfo for more information.
   420	        """
   421	        if isStr(subfontIndex): #bytes or unicode
   422	            sfi = 0
   423	            __dict__ = self.__dict__.copy()
   424	            while True:
   425	                TTFontParser.__init__(self, file, validate=validate,subfontIndex=sfi)
   426	                numSubfonts = self.numSubfonts = self.read_ulong()
   427	                self.extractInfo(charInfo)
   428	                if (isBytes(subfontIndex) and subfontIndex==self.name
   429	                    or subfontIndex==self.name.ustr): #we found it
   430	                    return
   431	                if not sfi:
   432	                    __dict__.update(dict(_ttf_data=self._ttf_data, filename=self.filename))
   433	                sfi += 1
   434	                if sfi>=numSubfonts:
   435	                    raise ValueError('cannot find %r subfont %r' % (self.filename, subfontIndex))
   436	                self.__dict__.clear()
   437	                self.__dict__.update(__dict__)
   438	        else:
   439	            TTFontParser.__init__(self, file, validate=validate,subfontIndex=subfontIndex)
   440	            self.extractInfo(charInfo)

   442	    def extractInfo(self, charInfo=1):
   443	        """
   444	        Extract typographic information from the loaded font file.
   445
   446	        The following attributes will be set::
   447
   448	            name         PostScript font name
   449	            flags        Font flags
   450	            ascent       Typographic ascender in 1/1000ths of a point
   451	            descent      Typographic descender in 1/1000ths of a point
   452	            capHeight    Cap height in 1/1000ths of a point (0 if not available)
   453	            bbox         Glyph bounding box [l,t,r,b] in 1/1000ths of a point
   454	            _bbox        Glyph bounding box [l,t,r,b] in unitsPerEm
   455	            unitsPerEm   Glyph units per em
   456	            italicAngle  Italic angle in degrees ccw
   457	            stemV        stem weight in 1/1000ths of a point (approximate)
   458
   459	        If charInfo is true, the following will also be set::
   460
   461	            defaultWidth   default glyph width in 1/1000ths of a point
   462	            charWidths     dictionary of character widths for every supported UCS character
   463	                           code
   464
   465	        This will only work if the font has a Unicode cmap (platform 3,
   466	        encoding 1, format 4 or platform 0 any encoding format 4).  Setting
   467	        charInfo to false avoids this requirement
   468
   469	        """
   470	        # name - Naming table
   471	        name_offset = self.seek_table("name")
   472	        format = self.read_ushort()
   473	        if format != 0:
   474	            raise TTFError("Unknown name table format (%d)" % format)
   475	        numRecords = self.read_ushort()
   476	        string_data_offset = name_offset + self.read_ushort()
   477	        names = {1:None,2:None,3:None,4:None,6:None}
   478	        K = list(names.keys())
   479	        nameCount = len(names)
   480	        for i in range(numRecords):
   481	            platformId = self.read_ushort()
   482	            encodingId = self.read_ushort()
   483	            languageId = self.read_ushort()
   484	            nameId = self.read_ushort()
   485	            length = self.read_ushort()
   486	            offset = self.read_ushort()
   487	            if nameId not in K: continue
   488	            N = None
   489	            if platformId == 3 and encodingId == 1 and languageId == 0x409: # Microsoft, Unicode, US English, PS Name
   490	                opos = self._pos
   491	                try:
   492	                    self.seek(string_data_offset + offset)
   493	                    if length % 2 != 0:
   494	                        raise TTFError("PostScript name is UTF-16BE string of odd length")
   495	                    N = TTFNameBytes(self.get_chunk(string_data_offset + offset, length),'utf_16_be')
   496	                finally:
   497	                    self._pos = opos
   498	            elif platformId == 1 and encodingId == 0 and languageId == 0: # Macintosh, Roman, English, PS Name
   499	                # According to OpenType spec, if PS name exists, it must exist
   500	                # both in MS Unicode and Macintosh Roman formats.  Apparently,
   501	                # you can find live TTF fonts which only have Macintosh format.
   502	                N = TTFNameBytes(self.get_chunk(string_data_offset + offset, length),'mac_roman')
   503	            if N and names[nameId]==None:
   504	                names[nameId] = N
   505	                nameCount -= 1
   506	                if nameCount==0: break
   507	        if names[6] is not None:
   508	            psName = names[6]
   509	        elif names[4] is not None:
   510	            psName = names[4]
   511	        # Fine, one last try before we bail.
   512	        elif names[1] is not None:
   513	            psName = names[1]
   514	        else:
   515	            psName = None
   516
   517	        # Don't just assume, check for None since some shoddy fonts cause crashes here...
   518	        if not psName:
   519	            if rl_config.autoGenerateTTFMissingTTFName:
   520	                fn = self.filename
   521	                if fn:
   522	                    bfn = os.path.splitext(os.path.basename(fn))[0]
   523	                if not fn:
   524	                    psName = bytestr('_RL_%s_%s_TTF' % (time.time(), self.__class__._agfnc))
   525	                    self.__class__._agfnc += 1
   526	                else:
   527	                    psName = self._agfnm.get(fn,'')
   528	                    if not psName:
   529	                        if bfn:
   530	                            psName = bytestr('_RL_%s_TTF' % bfn)
   531	                        else:
   532	                            psName = bytestr('_RL_%s_%s_TTF' % (time.time(), self.__class__._agfnc))
   533	                            self.__class__._agfnc += 1
   534	                        self._agfnm[fn] = psName
   535	            else:
   536	                raise TTFError("Could not find PostScript font name")
   537
   538	        psName = psName.__class__(psName.replace(b" ", b"-"))  #Dinu Gherman's fix for font names with spaces
   539
   540	        for c in psName:
   541	            if char2int(c)>126 or c in b' [](){}<>/%':
   542	                raise TTFError("psName=%r contains invalid character %s" % (psName,ascii(c)))
   543	        self.name = psName
   544	        self.familyName = names[1] or psName
   545	        self.styleName = names[2] or 'Regular'
   546	        self.fullName = names[4] or psName
   547	        self.uniqueFontID = names[3] or psName
   548
   549	        # head - Font header table
   550	        try:
   551	            self.seek_table("head")
   552	        except:
   553	            raise TTFError('head table not found ttf name=%s' % self.name)
   554	        ver_maj, ver_min = self.read_ushort(), self.read_ushort()
   555	        if ver_maj != 1:
   556	            raise TTFError('Unknown head table version %d.%04x' % (ver_maj, ver_min))
   557	        self.fontRevision = self.read_ushort(), self.read_ushort()
   558
   559	        self.skip(4)
   560	        magic = self.read_ulong()
   561	        if magic != 0x5F0F3CF5:
   562	            raise TTFError('Invalid head table magic %04x' % magic)
   563	        self.skip(2)
   564	        self.unitsPerEm = unitsPerEm = self.read_ushort()
   565	        scale = lambda x, unitsPerEm=unitsPerEm: x * 1000. / unitsPerEm
   566	        self.skip(16)
   567	        xMin = self.read_short()
   568	        yMin = self.read_short()
   569	        xMax = self.read_short()
   570	        yMax = self.read_short()
   571	        self.bbox = list(map(scale, [xMin, yMin, xMax, yMax]))
   572	        self.skip(3*2)
   573	        indexToLocFormat = self.read_ushort()
   574	        glyphDataFormat = self.read_ushort()
   575
   576	        # OS/2 - OS/2 and Windows metrics table
   577	        # (needs data from head table)
   578	        subsettingAllowed = True
   579	        if "OS/2" in self.table:
   580	            self.seek_table("OS/2")
   581	            version = self.read_ushort()
   582	            self.skip(2)
   583	            usWeightClass = self.read_ushort()
   584	            self.skip(2)
   585	            fsType = self.read_ushort()
   586	            if fsType==0x0002 or (fsType & 0x0300):
   587	                subsettingAllowed = os.path.basename(self.filename) not in rl_config.allowTTFSubsetting
   588	            self.skip(58)   #11*2 + 10 + 4*4 + 4 + 3*2
   589	            sTypoAscender = self.read_short()
   590	            sTypoDescender = self.read_short()
   591	            self.ascent = scale(sTypoAscender)      # XXX: for some reason it needs to be multiplied by 1.24--1.28
   592	            self.descent = scale(sTypoDescender)
   593
   594	            if version > 1:
   595	                self.skip(16)   #3*2 + 2*4 + 2
   596	                sCapHeight = self.read_short()
   597	                self.capHeight = scale(sCapHeight)
   598	            else:
   599	                self.capHeight = self.ascent
   600	        else:
   601	            # Microsoft TTFs require an OS/2 table; Apple ones do not.  Try to
   602	            # cope.  The data is not very important anyway.
   603	            usWeightClass = 500
   604	            self.ascent = scale(yMax)
   605	            self.descent = scale(yMin)
   606	            self.capHeight = self.ascent
   607
   608	        # There's no way to get stemV from a TTF file short of analyzing actual outline data
   609	        # This fuzzy formula is taken from pdflib sources, but we could just use 0 here
   610	        self.stemV = 50 + int((usWeightClass / 65.0) ** 2)
   611
   612	        # post - PostScript table
   613	        # (needs data from OS/2 table)
   614	        self.seek_table("post")
   615	        ver_maj, ver_min = self.read_ushort(), self.read_ushort()
   616	        if ver_maj not in (1, 2, 3, 4):
   617	            # Adobe/MS documents 1, 2, 2.5, 3; Apple also has 4.
   618	            # From Apple docs it seems that we do not need to care
   619	            # about the exact version, so if you get this error, you can
   620	            # try to remove this check altogether.
   621	            raise TTFError('Unknown post table version %d.%04x' % (ver_maj, ver_min))
   622	        self.italicAngle = self.read_short() + self.read_ushort() / 65536.0
   623	        self.underlinePosition = self.read_short()
   624	        self.underlineThickness = self.read_short()
   625	        isFixedPitch = self.read_ulong()
   626
   627	        self.flags = FF_SYMBOLIC        # All fonts that contain characters
   628	                                        # outside the original Adobe character
   629	                                        # set are considered "symbolic".
   630	        if self.italicAngle!= 0:
   631	            self.flags = self.flags | FF_ITALIC
   632	        if usWeightClass >= 600:        # FW_REGULAR == 500, FW_SEMIBOLD == 600
   633	            self.flags = self.flags | FF_FORCEBOLD
   634	        if isFixedPitch:
   635	            self.flags = self.flags | FF_FIXED
   636	        # XXX: FF_SERIF?  FF_SCRIPT?  FF_ALLCAP?  FF_SMALLCAP?
   637
   638	        # hhea - Horizontal header table
   639	        self.seek_table("hhea")
   640	        ver_maj, ver_min = self.read_ushort(), self.read_ushort()
   641	        if ver_maj != 1:
   642	            raise TTFError('Unknown hhea table version %d.%04x' % (ver_maj, ver_min))
   643	        self.skip(28)
   644	        metricDataFormat = self.read_ushort()
   645	        if metricDataFormat != 0:
   646	            raise TTFError('Unknown horizontal metric data format (%d)' % metricDataFormat)
   647	        numberOfHMetrics = self.read_ushort()
   648	        if numberOfHMetrics == 0:
   649	            raise TTFError('Number of horizontal metrics is 0')
   650
   651	        # maxp - Maximum profile table
   652	        self.seek_table("maxp")
   653	        ver_maj, ver_min = self.read_ushort(), self.read_ushort()
   654	        if ver_maj != 1:
   655	            raise TTFError('Unknown maxp table version %d.%04x' % (ver_maj, ver_min))
   656	        self.numGlyphs = numGlyphs = self.read_ushort()
   657	        if not subsettingAllowed:
   658	            if self.numGlyphs>0xFF:
   659	                raise TTFError('Font does not allow subsetting/embedding (%04X)' % fsType)
   660	            else:
   661	                self._full_font = True
   662	        else:
   663	            self._full_font = False
   664
   665	        if not charInfo:
   666	            self.charToGlyph = None
   667	            self.defaultWidth = None
   668	            self.charWidths = None
   669	            return
   670
   671	        if glyphDataFormat != 0:
   672	            raise TTFError('Unknown glyph data format (%d)' % glyphDataFormat)
   673
   674	        # cmap - Character to glyph index mapping table
   675	        cmap_offset = self.seek_table("cmap")
   676	        cmapVersion = self.read_ushort()
   677	        cmapTableCount = self.read_ushort()
   678	        if cmapTableCount==0 and cmapVersion!=0:
   679	            cmapTableCount, cmapVersion = cmapVersion, cmapTableCount
   680	        encoffs = None
   681	        enc = 0
   682	        for n in range(cmapTableCount):
   683	            platform = self.read_ushort()
   684	            encoding = self.read_ushort()
   685	            offset = self.read_ulong()
   686	            if platform==3:
   687	                enc = 1
   688	                encoffs = offset
   689	            elif platform==1 and encoding==0 and enc!=1:
   690	                enc = 2
   691	                encoffs = offset
   692	            elif platform==1 and encoding==1:
   693	                enc = 1
   694	                encoffs = offset
   695	            elif platform==0 and encoding!=5:
   696	                enc = 1
   697	                encoffs = offset
   698	        if encoffs is None:
   699	            raise TTFError('could not find a suitable cmap encoding')
   700	        encoffs += cmap_offset
   701	        self.seek(encoffs)
   702	        fmt = self.read_ushort()
   703	        self.charToGlyph = charToGlyph = {}
   704	        glyphToChar = {}
   705	        if fmt in (13,12,10,8):
   706	            self.skip(2)    #padding
   707	            length = self.read_ulong()
   708	            lang = self.read_ulong()
   709	        else:
   710	            length = self.read_ushort()
   711	            lang = self.read_ushort()
   712	        if fmt==0:
   713	            T = [self.read_uint8() for i in range(length-6)]
   714	            for unichar in range(min(256,self.numGlyphs,len(table))):
   715	                glyph = T[glyph]
   716	                charToGlyph[unichar] = glyph
   717	                glyphToChar.setdefault(glyph,[]).append(unichar)
   718	        elif fmt==4:
   719	            limit = encoffs + length
   720	            segCount = int(self.read_ushort() / 2.0)
   721	            self.skip(6)
   722	            endCount = [self.read_ushort() for _ in range(segCount)]
   723	            self.skip(2)
   724	            startCount = [self.read_ushort() for _ in range(segCount)]
   725	            idDelta = [self.read_short() for _ in range(segCount)]
   726	            idRangeOffset_start = self._pos
   727	            idRangeOffset = [self.read_ushort() for _ in range(segCount)]
   728
   729	            # Now it gets tricky.
   730	            for n in range(segCount):
   731	                for unichar in range(startCount[n], endCount[n] + 1):
   732	                    if idRangeOffset[n] == 0:
   733	                        glyph = (unichar + idDelta[n]) & 0xFFFF
   734	                    else:
   735	                        offset = (unichar - startCount[n]) * 2 + idRangeOffset[n]
   736	                        offset = idRangeOffset_start + 2 * n + offset
   737	                        if offset >= limit:
   738	                            # workaround for broken fonts (like Thryomanes)
   739	                            glyph = 0
   740	                        else:
   741	                            glyph = self.get_ushort(offset)
   742	                            if glyph != 0:
   743	                                glyph = (glyph + idDelta[n]) & 0xFFFF
   744	                    charToGlyph[unichar] = glyph
   745	                    glyphToChar.setdefault(glyph,[]).append(unichar)
   746	        elif fmt==6:
   747	            first = self.read_ushort()
   748	            count = self.read_ushort()
   749	            for glyph in range(first,first+count):
   750	                unichar = self.read_ushort()
   751	                charToGlyph[unichar] = glyph
   752	                glyphToChar.setdefault(glyph,[]).append(unichar)
   753	        elif fmt==10:
   754	            first = self.read_ulong()
   755	            count = self.read_ulong()
   756	            for glyph in range(first,first+
