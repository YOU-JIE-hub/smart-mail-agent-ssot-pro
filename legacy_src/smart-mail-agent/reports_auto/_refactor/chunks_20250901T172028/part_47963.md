   arg_label = str(n)
   793	                if isinstance(outer_context, CallExpr) and len(outer_context.arg_names) >= n:
   794	                    arg_name = outer_context.arg_names[n - 1]
   795	                    if arg_name is not None:
   796	                        arg_label = f'"{arg_name}"'
   797	                if (
   798	                    arg_kind == ARG_STAR2
   799	                    and isinstance(arg_type, TypedDictType)
   800	                    and m <= len(callee.arg_names)
   801	                    and callee.arg_names[m - 1] is not None
   802	                    and callee.arg_kinds[m - 1] != ARG_STAR2
   803	                ):
   804	                    arg_name = callee.arg_names[m - 1]
   805	                    assert arg_name is not None
   806	                    arg_type_str, expected_type_str = format_type_distinctly(
   807	                        arg_type.items[arg_name], expected_type, bare=True, options=self.options
   808	                    )
   809	                    arg_label = f'"{arg_name}"'
   810	                if isinstance(outer_context, IndexExpr) and isinstance(
   811	                    outer_context.index, StrExpr
   812	                ):
   813	                    msg = 'Value of "{}" has incompatible type {}; expected {}'.format(
   814	                        outer_context.index.value,
   815	                        quote_type_string(arg_type_str),
   816	                        quote_type_string(expected_type_str),
   817	                    )
   818	                else:
   819	                    msg = "Argument {} {}has incompatible type {}; expected {}".format(
   820	                        arg_label,
   821	                        target,
   822	                        quote_type_string(arg_type_str),
   823	                        quote_type_string(expected_type_str),
   824	                    )
   825	                expected_type = get_proper_type(expected_type)
   826	                if isinstance(expected_type, UnionType):
   827	                    expected_types = list(expected_type.items)
   828	                else:
   829	                    expected_types = [expected_type]
   830	                for type in get_proper_types(expected_types):
   831	                    if isinstance(arg_type, Instance) and isinstance(type, Instance):
   832	                        notes = append_invariance_notes(notes, arg_type, type)
   833	                        notes = append_numbers_notes(notes, arg_type, type)
   834	            object_type = get_proper_type(object_type)
   835	            if isinstance(object_type, TypedDictType):
   836	                code = codes.TYPEDDICT_ITEM
   837	            else:
   838	                code = codes.ARG_TYPE
   839	        error = self.fail(msg, context, code=code)
   840	        if notes:
   841	            for note_msg in notes:
   842	                self.note(note_msg, context, code=code)
   843	        return error
