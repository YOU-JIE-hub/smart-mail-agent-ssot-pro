   162	        provided in a configuration file.  Parameters that are missing here

   165	        `data_file` is the base name of the data file to use. The config value
   166	        defaults to ".coverage".  None can be provided to prevent writing a data
   167	        file.  `data_suffix` is appended (with a dot) to `data_file` to create
   168	        the final file name.  If `data_suffix` is simply True, then a suffix is
   169	        created with the machine and process identity included.
   170
   171	        `cover_pylib` is a boolean determining whether Python code installed
   172	        with the Python interpreter is measured.  This includes the Python
   173	        standard library and any packages installed with the interpreter.
   174
   175	        If `auto_data` is true, then any existing data file will be read when
   176	        coverage measurement starts, and data will be saved automatically when
   177	        measurement stops.
   178
   179	        If `timid` is true, then a slower and simpler trace function will be
   180	        used.  This is important for some environments where manipulation of
   181	        tracing functions breaks the faster trace function.
   182
   183	        If `branch` is true, then branch coverage will be measured in addition
   184	        to the usual statement coverage.
   185
   186	        `config_file` determines what configuration file to read:
   187
   188	            * If it is ".coveragerc", it is interpreted as if it were True,
   189	              for backward compatibility.
   190
   191	            * If it is a string, it is the name of the file to read.  If the
   192	              file can't be read, it is an error.
   193
   194	            * If it is True, then a few standard files names are tried
   195	              (".coveragerc", "setup.cfg", "tox.ini").  It is not an error for
   196	              these files to not be found.
   197
   198	            * If it is False, then no configuration file is read.
   199
   200	        `source` is a list of file paths or package names.  Only code located
   201	        in the trees indicated by the file paths or package names will be
   202	        measured.
   203
   204	        `source_pkgs` is a list of package names. It works the same as
   205	        `source`, but can be used to name packages where the name can also be
   206	        interpreted as a file path.
   207
   208	        `source_dirs` is a list of file paths. It works the same as
   209	        `source`, but raises an error if the path doesn't exist, rather
   210	        than being treated as a package name.

   213	        `include` will be measured, files that match `omit` will not.  Each
   214	        will also accept a single string argument.

   217	        desired.

   220	        in the measured code.  Without this, coverage.py will get incorrect
   221	        results if these libraries are in use.  Valid strings are "greenlet",
   222	        "eventlet", "gevent", "multiprocessing", or "thread" (the default).
   223	        This can also be a list of these strings.

   226	        already-imported files will be checked to see if they should be
   227	        measured by coverage.  Importing measured files before coverage is
   228	        started can mean that code is missed.

   230	        `context` is a string to use as the :ref:`static context

   233	        If `messages` is true, some messages will be printed to stdout

   238	        :ref:`api_plugin`.  When they are provided, they will override the

   244	        .. versionadded:: 4.2
   245	            The `concurrency` parameter can now be a list of strings.

   247	        .. versionadded:: 5.0
   248	            The `check_preimported` and `context` parameters.

   362	            return

   376	        # _exclude_re is a dict that maps exclusion list names to compiled regexes.

   384	        self._plugins = Plugins(self._debug)
   385	        if self._plugin_override:

   398	    def _post_init(self) -> None:
   399	        """Stuff to do after everything is initialized."""
   400	        if self._should_write_debug:
   401	            self._should_write_debug = False
   402	            self._write_startup_debug()
   403
   404	        # "[run] _crash" will raise an exception if the value is close by in
   405	        # the call stack, for testing error handling.
   406	        if self.config._crash and self.config._crash in short_stack():
   407	            raise RuntimeError(f"Crashing because called by {self.config._crash}")

   409	    def _write_startup_debug(self) -> None:
   410	        """Write out debug info at startup if needed."""
   411	        wrote_any = False
   412	        with self._debug.without_callers():
   413	            if self._debug.should("config"):
   414	                config_info = self.config.debug_info()
   415	                write_formatted_info(self._debug.write, "config", config_info)
   416	                wrote_any = True
   417
   418	            if self._debug.should("sys"):
   419	                write_formatted_info(self._debug.write, "sys", self.sys_info())
   420	                for plugin in self._plugins:
   421	                    header = "sys: " + plugin._coverage_plugin_name
   422	                    info = plugin.sys_info()
   423	                    write_formatted_info(self._debug.write, header, info)
   424	                wrote_any = True
   425
   426	            if self._debug.should("pybehave"):
   427	                write_formatted_info(self._debug.write, "pybehave", env.debug_info())
   428	                wrote_any = True
   429
   430	        if wrote_any:
   431	            write_formatted_info(self._debug.write, "end", ())

   445	    def _check_include_omit_etc(self, filename: str, frame: FrameType) -> bool:
   446	        """Check a file name against the include/omit/etc, rules, verbosely.
   447
   448	        Returns a boolean: True if the file should be traced, False if not.
   449
   450	        """
   451	        assert self._inorout is not None
   452	        reason = self._inorout.check_include_omit_etc(filename, frame)
   453	        if self._debug.should("trace"):
   454	            if not reason:
   455	                msg = f"Including {filename!r}"
   456	            else:
   457	                msg = f"Not including {filename!r}: {reason}"
   458	            self._debug.write(msg)
   459
   460	        return not reason

   462	    def _warn(self, msg: str, slug: str | None = None, once: bool = False) -> None:
   463	        """Use `msg` as a warning.
   464
   465	        For warning suppression, use `slug` as the shorthand.
   466
   467	        If `once` is true, only show this warning once (determined by the
   468	        slug.)
   469
   470	        """
   471	        if not self._no_warn_slugs:
   472	            self._no_warn_slugs = set(self.config.disable_warnings)
   473
   474	        if slug in self._no_warn_slugs:
   475	            # Don't issue the warning
   476	            return
   477
   478	        self._warnings.append(msg)
   479	        if slug:
   480	            msg = f"{msg} ({slug})"
   481	        if self._debug.should("pid"):
   482	            msg = f"[{os.getpid()}] {msg}"
   483	        warnings.warn(msg, category=CoverageWarning, stacklevel=2)
   484
   485	        if once:
   486	            assert slug is not None
   487	            self._no_warn_slugs.add(slug)
