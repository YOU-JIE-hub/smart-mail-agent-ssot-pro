o["resolution"] = xres, yres
  1552	            else:  # No absolute unit of measurement
  1553	                self.info["resolution"] = xres, yres
  1554
  1555	        # build tile descriptors
  1556	        x = y = layer = 0
  1557	        self.tile = []
  1558	        self.use_load_libtiff = READ_LIBTIFF or self._compression != "raw"
  1559	        if self.use_load_libtiff:
  1560	            # Decoder expects entire file as one tile.
  1561	            # There's a buffer size limit in load (64k)
  1562	            # so large g4 images will fail if we use that
  1563	            # function.
  1564	            #
  1565	            # Setup the one tile for the whole image, then
  1566	            # use the _load_libtiff function.
  1567
  1568	            # libtiff handles the fillmode for us, so 1;IR should
  1569	            # actually be 1;I. Including the R double reverses the
  1570	            # bits, so stripes of the image are reversed.  See
  1571	            # https://github.com/python-pillow/Pillow/issues/279
  1572	            if fillorder == 2:
  1573	                # Replace fillorder with fillorder=1
  1574	                key = key[:3] + (1,) + key[4:]
  1575	                logger.debug("format key: %s", key)
  1576	                # this should always work, since all the
  1577	                # fillorder==2 modes have a corresponding
  1578	                # fillorder=1 mode
  1579	                self._mode, rawmode = OPEN_INFO[key]
  1580	            # YCbCr images with new jpeg compression with pixels in one plane
  1581	            # unpacked straight into RGB values
  1582	            if (
  1583	                photo == 6
  1584	                and self._compression == "jpeg"
  1585	                and self._planar_configuration == 1
  1586	            ):
  1587	                rawmode = "RGB"
  1588	            # libtiff always returns the bytes in native order.
  1589	            # we're expecting image byte order. So, if the rawmode
  1590	            # contains I;16, we need to convert from native to image
  1591	            # byte order.
  1592	            elif rawmode == "I;16":
  1593	                rawmode = "I;16N"
  1594	            elif rawmode.endswith((";16B", ";16L")):
  1595	                rawmode = rawmode[:-1] + "N"
  1596
  1597	            # Offset in the tile tuple is 0, we go from 0,0 to
  1598	            # w,h, and we only do this once -- eds
  1599	            a = (rawmode, self._compression, False, self.tag_v2.offset)
  1600	            self.tile.append(ImageFile._Tile("libtiff", (0, 0, xsize, ysize), 0, a))
  1601
  1602	        elif STRIPOFFSETS in self.tag_v2 or TILEOFFSETS in self.tag_v2:
  1603	            # striped image
  1604	            if STRIPOFFSETS in self.tag_v2:
  1605	                offsets = self.tag_v2[STRIPOFFSETS]
  1606	                h = self.tag_v2.get(ROWSPERSTRIP, ysize)
  1607	                w = xsize
  1608	            else:
  1609	                # tiled image
  1610	                offsets = self.tag_v2[TILEOFFSETS]
  1611	                tilewidth = self.tag_v2.get(TILEWIDTH)
  1612	                h = self.tag_v2.get(TILELENGTH)
  1613	                if not isinstance(tilewidth, int) or not isinstance(h, int):
  1614	                    msg = "Invalid tile dimensions"
  1615	                    raise ValueError(msg)
  1616	                w = tilewidth
  1617
  1618	            if w == xsize and h == ysize and self._planar_configuration != 2:
  1619	                # Every tile covers the image. Only use the last offset
  1620	                offsets = offsets[-1:]
  1621
  1622	            for offset in offsets:
  1623	                if x + w > xsize:
  1624	                    stride = w * sum(bps_tuple) / 8  # bytes per line
  1625	                else:
  1626	                    stride = 0
  1627
  1628	                tile_rawmode = rawmode
  1629	                if self._planar_configuration == 2:
  1630	                    # each band on it's own layer
  1631	                    tile_rawmode = rawmode[layer]
  1632	                    # adjust stride width accordingly
  1633	                    stride /= bps_count
  1634
  1635	                args = (tile_rawmode, int(stride), 1)
  1636	                self.tile.append(
  1637	                    ImageFile._Tile(
  1638	                        self._compression,
  1639	                        (x, y, min(x + w, xsize), min(y + h, ysize)),
  1640	                        offset,
  1641	                        args,
  1642	                    )
  1643	                )
  1644	                x += w
  1645	                if x >= xsize:
  1646	                    x, y = 0, y + h
  1647	                    if y >= ysize:
  1648	                        y = 0
  1649	                        layer += 1
  1650	        else:
  1651	            logger.debug("- unsupported data organization")
  1652	            msg = "unknown data organization"
  1653	            raise SyntaxError(msg)
  1654
  1655	        # Fix up info.
  1656	        if ICCPROFILE in self.tag_v2:
  1657	            self.info["icc_profile"] = self.tag_v2[ICCPROFILE]
  1658
  1659	        # fixup palette descriptor
  1660
  1661	        if self.mode in ["P", "PA"]:
  1662	            palette = [o8(b // 256) for b in self.tag_v2[COLORMAP]]
  1663	            self.palette = ImagePalette.raw("RGB;L", b"".join(palette))
  1664

  1695	def _save(im: Image.Image, fp: IO[bytes], filename: str | bytes) -> None:
  1696	    try:
  1697	        rawmode, prefix, photo, format, bits, extra = SAVE_INFO[im.mode]
  1698	    except KeyError as e:
  1699	        msg = f"cannot write mode {im.mode} as TIFF"
  1700	        raise OSError(msg) from e
  1701
  1702	    encoderinfo = im.encoderinfo
  1703	    encoderconfig = im.encoderconfig
  1704
  1705	    ifd = ImageFileDirectory_v2(prefix=prefix)
  1706	    if encoderinfo.get("big_tiff"):
  1707	        ifd._bigtiff = True
  1708
  1709	    try:
  1710	        compression = encoderinfo["compression"]
  1711	    except KeyError:
  1712	        compression = im.info.get("compression")
  1713	        if isinstance(compression, int):
  1714	            # compression value may be from BMP. Ignore it
  1715	            compression = None
  1716	    if compression is None:
  1717	        compression = "raw"
  1718	    elif compression == "tiff_jpeg":
  1719	        # OJPEG is obsolete, so use new-style JPEG compression instead
  1720	        compression = "jpeg"
  1721	    elif compression == "tiff_deflate":
  1722	        compression = "tiff_adobe_deflate"
  1723
  1724	    libtiff = WRITE_LIBTIFF or compression != "raw"
  1725
  1726	    # required for color libtiff images
  1727	    ifd[PLANAR_CONFIGURATION] = 1
  1728
  1729	    ifd[IMAGEWIDTH] = im.size[0]
  1730	    ifd[IMAGELENGTH] = im.size[1]
  1731
  1732	    # write any arbitrary tags passed in as an ImageFileDirectory
  1733	    if "tiffinfo" in encoderinfo:
  1734	        info = encoderinfo["tiffinfo"]
  1735	    elif "exif" in encoderinfo:
  1736	        info = encoderinfo["exif"]
  1737	        if isinstance(info, bytes):
  1738	            exif = Image.Exif()
  1739	            exif.load(info)
  1740	            info = exif
  1741	    else:
  1742	        info = {}
  1743	    logger.debug("Tiffinfo Keys: %s", list(info))
  1744	    if isinstance(info, ImageFileDirectory_v1):
  1745	        info = info.to_v2()
  1746	    for key in info:
  1747	        if isinstance(info, Image.Exif) and key in TiffTags.TAGS_V2_GROUPS:
  1748	            ifd[key] = info.get_ifd(key)
  1749	        else:
  1750	            ifd[key] = info.get(key)
  1751	        try:
  1752	            ifd.tagtype[key] = info.tagtype[key]
  1753	        except Exception:
  1754	            pass  # might not be an IFD. Might not have populated type
  1755
  1756	    legacy_ifd = {}
  1757	    if hasattr(im, "tag"):
  1758	        legacy_ifd = im.tag.to_v2()
  1759
  1760	    supplied_tags = {**legacy_ifd, **getattr(im, "tag_v2", {})}
  1761	    for tag in (
  1762	        # IFD offset that may not be correct in the saved image
  1763	        EXIFIFD,
  1764	        # Determined by the image format and should not be copied from legacy_ifd.
  1765	        SAMPLEFORMAT,
  1766	    ):
  1767	        if tag in supplied_tags:
  1768	            del supplied_tags[tag]
  1769
  1770	    # additions written by Greg Couch, gregc@cgl.ucsf.edu
  1771	    # inspired by image-sig posting from Kevin Cazabon, kcazabon@home.com
  1772	    if hasattr(im, "tag_v2"):
  1773	        # preserve tags from original TIFF image file
  1774	        for key in (
  1775	            RESOLUTION_UNIT,
  1776	            X_RESOLUTION,
  1777	            Y_RESOLUTION,
  1778	            IPTC_NAA_CHUNK,
  1779	            PHOTOSHOP_CHUNK,
  1780	            XMP,
  1781	        ):
  1782	            if key in im.tag_v2:
  1783	                if key == IPTC_NAA_CHUNK and im.tag_v2.tagtype[key] not in (
  1784	                    TiffTags.BYTE,
  1785	                    TiffTags.UNDEFINED,
  1786	                ):
  1787	                    del supplied_tags[key]
  1788	                else:
  1789	                    ifd[key] = im.tag_v2[key]
  1790	                    ifd.tagtype[key] = im.tag_v2.tagtype[key]
  1791
  1792	    # preserve ICC profile (should also work when saving other formats
  1793	    # which support profiles as TIFF) -- 2008-06-06 Florian Hoech
  1794	    icc = encoderinfo.get("icc_profile", im.info.get("icc_profile"))
  1795	    if icc:
  1796	        ifd[ICCPROFILE] = icc
  1797
  1798	    for key, name in [
  1799	        (IMAGEDESCRIPTION, "description"),
  1800	        (X_RESOLUTION, "resolution"),
  1801	        (Y_RESOLUTION, "resolution"),
  1802	        (X_RESOLUTION, "x_resolution"),
  1803	        (Y_RESOLUTION, "y_resolution"),
  1804	        (RESOLUTION_UNIT, "resolution_unit"),
  1805	        (SOFTWARE, "software"),
  1806	        (DATE_TIME, "date_time"),
  1807	        (ARTIST, "artist"),
  1808	        (COPYRIGHT, "copyright"),
  1809	    ]:
  1810	        if name in encoderinfo:
  1811	            ifd[key] = encoderinfo[name]
  1812
  1813	    dpi = encoderinfo.get("dpi")
  1814	    if dpi:
  1815	        ifd[RESOLUTION_UNIT] = 2
  1816	        ifd[X_RESOLUTION] = dpi[0]
  1817	        ifd[Y_RESOLUTION] = dpi[1]
  1818
  1819	    if bits != (1,):
  1820	        ifd[BITSPERSAMPLE] = bits
  1821	        if len(bits) != 1:
  1822	            ifd[SAMPLESPERPIXEL] = len(bits)
  1823	    if extra is not None:
  1824	        ifd[EXTRASAMPLES] = extra
  1825	    if format != 1:
  1826	        ifd[SAMPLEFORMAT] = format
  1827
  1828	    if PHOTOMETRIC_INTERPRETATION not in ifd:
  1829	        ifd[PHOTOMETRIC_INTERPRETATION] = photo
  1830	    elif im.mode in ("1", "L") and ifd[PHOTOMETRIC_INTERPRETATION] == 0:
  1831	        if im.mode == "1":
  1832	            inverted_im = im.copy()
  1833	            px = inverted_im.load()
  1834	            if px is not None:
  1835	                for y in range(inverted_im.height):
  1836	                    for x in range(inverted_im.width):
  1837	                        px[x, y] = 0 if px[x, y] == 255 else 255
  1838	                im = inverted_im
  1839	        else:
  1840	            im = ImageOps.invert(im)
  1841
  1842	    if im.mode in ["P", "PA"]:
  1843	        lut = im.im.getpalette("RGB", "RGB;L")
  1844	        colormap = []
  1845	        colors = len(lut) // 3
  1846	        for i in range(3):
  1847	            colormap += [v * 256 for v in lut[colors * i : colors * (i + 1)]]
  1848	            colormap += [0] * (256 - colors)
  1849	        ifd[COLORMAP] = colormap
  1850	    # data orientation
  1851	    w, h = ifd[IMAGEWIDTH], ifd[IMAGELENGTH]
  1852	    stride = len(bits) * ((w * bits[0] + 7) // 8)
  1853	    if ROWSPERSTRIP not in ifd:
  1854	        # aim for given strip size (64 KB by default) when using libtiff writer
  1855	        if libtiff:
  1856	            im_strip_size = encoderinfo.get("strip_size", STRIP_SIZE)
  1857	            rows_per_strip = 1 if stride == 0 else min(im_strip_size // stride, h)
  1858	            # JPEG encoder expects multiple of 8 rows
  1859	            if compression == "jpeg":
  1860	                rows_per_strip = min(((rows_per_strip + 7) // 8) * 8, h)
  1861	        else:
  1862	            rows_per_strip = h
  1863	        if rows_per_strip == 0:
  1864	            rows_per_strip = 1
  1865	        ifd[ROWSPERSTRIP] = rows_per_strip
  1866	    strip_byte_counts = 1 if stride == 0 else stride * ifd[ROWSPERSTRIP]
  1867	    strips_per_image = (h + ifd[ROWSPERSTRIP] - 1) // ifd[ROWSPERSTRIP]
  1868	    if strip_byte_counts >= 2**16:
  1869	        ifd.tagtype[STRIPBYTECOUNTS] = TiffTags.LONG
  1870	    ifd[STRIPBYTECOUNTS] = (strip_byte_counts,) * (strips_per_image - 1) + (
  1871	        stride * h - strip_byte_counts * (strips_per_image - 1),
  1872	    )
  1873	    ifd[STRIPOFFSETS] = tuple(
  1874	        range(0, strip_byte_counts * strips_per_image, strip_byte_counts)
  1875	    )  # this is adjusted by IFD writer
  1876	    # no compression by default:
  1877	    ifd[COMPRESSION] = COMPRESSION_INFO_REV.get(compression, 1)
  1878
  1879	    if im.mode == "YCbCr":
  1880	        for tag, default_value in {
  1881	            YCBCRSUBSAMPLING: (1, 1),
  1882	            REFERENCEBLACKWHITE: (0, 255, 128, 255, 128, 255),
  1883	        }.items():
  1884	            ifd.setdefault(tag, default_value)
  1885
  1886	    blocklist = [TILEWIDTH, TILELENGTH, TILEOFFSETS, TILEBYTECOUNTS]
  1887	    if libtiff:
  1888	        if "quality" in encoderinfo:
  1889	            quality = encoderinfo["quality"]
  1890	            if not isinstance(quality, int) or quality < 0 or quality > 100:
  1891	                msg = "Invalid quality setting"
  1892	                raise ValueError(msg)
  1893	            if compression != "jpeg":
  1894	                msg = "quality setting only supported for 'jpeg' compression"
  1895	                raise ValueError(msg)
  1896	            ifd[JPEGQUALITY] = quality
  1897
  1898	        logger.debug("Saving using libtiff encoder")
  1899	        logger.debug("Items: %s", sorted(ifd.items()))
  1900	        _fp = 0
  1901	        if hasattr(fp, "fileno"):
  1902	            try:
  1903	                fp.seek(0)
  1904	                _fp = fp.fileno()
  1905	            except io.UnsupportedOperation:
  1906	                pass
  1907
  1908	        # optional types for non core tags
  1909	        types = {}
  1910	        # STRIPOFFSETS and STRIPBYTECOUNTS are added by the library
  1911	        # based on the data in the strip.
  1912	        # OSUBFILETYPE is deprecated.
  1913	        # The other tags expect arrays with a certain length (fixed or depending on
  1914	        # BITSPERSAMPLE, etc), passing arrays with a different length will result in
  1915	        # segfaults. Block these tags until we add extra validation.
  1916	        # SUBIFD may also cause a segfault.
  1917	        blocklist += [
  1918	            OSUBFILETYPE,
  1919	            REFERENCEBLACKWHITE,
  1920	            STRIPBYTECOUNTS,
  1921	            STRIPOFFSETS,
  1922	            TRANSFERFUNCTION,
  1923	            SUBIFD,
  1924	        ]
  1925
  1926	        # bits per sample is a single short in the tiff directory, not a list.
  1927	        atts: dict[int, Any] = {BITSPERSAMPLE: bits[0]}
  1928	        # Merge the ones that we have with (optional) more bits from
  1929	        # the original file, e.g x,y resolution so that we can
  1930	        # save(load('')) == original file.
  1931	        for tag, value in itertools.chain(ifd.items(), supplied_tags.items()):
  1932	            # Libtiff can only process certain core items without adding
  1933	            # them to the custom dictionary.
  1934	            # Custom items are supported for int, float, unicode, string and byte
  1935	            # values. Other types and tuples require a tagtype.
  1936	            if tag not in TiffTags.LIBTIFF_CORE:
  1937	                if not getattr(Image.core, "libtiff_support_custom_tags", False):
  1938	                    continue
  1939
  1940	                if tag in TiffTags.TAGS_V2_GROUPS:
  1941	                    types[tag] = TiffTags.LONG8
  1942	                elif tag in ifd.tagtype:
  1943	                    types[tag] = ifd.tagtype[tag]
  1944	                elif not (isinstance(value, (int, float, str, bytes))):
  1945	                    continue
  1946	                else:
  1947	                    type = TiffTags.lookup(tag).type
  1948	                    if type:
  1949	                        types[tag] = type
  1950	            if tag not in atts and tag not in blocklist:
  1951	                if isinstance(value, str):
  1952	                    atts[tag] = value.encode("ascii", "replace") + b"\0"
  1953	                elif isinstance(value, IFDRational):
  1954	                    atts[tag] = float(value)
  1955	                else:
  1956	                    atts[tag] = value
  1957
  1958	        if SAMPLEFORMAT in atts and len(atts[SAMPLEFORMAT]) == 1:
  1959	            atts[SAMPLEFORMAT] = atts[SAMPLEFORMAT][0]
  1960
  1961	        logger.debug("Converted items: %s", sorted(atts.items()))
  1962
  1963	        # libtiff always expects the bytes in native order.
  1964	        # we're storing image byte order. So, if the rawmode
  1965	        # contains I;16, we need to convert from native to image
  1966	        # byte order.
  1967	        if im.mode in ("I;16", "I;16B", "I;16L"):
  1968	            rawmode = "I;16N"
  1969
  1970	        # Pass tags as sorted list so that the tags are set in a fixed order.
  1971	        # This is required by libtiff for some tags. For example, the JPEGQUALITY
  1972	        # pseudo tag requires that the COMPRESS tag was already set.
  1973	        tags = list(atts.items())
  1974	        tags.sort()
  1975	        a = (rawmode, compression, _fp, filename, tags, types)
  1976	        encoder = Image._getencoder(im.mode, "libtiff", a, encoderconfig)
  1977	        encoder.setimage(im.im, (0, 0) + im.size)
  1978	        while True:
  1979	            errcode, data = encoder.encode(ImageFile.MAXBLOCK)[1:]
  1980	            if not _fp:
  1981	                fp.write(data)
  1982	            if errcode:
  1983	                break
  1984	        if errcode < 0:
  1985	            msg = f"encoder error {errcode} when writing image file"
  1986	            raise OSError(msg)
  1987
  1988	    else:
  1989	        for tag in blocklist:
  1990	            del ifd[tag]
  1991	        offset = ifd.save(fp)
  1992
  1993	        ImageFile._save(
  1994	            im,
  1995	            fp,
  1996	            [ImageFile._Tile("raw", (0, 0) + im.size, offset, (rawmode, stride, 1))],
  1997	        )
  1998
  1999	    # -- helper for multi-page save --
  2000	    if "_debug_multipage" in encoderinfo:
  2001	        # just to access o32 and o16 (using correct byte order)
  2002	        setattr(im, "_debug_multipage", ifd)
