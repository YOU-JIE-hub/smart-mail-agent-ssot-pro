 = list(self.empty_char * (self.width or 1))
   200	            if self.time_per_iteration != 0:
   201	                chars[
   202	                    int(
   203	                        (math.cos(self.pos * self.time_per_iteration) / 2.0 + 0.5)
   204	                        * self.width
   205	                    )
   206	                ] = self.fill_char
   207	            bar = "".join(chars)
   208	        return bar

   210	    def format_progress_line(self) -> str:
   211	        show_percent = self.show_percent
   212
   213	        info_bits = []
   214	        if self.length is not None and show_percent is None:
   215	            show_percent = not self.show_pos
   216
   217	        if self.show_pos:
   218	            info_bits.append(self.format_pos())
   219	        if show_percent:
   220	            info_bits.append(self.format_pct())
   221	        if self.show_eta and self.eta_known and not self.finished:
   222	            info_bits.append(self.format_eta())
   223	        if self.item_show_func is not None:
   224	            item_info = self.item_show_func(self.current_item)
   225	            if item_info is not None:
   226	                info_bits.append(item_info)
   227
   228	        return (
   229	            self.bar_template
   230	            % {
   231	                "label": self.label,
   232	                "bar": self.format_bar(),
   233	                "info": self.info_sep.join(info_bits),
   234	            }
   235	        ).rstrip()

   237	    def render_progress(self) -> None:
   238	        import shutil
   239
   240	        if self.hidden:
   241	            return
   242
   243	        if not self._is_atty:
   244	            # Only output the label once if the output is not a TTY.
   245	            if self._last_line != self.label:
   246	                self._last_line = self.label
   247	                echo(self.label, file=self.file, color=self.color)
   248	            return
   249
   250	        buf = []
   251	        # Update width in case the terminal has been resized
   252	        if self.autowidth:
   253	            old_width = self.width
   254	            self.width = 0
   255	            clutter_length = term_len(self.format_progress_line())
   256	            new_width = max(0, shutil.get_terminal_size().columns - clutter_length)
   257	            if new_width < old_width and self.max_width is not None:
   258	                buf.append(BEFORE_BAR)
   259	                buf.append(" " * self.max_width)
   260	                self.max_width = new_width
   261	            self.width = new_width
   262
   263	        clear_width = self.width
   264	        if self.max_width is not None:
   265	            clear_width = self.max_width
   266
   267	        buf.append(BEFORE_BAR)
   268	        line = self.format_progress_line()
   269	        line_len = term_len(line)
   270	        if self.max_width is None or self.max_width < line_len:
   271	            self.max_width = line_len
   272
   273	        buf.append(line)
   274	        buf.append(" " * (clear_width - line_len))
   275	        line = "".join(buf)
   276	        # Render the line only if it changed.
   277
   278	        if line != self._last_line:
   279	            self._last_line = line
   280	            echo(line, file=self.file, color=self.color, nl=False)
   281	            self.file.flush()

   283	    def make_step(self, n_steps: int) -> None:
   284	        self.pos += n_steps
   285	        if self.length is not None and self.pos >= self.length:
   286	            self.finished = True
   287
   288	        if (time.time() - self.last_eta) < 1.0:
   289	            return
   290
   291	        self.last_eta = time.time()
   292
   293	        # self.avg is a rolling list of length <= 7 of steps where steps are
   294	        # defined as time elapsed divided by the total progress through
   295	        # self.length.
   296	        if self.pos:
   297	            step = (time.time() - self.start) / self.pos
   298	        else:
   299	            step = time.time() - self.start
   300
   301	        self.avg = self.avg[-6:] + [step]
   302
   303	        self.eta_known = self.length is not None
   304
   305	    def update(self, n_steps: int, current_item: V | None = None) -> None:
   306	        """Update the progress bar by advancing a specified number of
   307	        steps, and optionally set the ``current_item`` for this new
   308	        position.
   309
   310	        :param n_steps: Number of steps to advance.
   311	        :param current_item: Optional item to set as ``current_item``
   312	            for the updated position.
   313
   314	        .. versionchanged:: 8.0
   315	            Added the ``current_item`` optional parameter.
   316
   317	        .. versionchanged:: 8.0
   318	            Only render when the number of steps meets the
   319	            ``update_min_steps`` threshold.
   320	        """
   321	        if current_item is not None:
   322	            self.current_item = current_item
   323
   324	        self._completed_intervals += n_steps
   325
   326	        if self._completed_intervals >= self.update_min_steps:
   327	            self.make_step(self._completed_intervals)
   328	            self.render_progress()
   329	            self._completed_intervals = 0
   330
   331	    def finish(self) -> None:
   332	        self.eta_known = False
   333	        self.current_item = None
   334	        self.finished = True
   335
   336	    def generator(self) -> cabc.Iterator[V]:
   337	        """Return a generator which yields the items added to the bar
   338	        during construction, and updates the progress bar *after* the
   339	        yielded block returns.
   340	        """
   341	        # WARNING: the iterator interface for `ProgressBar` relies on
   342	        # this and only works because this is a simple generator which
   343	        # doesn't create or manage additional state. If this function
   344	        # changes, the impact should be evaluated both against
   345	        # `iter(bar)` and `next(bar)`. `next()` in particular may call
   346	        # `self.generator()` repeatedly, and this must remain safe in
   347	        # order for that interface to work.
   348	        if not self.entered:
   349	            raise RuntimeError("You need to use progress bars in a with block.")
   350
   351	        if not self._is_atty:
   352	            yield from self.iter
   353	        else:
   354	            for rv in self.iter:
   355	                self.current_item = rv
   356
   357	                # This allows show_item_func to be updated before the
   358	                # item is processed. Only trigger at the beginning of
   359	                # the update interval.
   360	                if self._completed_intervals == 0:
   361	                    self.render_progress()
   362
   363	                yield rv
   364	                self.update(1)
   365
   366	            self.finish()
   367	            self.render_progress()

   370	def pager(generator: cabc.Iterable[str], color: bool | None = None) -> None:
   371	    """Decide what method to use for paging through text."""
   372	    stdout = _default_text_stdout()
   373
   374	    # There are no standard streams attached to write to. For example,
   375	    # pythonw on Windows.
   376	    if stdout is None:
   377	        stdout = StringIO()
   378
   379	    if not isatty(sys.stdin) or not isatty(stdout):
   380	        return _nullpager(stdout, generator, color)
   381
   382	    # Split and normalize the pager command into parts.
   383	    pager_cmd_parts = shlex.split(os.environ.get("PAGER", ""), posix=False)
   384	    if pager_cmd_parts:
   385	        if WIN:
   386	            if _tempfilepager(generator, pager_cmd_parts, color):
   387	                return
   388	        elif _pipepager(generator, pager_cmd_parts, color):
   389	            return
   390
   391	    if os.environ.get("TERM") in ("dumb", "emacs"):
   392	        return _nullpager(stdout, generator, color)
   393	    if (WIN or sys.platform.startswith("os2")) and _tempfilepager(
   394	        generator, ["more"], color
   395	    ):
   396	        return
   397	    if _pipepager(generator, ["less"], color):
   398	        return
   399
   400	    import tempfile
   401
   402	    fd, filename = tempfile.mkstemp()
   403	    os.close(fd)
   404	    try:
   405	        if _pipepager(generator, ["more"], color):
   406	            return
   407	        return _nullpager(stdout, generator, color)
   408	    finally:
   409	        os.unlink(filename)

   412	def _pipepager(
   413	    generator: cabc.Iterable[str], cmd_parts: list[str], color: bool | None
   414	) -> bool:
   415	    """Page through text by feeding it to another program. Invoking a
   416	    pager through this might support colors.
   417
   418	    Returns `True` if the command was found, `False` otherwise and thus another
   419	    pager should be attempted.
   420	    """
   421	    # Split the command into the invoked CLI and its parameters.
   422	    if not cmd_parts:
   423	        return False
   424	    cmd = cmd_parts[0]
   425	    cmd_params = cmd_parts[1:]
   426
   427	    cmd_filepath = which(cmd)
   428	    if not cmd_filepath:
   429	        return False
   430	    # Resolves symlinks and produces a normalized absolute path string.
   431	    cmd_path = Path(cmd_filepath).resolve()
   432	    cmd_name = cmd_path.name
   433
   434	    import subprocess
   435
   436	    # Make a local copy of the environment to not affect the global one.
   437	    env = dict(os.environ)
   438
   439	    # If we're piping to less and the user hasn't decided on colors, we enable
   440	    # them by default we find the -R flag in the command line arguments.
   441	    if color is None and cmd_name == "less":
   442	        less_flags = f"{os.environ.get('LESS', '')}{' '.join(cmd_params)}"
   443	        if not less_flags:
   444	            env["LESS"] = "-R"
   445	            color = True
   446	        elif "r" in less_flags or "R" in less_flags:
   447	            color = True
   448
   449	    c = subprocess.Popen(
   450	        [str(cmd_path)] + cmd_params,
   451	        shell=True,
   452	        stdin=subprocess.PIPE,
   453	        env=env,
   454	        errors="replace",
   455	        text=True,
   456	    )
   457	    assert c.stdin is not None
   458	    try:
   459	        for text in generator:
   460	            if not color:
   461	                text = strip_ansi(text)
   462
   463	            c.stdin.write(text)
   464	    except BrokenPipeError:
   465	        # In case the pager exited unexpectedly, ignore the broken pipe error.
   466	        pass
   467	    except Exception as e:
   468	        # In case there is an exception we want to close the pager immediately
   469	        # and let the caller handle it.
   470	        # Otherwise the pager will keep running, and the user may not notice
   471	        # the error message, or worse yet it may leave the terminal in a broken state.
   472	        c.terminate()
   473	        raise e
   474	    finally:
   475	        # We must close stdin and wait for the pager to exit before we continue
   476	        try:
   477	            c.stdin.close()
   478	        # Close implies flush, so it might throw a BrokenPipeError if the pager
   479	        # process exited already.
   480	        except BrokenPipeError:
   481	            pass
   482
   483	        # Less doesn't respect ^C, but catches it for its own UI purposes (aborting
   484	        # search or other commands inside less).
   485	        #
   486	        # That means when the user hits ^C, the parent process (click) terminates,
   487	        # but less is still alive, paging the output and messing up the terminal.
   488	        #
   489	        # If the user wants to make the pager exit on ^C, they should set
   490	        # `LESS='-K'`. It's not our decision to make.
   491	        while True:
   492	            try:
   493	                c.wait()
   494	            except KeyboardInterrupt:
   495	                pass
   496	            else:
   497	                break
   498
   499	    return True
   500

   555	    def __init__(
   556	        self,
   557	        editor: str | None = None,
   558	        env: cabc.Mapping[str, str] | None = None,
   559	        require_save: bool = True,
   560	        extension: str = ".txt",
   561	    ) -> None:
   562	        self.editor = editor
   563	        self.env = env
   564	        self.require_save = require_save
   565	        self.extension = extension
   566
   567	    def get_editor(self) -> str:
   568	        if self.editor is not None:
   569	            return self.editor
   570	        for key in "VISUAL", "EDITOR":
   571	            rv = os.environ.get(key)
   572	            if rv:
   573	                return rv
   574	        if WIN:
   575	            return "notepad"
   576	        for editor in "sensible-editor", "vim", "nano":
   577	            if which(editor) is not None:
   578	                return editor
   579	        return "vi"
   580
   581	    def edit_files(self, filenames: cabc.Iterable[str]) -> None:
   582	        import subprocess
   583
   584	        editor = self.get_editor()
   585	        environ: dict[str, str] | None = None
   586
   587	        if self.env:
   588	            environ = os.environ.copy()
   589	            environ.update(self.env)
   590
   591	        exc_filename = " ".join(f'"{filename}"' for filename in filenames)
   592
   593	        try:
   594	            c = subprocess.Popen(
   595	                args=f"{editor} {exc_filename}", env=environ, shell=True
   596	            )
   597	            exit_code = c.wait()
   598	            if exit_code != 0:
   599	                raise ClickException(
   600	                    _("{editor}: Editing failed").format(editor=editor)
   601	                )
   602	        except OSError as e:
   603	            raise ClickException(
   604	                _("{editor}: Editing failed: {e}").format(editor=editor, e=e)
   605	            ) from e

   607	    @t.overload
   608	    def edit(self, text: bytes | bytearray) -> bytes | None: ...
   609
   610	    # We cannot know whether or not the type expected is str or bytes when None
   611	    # is passed, so str is returned as that was what was done before.
   612	    @t.overload
   613	    def edit(self, text: str | None) -> str | None: ...
   614
   615	    def edit(self, text: str | bytes | bytearray | None) -> str | bytes | None:
   616	        import tempfile
   617
   618	        if text is None:
   619	            data = b""
   620	        elif isinstance(text, (bytes, bytearray)):
   621	            data = text
   622	        else:
   623	            if text and not text.endswith("\n"):
   624	                text += "\n"
   625
   626	            if WIN:
   627	                data = text.replace("\n", "\r\n").encode("utf-8-sig")
   628	            else:
   629	                data = text.encode("utf-8")
   630
   631	        fd, name = tempfile.mkstemp(prefix="editor-", suffix=self.extension)
   632	        f: t.BinaryIO
   633
   634	        try:
   635	            with os.fdopen(fd, "wb") as f:
   636	                f.write(data)
   637
   638	            # If the filesystem resolution is 1 second, like Mac OS
   639	            # 10.12 Extended, or 2 seconds, like FAT32, and the editor
   640	            # closes very fast, require_save can fail. Set the modified
   641	            # time to be 2 seconds in the past to work around this.
   642	            os.utime(name, (os.path.getatime(name), os.path.getmtime(name) - 2))
   643	            # Depending on the resolution, the exact value might not be
   644	            # recorded, so get the new recorded value.
   645	            timestamp = os.path.getmtime(name)
   646
   647	            self.edit_files((name,))
   648
   649	            if self.require_save and os.path.getmtime(name) == timestamp:
   650	                return None
   651
   652	            with open(name, "rb") as f:
   653	                rv = f.read()
   654
   655	            if isinstance(text, (bytes, bytearray)):
   656	                return rv
   657
   658	            return rv.decode("utf-8-sig").replace("\r\n", "\n")
   659	        finally:
   660	            os.unlink(name)

   663	def open_url(url: str, wait: bool = False, locate: bool = False) -> int:
   664	    import subprocess
   665
   666	    def _unquote_file(url: str) -> str:
   667	        from urllib.parse import unquote
   668
   669	        if url.startswith("file://"):
   670	            url = unquote(url[7:])
   671
   672	        return url
   673
   674	    if sys.platform == "darwin":
   675	        args = ["open"]
   676	        if wait:
   677	            args.append("-W")
   678	        if locate:
   679	            args.append("-R")
   680	        args.append(_unquote_file(url))
   681	        null = open("/dev/null", "w")
   682	        try:
   683	            return subprocess.Popen(args, stderr=null).wait()
   684	        finally:
   685	            null.close()
   686	    elif WIN:
   687	        if locate:
   688	            url = _unquote_file(url)
   689	            args = ["explorer", f"/select,{url}"]
   690	        else:
   691	            args = ["start"]
   692	            if wait:
   693	                args.append("/WAIT")
   694	            args.append("")
   695	            args.append(url)
   696	        try:
   697	            return subprocess.call(args)
   698	        except OSError:
   699	            # Command not found
   700	            return 127
   701	    elif CYGWIN:
   702	        if locate:
   703	            url = _unquote_file(url)
   704	            args = ["cygstart", os.path.dirname(url)]
   705	        else:
   706	            args = ["cygstart"]
   707	            if wait:
   708	                args.append("-w")
   709	            args.append(url)
   710	        try:
   711	            return subprocess.call(args)
   712	        except OSError:
   713	            # Command not found
   714	            return 127
   715
   716	    try:
   717	        if locate:
   718	            url = os.path.dirname(_unquote_file(url)) or "."
   719	        else:
   720	            url = _unquote_file(url)
   721	        c = subprocess.Popen(["xdg-open", url])
   722	        if wait:
   723
