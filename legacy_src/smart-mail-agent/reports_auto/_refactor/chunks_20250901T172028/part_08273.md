   190	class RPrimitive(RType):
   191	    """Primitive type such as 'object' or 'int'.
   192
   193	    These often have custom ops associated with them. The 'object'
   194	    primitive type can be used to hold arbitrary Python objects.
   195
   196	    Different primitive types have different representations, and
   197	    primitives may be unboxed or boxed. Primitive types don't need to
   198	    directly correspond to Python types, but most do.
   199
   200	    NOTE: All supported primitive types are defined below
   201	    (e.g. object_rprimitive).
   202	    """
   203
   204	    # Map from primitive names to primitive types and is used by deserialization
   205	    primitive_map: ClassVar[dict[str, RPrimitive]] = {}
   206
   207	    def __init__(
   208	        self,
   209	        name: str,
   210	        *,
   211	        is_unboxed: bool,
   212	        is_refcounted: bool,
   213	        is_native_int: bool = False,
   214	        is_signed: bool = False,
   215	        ctype: str = "PyObject *",
   216	        size: int = PLATFORM_SIZE,
   217	        error_overlap: bool = False,
   218	        may_be_immortal: bool = True,
   219	    ) -> None:
   220	        RPrimitive.primitive_map[name] = self
   221
   222	        self.name = name
   223	        self.is_unboxed = is_unboxed
   224	        self.is_refcounted = is_refcounted
   225	        self.is_native_int = is_native_int
   226	        self.is_signed = is_signed
   227	        self._ctype = ctype
   228	        self.size = size
   229	        self.error_overlap = error_overlap
   230	        self._may_be_immortal = may_be_immortal and HAVE_IMMORTAL
   231	        if ctype == "CPyTagged":
   232	            self.c_undefined = "CPY_INT_TAG"
   233	        elif ctype in ("int16_t", "int32_t", "int64_t"):
   234	            # This is basically an arbitrary value that is pretty
   235	            # unlikely to overlap with a real value.
   236	            self.c_undefined = "-113"
   237	        elif ctype == "CPyPtr":
   238	            # TODO: Invent an overlapping error value?
   239	            self.c_undefined = "0"
   240	        elif ctype == "PyObject *":
   241	            # Boxed types use the null pointer as the error value.
   242	            self.c_undefined = "NULL"
   243	        elif ctype == "char":
   244	            self.c_undefined = "2"
   245	        elif ctype in ("PyObject **", "void *"):
   246	            self.c_undefined = "NULL"
   247	        elif ctype == "double":
   248	            self.c_undefined = "-113.0"
   249	        elif ctype in ("uint8_t", "uint16_t", "uint32_t", "uint64_t"):
   250	            self.c_undefined = "239"  # An arbitrary number
   251	        else:
   252	            assert False, "Unrecognized ctype: %r" % ctype
   253
   254	    def accept(self, visitor: RTypeVisitor[T]) -> T:
   255	        return visitor.visit_rprimitive(self)
   256
   257	    @property
   258	    def may_be_immortal(self) -> bool:
   259	        return self._may_be_immortal
   260
   261	    def serialize(self) -> str:
   262	        return self.name
   263
   264	    def __repr__(self) -> str:
   265	        return "<RPrimitive %s>" % self.name
   266
   267	    def __eq__(self, other: object) -> bool:
   268	        return isinstance(other, RPrimitive) and other.name == self.name
   269
   270	    def __hash__(self) -> int:
   271	        return hash(self.name)

   612	class TupleNameVisitor(RTypeVisitor[str]):
   613	    """Produce a tuple name based on the concrete representations of types."""
   614
   615	    def visit_rinstance(self, t: RInstance) -> str:
   616	        return "O"
   617
   618	    def visit_runion(self, t: RUnion) -> str:
   619	        return "O"
   620
   621	    def visit_rprimitive(self, t: RPrimitive) -> str:
   622	        if t._ctype == "CPyTagged":
   623	            return "I"
   624	        elif t._ctype == "char":
   625	            return "C"
   626	        elif t._ctype == "int64_t":
   627	            return "8"  # "8 byte integer"
   628	        elif t._ctype == "int32_t":
   629	            return "4"  # "4 byte integer"
   630	        elif t._ctype == "int16_t":
   631	            return "2"  # "2 byte integer"
   632	        elif t._ctype == "uint8_t":
   633	            return "U1"  # "1 byte unsigned integer"
   634	        elif t._ctype == "double":
   635	            return "F"
   636	        assert not t.is_unboxed, f"{t} unexpected unboxed type"
   637	        return "O"
   638
   639	    def visit_rtuple(self, t: RTuple) -> str:
   640	        parts = [elem.accept(self) for elem in t.types]
   641	        return "T{}{}".format(len(parts), "".join(parts))
   642
   643	    def visit_rstruct(self, t: RStruct) -> str:
   644	        assert False, "RStruct not supported in tuple"
   645
   646	    def visit_rarray(self, t: RArray) -> str:
   647	        assert False, "RArray not supported in tuple"
   648
   649	    def visit_rvoid(self, t: RVoid) -> str:
   650	        assert False, "rvoid in tuple?"

   653	class RTuple(RType):
   654	    """Fixed-length unboxed tuple (represented as a C struct).
   655
   656	    These are used to represent mypy TupleType values (fixed-length
   657	    Python tuples). Since this is unboxed, the identity of a tuple
   658	    object is not preserved within compiled code. If the identity of a
   659	    tuple is important, or there is a need to have multiple references
   660	    to a single tuple object, a variable-length tuple should be used
   661	    (tuple_rprimitive or Tuple[T, ...]  with explicit '...'), as they
   662	    are boxed.
   663
   664	    These aren't immutable. However, user code won't be able to mutate
   665	    individual tuple items.
   666	    """
   667
   668	    is_unboxed = True
   669
   670	    def __init__(self, types: list[RType])
