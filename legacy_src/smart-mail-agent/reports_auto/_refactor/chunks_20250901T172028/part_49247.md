    69	                    version = tuple((int(i) if i != "*" else None) for i in ver_str.split("."))[0:2]

    72	                    break

    18	    def __init__(self, lock_file) -> None:
    19	        parent = os.path.dirname(lock_file)

    21	            with suppress(OSError):
