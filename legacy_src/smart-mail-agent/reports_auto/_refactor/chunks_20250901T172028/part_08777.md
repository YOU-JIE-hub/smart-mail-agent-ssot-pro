    35	    def calcLayout(self):
    36	        "Work out x and y positions for drawing"
    37
    38
    39	        rows = self.codePoints * 1.0 / self.charsPerRow
    40	        if rows == int(rows):
    41	            self.rows = int(rows)
    42	        else:
    43	            self.rows = int(rows) + 1
    44	        # size allows for a gray column of labels
    45	        self.width = self.boxSize * (1+self.charsPerRow)
    46	        self.height = self.boxSize * (1+self.rows)
    47
    48	        #handy lists
    49	        self.ylist = []
    50	        for row in range(self.rows + 2):
    51	            self.ylist.append(row * self.boxSize)
    52	        self.xlist = []
    53	        for col in range(self.charsPerRow + 2):
    54	            self.xlist.append(col * self.boxSize)

    56	    def formatByte(self, byt):
    57	        if self.hex:
    58	            return '%02X' % byt
    59	        else:
    60	            return '%d' % byt

    62	    def drawChars(self, charList):
    63	        """Fills boxes in order.  None means skip a box.
    64	        Empty boxes at end get filled with gray"""
    65	        extraNeeded = (self.rows * self.charsPerRow - len(charList))
    66	        for i in range(extraNeeded):
    67	            charList.append(None)
    68	        #charList.extend([None] * extraNeeded)
    69	        row = 0
    70	        col = 0
    71	        self.canv.setFont(self.fontName, self.boxSize * 0.75)
    72	        for ch in charList:  # may be 2 bytes or 1
    73	            if ch is None:
    74	                self.canv.setFillGray(0.9)
    75	                self.canv.rect((1+col) * self.boxSize, (self.rows - row - 1) * self.boxSize,
    76	                    self.boxSize, self.boxSize, stroke=0, fill=1)
    77	                self.canv.setFillGray(0.0)
    78	            else:
    79	                try:
    80	                    self.canv.drawCentredString(
    81	                            (col+1.5) * self.boxSize,
    82	                            (self.rows - row - 0.875) * self.boxSize,
    83	                            ch,
    84	                            )
    85	                except:
    86	                    self.canv.setFillGray(0.9)
    87	                    self.canv.rect((1+col) * self.boxSize, (self.rows - row - 1) * self.boxSize,
    88	                        self.boxSize, self.boxSize, stroke=0, fill=1)
    89	                    self.canv.drawCentredString(
    90	                            (col+1.5) * self.boxSize,
    91	                            (self.rows - row - 0.875) * self.boxSize,
    92	                            '?',
    93	                            )
    94	                    self.canv.setFillGray(0.0)
    95	            col = col + 1
    96	            if col == self.charsPerRow:
    97	                row = row + 1
    98	                col = 0

   102	        self.canv.setFillGray(0.8)
   103	        self.canv.rect(0, self.ylist[-2], self.width, self.boxSize, fill=1, stroke=0)
   104	        self.canv.rect(0, 0, self.boxSize, self.ylist[-2], fill=1, stroke=0)
   105	        self.canv.setFillGray(0.0)
   106
   107	        #label each row and column
   108	        self.canv.setFont('Helvetica-Oblique',0.375 * self.boxSize)

   110	        for row in range(self.rows):
   111	            if self.rowLabels:
   112	                label = self.rowLabels[row]
   113	            else: # format start bytes as hex or decimal
   114	                label = self.formatByte(row * self.charsPerRow)

   116	                                        (self.rows - row - 0.75) * self.boxSize,
   117	                                        label
   118	                                        )

   122	                                        self.formatByte(col)

   125	        if topLeft:

   134	                 charsPerRow=16, boxSize=14, hex=1):
   135	        self.codePoints = 256
   136	        self.faceName = faceName
   137	        self.encodingName = encodingName
   138	        self.fontName = self.faceName + '-' + self.encodingName
   139	        self.charsPerRow = charsPerRow
   140	        self.boxSize = boxSize
   141	        self.hex = hex
   142	        self.rowLabels = None
   143	        pdfmetrics.registerFont(pdfmetrics.Font(self.fontName,
   144	                                                self.faceName,
   145	                                                self.encodingName)
   146	                                )
   147
   148	        self.calcLayout()
   149
   150
   151	    def draw(self):
   152	        self.drawLabels()
   153	        charList = [None] * 32 + list(map(int2Byte, list(range(32, 256))))
   154
   155	        #we need to convert these to Unicode, since ReportLab
   156	        #2.0 can only draw in Unicode.
   157
   158	        encName = self.encodingName
   159	        #apply some common translations
   160	        encName = adobe2codec.get(encName, encName)
   161	        decoder = codecs.lookup(encName)[1]
   162	        def decodeFunc(txt):
   163	            if txt is None:
   164	                return None
   165	            else:
   166	                return decoder(txt, errors='replace')[0]
   167
   168	        charList = [decodeFunc(ch) for ch in charList]
   169
   170
   171
   172	        self.drawChars(charList)
   173	        self.canv.grid(self.xlist, self.ylist)

   305	class CodeWidget(Widget):
   306	    """Block showing all the characters"""
   307	    def __init__(self):
   308	        self.x = 0
   309	        self.y = 0
   310	        self.width = 160
   311	        self.height = 160
   312
   313	    def draw(self):
   314	        dx = self.width / 16.0
   315	        dy = self.height / 16.0
   316	        g = Group()
   317	        g.add(Rect(self.x, self.y, self.width, self.height,
   318	                   fillColor=None, strokeColor=colors.black))
   319	        for x in range(16):
   320	            for y in range(16):
   321	                charValue = y * 16 + x
   322	                if charValue > 32:
   323	                    s = String(self.x + x * dx,
   324	                               self.y + (self.height - y*dy), int2Byte(charValue))
   325	                    g.add(s)
   326	        return g
