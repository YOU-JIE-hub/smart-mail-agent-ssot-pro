  2492	            if not emitted_msg:

  2494	                self.msg.signature_incompatible_with_supertype(

  2613	        """Check that keywords in a class definition are valid arguments for __init_subclass__().

  2617	            2       def __init_subclass__(cls, thing: int):
  2618	            3           pass

  2621	            6           pass
  2622	            7   Child()

  2625	        Child.__init_subclass__ is never called.
  2626	        """
  2627	        if defn.info.metaclass_type and defn.info.metaclass_type.type.fullname not in (
  2628	            "builtins.type",
  2629	            "abc.ABCMeta",
  2630	        ):

  2632	            # metaclass are present.

  2635	        # we skip the current class itself.
  2636	        for base in defn.info.mro[1:]:

  2645	            # and is never passed to __init_subclass__ implementations
  2646	            if "metaclass" in arg_names:
  2647	                idx = arg_names.index("metaclass")
  2648	                arg_names.pop(idx)
  2649	                args.pop(idx)

  2654	            call_expr.end_line = defn.end_line

  2656	            # We are only interested in the first Base having __init_subclass__,
  2657	            # all other bases have already been checked.
