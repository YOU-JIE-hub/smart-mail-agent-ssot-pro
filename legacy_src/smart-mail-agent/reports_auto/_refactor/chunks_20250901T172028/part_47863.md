   208	    """Make free type variables generic in the type if possible.
   209
   210	    See docstring for apply_poly() for details.
   211	    """
   212
   213	    def __init__(
   214	        self,
   215	        poly_tvars: Iterable[TypeVarLikeType],
   216	        bound_tvars: frozenset[TypeVarLikeType] = frozenset(),
   217	        seen_aliases: frozenset[TypeInfo] = frozenset(),
   218	    ) -> None:
   219	        super().__init__()
   220	        self.poly_tvars = set(poly_tvars)
   221	        # This is a simplified version of TypeVarScope used during semantic analysis.
   222	        self.bound_tvars = bound_tvars
   223	        self.seen_aliases = seen_aliases
   224
   225	    def collect_vars(self, t: CallableType | Parameters) -> list[TypeVarLikeType]:
   226	        found_vars = []
   227	        for arg in t.arg_types:
   228	            for tv in get_all_type_vars(arg):
   229	                if isinstance(tv, ParamSpecType):
   230	                    normalized: TypeVarLikeType = tv.copy_modified(
   231	                        flavor=ParamSpecFlavor.BARE, prefix=Parameters([], [], [])
   232	                    )
   233	                else:
   234	                    normalized = tv
   235	                if normalized in self.poly_tvars and normalized not in self.bound_tvars:
   236	                    found_vars.append(normalized)
   237	        return remove_dups(found_vars)
   238
   239	    def visit_callable_type(self, t: CallableType) -> Type:
   240	        found_vars = self.collect_vars(t)
   241	        self.bound_tvars |= set(found_vars)
   242	        result = super().visit_callable_type(t)
   243	        self.bound_tvars -= set(found_vars)
   244
   245	        assert isinstance(result, ProperType) and isinstance(result, CallableType)
   246	        result.variables = list(result.variables) + found_vars
   247	        return result
   248
   249	    def visit_type_var(self, t: TypeVarType) -> Type:
   250	        if t in self.poly_tvars and t not in self.bound_tvars:
   251	            raise PolyTranslationError()
   252	        return super().visit_type_var(t)
   253
   254	    def visit_param_spec(self, t: ParamSpecType) -> Type:
   255	        if t in self.poly_tvars and t not in self.bound_tvars:
   256	            raise PolyTranslationError()
   257	        return super().visit_param_spec(t)
   258
   259	    def visit_type_var_tuple(self, t: TypeVarTupleType) -> Type:
   260	        if t in self.poly_tvars and t not in self.bound_tvars:
   261	            raise PolyTranslationError()
   262	        return super().visit_type_var_tuple(t)
   263
   264	    def visit_type_alias_type(self, t: TypeAliasType) -> Type:
   265	        if not t.args:
   266	            return t.copy_modified()
   267	        if not t.is_recursive:
   268	            return get_proper_type(t).accept(self)
   269	        # We can't handle polymorphic application for recursive generic aliases
   270	        # without risking an infinite recursion, just give up for now.
   271	        raise PolyTranslationError()
   272
   273	    def visit_instance(self, t: Instance) -> Type:
   274	        if t.type.has_param_spec_type:
   275	            # We need this special-casing to preserve the possibility to store a
   276	            # generic function in an instance type. Things like
   277	            #     forall T . Foo[[x: T], T]
   278	            # are not really expressible in current type system, but this looks like
   279	            # a useful feature, so let's keep it.
   280	            param_spec_index = next(
   281	                i for (i, tv) in enumerate(t.type.defn.type_vars) if isinstance(tv, ParamSpecType)
   282	            )
   283	            p = get_proper_type(t.args[param_spec_index])
   284	            if isinstance(p, Parameters):
   285	                found_vars = self.collect_vars(p)
   286	                self.bound_tvars |= set(found_vars)
   287	                new_args = [a.accept(self) for a in t.args]
   288	                self.bound_tvars -= set(found_vars)
   289
   290	                repl = new_args[param_spec_index]
   291	                assert isinstance(repl, ProperType) and isinstance(repl, Parameters)
   292	                repl.variables = list(repl.variables) + list(found_vars)
   293	                return t.copy_modified(args=new_args)
   294	        # There is the same problem with callback protocols as with aliases
   295	        # (callback protocols are essentially more flexible aliases to callables).
   296	        if t.args and t.type.is_protocol and t.type.protocol_members == ["__call__"]:
   297	            if t.type in self.seen_aliases:
   298	                raise PolyTranslationError()
   299	            call = mypy.subtypes.find_member("__call__", t, t, is_operator=True)
   300	            assert call is not None
   301	            return call.accept(
   302	                PolyTranslator(self.poly_tvars, self.bound_tvars, self.seen_aliases | {t.type})
   303	            )
   304	        return super().visit_instance(t)
