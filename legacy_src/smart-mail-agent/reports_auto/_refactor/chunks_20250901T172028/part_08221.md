   112	class TransformVisitor(NodeVisitor[Node]):
   113	    """Transform a semantically analyzed AST (or subtree) to an identical copy.
   114
   115	    Use the node() method to transform an AST node.
   116
   117	    Subclass to perform a non-identity transform.
   118
   119	    Notes:
   120
   121	     * This can only be used to transform functions or classes, not top-level
   122	       statements, and/or modules as a whole.
   123	     * Do not duplicate TypeInfo nodes. This would generally not be desirable.
   124	     * Only update some name binding cross-references, but only those that
   125	       refer to Var, Decorator or FuncDef nodes, not those targeting ClassDef or
   126	       TypeInfo nodes.
   127	     * Types are not transformed, but you can override type() to also perform
   128	       type transformation.
   129
   130	    TODO nested classes and functions have not been tested well enough
   131	    """

   133	    def __init__(self) -> None:
   134	        # To simplify testing, set this flag to True if you want to transform
   135	        # all statements in a file (this is prohibited in normal mode).
   136	        self.test_only = False
   137	        # There may be multiple references to a Var node. Keep track of
   138	        # Var translations using a dictionary.
   139	        self.var_map: dict[Var, Var] = {}
   140	        # These are uninitialized placeholder nodes used temporarily for nested
   141	        # functions while we are transforming a top-level function. This maps an
   142	        # untransformed node to a placeholder (which will later become the
   143	        # transformed node).
   144	        self.func_placeholder_map: dict[FuncDef, FuncDef] = {}

   146	    def visit_mypy_file(self, node: MypyFile) -> MypyFile:
   147	        assert self.test_only, "This visitor should not be used for whole files."

   149	        ignored_lines = {line: codes.copy() for line, codes in node.ignored_lines.items()}
   150	        new = MypyFile(self.statements(node.defs), [], node.is_bom, ignored_lines=ignored_lines)
   151	        new._fullname = node._fullname
   152	        new.path = node.path
   153	        new.names = SymbolTable()
   154	        return new

   178	    def visit_func_def(self, node: FuncDef) -> FuncDef:
   179	        # Note that a FuncDef must be transformed to a FuncDef.
   180
   181	        # These contortions are needed to handle the case of recursive
   182	        # references inside the function being transformed.
   183	        # Set up placeholder nodes for references within this function
   184	        # to other functions defined inside it.
   185	        # Don't create an entry for this function itself though,
   186	        # since we want self-references to point to the original
   187	        # function if this is the top-level node we are transforming.
   188	        init = FuncMapInitializer(self)
   189	        for stmt in node.body.body:
   190	            stmt.accept(init)
   191
   192	        new = FuncDef(
   193	            node.name,
   194	            [self.copy_argument(arg) for arg in node.arguments],
   195	            self.block(node.body),
   196	            cast(Optional[FunctionLike], self.optional_type(node.type)),
   197	        )
   198
   199	        self.copy_function_attributes(new, node)
   200
   201	        new._fullname = node._fullname
   202	        new.is_decorated = node.is_decorated
   203	        new.is_conditional = node.is_conditional
   204	        new.abstract_status = node.abstract_status
   205	        new.is_static = node.is_static
   206	        new.is_class = node.is_class
   207	        new.is_property = node.is_property
   208	        new.is_final = node.is_final
   209	        new.original_def = node.original_def
   210
   211	        if node in self.func_placeholder_map:
   212	            # There is a placeholder definition for this function. Replace
   213	            # the attributes of the placeholder with those form the transformed
   214	            # function. We know that the classes will be identical (otherwise
   215	            # this wouldn't work).
   216	            result = self.func_placeholder_map[node]
   217	            replace_object_state(result, new)
   218	            return result
   219	        else:
   220	            return new

   243	        items = [cast(OverloadPart, item.accept(self)) for item in node.items]

   253	        new.is_class = node.is_class
   254	        new.is_property = node.is_property
   255	        new.is_final = node.is_final

   301	        new.is_staticmethod = node.is_staticmethod
   302	        new.is_classmethod = node.is_classmethod
   303	        new.is_property = node.is_property
   304	        new.is_final = node.is_final
   305	        new.final_value = node.final_value
   306	        new.final_unset_in_class = node.final_unset_in_class
   307	        new.final_set_in_init = node.final_set_in_init
   308	        new.set_line(node)

   489	    def copy_ref(self, new: RefExpr, original: RefExpr) -> None:
   490	        new.kind = original.kind
   491	        new.fullname = original.fullname
   492	        target = original.node
   493	        if isinstance(target, Var):
   494	            # Do not transform references to global variables. See
   495	            # testGenericFunctionAliasExpand for an example where this is important.
   496	            if original.kind != GDEF:
   497	                target = self.visit_var(target)
   498	        elif isinstance(target, Decorator):
   499	            target = self.visit_var(target.var)
   500	        elif isinstance(target, FuncDef):
   501	            # Use a placeholder node for the function if it exists.
   502	            target = self.func_placeholder_map.get(target, target)
   503	        new.node = target
   504	        new.is_new_def = original.is_new_def
   505	        new.is_inferred_def = original.is_inferred_def
