   744	    def visit_parameters(self, template: Parameters) -> list[Constraint]:
   745	        # Constraining Any against C[P] turns into infer_against_any([P], Any)
   746	        if isinstance(self.actual, AnyType):
   747	            return self.infer_against_any(template.arg_types, self.actual)
   748	        if type_state.infer_polymorphic and isinstance(self.actual, Parameters):
   749	            # For polymorphic inference we need to be able to infer secondary constraints
   750	            # in situations like [x: T] <: P <: [x: int].
   751	            return infer_callable_arguments_constraints(template, self.actual, self.direction)
   752	        if type_state.infer_polymorphic and isinstance(self.actual, ParamSpecType):
   753	            # Similar for [x: T] <: Q <: Concatenate[int, P].
   754	            return infer_callable_arguments_constraints(
   755	                template, self.actual.prefix, self.direction
   756	            )
   757	        # There also may be unpatched types after a user error, simply ignore them.
   758	        return []

   761
   762	    def visit_instance(self, template: Instance) -> list[Constraint]:
   763	        original_actual = actual = self.actual
   764	        res: list[Constraint] = []
   765	        if isinstance(actual, (CallableType, Overloaded)) and template.type.is_protocol:
   766	            if "__call__" in template.type.protocol_members:
   767	                # Special case: a generic callback protocol
   768	                if not any(template == t for t in template.type.inferring):
   769	                    template.type.inferring.append(template)
   770	                    call = mypy.subtypes.find_member(
   771	                        "__call__", template, actual, is_operator=True
   772	                    )
   773	                    assert call is not None
   774	                    if (
   775	                        self.direction == SUPERTYPE_OF
   776	                        and mypy.subtypes.is_subtype(actual, erase_typevars(call))
   777	                        or self.direction == SUBTYPE_OF
   778	                        and mypy.subtypes.is_subtype(erase_typevars(call), actual)
   779	                    ):
   780	                        res.extend(infer_constraints(call, actual, self.direction))
   781	                    template.type.inferring.pop()
   782	        if isinstance(actual, CallableType) and actual.fallback is not None:
   783	            if (
   784	                actual.is_type_obj()
   785	                and template.type.is_protocol
   786	                and self.direction == SUPERTYPE_OF
   787	            ):
   788	                ret_type = get_proper_type(actual.ret_type)
   789	                if isinstance(ret_type, TupleType):
   790	                    ret_type = mypy.typeops.tuple_fallback(ret_type)
   791	                if isinstance(ret_type, Instance):
   792	                    res.extend(
   793	                        self.infer_constraints_from_protocol_members(
   794	                            ret_type, template, ret_type, template, class_obj=True
   795	                        )
   796	                    )
   797	            actual = actual.fallback
   798	        if isinstance(actual, TypeType) and template.type.is_protocol:
   799	            if self.direction == SUPERTYPE_OF:
   800	                a_item = actual.item
   801	                if isinstance(a_item, Instance):
   802	                    res.extend(
   803	                        self.infer_constraints_from_protocol_members(
   804	                            a_item, template, a_item, template, class_obj=True
   805	                        )
   806	                    )
   807	                # Infer constraints for Type[T] via metaclass of T when it makes sense.
   808	                if isinstance(a_item, TypeVarType):
   809	                    a_item = get_proper_type(a_item.upper_bound)
   810	                if isinstance(a_item, Instance) and a_item.type.metaclass_type:
   811	                    res.extend(
   812	                        self.infer_constraints_from_protocol_members(
   813	                            a_item.type.metaclass_type, template, actual, template
   814	                        )
   815	                    )
   816
   817	        if isinstance(actual, Overloaded) and actual.fallback is not None:
   818	            actual = actual.fallback
   819	        if isinstance(actual, TypedDictType):
   820	            actual = actual.as_anonymous().fallback
   821	        if isinstance(actual, LiteralType):
   822	            actual = actual.fallback
   823	        if isinstance(actual, Instance):
   824	            instance = actual
   825	            erased = erase_typevars(template)
   826	            assert isinstance(erased, Instance)  # type: ignore[misc]
   827	            # We always try nominal inference if possible,
   828	            # it is much faster than the structural one.
   829	            if self.direction == SUBTYPE_OF and template.type.has_base(instance.type.fullname):
   830	                mapped = map_instance_to_supertype(template, instance.type)
   831	                tvars = mapped.type.defn.type_vars
   832
   833	                if instance.type.has_type_var_tuple_type:
   834	                    # Variadic types need special handling to map each type argument to
   835	                    # the correct corresponding type variable.
   836	                    assert instance.type.type_var_tuple_prefix is not None
   837	                    assert instance.type.type_var_tuple_suffix is not None
   838	                    prefix_len = instance.type.type_var_tuple_prefix
   839	                    suffix_len = instance.type.type_var_tuple_suffix
   840	                    tvt = instance.type.defn.type_vars[prefix_len]
   841	                    assert isinstance(tvt, TypeVarTupleType)
   842	                    fallback = tvt.tuple_fallback
   843	                    i_prefix, i_middle, i_suffix = split_with_prefix_and_suffix(
   844	                        instance.args, prefix_len, suffix_len
   845	                    )
   846	                    m_prefix, m_middle, m_suffix = split_with_prefix_and_suffix(
   847	                        mapped.args, prefix_len, suffix_len
   848	                    )
   849	                    instance_args = i_prefix + (TupleType(list(i_middle), fallback),) + i_suffix
   850	                    mapped_args = m_prefix + (TupleType(list(m_middle), fallback),) + m_suffix
   851	                else:
   852	                    mapped_args = mapped.args
   853	                    instance_args = instance.args
   854
   855	                # N.B: We use zip instead of indexing because the lengths might have
   856	                # mismatches during daemon reprocessing.
   857	                for tvar, mapped_arg, instance_arg in zip(tvars, mapped_args, instance_args):
   858	                    if isinstance(tvar, TypeVarType):
   859	                        # The constraints for generic type parameters depend on variance.
   860	                        # Include constraints from both directions if invariant.
   861	                        if tvar.variance != CONTRAVARIANT:
   862	                            res.extend(infer_constraints(mapped_arg, instance_arg, self.direction))
   863	                        if tvar.variance != COVARIANT:
   864	                            res.extend(
   865	                                infer_constraints(mapped_arg, instance_arg, neg_op(self.direction))
   866	                            )
   867	                    elif isinstance(tvar, ParamSpecType) and isinstance(mapped_arg, ParamSpecType):
   868	                        prefix = mapped_arg.prefix
   869	                        if isinstance(instance_arg, Parameters):
   870	                            # No such thing as variance for ParamSpecs, consider them invariant
   871	                            # TODO: constraints between prefixes using
   872	                            # infer_callable_arguments_constraints()
   873	                            suffix: Type = instance_arg.copy_modified(
   874	                                instance_arg.arg_types[len(prefix.arg_types) :],
   875	                                instance_arg.arg_kinds[len(prefix.arg_kinds) :],
   876	                                instance_arg.arg_names[len(prefix.arg_names) :],
   877	                            )
   878	                            res.append(Constraint(mapped_arg, SUBTYPE_OF, suffix))
   879	                            res.append(Constraint(mapped_arg, SUPERTYPE_OF, suffix))
   880	                        elif isinstance(instance_arg, ParamSpecType):
   881	                            suffix = instance_arg.copy_modified(
   882	                                prefix=Parameters(
   883	                                    instance_arg.prefix.arg_types[len(prefix.arg_types) :],
   884	                                    instance_arg.prefix.arg_kinds[len(prefix.arg_kinds) :],
   885	                                    instance_arg.prefix.arg_names[len(prefix.arg_names) :],
   886	                                )
   887	                            )
   888	                            res.append(Constraint(mapped_arg, SUBTYPE_OF, suffix))
   889	                            res.append(Constraint(mapped_arg, SUPERTYPE_OF, suffix))
   890	                    elif isinstance(tvar, TypeVarTupleType):
   891	                        # Handle variadic type variables covariantly for consistency.
   892	                        res.extend(infer_constraints(mapped_arg, instance_arg, self.direction))
   893
   894	                return res
   895	            elif self.direction == SUPERTYPE_OF and instance.type.has_base(template.type.fullname):
   896	                mapped = map_instance_to_supertype(instance, template.type)
   897	                tvars = template.type.defn.type_vars
   898	                if template.type.has_type_var_tuple_type:
   899	                    # Variadic types need special handling to map each type argument to
   900	                    # the correct corresponding type variable.
   901	                    assert template.type.type_var_tuple_prefix is not None
   902	                    assert template.type.type_var_tuple_suffix is not None
   903	                    prefix_len = template.type.type_var_tuple_prefix
   904	                    suffix_len = template.type.type_var_tuple_suffix
   905	                    tvt = template.type.defn.type_vars[prefix_len]
   906	                    assert isinstance(tvt, TypeVarTupleType)
   907	                    fallback = tvt.tuple_fallback
   908	                    t_prefix, t_middle, t_suffix = split_with_prefix_and_suffix(
   909	                        template.args, prefix_len, suffix_len
   910	                    )
   911	                    m_prefix, m_middle, m_suffix = split_with_prefix_and_suffix(
   912	                        mapped.args, prefix_len, suffix_len
   913	                    )
   914	                    template_args = t_prefix + (TupleType(list(t_middle), fallback),) + t_suffix
   915	                    mapped_args = m_prefix + (TupleType(list(m_middle), fallback),) + m_suffix
   916	                else:
   917	                    mapped_args = mapped.args
   918	                    template_args = template.args
   919	                # N.B: We use zip instead of indexing because the lengths might have
   920	                # mismatches during daemon reprocessing.
   921	                for tvar, mapped_arg, template_arg in zip(tvars, mapped_args, template_args):
   922	                    if isinstance(tvar, TypeVarType):
   923	                        # The constraints for generic type parameters depend on variance.
   924	                        # Include constraints from both directions if invariant.
   925	                        if tvar.variance != CONTRAVARIANT:
   926	                            res.extend(infer_constraints(template_arg, mapped_arg, self.direction))
   927	                        if tvar.variance != COVARIANT:
   928	                            res.extend(
   929	                                infer_constraints(template_arg, mapped_arg, neg_op(self.direction))
   930	                            )
   931	                    elif isinstance(tvar, ParamSpecType) and isinstance(
   932	                        template_arg, ParamSpecType
   933	                    ):
   934	                        prefix = template_arg.prefix
   935	                        if isinstance(mapped_arg, Parameters):
   936	                            # No such thing as variance for ParamSpecs, consider them invariant
   937	                            # TODO: constraints between prefixes using
   938	                            # infer_callable_arguments_constraints()
   939	                            suffix = mapped_arg.copy_modified(
   940	                                mapped_arg.arg_types[len(prefix.arg_types) :],
   941	                                mapped_arg.arg_kinds[len(prefix.arg_kinds) :],
   942	                                mapped_arg.arg_names[len(prefix.arg_names) :],
   943	                            )
   944	                            res.append(Constraint(template_arg, SUBTYPE_OF, suffix))
   945	                            res.append(Constraint(template_arg, SUPERTYPE_OF, suffix))
   946	                        elif isinstance(mapped_arg, ParamSpecType):
   947	                            suffix = mapped_arg.copy_modified(
   948	                                prefix=Parameters(
   949	                                    mapped_arg.prefix.arg_types[len(prefix.arg_types) :],
   950	                                    mapped_arg.prefix.arg_kinds[len(prefix.arg_kinds) :],
   951	                                    mapped_arg.prefix.arg_names[len(prefix.arg_names) :],
   952	                                )
   953	                            )
   954	                            res.append(Constraint(template_arg, SUBTYPE_OF, suffix))
   955	                            res.append(Constraint(template_arg, SUPERTYPE_OF, suffix))
   956	                    elif isinstance(tvar, TypeVarTupleType):
   957	                        # Consider variadic type variables to be invariant.
   958	                        res.extend(infer_constraints(template_arg, mapped_arg, SUBTYPE_OF))
   959	                        res.extend(infer_constraints(template_arg, mapped_arg, SUPERTYPE_OF))
   960	                return res
   961	            if (
   962	                template.type.is_protocol
   963	                and self.direction == SUPERTYPE_OF
   964	                and
   965	                # We avoid infinite recursion for structural subtypes by checking
   966	                # whether this type already appeared in the inference chain.
   967	                # This is a conservative way to break the inference cycles.
   968	                # It never produces any "false" constraints but gives up soon
   969	                # on purely structural inference cycles, see #3829.
   970	                # Note that we use is_protocol_implementation instead of is_subtype
   971	                # because some type may be considered a subtype of a protocol
   972	                # due to _promote, but still not implement the protocol.
   973	                not any(template == t for t in reversed(template.type.inferring))
   974	                and mypy.subtypes.is_protocol_implementation(instance, erased, skip=["__call__"])
   975	            ):
   976	                template.type.inferring.append(template)
   977	                res.extend(
   978	                    self.infer_constraints_from_protocol_members(
   979	                        instance, template, original_actual, template
   980	                    )
   981	                )
   982	                template.type.inferring.pop()
   983	                return res
   984	            elif (
   985	                instance.type.is_protocol
   986	                and self.direction == SUBTYPE_OF
   987	                and
   988	                # We avoid infinite recursion for structural subtypes also here.
   989	                not any(instance == i for i in reversed(instance.type.inferring))
   990	                and mypy.subtypes.is_protocol_implementation(erased, instance, skip=["__call__"])
   991	            ):
   992	                instance.type.inferring.append(instance)
   993	                res.extend(
   994	                    self.infer_constraints_from_protocol_members(
   995	                        instance, template, template, instance
   996	                    )
   997	                )
   998	                instance.type.inferring.pop()
   999	                return res
  1000	        if res:
  1001	            return res
  1002
  1003	        if isinstance(actual, AnyType):
  1004	            return self.infer_against_any(template.args, actual)
  1005	        if (
  1006	            isinstance(actual, TupleType)
  1007	            and is_named_instance(template, TUPLE_LIKE_INSTANCE_NAMES)
  1008	            and self.direction == SUPERTYPE_OF
  1009	        ):
  1010	            for item in actual.items:
  1011	                if isinstance(item, UnpackType):
  1012	                    unpacked = get_proper_type(item.type)
  1013	                    if isinstance(unpacked, TypeVarTupleType):
  1014	                        # Cannot infer anything for T from [T, ...] <: *Ts
  1015	                        continue
  1016	                    assert (
  1017	                        isinstance(unpacked, Instance)
  1018	                        and unpacked.type.fullname == "builtins.tuple"
  1019	                    )
  1020	                    item = unpacked.args[0]
  1021	                cb = infer_constraints(template.args[0], item, SUPERTYPE_OF)
  1022	                res.extend(cb)
  1023	            return res
  1024	        elif isinstance(actual, TupleType) and self.direction == SUPERTYPE_OF:
  1025	            return infer_constraints(template, mypy.typeops.tuple_fallback(actual), self.direction)
  1026	        elif isinstance(actual, TypeVarType):
  1027	            if not actual.values and not actual.id.is_meta_var():
  1028	                return infer_constraints(template, actual.upper_bound, self.direction)
  1029	            return []
  1030	        elif isinstance(actual, ParamSpecType):
  1031	            return infer_constraints(template, actual.upper_bound, self.direction)
  1032	        elif isinstance(actual, TypeVarTupleType):
  1033	            raise NotImplementedError
  1034	        else:
  1035	            return []
