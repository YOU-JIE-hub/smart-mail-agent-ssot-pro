  2094	def instantiate_type_alias(
  2095	    node: TypeAlias,
  2096	    args: list[Type],

  2098	    no_args: bool,
  2099	    ctx: Context,
  2100	    options: Options,
  2101	    *,
  2102	    unexpanded_type: Type | None = None,
  2103	    disallow_any: bool = False,
  2104	    use_standard_error: bool = False,
  2105	    empty_tuple_index: bool = False,
  2106	) -> Type:
  2107	    """Create an instance of a (generic) type alias from alias node and type arguments.
  2108
  2109	    We are following the rules outlined in TypeAlias docstring.
  2110	    Here:
  2111	        node: type alias node (definition)
  2112	        args: type arguments (types to be substituted in place of type variables
  2113	              when expanding the alias)
  2114	        fail: error reporter callback

  2117	        unexpanded_type, disallow_any, use_standard_error: used to customize error messages

  2120	    # so we need to normalize them as soon as possible.
  2121	    # TODO: can this cause an infinite recursion?
  2122	    args = flatten_nested_tuples(args)
  2123	    if any(unknown_unpack(a) for a in args):
  2124	        # This type is not ready to be validated, because of unknown total count.
  2125	        # Note that we keep the kind of Any for consistency.
  2126	        return set_any_tvars(node, [], ctx.line, ctx.column, options, special_form=True)
  2127
  2128	    max_tv_count = len(node.alias_tvars)
  2129	    act_len = len(args)
  2130	    if (
  2131	        max_tv_count > 0
  2132	        and act_len == 0
  2133	        and not (empty_tuple_index and node.tvar_tuple_index is not None)
  2134	    ):
  2135	        # Interpret bare Alias same as normal generic, i.e., Alias[Any, Any, ...]
  2136	        return set_any_tvars(
  2137	            node,
  2138	            args,
  2139	            ctx.line,
  2140	            ctx.column,
  2141	            options,
  2142	            disallow_any=disallow_any,
  2143	            fail=fail,
  2144	            unexpanded_type=unexpanded_type,
  2145	        )
  2146	    if max_tv_count == 0 and act_len == 0:
  2147	        if no_args:
  2148	            assert isinstance(node.target, Instance)  # type: ignore[misc]
  2149	            # Note: this is the only case where we use an eager expansion. See more info about
  2150	            # no_args aliases like L = List in the docstring for TypeAlias class.
  2151	            return Instance(node.target.type, [], line=ctx.line, column=ctx.column)
  2152	        return TypeAliasType(node, [], line=ctx.line, column=ctx.column)
  2153	    if (
  2154	        max_tv_count == 0
  2155	        and act_len > 0
  2156	        and isinstance(node.target, Instance)  # type: ignore[misc]
  2157	        and no_args
  2158	    ):
  2159	        tp = Instance(node.target.type, args)
  2160	        tp.line = ctx.line
  2161	        tp.column = ctx.column
  2162	        tp.end_line = ctx.end_line
  2163	        tp.end_column = ctx.end_column
  2164	        return tp
  2165	    if node.tvar_tuple_index is None:
  2166	        if any(isinstance(a, UnpackType) for a in args):
  2167	            # A variadic unpack in fixed size alias (fixed unpacks must be flattened by the caller)
  2168	            fail(message_registry.INVALID_UNPACK_POSITION, ctx, code=codes.VALID_TYPE)
  2169	            return set_any_tvars(node, [], ctx.line, ctx.column, options, from_error=True)
  2170	        min_tv_count = sum(not tv.has_default() for tv in node.alias_tvars)
  2171	        fill_typevars = act_len != max_tv_count
  2172	        correct = min_tv_count <= act_len <= max_tv_count
  2173	    else:
  2174	        min_tv_count = sum(
  2175	            not tv.has_default() and not isinstance(tv, TypeVarTupleType)
  2176	            for tv in node.alias_tvars
  2177	        )
  2178	        correct = act_len >= min_tv_count
  2179	        for a in args:
  2180	            if isinstance(a, UnpackType):
  2181	                unpacked = get_proper_type(a.type)
  2182	                if isinstance(unpacked, Instance) and unpacked.type.fullname == "builtins.tuple":
  2183	                    # Variadic tuple is always correct.
  2184	                    correct = True
  2185	        fill_typevars = not correct
  2186	    if fill_typevars:
  2187	        if not correct:
  2188	            if use_standard_error:
  2189	                # This is used if type alias is an internal representation of another type,
  2190	                # for example a generic TypedDict or NamedTuple.
  2191	                msg = wrong_type_arg_count(max_tv_count, max_tv_count, str(act_len), node.name)
  2192	            else:
  2193	                if node.tvar_tuple_index is not None:
  2194	                    msg = (
  2195	                        "Bad number of arguments for type alias,"
  2196	                        f" expected at least {min_tv_count}, given {act_len}"
  2197	                    )
  2198	                elif min_tv_count != max_tv_count:
  2199	                    msg = (
  2200	                        "Bad number of arguments for type alias,"
  2201	                        f" expected between {min_tv_count} and {max_tv_count}, given {act_len}"
  2202	                    )
  2203	                else:
  2204	                    msg = (
  2205	                        "Bad number of arguments for type alias,"
  2206	                        f" expected {min_tv_count}, given {act_len}"
  2207	                    )
  2208	            fail(msg, ctx, code=codes.TYPE_ARG)
  2209	            args = []
  2210	        return set_any_tvars(node, args, ctx.line, ctx.column, options, from_error=True)
  2211	    elif node.tvar_tuple_index is not None:
  2212	        # We also need to check if we are not performing a type variable tuple split.
  2213	        unpack = find_unpack_in_list(args)
  2214	        if unpack is not None:
  2215	            unpack_arg = args[unpack]
  2216	            assert isinstance(unpack_arg, UnpackType)
  2217	            if isinstance(unpack_arg.type, TypeVarTupleType):
  2218	                exp_prefix = node.tvar_tuple_index
  2219	                act_prefix = unpack
  2220	                exp_suffix = len(node.alias_tvars) - node.tvar_tuple_index - 1
  2221	                act_suffix = len(args) - unpack - 1
  2222	                if act_prefix < exp_prefix or act_suffix < exp_suffix:
  2223	                    fail("TypeVarTuple cannot be split", ctx, code=codes.TYPE_ARG)
  2224	                    return set_any_tvars(node, [], ctx.line, ctx.column, options, from_error=True)
  2225	    # TODO: we need to check args validity w.r.t alias.alias_tvars.
  2226	    # Otherwise invalid instantiations will be allowed in runtime context.
  2227	    # Note: in type context, these will be still caught by semanal_typeargs.
  2228	    typ = TypeAliasType(node, args, ctx.line, ctx.column)
  2229	    assert typ.alias is not None
  2230	    # HACK: Implement FlexibleAlias[T, typ] by expanding it to typ here.
  2231	    if (
  2232	        isinstance(typ.alias.target, Instance)  # type: ignore[misc]
  2233	        and typ.alias.target.type.fullname == "mypy_extensions.FlexibleAlias"

  2343	def detect_diverging_alias(
  2344	    node: TypeAlias,
  2345	    target: Type,
  2346	    lookup: Callable[[str, Context], SymbolTableNode | None],
  2347	    scope: TypeVarLikeScope,
  2348	) -> bool:
  2349	    """This detects type aliases that will diverge during type checking.
  2350
  2351	    For example F = Something[..., F[List[T]]]. At each expansion step this will produce
  2352	    *new* type aliases: e.g. F[List[int]], F[List[List[int]]], etc. So we can't detect
  2353	    recursion. It is a known problem in the literature, recursive aliases and generic types
  2354	    don't always go well together. It looks like there is no known systematic solution yet.
  2355
  2356	    # TODO: should we handle such aliases using type_recursion counter and some large limit?
  2357	    They may be handy in rare cases, e.g. to express a union of non-mixed nested lists:
  2358	    Nested = Union[T, Nested[List[T]]] ~> Union[T, List[T], List[List[T]], ...]
  2359	    """
  2360	    visitor = DivergingAliasDetector({node}, lookup, scope)
  2361	    _ = target.accept(visitor)
  2362	    return visitor.diverging
