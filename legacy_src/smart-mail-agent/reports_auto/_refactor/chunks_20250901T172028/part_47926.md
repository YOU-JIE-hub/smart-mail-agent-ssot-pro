  1013	                    if isinstance(unpacked, TypeVarTupleType):

  1021	                cb = infer_constraints(template.args[0], item, SUPERTYPE_OF)

  1024	        elif isinstance(actual, TupleType) and self.direction == SUPERTYPE_OF:
  1025	            return infer_constraints(template, mypy.typeops.tuple_fallback(actual), self.direction)
  1026	        elif isinstance(actual, TypeVarType):
  1027	            if not actual.values and not actual.id.is_meta_var():
  1028	                return infer_constraints(template, actual.upper_bound, self.direction)

  1031	            return infer_constraints(template, actual.upper_bound, self.direction)
  1032	        elif isinstance(actual, TypeVarTupleType):

  1034	        else:

  1037	    def infer_constraints_from_protocol_members(
  1038	        self,
  1039	        instance: Instance,
  1040	        template: Instance,
  1041	        subtype: Type,
  1042	        protocol: Instance,
  1043	        class_obj: bool = False,
  1044	    ) -> list[Constraint]:
  1045	        """Infer constraints for situations where either 'template' or 'instance' is a protocol.
  1046
  1047	        The 'protocol' is the one of two that is an instance of protocol type, 'subtype'
  1048	        is the type used to bind self during inference. Currently, we just infer constrains for
  1049	        every protocol member type (both ways for settable members).
  1050	        """
  1051	        res = []
  1052	        for member in protocol.type.protocol_members:
  1053	            inst = mypy.subtypes.find_member(member, instance, subtype, class_obj=class_obj)
  1054	            temp = mypy.subtypes.find_member(member, template, subtype)
  1055	            if inst is None or temp is None:
  1056	                if member == "__call__":
  1057	                    continue
  1058	                return []  # See #11020
  1059	            # The above is safe since at this point we know that 'instance' is a subtype
  1060	            # of (erased) 'template', therefore it defines all protocol members
  1061	            if class_obj:
  1062	                # For class objects we must only infer constraints if possible, otherwise it
  1063	                # can lead to confusion between class and instance, for example StrEnum is
  1064	                # Iterable[str] for an instance, but Iterable[StrEnum] for a class object.
  1065	                if not mypy.subtypes.is_subtype(
  1066	                    inst, erase_typevars(temp), ignore_pos_arg_names=True
  1067	                ):
  1068	                    continue
  1069	            # This exception matches the one in typeops.py, see PR #14121 for context.
  1070	            if member == "__call__" and instance.type.is_metaclass(precise=True):
  1071	                continue
  1072	            res.extend(infer_constraints(temp, inst, self.direction))
  1073	            if mypy.subtypes.IS_SETTABLE in mypy.subtypes.get_member_flags(member, protocol):
  1074	                # Settable members are invariant, add opposite constraints
  1075	                res.extend(infer_constraints(temp, inst, neg_op(self.direction)))
  1076	        return res
