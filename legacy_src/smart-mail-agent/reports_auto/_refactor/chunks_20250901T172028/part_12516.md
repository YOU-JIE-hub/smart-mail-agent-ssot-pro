    23	def _phys_tokens(toks: TokenInfos) -> TokenInfos:
    24	    """Return all physical tokens, even line continuations.
    25
    26	    tokenize.generate_tokens() doesn't return a token for the backslash that
    27	    continues lines.  This wrapper provides those tokens so that we can
    28	    re-create a faithful representation of the original source.
    29
    30	    Returns the same values as generate_tokens()
    31
    32	    """
    33	    last_line: str | None = None
    34	    last_lineno = -1
    35	    last_ttext: str = ""
    36	    for ttype, ttext, (slineno, scol), (elineno, ecol), ltext in toks:
    37	        if last_lineno != elineno:
    38	            if last_line and last_line.endswith("\\\n"):
    39	                # We are at the beginning of a new line, and the last line
    40	                # ended with a backslash.  We probably have to inject a
    41	                # backslash token into the stream. Unfortunately, there's more
    42	                # to figure out.  This code::
    43	                #
    44	                #   usage = """\
    45	                #   HEY THERE
    46	                #   """
    47	                #
    48	                # triggers this condition, but the token text is::
    49	                #
    50	                #   '"""\\\nHEY THERE\n"""'
    51	                #
    52	                # so we need to figure out if the backslash is already in the
    53	                # string token or not.
    54	                inject_backslash = True
    55	                if last_ttext.endswith("\\"):
    56	                    inject_backslash = False
    57	                elif ttype == token.STRING:
    58	                    if (last_line.endswith("\\\n") and  # pylint: disable=simplifiable-if-statement
    59	                        last_line.rstrip(" \\\n").endswith(last_ttext)):
    60	                        # Deal with special cases like such code::
    61	                        #
    62	                        #   a = ["aaa",\ # there may be zero or more blanks between "," and "\".
    63	                        #        "bbb \
    64	                        #        ccc"]
    65	                        #
    66	                        inject_backslash = True
    67	                    else:
    68	                        # It's a multi-line string and the first line ends with
    69	                        # a backslash, so we don't need to inject another.
    70	                        inject_backslash = False
    71	                elif env.PYBEHAVIOR.fstring_syntax and ttype == token.FSTRING_MIDDLE:
    72	                    inject_backslash = False
    73	                if inject_backslash:
    74	                    # Figure out what column the backslash is in.
    75	                    ccol = len(last_line.split("\n")[-2]) - 1
    76	                    # Yield the token, with a fake token type.
    77	                    yield tokenize.TokenInfo(
    78	                        99999, "\\\n",
    79	                        (slineno, ccol), (slineno, ccol+2),
    80	                        last_line,
    81	                    )
    82	            last_line = ltext
    83	        if ttype not in (tokenize.NEWLINE, tokenize.NL):
    84	            last_ttext = ttext
    85	        yield tokenize.TokenInfo(ttype, ttext, (slineno, scol), (elineno, ecol), ltext)
    86	        last_lineno = elineno

   104	def source_token_lines(source: str) -> TSourceTokenLines:
   105	    """Generate a series of lines, one for each line in `source`.
   106
   107	    Each line is a list of pairs, each pair is a token::
   108
   109	        [('key', 'def'), ('ws', ' '), ('nam', 'hello'), ('op', '('), ... ]
   110
   111	    Each pair has a token class, and the token text.
   112
   113	    If you concatenate all the token texts, and then join them with newlines,
   114	    you should have your original `source` back, with two differences:
   115	    trailing white space is not preserved, and a final line with no newline
   116	    is indistinguishable from a final line with a newline.
   117
   118	    """
   119
   120	    ws_tokens = {token.INDENT, token.DEDENT, token.NEWLINE, tokenize.NL}
   121	    line: list[tuple[str, str]] = []
   122	    col = 0
   123
   124	    source = source.expandtabs(8).replace("\r\n", "\n")
   125	    tokgen = generate_tokens(source)
   126
   127	    if env.PYBEHAVIOR.soft_keywords:
   128	        soft_key_lines = find_soft_key_lines(source)
   129	    else:
   130	        soft_key_lines = set()
   131
   132	    for ttype, ttext, (sline, scol), (_, ecol), _ in _phys_tokens(tokgen):
   133	        mark_start = True
   134	        for part in re.split("(\n)", ttext):
   135	            if part == "\n":
   136	                yield line
   137	                line = []
   138	                col = 0
   139	                mark_end = False
   140	            elif part == "":
   141	                mark_end = False
   142	            elif ttype in ws_tokens:
   143	                mark_end = False
   144	            else:
   145	                if env.PYBEHAVIOR.fstring_syntax and ttype == token.FSTRING_MIDDLE:
   146	                    part = part.replace("{", "{{").replace("}", "}}")
   147	                    ecol = scol + len(part)
   148	                if mark_start and scol > col:
   149	                    line.append(("ws", " " * (scol - col)))
   150	                    mark_start = False
   151	                tok_class = tokenize.tok_name.get(ttype, "xx").lower()[:3]
   152	                if ttype == token.NAME:
   153	                    if keyword.iskeyword(ttext):
   154	                        # Hard keywords are always keywords.
   155	                        tok_class = "key"
   156	                    elif env.PYBEHAVIOR.soft_keywords and keyword.issoftkeyword(ttext):
   157	                        # Soft keywords appear at the start of their line.
   158	                        if len(line) == 0:
   159	                            is_start_of_line = True
   160	                        elif (len(line) == 1) and line[0][0] == "ws":
   161	                            is_start_of_line = True
   162	                        else:
   163	                            is_start_of_line = False
   164	                        if is_start_of_line and sline in soft_key_lines:
   165	                            tok_class = "key"
   166	                line.append((tok_class, part))
   167	                mark_end = True
   168	            scol = 0
   169	        if mark_end:
   170	            col = ecol
   171
   172	    if line:
   173	        yield line
