   502	def generate_richcompare_wrapper(cl: ClassIR, emitter: Emitter) -> str | None:
   503	    """Generates a wrapper for richcompare dunder methods."""
   504	    # Sort for determinism on Python 3.5
   505	    matches = sorted(name for name in RICHCOMPARE_OPS if cl.has_method(name))
   506	    if not matches:
   507	        return None
   508
   509	    name = f"{DUNDER_PREFIX}_RichCompare_{cl.name_prefix(emitter.names)}"
   510	    emitter.emit_line(
   511	        "static PyObject *{name}(PyObject *obj_lhs, PyObject *obj_rhs, int op) {{".format(
   512	            name=name
   513	        )
   514	    )
   515	    emitter.emit_line("switch (op) {")
   516	    for func in matches:
   517	        emitter.emit_line(f"case {RICHCOMPARE_OPS[func]}: {{")
   518	        method = cl.get_method(func)
   519	        assert method is not None
   520	        generate_wrapper_core(method, emitter, arg_names=["lhs", "rhs"])
   521	        emitter.emit_line("}")
   522	    emitter.emit_line("}")
   523
   524	    emitter.emit_line("Py_INCREF(Py_NotImplemented);")
   525	    emitter.emit_line("return Py_NotImplemented;")
   526
   527	    emitter.emit_line("}")
   528
   529	    return name

   631	def generate_set_del_item_wrapper(cl: ClassIR, fn: FuncIR, emitter: Emitter) -> str:
   632	    """Generates a wrapper for native __setitem__ method (also works for __delitem__).
   633
   634	    This is used with the mapping protocol slot. Arguments are taken as *PyObjects and we
   635	    return a negative C int on error.
   636
   637	    Create a separate wrapper function for __delitem__ as needed and have the
   638	    __setitem__ wrapper call it if the value is NULL. Return the name
   639	    of the outer (__setitem__) wrapper.
   640	    """
   641	    method_cls = cl.get_method_and_class("__delitem__")
   642	    del_name = None
   643	    if method_cls and method_cls[1] == cl:
   644	        # Generate a separate wrapper for __delitem__
   645	        del_name = generate_del_item_wrapper(cl, method_cls[0], emitter)
   646
   647	    args = fn.args
   648	    if fn.name == "__delitem__":
   649	        # Add an extra argument for value that we expect to be NULL.
   650	        args = list(args) + [RuntimeArg("___value", object_rprimitive, ARG_POS)]
   651
   652	    name = "{}{}{}".format(DUNDER_PREFIX, "__setitem__", cl.name_prefix(emitter.names))
   653	    input_args = ", ".join(f"PyObject *obj_{arg.name}" for arg in args)
   654	    emitter.emit_line(f"static int {name}({input_args}) {{")
   655
   656	    # First check if this is __delitem__
   657	    emitter.emit_line(f"if (obj_{args[2].name} == NULL) {{")
   658	    if del_name is not None:
   659	        # We have a native implementation, so call it
   660	        emitter.emit_line(f"return {del_name}(obj_{args[0].name}, obj_{args[1].name});")
   661	    else:
   662	        # Try to call superclass method instead
   663	        emitter.emit_line(f"PyObject *super = CPy_Super(CPyModule_builtins, obj_{args[0].name});")
   664	        emitter.emit_line("if (super == NULL) return -1;")
   665	        emitter.emit_line(
   666	            'PyObject *result = PyObject_CallMethod(super, "__delitem__", "O", obj_{});'.format(
   667	                args[1].name
   668	            )
   669	        )
   670	        emitter.emit_line("Py_DECREF(super);")
   671	        emitter.emit_line("Py_XDECREF(result);")
   672	        emitter.emit_line("return result == NULL ? -1 : 0;")
   673	    emitter.emit_line("}")
   674
   675	    method_cls = cl.get_method_and_class("__setitem__")
   676	    if method_cls and method_cls[1] == cl:
   677	        generate_set_del_item_wrapper_inner(fn, emitter, args)
   678	    else:
   679	        emitter.emit_line(f"PyObject *super = CPy_Super(CPyModule_builtins, obj_{args[0].name});")
   680	        emitter.emit_line("if (super == NULL) return -1;")
   681	        emitter.emit_line("PyObject *result;")
   682
   683	        if method_cls is None and cl.builtin_base is None:
   684	            msg = f"'{cl.name}' object does not support item assignment"
   685	            emitter.emit_line(f'PyErr_SetString(PyExc_TypeError, "{msg}");')
   686	            emitter.emit_line("result = NULL;")
   687	        else:
   688	            # A base class may have __setitem__
   689	            emitter.emit_line(
   690	                'result = PyObject_CallMethod(super, "__setitem__", "OO", obj_{}, obj_{});'.format(
   691	                    args[1].name, args[2].name
   692	                )
   693	            )
   694	        emitter.emit_line("Py_DECREF(super);")
   695	        emitter.emit_line("Py_XDECREF(result);")
   696	        emitter.emit_line("return result == NULL ? -1 : 0;")
   697	        emitter.emit_line("}")
   698	    return name

   701	def generate_set_del_item_wrapper_inner(
   702	    fn: FuncIR, emitter: Emitter, args: Sequence[RuntimeArg]
   703	) -> None:
   704	    for arg in args:
   705	        generate_arg_check(arg.name, arg.type, emitter, GotoHandler("fail"))
   706	    native_args = ", ".join(f"arg_{arg.name}" for arg in args)
   707	    emitter.emit_line(
   708	        "{}val = {}{}({});".format(
   709	            emitter.ctype_spaced(fn.ret_type), NATIVE_PREFIX, fn.cname(emitter.names), native_args
   710	        )
   711	    )
   712	    emitter.emit_error_check("val", fn.ret_type, "goto fail;")
   713	    emitter.emit_dec_ref("val", fn.ret_type)
   714	    emitter.emit_line("return 0;")
   715	    emitter.emit_label("fail")
   716	    emitter.emit_line("return -1;")
   717	    emitter.emit_line("}")

   721	    """Generates a wrapper for a native __contains__ method."""
   722	    name = f"{DUNDER_PREFIX}{fn.name}{cl.name_prefix(emitter.names)}"
   723	    emitter.emit_line(f"static int {name}(PyObject *self, PyObject *obj_item) {{")

   745	def generate_wrapper_core(
   746	    fn: FuncIR,
   747	    emitter: Emitter,
   748	    optional_args: list[RuntimeArg] | None = None,
   749	    arg_names: list[str] | None = None,
   750	    cleanups: list[str] | None = None,
   751	    traceback_code: str | None = None,
   752	) -> None:
   753	    """Generates the core part of a wrapper function for a native function.
   754
   755	    This expects each argument as a PyObject * named obj_{arg} as a precondition.
   756	    It converts the PyObject *s to the necessary types, checking and unboxing if necessary,
   757	    makes the call, then boxes the result if necessary and returns it.
   758	    """
   759	    gen = WrapperGenerator(None, emitter)
   760	    gen.set_target(fn)
   761	    if arg_names:
   762	        gen.arg_names = arg_names
   763	    gen.cleanups = cleanups or []
   764	    gen.optional_args = optional_args or []
   765	    gen.traceback_code = traceback_code or ""
   766
   767	    error = ReturnHandler("NULL") if not gen.use_goto() else GotoHandler("fail")
   768	    gen.emit_arg_processing(error=error)
   769	    gen.emit_call()
   770	    gen.emit_error_handling()

   773	def generate_arg_check(
   774	    name: str,
   775	    typ: RType,
   776	    emitter: Emitter,
   777	    error: ErrorHandler | None = None,
   778	    *,
   779	    optional: bool = False,
   780	    raise_exception: bool = True,
   781	    bitmap_arg_index: int = 0,
   782	) -> None:
   783	    """Insert a runtime check for argument and unbox if necessary.
   784
   785	    The object is named PyObject *obj_{}. This is expected to generate
   786	    a value of name arg_{} (unboxed if necessary). For each primitive a runtime
   787	    check ensures the correct type.
   788	    """
   789	    error = error or AssignHandler()
   790	    if typ.is_unboxed:
   791	        if typ.error_overlap and optional:
   792	            # Update bitmap is value is provided.
   793	            init = emitter.c_undefined_value(typ)
   794	            emitter.emit_line(f"{emitter.ctype(typ)} arg_{name} = {init};")
   795	            emitter.emit_line(f"if (obj_{name} != NULL) {{")
   796	            bitmap = bitmap_name(bitmap_arg_index // BITMAP_BITS)
   797	            emitter.emit_line(f"{bitmap} |= 1 << {bitmap_arg_index & (BITMAP_BITS - 1)};")
   798	            emitter.emit_unbox(
   799	                f"obj_{name}",
   800	                f"arg_{name}",
   801	                typ,
   802	                declare_dest=False,
   803	                raise_exception=raise_exception,
   804	                error=error,
   805	                borrow=True,
   806	            )
   807	            emitter.emit_line("}")
   808	        else:
   809	            # Borrow when unboxing to avoid reference count manipulation.
   810	            emitter.emit_unbox(
   811	                f"obj_{name}",
   812	                f"arg_{name}",
   813	                typ,
   814	                declare_dest=True,
   815	                raise_exception=raise_exception,
   816	                error=error,
   817	                borrow=True,
   818	                optional=optional,
   819	            )
   820	    elif is_object_rprimitive(typ):
   821	        # Object is trivial since any object is valid
   822	        if optional:
   823	            emitter.emit_line(f"PyObject *arg_{name};")
   824	            emitter.emit_line(f"if (obj_{name} == NULL) {{")
   825	            emitter.emit_line(f"arg_{name} = {emitter.c_error_value(typ)};")
   826	            emitter.emit_lines("} else {", f"arg_{name} = obj_{name}; ", "}")
   827	        else:
   828	            emitter.emit_line(f"PyObject *arg_{name} = obj_{name};")
   829	    else:
   830	        emitter.emit_cast(
   831	            f"obj_{name}",
   832	            f"arg_{name}",
   833	            typ,
   834	            declare_dest=True,
   835	            raise_exception=raise_exception,
   836	            error=error,
   837	            optional=optional,
   838	        )

   841	class WrapperGenerator:
   842	    """Helper that simplifies the generation of wrapper functions."""
   843
   844	    # TODO: Use this for more wrappers
   845
   846	    def __init__(self, cl: ClassIR | None, emitter: Emitter) -> None:
   847	        self.cl = cl
   848	        self.emitter = emitter
   849	        self.cleanups: list[str] = []
   850	        self.optional_args: list[RuntimeArg] = []
   851	        self.traceback_code = ""
   852
   853	    def set_target(self, fn: FuncIR) -> None:
   854	        """Set the wrapped function.
   855
   856	        It's fine to modify the attributes initialized here later to customize
   857	        the wrapper function.
   858	        """
   859	        self.target_name = fn.name
   860	        self.target_cname = fn.cname(self.emitter.names)
   861	        self.num_bitmap_args = fn.sig.num_bitmap_args
   862	        if self.num_bitmap_args:
   863	            self.args = fn.args[: -self.num_bitmap_args]
   864	        else:
   865	            self.args = fn.args
   866	        self.arg_names = [arg.name for arg in self.args]
   867	        self.ret_type = fn.ret_type
   868
   869	    def wrapper_name(self) -> str:
   870	        """Return the name of the wrapper function."""
   871	        return "{}{}{}".format(
   872	            DUNDER_PREFIX,
   873	            self.target_name,
   874	            self.cl.name_prefix(self.emitter.names) if self.cl else "",
   875	        )
   876
   877	    def use_goto(self) -> bool:
   878	        """Do we use a goto for error handling (instead of straight return)?"""
   879	        return bool(self.cleanups or self.traceback_code)
   880
   881	    def emit_header(self) -> None:
   882	        """Emit the function header of the wrapper implementation."""
   883	        input_args = ", ".join(f"PyObject *obj_{arg}" for arg in self.arg_names)
   884	        self.emitter.emit_line(
   885	            "static PyObject *{name}({input_args}) {{".format(
   886	                name=self.wrapper_name(), input_args=input_args
   887	            )
   888	        )
   889
   890	    def emit_arg_processing(
   891	        self, error: ErrorHandler | None = None, raise_exception: bool = True
   892	    ) -> None:
   893	        """Emit validation and unboxing of arguments."""
   894	        error = error or self.error()
   895	        bitmap_arg_index = 0
   896	        for arg_name, arg in zip(self.arg_names, self.args):
   897	            # Suppress the argument check for *args/**kwargs, since we know it must be right.
   898	            typ = arg.type if arg.kind not in (ARG_STAR, ARG_STAR2) else object_rprimitive
   899	            optional = arg in self.optional_args
   900	            generate_arg_check(
   901	                arg_name,
   902	                typ,
   903	                self.emitter,
   904	                error,
   905	                raise_exception=raise_exception,
   906	                optional=optional,
   907	                bitmap_arg_index=bitmap_arg_index,
   908	            )
   909	            if optional and typ.error_overlap:
   910	                bitmap_arg_index += 1
   911
   912	    def emit_call(self, not_implemented_handler: str = "") -> None:
   913	        """Emit call to the wrapper function.
   914
   915	        If not_implemented_handler is non-empty, use this C code to handle
   916	        a NotImplemented return value (if it's possible based on the return type).
   917	        """
   918	        native_args = ", ".join(f"arg_{arg}" for arg in self.arg_names)
   919	        if self.num_bitmap_args:
   920	            bitmap_args = ", ".join(
   921	                [bitmap_name(i) for i in reversed(range(self.num_bitmap_args))]
   922	            )
   923	            native_args = f"{native_args}, {bitmap_args}"
   924
   925	        ret_type = self.ret_type
   926	        emitter = self.emitter
   927	        if ret_type.is_unboxed or self.use_goto():
   928	            # TODO: The Py_RETURN macros return the correct PyObject * with reference count
   929	            #       handling. Are they relevant?
   930	            emitter.emit_line(
   931	                "{}retval = {}{}({});".format(
   932	                    emitter.ctype_spaced(ret_type), NATIVE_PREFIX, self.target_cname, native_args
   933	                )
   934	            )
   935	            emitter.emit_lines(*self.cleanups)
   936	            if ret_type.is_unboxed:
   937	                emitter.emit_error_check("retval", ret_type, "return NULL;")
   938	                emitter.emit_box("retval", "retbox", ret_type, declare_dest=True)
   939
   940	            emitter.emit_line("return {};".format("retbox" if ret_type.is_unboxed else "retval"))
   941	        else:
   942	            if not_implemented_handler and not isinstance(ret_type, RInstance):
   943	                # The return value t
