def score_spam(subject: str, content: str, sender: str = "") -> Dict[str, float | List[str]]:

        score += 0.15
        reasons.append("money")

        score += 0.10
        reasons.append("caps")
    if snd.endswith("@unknown-domain.com"):

        reasons.append("suspicious_sender")

def _write_minimal_pdf(lines: Sequence[str], out_dir: str | Path, base_filename: str) -> Path:
    out_dir = Path(out_dir)
    out_dir.mkdir(parents=True, exist_ok=True)
    path = out_dir / f"{_sanitize_name(base_filename)}.pdf"
    content = "\n".join(lines or [""]).encode("utf-8")
