   108	    def finished(self, filename):
   109	        """Write the HTML reports for filename."""
   110
   111	        # Normalize the filename path
   112	        filename = os.path.normpath(filename)
   113
   114	        report_filename = self.get_report_filename(filename, suffix='.report')
   115	        source_filename = self.get_report_filename(filename, suffix='.source')
   116
   117	        if not self.errors:
   118	            # If the files exist, they are out of date; remove them
   119	            for f in [report_filename, source_filename]:
   120	                if os.path.exists(f):
   121	                    os.unlink(f)
   122	            return
   123
   124	        with open(filename, 'rb') as f:
   125	            source = f.read()
   126	        if not self.pep8report:
   127	            orig_filename = filename
   128	        filename = re.sub(r'^\./', '', filename)
   129
   130	        highest_sev = min(sev for e, sev in self.errors)
   131	        self.files.append(IndexEntry(
   132	            filename=filename,
   133	            report_name=os.path.basename(report_filename),
   134	            error_count=len(self.errors),
   135	            highest_sev=highest_sev
   136	        ))
   137
   138	        # Build an index of errors by code/description
   139	        index = []
   140	        counts = Counter()
   141	        pep8_report_errs = []
   142	        for code, errors in self.by_code.items():
   143	            sev = find_severity(code)
   144	            counts[sev] += len(errors)
   145	            e = min(errors, key=attrgetter('line_number'))
   146	            unique_messages = len({e.text for e in errors})
   147
   148	            errcounts = Counter((e.line_number, e.text) for e in errors)
   149	            errs = sorted(
   150	                (line, text, count)
   151	                for (line, text), count in errcounts.items()
   152	            )
   153	            index.append((
   154	                sev,
   155	                len(errors),
   156	                code,
   157	                e.text,
   158	                e.line_number,
   159	                unique_messages,
   160	                unique_messages > 1 or any(e[2] > 1 for e in errs),
   161	                errs
   162	            ))
   163	            if self.pep8report:
   164	                # pep8 report - Step 1: gather errors
   165	                for err in errors:
   166	                    pep8_report_errs.append(err)

   168	        if self.pep8report:
   169	            # pep8 report - Step 2: sort gathered errors by
   170	            # filename, line- and column number
   171	            pep8_report_errs.sort(key=lambda err: (
   172	                err.filename, err.line_number, err.column_number))
   173	            # Step 3: Profit. Print the errors in pep8 report format
   174	            for e in pep8_report_errs:
   175	                print((
   176	                    "{e.filename}:{e.line_number}:{e.column_number} "
   177	                    "{e.code} {e.text}"
   178	                ).format(e=e))
   179
   180	        index.sort(key=lambda r: (r[0], -r[1], r[2]))
   181
   182	        scores = []
   183	        for sev, count in sorted(counts.items()):
   184	            scores.append(
   185	                '%s: %d' % (SEVERITY_NAMES[sev - 1], count)
   186	            )
   187	        if not self.pep8report:
   188	            print(orig_filename, "has issues:", *scores)
   189
   190	        # Build a mapping of errors by line
   191	        by_line = defaultdict(Counter)
   192	        for error, sev in self.errors:
   193	            line_errors = by_line[error.line_number]
   194	            line_errors[sev, error.code, error.text] += 1
   195
   196	        # Build a table of severities by line
   197	        line_sevs = {}
   198	        for line, errs in by_line.items():
   199	            line_sevs[line] = min(e[0] for e in errs)
   200
   201	        params = self._format_source(source)
   202	        params.update(
   203	            filename=filename,
   204	            report_filename=os.path.basename(report_filename),
   205	            source_filename=os.path.basename(source_filename),
   206	            errors=by_line,
   207	            line_sevs=line_sevs,
   208	            highest_sev=highest_sev,
   209	            index=index,
   210	            title=self.options.htmltitle,
   211	        )
   212	        rendered = self.report_template.render(**params)
   213	        with codecs.open(report_filename, 'w', encoding='utf8') as f:
   214	            f.write(rendered)
   215	        rendered = self.source_template.render(**params)
   216	        with codecs.open(source_filename, 'w', encoding='utf8') as f:
   217	            f.write(rendered)
