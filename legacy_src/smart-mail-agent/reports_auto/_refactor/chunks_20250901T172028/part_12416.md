   454
   455	        pop item1 from stack, run its teardowns
   456	        pop mod1 from stack, run its teardowns

   458	    mod1 was popped because it ended its purpose with item1. The stack is:
   459
   460	        [session]
   461

   466	        push item2 to stack, run item2.setup()

   473	    because item2 is the last item. What it does is:

   477	        pop session from stack, run its teardowns

   480
   481	        []

   488	        self.stack: dict[

   500	        needed_collectors = item.listchain()
   501
   502	        # If a collector fails its setup, fail its entire subtree of items.
   503	        # The setup is not retried for each item - the same exception is used.
   504	        for col, (finalizers, exc) in self.stack.items():

   506	            if exc:
   507	                raise exc[0].with_traceback(exc[1])

   512	            self.stack[col] = ([col.teardown], None)
   513	            try:

   533	        When nextitem is None (meaning we're at the last item), the entire
   534	        stack is torn down.
   535	        """
   536	        needed_collectors = (nextitem and nextitem.listchain()) or []
   537	        exceptions: list[BaseException] = []

   539	            if list(self.stack.keys()) == needed_collectors[: len(self.stack)]:
   540	                break

   544	                fin = finalizers.pop()

   546	                    fin()
   547	                except TEST_OUTCOME as e:
   548	                    these_exceptions.append(e)
   549
   550	            if len(these_exceptions) == 1:
   551	                exceptions.extend(these_exceptions)
   552	            elif these_exceptions:
   553	                msg = f"errors while tearing down {node!r}"
   554	                exceptions.append(BaseExceptionGroup(msg, these_exceptions[::-1]))

   556	        if len(exceptions) == 1:
   557	            raise exceptions[0]
   558	        elif exceptions:
   559	            raise BaseExceptionGroup("errors during test teardown", exceptions[::-1])

   561	            assert not self.stack

    24	    Represents one of the possible fixture scopes in pytest.

    27

    57	        self_index = _SCOPE_INDICES[self]

    59	        return self_index < other_index
