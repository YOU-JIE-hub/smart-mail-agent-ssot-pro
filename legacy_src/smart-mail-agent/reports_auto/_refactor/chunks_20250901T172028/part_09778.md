    29	                       subtype="1",
    30	                       border=None,
    31	                       checksum=False,
    32	                       availheight=None) :
    33	        """
    34	           Initializes an LTO label.
    35
    36	           prefix : Up to six characters from [A-Z][0-9]. Defaults to "".
    37	           number : Label's number or None. Defaults to None.
    38	           subtype : LTO subtype string , e.g. "1" for LTO1. Defaults to "1".
    39	           border : None, or the width of the label's border. Defaults to None.

    44	        self.border = border
    45	        if (len(subtype) != 1) \
    46	            or (subtype not in ascii_uppercase + string_digits) :

    48	        if ((not number) and (len(prefix) > 6)) \
    49	           or not prefix.isalnum() :
    50	            raise ValueError("Invalid prefix '%s'" % prefix)
    51	        label = "%sL%s" % ((prefix + str(number or 0).zfill(6 - len(prefix)))[:6],
    52	                           subtype)
    53	        if len(label) != 8 :
    54	            raise ValueError("Invalid set of parameters (%s, %s, %s)" \
    55	                                % (prefix, number, subtype))
    56	        self.label = label
    57	        Standard39.__init__(self,
    58	                            label,
    59	                            ratio=self.CODERATIO,
    60	                            barHeight=self.height,
    61	                            barWidth=self.CODEBARWIDTH,
    62	                            gap=self.CODEGAP,
    63	                            lquiet=self.CODELQUIET,
    64	                            rquiet=self.CODERQUIET,
    65	                            quiet=True,
    66	                            checksum=checksum)
