    96	def _calc_pc(V,avail):
    97	    '''check list V for percentage or * values
    98	    1) absolute values go through unchanged
    99	    2) percentages are used as weights for unconsumed space
   100	    3) if no None values were seen '*' weights are
   101	    set equally with unclaimed space
   102	    otherwise * weights are assigned as None'''
   103	    R = []
   104	    r = R.append
   105	    I = []
   106	    i = I.append
   107	    J = []
   108	    j = J.append
   109	    s = avail
   110	    w = n = 0.
   111	    for v in V:
   112	        if isinstance(v,strTypes):
   113	            v = str(v).strip()
   114	            if not v:
   115	                v = None
   116	                n += 1
   117	            elif v.endswith('%'):
   118	                v = float(v[:-1])
   119	                w += v
   120	                i(len(R))
   121	            elif v=='*':
   122	                j(len(R))
   123	            else:
   124	                v = float(v)
   125	                s -= v
   126	        elif v is None:
   127	            n += 1
   128	        else:
   129	            s -= v
   130	        r(v)
   131	    s = max(0.,s)
   132	    f = s/max(100.,w)
   133	    for i in I:
   134	        R[i] *= f
   135	        s -= R[i]
   136	    s = max(0.,s)
