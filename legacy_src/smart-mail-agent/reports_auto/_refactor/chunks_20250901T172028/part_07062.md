  6153	                        # considered "always right" (i.e. even if the types are not overlapping).

  6157	                            return {expr: TypeGuardedType(node.callee.type_guard)}, {}
  6158	                        else:
  6159	                            assert node.callee.type_is is not None

  6161	                                expr,
  6162	                                *self.conditional_types_with_intersection(
  6163	                                    self.lookup_type(expr),
  6164	                                    [TypeRange(node.callee.type_is, is_upper_bound=False)],
  6165	                                    expr,
  6166	                                ),
  6167	                            )
  6168	        elif isinstance(node, ComparisonExpr):
  6169	            return self.comparison_type_narrowing_helper(node)
  6170	        elif isinstance(node, AssignmentExpr):
  6171	            if_map: dict[Expression, Type] | None
  6172	            else_map: dict[Expression, Type] | None
  6173	            if_map = {}
  6174	            else_map = {}
  6175
  6176	            if_assignment_map, else_assignment_map = self.find_isinstance_check(node.target)
  6177
  6178	            if if_assignment_map is not None:
  6179	                if_map.update(if_assignment_map)
  6180	            if else_assignment_map is not None:
  6181	                else_map.update(else_assignment_map)
  6182
  6183	            if_condition_map, else_condition_map = self.find_isinstance_check(
  6184	                node.value, in_boolean_context=False
  6185	            )
  6186
  6187	            if if_condition_map is not None:
  6188	                if_map.update(if_condition_map)
  6189	            if else_condition_map is not None:
  6190	                else_map.update(else_condition_map)
  6191
  6192	            return (
  6193	                (None if if_assignment_map is None or if_condition_map is None else if_map),
  6194	                (None if else_assignment_map is None or else_condition_map is None else else_map),
  6195	            )
  6196	        elif isinstance(node, OpExpr) and node.op == "and":
  6197	            left_if_vars, left_else_vars = self.find_isinstance_check(node.left)
  6198	            right_if_vars, right_else_vars = self.find_isinstance_check(node.right)
  6199
  6200	            # (e1 and e2) is true if both e1 and e2 are true,
  6201	            # and false if at least one of e1 and e2 is false.
  6202	            return (
  6203	                and_conditional_maps(left_if_vars, right_if_vars),
  6204	                # Note that if left else type is Any, we can't add any additional
  6205	                # types to it, since the right maps were computed assuming
  6206	                # the left is True, which may be not the case in the else branch.

  6208	            )
  6209	        elif isinstance(node, OpExpr) and node.op == "or":
  6210	            left_if_vars, left_else_vars = self.find_isinstance_check(node.left)
  6211	            right_if_vars, right_else_vars = self.find_isinstance_check(node.right)
  6212
  6213	            # (e1 or e2) is true if at least one of e1 or e2 is true,
  6214	            # and false if both e1 and e2 are false.
  6215	            return (
  6216	                or_conditional_maps(left_if_vars, right_if_vars),
  6217	                and_conditional_maps(left_else_vars, right_else_vars),
  6218	            )
  6219	        elif isinstance(node, UnaryExpr) and node.op == "not":
  6220	            left, right = self.find_isinstance_check(node.expr)
  6221	            return right, left
  6222	        elif (
  6223	            literal(node) == LITERAL_TYPE
  6224	            and self.has_type(node)
  6225	            and self.can_be_narrowed_with_len(self.lookup_type(node))

  6227	            and not custom_special_method(self.lookup_type(node), "__bool__")
  6228	            and self.options.strict_optional
  6229	        ):
  6230	            # Combine a `len(x) > 0` check with the default logic below.
  6231	            yes_type, no_type = self.narrow_with_len(self.lookup_type(node), ">", 0)
  6232	            if yes_type is not None:
  6233	                yes_type = true_only(yes_type)
  6234	            else:
  6235	                yes_type = UninhabitedType()

  6237	                no_type = false_only(no_type)
  6238	            else:
  6239	                no_type = UninhabitedType()

  6241	            else_map = {node: no_type} if not isinstance(no_type, UninhabitedType) else None
  6242	            return if_map, else_map
  6243
  6244	        # Restrict the type of the variable to True-ish/False-ish in the if and else branches
  6245	        # respectively
  6246	        original_vartype = self.lookup_type(node)
  6247	        if in_boolean_context:
  6248	            # We don't check `:=` values in expressions like `(a := A())`,
  6249	            # because they produce two error messages.
  6250	            self.check_for_truthy_type(original_vartype, node)
  6251	        vartype = try_expanding_sum_type_to_union(original_vartype, "builtins.bool")
  6252
  6253	        if_type = true_only(vartype)
  6254	        else_type = false_only(vartype)
  6255	        if_map = {node: if_type} if not isinstance(if_type, UninhabitedType) else None
  6256	        else_map = {node: else_type} if not isinstance(else_type, UninhabitedType) else None
  6257	        return if_map, else_map

  6259	    def comparison_type_narrowing_helper(self, node: ComparisonExpr) -> tuple[TypeMap, TypeMap]:
  6260	        """Infer type narrowing from a comparison expression."""
  6261	        # Step 1: Obtain the types of each operand and whether or not we can
  6262	        # narrow their types. (For example, we shouldn't try narrowing the
  6263	        # types of literal string or enum expressions).
  6264
  6265	        operands = [collapse_walrus(x) for x in node.operands]
  6266	        operand_types = []
  6267	        narrowable_operand_index_to_hash = {}
  6268	        for i, expr in enumerate(operands):
  6269	            if not self.has_type(expr):
  6270	                return {}, {}
  6271	            expr_type = self.lookup_type(expr)
  6272	            operand_types.append(expr_type)
  6273
  6274	            if (
  6275	                literal(expr) == LITERAL_TYPE
  6276	                and not is_literal_none(expr)
  6277	                and not self.is_literal_enum(expr)
  6278	            ):
  6279	                h = literal_hash(expr)
  6280	                if h is not None:
  6281	                    narrowable_operand_index_to_hash[i] = h
  6282
  6283	        # Step 2: Group operands chained by either the 'is' or '==' operands
  6284	        # together. For all other operands, we keep them in groups of size 2.
  6285	        # So the expression:
  6286	        #
  6287	        #   x0 == x1 == x2 < x3 < x4 is x5 is x6 is not x7 is not x8
  6288	        #
  6289	        # ...is converted into the simplified operator list:
  6290	        #
  6291	        #  [("==", [0, 1, 2]), ("<", [2, 3]), ("<", [3, 4]),
  6292	        #   ("is", [4, 5, 6]), ("is not", [6, 7]), ("is not", [7, 8])]
  6293	        #
  6294	        # We group identity/equality expressions so we can propagate information
  6295	        # we discover about one operand across the entire chain. We don't bother
  6296	        # handling 'is not' and '!=' chains in a special way: those are very rare
  6297	        # in practice.
  6298
  6299	        simplified_operator_list = group_comparison_operands(
  6300	            node.pairwise(), narrowable_operand_index_to_hash, {"==", "is"}
  6301	        )
  6302
  6303	        # Step 3: Analyze each group and infer more precise type maps for each
  6304	        # assignable operand, if possible. We combine these type maps together
  6305	        # in the final step.
  6306
  6307	        partial_type_maps = []
  6308	        for operator, expr_indices in simplified_operator_list:
  6309	            if operator in {"is", "is not", "==", "!="}:
  6310	                if_map, else_map = self.equality_type_narrowing_helper(
  6311	                    node,
  6312	                    operator,
  6313	                    operands,
  6314	                    operand_types,
  6315	                    expr_indices,
  6316	                    narrowable_operand_index_to_hash,
  6317	                )
  6318	            elif operator in {"in", "not in"}:
  6319	                assert len(expr_indices) == 2
  6320	                left_index, right_index = expr_indices
  6321	                item_type = operand_types[left_index]
  6322	                iterable_type = operand_types[right_index]
  6323
  6324	                if_map, else_map = {}, {}
  6325
  6326	                if left_index in narrowable_operand_index_to_hash:
  6327	                    # We only try and narrow away 'None' for now
  6328	                    if is_overlapping_none(item_type):
  6329	                        collection_item_type = get_proper_type(builtin_item_type(iterable_type))
  6330	                        if (
  6331	                            collection_item_type is not None
  6332	                            and not is_overlapping_none(collection_item_type)
  6333	                            and not (
  6334	                                isinstance(collection_item_type, Instance)
  6335	                                and collection_item_type.type.fullname == "builtins.object"
  6336	                            )
  6337	                            and is_overlapping_erased_types(item_type, collection_item_type)
  6338	                        ):
  6339	                            if_map[operands[left_index]] = remove_optional(item_type)
  6340
  6341	                if right_index in narrowable_operand_index_to_hash:
  6342	                    if_type, else_type = self.conditional_types_for_iterable(
  6343	                        item_type, iterable_type
  6344	                    )
  6345	                    expr = operands[right_index]
  6346	                    if if_type is None:
  6347	                        if_map = None
  6348	                    else:
  6349	                        if_map[expr] = if_type
  6350	                    if else_type is None:
  6351	                        else_map = None
  6352	                    else:
  6353	                        else_map[expr] = else_type
  6354
  6355	            else:
  6356	                if_map = {}
  6357	                else_map = {}
  6358
  6359	            if operator in {"is not", "!=", "not in"}:
  6360	                if_map, else_map = else_map, if_map
  6361
  6362	            partial_type_maps.append((if_map, else_map))
  6363
  6364	        # If we have found non-trivial restrictions from the regular comparisons,
  6365	        # then return soon. Otherwise try to infer restrictions involving `len(x)`.
  6366	        # TODO: support regular and len() narrowing in the same chain.
  6367	        if any(m != ({}, {}) for m in partial_type_maps):
  6368	            return reduce_conditional_maps(partial_type_maps)
  6369	        else:
  6370	            # Use meet for `and` maps to get correct results for chained checks
  6371	            # like `if 1 < len(x) < 4: ...`
  6372	            return reduce_conditional_maps(self.find_tuple_len_narrowing(node), use_meet=True)

  6374	    def equality_type_narrowing_helper(
  6375	        self,
  6376	        node: ComparisonExpr,
  6377	        operator: str,
  6378	        operands: list[Expression],
  6379	        operand_types: list[Type],
  6380	        expr_indices: list[int],
  6381	        narrowable_operand_index_to_hash: dict[int, tuple[Key, ...]],
  6382	    ) -> tuple[TypeMap, TypeMap]:
  6383	        """Calculate type maps for '==', '!=', 'is' or 'is not' expression."""
  6384	        # is_valid_target:
  6385	        #   Controls which types we're allowed to narrow exprs to. Note that
  6386	        #   we cannot use 'is_literal_type_like' in both cases since doing
  6387	        #   'x = 10000 + 1; x is 10001' is not always True in all Python
  6388	        #   implementations.

  6390	        # coerce_only_in_literal_context:

  6392	        #   the provided exprs contains an explicit Literal type. This could
  6393	        #   technically be set to any arbitrary value, but it seems being liberal
  6394	        #   with narrowing when using 'is' and conservative when using '==' seems
  6395	        #   to break the least amount of real-world code.
  6396	        #
  6397	        # should_narrow_by_identity:

  6399	        #   that could cause identity-based narrowing to produce invalid results.

  6402	            coerce_only_in_literal_context = False

  6410	                return not custom_special_method(

  6412	                ) and not custom_special_method(t, "__ne__", check_all=False)

  6414	            is_valid_target = is_exactly_literal_type
  6415	            coerce_only_in_literal_context = True

  6417	            expr_types = [operand_types[i] for i in expr_indices]
  6418	            should_narrow_by_identity = all(
  6419	                map(has_no_custom_eq_checks, expr_types)
  6420	            ) and not is_ambiguous_mix_of_enums(expr_types)

  6422	        if_map: TypeMap = {}
  6423	        else_map: TypeMap = {}

  6428	                expr_indices,
  6429	                narrowable_operand_index_to_hash.keys(),
  6430	                is_valid_target,
  6431	                coerce_only_in_literal_context,
  6432	            )
  6433
  6434	        if if_map == {} and else_map == {}:
  6435	            if_map, else_map = self.refine_away_none_in_comparison(
  6436	                operands, operand_types, expr_indices, narrowable_operand_index_to_hash.keys()

  6445	    def propagate_up_typemap_info(self, new_types: TypeMap) -> TypeMap:
  6446	        """Attempts refining parent expressions of any MemberExpr or IndexExprs in new_types.
  6447
  6448	        Specifically, this function accepts two mappings of expression to original types:
  6449	        the original mapping (existing_types), and a new mapping (new_types) intended to
  6450	        update the original.
  6451
  6452	        This function iterates through new_types and attempts to use the information to try
  6453	        refining any parent types that happen to be unions.
  6454
  6455	        For example, suppose there are two types "A = Tuple[int, int]" and "B = Tuple[str, str]".
  6456	        Next, suppose that 'new_types' specifies the expression 'foo[0]' has a refined type
  6457	        of 'int' and that 'foo' was previously deduced to be of type Union[A, B].
  6458
  6459	        Then, this function will observe that since A[0] is an int and B[0] is not, the type of
  6460	        'foo' can be further refined from Union[A, B] into just B.
  6461
  6462	        We perform this kind of "parent narrowing" for member lookup expressions and indexing
  6463	        expressions into tuples, namedtuples, and typeddicts. We repeat this narrowing
  6464	        recursively if the parent is also a "lookup expression". So for example, if we have
  6465	        the expression "foo['bar'].baz[0]", we'd potentially end up refining types for the
  6466	        expressions "foo", "foo['bar']", and "foo['bar'].baz".
  6467
  6468	        We return the newly refined map. This map is guaranteed to be a superset of 'new_types'.
  6469	        """
  6470	        if new_types is None:
  6471	            return None
  6472	        output_map = {}
  6473	        for expr, expr_type in new_types.items():
  6474	            # The original inferred type should always be present in the output map, of course
  6475	            output_map[expr] = expr_type
  6476
  6477	            # Next, try using this information to refine the parent types, if applicable.
  6478	            new_mapping = self.refine_parent_types(expr, expr_type)
  6479	            for parent_expr, proposed_parent_type in new_mapping.items():
  6480	                # We don't try inferring anything if we've already inferred something for
  6481	                # the parent expression.
  6482	                # TODO: Consider picking the narrower type instead of always discarding this?
  6483	                if parent_expr in new_types:
  6484	                    continue
  6485	                output_map[parent_expr] = proposed_parent_type
  6486	        return output_map
  6487
  6488	    def refine_parent_types(self, expr: Expression, expr_type: Type) -> Mapping[Expression, Type]:
  6489	        """Checks if the given expr is a 'lookup operation' into a union and iteratively refines
  6490	        the parent types based on the 'expr_type'.
  6491
  6492	        For example, if 'expr' is an expression like 'a.b.c.d', we'll potentially return refined
  6493	        types for expressions 'a', 'a.b', and 'a.b.c'.
  6494
  6495	        For more details about what a 'lookup operation' is and how we use the expr_type to refine
  6496	        the parent types of lookup_expr, see the docstring in 'propagate_up_typemap_info'.
  6497	        """
  6498	        output: dict[Expression, Type] = {}
  6499
  6500	        # Note: parent_expr and parent_type are progressively refined as we crawl up the
  6501	        # parent lookup chain.
  6502	        while True:
  6503	            # First, check if this expression is one that's attempting to
  6504	            # "lookup" some key in the parent type. If so, save the parent type
  6505	            # and create function that will try replaying the same lookup
  6506	            # operation against arbitrary types.
  6507	            if isinstance(expr, MemberExpr):
  6508	                parent_expr = self._propagate_walrus_assignments(expr.expr, output)
  6509	                parent_type = self.lookup_type_or_none(parent_expr)
  6510	                member_name = expr.name
  6511
  6512	                def replay_lookup(new_parent_type: ProperType) -> Type | None:
  6513	                    with self.msg.filter_errors() as w:
  6514	                        member_type = analyze_member_access(
  6515	                            name=member_name,
  6516	                            typ=new_parent_type,
  6517	                            context=parent_expr,
  6518	                            is_lvalue=False,
  6519	                            is_super=False,
  6520	                            is_operator=False,
  6521
