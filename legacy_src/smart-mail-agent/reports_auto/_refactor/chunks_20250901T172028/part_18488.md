   376	def find_dataclass_transform_spec(node: Node | None) -> DataclassTransformSpec | None:
   377	    """
   378	    Find the dataclass transform spec for the given node, if any exists.
   379
   380	    Per PEP 681 (https://peps.python.org/pep-0681/#the-dataclass-transform-decorator), dataclass
   381	    transforms can be specified in multiple ways, including decorator functions and
   382	    metaclasses/base classes. This function resolves the spec from any of these variants.
   383	    """
   384
   385	    # The spec only lives on the function/class definition itself, so we need to unwrap down to that
   386	    # point
   387	    if isinstance(node, CallExpr):
   388	        # Like dataclasses.dataclass, transform-based decorators can be applied either with or
   389	        # without parameters; ie, both of these forms are accepted:
   390	        #
   391	        # @typing.dataclass_transform
   392	        # class Foo: ...
   393	        # @typing.dataclass_transform(eq=True, order=True, ...)
   394	        # class Bar: ...
   395	        #
   396	        # We need to unwrap the call for the second variant.
   397	        node = node.callee
   398
   399	    if isinstance(node, RefExpr):
   400	        node = node.node
   401
   402	    if isinstance(node, Decorator):
   403	        # typing.dataclass_transform usage must always result in a Decorator; it always uses the
   404	        # `@dataclass_transform(...)` syntax and never `@dataclass_transform`
   405	        node = node.func
   406
   407	    if isinstance(node, OverloadedFuncDef):
   408	        # The dataclass_transform decorator may be attached to any single overload, so we must
   409	        # search them all.
   410	        # Note that using more than one decorator is undefined behavior, so we can just take the
   411	        # first that we find.
   412	        for candidate in node.items:
   413	            spec = find_dataclass_transform_spec(candidate)
   414	            if spec is not None:
   415	                return spec
   416	        return find_dataclass_transform_spec(node.impl)
   417
   418	    # For functions, we can directly consult the AST field for the spec
   419	    if isinstance(node, FuncDef):
   420	        return node.dataclass_transform_spec
   421
   422	    if isinstance(node, ClassDef):
   423	        node = node.info
   424	    if isinstance(node, TypeInfo):
   425	        # Search all parent classes to see if any are decorated with `typing.dataclass_transform`
   426	        for base in node.mro[1:]:
   427	            if base.dataclass_transform_spec is not None:
   428	                return base.dataclass_transform_spec
   429
   430	        # Check if there is a metaclass that is decorated with `typing.dataclass_transform`
   431	        #
   432	        # Note that PEP 681 only discusses using a metaclass that is directly decorated with
   433	        # `typing.dataclass_transform`; subclasses thereof should be treated with dataclass
   434	        # semantics rather than as transforms:
   435	        #
   436	        # > If dataclass_transform is applied to a class, dataclass-like semantics will be assumed
   437	        # > for any class that directly or indirectly derives from the decorated class or uses the
   438	        # > decorated class as a metaclass.
   439	        #
   440	        # The wording doesn't make this entirely explicit, but Pyright (the reference
   441	        # implementation for this PEP) only handles directly-decorated metaclasses.
   442	        metaclass_type = node.metaclass_type
   443	        if metaclass_type is not None and metaclass_type.type.dataclass_transform_spec is not None:
   444	            return metaclass_type.type.dataclass_transform_spec
   445
   446	    return None
