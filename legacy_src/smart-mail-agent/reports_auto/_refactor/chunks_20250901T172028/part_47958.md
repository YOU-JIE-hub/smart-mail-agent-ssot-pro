    #
   409	    # If both types are singleton variants (and are not TypeVarLikes), we've hit the base case:
   410	    # we skip these checks to avoid infinitely recursing.
   411
   412	    def is_none_typevarlike_overlap(t1: Type, t2: Type) -> bool:
   413	        t1, t2 = get_proper_types((t1, t2))
   414	        return isinstance(t1, NoneType) and isinstance(t2, TypeVarLikeType)
   415
   416	    if prohibit_none_typevar_overlap:
   417	        if is_none_typevarlike_overlap(left, right) or is_none_typevarlike_overlap(right, left):
   418	            return False
   419
   420	    if (
   421	        len(left_possible) > 1
   422	        or len(right_possible) > 1
   423	        or isinstance(left, TypeVarLikeType)
   424	        or isinstance(right, TypeVarLikeType)
   425	    ):
   426	        for l in left_possible:
   427	            for r in right_possible:
   428	                if _is_overlapping_types(l, r):
   429	                    return True
   430	        return False
   431
   432	    # Now that we've finished handling TypeVarLikes, we're free to end early
   433	    # if one one of the types is None and we're running in strict-optional mode.
   434	    # (None only overlaps with None in strict-optional mode).
   435	    #
   436	    # We must perform this check after the TypeVarLike checks because
   437	    # a TypeVar could be bound to None, for example.
   438
   439	    if state.strict_optional and isinstance(left, NoneType) != isinstance(right, NoneType):
   440	        return False
   441
   442	    # Next, we handle single-variant types that may be inherently partially overlapping:
   443	    #
   444	    # - TypedDicts
   445	    # - Tuples
   446	    #
   447	    # If we cannot identify a partial overlap and end early, we degrade these two types
   448	    # into their 'Instance' fallbacks.
   449
   450	    if isinstance(left, TypedDictType) and isinstance(right, TypedDictType):
   451	        return are_typed_dicts_overlapping(left, right, _is_overlapping_types)
   452	    elif typed_dict_mapping_pair(left, right):
   453	        # Overlaps between TypedDicts and Mappings require dedicated logic.
   454	        return typed_dict_mapping_overlap(left, right, overlapping=_is_overlapping_types)
   455	    elif isinstance(left, TypedDictType):
   456	        left = left.fallback
   457	    elif isinstance(right, TypedDictType):
   458	        right = right.fallback
   459
   460	    if is_tuple(left) and is_tuple(right):
   461	        return are_tuples_overlapping(left, right, _is_overlapping_types)
   462	    elif isinstance(left, TupleType):
   463	        left = tuple_fallback(left)
   464	    elif isinstance(right, TupleType):
   465	        right = tuple_fallback(right)
   466
   467	    # Next, we handle single-variant types that cannot be inherently partially overlapping,
   468	    # but do require custom logic to inspect.
   469	    #
   470	    # As before, we degrade into 'Instance' whenever possible.
   471
   472	    if isinstance(left, TypeType) and isinstance(right, TypeType):
   473	        return _is_overlapping_types(left.item, right.item)
   474
   475	    def _type_object_overlap(left: Type, right: Type) -> bool:
   476	        """Special cases for type object types overlaps."""
   477	        # TODO: these checks are a bit in gray area, adjust if they cause problems.
   478	        left, right = get_proper_types((left, right))
   479	        # 1. Type[C] vs Callable[..., C] overlap even if the latter is not class object.
   480	        if isinstance(left, TypeType) and isinstance(right, CallableType):
   481	            return _is_overlapping_types(left.item, right.ret_type)
   482	        # 2. Type[C] vs Meta, where Meta is a metaclass for C.
   483	        if isinstance(left, TypeType) and isinstance(right, Instance):
   484	            if isinstance(left.item, Instance):
   485	                left_meta = left.item.type.metaclass_type
   486	                if left_meta is not None:
   487	                    return _is_overlapping_types(left_meta, right)
   488	                # builtins.type (default metaclass) overlaps with all metaclasses
   489	                return right.type.has_base("builtins.type")
   490	            elif isinstance(left.item, AnyType):
   491	                return right.type.has_base("builtins.type")
   492	        # 3. Callable[..., C] vs Meta is considered below, when we switch to fallbacks.
   493	        return False
   494
   495	    if isinstance(left, TypeType) or isinstance(right, TypeType):
   496	        return _type_object_overlap(left, right) or _type_object_overlap(right, left)
   497
   498	    if isinstance(left, Parameters) and isinstance(right, Parameters):
   499	        return are_parameters_compatible(
   500	            left,
   501	            right,
   502	            is_compat=_is_overlapping_types,
   503	            is_proper_subtype=False,
   504	            ignore_pos_arg_names=not overlap_for_overloads,
   505	            allow_partial_overlap=True,
   506	        )
   507	    # A `Parameters` does not overlap with anything else, however
   508	    if isinstance(left, Parameters) or isinstance(right, Parameters):
   509	        return False
   510
   511	    if isinstance(left, CallableType) and isinstance(right, CallableType):
   512	        return is_callable_compatible(
   513	            left,
   514	            right,
   515	            is_compat=_is_overlapping_types,
   516	            is_proper_subtype=False,
   517	            ignore_pos_arg_names=not overlap_for_overloads,
   518	            allow_partial_overlap=True,
   519	        )
   520
   521	    call = None
   522	    other = None
   523	    if isinstance(left, CallableType) and isinstance(right, Instance):
   524	        call = find_member("__call__", right, right, is_operator=True)
   525	        other = left
   526	    if isinstance(right, CallableType) and isinstance(left, Instance):
   527	        call = find_member("__call__", left, left, is_operator=True)
   528	        other = right
   529	    if isinstance(get_proper_type(call), FunctionLike):
   530	        assert call is not None and other is not None
   531	        return _is_overlapping_types(call, other)
   532
   533	    if isinstance(left, CallableType):
   534	        left = left.fallback
   535	    if isinstance(right, CallableType):
   536	        right = right.fallback
   537
   538	    if isinstance(left, LiteralType) and isinstance(right, LiteralType):
   539	        if left.value == right.value:
   540	            # If values are the same, we still need to check if fallbacks are overlapping,
   541	            # this is done below.
   542	            left = left.fallback
   543	            right = right.fallback
   544	        else:
   545	            return False
   546	    elif isinstance(left, LiteralType):
   547	        left = left.fallback
   548	    elif isinstance(right, LiteralType):
   549	        right = right.fallback
   550
   551	    # Finally, we handle the case where left and right are instances.
   552
   553	    if isinstance(left, Instance) and isinstance(right, Instance):
   554	        # First we need to handle promotions and structural compatibility for instances
   555	        # that came as fallbacks, so simply call is_subtype() to avoid code duplication.
   556	        if _is_subtype(left, right) or _is_subtype(right, left):
   557	            return True
   558
   559	        if right.type.fullname == "builtins.int" and left.type.fullname in MYPYC_NATIVE_INT_NAMES:
   560	            return True
   561
   562	        # Two unrelated types cannot be partially overlapping: they're disjoint.
   563	        if left.type.has_base(right.type.fullname):
   564	            left = map_instance_to_supertype(left, right.type)
   565	        elif right.type.has_base(left.type.fullname):
   566	            right = map_instance_to_supertype(right, left.type)
   567	        else:
   568	            return False
   569
   570	        if right.type.has_type_var_tuple_type:
   571	            # Similar to subtyping, we delegate the heavy lifting to the tuple overlap.
   572	            assert right.type.type_var_tuple_prefix is not None
   573	            assert right.type.type_var_tuple_suffix is not None
   574	            prefix = right.type.type_var_tuple_prefix
   575	            suffix = right.type.type_var_tuple_suffix
   576	            tvt = right.type.defn.type_vars[prefix]
   577	            assert isinstance(tvt, TypeVarTupleType)
   578	            fallback = tvt.tuple_fallback
   579	            left_prefix, left_middle, left_suffix = split_with_prefix_and_suffix(
   580	                left.args, prefix, suffix
   581	            )
   582	            right_prefix, right_middle, right_suffix = split_with_prefix_and_suffix(
   583	                right.args, prefix, suffix
   584	            )
   585	            left_args = left_prefix + (TupleType(list(left_middle), fallback),) + left_suffix
   586	            right_args = right_prefix + (TupleType(list(right_middle), fallback),) + right_suffix
   587	        else:
   588	            left_args = left.args
   589	            right_args = right.args
   590	        if len(left_args) == len(right_args):
   591	            # Note: we don't really care about variance here, since the overlapping check
   592	            # is symmetric and since we want to return 'True' even for partial overlaps.
   593	            #
   594	            # For example, suppose we have two types Wrapper[Parent] and Wrapper[Child].
   595	            # It doesn't matter whether Wrapper is covariant or contravariant since
   596	            # either way, one of the two types will overlap with the other.
   597	            #
   598	            # Similarly, if Wrapper was invariant, the two types could still be partially
   599	            # overlapping -- what if Wrapper[Parent] happened to contain only instances of
   600	            # specifically Child?
   601	            #
   602	            # Or, to use a more concrete example, List[Union[A, B]] and List[Union[B, C]]
   603	            # would be considered partially overlapping since it's possible for both lists
   604	            # to contain only instances of B at runtime.
   605	            if all(
   606	                _is_overlapping_types(left_arg, right_arg)
   607	                for left_arg, right_arg in zip(left_args, right_args)
   608	            ):
   609	                return True
   610
   611	        return False
   612
   613	    # We ought to have handled every case by now: we conclude the
   614	    # two types are not overlapping, either completely or partially.
   615	    #
   616	    # Note: it's unclear however, whether returning False is the right thing
   617	    # to do when inferring reachability -- see  https://github.com/python/mypy/issues/5529
   618
   619	    assert type(left) != type(right), f"{type(left)} vs {type(right)}"
   620	    return False

   635	def are_typed_dicts_overlapping(
   636	    left: TypedDictType, right: TypedDictType, is_overlapping: Callable[[Type, Type], bool]
   637	) -> bool:
   638	    """Returns 'true' if left and right are overlapping TypeDictTypes."""
   639	    # All required keys in left are present and overlapping with something in right
   640	    for key in left.required_keys:
   641	        if key not in right.items:
   642	            return False
   643	        if not is_overlapping(left.items[key], right.items[key]):
   644	            return False
   645
   646	    # Repeat check in the other direction
   647	    for key in right.required_keys:
   648	        if key not in left.items:
   649	            return False
   650	        if not is_overlapping(left.items[key], right.items[key]):
   651	            return False
   652
   653	    # The presence of any additional optional keys does not affect whether the two
   654	    # TypedDicts are partially overlapping: the dicts would be overlapping if the
   655	    # keys happened to be missing.
   656	    return True

   659	def are_tuples_overlapping(
   660	    left: Type, right: Type, is_overlapping: Callable[[Type, Type], bool]
   661	) -> bool:
   662	    """Returns true if left and right are overlapping tuples."""
   663	    left, right = get_proper_types((left, right))
   664	    left = adjust_tuple(left, right) or left
   665	    right = adjust_tuple(right, left) or right
   666	    assert isinstance(left, TupleType), f"Type {left} is not a tuple"
   667	    assert isinstance(right, TupleType), f"Type {right} is not a tuple"
   668
   669	    # This algorithm works well if only one tuple is variadic, if both are
   670	    # variadic we may get rare false negatives for overlapping prefix/suffix.
   671	    # Also, this ignores empty unpack case, but it is probably consistent with
   672	    # how we handle e.g. empty lists in overload overlaps.
   673	    # TODO: write a more robust algorithm for cases where both types are variadic.
   674	    left_unpack = find_unpack_in_list(left.items)
   675	    right_unpack = find_unpack_in_list(right.items)
   676	    if left_unpack is not None:
   677	        left = expand_tuple_if_possible(left, len(right.items))
   678	    if right_unpack is not None:
   679	        right = expand_tuple_if_possible(right, len(left.items))
   680
   681	    if len(left.items) != len(right.items):
   682	        return False
   683	    if not all(is_overlapping(l, r) for l, r in zip(left.items, right.items)):
   684	        return False
   685
   686	    # Check that the tuples aren't from e.g. different NamedTuples.
   687	    if is_named_instance(right.partial_fallback, "builtins.tuple") or is_named_instance(
   688	        left.partial_fallback, "builtins.tuple"
   689	    ):
   690	        return True
   691	    else:
   692	        return is_overlapping(left.partial_fallback, right.partial_fallback)
   693

   695	def expand_tuple_if_possible(tup: TupleType, target: int) -> TupleType:
   696	    if len(tup.items) > target + 1:
   697	        return tup
   698	    extra = target + 1 - len(tup.items)
   699	    new_items = []
   700	    for it in tup.items:
   701	        if not isinstance(it, UnpackType):
   702	            new_items.append(it)
   703	            continue
   704	        unpacked = get_proper_type(it.type)
   705	        if isinstance(unpacked, TypeVarTupleType):
   706	            instance = unpacked.tuple_fallback
   707	        else:
   708	            # Nested non-variadic tuples should be normalized at this point.
   709	            assert isinstance(unpacked, Instance)
   710	            instance = unpacked
   711	        assert instance.type.fullname == "builtins.tuple"
   712	        new_items.extend([instance.args[0]] * extra)
   713	    return tup.copy_modified(items=new_items)

   716	def adjust_tuple(left: ProperType, r: ProperType) -> TupleType | None:
   717	    """Find out if `left` is a Tuple[A, ...], and adjust its length to `right`"""
   718	    if isinstance(left, Instance) and left.type.fullname == "builtins.tuple":
   719	        n = r.length() if isinstance(r, TupleType) else 1
   720	        return TupleType([left.args[0]] * n, left)
   721	    return None

   724	def is_tuple(typ: Type) -> bool:
   725	    typ = get_proper_type(typ)
   726	    return isinstance(typ, TupleType) or (
   727	        isinstance(typ, Instance) and typ.type.fullname == "builtins.tuple"
   728	    )

   731	class TypeMeetVisitor(TypeVisitor[ProperType]):
   732	    def __init__(self, s: ProperType) -> None:
   733	        self.s = s
   734
   735	    def visit_unbound_type(self, t: UnboundType) -> ProperType:
   736	        if isinstance(self.s, NoneType):
   737	            if state.strict_optional:
   738	                return UninhabitedType()
   739	            else:
   740	                return self.s
   741	        elif isinstance(self.s, UninhabitedType):
   742	            return self.s
   743	        else:
   744	            return AnyType(TypeOfAny.special_form)
   745
   746	    def visit_any(self, t: AnyType) -> ProperType:
   747	        return self.s
   748
   749	    def visit_union_type(self, t: UnionType) -> ProperType:
   750	        if isinstance(self.s, UnionType):
   751	            meets: list[Type] = []
   752	            for x in t.items:
   753	                for y in self.s.items:
   754	                    meets.append(meet_types(x, y))
   755	        else:
   756	            meets = [meet_types(x, self.s) for x in t.items]
   757	        return make_simplified_union(meets)
   758
   759	    def visit_none_type(self, t: NoneType) -> ProperType:
   760	        if state.strict_optional:
   761	            if isinstance(self.s, NoneType) or (
   762	                isinstance(self.s, Instance) and self.s.type.fullname == "builtins.object"
   763	            ):
   764	                return t
   765	            else:
   766	                return UninhabitedType()
   767	        else:
   768	            return t
   769
   770	    def visit_uninhabited_type(self, t: UninhabitedType) -> ProperType:
   771	        return t
   772
   773	    def visit_deleted_type(self, t: DeletedType) -> ProperType:
   774	        if isinstance(self.s, NoneType):
   775	            if state.strict_optional:
   776	                return t
   777	            else:
   778	                return self.s
   779	        elif isinstance(self.s, UninhabitedType):
   780	            return self.s
   781	        else:
   782	            return t
   783
   784	    def visit_erased_type(self, t: ErasedType) -> ProperType:
   785	        return self.s
   786
   787	    def visit_type_var(self, t: TypeVarType) -> ProperType:
   788	        if isinstance(self.s, TypeVarType) and self.s.id == t.id:
   789	            if self.s.upper_bound == t.upper_bound:
   790	                return self.s
   791	            return self.s.copy_modified(upper_bound=self.meet(self.s.upper_bound, t.upper_bound))
   792	        else:
   793	            return self.default(self.s)
   794
   795	    def visit_param_spec(self, t: ParamSpecType) -> ProperType:
   796	        if self.s == t:
   797	            return self.s
   798	        else:
   799	            return self.default(self.s)
   800
   801	    def visit_type_var_tuple(self, t: TypeVarTupleType) -> ProperType:
   802	        if isinstance(self.s, TypeVarTupleType) and self.s.id == t.id:
   803	            return self.s if self.s.min_len > t.min_len else t
   804	        else:
   805	            return self.default(self.s)
   806
   807	    def visit_unpack_type(self, t: UnpackType) -> ProperType:
   808	        raise NotImplementedError
   809
   810	    def visit_parameters(self, t: Parameters) -> ProperType:
   811	        if isinstance(self.s, Parameters):
   812	            if len(t.arg_types) != len(self.s.arg_types):
   813	                return self.default(self.s)
   814	            from mypy.join import join_types
   815
   816	            return t.copy_modified(
   817	                arg_types=[join_types(s_a, t_a) for s_a, t_a in zip(self.s.arg_types, t.arg_types)]
   818	            )
   819	        else:
   820	            return self.default(self.s)
   821
   822	    def visit_instance(self, t: Instance) -> ProperType:
   823	        if isinstance(self.s, Instance):
   824	            if t.type == self.s.type:
   825	                if is_subtype(t, self.s) or is_subtype(self.s, t):
   826	                    # Combine type arguments. We could have used join below
   827	                    # equivalently.
   828	                    args: list[Type] = []
   829	                    # N.B: We use zip instead of indexing because the lengths might have
   830	                    # mismatches during daemon reprocessing.
   831	                    if t.type.has_type_var_tuple_type:
   832
