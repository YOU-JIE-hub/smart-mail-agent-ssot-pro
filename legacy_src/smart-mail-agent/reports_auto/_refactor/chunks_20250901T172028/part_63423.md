es isn't specified, make it the same as the set
   208	            # of modules with a stale public interface.
   209	            rechecked_modules[passnum] = stale_modules[passnum]
   210	        if (
   211	            passnum in stale_modules
   212	            and passnum in rechecked_modules
   213	            and not stale_modules[passnum].issubset(rechecked_modules[passnum])
   214	        ):
   215	            _case_fail(f"Stale modules after pass {passnum} must be a subset of rechecked modules")
   216
   217	    output_inline_start = len(output)
   218	    input = first_item.data
   219	    expand_errors(input, output, "main")
   220	    for file_path, contents in files:
   221	        expand_errors(contents.split("\n"), output, file_path)
   222
   223	    seen_files = set()
   224	    for file, _ in files:
   225	        if file in seen_files:
   226	            _case_fail(f"Duplicated filename {file}. Did you include it multiple times?")
   227
   228	        seen_files.add(file)
   229
   230	    case.input = input
   231	    case.output = output
   232	    case.output_inline_start = output_inline_start
   233	    case.output2 = output2
   234	    case.last_line = case.line + item.line + len(item.data) - 2
   235	    case.files = files
   236	    case.output_files = output_files
   237	    case.expected_stale_modules = stale_modules
   238	    case.expected_rechecked_modules = rechecked_modules
   239	    case.deleted_paths = deleted_paths
   240	    case.triggered = triggered or []
   241	    case.expected_fine_grained_targets = targets
   242	    case.test_modules = test_modules

   338	        parse_test_case(case=self)
   339	        self.old_cwd = os.getcwd()

   341	        os.chdir(self.tmpdir)
   342	        os.mkdir(test_temp_dir)

   344	        # Precalculate steps for find_steps()
   345	        steps: dict[int, list[FileOperation]] = {}

   347	        for path, content in self.files:
   348	            m = re.match(r".*\.([0-9]+)$", path)
   349	            if m:
   350	                # Skip writing subsequent incremental steps - rather
   351	                # store them as operations.
   352	                num = int(m.group(1))
   353	                assert num >= 2
   354	                target_path = re.sub(r"\.[0-9]+$", "", path)
   355	                module = module_from_path(target_path)
   356	                operation = UpdateFile(module, content, target_path)
   357	                steps.setdefault(num, []).append(operation)
   358	            else:
   359	                # Write the first incremental steps
   360	                dir = os.path.dirname(path)
   361	                os.makedirs(dir, exist_ok=True)
   362	                with open(path, "w", encoding="utf8") as f:
   363	                    f.write(content)

   365	        for num, paths in self.deleted_paths.items():
   366	            assert num >= 2
   367	            for path in paths:
   368	                module = module_from_path(path)
   369	                steps.setdefault(num, []).append(DeleteFile(module, path))
   370	        max_step = max(steps) if steps else 2
   371	        self.steps = [steps.get(num, []) for num in range(2, max_step + 1)]

   375	            os.chdir(self.old_cwd)
   376	        if self.tmpdir is not None:
   377	            shutil.rmtree(self.tmpdir, ignore_errors=True)
   378	        self.old_cwd = None
   379	        self.tmpdir = None

   384	    def repr_failure(
   385	        self, excinfo: pytest.ExceptionInfo[BaseException], style: Any | None = None
   386	    ) -> str:
   387	        excrepr: object
   388	        if isinstance(excinfo.value, SystemExit):
   389	            # We assume that before doing exit() (which raises SystemExit) we've printed
   390	            # enough context about what happened so that a stack trace is not useful.
   391	            # In particular, uncaught exceptions during semantic analysis or type checking
   392	            # call exit() and they already print out a stack trace.
   393	            excrepr = excinfo.exconly()
   394	        elif isinstance(excinfo.value, pytest.fail.Exception) and not excinfo.value.pytrace:
   395	            excrepr = excinfo.exconly()
   396	        else:
   397	            excinfo.traceback = self.parent._traceback_filter(excinfo)
   398	            excrepr = excinfo.getrepr(style="short")
   399
   400	        return f"data: {self.file}:{self.line}:\n{excrepr}"

   406	        second step, etc. Each operation can either be a file modification/creation (UpdateFile)

   414	def module_from_path(path: str) -> str:
   415	    path = re.sub(r"\.pyi?$", "", path)
   416	    # We can have a mix of Unix-style and Windows-style separators.
   417	    parts = re.split(r"[/\\]", path)
   418	    del parts[0]
   419	    module = ".".join(parts)
   420	    module = re.sub(r"\.__init__$", "", module)
   421	    return module

   424	@dataclass
   425	class TestItem:
   426	    """Parsed test caseitem.
   427
   428	    An item is of the form
   429	      [id arg]
   430	      .. data ..
   431	    """
   432
   433	    id: str
   434	    arg: str | None
   435	    # Processed, collapsed text data
   436	    data: list[str]
   437	    # Start line: 1-based, inclusive, relative to testcase
   438	    line: int
   439	    # End line: 1-based, exclusive, relative to testcase; not same as `line + len(test_item.data)` due to collapsing
   440	    end_line: int
   441
   442	    @property
   443	    def trimmed_newlines(self) -> int:  # compensates for strip_list
   444	        return self.end_line - self.line - len(self.data)

   446
   447	def parse_test_data(raw_data: str, name: str) -> list[TestItem]:
   448	    """Parse a list of lines that represent a sequence of test items."""
   449
   450	    lines = ["", "[case " + name + "]"] + raw_data.split("\n")
   451	    ret: list[TestItem] = []
   452	    data: list[str] = []
   453
   454	    id: str | None = None
   455	    arg: str | None = None
   456
   457	    i = 0
   458	    i0 = 0
   459	    while i < len(lines):
   460	        s = lines[i].strip()
   461
   462	        if lines[i].startswith("[") and s.endswith("]"):
   463	            if id:
   464	                data = collapse_line_continuation(data)
   465	                data = strip_list(data)
   466	                ret.append(TestItem(id, arg, data, i0 + 1, i))
   467
   468	            i0 = i
   469	            id = s[1:-1]
   470	            arg = None
   471	            if " " in id:
   472	                arg = id[id.index(" ") + 1 :]
   473	                id = id[: id.index(" ")]
   474	            data = []
   475	        elif lines[i].startswith("\\["):
   476	            data.append(lines[i][1:])
   477	        elif not lines[i].startswith("--"):
   478	            data.append(lines[i])
   479	        elif lines[i].startswith("----"):
   480	            data.append(lines[i][2:])
   481	        i += 1
   482
   483	    # Process the last item.
   484	    if id:
   485	        data = collapse_line_continuation(data)
   486	        data = strip_list(data)
   487	        ret.append(TestItem(id, arg, data, i0 + 1, i - 1))
   488
   489	    return ret

   492	def strip_list(l: list[str]) -> list[str]:
   493	    """Return a stripped copy of l.
   494
   495	    Strip whitespace at the end of all lines, and strip all empty
   496	    lines from the end of the array.
   497	    """
   498
   499	    r: list[str] = []
   500	    for s in l:
   501	        # Strip spaces at end of line
   502	        r.append(re.sub(r"\s+$", "", s))
   503
   504	    while r and r[-1] == "":
   505	        r.pop()
   506
   507	    return r

   510	def collapse_line_continuation(l: list[str]) -> list[str]:
   511	    r: list[str] = []
   512	    cont = False
   513	    for s in l:
   514	        ss = re.sub(r"\\$", "", s)
   515	        if cont:
   516	            r[-1] += re.sub("^ +", "", ss)
   517	        else:
   518	            r.append(ss)
   519	        cont = s.endswith("\\")
   520	    return r

   527	def expand_errors(input: list[str], output: list[str], fnam: str) -> None:
   528	    """Transform comments such as '# E: message' or
   529	    '# E:3: message' in input.
   530
   531	    The result is lines like 'fnam:line: error: message'.
   532	    """
   533
   534	    for i in range(len(input)):
   535	        # The first in the split things isn't a comment
   536	        for possible_err_comment in input[i].split(" # ")[1:]:
   537	            m = re.search(
   538	                r"^([ENW]):((?P<col>\d+):)? (?P<message>.*)$", possible_err_comment.strip()
   539	            )
   540	            if m:
   541	                if m.group(1) == "E":
   542	                    severity = "error"
   543	                elif m.group(1) == "N":
   544	                    severity = "note"
   545	                elif m.group(1) == "W":
   546	                    severity = "warning"
   547	                col = m.group("col")
   548	                message = m.group("message")
   549	                message = message.replace("\\#", "#")  # adds back escaped # character
   550	                if col is None:
   551	                    output.append(f"{fnam}:{i + 1}: {severity}: {message}")
   552	                else:
   553	                    output.append(f"{fnam}:{i + 1}:{col}: {severity}: {message}")

   556	def fix_win_path(line: str) -> str:
   557	    r"""Changes Windows paths to Linux paths in error messages.
   558
   559	    E.g. foo\bar.py -> foo/bar.py.
   560	    """
   561	    line = line.replace(root_dir, root_dir.replace("\\", "/"))
   562	    m = re.match(r"^([\S/]+):(\d+:)?(\s+.*)", line)
   563	    if not m:
   564	        return line
   565	    else:
   566	        filename, lineno, message = m.groups()
   567	        return "{}:{}{}".format(filename.replace("\\", "/"), lineno or "", message)

   571	    r"""Changes filename paths to Linux paths in Cobertura output files.
   572
   573	    E.g. filename="pkg\subpkg\a.py" -> filename="pkg/subpkg/a.py".

   667	def split_test_cases(
   668	    parent: DataFileCollector, suite: DataSuite, file: str
   669	) -> Iterator[DataDrivenTestCase]:
   670	    """Iterate over raw test cases in file, at collection time, ignoring sub items.
   671
   672	    The collection phase is slow, so any heavy processing should be deferred to after
   673	    uninteresting tests are filtered (when using -k PATTERN switch).
   674	    """
   675	    with open(file, encoding="utf-8") as f:
   676	        data = f.read()
   677	    cases = re.split(r"^\[case ([^]+)]+)\][ \t]*$\n", data, flags=re.DOTALL | re.MULTILINE)
   678	    cases_iter = iter(cases)
   679	    line_no = next(cases_iter).count("\n") + 1
   680	    test_names = set()
   681	    for case_id in cases_iter:
   682	        data = next(cases_iter)
   683
   684	        m = _case_name_pattern.fullmatch(case_id)
   685	        if not m:
   686	            raise RuntimeError(f"Invalid testcase id {case_id!r}")
   687	        name = m.group("name")
   688	        if name in test_names:
   689	            raise RuntimeError(
   690	                'Found a duplicate test name "{}" in {} on line {}'.format(
   691	                    name, parent.name, line_no
   692	                )
   693	            )
   694	        yield DataDrivenTestCase.from_parent(
   695	            parent=parent,
   696	            suite=suite,
   697	            file=file,
   698	            name=add_test_name_suffix(name, suite.test_name_suffix),
   699	            writescache=bool(m.group("writescache")),
   700	            only_when=m.group("only_when"),
   701	            platform=m.group("platform"),
   702	            skip=bool(m.group("skip")),
   703	            xfail=bool(
