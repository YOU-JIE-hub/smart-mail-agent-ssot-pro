   746	    def _get_link_info_dict(self, obj):
   747	        #We do not want None or False as the link, even if it is the
   748	        #attribute's value - use the empty string instead.
   749	        url = getattr(obj, "hrefURL", "") or ""
   750	        title = getattr(obj, "hrefTitle", "") or ""
   751	        if url :
   752	            #Is it valid to have a link with no href?  The XML requires
   753	            #the xlink:href to be present, but you might just want a
   754	            #tool tip shown (via the xlink:title attribute).  Note that
   755	            #giving an href of "" is equivalent to "the current page"
   756	            #(a relative link saying go nowhere).
   757	            return {"xlink:href":url, "xlink:title":title, "target":"_top"}
   758	            #Currently of all the mainstream browsers I have tested, only Safari/webkit
   759	            #will show  SVG images embedded in HTML using a simple <img src="..." /> tag.
   760	            #However, the links don't work (Safari 3.2.1 on the Mac).
   761	            #
   762	            #Therefore I use the following, which also works for Firefox, Opera, and
   763	            #IE 6.0 with Adobe SVG Viewer 6 beta:
   764	            #<object data="..." type="image/svg+xml" width="430" height="150" class="img">
   765	            #
   766	            #Once displayed, Firefox and Safari treat the SVG like a frame, and
   767	            #by default clicking on links acts "in frame" and replaces the image.
   768	            #Opera does what I expect, and replaces the whole page with the link.
   769	            #
   770	            #Therefore I use target="_top" to force the links to replace the whole page.
   771	            #This now works as expected on Safari 3.2.1, Firefox 3.0.6, Opera 9.20.
   772	            #Perhaps the target attribute should be an option, perhaps defaulting to
   773	            #"_top" as used here?
   774	        else :
   775	            return None
