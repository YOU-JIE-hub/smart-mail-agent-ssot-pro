   309	# None as a sub means disallowed.
   310	G2RX_TOKENS = [(re.compile(rx), sub) for rx, sub in [
   311	    (r"\*\*\*+", None),             # Can't have ***
   312	    (r"[^/]+\*\*+", None),          # Can't have x**
   313	    (r"\*\*+[^/]+", None),          # Can't have **x
   314	    (r"\*\*/\*\*", None),           # Can't have **/**
   315	    (r"^\*+/", r"(.*[/\\\\])?"),    # ^*/ matches any prefix-slash, or nothing.
   316	    (r"/\*+$", r"[/\\\\].*"),       # /*$ matches any slash-suffix.
   317	    (r"\*\*/", r"(.*[/\\\\])?"),    # **/ matches any subdirs, including none
   318	    (r"/", r"[/\\\\]"),             # / matches either slash or backslash
   319	    (r"\*", r"[^/\\\\]*"),          # * matches any number of non slash-likes
   320	    (r"\?", r"[^/\\\\]"),           # ? matches one non slash-like
   321	    (r"\[.*?\]", r"\g<0>"),         # [a-f] matches [a-f]
   322	    (r"[a-zA-Z0-9_-]+", r"\g<0>"),  # word chars match themselves
   323	    (r"[\[\]]", None),              # Can't have single square brackets
   324	    (r".", r"\\\g<0>"),             # Anything else is escaped to be safe

   327	def _glob_to_regex(pattern: str) -> str:
   328	    """Convert a file-path glob pattern into a regex."""
   329	    # Turn all backslashes into slashes to simplify the tokenizer.
   330	    pattern = pattern.replace("\\", "/")
   331	    if "/" not in pattern:
   332	        pattern = f"**/{pattern}"
   333	    path_rx = []
   334	    pos = 0
   335	    while pos < len(pattern):
   336	        for rx, sub in G2RX_TOKENS:                     # pragma: always breaks
   337	            if m := rx.match(pattern, pos=pos):
   338	                if sub is None:
   339	                    raise ConfigError(f"File pattern can't include {m[0]!r}")
   340	                path_rx.append(m.expand(sub))
   341	                pos = m.end()
   342	                break
   343	    return "".join(path_rx)
