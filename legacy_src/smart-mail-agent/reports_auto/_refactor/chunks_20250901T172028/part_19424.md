   150	def _getBezierExtrema(y0,y1,y2,y3):
   151	    '''
   152	    this is used to find if a curveTo path operator has extrema in its range
   153	    The curveTo operator is defined by the points y0, y1, y2, y3
   154
   155	        B(t):=(1-t)^3*y0+3*(1-t)^2*t*y1+3*(1-t)*t^2*y2+t^3*y3
   156	            :=t^3*(y3-3*y2+3*y1-y0)+t^2*(3*y2-6*y1+3*y0)+t*(3*y1-3*y0)+y0
   157	    and is a cubic bezier curve.
   158
   159	    The differential is a quadratic
   160	        t^2*(3*y3-9*y2+9*y1-3*y0)+t*(6*y2-12*y1+6*y0)+3*y1-3*y0
   161
   162	    The extrema must be at real roots, r, of the above which lie in 0<=r<=1
   163
   164	    The quadratic coefficients are
   165	        a=3*y3-9*y2+9*y1-3*y0 b=6*y2-12*y1+6*y0 c=3*y1-3*y0
   166	    or
   167	        a=y3-3*y2+3*y1-y0 b=2*y2-4*y1+2*y0 c=y1-y0  (remove common factor of 3)
   168	    or
   169	        a=y3-3*(y2-y1)-y0 b=2*(y2-2*y1+y0) c=y1-y0
   170
   171	    The returned value is [y0,x1,x2,y3] where if found x1, x2 are any extremals that were found;
   172	    there can be 0, 1 or 2 extremals
   173	    '''
   174	    a=y3-3*(y2-y1)-y0
   175	    b=2*(y2-2*y1+y0)
   176	    c=y1-y0
   177	    Y = [y0] #the set of points
   178
   179	    #standard method to find roots of quadratic
   180	    d = b*b - 4*a*c
   181	    if d>=0:
   182	        d = sqrt(d)
   183	        if b<0: d = -d
   184	        q = -0.5*(b+d)
   185	        R = []
   186	        try:
   187	            R.append(q/a)
   188	        except:
   189	            pass
   190	        try:
   191	            R.append(c/q)
   192	        except:
   193	            pass
   194	        b *= 1.5
   195	        c *= 3
   196	        for t in R:
   197	            if 0<=t<=1:
   198	                #real root in range evaluate spline there and add to X
   199	                Y.append(t*(t*(t*a+b)+c)+y0)
   200	    Y.append(y3)
   201	    return Y

   503	    '''return a repr style string with named fixed args first, then keywords'''
   504	    if isinstance(self,float):
   505	        return fp_str(self)
   506	    elif isSeq(self):
   507	        s = ''
   508	        for v in self:
