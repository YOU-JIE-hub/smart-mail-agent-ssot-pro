   495	    def _force_enable_logging(
   496	        self, level: int | str, logger_obj: logging.Logger
   497	    ) -> int:
   498	        """Enable the desired logging level if the global level was disabled via ``logging.disabled``.
   499
   500	        Only enables logging levels greater than or equal to the requested ``level``.
   501
   502	        Does nothing if the desired ``level`` wasn't disabled.
   503
   504	        :param level:
   505	            The logger level caplog should capture.
   506	            All logging is enabled if a non-standard logging level string is supplied.
   507	            Valid level strings are in :data:`logging._nameToLevel`.
   508	        :param logger_obj: The logger object to check.
   509
   510	        :return: The original disabled logging level.
   511	        """
   512	        original_disable_level: int = logger_obj.manager.disable
   513
   514	        if isinstance(level, str):
   515	            # Try to translate the level string to an int for `logging.disable()`
   516	            level = logging.getLevelName(level)
   517
   518	        if not isinstance(level, int):
   519	            # The level provided was not valid, so just un-disable all logging.
   520	            logging.disable(logging.NOTSET)
   521	        elif not logger_obj.isEnabledFor(level):
   522	            # Each level is `10` away from other levels.
   523	            # https://docs.python.org/3/library/logging.html#logging-levels
   524	            disable_level = max(level - 10, logging.NOTSET)
   525	            logging.disable(disable_level)
   526
   527	        return original_disable_level

   529	    def set_level(self, level: int | str, logger: str | None = None) -> None:
   530	        """Set the threshold level of a logger for the duration of a test.
   531
   532	        Logging messages which are less severe than this level will not be captured.
   533
   534	        .. versionchanged:: 3.4
   535	            The levels of the loggers changed by this function will be
   536	            restored to their initial values at the end of the test.
   537
   538	        Will enable the requested logging level if it was disabled via :func:`logging.disable`.
   539
   540	        :param level: The level.
   541	        :param logger: The logger to update. If not given, the root logger.
   542	        """
   543	        logger_obj = logging.getLogger(logger)
   544	        # Save the original log-level to restore it during teardown.
   545	        self._initial_logger_levels.setdefault(logger, logger_obj.level)
   546	        logger_obj.setLevel(level)
   547	        if self._initial_handler_level is None:
   548	            self._initial_handler_level = self.handler.level
   549	        self.handler.setLevel(level)
   550	        initial_disabled_logging_level = self._force_enable_logging(level, logger_obj)
   551	        if self._initial_disabled_logging_level is None:
   552	            self._initial_disabled_logging_level = initial_disabled_logging_level

   554	    @contextmanager
   555	    def at_level(self, level: int | str, logger: str | None = None) -> Generator[None]:
   556	        """Context manager that sets the level for capturing of logs. After
   557	        the end of the 'with' statement the level is restored to its original
   558	        value.
   559
   560	        Will enable the requested logging level if it was disabled via :func:`logging.disable`.
   561
   562	        :param level: The level.
   563	        :param logger: The logger to update. If not given, the root logger.
   564	        """
   565	        logger_obj = logging.getLogger(logger)
   566	        orig_level = logger_obj.level
   567	        logger_obj.setLevel(level)
   568	        handler_orig_level = self.handler.level
   569	        self.handler.setLevel(level)
   570	        original_disable_level = self._force_enable_logging(level, logger_obj)
   571	        try:
   572	            yield
   573	        finally:
   574	            logger_obj.setLevel(orig_level)
   575	            self.handler.setLevel(handler_orig_level)
   576	            logging.disable(original_disable_level)

   578	    @contextmanager
   579	    def filtering(self, filter_: logging.Filter) -> Generator[None]:
   580	        """Context manager that temporarily adds the given filter to the caplog's
   581	        :meth:`handler` for the 'with' statement block, and removes that filter at the
   582	        end of the block.
   583
   584	        :param filter_: A custom :class:`logging.Filter` object.
   585
   586	        .. versionadded:: 7.5
   587	        """
   588	        self.handler.addFilter(filter_)
   589	        try:
   590	            yield
   591	        finally:
   592	            self.handler.removeFilter(filter_)

   612	def get_log_level_for_setting(config: Config, *setting_names: str) -> int | None:
   613	    for setting_name in setting_names:
   614	        log_level = config.getoption(setting_name)
   615	        if log_level is None:
   616	            log_level = config.getini(setting_name)
   617	        if log_level:
   618	            break
   619	    else:
   620	        return None
   621
   622	    if isinstance(log_level, str):
   623	        log_level = log_level.upper()
   624	    try:
   625	        return int(getattr(logging, log_level, log_level))
   626	    except ValueError as e:
   627	        # Python logging does not recognise this as a logging level
   628	        raise UsageError(
   629	            f"'{log_level}' is not recognized as a logging level name for "
   630	            f"'{setting_name}'. Please consider passing the "
   631	            "logging level num instead."
   632	        ) from e

   644	    def __init__(self, config: Config) -> None:
   645	        """Create a new plugin to capture log messages.
   646
   647	        The formatter can be safely shared across all handlers so
   648	        create a single one for the entire test session here.
   649	        """
   650	        self._config = config
   651
   652	        # Report logging.
   653	        self.formatter = self._create_formatter(
   654	            get_option_ini(config, "log_format"),
   655	            get_option_ini(config, "log_date_format"),
   656	            get_option_ini(config, "log_auto_indent"),
   657	        )
   658	        self.log_level = get_log_level_for_setting(config, "log_level")
   659	        self.caplog_handler = LogCaptureHandler()
   660	        self.caplog_handler.setFormatter(self.formatter)
   661	        self.report_handler = LogCaptureHandler()
   662	        self.report_handler.setFormatter(self.formatter)
   663
   664	        # File logging.
   665	        self.log_file_level = get_log_level_for_setting(
   666	            config, "log_file_level", "log_level"
   667	        )
   668	        log_file = get_option_ini(config, "log_file") or os.devnull
   669	        if log_file != os.devnull:
   670	            directory = os.path.dirname(os.path.abspath(log_file))
   671	            if not os.path.isdir(directory):
   672	                os.makedirs(directory)
   673
   674	        self.log_file_mode = get_option_ini(config, "log_file_mode") or "w"
   675	        self.log_file_handler = _FileHandler(
   676	            log_file, mode=self.log_file_mode, encoding="UTF-8"
   677	        )
   678	        log_file_format = get_option_ini(config, "log_file_format", "log_format")
   679	        log_file_date_format = get_option_ini(
   680	            config, "log_file_date_format", "log_date_format"
   681	        )

   683	        log_file_formatter = DatetimeFormatter(
   684	            log_file_format, datefmt=log_file_date_format

   686	        self.log_file_handler.setFormatter(log_file_formatter)

   688	        # CLI/live logging.
   689	        self.log_cli_level = get_log_level_for_setting(
   690	            config, "log_cli_level", "log_level"
   691	        )
   692	        if self._log_cli_enabled():
   693	            terminal_reporter = config.pluginmanager.get_plugin("terminalreporter")
   694	            # Guaranteed by `_log_cli_enabled()`.
   695	            assert terminal_reporter is not None

   697	            # if capturemanager plugin is disabled, live logging still works.
   698	            self.log_cli_handler: (
   699	                _LiveLoggingStreamHandler | _LiveLoggingNullHandler
   700	            ) = _LiveLoggingStreamHandler(terminal_reporter, capture_manager)
   701	        else:
   702	            self.log_cli_handler = _LiveLoggingNullHandler()
   703	        log_cli_formatter = self._create_formatter(
   704	            get_option_ini(config, "log_cli_format", "log_format"),
   705	            get_option_ini(config, "log_cli_date_format", "log_date_format"),
   706	            get_option_ini(config, "log_auto_indent"),
   707	        )
   708	        self.log_cli_handler.setFormatter(log_cli_formatter)
   709	        self._disable_loggers(loggers_to_disable=config.option.logger_disable)

   711	    def _disable_loggers(self, loggers_to_disable: list[str]) -> None:
   712	        if not loggers_to_disable:
   713	            return
   714
   715	        for name in loggers_to_disable:
   716	            logger = logging.getLogger(name)
   717	            logger.disabled = True

   719	    def _create_formatter(self, log_format, log_date_format, auto_indent):
   720	        # Color option doesn't exist if terminal plugin is disabled.
   721	        color = getattr(self._config.option, "color", "no")
   722	        if color != "no" and ColoredLevelFormatter.LEVELNAME_FMT_REGEX.search(
   723	            log_format
   724	        ):
   725	            formatter: logging.Formatter = ColoredLevelFormatter(
   726	                create_terminal_writer(self._config), log_format, log_date_format
   727	            )
   728	        else:
   729	            formatter = DatetimeFormatter(log_format, log_date_format)
   730
   731	        formatter._style = PercentStyleMultiline(
   732	            formatter._style._fmt, auto_indent=auto_indent
   733	        )
   734
   735	        return formatter

   737	    def set_log_path(self, fname: str) -> None:
   738	        """Set the filename parameter for Logging.FileHandler().
   739
   740	        Creates parent directory if it does not exist.

   742	        .. warning::
   743	            This is an experimental API.

   747	        if not fpath.is_absolute():
   748	            fpath = self._config.rootpath / fpath

   750	        if not fpath.parent.exists():
   751	            fpath.parent.mkdir(exist_ok=True, parents=True)

   754	        stream: io.TextIOWrapper = fpath.open(mode=self.log_file_mode, encoding="UTF-8")  # type: ignore[assignment]

   762	            "--log-cli-level"

   812	    @contextmanager
   813	    def _runtest_for(self, item: nodes.Item, when: str) -> Generator[None]:
   814	        """Implement the internals of the pytest_runtest_xxx() hooks."""

   816	            catching_logs(
