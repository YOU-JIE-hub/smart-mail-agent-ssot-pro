  if param_spec and callee.arg_kinds[i] in (ARG_STAR, ARG_STAR2):
   230	                    # If actual arguments are mapped to ParamSpec type, we can't infer individual
   231	                    # constraints, instead store them and infer single constraint at the end.
   232	                    # It is impossible to map actual kind to formal kind, so use some heuristic.
   233	                    # This inference is used as a fallback, so relying on heuristic should be OK.
   234	                    if not incomplete_star_mapping:
   235	                        param_spec_arg_types.append(
   236	                            mapper.expand_actual_type(
   237	                                actual_arg_type, arg_kinds[actual], None, arg_kinds[actual]
   238	                            )
   239	                        )
   240	                        actual_kind = arg_kinds[actual]
   241	                        param_spec_arg_kinds.append(
   242	                            ARG_POS if actual_kind not in (ARG_STAR, ARG_STAR2) else actual_kind
   243	                        )
   244	                        param_spec_arg_names.append(arg_names[actual] if arg_names else None)
   245	                else:
   246	                    actual_type = mapper.expand_actual_type(
   247	                        actual_arg_type,
   248	                        arg_kinds[actual],
   249	                        callee.arg_names[i],
   250	                        callee.arg_kinds[i],
   251	                    )
   252	                    c = infer_constraints(callee.arg_types[i], actual_type, SUPERTYPE_OF)
   253	                    constraints.extend(c)
   254	    if (
   255	        param_spec
   256	        and not any(c.type_var == param_spec.id for c in constraints)
   257	        and not incomplete_star_mapping
   258	    ):
   259	        # Use ParamSpec constraint from arguments only if there are no other constraints,
   260	        # since as explained above it is quite ad-hoc.
   261	        constraints.append(
   262	            Constraint(
   263	                param_spec,
   264	                SUPERTYPE_OF,
   265	                Parameters(
   266	                    arg_types=param_spec_arg_types,
   267	                    arg_kinds=param_spec_arg_kinds,
   268	                    arg_names=param_spec_arg_names,
   269	                    imprecise_arg_kinds=True,
   270	                ),
   271	            )
   272	        )
   273	    if any(isinstance(v, ParamSpecType) for v in callee.variables):
   274	        # As a perf optimization filter imprecise constraints only when we can have them.
   275	        constraints = filter_imprecise_kinds(constraints)
   276	    return constraints

   279	def infer_constraints(
   280	    template: Type, actual: Type, direction: int, skip_neg_op: bool = False
   281	) -> list[Constraint]:
   282	    """Infer type constraints.
   283
   284	    Match a template type, which may contain type variable references,
   285	    recursively against a type which does not contain (the same) type
   286	    variable references. The result is a list of type constrains of
   287	    form 'T is a supertype/subtype of x', where T is a type variable
   288	    present in the template and x is a type without reference to type
   289	    variables present in the template.
   290
   291	    Assume T and S are type variables. Now the following results can be
   292	    calculated (read as '(template, actual) --> result'):
   293
   294	      (T, X)            -->  T :> X
   295	      (X[T], X[Y])      -->  T <: Y and T :> Y
   296	      ((T, T), (X, Y))  -->  T :> X and T :> Y
   297	      ((T, S), (X, Y))  -->  T :> X and S :> Y
   298	      (X[T], Any)       -->  T <: Any and T :> Any
   299
   300	    The constraints are represented as Constraint objects. If skip_neg_op == True,
   301	    then skip adding reverse (polymorphic) constraints (since this is already a call
   302	    to infer such constraints).
   303	    """
   304	    if any(
   305	        get_proper_type(template) == get_proper_type(t)
   306	        and get_proper_type(actual) == get_proper_type(a)
   307	        for (t, a) in reversed(type_state.inferring)
   308	    ):
   309	        return []
   310	    if has_recursive_types(template) or isinstance(get_proper_type(template), Instance):
   311	        # This case requires special care because it may cause infinite recursion.
   312	        # Note that we include Instances because the may be recursive as str(Sequence[str]).
   313	        if not has_type_vars(template):
   314	            # Return early on an empty branch.
   315	            return []
   316	        type_state.inferring.append((template, actual))
   317	        res = _infer_constraints(template, actual, direction, skip_neg_op)
   318	        type_state.inferring.pop()
   319	        return res
   320	    return _infer_constraints(template, actual, direction, skip_neg_op)
   321
   322
   323	def _infer_constraints(
   324	    template: Type, actual: Type, direction: int, skip_neg_op: bool
   325	) -> list[Constraint]:
   326	    orig_template = template
   327	    template = get_proper_type(template)
   328	    actual = get_proper_type(actual)
   329
   330	    # Type inference shouldn't be affected by whether union types have been simplified.
   331	    # We however keep any ErasedType items, so that the caller will see it when using
   332	    # checkexpr.has_erased_component().
   333	    if isinstance(template, UnionType):
   334	        template = mypy.typeops.make_simplified_union(template.items, keep_erased=True)
   335	    if isinstance(actual, UnionType):
   336	        actual = mypy.typeops.make_simplified_union(actual.items, keep_erased=True)
   337
   338	    # Ignore Any types from the type suggestion engine to avoid them
   339	    # causing us to infer Any in situations where a better job could
   340	    # be done otherwise. (This can produce false positives but that
   341	    # doesn't really matter because it is all heuristic anyway.)
   342	    if isinstance(actual, AnyType) and actual.type_of_any == TypeOfAny.suggestion_engine:
   343	        return []
   344
   345	    # type[A | B] is always represented as type[A] | type[B] internally.
   346	    # This makes our constraint solver choke on type[T] <: type[A] | type[B],
   347	    # solving T as generic meet(A, B) which is often `object`. Force unwrap such unions
   348	    # if both sides are type[...] or unions thereof. See `testTypeVarType` test
   349	    type_type_unwrapped = False
   350	    if _is_type_type(template) and _is_type_type(actual):
   351	        type_type_unwrapped = True
   352	        template = _unwrap_type_type(template)
   353	        actual = _unwrap_type_type(actual)
   354
   355	    # If the template is simply a type variable, emit a Constraint directly.
   356	    # We need to handle this case before handling Unions for two reasons:
   357	    #  1. "T <: Union[U1, U2]" is not equivalent to "T <: U1 or T <: U2",
   358	    #     because T can itself be a union (notably, Union[U1, U2] itself).
   359	    #  2. "T :> Union[U1, U2]" is logically equivalent to "T :> U1 and
   360	    #     T :> U2", but they are not equivalent to the constraint solver,
   361	    #     which never introduces new Union types (it uses join() instead).
   362	    if isinstance(template, TypeVarType):
   363	        return [Constraint(template, direction, actual)]
   364
   365	    if (
   366	        isinstance(actual, TypeVarType)
   367	        and not actual.id.is_meta_var()
   368	        and direction == SUPERTYPE_OF
   369	    ):
   370	        # Unless template is also a type variable (or a union that contains one), using the upper
   371	        # bound for inference will usually give better result for actual that is a type variable.
   372	        if not isinstance(template, UnionType) or not any(
   373	            isinstance(t, TypeVarType) for t in template.items
   374	        ):
   375	            actual = get_proper_type(actual.upper_bound)
   376
   377	    # Now handle the case of either template or actual being a Union.
   378	    # For a Union to be a subtype of another type, every item of the Union
   379	    # must be a subtype of that type, so concatenate the constraints.
   380	    if direction == SUBTYPE_OF and isinstance(template, UnionType):
   381	        res = []
   382	        for t_item in template.items:
   383	            res.extend(infer_constraints(t_item, actual, direction))
   384	        return res
   385	    if direction == SUPERTYPE_OF and isinstance(actual, UnionType):
   386	        res = []
   387	        for a_item in actual.items:
   388	            # `orig_template` has to be preserved intact in case it's recursive.
   389	            # If we unwrapped ``type[...]`` previously, wrap the item back again,
   390	            # as ``type[...]`` can't be removed from `orig_template`.
   391	            if type_type_unwrapped:
   392	                a_item = TypeType.make_normalized(a_item)
   393	            res.extend(infer_constraints(orig_template, a_item, direction))
   394	        return res
   395
   396	    # Now the potential subtype is known not to be a Union or a type
   397	    # variable that we are solving for. In that case, for a Union to
   398	    # be a supertype of the potential subtype, some item of the Union
   399	    # must be a supertype of it.
   400	    if direction == SUBTYPE_OF and isinstance(actual, UnionType):
   401	        # If some of items is not a complete type, disregard that.
   402	        items = simplify_away_incomplete_types(actual.items)
   403	        # We infer constraints eagerly -- try to find constraints for a type
   404	        # variable if possible. This seems to help with some real-world
   405	        # use cases.
   406	        return any_constraints(
   407	            [infer_constraints_if_possible(template, a_item, direction) for a_item in items],
   408	            eager=True,
   409	        )
   410	    if direction == SUPERTYPE_OF and isinstance(template, UnionType):
   411	        # When the template is a union, we are okay with leaving some
   412	        # type variables indeterminate. This helps with some special
   413	        # cases, though this isn't very principled.
   414	        result = any_constraints(
   415	            [
   416	                infer_constraints_if_possible(t_item, actual, direction)
   417	                for t_item in template.items
   418	            ],
   419	            eager=isinstance(actual, AnyType),
   420	        )
   421	        if result:
   422	            return result
   423	        elif has_recursive_types(template) and not has_recursive_types(actual):
   424	            return handle_recursive_union(template, actual, direction)
   425	        return []
   426
   427	    # Remaining cases are handled by ConstraintBuilderVisitor.
   428	    return template.accept(ConstraintBuilderVisitor(actual, direction, skip_neg_op))
   429

   431	def _is_type_type(tp: ProperType) -> TypeGuard[TypeType | UnionType]:
   432	    """Is ``tp`` a ``type[...]`` or a union thereof?
   433
   434	    ``Type[A | B]`` is internally represented as ``type[A] | type[B]``, and this
   435	    troubles the solver sometimes.
   436	    """
   437	    return (
   438	        isinstance(tp, TypeType)
   439	        or isinstance(tp, UnionType)
   440	        and all(isinstance(get_proper_type(o), TypeType) for o in tp.items)
   441	    )
   442

   444	def _unwrap_type_type(tp: TypeType | UnionType) -> ProperType:
   445	    """Extract the inner type from ``type[...]`` expression or a union thereof."""
   446	    if isinstance(tp, TypeType):
   447	        return tp.item
   448	    return UnionType.make_union([cast(TypeType, get_proper_type(o)).item for o in tp.items])

   451	def infer_constraints_if_possible(
   452	    template: Type, actual: Type, direction: int
   453	) -> list[Constraint] | None:
   454	    """Like infer_constraints, but return None if the input relation is
   455	    known to be unsatisfiable, for example if template=List[T] and actual=int.
   456	    (In this case infer_constraints would return [], just like it would for
   457	    an automatically satisfied relation like template=List[T] and actual=object.)
   458	    """
   459	    if direction == SUBTYPE_OF and not mypy.subtypes.is_subtype(erase_typevars(template), actual):
   460	        return None
   461	    if direction == SUPERTYPE_OF and not mypy.subtypes.is_subtype(
   462	        actual, erase_typevars(template)
   463	    ):
   464	        return None
   465	    if (
   466	        direction == SUPERTYPE_OF
   467	        and isinstance(template, TypeVarType)
   468	        and not mypy.subtypes.is_subtype(actual, erase_typevars(template.upper_bound))
   469	    ):
   470	        # This is not caught by the above branch because of the erase_typevars() call,
   471	        # that would return 'Any' for a type variable.
   472	        return None
   473	    return infer_constraints(template, actual, direction)

   476	def select_trivial(options: Sequence[list[Constraint] | None]) -> list[list[Constraint]]:
   477	    """Select only those lists where each item is a constraint against Any."""
   478	    res = []
   479	    for option in options:
   480	        if option is None:
   481	            continue
   482	        if all(isinstance(get_proper_type(c.target), AnyType) for c in option):
   483	            res.append(option)
   484	    return res

   487	def merge_with_any(constraint: Constraint) -> Constraint:
   488	    """Transform a constraint target into a union with given Any type."""
   489	    target = constraint.target
   490	    if is_union_with_any(target):
   491	        # Do not produce redundant unions.
   492	        return constraint
   493	    # TODO: if we will support multiple sources Any, use this here instead.
   494	    any_type = AnyType(TypeOfAny.implementation_artifact)
   495	    return Constraint(
   496	        constraint.origin_type_var,
   497	        constraint.op,
   498	        UnionType.make_union([target, any_type], target.line, target.column),
   499	    )

   502	def handle_recursive_union(template: UnionType, actual: Type, direction: int) -> list[Constraint]:
   503	    # This is a hack to special-case things like Union[T, Inst[T]] in recursive types. Although
   504	    # it is quite arbitrary, it is a relatively common pattern, so we should handle it well.
   505	    # This function may be called when inferring against such union resulted in different
   506	    # constraints for each item. Normally we give up in such case, but here we instead split
   507	    # the union in two parts, and try inferring sequentially.
   508	    non_type_var_items = [t for t in template.items if not isinstance(t, TypeVarType)]
   509	    type_var_items = [t for t in template.items if isinstance(t, TypeVarType)]
   510	    return infer_constraints(
   511	        UnionType.make_union(non_type_var_items), actual, direction
   512	    ) or infer_constraints(UnionType.make_union(type_var_items), actual, direction)

   515	def any_constraints(options: list[list[Constraint] | None], *, eager: bool) -> list[Constraint]:
   516	    """Deduce what we can from a collection of constraint lists.
   517
   518	    It's a given that at least one of the lists must be satisfied. A
   519	    None element in the list of options represents an unsatisfiable
   520	    constraint and is ignored.  Ignore empty constraint lists if eager
   521	    is true -- they are always trivially satisfiable.
   522	    """
   523	    if eager:
   524	        valid_options = [option for option in options if option]
   525	    else:
   526	        valid_options = [option for option in options if option is not None]
   527
   528	    if not valid_options:
   529	        return []
   530
   531	    if len(valid_options) == 1:
   532	        return valid_options[0]
   533
   534	    if all(is_same_constraints(valid_options[0], c) for c in valid_options[1:]):
   535	        # Multiple sets of constraints that are all the same. Just pick any one of them.
   536	        return valid_options[0]
   537
   538	    if all(is_similar_constraints(valid_options[0], c) for c in valid_options[1:]):
   539	        # All options have same structure. In this case we can merge-in trivial
   540	        # options (i.e. those that only have Any) and try again.
   541	        # TODO: More generally, if a given (variable, direction) pair appears in
   542	        # every option, combine the bounds with meet/join always, not just for Any.
   543	        trivial_options = select_trivial(valid_options)
   544	        if trivial_options and len(trivial_options) < len(valid_options):
   545	            merged_options = []
   546	            for option in valid_options:
   547	                if option in trivial_options:
   548	                    continue
   549	                merged_options.append([merge_with_any(c) for c in option])
   550	            return any_constraints(list(merged_options), eager=eager)
   551
   552	    # If normal logic didn't work, try excluding trivially unsatisfiable constraint (due to
   553	    # upper bounds) from each option, and comparing them again.
   554	    filtered_options = [filter_satisfiable(o) for o in options]
   555	    if filtered_options != options:
   556	        return any_constraints(filtered_options, eager=eager)
   557
   558	    # Try harder: if that didn't work, try to strip typevars that aren't meta vars.
   559	    # Note this is what we would always do, but unfortunately some callers may not
   560	    # set the meta var status correctly (for historical reasons), so we use this as
   561	    # a fallback only.
   562	    filtered_options = [exclude_non_meta_vars(o) for o in options]
   563	    if filtered_options != options:
   564	        return any_constraints(filtered_options, eager=eager)
   565
   566	    # Otherwise, there are either no valid options or multiple, inconsistent valid
   567	    # options. Give up and deduce nothing.
   568	    return []
   569

   571	def filter_satisfiable(option: list[Constraint] | None) -> list[Constraint] | None:
   572	    """Keep only constraints that can possibly be satisfied.
   573
   574	    Currently, we filter out constraints where target is not a subtype of the upper bound.
   575	    Since those can be never satisfied. We may add more cases in future if it improves type
   576	    inference.
   577	    """
   578	    if not option:
   579	        return option
   580
   581	    satisfiable = []
   582	    for c in option:
   583	        if isinstance(c.origin_type_var, TypeVarType) and c.origin_type_var.values:
   584	            if any(
   585	                mypy.subtypes.is_subtype(c.target, value) for value in c.origin_type_var.values
   586	            ):
   587	                satisfiable.append(c)
   588	        elif mypy.subtypes.is_subtype(c.target, c.origin_type_var.upper_bound):
   589	            satisfiable.append(c)
   590	    if not satisfiable:
   591	        return None
   592	    return satisfiable

   595	def exclude_non_meta_vars(option: list[Constraint] | None) -> list[Constraint] | None:
   596	    # If we had an empty list, keep it intact
   597	    if not option:
   598	        return option
   599	    # However, if none of the options actually references meta vars, better remove
   600	    # this constraint entirely.
   601	    return [c for c in option if c.type_var.is_meta_var()] or None

   604	def is_same_constraints(x: list[Constraint], y: list[Constraint]) -> bool:
   605	    for c1 in x:
   606	        if not any(is_same_constraint(c1, c2) for c2 in y):
   607	            return False
   608	    for c1 in y:
   609	        if not any(is_same_constraint(c1, c2) for c2 in x):
   610	            return False
   611	    return True

   620	        c1.type_var
