   931	class UnboundType(ProperType):
   932	    """Instance type that has not been bound during semantic analysis."""
   933
   934	    __slots__ = (
   935	        "name",
   936	        "args",
   937	        "optional",
   938	        "empty_tuple_index",
   939	        "original_str_expr",
   940	        "original_str_fallback",
   941	    )
   942
   943	    def __init__(
   944	        self,
   945	        name: str,
   946	        args: Sequence[Type] | None = None,
   947	        line: int = -1,
   948	        column: int = -1,
   949	        optional: bool = False,
   950	        empty_tuple_index: bool = False,
   951	        original_str_expr: str | None = None,
   952	        original_str_fallback: str | None = None,
   953	    ) -> None:
   954	        super().__init__(line, column)
   955	        if not args:
   956	            args = []
   957	        self.name = name
   958	        self.args = tuple(args)
   959	        # Should this type be wrapped in an Optional?
   960	        self.optional = optional
   961	        # Special case for X[()]
   962	        self.empty_tuple_index = empty_tuple_index
   963	        # If this UnboundType was originally defined as a str or bytes, keep track of
   964	        # the original contents of that string-like thing. This way, if this UnboundExpr
   965	        # ever shows up inside of a LiteralType, we can determine whether that
   966	        # Literal[...] is valid or not. E.g. Literal[foo] is most likely invalid
   967	        # (unless 'foo' is an alias for another literal or something) and
   968	        # Literal["foo"] most likely is.

   970	        # We keep track of the entire string instead of just using a boolean flag
   971	        # so we can distinguish between things like Literal["foo"] vs
   972	        # Literal["    foo   "].
   973	        #
   974	        # We also keep track of what the original base fallback type was supposed to be
   975	        # so we don't have to try and recompute it later
   976	        self.original_str_expr = original_str_expr
   977	        self.original_str_fallback = original_str_fallback
   978
   979	    def copy_modified(self, args: Bogus[Sequence[Type] | None] = _dummy) -> UnboundType:
   980	        if args is _dummy:
   981	            args = self.args
   982	        return UnboundType(
   983	            name=self.name,
   984	            args=args,
   985	            line=self.line,
   986	            column=self.column,
   987	            optional=self.optional,
   988	            empty_tuple_index=self.empty_tuple_index,

   991	        )

   994	        return visitor.visit_unbound_type(self)
   995
   996	    def __hash__(self) -> int:
   997	        return hash((self.name, self.optional, tuple(self.args), self.original_str_expr))

  1001	            return NotImplemented
  1002	        return (
  1003	            self.name == other.name
  1004	            and self.optional == other.optional
  1005	            and self.args == other.args
  1006	            and self.original_str_expr == other.original_str_expr
  1007	            and self.original_str_fallback == other.original_str_fallback

  1012	            ".class": "UnboundType",
  1013	            "name": self.name,
  1014	            "args": [a.serialize() for a in self.args],
