  1247	    def visit_tuple_type(self, template: TupleType) -> list[Constraint]:

  1249	        unpack_index = find_unpack_in_list(template.items)
  1250	        is_varlength_tuple = (
  1251	            isinstance(actual, Instance) and actual.type.fullname == "builtins.tuple"

  1259	                    # direction is opposite, inferring something may give better error messages.

  1264	                        res = [
  1265	                            Constraint(type_var=unpacked_type, op=self.direction, target=actual)
  1266	                        ]

  1268	                        assert (
  1269	                            isinstance(unpacked_type, Instance)
  1270	                            and unpacked_type.type.fullname == "builtins.tuple"
  1271	                        )

  1420	def find_matching_overload_item(overloaded: Overloaded, template: CallableType) -> CallableType:
  1421	    """Disambiguate overload item against a template."""
  1422	    items = overloaded.items
  1423	    for item in items:
  1424	        # Return type may be indeterminate in the template, so ignore it when performing a
  1425	        # subtype check.
  1426	        if mypy.subtypes.is_callable_compatible(
  1427	            item,
  1428	            template,
  1429	            is_compat=mypy.subtypes.is_subtype,
  1430	            is_proper_subtype=False,
  1431	            ignore_return=True,
  1432	        ):
  1433	            return item
  1434	    # Fall back to the first item if we can't find a match. This is totally arbitrary --
  1435	    # maybe we should just bail out at this point.
  1436	    return items[0]
