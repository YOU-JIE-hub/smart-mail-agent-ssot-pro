   395	        """Record assignment to given name and return True if it defines a new variable.

   405	            # Can't redefine within try/with a block.

   417	                var_blocks[name] = -1

   420	            # Redefinition -- defines a new variable with the same name.
   421	            return True

   423	            # Assigns to an existing variable.
   424	            return False

   428	    """Perform some limited variable renaming in with statements.
   429
   430	    This allows reusing a variable in multiple with statements with
   431	    different types. For example, the two instances of 'x' can have
   432	    incompatible types:
   433
   434	       with C() as x:
   435	           f(x)
   436	       with D() as x:
   437	           g(x)

   440
   441	       with C() as x':
   442	           f(x')
