e:
   370	       if [not] is_error(r1) goto L1 else goto L2
   371	    """
   372
   373	    # Branch ops never raise an exception.
   374	    error_kind = ERR_NEVER
   375
   376	    BOOL: Final = 100
   377	    IS_ERROR: Final = 101
   378
   379	    def __init__(
   380	        self,
   381	        value: Value,
   382	        true_label: BasicBlock,
   383	        false_label: BasicBlock,
   384	        op: int,
   385	        line: int = -1,
   386	        *,
   387	        rare: bool = False,
   388	    ) -> None:
   389	        super().__init__(line)
   390	        # Target value being checked
   391	        self.value = value
   392	        # Branch here if the condition is true
   393	        self.true = true_label
   394	        # Branch here if the condition is false
   395	        self.false = false_label
   396	        # Branch.BOOL (boolean check) or Branch.IS_ERROR (error value check)
   397	        self.op = op
   398	        # If True, the condition is negated
   399	        self.negated = False
   400	        # If not None, the true label should generate a traceback entry (func name, line number)
   401	        self.traceback_entry: tuple[str, int] | None = None
   402	        # If True, we expect to usually take the false branch (for optimization purposes);
   403	        # this is implicitly treated as true if there is a traceback entry
   404	        self.rare = rare
   405
   406	    def targets(self) -> Sequence[BasicBlock]:
   407	        return (self.true, self.false)
   408
   409	    def set_target(self, i: int, new: BasicBlock) -> None:
   410	        assert i == 0 or i == 1
   411	        if i == 0:
   412	            self.true = new
   413	        else:
   414	            self.false = new
   415
   416	    def sources(self) -> list[Value]:
   417	        return [self.value]
   418
   419	    def set_sources(self, new: list[Value]) -> None:
   420	        (self.value,) = new
   421
   422	    def invert(self) -> None:
   423	        self.negated = not self.negated
   424
   425	    def accept(self, visitor: OpVisitor[T]) -> T:
   426	        return visitor.visit_branch(self)

   621	class PrimitiveDescription:
   622	    """Description of a primitive op.
   623
   624	    Primitives get lowered into lower-level ops before code generation.
   625
   626	    If c_function_name is provided, a primitive will be lowered into a CallC op.
   627	    Otherwise custom logic will need to be implemented to transform the
   628	    primitive into lower-level ops.
   629	    """
   630
   631	    def __init__(
   632	        self,
   633	        name: str,
   634	        arg_types: list[RType],
   635	        return_type: RType,  # TODO: What about generic?
   636	        var_arg_type: RType | None,
   637	        truncated_type: RType | None,
   638	        c_function_name: str | None,
   639	        error_kind: int,
   640	        steals: StealsDescription,
   641	        is_borrowed: bool,
   642	        ordering: list[int] | None,
   643	        extra_int_constants: list[tuple[int, RType]],
   644	        priority: int,
   645	        is_pure: bool,
   646	    ) -> None:
   647	        # Each primitive much have a distinct name, but otherwise they are arbitrary.
   648	        self.name: Final = name
   649	        self.arg_types: Final = arg_types
   650	        self.return_type: Final = return_type
   651	        self.var_arg_type: Final = var_arg_type
   652	        self.truncated_type: Final = truncated_type
   653	        # If non-None, this will map to a call of a C helper function; if None,
   654	        # there must be a custom handler function that gets invoked during the lowering
   655	        # pass to generate low-level IR for the primitive (in the mypyc.lower package)
   656	        self.c_function_name: Final = c_function_name
   657	        self.error_kind: Final = error_kind
   658	        self.steals: Final = steals
   659	        self.is_borrowed: Final = is_borrowed
   660	        self.ordering: Final = ordering
   661	        self.extra_int_constants: Final = extra_int_constants
   662	        self.priority: Final = priority
   663	        # Pure primitives have no side effects, take immutable arguments, and
   664	        # never fail. They support additional optimizations.
   665	        self.is_pure: Final = is_pure
   666	        if is_pure:
   667	            assert error_kind == ERR_NEVER
   668
   669	    def __repr__(self) -> str:
   670	        return f"<PrimitiveDescription {self.name!r}: {self.arg_types}>"

   673	class PrimitiveOp(RegisterOp):
   674	    """A higher-level primitive operation.
   675
   676	    Some of these have special compiler support. These will be lowered
   677	    (transformed) into lower-level IR ops before code generation, and after
   678	    reference counting op insertion. Others will be transformed into CallC
   679	    ops.
   680
   681	    Tagged integer equality is a typical primitive op with non-trivial
   682	    lowering. It gets transformed into a tag check, followed by different
   683	    code paths for short and long representations.
   684	    """
   685
   686	    def __init__(self, args: list[Value], desc: PrimitiveDescription, line: int = -1) -> None:
   687	        self.args = args
   688	        self.type = desc.return_type
   689	        self.error_kind = desc.error_kind
   690	        self.desc = desc
   691
   692	    def sources(self) -> list[Value]:
   693	        return self.args
   694
   695	    def set_sources(self, new: list[Value]) -> None:
   696	        self.args = new[:]
   697
   698	    def stolen(self) -> list[Value]:
   699	        steals = self.desc.steals
   700	        if isinstance(steals, list):
   701	            assert len(steals) == len(self.args)
   702	            return [arg for arg, steal in zip(self.args, steals) if steal]
   703	        else:
   704	            return [] if not steals else self.sources()
   705
   706	    def accept(self, visitor: OpVisitor[T]) -> T:
   707	        return visitor.visit_primitive_op(self)
