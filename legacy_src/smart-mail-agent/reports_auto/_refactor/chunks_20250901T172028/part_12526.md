    80	class Templite:
    81	    """A simple template renderer, for a nano-subset of Django syntax.
    82
    83	    Supported constructs are extended variable access::
    84
    85	        {{var.modifier.modifier|filter|filter}}
    86
    87	    loops::
    88
    89	        {% for var in list %}...{% endfor %}
    90
    91	    and ifs::
    92
    93	        {% if var %}...{% endif %}
    94
    95	    Comments are within curly-hash markers::
    96
    97	        {# This will be ignored #}
    98
    99	    Lines between `{% joined %}` and `{% endjoined %}` will have lines stripped
   100	    and joined.  Be careful, this could join words together!
   101
   102	    Any of these constructs can have a hyphen at the end (`-}}`, `-%}`, `-#}`),
   103	    which will collapse the white space following the tag.
   104
   105	    Construct a Templite with the template text, then use `render` against a
   106	    dictionary context to create a finished string::
   107
   108	        templite = Templite('''
   109	            <h1>Hello {{name|upper}}!</h1>
   110	            {% for topic in topics %}
   111	                <p>You are interested in {{topic}}.</p>
   112	            {% endif %}
   113	            ''',
   114	            {"upper": str.upper},
   115	        )
   116	        text = templite.render({
   117	            "name": "Ned",
   118	            "topics": ["Python", "Geometry", "Juggling"],
   119	        })
   120
   121	    """
   122	    def __init__(self, text: str, *contexts: dict[str, Any]) -> None:
   123	        """Construct a Templite with the given `text`.
   124
   125	        `contexts` are dictionaries of values to use for future renderings.
   126	        These are good for filters and global values.
   127
   128	        """
   129	        self.context = {}
   130	        for context in contexts:
   131	            self.context.update(context)
   132
   133	        self.all_vars: set[str] = set()
   134	        self.loop_vars: set[str] = set()
   135
   136	        # We construct a function in source form, then compile it and hold onto
   137	        # it, and execute it to render the template.
   138	        code = CodeBuilder()
   139
   140	        code.add_line("def render_function(context, do_dots):")
   141	        code.indent()
   142	        vars_code = code.add_section()
   143	        code.add_line("result = []")
   144	        code.add_line("append_result = result.append")
   145	        code.add_line("extend_result = result.extend")
   146	        code.add_line("to_str = str")
   147
   148	        buffered: list[str] = []
   149
   150	        def flush_output() -> None:
   151	            """Force `buffered` to the code builder."""
   152	            if len(buffered) == 1:
   153	                code.add_line("append_result(%s)" % buffered[0])
   154	            elif len(buffered) > 1:
   155	                code.add_line("extend_result([%s])" % ", ".join(buffered))
   156	            del buffered[:]
   157
   158	        ops_stack = []
   159
   160	        # Split the text to form a list of tokens.
   161	        tokens = re.split(r"(?s)({{.*?}}|{%.*?%}|{#.*?#})", text)
   162
   163	        squash = in_joined = False
   164
   165	        for token in tokens:
   166	            if token.startswith("{"):
   167	                start, end = 2, -2
   168	                squash = (token[-3] == "-")
   169	                if squash:
   170	                    end = -3
   171
   172	                if token.startswith("{#"):
   173	                    # Comment: ignore it and move on.
   174	                    continue
   175	                elif token.startswith("{{"):
   176	                    # An expression to evaluate.
   177	                    expr = self._expr_code(token[start:end].strip())
   178	                    buffered.append("to_str(%s)" % expr)
   179	                else:
   180	                    # token.startswith("{%")
   181	                    # Action tag: split into words and parse further.
   182	                    flush_output()
   183
   184	                    words = token[start:end].strip().split()
   185	                    if words[0] == "if":
   186	                        # An if statement: evaluate the expression to determine if.
   187	                        if len(words) != 2:
   188	                            self._syntax_error("Don't understand if", token)
   189	                        ops_stack.append("if")
   190	                        code.add_line("if %s:" % self._expr_code(words[1]))
   191	                        code.indent()
   192	                    elif words[0] == "for":
   193	                        # A loop: iterate over expression result.
   194	                        if len(words) != 4 or words[2] != "in":
   195	                            self._syntax_error("Don't understand for", token)
   196	                        ops_stack.append("for")
   197	                        self._variable(words[1], self.loop_vars)
   198	                        code.add_line(
   199	                            f"for c_{words[1]} in {self._expr_code(words[3])}:",
   200	                        )
   201	                        code.indent()
   202	                    elif words[0] == "joined":
   203	                        ops_stack.append("joined")
   204	                        in_joined = True
   205	                    elif words[0].startswith("end"):
   206	                        # Endsomething.  Pop the ops stack.
   207	                        if len(words) != 1:
   208	                            self._syntax_error("Don't understand end", token)
   209	                        end_what = words[0][3:]
   210	                        if not ops_stack:
   211	                            self._syntax_error("Too many ends", token)
   212	                        start_what = ops_stack.pop()
   213	                        if start_what != end_what:
   214	                            self._syntax_error("Mismatched end tag", end_what)
   215	                        if end_what == "joined":
   216	                            in_joined = False
   217	                        else:
   218	                            code.dedent()
   219	                    else:
   220	                        self._syntax_error("Don't understand tag", words[0])
   221	            else:
   222	                # Literal content.  If it isn't empty, output it.
   223	                if in_joined:
   224	                    token = re.sub(r"\s*\n\s*", "", token.strip())
   225	                elif squash:
   226	                    token = token.lstrip()
   227	                if token:
   228	                    buffered.append(repr(token))
   229
   230	        if ops_stack:
   231	            self._syntax_error("Unmatched action tag", ops_stack[-1])
   232
   233	        flush_output()
   234
   235	        for var_name in self.all_vars - self.loop_vars:
   236	            vars_code.add_line(f"c_{var_name} = context[{var_name!r}]")
   237
   238	        code.add_line("return ''.join(result)")
   239	        code.dedent()
   240	        self._render_function = cast(
   241	            Callable[
   242	                [dict[str, Any], Callable[..., Any]],
   243	                str,
   244	            ],
   245	            code.get_globals()["render_function"],
   246	        )

   248	    def _expr_code(self, expr: str) -> str:
   249	        """Generate a Python expression for `expr`."""
   250	        if "|" in expr:
   251	            pipes = expr.split("|")
   252	            code = self._expr_code(pipes[0])
   253	            for func in pipes[1:]:
   254	                self._variable(func, self.all_vars)
   255	                code = f"c_{func}({code})"
   256	        elif "." in expr:
   257	            dots = expr.split(".")

   259	            args = ", ".join(repr(d) for d in dots[1:])
   260	            code = f"do_dots({code}, {args})"
   261	        else:
   262	            self._variable(expr, self.all_vars)
   263	            code = "c_%s" % expr
   264	        return code
