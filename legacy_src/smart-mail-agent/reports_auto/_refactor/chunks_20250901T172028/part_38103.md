    82	    def init_under_package_root(self, path: str) -> bool:
    83	        """Is this path an __init__.py under a package root?
    84
    85	        This is used to detect packages that don't contain __init__.py
    86	        files, which is needed to support Bazel.  The function should
    87	        only be called for non-existing files.

    89	        It will return True if it refers to a __init__.py file that
    90	        Bazel would create, so that at runtime Python would think the
    91	        directory containing it is a package.  For this to work you
    92	        must pass one or more package roots using the --package-root

    96	        itself will not be considered to contain a __init__.py file.

    98	        necessary for mypy to properly distinguish packages from other
    99	        directories.

   101	        See https://docs.bazel.build/versions/master/be/python.html,
   102	        where this behavior is described under legacy_create_init.

   105	            return False
   106	        dirname, basename = os.path.split(path)
   107	        if basename != "__init__.py":
   108	            return False
   109	        if not os.path.basename(dirname).isidentifier():
   110	            # Can't put an __init__.py in a place that's not an identifier
   111	            return False

   118	                return False

   121	        if os.path.isabs(path):

   123	        path = os.path.normpath(path)
   124	        for root in self.package_root:

   127	                    # A package root itself is never a package.
   128	                    ok = False
   129	                    break
   130	                else:
   131	                    ok = True

   134	    def _fake_init(self, path: str) -> os.stat_result:
   135	        """Prime the cache with a fake __init__.py file.
   136
   137	        This makes code that looks for path believe an empty file by
   138	        that name exists.  Should only be called after
   139	        init_under_package_root() returns True.

   141	        dirname, basename = os.path.split(path)
   142	        assert basename == "__init__.py", path

   144	        dirname = os.path.normpath(dirname)

   146	        # Get stat result as a list so we can modify it.

   149	        seq[stat.ST_INO] = 1

   171	            self.listdir_error_cache[path] = copy_os_error(err)

   174	        # Check the fake cache.
   175	        if path in self.fake_package_cache and "__init__.py" not in results:
   176	            results.append("__init__.py")
   177	        return results

   182	            return False

   185	    def isfile_case(self, path: str, prefix: str) -> bool:
   186	        """Return whether path exists and is a file.
   187
   188	        On case-insensitive filesystems (like Mac or Windows) this returns
   189	        False if the case of path's last component does not exactly match
   190	        the case found in the filesystem.
   191
   192	        We check also the case of other path components up to prefix.
   193	        For example, if path is 'user-stubs/pack/mod.pyi' and prefix is 'user-stubs',
   194	        we check that the case of 'pack' and 'mod.py' matches exactly, 'user-stubs' will be
   195	        case insensitive on case insensitive filesystems.

   201	            return False

   207	            return False
   208	        try:

   210	            # This allows one to check file name case sensitively in
   211	            # case-insensitive filesystems.

   213	        except OSError:
   214	            res = False
   215	        if res:
   216	            # Also recursively check the other path components in case sensitive way.
   217	            res = self.exists_case(head, prefix)
   218	        self.isfile_case_cache[path] = res
   219	        return res

   221	    def exists_case(self, path: str, prefix: str) -> bool:

   225	        if path in self.exists_case_cache:

   242	        self.exists_case_cache[path] = res
   243	        return res

   255	    def read(self, path: str) -> bytes:

   257	            return self.read_cache[path]

   265	        dirname, basename = os.path.split(path)

   268	        if basename == "__init__.py" and dirname in self.fake_package_cache:
   269	            data = b""

   272	                with open(path, "rb") as f:
   273	                    data = f.read()

   275	                self.read_error_cache[path] = err
   276	                raise

   278	        self.read_cache[path] = data
   279	        self.hash_cache[path] = hash_digest(data)

   285	        return self.hash_cache[path]

    18	class FileSystemWatcher:

    20
    21	    All file system access is performed using FileSystemCache. We
    22	    detect changed files by stat()ing them all and comparing hashes
    23	    of potentially changed files. If a file has both size and mtime
    24	    unmodified, the file is assumed to be unchanged.
    25
    26	    An important goal of this class is to make it easier to eventually
    27	    use file system events to detect file changes.
    28
    29	    Note: This class doesn't flush the file system cache. If you don't
    30	    manually flush it, changes won't be seen.
    31	    """
    32
    33	    # TODO: Watching directories?
    34	    # TODO: Handle non-files
    35
    36	    def __init__(self, fs: FileSystemCache) -> None:
    37	        self.fs = fs
    38	        self._paths: set[str] = set()
    39	        self._file_data: dict[str, FileData | None] = {}
    40
    41	    def dump_file_data(self) -> dict[str, tuple[float, int, str]]:
    42	        return {k: v for k, v in self._file_data.items() if v is not None}
    43
    44	    def set_file_data(self, path: str, data: FileData) -> None:
    45	        self._file_data[path] = data
    46
    47	    def add_watched_paths(self, paths: Iterable[str]) -> None:
    48	        for path in paths:
    49	            if path not in self._paths:
    50	                # By storing None this path will get reported as changed by
    51	                # find_changed if it exists.
    52	                self._file_data[path] = None
    53	        self._paths |= set(paths)
    54
    55	    def remove_watched_paths(self, paths: Iterable[str]) -> None:
    56	        for path in paths:
    57	            if path in self._file_data:
    58	                del self._file_data[path]
    59	        self._paths -= set(paths)
    60
    61	    def _update(self, path: str, st: os.stat_result) -> None:
    62	        hash_digest = self.fs.hash_digest(path)
    63	        self._file_data[path] = FileData(st.st_mtime, st.st_size, hash_digest)
    64
    65	    def _find_changed(self, paths: Iterable[str]) -> AbstractSet[str]:
    66	        changed = set()
    67	        for path in paths:
    68	            old = self._file_data[path]
    69	            st = self.fs.stat_or_none(path)
    70	            if st is None:
    71	                if old is not None:
    72	                    # File was deleted.
    73	                    changed.add(path)
    74	                    self._file_data[path] = None
    75	            else:
    76	                if old is None:
    77	                    # File is new.
    78	                    changed.add(path)
    79	                    self._update(path, st)
    80	                # Round mtimes down, to match the mtimes we write to meta files
    81	                elif st.st_size != old.st_size or int(st.st_mtime) != int(old.st_mtime):
    82	                    # Only look for changes if size or mtime has changed as an
    83	                    # optimization, since calculating hash is expensive.
    84	                    new_hash = self.fs.hash_digest(path)
    85	                    self._update(path, st)
    86	                    if st.st_size != old.st_size or new_hash != old.hash:
    87	                        # Changed file.
    88	                        changed.add(path)
    89	        return changed

    92	        """Return paths that have changes since the last call, in the watched set."""
    93	        return self._find_changed(self._paths)

    98	        This only calls self.fs.stat() on added or updated files, not
    99	        on all files.  It believes all other files are unchanged!

   102	        remove_watched_paths() for remove.

   105	        self.add_watched_paths(update)
   106	        return self._find_changed(update)
