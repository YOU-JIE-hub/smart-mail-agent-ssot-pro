 656	    except CompileError as err:
   657	        # There was a blocking error, so module AST is incomplete. Restore old modules.
   658	        restore([module])
   659	        return BlockedUpdate(module, path, remaining_modules, err.messages)
   660
   661	    # Merge old and new ASTs.
   662	    new_modules_dict: dict[str, MypyFile | None] = {module: state.tree}
   663	    replace_modules_with_new_variants(manager, graph, {orig_module: orig_tree}, new_modules_dict)
   664
   665	    t1 = time.time()
   666	    # Perform type checking.
   667	    state.type_checker().reset()
   668	    state.type_check_first_pass()
   669	    state.type_check_second_pass()
   670	    state.detect_possibly_undefined_vars()
   671	    state.generate_unused_ignore_notes()
   672	    state.generate_ignore_without_code_notes()
   673	    t2 = time.time()
   674	    state.finish_passes()
   675	    t3 = time.time()
   676	    manager.add_stats(semanal_time=t1 - t0, typecheck_time=t2 - t1, finish_passes_time=t3 - t2)
   677
   678	    graph[module] = state
   679
   680	    return NormalUpdate(module, path, remaining_modules, state.tree)

   683	def find_relative_leaf_module(modules: list[tuple[str, str]], graph: Graph) -> tuple[str, str]:
   684	    """Find a module in a list that directly imports no other module in the list.
   685
   686	    If no such module exists, return the lexicographically first module from the list.
   687	    Always return one of the items in the modules list.
   688
   689	    NOTE: If both 'abc' and 'typing' have changed, an effect of the above rule is that
   690	        we prefer 'abc', even if both are in the same SCC. This works around a false
   691	        positive in 'typing', at least in tests.
   692
   693	    Args:
   694	        modules: List of (module, path) tuples (non-empty)
   695	        graph: Program import graph that contains all modules in the module list
   696	    """
   697	    assert modules
   698	    # Sort for repeatable results.
   699	    modules = sorted(modules)
   700	    module_set = {module for module, _ in modules}
   701	    for module, path in modules:
   702	        state = graph[module]
   703	        if len(set(state.dependencies) & module_set) == 0:
   704	            # Found it!
   705	            return module, path
   706	    # Could not find any. Just return the first module (by lexicographic order).
   707	    return modules[0]

   759	def calculate_active_triggers(
   760	    manager: BuildManager,
   761	    old_snapshots: dict[str, dict[str, SymbolSnapshot]],
   762	    new_modules: dict[str, MypyFile | None],
   763	) -> set[str]:
   764	    """Determine activated triggers by comparing old and new symbol tables.
   765
   766	    For example, if only the signature of function m.f is different in the new
   767	    symbol table, return {'<m.f>'}.
   768	    """
   769	    names: set[str] = set()
   770	    for id in new_modules:
   771	        snapshot1 = old_snapshots.get(id)
   772	        if snapshot1 is None:
   773	            names.add(id)
   774	            snapshot1 = {}
   775	        new = new_modules[id]
   776	        if new is None:
   777	            snapshot2 = snapshot_symbol_table(id, SymbolTable())
   778	            names.add(id)
   779	        else:
   780	            snapshot2 = snapshot_symbol_table(id, new.names)
   781	        diff = compare_symbol_table_snapshots(id, snapshot1, snapshot2)
   782	        package_nesting_level = id.count(".")
   783	        for item in diff.copy():
   784	            if item.count(".") <= package_nesting_level + 1 and item.split(".")[-1] not in (
   785	                "__builtins__",
   786	                "__file__",
   787	                "__name__",
   788	                "__package__",
   789	                "__doc__",
   790	            ):
   791	                # Activate catch-all wildcard trigger for top-level module changes (used for
   792	                # "from m import *"). This also gets triggered by changes to module-private
   793	                # entries, but as these unneeded dependencies only result in extra processing,
   794	                # it's a minor problem.
   795	                #
   796	                # TODO: Some __* names cause mistriggers. Fix the underlying issue instead of
   797	                #     special casing them here.
   798	                diff.add(id + WILDCARD_TAG)
   799	            if item.count(".") > package_nesting_level + 1:
   800	                # These are for changes within classes, used by protocols.
   801	                diff.add(item.rsplit(".", 1)[0] + WILDCARD_TAG)
   802
   803	        names |= diff
   804	    return {make_trigger(name) for name in names}

   807	def replace_modules_with_new_variants(
   808	    manager: BuildManager,
   809	    graph: dict[str, State],
   810	    old_modules: dict[str, MypyFile | None],
   811	    new_modules: dict[str, MypyFile | None],
   812	) -> None:
   813	    """Replace modules with newly builds versions.
   814
   815	    Retain the identities of externally visible AST nodes in the
   816	    old ASTs so that references to the affected modules from other
   817	    modules will still be valid (unless something was deleted or
   818	    replaced with an incompatible definition, in which case there
   819	    will be dangling references that will be handled by
   820	    propagate_changes_using_dependencies).
   821	    """
   822	    for id in new_modules:
   823	        preserved_module = old_modules.get(id)
   824	        new_module = new_modules[id]
   825	        if preserved_module and new_module is not None:
   826	            merge_asts(preserved_module, preserved_module.names, new_module, new_module.names)
   827	            manager.modules[id] = preserved_module
   828	            graph[id].tree = preserved_module

   831	def propagate_changes_using_dependencies(
   832	    manager: BuildManager,
   833	    graph: dict[str, State],
   834	    deps: dict[str, set[str]],
   835	    triggered: set[str],
   836	    up_to_date_modules: set[str],
   837	    targets_with_errors: set[str],
   838	    processed_targets: list[str],
   839	) -> list[tuple[str, str]]:
   840	    """Transitively rechecks targets based on triggers and the dependency map.
   841
   842	    Returns a list (module id, path) tuples representing modules that contain
   843	    a target that needs to be reprocessed but that has not been parsed yet.
   844
   845	    Processed targets should be appended to processed_targets (used in tests only,
   846	    to test the order of processing targets).
   847	    """
   848
   849	    num_iter = 0
   850	    remaining_modules: list[tuple[str, str]] = []
   851
   852	    # Propagate changes until nothing visible has changed during the last
   853	    # iteration.
   854	    while triggered or targets_with_errors:
   855	        num_iter += 1
   856	        if num_iter > MAX_ITER:
   857	            raise RuntimeError("Max number of iterations (%d) reached (endless loop?)" % MAX_ITER)
   858
   859	        todo, unloaded, stale_protos = find_targets_recursive(
   860	            manager, graph, triggered, deps, up_to_date_modules
   861	        )
   862	        # TODO: we sort to make it deterministic, but this is *incredibly* ad hoc
   863	        remaining_modules.extend((id, graph[id].xpath) for id in sorted(unloaded))
   864	        # Also process targets that used to have errors, as otherwise some
   865	        # errors might be lost.
   866	        for target in targets_with_errors:
   867	            id = module_prefix(graph, target)
   868	            if id is not None and id not in up_to_date_modules:
   869	                if id not in todo:
   870	                    todo[id] = set()
   871	                manager.log_fine_grained(f"process target with error: {target}")
   872	                more_nodes, _ = lookup_target(manager, target)
   873	                todo[id].update(more_nodes)
   874	        triggered = set()
   875	        # First invalidate subtype caches in all stale protocols.
   876	        # We need to do this to avoid false negatives if the protocol itself is
   877	        # unchanged, but was marked stale because its sub- (or super-) type changed.
   878	        for info in stale_protos:
   879	            type_state.reset_subtype_caches_for(info)
   880	        # Then fully reprocess all targets.
   881	        # TODO: Preserve order (set is not optimal)
   882	        for id, nodes in sorted(todo.items(), key=lambda x: x[0]):
   883	            assert id not in up_to_date_modules
   884	            triggered |= reprocess_nodes(manager, graph, id, nodes, deps, processed_targets)
   885	        # Changes elsewhere may require us to reprocess modules that were
   886	        # previously considered up to date. For example, there may be a
   887	        # dependency loop that loops back to an originally processed module.
   888	        up_to_date_modules = set()
   889	        targets_with_errors = set()
   890	        if is_verbose(manager):
   891	            manager.log_fine_grained(f"triggered: {list(triggered)!r}")
   892
   893	    return remaining_modules
   894

   896	def find_targets_recursive(
   897	    manager: BuildManager,
   898	    graph: Graph,
   899	    triggers: set[str],
   900	    deps: dict[str, set[str]],
   901	    up_to_date_modules: set[str],
   902	) -> tuple[dict[str, set[FineGrainedDeferredNode]], set[str], set[TypeInfo]]:
   903	    """Find names of all targets that need to reprocessed, given some triggers.
   904
   905	    Returns: A tuple containing a:
   906	     * Dictionary from module id to a set of stale targets.
   907	     * A set of module ids for unparsed modules with stale targets.
   908	    """
   909	    result: dict[str, set[FineGrainedDeferredNode]] = {}
   910	    worklist = triggers
   911	    processed: set[str] = set()
   912	    stale_protos: set[TypeInfo] = set()
   913	    unloaded_files: set[str] = set()
   914
   915	    # Find AST nodes corresponding to each target.
   916	    #
   917	    # TODO: Don't rely on a set, since the items are in an unpredictable order.
   918	    while worklist:
   919	        processed |= worklist
   920	        current = worklist
   921	        worklist = set()
   922	        for target in current:
   923	            if target.startswith("<"):
   924	                module_id = module_prefix(graph, trigger_to_target(target))
   925	                if module_id:
   926	                    ensure_deps_loaded(module_id, deps, graph)
   927
   928	                worklist |= deps.get(target, set()) - processed
   929	            else:
   930	                module_id = module_prefix(graph, target)
   931	                if module_id is None:
   932	                    # Deleted module.
   933	                    continue
   934	                if module_id in up_to_date_modules:
   935	                    # Already processed.
   936	                    continue
   937	                if (
   938	                    module_id not in manager.modules
   939	                    or manager.modules[module_id].is_cache_skeleton
   940	                ):
   941	                    # We haven't actually parsed and checked the module, so we don't have
   942	                    # access to the actual nodes.
   943	                    # Add it to the queue of files that need to be processed fully.
   944	                    unloaded_files.add(module_id)
   945	                    continue
   946
   947	                if module_id not in result:
   948	                    result[module_id] = set()
   949	                manager.log_fine_grained(f"process: {target}")
   950	                deferred, stale_proto = lookup_target(manager, target)
   951	                if stale_proto:
   952	                    stale_protos.add(stale_proto)
   953	                result[module_id].update(deferred)
   954
   955	    return result, unloaded_files, stale_protos

   958	def reprocess_nodes(
   959	    manager: BuildManager,
   960	    graph: dict[str, State],
   961	    module_id: str,
   962	    nodeset: set[FineGrainedDeferredNode],
   963	    deps: dict[str, set[str]],
   964	    processed_targets: list[str],
   965	) -> set[str]:
   966	    """Reprocess a set of nodes within a single module.
   967
   968	    Return fired triggers.
   969	    """
   970	    if module_id not in graph:
   971	        manager.log_fine_grained("%s not in graph (blocking errors or deleted?)" % module_id)
   972	        return set()
   973
   974	    file_node = manager.modules[module_id]
   975	    old_symbols = find_symbol_tables_recursive(file_node.fullname, file_node.names)
   976	    old_symbols = {name: names.copy() for name, names in old_symbols.items()}
   977	    old_symbols_snapshot = snapshot_symbol_table(file_node.fullname, file_node.names)
   978
   979	    def key(node: FineGrainedDeferredNode) -> int:
   980	        # Unlike modules which are sorted by name within SCC,
   981	        # nodes within the same module are sorted by line number, because
   982	        # this is how they are processed in normal mode.
   983	        return node.node.line
   984
   985	    nodes = sorted(nodeset, key=key)
   986
   987	    state = graph[module_id]
   988	    options = state.options
   989	    manager.errors.set_file_ignored_lines(
   990	        file_node.path, file_node.ignored_lines, options.ignore_errors or state.ignore_all
   991	    )
   992	    manager.errors.set_skipped_lines(file_node.path, file_node.skipped_lines)
   993
   994	    targets = set()
   995	    for node in nodes:
   996	        target = target_from_node(module_id, node.node)
   997	        if target is not None:
   998	            targets.add(target)
   999	    manager.errors.clear_errors_in_targets(file_node.path, targets)
  1000
  1001	    # If one of the nodes is the module itself, emit any errors that
  1002	    # happened before semantic analysis.
  1003	    for target in targets:
  1004	        if target == module_id:
  1005	            for info in graph[module_id].early_errors:
  1006	                manager.errors.add_error_info(info)
  1007
  1008	    # Strip semantic analysis information.
  1009	    saved_attrs: SavedAttributes = {}
  1010	    for deferred in nodes:
  1011	        processed_targets.append(deferred.node.fullname)
  1012	        strip_target(deferred.node, saved_attrs)
  1013	    semantic_analysis_for_targets(graph[module_id], nodes, graph, saved_attrs)
  1014	    # Merge symbol tables to preserve identities of AST nodes. The file node will remain
  1015	    # the same, but other nodes may have been recreated with different identities, such as
  1016	    # NamedTuples defined using assignment statements.
  1017	    new_symbols = find_symbol_tables_recursive(file_node.fullname, file_node.names)
  1018	    for name in old_symbols:
  1019	        if name in new_symbols:
  1020	            merge_asts(file_node, old_symbols[name], file_node, new_symbols[name])
  1021
  1022	    # Type check.
  1023	    checker = graph[module_id].type_checker()
  1024	    checker.reset()
  1025	    # We seem to need additional passes in fine-grained incremental mode.
  1026	    checker.pass_num = 0
  1027	    checker.last_pass = 3
  1028	    more = checker.check_second_pass(nodes)
  1029	    while more:
  1030	        more = False
  1031	        if graph[module_id].type_checker().check_second_pass():
  1032	            more = True
  1033
  1034	    if manager.options.export_types:
  1035	        manager.all_types.update(graph[module_id].type_map())
  1036
  1037	    new_symbols_snapshot = snapshot_symbol_table(file_node.fullname, file_node.names)
  1038	    # Check if any attribute types were changed and need to be propagated further.
  1039	    changed = compare_symbol_table_snapshots(
  1040	        file_node.fullname, old_symbols_snapshot, new_symbols_snapshot
  1041	    )
  1042	    new_triggered = {make_trigger(name) for name in changed}
  1043
  1044	    # Dependencies may have changed.
  1045	    update_deps(module_id, nodes, graph, deps, options)
  1046
  1047	    # Report missing imports.
  1048	    graph[module_id].verify_dependencies()
  1049
  1050	    graph[module_id].free_state()
  1051
  1052	    return new_triggered
  1053

  1055	def find_symbol_tables_recursive(prefix: str, symbols: SymbolTable) -> dict[str, SymbolTable]:
  1056	    """Find all nested symbol tables.
  1057
  1058	    Args:
  1059	        prefix: Full name prefix (used for return value keys and to filter result so that
  1060	            cross references to other modules aren't included)
  1061	        symbols: Root symbol table
  1062
  1063	    Returns a dictionary from full name to corresponding symbol table.
  1064	    """
  1065	    result = {prefix: symbols}
  1066	    for name, node in symbols.items():
  1067	        if isinstance(node.node, TypeInfo) and node.node.fullname.startswith(prefix + "."):
  1068	            more = find_symbol_tables_recursive(prefix + "." + name, node.node.names)
  1069	            result.update(more)
  1070	    return result
