ches[-1].may_be_defined
   134
   135	    def is_undefined(self, name: str) -> bool:
   136	        assert len(self.branches) > 0
   137	        branch = self.branches[-1]
   138	        return name not in branch.may_be_defined and name not in branch.must_be_defined
   139
   140	    def is_defined_in_a_branch(self, name: str) -> bool:
   141	        assert len(self.branches) > 0
   142	        for b in self.branches:
   143	            if name in b.must_be_defined or name in b.may_be_defined:
   144	                return True
   145	        return False
   146
   147	    def done(self) -> BranchState:
   148	        # First, compute all vars, including skipped branches. We include skipped branches
   149	        # because our goal is to capture all variables that semantic analyzer would
   150	        # consider defined.
   151	        all_vars = set()
   152	        for b in self.branches:
   153	            all_vars.update(b.may_be_defined)
   154	            all_vars.update(b.must_be_defined)
   155	        # For the rest of the things, we only care about branches that weren't skipped.
   156	        non_skipped_branches = [b for b in self.branches if not b.skipped]
   157	        if non_skipped_branches:
   158	            must_be_defined = non_skipped_branches[0].must_be_defined
   159	            for b in non_skipped_branches[1:]:
   160	                must_be_defined.intersection_update(b.must_be_defined)
   161	        else:
   162	            must_be_defined = set()
   163	        # Everything that wasn't defined in all branches but was defined
   164	        # in at least one branch should be in `may_be_defined`!
   165	        may_be_defined = all_vars.difference(must_be_defined)
   166	        return BranchState(
   167	            must_be_defined=must_be_defined,
   168	            may_be_defined=may_be_defined,
   169	            skipped=len(non_skipped_branches) == 0,
   170	        )

   180	class Scope:
   181	    def __init__(self, stmts: list[BranchStatement], scope_type: ScopeType) -> None:
   182	        self.branch_stmts: list[BranchStatement] = stmts
   183	        self.scope_type = scope_type
   184	        self.undefined_refs: dict[str, set[NameExpr]] = {}
   185
   186	    def copy(self) -> Scope:
   187	        result = Scope([s.copy() for s in self.branch_stmts], self.scope_type)
   188	        result.undefined_refs = self.undefined_refs.copy()
   189	        return result
   190
   191	    def record_undefined_ref(self, o: NameExpr) -> None:
   192	        if o.name not in self.undefined_refs:
   193	            self.undefined_refs[o.name] = set()
   194	        self.undefined_refs[o.name].add(o)
   195
   196	    def pop_undefined_ref(self, name: str) -> set[NameExpr]:
   197	        return self.undefined_refs.pop(name, set())

   200	class DefinedVariableTracker:
   201	    """DefinedVariableTracker manages the state and scope for the UndefinedVariablesVisitor."""
   202
   203	    def __init__(self) -> None:
   204	        # There's always at least one scope. Within each scope, there's at least one "global" BranchingStatement.
   205	        self.scopes: list[Scope] = [Scope([BranchStatement()], ScopeType.Global)]
   206	        # disable_branch_skip is used to disable skipping a branch due to a return/raise/etc. This is useful
   207	        # in things like try/except/finally statements.
   208	        self.disable_branch_skip = False
   209
   210	    def copy(self) -> DefinedVariableTracker:
   211	        result = DefinedVariableTracker()
   212	        result.scopes = [s.copy() for s in self.scopes]
   213	        result.disable_branch_skip = self.disable_branch_skip
   214	        return result
   215
   216	    def _scope(self) -> Scope:
   217	        assert len(self.scopes) > 0
   218	        return self.scopes[-1]
   219
   220	    def enter_scope(self, scope_type: ScopeType) -> None:
   221	        assert len(self._scope().branch_stmts) > 0
   222	        initial_state = None
   223	        if scope_type == ScopeType.Generator:
   224	            # Generators are special because they inherit the outer scope.
   225	            initial_state = self._scope().branch_stmts[-1].branches[-1]
   226	        self.scopes.append(Scope([BranchStatement(initial_state)], scope_type))
   227
   228	    def exit_scope(self) -> None:
   229	        self.scopes.pop()
   230
   231	    def in_scope(self, scope_type: ScopeType) -> bool:
   232	        return self._scope().scope_type == scope_type
   233
   234	    def start_branch_statement(self) -> None:
   235	        assert len(self._scope().branch_stmts) > 0
   236	        self._scope().branch_stmts.append(
   237	            BranchStatement(self._scope().branch_stmts[-1].branches[-1])
   238	        )
   239
   240	    def next_branch(self) -> None:
   241	        assert len(self._scope().branch_stmts) > 1
   242	        self._scope().branch_stmts[-1].next_branch()
   243
   244	    def end_branch_statement(self) -> None:
   245	        assert len(self._scope().branch_stmts) > 1
   246	        result = self._scope().branch_stmts.pop().done()
   247	        self._scope().branch_stmts[-1].record_nested_branch(result)
   248
   249	    def skip_branch(self) -> None:
   250	        # Only skip branch if we're outside of "root" branch statement.
   251	        if len(self._scope().branch_stmts) > 1 and not self.disable_branch_skip:
   252	            self._scope().branch_stmts[-1].skip_branch()
   253
   254	    def record_definition(self, name: str) -> None:
   255	        assert len(self.scopes) > 0
   256	        assert len(self.scopes[-1].branch_stmts) > 0
   257	        self._scope().branch_stmts[-1].record_definition(name)
   258
   259	    def delete_var(self, name: str) -> None:
   260	        assert len(self.scopes) > 0
   261	        assert len(self.scopes[-1].branch_stmts) > 0
   262	        self._scope().branch_stmts[-1].delete_var(name)
   263
   264	    def record_undefined_ref(self, o: NameExpr) -> None:
   265	        """Records an undefined reference. These can later be retrieved via `pop_undefined_ref`."""
   266	        assert len(self.scopes) > 0
   267	        self._scope().record_undefined_ref(o)
   268
   269	    def pop_undefined_ref(self, name: str) -> set[NameExpr]:
   270	        """If name has previously been reported as undefined, the NameExpr that was called will be returned."""
   271	        assert len(self.scopes) > 0
   272	        return self._scope().pop_undefined_ref(name)
   273
   274	    def is_possibly_undefined(self, name: str) -> bool:
   275	        assert len(self._scope().branch_stmts) > 0
   276	        # A variable is undefined if it's in a set of `may_be_defined` but not in `must_be_defined`.
   277	        return self._scope().branch_stmts[-1].is_possibly_undefined(name)
   278
   279	    def is_defined_in_different_branch(self, name: str) -> bool:
   280	        """This will return true if a variable is defined in a branch that's not the current branch."""
   281	        assert len(self._scope().branch_stmts) > 0
   282	        stmt = self._scope().branch_stmts[-1]
   283	        if not stmt.is_undefined(name):
   284	            return False
   285	        for stmt in self._scope().branch_stmts:
   286	            if stmt.is_defined_in_a_branch(name):
   287	                return True
   288	        return False
   289
   290	    def is_undefined(self, name: str) -> bool:
   291	        assert len(self._scope().branch_stmts) > 0
   292	        return self._scope().branch_stmts[-1].is_undefined(name)

   300	class PossiblyUndefinedVariableVisitor(ExtendedTraverserVisitor):
   301	    """Detects the following cases:
   302	    - A variable that's defined only part of the time.
   303	    - If a variable is used before definition
   304
   305	    An example of a partial definition:
   306	    if foo():
   307	        x = 1
   308	    print(x)  # Error: "x" may be undefined.
   309
   310	    Example of a used before definition:
   311	    x = y
   312	    y: int = 2
   313
   314	    Note that this code does not detect variables not defined in any of the branches -- that is
   315	    handled by the semantic analyzer.
   316	    """
   317
   318	    def __init__(
   319	        self,
   320	        msg: MessageBuilder,
   321	        type_map: dict[Expression, Type],
   322	        options: Options,
   323	        names: SymbolTable,
   324	    ) -> None:
   325	        self.msg = msg
   326	        self.type_map = type_map
   327	        self.options = options
   328	        self.builtins = SymbolTable()
   329	        builtins_mod = names.get("__builtins__", None)
   330	        if builtins_mod:
   331	            assert isinstance(builtins_mod.node, MypyFile)
   332	            self.builtins = builtins_mod.node.names
   333	        self.loops: list[Loop] = []
   334	        self.try_depth = 0
   335	        self.tracker = DefinedVariableTracker()
   336	        for name in implicit_module_attrs:
   337	            self.tracker.record_definition(name)
   338
   339	    def var_used_before_def(self, name: str, context: Context) -> None:
   340	        if self.msg.errors.is_error_code_enabled(errorcodes.USED_BEFORE_DEF):
   341	            self.msg.var_used_before_def(name, context)
   342
   343	    def variable_may_be_undefined(self, name: str, context: Context) -> None:
   344	        if self.msg.errors.is_error_code_enabled(errorcodes.POSSIBLY_UNDEFINED):
   345	            self.msg.variable_may_be_undefined(name, context)
   346
   347	    def process_definition(self, name: str) -> None:
   348	        # Was this name previously used? If yes, it's a used-before-definition error.
   349	        if not self.tracker.in_scope(ScopeType.Class):
   350	            refs = self.tracker.pop_undefined_ref(name)
   351	            for ref in refs:
   352	                if self.loops:
   353	                    self.variable_may_be_undefined(name, ref)
   354	                else:
   355	                    self.var_used_before_def(name, ref)
   356	        else:
   357	            # Errors in class scopes are caught by the semantic analyzer.
   358	            pass
   359	        self.tracker.record_definition(name)

   361	    def visit_global_decl(self, o: GlobalDecl) -> None:
   362	        for name in o.names:
   363	            self.process_definition(name)
   364	        super().visit_global_decl(o)

   366	    def visit_nonlocal_decl(self, o: NonlocalDecl) -> None:

   372	        if isinstance(lvalue, NameExpr):
   373	            self.process_definition(lvalue.name)

   375	            self.process_lvalue(lvalue.expr)

   389	    def visit_if_stmt(self, o: IfStmt) -> None:
   390	        for e in o.expr:
   391	            e.accept(self)
   392	        self.tracker.start_branch_statement()
   393	        for b in o.body:
   394	            if b.is_unreachable:
   395	                continue
   396	            b.accept(self)
   397	            self.tracker.next_branch()
   398	        if o.else_body:
   399	            if not o.else_body.is_unreachable:
   400	                o.else_body.accept(self)
   401	            else:
   402	                self.tracker.skip_branch()

   405	    def visit_match_stmt(self, o: MatchStmt) -> None:
   406	        o.subject.accept(self)
   407	        self.tracker.start_branch_statement()
   408	        for i in range(len(o.patterns)):
   409	            pattern = o.patterns[i]
   410	            pattern.accept(self)
   411	            guard = o.guards[i]
   412	            if guard is not None:
   413	                guard.accept(self)
   414	            if not o.bodies[i].is_unreachable:
   415	                o.bodies[i].accept(self)
   416	            else:
   417	                self.tracker.skip_branch()
   418	            is_catchall = infer_pattern_value(pattern) == ALWAYS_TRUE
   419	            if not is_catchall:
   420	                self.tracker.next_branch()
   421	        self.tracker.end_branch_statement()

   427	    def visit_func(self, o: FuncItem) -> None:
   428	        if o.is_dynamic() and not self.options.check_untyped_defs:
   429	            return

   458	    def visit_for_stmt(self, o: ForStmt) -> None:
   459	        o.expr.accept(self)
   460	        self.process_lvalue(o.index)
   461	        o.index.accept(self)
   462	        self.tracker.start_branch_statement()
   463	        loop = Loop()
   464	        self.loops.append(loop)

   466	        self.tracker.next_branch()
   467	        self.tracker.end_branch_statement()
   468	        if o.else_body is not None:
   469	            # If the loop has a `break` inside, `else` is executed conditionally.
   470	            # If the loop doesn't have a `break` either the function will return or
   471	            # execute the `else`.
   472	            has_break = loop.has_break
   473	            if has_break:
   474	                self.tracker.start_branch_statement()
   475	                self.tracker.next_branch()
   476	            o.else_body.accept(self)
   477	            if has_break:
   478	                self.tracker.end_branch_statement()
   479	        self.loops.pop()

   515	    def visit_try_stmt(self, o: TryStmt) -> None:
   516	        """
   517	        Note that finding undefined vars in `finally` requires different handling from
   518	        the rest of the code. In particular, we want to disallow skipping branches due to jump
   519	        statements in except/else clauses for finally but not for other cases. Imagine a case like:
   520	        def f() -> int:
   521	            try:
   522	                x = 1
   523	            except:
   524	                # This jump statement needs to be handled differently depending on whether or
   525	                # not we're trying to process `finally` or not.
   526	                return 0
   527	            finally:
   528	                # `x` may be undefined here.
   529	                pass
   530	            # `x` is always defined here.
   531	            return x
   532	        """
   533	        self.try_depth += 1
   534	        if o.finally_body is not None:
   535	            # In order to find undefined vars in `finally`, we need to
   536	            # process try/except with branch skipping disabled. However, for the rest of the code
   537	            # after finally, we need to process try/except with branch skipping enabled.
   538	            # Therefore, we need to process try/finally twice.
   539	            # Because processing is not idempotent, we should make a copy of the tracker.
   540	            old_tracker = self.tracker.copy()
   541	            self.tracker.disable_branch_skip = True
   542	            self.process_try_stmt(o)
   543	            self.tracker = old_tracker
   544	        self.process_try_stmt(o)
   545	        self.try_depth -= 1
   546
   547	    def process_try_stmt(self, o: TryStmt) -> None:
   548	        """
   549	        Processes try statement decomposing it into the following:
   550	        if ...:
   551	            body
   552	            else_body
   553	        elif ...:
   554	            except 1
   555	        elif ...:
   556	            except 2
   557	        else:
   558	            except n
   559	        finally
   560	        """
   561	        self.tracker.start_branch_statement()
   562	        o.body.accept(self)
   563	        if o.else_body is not None:
   564	            o.else_body.accept(self)
   565	        if len(o.handlers) > 0:
   566	            assert len(o.handlers) == len(o.vars) == len(o.types)
   567	            for i in range(len(o.handlers)):
   568	                self.tracker.next_branch()
   569	                exc_type = o.types[i]
   570	                if exc_type is not None:
   571	                    exc_type.accept(self)
   572	                var = o.vars[i]
   573	                if var is not None:
   574	                    self.process_definition(var.name)
   575	                    var.accept(self)
   576	                o.handlers[i].accept(self)
   577	                if var is not None:
   578	                    self.tracker.delete_var(var.name)
   579	        self.tracker.end_branch_statement()
   580
   581	        if o.finally_body is not None:
   582	            o.finally_body.accept(self)
   583

   586	        self.tracker.start_branch_statement()
   587	        loop
