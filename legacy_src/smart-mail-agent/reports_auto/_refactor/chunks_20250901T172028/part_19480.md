   911	class PDFCrossReferenceSubsection(PDFObject):
   912	    def __init__(self, firstentrynumber, idsequence):
   913	        self.firstentrynumber = firstentrynumber
   914	        self.idsequence = idsequence
   915	    def format(self, document):
   916	        """id sequence should represent contiguous object nums else error. free numbers not supported (yet)"""
   917	        firstentrynumber = self.firstentrynumber
   918	        idsequence = self.idsequence
   919	        entries = list(idsequence)
   920	        nentries = len(idsequence)
   921	        # special case: object number 0 is always free
   922	        taken = {}
   923	        if firstentrynumber==0:
   924	            taken[0] = "standard free entry"
   925	            nentries = nentries+1
   926	            entries.insert(0, "0000000000 65535 f ")
   927	        idToNV = document.idToObjectNumberAndVersion
   928	        idToOffset = document.idToOffset
   929	        lastentrynumber = firstentrynumber+nentries-1
   930	        for id in idsequence:
   931	            (num, version) = idToNV[id]
   932	            if num in taken:
   933	                raise ValueError("object number collision %s %s %s" % (num, repr(id), repr(taken[id])))
   934	            if num>lastentrynumber or num<firstentrynumber:
   935	                raise ValueError("object number %s not in range %s..%s" % (num, firstentrynumber, lastentrynumber))
   936	            # compute position in list
   937	            rnum = num-firstentrynumber
   938	            taken[num] = id
   939	            offset = idToOffset[id]
   940	            entries[num] = '%0.10d %0.5d n ' % (offset, version)
   941	        # now add the initial line
   942	        firstline = "%s %s" % (firstentrynumber, nentries)
   943	        entries.insert(0, firstline)
   944	        # make sure it ends with \n
   945	        entries.append("")
   946	        return pdfdocEnc('\n'.join(entries))
