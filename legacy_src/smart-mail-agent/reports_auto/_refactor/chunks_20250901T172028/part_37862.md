462	        # Ignore names which cannot be hooks.
   463	        if name == "pytest_plugins":
   464	            return None

   466	        opts = super().parse_hookimpl_opts(plugin, name)
   467	        if opts is not None:

   474	        # Collect unmarked hooks as long as they have the `pytest_' prefix.

   476	            method, "impl", ("tryfirst", "trylast", "optionalhook", "hookwrapper")
   477	        )
   478	        return cast(HookimplOpts, legacy)

   547	    def _set_initial_conftests(
   548	        self,
   549	        args: Sequence[str | pathlib.Path],
   550	        pyargs: bool,
   551	        noconftest: bool,
   552	        rootpath: pathlib.Path,
   553	        confcutdir: pathlib.Path | None,
   554	        invocation_dir: pathlib.Path,
   555	        importmode: ImportMode | str,
   556	        *,
   557	        consider_namespace_packages: bool,
   558	    ) -> None:
   559	        """Load initial conftest files given a preparsed "namespace".
   560
   561	        As conftest files may add their own command line options which have
   562	        arguments ('--my-opt somepath') we might get some false positives.
   563	        All builtin and 3rd party plugins will have been loaded, however, so
   564	        common options will not confuse our logic here.
   565	        """
   566	        self._confcutdir = (
   567	            absolutepath(invocation_dir / confcutdir) if confcutdir else None
   568	        )
   569	        self._noconftest = noconftest
   570	        self._using_pyargs = pyargs
   571	        foundanchor = False
   572	        for initial_path in args:
   573	            path = str(initial_path)
   574	            # remove node-id syntax
   575	            i = path.find("::")
   576	            if i != -1:
   577	                path = path[:i]
   578	            anchor = absolutepath(invocation_dir / path)
   579
   580	            # Ensure we do not break if what appears to be an anchor
   581	            # is in fact a very long option (#10169, #11394).
   582	            if safe_exists(anchor):
   583	                self._try_load_conftest(
   584	                    anchor,
   585	                    importmode,
   586	                    rootpath,
   587	                    consider_namespace_packages=consider_namespace_packages,
   588	                )
   589	                foundanchor = True
   590	        if not foundanchor:
   591	            self._try_load_conftest(
   592	                invocation_dir,
   593	                importmode,
   594	                rootpath,
   595	                consider_namespace_packages=consider_namespace_packages,
   596	            )

   598	    def _is_in_confcutdir(self, path: pathlib.Path) -> bool:
   599	        """Whether to consider the given path to load conftests from."""
   600	        if self._confcutdir is None:
   601	            return True
   602	        # The semantics here are literally:
   603	        #   Do not load a conftest if it is found upwards from confcut dir.
   604	        # But this is *not* the same as:
   605	        #   Load only conftests from confcutdir or below.
   606	        # At first glance they might seem the same thing, however we do support use cases where
   607	        # we want to load conftests that are not found in confcutdir or below, but are found
   608	        # in completely different directory hierarchies like packages installed
   609	        # in out-of-source trees.
   610	        # (see #9767 for a regression where the logic was inverted).
   611	        return path not in self._confcutdir.parents

   613	    def _try_load_conftest(
   614	        self,
   615	        anchor: pathlib.Path,
   616	        importmode: str | ImportMode,
   617	        rootpath: pathlib.Path,
   618	        *,
   619	        consider_namespace_packages: bool,
   620	    ) -> None:
   621	        self._loadconftestmodules(
   622	            anchor,
   623	            importmode,
   624	            rootpath,
   625	            consider_namespace_packages=consider_namespace_packages,
   626	        )
   627	        # let's also consider test* subdirs
   628	        if anchor.is_dir():
   629	            for x in anchor.glob("test*"):
   630	                if x.is_dir():
   631	                    self._loadconftestmodules(
   632	                        x,
   633	                        importmode,
   634	                        rootpath,
   635	                        consider_namespace_packages=consider_namespace_packages,
   636	                    )

   638	    def _loadconftestmodules(
   639	        self,
   640	        path: pathlib.Path,
   641	        importmode: str | ImportMode,
   642	        rootpath: pathlib.Path,
   643	        *,
   644	        consider_namespace_packages: bool,
   645	    ) -> None:
   646	        if self._noconftest:
   647	            return
   648
   649	        directory = self._get_directory(path)
   650
   651	        # Optimization: avoid repeated searches in the same directory.
   652	        # Assumes always called with same importmode and rootpath.
   653	        if directory in self._dirpath2confmods:
   654	            return
   655
   656	        clist = []
   657	        for parent in reversed((directory, *directory.parents)):
   658	            if self._is_in_confcutdir(parent):
   659	                conftestpath = parent / "conftest.py"
   660	                if conftestpath.is_file():
   661	                    mod = self._importconftest(
   662	                        conftestpath,
   663	                        importmode,
   664	                        rootpath,
   665	                        consider_namespace_packages=consider_namespace_packages,
   666	                    )
   667	                    clist.append(mod)
   668	        self._dirpath2confmods[directory] = clist
   669
   670	    def _getconftestmodules(self, path: pathlib.Path) -> Sequence[types.ModuleType]:
   671	        directory = self._get_directory(path)
   672	        return self._dirpath2confmods.get(directory, ())

   687	    def _importconftest(
   688	        self,
   689	        conftestpath: pathlib.Path,
   690	        importmode: str | ImportMode,
   691	        rootpath: pathlib.Path,
   692	        *,
   693	        consider_namespace_packages: bool,
   694	    ) -> types.ModuleType:
   695	        conftestpath_plugin_name = str(conftestpath)
   696	        existing = self.get_plugin(conftestpath_plugin_name)
   697	        if existing is not None:
   698	            return cast(types.ModuleType, existing)
   699
   700	        # conftest.py files there are not in a Python package all have module
   701	        # name "conftest", and thus conflict with each other. Clear the existing
   702	        # before loading the new one, otherwise the existing one will be
   703	        # returned from the module cache.
   704	        pkgpath = resolve_package_path(conftestpath)
   705	        if pkgpath is None:
   706	            try:
   707	                del sys.modules[conftestpath.stem]
   708	            except KeyError:
   709	                pass
   710
   711	        try:
   712	            mod = import_path(
   713	                conftestpath,
   714	                mode=importmode,
   715	                root=rootpath,
   716	                consider_namespace_packages=consider_namespace_packages,
   717	            )
   718	        except Exception as e:
   719	            assert e.__traceback__ is not None
   720	            raise ConftestImportFailure(conftestpath, cause=e) from e
   721
   722	        self._check_non_top_pytest_plugins(mod, conftestpath)
   723
   724	        self._conftest_plugins.add(mod)
   725	        dirpath = conftestpath.parent
   726	        if dirpath in self._dirpath2confmods:
   727	            for path, mods in self._dirpath2confmods.items():
   728	                if dirpath in path.parents or path == dirpath:
   729	                    if mod in mods:
   730	                        raise AssertionError(
   731	                            f"While trying to load conftest path {conftestpath!s}, "
   732	                            f"found that the module {mod} is already loaded with path {mod.__file__}. "
   733	                            "This is not supposed to happen. Please report this issue to pytest."
   734	                        )
   735	                    mods.append(mod)
   736	        self.trace(f"loading conftestmodule {mod!r}")
   737	        self.consider_conftest(mod, registration_name=conftestpath_plugin_name)
   738	        return mod

   741	        self,
   742	        mod: types.ModuleType,
   743	        conftestpath: pathlib.Path,

   745	        if (
   746	            hasattr(mod, "pytest_plugins")

   767	        self, args: Sequence[str], *, exclude_only: bool = False
   768	    ) -> None:
   769	        """:meta private:"""
   770	        i = 0
   771	        n = len(args)
   772	        while i < n:
   773	            opt = args[i]
   774	            i += 1
   775	            if isinstance(opt, str):
   776	                if opt == "-p":
   777	                    try:
   778	                        parg = args[i]
   779	                    except IndexError:
   780	                        return
   781	                    i += 1
   782	                elif opt.startswith("-p"):
   783	                    parg = opt[2:]
   784	                else:
   785	                    continue
   786	                parg = parg.strip()
   787	                if exclude_only and not parg.startswith("no:"):
   788	                    continue
   789	                self.consider_pluginarg(parg)

   791	    def consider_pluginarg(self, arg: str) -> None:
   792	        """:meta private:"""
   793	        if arg.startswith("no:"):
   794	            name = arg[3:]
   795	            if name in essential_plugins:
   796	                raise UsageError(f"plugin {name} cannot be disabled")
   797
   798	            # PR #4304: remove stepwise if cacheprovider is blocked.
   799	            if name == "cacheprovider":
   800	                self.set_blocked("stepwise")
   801	                self.set_blocked("pytest_stepwise")
   802
   803	            self.set_blocked(name)
   804	            if not name.startswith("pytest_"):
   805	                self.set_blocked("pytest_" + name)
   806	        else:
   807	            name = arg
   808	            # Unblock the plugin.
   809	            self.unblock(name)
   810	            if not name.startswith("pytest_"):
   811	                self.unblock("pytest_" + name)
   812	            self.import_plugin(arg, consider_entry_points=True)

   835	    def import_plugin(self, modname: str, consider_entry_points: bool = False) -> None:
   836	        """Import a plugin with ``modname``.
   837
   838	        If ``consider_entry_points`` is True, entry point names are also
   839	        considered to find a plugin.
   840	        """
   841	        # Most often modname refers to builtin modules, e.g. "pytester",
   842	        # "terminal" or "capture".  Those plugins are registered under their
   843	        # basename for historic purposes but must be imported with the
   844	        # _pytest prefix.
   845	        assert isinstance(modname, str), (
   846	            f"module name as text required, got {modname!r}"
   847	        )
   848	        if self.is_blocked(modname) or self.get_plugin(modname) is not None:
   849	            return
   850
   851	        importspec = "_pytest." + modname if modname in builtin_plugins else modname
   852	        self.rewrite_hook.mark_rewrite(importspec)
   853
   854	        if consider_entry_points:
   855	            loaded = self.load_setuptools_entrypoints("pytest11", name=modname)
   856	            if loaded:
   857	                return
   858
   859	        try:
   860	            __import__(importspec)
   861	        except ImportError as e:
   862	            raise ImportError(
   863	                f'Error importing plugin "{modname}": {e.args[0]}'
   864	            ).with_traceback(e.__traceback__) from e
   865
   866	        except Skipped as e:
   867	            self.skipped_plugins.append((modname, e.msg or ""))
   868	        else:
   869	            mod = sys.modules[importspec]
   870	            self.register(mod, modname)

   873	def _get_plugin_specs_as_list(
   874	    specs: None | types.ModuleType | str | Sequence[str],
   875	) -> list[str]:
   876	    """Parse a plugins specification into a list of plugin names."""
   877	    # None means empty.
   878	    if specs is None:
   879	        return []
   880	    # Workaround for #3899 - a submodule which happens to be called "pytest_plugins".
   881	    if isinstance(specs, types.ModuleType):
   882	        return []
   883	    # Comma-separated list.
   884	    if isinstance(specs, str):
   885	        return specs.split(",") if specs else []
   886	    # Direct specification.
   887	    if isinstance(specs, collections.abc.Sequence):
   888	        return list(specs)
   889	    raise UsageError(
   890	        f"Plugins may be specified as a sequence or a ','-separated string of plugin names. Got: {specs!r}"
   891	    )
