   699	    def method_fullname(self, object_type: Type, method_name: str) -> str | None:
   700	        """Convert a method name to a fully qualified name, based on the type of the object that
   701	        it is invoked on. Return `None` if the name of `object_type` cannot be determined.
   702	        """
   703	        object_type = get_proper_type(object_type)
   704
   705	        if isinstance(object_type, CallableType) and object_type.is_type_obj():
   706	            # For class method calls, object_type is a callable representing the class object.
   707	            # We "unwrap" it to a regular type, as the class/instance method difference doesn't
   708	            # affect the fully qualified name.
   709	            object_type = get_proper_type(object_type.ret_type)
   710	        elif isinstance(object_type, TypeType):
   711	            object_type = object_type.item
   712
   713	        type_name = None
   714	        if isinstance(object_type, Instance):
   715	            type_name = object_type.type.fullname
   716	        elif isinstance(object_type, (TypedDictType, LiteralType)):
   717	            info = object_type.fallback.type.get_containing_type_info(method_name)
   718	            type_name = info.fullname if info is not None else None
   719	        elif isinstance(object_type, TupleType):
   720	            type_name = tuple_fallback(object_type).type.fullname
   721
   722	        if type_name:
   723	            return f"{type_name}.{method_name}"
   724	        else:
   725	            return None

   727	    def always_returns_none(self, node: Expression) -> bool:
   728	        """Check if `node` refers to something explicitly annotated as only returning None."""
   729	        if isinstance(node, RefExpr):
   730	            if self.defn_returns_none(node.node):
   731	                return True
   732	        if isinstance(node, MemberExpr) and node.node is None:  # instance or class attribute
   733	            typ = get_proper_type(self.chk.lookup_type(node.expr))
   734	            if isinstance(typ, Instance):
   735	                info = typ.type
   736	            elif isinstance(typ, CallableType) and typ.is_type_obj():
   737	                ret_type = get_proper_type(typ.ret_type)
   738	                if isinstance(ret_type, Instance):
   739	                    info = ret_type.type
   740	                else:
   741	                    return False
   742	            else:
   743	                return False
   744	            sym = info.get(node.name)
   745	            if sym and self.defn_returns_none(sym.node):
   746	                return True
   747	        return False

   749	    def defn_returns_none(self, defn: SymbolNode | None) -> bool:
   750	        """Check if `defn` can _only_ return None."""
   751	        if isinstance(defn, FuncDef):
   752	            return isinstance(defn.type, CallableType) and isinstance(
   753	                get_proper_type(defn.type.ret_type), NoneType
   754	            )
   755	        if isinstance(defn, OverloadedFuncDef):
   756	            return all(self.defn_returns_none(item) for item in defn.items)
   757	        if isinstance(defn, Var):
   758	            typ = get_proper_type(defn.type)
   759	            if (
   760	                not defn.is_inferred
   761	                and isinstance(typ, CallableType)
   762	                and isinstance(get_proper_type(typ.ret_type), NoneType)
   763	            ):
   764	                return True
   765	            if isinstance(typ, Instance):
   766	                sym = typ.type.get("__call__")
   767	                if sym and self.defn_returns_none(sym.node):
   768	                    return True
   769	        return False

   790	    def check_typeddict_call(
   791	        self,
   792	        callee: TypedDictType,
   793	        arg_kinds: list[ArgKind],
   794	        arg_names: Sequence[str | None],
   795	        args: list[Expression],
   796	        context: Context,
   797	        orig_callee: Type | None,
   798	    ) -> Type:
   799	        if args and all(ak in (ARG_NAMED, ARG_STAR2) for ak in arg_kinds):
   800	            # ex: Point(x=42, y=1337, **extras)
   801	            # This is a bit ugly, but this is a price for supporting all possible syntax
   802	            # variants for TypedDict constructors.
   803	            kwargs = zip([StrExpr(n) if n is not None else None for n in arg_names], args)
   804	            result = self.validate_typeddict_kwargs(kwargs=kwargs, callee=callee)
   805	            if result is not None:
   806	                validated_kwargs, always_present_keys = result
   807	                return self.check_typeddict_call_with_kwargs(
   808	                    callee, validated_kwargs, context, orig_callee, always_present_keys
   809	                )
   810	            return AnyType(TypeOfAny.from_error)
   811
   812	        if len(args) == 1 and arg_kinds[0] == ARG_POS:
   813	            unique_arg = args[0]
   814	            if isinstance(unique_arg, DictExpr):
   815	                # ex: Point({'x': 42, 'y': 1337, **extras})
   816	                return self.check_typeddict_call_with_dict(
   817	                    callee, unique_arg.items, context, orig_callee
   818	                )
   819	            if isinstance(unique_arg, CallExpr) and isinstance(unique_arg.analyzed, DictExpr):
   820	                # ex: Point(dict(x=42, y=1337, **extras))
   821	                return self.check_typeddict_call_with_dict(
   822	                    callee, unique_arg.analyzed.items, context, orig_callee
   823	                )
   824
   825	        if not args:
   826	            # ex: EmptyDict()
   827	            return self.check_typeddict_call_with_kwargs(callee, {}, context, orig_callee, set())
   828
   829	        self.chk.fail(message_registry.INVALID_TYPEDDICT_ARGS, context)
   830	        return AnyType(TypeOfAny.from_error)

   832	    def validate_typeddict_kwargs(
   833	        self, kwargs: Iterable[tuple[Expression | None, Expression]], callee: TypedDictType
   834	    ) -> tuple[dict[str, list[Expression]], set[str]] | None:
   835	        # All (actual or mapped from ** unpacks) expressions that can match given key.
   836	        result = defaultdict(list)
   837	        # Keys that are guaranteed to be present no matter what (e.g. for all items of a union)
   838	        always_present_keys = set()
   839	        # Indicates latest encountered ** unpack among items.
   840	        last_star_found = None
   841
   842	        for item_name_expr, item_arg in kwargs:
   843	            if item_name_expr:
   844	                key_type = self.accept(item_name_expr)
   845	                values = try_getting_str_literals(item_name_expr, key_type)
   846	                literal_value = None
   847	                if values and len(values) == 1:
   848	                    literal_value = values[0]
   849	                if literal_value is None:
   850	                    key_context = item_name_expr or item_arg
   851	                    self.chk.fail(
   852	                        message_registry.TYPEDDICT_KEY_MUST_BE_STRING_LITERAL,
   853	                        key_context,
   854	                        code=codes.LITERAL_REQ,
   855	                    )
   856	                    return None
   857	                else:
   858	                    # A directly present key unconditionally shadows all previously found
   859	                    # values from ** items.
   860	                    # TODO: for duplicate keys, type-check all values.
   861	                    result[literal_value] = [item_arg]
   862	                    always_present_keys.add(literal_value)
   863	            else:
   864	                last_star_found = item_arg
   865	                if not self.validate_star_typeddict_item(
   866	                    item_arg, callee, result, always_present_keys
   867	                ):
   868	                    return None
   869	        if self.chk.options.extra_checks and last_star_found is not None:
   870	            absent_keys = []
   871	            for key in callee.items:
   872	                if key not in callee.required_keys and key not in result:
   873	                    absent_keys.append(key)
   874	            if absent_keys:
   875	                # Having an optional key not explicitly declared by a ** unpacked
   876	                # TypedDict is unsafe, it may be an (incompatible) subtype at runtime.
   877	                # TODO: catch the cases where a declared key is overridden by a subsequent
   878	                # ** item without it (and not again overridden with complete ** item).
   879	                self.msg.non_required_keys_absent_with_star(absent_keys, last_star_found)
   880	        return result, always_present_keys

   882	    def validate_star_typeddict_item(
   883	        self,
   884	        item_arg: Expression,
   885	        callee: TypedDictType,
   886	        result: dict[str, list[Expression]],
   887	        always_present_keys: set[str],
   888	    ) -> bool:
   889	        """Update keys/expressions from a ** expression in TypedDict constructor.
   890
   891	        Note `result` and `always_present_keys` are updated in place. Return true if the
   892	        expression `item_arg` may valid in `callee` TypedDict context.
   893	        """
   894	        inferred = get_proper_type(self.accept(item_arg, type_context=callee))
   895	        possible_tds = []
   896	        if isinstance(inferred, TypedDictType):
   897	            possible_tds = [inferred]
   898	        elif isinstance(inferred, UnionType):
   899	            for item in get_proper_types(inferred.relevant_items()):
   900	                if isinstance(item, TypedDictType):
   901	                    possible_tds.append(item)
   902	                elif not self.valid_unpack_fallback_item(item):
   903	                    self.msg.unsupported_target_for_star_typeddict(item, item_arg)
   904	                    return False
   905	        elif not self.valid_unpack_fallback_item(inferred):
   906	            self.msg.unsupported_target_for_star_typeddict(inferred, item_arg)
   907	            return False
   908	        all_keys: set[str] = set()
   909	        for td in possible_tds:
   910	            all_keys |= td.items.keys()
   911	        for key in all_keys:
   912	            arg = TempNode(
   913	                UnionType.make_union([td.items[key] for td in possible_tds if key in td.items])
   914	            )
   915	            arg.set_line(item_arg)
   916	            if all(key in td.required_keys for td in possible_tds):
   917	                always_present_keys.add(key)
   918	                # Always present keys override previously found values. This is done
   919	                # to support use cases like `Config({**defaults, **overrides})`, where
   920	                # some `overrides` types are narrower that types in `defaults`, and
   921	                # former are too wide for `Config`.
   922	                if result[key]:
   923	                    first = result[key][0]
   924	                    if not isinstance(first, TempNode):
   925	                        # We must always preserve any non-synthetic values, so that
   926	                        # we will accept them even if they are shadowed.
   927	                        result[key] = [first, arg]
   928	                    else:
   929	                        result[key] = [arg]
   930	                else:
   931	                    result[key] = [arg]
   932	            else:
   933	                # If this key is not required at least in some item of a union
   934	                # it may not shadow previous item, so we need to type check both.
   935	                result[key].append(arg)
   936	        return True

   938	    def valid_unpack_fallback_item(self, typ: ProperType) -> bool:
   939	        if isinstance(typ, AnyType):
   940	            return True
   941	        if not isinstance(typ, Instance) or not typ.type.has_base("typing.Mapping"):
   942	            return False
   943	        mapped = map_instance_to_supertype(typ, self.chk.lookup_typeinfo("typing.Mapping"))
   944	        return all(isinstance(a, AnyType) for a in get_proper_types(mapped.args))

   980	        """Construct a reasonable type for a TypedDict type in runtime context.
   981
   982	        If it appears as a callee, it will be special-cased anyway, e.g. it is
   983	        also allowed to accept a single positional argument if it is a dict literal.
   984

   989	        target = info.special_alias.target
   990	        assert isinstance(target, ProperType) and isinstance(target, TypedDictType)
   991	        return self.typeddict_callable_from_context(target, info.defn.type_vars)
