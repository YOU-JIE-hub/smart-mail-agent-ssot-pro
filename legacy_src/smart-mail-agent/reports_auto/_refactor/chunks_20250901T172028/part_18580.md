   226	    def record_protocol_subtype_check(self, left_type: TypeInfo, right_type: TypeInfo) -> None:
   227	        assert right_type.is_protocol
   228	        self._rechecked_types.add(left_type)
   229	        self._attempted_protocols.setdefault(left_type.fullname, set()).add(right_type.fullname)
   230	        self._checked_against_members.setdefault(left_type.fullname, set()).update(
   231	            right_type.protocol_members
   232	        )

   234	    def _snapshot_protocol_deps(self) -> dict[str, set[str]]:
   235	        """Collect protocol attribute dependencies found so far from registered subtype checks.
   236
   237	        There are three kinds of protocol dependencies. For example, after a subtype check:
   238
   239	            x: Proto = C()
   240
   241	        the following dependencies will be generated:
   242	            1. ..., <SuperProto[wildcard]>, <Proto[wildcard]> -> <Proto>
   243	            2. ..., <B.attr>, <C.attr> -> <C> [for every attr in Proto members]
   244	            3. <C> -> Proto  # this one to invalidate the subtype cache
   245
   246	        The first kind is generated immediately per-module in deps.py (see also an example there
   247	        for motivation why it is needed). While two other kinds are generated here after all
   248	        modules are type checked and we have recorded all the subtype checks. To understand these
   249	        two kinds, consider a simple example:
   250
   251	            class A:
   252	                def __iter__(self) -> Iterator[int]:
   253	                    ...
   254
   255	            it: Iterable[int] = A()
   256
   257	        We add <a.A.__iter__> -> <a.A> to invalidate the assignment (module target in this case),
   258	        whenever the signature of a.A.__iter__ changes. We also add <a.A> -> typing.Iterable,
   259	        to invalidate the subtype caches of the latter. (Note that the same logic applies to
   260	        proper subtype checks, and calculating meets and joins, if this involves calling
   261	        'subtypes.is_protocol_implementation').
   262	        """
   263	        deps: dict[str, set[str]] = {}
   264	        for info in self._rechecked_types:
   265	            for attr in self._checked_against_members[info.fullname]:
   266	                # The need for full MRO here is subtle, during an update, base classes of
   267	                # a concrete class may not be reprocessed, so not all <B.x> -> <C.x> deps
   268	                # are added.
   269	                for base_info in info.mro[:-1]:
   270	                    trigger = make_trigger(f"{base_info.fullname}.{attr}")
   271	                    if "typing" in trigger or "builtins" in trigger:
   272	                        # TODO: avoid everything from typeshed
   273	                        continue
   274	                    deps.setdefault(trigger, set()).add(make_trigger(info.fullname))
   275	            for proto in self._attempted_protocols[info.fullname]:
   276	                trigger = make_trigger(info.fullname)
   277	                if "typing" in trigger or "builtins" in trigger:
   278	                    continue
   279	                # If any class that was checked against a protocol changes,
   280	                # we need to reset the subtype cache for the protocol.
   281	                #
   282	                # Note: strictly speaking, the protocol doesn't need to be
   283	                # re-checked, we only need to reset the cache, and its uses
   284	                # elsewhere are still valid (unless invalidated by other deps).
   285	                deps.setdefault(trigger, set()).add(proto)
   286	        return deps
   287
   288	    def update_protocol_deps(self, second_map: dict[str, set[str]] | None = None) -> None:
   289	        """Update global protocol dependency map.
   290
   291	        We update the global map incrementally, using a snapshot only from recently
   292	        type checked types. If second_map is given, update it as well. This is currently used
   293	        by FineGrainedBuildManager that maintains normal (non-protocol) dependencies.
   294	        """
   295	        assert self.proto_deps is not None, "This should not be called after failed cache load"
   296	        new_deps = self._snapshot_protocol_deps()
   297	        for trigger, targets in new_deps.items():
   298	            self.proto_deps.setdefault(trigger, set()).update(targets)
   299	        if second_map is not None:
   300	            for trigger, targets in new_deps.items():
   301	                second_map.setdefault(trigger, set()).update(targets)
   302	        self._rechecked_types.clear()
   303	        self._attempted_protocols.clear()
   304	        self._checked_against_members.clear()
   305
   306	    def add_all_protocol_deps(self, deps: dict[str, set[str]]) -> None:
   307	        """Add all known protocol dependencies to deps.
   308
   309	        This is used by tests and debug output, and also when collecting
   310	        all collected or loaded dependencies as part of build.
   311	        """
   312	        self.update_protocol_deps()  # just in case
   313	        if self.proto_deps is not None:
   314	            for trigger, targets in self.proto_deps.items():
   315	                deps.setdefault(trigger, set()).update(targets)
