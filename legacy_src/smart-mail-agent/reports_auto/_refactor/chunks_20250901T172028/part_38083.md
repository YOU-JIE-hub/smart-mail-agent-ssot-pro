1038	        self,
  1039	        instance: Instance,
  1040	        template: Instance,
  1041	        subtype: Type,
  1042	        protocol: Instance,
  1043	        class_obj: bool = False,
  1044	    ) -> list[Constraint]:
  1045	        """Infer constraints for situations where either 'template' or 'instance' is a protocol.
  1046
  1047	        The 'protocol' is the one of two that is an instance of protocol type, 'subtype'
  1048	        is the type used to bind self during inference. Currently, we just infer constrains for
  1049	        every protocol member type (both ways for settable members).
  1050	        """
  1051	        res = []
  1052	        for member in protocol.type.protocol_members:
  1053	            inst = mypy.subtypes.find_member(member, instance, subtype, class_obj=class_obj)
  1054	            temp = mypy.subtypes.find_member(member, template, subtype)
  1055	            if inst is None or temp is None:
  1056	                if member == "__call__":
  1057	                    continue
  1058	                return []  # See #11020
  1059	            # The above is safe since at this point we know that 'instance' is a subtype
  1060	            # of (erased) 'template', therefore it defines all protocol members
  1061	            if class_obj:
  1062	                # For class objects we must only infer constraints if possible, otherwise it
  1063	                # can lead to confusion between class and instance, for example StrEnum is
  1064	                # Iterable[str] for an instance, but Iterable[StrEnum] for a class object.
  1065	                if not mypy.subtypes.is_subtype(
  1066	                    inst, erase_typevars(temp), ignore_pos_arg_names=True
  1067	                ):
  1068	                    continue
  1069	            # This exception matches the one in typeops.py, see PR #14121 for context.
  1070	            if member == "__call__" and instance.type.is_metaclass(precise=True):
  1071	                continue
  1072	            res.extend(infer_constraints(temp, inst, self.direction))
  1073	            if mypy.subtypes.IS_SETTABLE in mypy.subtypes.get_member_flags(member, protocol):
  1074	                # Settable members are invariant, add opposite constraints
  1075	                res.extend(infer_constraints(temp, inst, neg_op(self.direction)))
  1076	        return res

  1078	    def visit_callable_type(self, template: CallableType) -> list[Constraint]:
  1079	        # Normalize callables before matching against each other.
  1080	        # Note that non-normalized callables can be created in annotations
  1081	        # using e.g. callback protocols.
  1082	        # TODO: check that callables match? Ideally we should not infer constraints
  1083	        # callables that can never be subtypes of one another in given direction.
  1084	        template = template.with_unpacked_kwargs().with_normalized_var_args()
  1085	        extra_tvars = False
  1086	        if isinstance(self.actual, CallableType):
  1087	            res: list[Constraint] = []
  1088	            cactual = self.actual.with_unpacked_kwargs().with_normalized_var_args()
  1089	            param_spec = template.param_spec()
  1090
  1091	            template_ret_type, cactual_ret_type = template.ret_type, cactual.ret_type
  1092	            if template.type_guard is not None and cactual.type_guard is not None:
  1093	                template_ret_type = template.type_guard
  1094	                cactual_ret_type = cactual.type_guard
  1095
  1096	            if template.type_is is not None and cactual.type_is is not None:
  1097	                template_ret_type = template.type_is
  1098	                cactual_ret_type = cactual.type_is
  1099
  1100	            res.extend(infer_constraints(template_ret_type, cactual_ret_type, self.direction))
  1101
  1102	            if param_spec is None:
  1103	                # TODO: Erase template variables if it is generic?
  1104	                if (
  1105	                    type_state.infer_polymorphic
  1106	                    and cactual.variables
  1107	                    and not self.skip_neg_op
  1108	                    # Technically, the correct inferred type for application of e.g.
  1109	                    # Callable[..., T] -> Callable[..., T] (with literal ellipsis), to a generic
  1110	                    # like U -> U, should be Callable[..., Any], but if U is a self-type, we can
  1111	                    # allow it to leak, to be later bound to self. A bunch of existing code
  1112	                    # depends on this old behaviour.
  1113	                    and not any(tv.id.is_self() for tv in cactual.variables)
  1114	                ):
  1115	                    # If the actual callable is generic, infer constraints in the opposite
  1116	                    # direction, and indicate to the solver there are extra type variables
  1117	                    # to solve for (see more details in mypy/solve.py).
  1118	                    res.extend(
  1119	                        infer_constraints(
  1120	                            cactual, template, neg_op(self.direction), skip_neg_op=True
  1121	                        )
  1122	                    )
  1123	                    extra_tvars = True
  1124
  1125	                # We can't infer constraints from arguments if the template is Callable[..., T]
  1126	                # (with literal '...').
  1127	                if not template.is_ellipsis_args:
  1128	                    unpack_present = find_unpack_in_list(template.arg_types)
  1129	                    # When both ParamSpec and TypeVarTuple are present, things become messy
  1130	                    # quickly. For now, we only allow ParamSpec to "capture" TypeVarTuple,
  1131	                    # but not vice versa.
  1132	                    # TODO: infer more from prefixes when possible.
  1133	                    if unpack_present is not None and not cactual.param_spec():
  1134	                        # We need to re-normalize args to the form they appear in tuples,
  1135	                        # for callables we always pack the suffix inside another tuple.
  1136	                        unpack = template.arg_types[unpack_present]
  1137	                        assert isinstance(unpack, UnpackType)
  1138	                        tuple_type = get_tuple_fallback_from_unpack(unpack)
  1139	                        template_types = repack_callable_args(template, tuple_type)
  1140	                        actual_types = repack_callable_args(cactual, tuple_type)
  1141	                        # Now we can use the same general helper as for tuple types.
  1142	                        unpack_constraints = build_constraints_for_simple_unpack(
  1143	                            template_types, actual_types, neg_op(self.direction)
  1144	                        )
  1145	                        res.extend(unpack_constraints)
  1146	                    else:
  1147	                        # TODO: do we need some special-casing when unpack is present in actual
  1148	                        # callable but not in template callable?
  1149	                        res.extend(
  1150	                            infer_callable_arguments_constraints(template, cactual, self.direction)
  1151	                        )
  1152	            else:
  1153	                prefix = param_spec.prefix
  1154	                prefix_len = len(prefix.arg_types)
  1155	                cactual_ps = cactual.param_spec()
  1156
  1157	                if type_state.infer_polymorphic and cactual.variables and not self.skip_neg_op:
  1158	                    # Similar logic to the branch above.
  1159	                    res.extend(
  1160	                        infer_constraints(
  1161	                            cactual, template, neg_op(self.direction), skip_neg_op=True
  1162	                        )
  1163	                    )
  1164	                    extra_tvars = True
  1165
  1166	                # Compare prefixes as well
  1167	                cactual_prefix = cactual.copy_modified(
  1168	                    arg_types=cactual.arg_types[:prefix_len],
  1169	                    arg_kinds=cactual.arg_kinds[:prefix_len],
  1170	                    arg_names=cactual.arg_names[:prefix_len],
  1171	                )
  1172	                res.extend(
  1173	                    infer_callable_arguments_constraints(prefix, cactual_prefix, self.direction)
  1174	                )
  1175
  1176	                param_spec_target: Type | None = None
  1177	                if not cactual_ps:
  1178	                    max_prefix_len = len([k for k in cactual.arg_kinds if k in (ARG_POS, ARG_OPT)])
  1179	                    prefix_len = min(prefix_len, max_prefix_len)
  1180	                    param_spec_target = Parameters(
  1181	                        arg_types=cactual.arg_types[prefix_len:],
  1182	                        arg_kinds=cactual.arg_kinds[prefix_len:],
  1183	                        arg_names=cactual.arg_names[prefix_len:],
  1184	                        variables=cactual.variables if not type_state.infer_polymorphic else [],
  1185	                        imprecise_arg_kinds=cactual.imprecise_arg_kinds,
  1186	                    )
  1187	                else:
  1188	                    if len(param_spec.prefix.arg_types) <= len(cactual_ps.prefix.arg_types):
  1189	                        param_spec_target = cactual_ps.copy_modified(
  1190	                            prefix=Parameters(
  1191	                                arg_types=cactual_ps.prefix.arg_types[prefix_len:],
  1192	                                arg_kinds=cactual_ps.prefix.arg_kinds[prefix_len:],
  1193	                                arg_names=cactual_ps.prefix.arg_names[prefix_len:],
  1194	                                imprecise_arg_kinds=cactual_ps.prefix.imprecise_arg_kinds,
  1195	                            )
  1196	                        )
  1197	                if param_spec_target is not None:
  1198	                    res.append(Constraint(param_spec, self.direction, param_spec_target))
  1199	            if extra_tvars:
  1200	                for c in res:
  1201	                    c.extra_tvars += cactual.variables
  1202	            return res
  1203	        elif isinstance(self.actual, AnyType):
  1204	            param_spec = template.param_spec()
  1205	            any_type = AnyType(TypeOfAny.from_another_any, source_any=self.actual)
  1206	            if param_spec is None:
  1207	                # FIX what if generic
  1208	                res = self.infer_against_any(template.arg_types, self.actual)
  1209	            else:
  1210	                res = [
  1211	                    Constraint(
  1212	                        param_spec,
  1213	                        SUBTYPE_OF,
  1214	                        Parameters([any_type, any_type], [ARG_STAR, ARG_STAR2], [None, None]),
  1215	                    )
  1216	                ]
  1217	            res.extend(infer_constraints(template.ret_type, any_type, self.direction))
  1218	            return res
  1219	        elif isinstance(self.actual, Overloaded):
  1220	            return self.infer_against_overloaded(self.actual, template)
  1221	        elif isinstance(self.actual, TypeType):
  1222	            return infer_constraints(template.ret_type, self.actual.item, self.direction)
  1223	        elif isinstance(self.actual, Instance):
  1224	            # Instances with __call__ method defined are considered structural
  1225	            # subtypes of Callable with a compatible signature.
  1226	            call = mypy.subtypes.find_member(
  1227	                "__call__", self.actual, self.actual, is_operator=True
  1228	            )
  1229	            if call:
  1230	                return infer_constraints(template, call, self.direction)
  1231	            else:
  1232	                return []
  1233	        else:
  1234	            return []

  1247	    def visit_tuple_type(self, template: TupleType) -> list[Constraint]:
  1248	        actual = self.actual
  1249	        unpack_index = find_unpack_in_list(template.items)
  1250	        is_varlength_tuple = (
  1251	            isinstance(actual, Instance) and actual.type.fullname == "builtins.tuple"
  1252	        )
  1253
  1254	        if isinstance(actual, TupleType) or is_varlength_tuple:
  1255	            res: list[Constraint] = []
  1256	            if unpack_index is not None:
  1257	                if is_varlength_tuple:
  1258	                    # Variadic tuple can be only a supertype of a tuple type, but even if
  1259	                    # direction is opposite, inferring something may give better error messages.
  1260	                    unpack_type = template.items[unpack_index]
  1261	                    assert isinstance(unpack_type, UnpackType)
  1262	                    unpacked_type = get_proper_type(unpack_type.type)
  1263	                    if isinstance(unpacked_type, TypeVarTupleType):
  1264	                        res = [
  1265	                            Constraint(type_var=unpacked_type, op=self.direction, target=actual)
  1266	                        ]
  1267	                    else:
  1268	                        assert (
  1269	                            isinstance(unpacked_type, Instance)
  1270	                            and unpacked_type.type.fullname == "builtins.tuple"
  1271	                        )
  1272	                        res = infer_constraints(unpacked_type, actual, self.direction)
  1273	                    assert isinstance(actual, Instance)  # ensured by is_varlength_tuple == True
  1274	                    for i, ti in enumerate(template.items):
  1275	                        if i == unpack_index:
  1276	                            # This one we just handled above.
  1277	                            continue
  1278	                        # For Tuple[T, *Ts, S] <: tuple[X, ...] infer also T <: X and S <: X.
  1279	                        res.extend(infer_constraints(ti, actual.args[0], self.direction))
  1280	                    return res
  1281	                else:
  1282	                    assert isinstance(actual, TupleType)
  1283	                    unpack_constraints = build_constraints_for_simple_unpack(
  1284	                        template.items, actual.items, self.direction
  1285	                    )
  1286	                    actual_items: tuple[Type, ...] = ()
  1287	                    template_items: tuple[Type, ...] = ()
  1288	                    res.extend(unpack_constraints)
  1289	            elif isinstance(actual, TupleType):
  1290	                a_unpack_index = find_unpack_in_list(actual.items)
  1291	                if a_unpack_index is not None:
  1292	                    # The case where template tuple doesn't have an unpack, but actual tuple
  1293	                    # has an unpack. We can infer something if actual unpack is a variadic tuple.
  1294	                    # Tuple[T, S, U] <: tuple[X, *tuple[Y, ...], Z] => T <: X, S <: Y, U <: Z.
  1295	                    a_unpack = actual.items[a_unpack_index]
  1296	                    assert isinstance(a_unpack, UnpackType)
  1297	                    a_unpacked = get_proper_type(a_unpack.type)
  1298	                    if len(actual.items) + 1 <= len(template.items):
  1299	                        a_prefix_len = a_unpack_index
  1300	                        a_suffix_len = len(actual.items) - a_unpack_index - 1
  1301	                        t_prefix, t_middle, t_suffix = split_with_prefix_and_suffix(
  1302	                            tuple(template.items), a_prefix_len, a_suffix_len
  1303	                        )
  1304	                        actual_items = tuple(actual.items[:a_prefix_len])
  1305	                        if a_suffix_len:
  1306	                            actual_items += tuple(actual.items[-a_suffix_len:])
  1307	                        template_items = t_prefix + t_suffix
  1308	                        if isinstance(a_unpacked, Instance):
  1309	                            assert a_unpacked.type.fullname == "builtins.tuple"
  1310	                            for tm in t_middle:
  1311	                                res.extend(
  1312	                                    infer_constraints(tm, a_unpacked.args[0], self.direction)
  1313	                                )
  1314	                    else:
  1315	                        actual_items = ()
  1316	                        template_items = ()
  1317	                else:
  1318	                    actual_items = tuple(actual.items)
  1319	                    template_items = tuple(template.items)
  1320	            else:
  1321	                return res
  1322
  1323	            # Cases above will return if actual wasn't a TupleType.
  1324	            assert isinstance(actual, TupleType)
  1325	            if len(actual_items) == len(template_items):
  1326	                if (
  1327	                    actual.partial_fallback.type.is_named_tuple
  1328	                    and template.partial_fallback.type.is_named_tuple
  1329	                ):
  1330	                    # For named tuples using just the fallbacks usually gives better results.
  1331	                    return res + infer_constraints(
  1332	                        template.partial_fallback, actual.partial_fallback, self.direction
  1333	                    )
  1334	                for i in range(len(template_items)):
  1335	                    res.extend(
  1336	                        infer_constraints(template_items[i], actual_items[i], self.direction)
  1337	                    )
  1338	            return res
  1339	        elif isinstance(actual, AnyType):
  1340	            return self.infer_against_any(template.items, actual)
  1341	        else:
  1342	            return []
