  1035	    def _calcSpanRects(self):
  1036	        """Work out rects for tables which do row and column spanning.
  1037
  1038	        Based on self._spanRanges, which is already known,
  1039	        and the widths which were given or previously calculated,
  1040	        self._spanRects shows the real coords for drawing:
  1041
  1042	            (col, row) -> (x, y, width, height)
  1043
  1044	        for each cell.  Any cell which 'does not exist' as another
  1045	        has spanned over it will get a None entry on the right
  1046	        """
  1047	        spanRects = getattr(self,'_spanRects',{})
  1048	        hmax = getattr(self,'_hmax',None)
  1049	        longTable = self._longTableOptimize
  1050	        if spanRects and (longTable and hmax==self._hmax_spanRects or not longTable):
  1051	            return
  1052	        colpositions = self._colpositions
  1053	        rowpositions = self._rowpositions
  1054	        vBlocks = {}
  1055	        hBlocks = {}
  1056	        rlim = len(rowpositions)-1
  1057	        for (coord, value) in self._spanRanges.items():
  1058	            if value is None:
  1059	                spanRects[coord] = None
  1060	            else:
  1061	                try:
  1062	                    col0, row0, col1, row1 = value
  1063	                    if row1>=rlim: continue

  1065	                    if col1-col0>0:
  1066	                        for _ in range(col0+1,col1+1):
  1067	                            vBlocks.setdefault(colpositions[_],[]).append((rowpositions[row1+1],rowpositions[row0]))
  1068	                    if row1-row0>0:
  1069	                        for _ in range(row0+1,row1+1):
  1070	                            hBlocks.setdefault(rowpositions[_],[]).append((colpositions[col0],colpositions[col1+1]))
  1071	                    x = colpositions[col0]
  1072	                    y = rowpositions[row1+1]
  1073	                    width = colpositions[col1+1] - x
  1074	                    height = rowpositions[row0] - y
  1075	                    spanRects[coord] = (x, y, width, height)
  1076	                except:
  1077	                    annotateException('\nspanning problem in %s' % (self.identity(),))
  1078
  1079	        for _ in hBlocks, vBlocks:
  1080	            for value in _.values():
  1081	                value.sort()
  1082	        self._spanRects = spanRects
  1083	        self._vBlocks = vBlocks
  1084	        self._hBlocks = hBlocks
  1085	        self._hmax_spanRects = hmax
