  1140	                   maxPasses = 10,
  1141	                   **buildKwds
  1142	                   ):
  1143	        """Makes multiple passes until all indexing flowables
  1144	        are happy.
  1145
  1146	        Returns number of passes"""
  1147	        self._indexingFlowables = []
  1148	        #scan the story and keep a copy
  1149	        for thing in story:
  1150	            if thing.isIndexing():
  1151	                self._indexingFlowables.append(thing)
  1152
  1153	        #better fix for filename is a 'file' problem
  1154	        self._doSave = 0
  1155	        passes = 0
  1156	        mbe = []
  1157	        self._multiBuildEdits = mbe.append
  1158	        while 1:
  1159	            passes += 1
  1160	            if self._onProgress:
  1161	                self._onProgress('PASS', passes)
  1162	            if verbose: sys.stdout.write('building pass '+str(passes) + '...')
  1163
  1164	            for fl in self._indexingFlowables:
  1165	                fl.beforeBuild()
  1166
  1167	            # work with a copy of the story, since it is consumed
  1168	            tempStory = story[:]
  1169	            self.build(tempStory, **buildKwds)
  1170	            #self.notify('debug',None)
  1171
  1172	            for fl in self._indexingFlowables:
  1173	                fl.afterBuild()
  1174
  1175	            happy = self._allSatisfied()
  1176
  1177	            if happy:
  1178	                self._doSave = 0
  1179	                self.canv.save()
  1180	                break
  1181	            if passes > maxPasses:
  1182	                raise IndexError("Index entries not resolved after %d passes" % maxPasses)
  1183
  1184	            #work through any edits
  1185	            while mbe:
  1186	                e = mbe.pop(0)
  1187	                e[0](*e[1:])
  1188
  1189	        del self._multiBuildEdits
  1190	        if verbose: print('saved')
  1191	        return passes

  1237	    def docExec(self,stmt,lifetime):
  1238	        stmt=stmt.strip()
  1239	        NS=self._nameSpace
  1240	        K0=list(NS.keys())
  1241	        try:
  1242	            if lifetime not in self._allowedLifetimes:

  1256	            for v in LT.values():
  1257	                if var in v:
  1258	                    v.remove(var)

  1319	def progressCB(typ, value):
  1320	    """Example prototype for progress monitoring.
  1321
  1322	    This aims to provide info about what is going on
  1323	    during a big job.  It should enable, for example, a reasonably
  1324	    smooth progress bar to be drawn.  We design the argument
  1325	    signature to be predictable and conducive to programming in
  1326	    other (type safe) languages.  If set, this will be called
  1327	    repeatedly with pairs of values.  The first is a string
  1328	    indicating the type of call; the second is a numeric value.
  1329
  1330	    typ 'STARTING', value = 0
  1331	    typ 'SIZE_EST', value = numeric estimate of job size
  1332	    typ 'PASS', value = number of this rendering pass
  1333	    typ 'PROGRESS', value = number between 0 and SIZE_EST
  1334	    typ 'PAGE', value = page number of page
  1335	    type 'FINISHED', value = 0
  1336
  1337	    The sequence is
  1338	        STARTING - always called once
  1339	        SIZE_EST - always called once
  1340	        PROGRESS - called often
  1341	        PAGE - called often when page is emitted
  1342	        FINISHED - called when really, really finished
  1343
  1344	    some juggling is needed to accurately estimate numbers of
  1345	    pages in pageDrawing mode.
  1346
  1347	    NOTE: the SIZE_EST is a guess.  It is possible that the
  1348	    PROGRESS value may slightly exceed it, or may even step
  1349	    back a little on rare occasions.  The only way to be
  1350	    really accurate would be to do two passes, and I don't
  1351	    want to take that performance hit.
  1352	    """
  1353	    print('PROGRESS MONITOR:  %-10s   %d' % (typ, value))
