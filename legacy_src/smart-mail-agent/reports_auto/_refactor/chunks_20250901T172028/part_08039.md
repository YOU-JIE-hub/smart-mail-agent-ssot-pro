l_type,
   216	        context=context,
   217	        chk=chk,
   218	        self_type=self_type,
   219	        module_symbol_table=module_symbol_table,
   220	        no_deferral=no_deferral,
   221	        is_self=is_self,
   222	        rvalue=rvalue,
   223	        suppress_errors=suppress_errors,
   224	    )
   225	    result = _analyze_member_access(name, typ, mx, override_info)
   226	    possible_literal = get_proper_type(result)
   227	    if (
   228	        in_literal_context
   229	        and isinstance(possible_literal, Instance)
   230	        and possible_literal.last_known_value is not None
   231	    ):
   232	        return possible_literal.last_known_value
   233	    else:
   234	        return result

   237	def _analyze_member_access(
   238	    name: str, typ: Type, mx: MemberContext, override_info: TypeInfo | None = None
   239	) -> Type:
   240	    typ = get_proper_type(typ)
   241	    if isinstance(typ, Instance):
   242	        return analyze_instance_member_access(name, typ, mx, override_info)
   243	    elif isinstance(typ, AnyType):
   244	        # The base object has dynamic type.
   245	        return AnyType(TypeOfAny.from_another_any, source_any=typ)
   246	    elif isinstance(typ, UnionType):
   247	        return analyze_union_member_access(name, typ, mx)
   248	    elif isinstance(typ, FunctionLike) and typ.is_type_obj():
   249	        return analyze_type_callable_member_access(name, typ, mx)
   250	    elif isinstance(typ, TypeType):
   251	        return analyze_type_type_member_access(name, typ, mx, override_info)
   252	    elif isinstance(typ, TupleType):
   253	        # Actually look up from the fallback instance type.
   254	        return _analyze_member_access(name, tuple_fallback(typ), mx, override_info)
   255	    elif isinstance(typ, (LiteralType, FunctionLike)):
   256	        # Actually look up from the fallback instance type.
   257	        return _analyze_member_access(name, typ.fallback, mx, override_info)
   258	    elif isinstance(typ, TypedDictType):
   259	        return analyze_typeddict_access(name, typ, mx, override_info)
   260	    elif isinstance(typ, NoneType):
   261	        return analyze_none_member_access(name, typ, mx)
   262	    elif isinstance(typ, TypeVarLikeType):
   263	        if isinstance(typ, TypeVarType) and typ.values:
   264	            return _analyze_member_access(
   265	                name, make_simplified_union(typ.values), mx, override_info
   266	            )
   267	        return _analyze_member_access(name, typ.upper_bound, mx, override_info)
   268	    elif isinstance(typ, DeletedType):
   269	        if not mx.suppress_errors:
   270	            mx.msg.deleted_as_rvalue(typ, mx.context)
   271	        return AnyType(TypeOfAny.from_error)
   272	    return report_missing_attribute(mx.original_type, typ, name, mx)

   275	def may_be_awaitable_attribute(
   276	    name: str, typ: Type, mx: MemberContext, override_info: TypeInfo | None = None
   277	) -> bool:
   278	    """Check if the given type has the attribute when awaited."""
   279	    if mx.chk.checking_missing_await:
   280	        # Avoid infinite recursion.
   281	        return False
   282	    with mx.chk.checking_await_set(), mx.msg.filter_errors() as local_errors:
   283	        aw_type = mx.chk.get_precise_awaitable_type(typ, local_errors)
   284	        if aw_type is None:
   285	            return False
   286	        _ = _analyze_member_access(
   287	            name, aw_type, mx.copy_modified(self_type=aw_type), override_info
   288	        )
   289	        return not local_errors.has_new_errors()

   292	def report_missing_attribute(
   293	    original_type: Type,
   294	    typ: Type,
   295	    name: str,
   296	    mx: MemberContext,
   297	    override_info: TypeInfo | None = None,
   298	) -> Type:
   299	    if mx.suppress_errors:
   300	        return AnyType(TypeOfAny.from_error)
   301	    error_code = mx.msg.has_no_attr(original_type, typ, name, mx.context, mx.module_symbol_table)
   302	    if not mx.msg.prefer_simple_messages():
   303	        if may_be_awaitable_attribute(name, typ, mx, override_info):
   304	            mx.msg.possible_missing_await(mx.context, error_code)
   305	    return AnyType(TypeOfAny.from_error)

   308	# The several functions that follow implement analyze_member_access for various
   309	# types and aren't documented individually.

   311
   312	def analyze_instance_member_access(
   313	    name: str, typ: Instance, mx: MemberContext, override_info: TypeInfo | None
   314	) -> Type:
   315	    info = typ.type
   316	    if override_info:
   317	        info = override_info
   318
   319	    method = info.get_method(name)
   320
   321	    if name == "__init__" and not mx.is_super and not info.is_final:
   322	        if not method or not method.is_final:
   323	            # Accessing __init__ in statically typed code would compromise
   324	            # type safety unless used via super() or the method/class is final.
   325	            mx.fail(message_registry.CANNOT_ACCESS_INIT)
   326	            return AnyType(TypeOfAny.from_error)
   327
   328	    # The base object has an instance type.
   329
   330	    if (
   331	        state.find_occurrences
   332	        and info.name == state.find_occurrences[0]
   333	        and name == state.find_occurrences[1]
   334	        and not mx.suppress_errors
   335	    ):
   336	        mx.msg.note("Occurrence of '{}.{}'".format(*state.find_occurrences), mx.context)
   337
   338	    # Look up the member. First look up the method dictionary.
   339	    if method and not isinstance(method, Decorator):
   340	        if mx.is_super and not mx.suppress_errors:
   341	            validate_super_call(method, mx)
   342
   343	        if method.is_property:
   344	            assert isinstance(method, OverloadedFuncDef)
   345	            getter = method.items[0]
   346	            assert isinstance(getter, Decorator)
   347	            if mx.is_lvalue and getter.var.is_settable_property:
   348	                mx.chk.warn_deprecated(method.setter, mx.context)
   349	            return analyze_var(name, getter.var, typ, mx)
   350
   351	        if mx.is_lvalue and not mx.suppress_errors:
   352	            mx.msg.cant_assign_to_method(mx.context)
   353	        if not isinstance(method, OverloadedFuncDef):
   354	            signature = function_type(method, mx.named_type("builtins.function"))
   355	        else:
   356	            if method.type is None:
   357	                # Overloads may be not ready if they are decorated. Handle this in same
   358	                # manner as we would handle a regular decorated function: defer if possible.
   359	                if not mx.no_deferral and method.items:
   360	                    mx.not_ready_callback(method.name, mx.context)
   361	                return AnyType(TypeOfAny.special_form)
   362	            assert isinstance(method.type, Overloaded)
   363	            signature = method.type
   364	        if not mx.preserve_type_var_ids:
   365	            signature = freshen_all_functions_type_vars(signature)
   366	        if not method.is_static:
   367	            if isinstance(method, (FuncDef, OverloadedFuncDef)) and method.is_trivial_self:
   368	                signature = bind_self_fast(signature, mx.self_type)
   369	            else:
   370	                signature = check_self_arg(
   371	                    signature, mx.self_type, method.is_class, mx.context, name, mx.msg
   372	                )
   373	                signature = bind_self(signature, mx.self_type, is_classmethod=method.is_class)
   374	        # TODO: should we skip these steps for static methods as well?
   375	        # Since generic static methods should not be allowed.
   376	        typ = map_instance_to_supertype(typ, method.info)
   377	        member_type = expand_type_by_instance(signature, typ)
   378	        freeze_all_type_vars(member_type)
   379	        return member_type
   380	    else:
   381	        # Not a method.
   382	        return analyze_member_var_access(name, typ, info, mx)
   383

   385	def validate_super_call(node: FuncBase, mx: MemberContext) -> None:
   386	    unsafe_super = False
   387	    if isinstance(node, FuncDef) and node.is_trivial_body:
   388	        unsafe_super = True
   389	    elif isinstance(node, OverloadedFuncDef):
   390	        if node.impl:
   391	            impl = node.impl if isinstance(node.impl, FuncDef) else node.impl.func
   392	            unsafe_super = impl.is_trivial_body
   393	        elif not node.is_property and node.items:
   394	            assert isinstance(node.items[0], Decorator)
   395	            unsafe_super = node.items[0].func.is_trivial_body
   396	    if unsafe_super:
   397	        mx.msg.unsafe_super(node.name, node.info.name, mx.context)

   400	def analyze_type_callable_member_access(name: str, typ: FunctionLike, mx: MemberContext) -> Type:
   401	    # Class attribute.
   402	    # TODO super?
   403	    ret_type = typ.items[0].ret_type
   404	    assert isinstance(ret_type, ProperType)
   405	    if isinstance(ret_type, TupleType):
   406	        ret_type = tuple_fallback(ret_type)
   407	    if isinstance(ret_type, TypedDictType):
   408	        ret_type = ret_type.fallback
   409	    if isinstance(ret_type, Instance):
   410	        if not mx.is_operator:
   411	            # When Python sees an operator (eg `3 == 4`), it automatically translates that
   412	            # into something like `int.__eq__(3, 4)` instead of `(3).__eq__(4)` as an
   413	            # optimization.
   414	            #
   415	            # While it normally it doesn't matter which of the two versions are used, it
   416	            # does cause inconsistencies when working with classes. For example, translating
   417	            # `int == int` to `int.__eq__(int)` would not work since `int.__eq__` is meant to
   418	            # compare two int _instances_. What we really want is `type(int).__eq__`, which
   419	            # is meant to compare two types or classes.
   420	            #
   421	            # This check makes sure that when we encounter an operator, we skip looking up
   422	            # the corresponding method in the current instance to avoid this edge case.
   423	            # See https://github.com/python/mypy/pull/1787 for more info.
   424	            # TODO: do not rely on same type variables being present in all constructor overloads.
   425	            result = analyze_class_attribute_access(
   426	                ret_type, name, mx, original_vars=typ.items[0].variables, mcs_fallback=typ.fallback
   427	            )
   428	            if result:
   429	                return result
   430	        # Look up from the 'type' type.
   431	        return _analyze_member_access(name, typ.fallback, mx)
   432	    else:
   433	        assert False, f"Unexpected type {ret_type!r}"

   436	def analyze_type_type_member_access(
   437	    name: str, typ: TypeType, mx: MemberContext, override_info: TypeInfo | None
   438	) -> Type:
   439	    # Similar to analyze_type_callable_attribute_access.
   440	    item = None
   441	    fallback = mx.named_type("builtins.type")
   442	    if isinstance(typ.item, Instance):
   443	        item = typ.item
   444	    elif isinstance(typ.item, AnyType):
   445	        with mx.msg.filter_errors():
   446	            return _analyze_member_access(name, fallback, mx, override_info)
   447	    elif isinstance(typ.item, TypeVarType):
   448	        upper_bound = get_proper_type(typ.item.upper_bound)
   449	        if isinstance(upper_bound, Instance):
   450	            item = upper_bound
   451	        elif isinstance(upper_bound, UnionType):
   452	            return _analyze_member_access(
   453	                name,
   454	                TypeType.make_normalized(upper_bound, line=typ.line, column=typ.column),
   455	                mx,
   456	                override_info,
   457	            )
   458	        elif isinstance(upper_bound, TupleType):
   459	            item = tuple_fallback(upper_bound)
   460	        elif isinstance(upper_bound, AnyType):
   461	            with mx.msg.filter_errors():
   462	                return _analyze_member_access(name, fallback, mx, override_info)
   463	    elif isinstance(typ.item, TupleType):
   464	        item = tuple_fallback(typ.item)
   465	    elif isinstance(typ.item, FunctionLike) and typ.item.is_type_obj():
   466	        item = typ.item.fallback
   467	    elif isinstance(typ.item, TypeType):
   468	        # Access member on metaclass object via Type[Type[C]]
   469	        if isinstance(typ.item.item, Instance):
   470	            item = typ.item.item.type.metaclass_type
   471	    ignore_messages = False
   472
   473	    if item is not None:
   474	        fallback = item.type.metaclass_type or fallback
   475
   476	    if item and not mx.is_operator:
   477	        # See comment above for why operators are skipped
   478	        result = analyze_class_attribute_access(
   479	            item, name, mx, mcs_fallback=fallback, override_info=override_info
   480	        )
   481	        if result:
   482	            if not (isinstance(get_proper_type(result), AnyType) and item.type.fallback_to_any):
   483	                return result
   484	            else:
   485	                # We don't want errors on metaclass lookup for classes with Any fallback
   486	                ignore_messages = True
   487
   488	    with mx.msg.filter_errors(filter_errors=ignore_messages):
   489	        return _analyze_member_access(name, fallback, mx, override_info)
   490

   492	def analyze_union_member_access(name: str, typ: UnionType, mx: MemberContext) -> Type:
   493	    with mx.msg.disable_type_names():
   494	        results = []
   495	        for subtype in typ.relevant_items():
   496	            # Self types should be bound to every individual item of a union.
   497	            item_mx = mx.copy_modified(self_type=subtype)
   498	            results.append(_analyze_member_access(name, subtype, item_mx))
   499	    return make_simplified_union(results)

   516	def analyze_member_var_access(
   517	    name: str, itype: Instance, info: TypeInfo, mx: MemberContext
   518	) -> Type:
   519	    """Analyse attribute access that does not target a method.
   520
   521	    This is logically part of analyze_member_access and the arguments are similar.
   522
   523	    original_type is the type of E in the expression E.var
   524	    """
   525	    # It was not a method. Try looking up a variable.
   526	    node = info.get(name)
   527	    v = node.node if node else None
   528
   529	    mx.chk.warn_deprecated(v, mx.context)
   530
   531	    vv = v
   532	    is_trivial_self = False
   533	    if isinstance(vv, Decorator):
   534	        # The associated Var node of a decorator contains the type.
   535	        v = vv.var
   536	        is_trivial_self = vv.func.is_trivial_self and not vv.decorators
   537	        if mx.is_super and not mx.suppress_errors:
   538	            validate_super_call(vv.func, mx)
   539	    if isinstance(v, FuncDef):
   540	        assert False, "Did not expect a function"
   541	    if isinstance(v, MypyFile):
   542	        mx.chk.module_refs.add(v.fullname)
   543
   544	    if isinstance(vv, (TypeInfo, TypeAlias, MypyFile, TypeVarLikeExpr)):
   545	        # If the associated variable is a TypeInfo synthesize a Var node for
   546	        # the purposes of type checking.  This enables us to type check things
   547	        # like accessing class attributes on an inner class. Similar we allow
   548	        # using qualified type aliases in runtime context. For example:
   549	        #     class C:
   550	        #         A = List[int]
   551	        #     x = C.A() <- this is OK
   552	        typ = mx.chk.expr_checker.analyze_static_reference(vv, mx.context, mx.is_lvalue)
   553	        v = Var(name, type=typ)
   554	        v.info = info
   555
   556	    if isinstance(v, Var):
   557	        implicit = info[name].implicit
   558
   559	        # An assignment to final attribute is always an error,
   560	        # independently of types.
   561	        if mx.is_lvalue and not mx.chk.get_final_context():
   562	            check_final_member(name, info, mx.msg, mx.context)
   563
   564	        return analyze_var(name, v, itype, mx, implicit=implicit, is_trivial_self=is_trivial_self)
   565	    elif (
   566	        not v
   567	        and name not in ["__getattr__", "__setattr__", "__getattribute__"]
   568	        and not mx.is_operator
   569	        and mx.module_symbol_table is None
   570	    ):
   571	        # Above we skip ModuleType.__getattr__ etc. if we have a
   572	        # module symbol table, since the symbol table allows precise
   573	        # checking.
   574	        if not mx.is_lvalue:
   575	            for method_name in ("__getattribute__", "__getattr__"):
   576	                method = info.get_method(method_name)
   577
   578	                # __getattribute__ is defined on builtins.object and returns Any, so without
   579	                # the guard this search will always find object.__getattribute__ and conclude
   580	                # that the attribute exists
   581	                if method and method.info.fullname != "builtins.object":
   582	                    bound_method = analyze_decorator_or_funcbase_access(
   583	                        defn=method, itype=itype, name=method_name, mx=mx
   584	                    )
   585	                    typ = map_instance_to_supertype(itype, method.info)
   586	                    getattr_type = get_proper_type(expand_type_by_instance(bound_method, typ))
   587	                    if isinstance(getattr_type, CallableType):
   588	                        result = getattr_type.ret_type
   589	                    else:
   590	                        result = getattr_type
   591
   592	                    # Call the attribute hook before returning.
   593	                    fullname = f"{method.info.fullname}.{name}"
   594	                    hook = mx.chk.plugin.get_attribute_hook(fullname)
   595	                    if hook:
   596	                        result = hook(
   597	                            AttributeContext(
   598	                                get_proper_type(mx.original_type),
   599	                                result,
   600	                                mx.is_lvalue,
   601	                                mx.context,
   602	                                mx.chk,
   603	                            )
   604	                        )
   605	                    return result
   606	        else:
   607	            setattr_meth = info.get_method("__setattr__")
   608	            if setattr_meth and setattr_meth.info.fullname != "builtins.object":
   609	                bound_type = analyze_decorator_or_funcbase_access(
   610	                    defn=setattr_meth,
   611	                    itype=itype,
   612	                    name="__setattr__",
   613	                    mx=mx.copy_modified(is_lvalue=False),
   614	                )
   615	                typ = map_instance_to_supertype(itype, setattr_meth.info)
   616	                setattr_type = get_proper_type(expand_type_by_instance(bound_type, typ))
   617	                if isinstance(setattr_type, CallableType) and len(setattr_type.arg_types) > 0:
   618	                    return setattr_type.arg_types[-1]
   619
   620	    if itype.type.fallback_to_any:
   621	        return AnyType(TypeOfAny.special_form)
   622
   623	    # Could not find the member.
   624	    if itype.extra_attrs and name in itype.extra_attrs.attrs:
   625
