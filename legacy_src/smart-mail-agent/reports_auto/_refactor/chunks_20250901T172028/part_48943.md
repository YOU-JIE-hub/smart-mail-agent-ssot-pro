   248	def _findOverlapRun(B,i,wrap):
   249	    '''find overlap run containing B[i]'''
   250	    n = len(B)
   251	    R = [i]
   252	    while 1:
   253	        i = R[-1]
   254	        j = (i+1)%n
   255	        if j in R or not boundsOverlap(B[i],B[j]): break
   256	        R.append(j)
   257	    while 1:
   258	        i = R[0]
   259	        j = (i-1)%n
   260	        if j in R or not boundsOverlap(B[i],B[j]): break
   261	        R.insert(0,j)
   262	    return R
