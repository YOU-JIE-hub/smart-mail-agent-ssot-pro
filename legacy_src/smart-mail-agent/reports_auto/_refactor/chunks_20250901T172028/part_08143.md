  6926	    def defer(self, debug_context: Context | None = None, force_progress: bool = False) -> None:
  6927	        """Defer current analysis target to be analyzed again.
  6928
  6929	        This must be called if something in the current target is
  6930	        incomplete or has a placeholder node. However, this must *not*
  6931	        be called during the final analysis iteration! Instead, an error
  6932	        should be generated. Often 'process_placeholder' is a good
  6933	        way to either defer or generate an error.
  6934
  6935	        NOTE: Some methods, such as 'anal_type', 'mark_incomplete' and

  6937	              They are usually preferable to a direct defer() call.

  6939	        assert not self.final_iteration, "Must not defer during final iteration"
  6940	        if force_progress:
  6941	            # Usually, we report progress if we have replaced a placeholder node
  6942	            # with an actual valid node. However, sometimes we need to update an
  6943	            # existing node *in-place*. For example, this is used by type aliases
  6944	            # in context of forward references and/or recursive aliases, and in
  6945	            # similar situations (recursive named tuples etc).
  6946	            self.progress = True
  6947	        self.deferred = True
  6948	        # Store debug info for this deferral.
  6949	        line = (
  6950	            debug_context.line if debug_context else self.statement.line if self.statement else -1
  6951	        )

  7045	        """Enter a function, generator or comprehension scope."""

  7092	    def current_symbol_table(
  7093	        self, escape_comprehensions: bool = False, type_param: bool = False
  7094	    ) -> SymbolTable:
  7095	        if type_param and self.scope_stack[-1] == SCOPE_ANNOTATION:
  7096	            n = self.locals[-1]
  7097	            assert n is not None
  7098	            return n
  7099	        elif self.is_func_scope():
  7100	            if self.scope_stack[-1] == SCOPE_ANNOTATION:
  7101	                n = self.locals[-2]

  7105	            if escape_comprehensions:
  7106	                assert len(self.locals) == len(self.scope_stack)
  7107	                # Retrieve the symbol table from the enclosing non-comprehension scope.
  7108	                for i, scope_type in enumerate(reversed(self.scope_stack)):
  7109	                    if scope_type != SCOPE_COMPREHENSION:
  7110	                        if i == len(self.locals) - 1:  # The last iteration.
  7111	                            # The caller of the comprehension is in the global space.
  7112	                            names = self.globals
  7113	                        else:
  7114	                            names_candidate = self.locals[-1 - i]
  7115	                            assert (
  7116	                                names_candidate is not None
  7117	                            ), "Escaping comprehension from invalid scope"
  7118	                            names = names_candidate
  7119	                        break
  7120	                else:
  7121	                    assert False, "Should have at least one non-comprehension scope"
  7122	            else:
  7123	                names = n
  7124	            assert names is not None
  7125	        elif self.type is not None:
  7126	            names = self.type.names
  7127	        else:
  7128	            names = self.globals
