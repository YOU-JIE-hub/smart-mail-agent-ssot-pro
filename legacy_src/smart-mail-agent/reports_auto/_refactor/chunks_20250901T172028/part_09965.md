   599	        swig_cmd.extend(self.swig_opts)
   600	        if self.swig_cpp:

   603	        # Do not override commandline arguments

   605	            for o in extension.swig_opts:
   606	                swig_cmd.append(o)

   608	        for source in swig_sources:
   609	            target = swig_targets[source]

   611	            self.spawn(swig_cmd + ["-o", target, source])
