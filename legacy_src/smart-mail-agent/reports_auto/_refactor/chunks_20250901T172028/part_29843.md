 is the union of the true versions of its components
   768	        new_items = [true_only(item) for item in t.items]
   769	        can_be_true_items = [item for item in new_items if item.can_be_true]
   770	        return make_simplified_union(can_be_true_items, line=t.line, column=t.column)
   771	    else:
   772	        ret_type = _get_type_method_ret_type(t, name="__bool__") or _get_type_method_ret_type(
   773	            t, name="__len__"
   774	        )
   775
   776	        if ret_type and not ret_type.can_be_true:
   777	            return UninhabitedType(line=t.line, column=t.column)
   778
   779	        new_t = copy_type(t)
   780	        new_t.can_be_false = False
   781	        return new_t

   784	def false_only(t: Type) -> ProperType:
   785	    """
   786	    Restricted version of t with only False-ish values
   787	    """
   788	    t = get_proper_type(t)
   789
   790	    if not t.can_be_false:
   791	        if state.strict_optional:
   792	            # All values of t are True-ish, so there are no false values in it
   793	            return UninhabitedType(line=t.line)
   794	        else:
   795	            # When strict optional checking is disabled, everything can be
   796	            # False-ish since anything can be None
   797	            return NoneType(line=t.line)
   798	    elif not t.can_be_true:
   799	        # All values of t are already False-ish, so false_only is idempotent in this case
   800	        return t
   801	    elif isinstance(t, UnionType):
   802	        # The false version of a union type is the union of the false versions of its components
   803	        new_items = [false_only(item) for item in t.items]
   804	        can_be_false_items = [item for item in new_items if item.can_be_false]
   805	        return make_simplified_union(can_be_false_items, line=t.line, column=t.column)
   806	    elif isinstance(t, Instance) and t.type.fullname in ("builtins.str", "builtins.bytes"):
   807	        return LiteralType("", fallback=t)
   808	    elif isinstance(t, Instance) and t.type.fullname == "builtins.int":
   809	        return LiteralType(0, fallback=t)
   810	    else:
   811	        ret_type = _get_type_method_ret_type(t, name="__bool__") or _get_type_method_ret_type(
   812	            t, name="__len__"
   813	        )
   814
   815	        if ret_type:
   816	            if not ret_type.can_be_false:
   817	                return UninhabitedType(line=t.line)
   818	        elif isinstance(t, Instance):
   819	            if t.type.is_final or t.type.is_enum:
   820	                return UninhabitedType(line=t.line)
   821	        elif isinstance(t, LiteralType) and t.is_enum_literal():
   822	            return UninhabitedType(line=t.line)
   823
   824	        new_t = copy_type(t)
   825	        new_t.can_be_true = False
   826	        return new_t

   961	    typ: Type, target_literal_type: type[T], target_fullname: str
   962	) -> list[T] | None:
   963	    """If the given expression or type corresponds to a Literal or

  1029	def try_expanding_sum_type_to_union(typ: Type, target_fullname: str) -> ProperType:
  1030	    """Attempts to recursively expand any enum Instances with the given target_fullname
  1031	    into a Union of all of its component LiteralTypes.
  1032
  1033	    For example, if we have:
  1034
  1035	        class Color(Enum):
  1036	            RED = 1
  1037	            BLUE = 2
  1038	            YELLOW = 3
  1039
  1040	        class Status(Enum):
  1041	            SUCCESS = 1
  1042	            FAILURE = 2
  1043	            UNKNOWN = 3
  1044
  1045	    ...and if we call `try_expanding_sum_type_to_union(Union[Color, Status], 'module.Color')`,
  1046	    this function will return Literal[Color.RED, Color.BLUE, Color.YELLOW, Status].
  1047	    """
  1048	    typ = get_proper_type(typ)
  1049
  1050	    if isinstance(typ, UnionType):
  1051	        items = [
  1052	            try_expanding_sum_type_to_union(item, target_fullname) for item in typ.relevant_items()
  1053	        ]
  1054	        return make_simplified_union(items, contract_literals=False)
  1055
  1056	    if isinstance(typ, Instance) and typ.type.fullname == target_fullname:
  1057	        if typ.type.fullname == "builtins.bool":
  1058	            items = [LiteralType(True, typ), LiteralType(False, typ)]
  1059	            return make_simplified_union(items, contract_literals=False)
  1060
  1061	        if typ.type.is_enum:
  1062	            items = [LiteralType(name, typ) for name in typ.type.enum_members]
  1063	            if not items:
  1064	                return typ
  1065	            return make_simplified_union(items, contract_literals=False)
  1066
  1067	    return typ

  1070	def try_contracting_literals_in_union(types: Sequence[Type]) -> list[ProperType]:
  1071	    """Contracts any literal types back into a sum type if possible.
  1072
  1073	    Requires a flattened union and does not descend into children.
  1074
  1075	    Will replace the first instance of the literal with the sum type and
  1076	    remove all others.
  1077
  1078	    If we call `try_contracting_union(Literal[Color.RED, Color.BLUE, Color.YELLOW])`,
  1079	    this function will return Color.
  1080
  1081	    We also treat `Literal[True, False]` as `bool`.
  1082	    """
  1083	    proper_types = [get_proper_type(typ) for typ in types]
  1084	    sum_types: dict[str, tuple[set[Any], list[int]]] = {}
  1085	    marked_for_deletion = set()
  1086	    for idx, typ in enumerate(proper_types):
  1087	        if isinstance(typ, LiteralType):
  1088	            fullname = typ.fallback.type.fullname
  1089	            if typ.fallback.type.is_enum or isinstance(typ.value, bool):
  1090	                if fullname not in sum_types:
  1091	                    sum_types[fullname] = (
  1092	                        (
  1093	                            set(typ.fallback.type.enum_members)
  1094	                            if typ.fallback.type.is_enum
  1095	                            else {True, False}
  1096	                        ),
  1097	                        [],
  1098	                    )
  1099	                literals, indexes = sum_types[fullname]
  1100	                literals.discard(typ.value)
  1101	                indexes.append(idx)
  1102	                if not literals:
  1103	                    first, *rest = indexes
  1104	                    proper_types[first] = typ.fallback
  1105	                    marked_for_deletion |= set(rest)
  1106	    return list(
  1107	        itertools.compress(
  1108	            proper_types, [(i not in marked_for_deletion) for i in range(len(proper_types))]
  1109	        )
  1110	    )
