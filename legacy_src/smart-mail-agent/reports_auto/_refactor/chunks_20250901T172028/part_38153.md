   127	def singledispatch_register_callback(ctx: MethodContext) -> Type:
   128	    """Called for functools._SingleDispatchCallable.register"""
   129	    assert isinstance(ctx.type, Instance)
   130	    # TODO: check that there's only one argument
   131	    first_arg_type = get_proper_type(get_first_arg(ctx.arg_types))
   132	    if isinstance(first_arg_type, (CallableType, Overloaded)) and first_arg_type.is_type_obj():
   133	        # HACK: We received a class as an argument to register. We need to be able
   134	        # to access the function that register is being applied to, and the typeshed definition
   135	        # of register has it return a generic Callable, so we create a new
   136	        # SingleDispatchRegisterCallable class, define a __call__ method, and then add a
   137	        # plugin hook for that.

   139	        # is_subtype doesn't work when the right type is Overloaded, so we need the
   140	        # actual type
   141	        register_type = first_arg_type.items[0].ret_type
   142	        type_args = RegisterCallableInfo(register_type, ctx.type)
   143	        register_callable = make_fake_register_class_instance(ctx.api, type_args)
   144	        return register_callable
   145	    elif isinstance(first_arg_type, CallableType):
   146	        # TODO: do more checking for registered functions
   147	        register_function(ctx, ctx.type, first_arg_type, ctx.api.options)
   148	        # The typeshed stubs for register say that the function returned is Callable[..., T], even
   149	        # though the function returned is the same as the one passed in. We return the type of the
   150	        # function so that mypy can properly type check cases where the registered function is used
   151	        # directly (instead of through singledispatch)
   152	        return first_arg_type

   158	def register_function(
   159	    ctx: PluginContext,
   160	    singledispatch_obj: Instance,
   161	    func: Type,
   162	    options: Options,
   163	    register_arg: Type | None = None,
   164	) -> None:
   165	    """Register a function"""
   166
   167	    func = get_proper_type(func)
   168	    if not isinstance(func, CallableType):
   169	        return
   170	    metadata = get_singledispatch_info(singledispatch_obj)
   171	    if metadata is None:
   172	        # if we never added the fallback to the type variables, we already reported an error, so
   173	        # just don't do anything here
   174	        return
   175	    dispatch_type = get_dispatch_type(func, register_arg)
   176	    if dispatch_type is None:
   177	        # TODO: report an error here that singledispatch requires at least one argument
   178	        # (might want to do the error reporting in get_dispatch_type)
   179	        return
   180	    fallback = metadata.fallback
   181
   182	    fallback_dispatch_type = fallback.arg_types[0]
   183	    if not is_subtype(dispatch_type, fallback_dispatch_type):
   184	        fail(
   185	            ctx,
   186	            "Dispatch type {} must be subtype of fallback function first argument {}".format(
   187	                format_type(dispatch_type, options), format_type(fallback_dispatch_type, options)
   188	            ),
   189	            func.definition,
   190	        )
   191	        return
   192	    return

    54	    for i in range(len(s.expr)):
    55	        result = infer_condition_value(s.expr[i], options)
