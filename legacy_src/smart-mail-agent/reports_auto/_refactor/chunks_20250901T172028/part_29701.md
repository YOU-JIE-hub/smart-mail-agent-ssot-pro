  2117	        parent_error: ErrorInfo,

  2126	        # note: 'Cls' is missing following 'Proto' members:
  2127	        # note:     method, attr
  2128	        MAX_ITEMS = 2  # Maximum number of conflicts, missing members, and overloads shown

  2130	        exclusions: dict[type, list[str]] = {
  2131	            TypedDictType: ["typing.Mapping"],
  2132	            TupleType: ["typing.Iterable", "typing.Sequence"],
  2133	        }
  2134	        if supertype.type.fullname in exclusions.get(type(subtype), []):
  2135	            return
  2136	        if any(isinstance(tp, UninhabitedType) for tp in get_proper_types(supertype.args)):
  2137	            # We don't want to add notes for failed inference (e.g. Iterable[Never]).
  2138	            # This will be only confusing a user even more.
  2139	            return

  2142	        is_module = False
  2143	        skip = []
  2144	        if isinstance(subtype, TupleType):
  2145	            subtype = subtype.partial_fallback
  2146	        elif isinstance(subtype, TypedDictType):
  2147	            subtype = subtype.fallback
  2148	        elif isinstance(subtype, TypeType):
  2149	            if not isinstance(subtype.item, Instance):
  2150	                return
  2151	            class_obj = True
  2152	            subtype = subtype.item
  2153	        elif isinstance(subtype, CallableType):
  2154	            if subtype.is_type_obj():
  2155	                ret_type = get_proper_type(subtype.ret_type)
  2156	                if isinstance(ret_type, TupleType):
  2157	                    ret_type = ret_type.partial_fallback
  2158	                if not isinstance(ret_type, Instance):
  2159	                    return
  2160	                class_obj = True
  2161	                subtype = ret_type
  2162	            else:
  2163	                subtype = subtype.fallback
  2164	                skip = ["__call__"]
  2165	        if subtype.extra_attrs and subtype.extra_attrs.mod_name:
  2166	            is_module = True

  2169	        missing = get_missing_protocol_members(subtype, supertype, skip=skip)
  2170	        if (
  2171	            missing
  2172	            and (len(missing) < len(supertype.type.protocol_members) or missing == ["__call__"])
  2173	            and len(missing) <= MAX_ITEMS
  2174	        ):

  2176	                self.note(

  2181	                    parent_error=parent_error,
