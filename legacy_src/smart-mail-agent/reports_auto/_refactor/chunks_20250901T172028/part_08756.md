   710	    def drawNode(self, node):
   711	        """This is the recursive method called for each node in the tree.
   712	        """
   713
   714	        if self.verbose: print("### begin _SVGRenderer.drawNode(%r)" % node)
   715
   716	        self._canvas.comment('begin node %r'%node)
   717	        style = self._canvas.style.copy()
   718	        if not (isinstance(node, Path) and node.isClipPath):
   719	            pass # self._canvas.saveState()
   720
   721	        #apply state changes
   722	        deltas = getStateDelta(node)
   723	        self._tracker.push(deltas)
   724	        self.applyStateChanges(deltas, {})
   725
   726	        #draw the object, or recurse
   727	        self.drawNodeDispatcher(node)
   728
   729	        rDeltas = self._tracker.pop()
   730	        if not (isinstance(node, Path) and node.isClipPath):
   731	            pass #self._canvas.restoreState()
   732	        self._canvas.comment('end node %r'%node)
   733
   734	        #restore things we might have lost (without actually doing anything).
   735	        for k, v in rDeltas.items():
   736	            if k in self._restores:
   737	                setattr(self._canvas,self._restores[k],v)
   738	        self._canvas.style = style
   739
   740	        if self.verbose: print("### end _SVGRenderer.drawNode(%r)" % node)
