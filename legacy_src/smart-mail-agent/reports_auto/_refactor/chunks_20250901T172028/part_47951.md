upper_bound:
   302	                return self.s
   303	            return self.s.copy_modified(upper_bound=join_types(self.s.upper_bound, t.upper_bound))
   304	        else:
   305	            return self.default(self.s)

   308	        if self.s == t:
   309	            return t

   323	    def visit_parameters(self, t: Parameters) -> ProperType:
   324	        if isinstance(self.s, Parameters):
   325	            if not is_similar_params(t, self.s):
   326	                # TODO: it would be prudent to return [*object, **object] instead of Any.
   327	                return self.default(self.s)
   328	            from mypy.meet import meet_types

   330	            return t.copy_modified(
   331	                arg_types=[
   332	                    meet_types(s_a, t_a) for s_a, t_a in zip(self.s.arg_types, t.arg_types)

   339	    def visit_instance(self, t: Instance) -> ProperType:
   340	        if isinstance(self.s, Instance):
   341	            if self.instance_joiner is None:
   342	                self.instance_joiner = InstanceJoiner()
   343	            nominal = self.instance_joiner.join_instances(t, self.s)
   344	            structural: Instance | None = None
   345	            if t.type.is_protocol and is_protocol_implementation(self.s, t):
   346	                structural = t
   347	            elif self.s.type.is_protocol and is_protocol_implementation(t, self.s):
   348	                structural = self.s
   349	            # Structural join is preferred in the case where we have found both
   350	            # structural and nominal and they have same MRO length (see two comments
   351	            # in join_instances_via_supertype). Otherwise, just return the nominal join.
   352	            if not structural or is_better(nominal, structural):
   353	                return nominal
   354	            return structural
   355	        elif isinstance(self.s, FunctionLike):
   356	            if t.type.is_protocol:
   357	                call = unpack_callback_protocol(t)
   358	                if call:
   359	                    return join_types(call, self.s)
   360	            return join_types(t, self.s.fallback)
   361	        elif isinstance(self.s, TypeType):
   362	            return join_types(t, self.s)
   363	        elif isinstance(self.s, TypedDictType):
   364	            return join_types(t, self.s)
   365	        elif isinstance(self.s, TupleType):
   366	            return join_types(t, self.s)
   367	        elif isinstance(self.s, LiteralType):
   368	            return join_types(t, self.s)
   369	        elif isinstance(self.s, TypeVarTupleType) and is_subtype(self.s.upper_bound, t):
   370	            return t
   371	        else:
   372	            return self.default(self.s)

   374	    def visit_callable_type(self, t: CallableType) -> ProperType:
   375	        if isinstance(self.s, CallableType) and is_similar_callables(t, self.s):
   376	            if is_equivalent(t, self.s):
   377	                return combine_similar_callables(t, self.s)
   378	            result = join_similar_callables(t, self.s)
   379	            # We set the from_type_type flag to suppress error when a collection of
   380	            # concrete class objects gets inferred as their common abstract superclass.
   381	            if not (
   382	                (t.is_type_obj() and t.type_object().is_abstract)
   383	                or (self.s.is_type_obj() and self.s.type_object().is_abstract)
   384	            ):
   385	                result.from_type_type = True
   386	            if any(
   387	                isinstance(tp, (NoneType, UninhabitedType))
   388	                for tp in get_proper_types(result.arg_types)
   389	            ):
   390	                # We don't want to return unusable Callable, attempt fallback instead.
   391	                return join_types(t.fallback, self.s)
   392	            return result
   393	        elif isinstance(self.s, Overloaded):
   394	            # Switch the order of arguments to that we'll get to visit_overloaded.
   395	            return join_types(t, self.s)
   396	        elif isinstance(self.s, Instance) and self.s.type.is_protocol:
   397	            call = unpack_callback_protocol(self.s)
   398	            if call:
   399	                return join_types(t, call)
   400	        return join_types(t.fallback, self.s)

   402	    def visit_overloaded(self, t: Overloaded) -> ProperType:
   403	        # This is more complex than most other cases. Here are some
   404	        # examples that illustrate how this works.
   405	        #
   406	        # First let's define a concise notation:
   407	        #  - Cn are callable types (for n in 1, 2, ...)
   408	        #  - Ov(C1, C2, ...) is an overloaded type with items C1, C2, ...
   409	        #  - Callable[[T, ...], S] is written as [T, ...] -> S.
   410	        #
   411	        # We want some basic properties to hold (assume Cn are all
   412	        # unrelated via Any-similarity):
   413	        #
   414	        #   join(Ov(C1, C2), C1) == C1
   415	        #   join(Ov(C1, C2), Ov(C1, C2)) == Ov(C1, C2)
   416	        #   join(Ov(C1, C2), Ov(C1, C3)) == C1
   417	        #   join(Ov(C2, C2), C3) == join of fallback types
   418	        #
   419	        # The presence of Any types makes things more interesting. The join is the
   420	        # most general type we can get with respect to Any:
   421	        #
   422	        #   join(Ov([int] -> int, [str] -> str), [Any] -> str) == Any -> str
   423	        #
   424	        # We could use a simplification step that removes redundancies, but that's not
   425	        # implemented right now. Consider this example, where we get a redundancy:
   426	        #
   427	        #   join(Ov([int, Any] -> Any, [str, Any] -> Any), [Any, int] -> Any) ==
   428	        #       Ov([Any, int] -> Any, [Any, int] -> Any)
   429	        #
   430	        # TODO: Consider more cases of callable subtyping.
   431	        result: list[CallableType] = []
   432	        s = self.s
   433	        if isinstance(s, FunctionLike):
   434	            # The interesting case where both types are function types.
   435	            for t_item in t.items:
   436	                for s_item in s.items:
   437	                    if is_similar_callables(t_item, s_item):
   438	                        if is_equivalent(t_item, s_item):
   439	                            result.append(combine_similar_callables(t_item, s_item))
   440	                        elif is_subtype(t_item, s_item):
   441	                            result.append(s_item)
   442	            if result:
   443	                # TODO: Simplify redundancies from the result.
   444	                if len(result) == 1:
   445	                    return result[0]
   446	                else:
   447	                    return Overloaded(result)
   448	            return join_types(t.fallback, s.fallback)
   449	        elif isinstance(s, Instance) and s.type.is_protocol:
   450	            call = unpack_callback_protocol(s)
   451	            if call:
   452	                return join_types(t, call)
   453	        return join_types(t.fallback, s)

   455	    def join_tuples(self, s: TupleType, t: TupleType) -> list[Type] | None:
   456	        """Join two tuple types while handling variadic entries.
   457
   458	        This is surprisingly tricky, and we don't handle some tricky corner cases.
   459	        Most of the trickiness comes from the variadic tuple items like *tuple[X, ...]
   460	        since they can have arbitrary partial overlaps (while *Ts can't be split).
   461	        """
   462	        s_unpack_index = find_unpack_in_list(s.items)
   463	        t_unpack_index = find_unpack_in_list(t.items)
   464	        if s_unpack_index is None and t_unpack_index is None:
   465	            if s.length() == t.length():
   466	                items: list[Type] = []
   467	                for i in range(t.length()):
   468	                    items.append(join_types(t.items[i], s.items[i]))
   469	                return items
   470	            return None
   471	        if s_unpack_index is not None and t_unpack_index is not None:
   472	            # The most complex case: both tuples have an unpack item.
   473	            s_unpack = s.items[s_unpack_index]
   474	            assert isinstance(s_unpack, UnpackType)
   475	            s_unpacked = get_proper_type(s_unpack.type)
   476	            t_unpack = t.items[t_unpack_index]
   477	            assert isinstance(t_unpack, UnpackType)
   478	            t_unpacked = get_proper_type(t_unpack.type)
   479	            if s.length() == t.length() and s_unpack_index == t_unpack_index:
   480	                # We can handle a case where arity is perfectly aligned, e.g.
   481	                # join(Tuple[X1, *tuple[Y1, ...], Z1], Tuple[X2, *tuple[Y2, ...], Z2]).
   482	                # We can essentially perform the join elementwise.
   483	                prefix_len = t_unpack_index
   484	                suffix_len = t.length() - t_unpack_index - 1
   485	                items = []
   486	                for si, ti in zip(s.items[:prefix_len], t.items[:prefix_len]):
   487	                    items.append(join_types(si, ti))
   488	                joined = join_types(s_unpacked, t_unpacked)
   489	                if isinstance(joined, TypeVarTupleType):
   490	                    items.append(UnpackType(joined))
   491	                elif isinstance(joined, Instance) and joined.type.fullname == "builtins.tuple":
   492	                    items.append(UnpackType(joined))
   493	                else:
   494	                    if isinstance(t_unpacked, Instance):
   495	                        assert t_unpacked.type.fullname == "builtins.tuple"
   496	                        tuple_instance = t_unpacked
   497	                    else:
   498	                        assert isinstance(t_unpacked, TypeVarTupleType)
   499	                        tuple_instance = t_unpacked.tuple_fallback
   500	                    items.append(
   501	                        UnpackType(
   502	                            tuple_instance.copy_modified(
   503	                                args=[object_from_instance(tuple_instance)]
   504	                            )
   505	                        )
   506	                    )
   507	                if suffix_len:
   508	                    for si, ti in zip(s.items[-suffix_len:], t.items[-suffix_len:]):
   509	                        items.append(join_types(si, ti))
   510	                return items
   511	            if s.length() == 1 or t.length() == 1:
   512	                # Another case we can handle is when one of tuple is purely variadic
   513	                # (i.e. a non-normalized form of tuple[X, ...]), in this case the join
   514	                # will be again purely variadic.
   515	                if not (isinstance(s_unpacked, Instance) and isinstance(t_unpacked, Instance)):
   516	                    return None
   517	                assert s_unpacked.type.fullname == "builtins.tuple"
   518	                assert t_unpacked.type.fullname == "builtins.tuple"
   519	                mid_joined = join_types(s_unpacked.args[0], t_unpacked.args[0])
   520	                t_other = [a for i, a in enumerate(t.items) if i != t_unpack_index]
   521	                s_other = [a for i, a in enumerate(s.items) if i != s_unpack_index]
   522	                other_joined = join_type_list(s_other + t_other)
   523	                mid_joined = join_types(mid_joined, other_joined)
   524	                return [UnpackType(s_unpacked.copy_modified(args=[mid_joined]))]
   525	            # TODO: are there other case we can handle (e.g. both prefix/suffix are shorter)?
   526	            return None
   527	        if s_unpack_index is not None:
   528	            variadic = s
   529	            unpack_index = s_unpack_index
   530	            fixed = t
   531	        else:
   532	            assert t_unpack_index is not None
   533	            variadic = t
   534	            unpack_index = t_unpack_index
   535	            fixed = s
   536	        # Case where one tuple has variadic item and the other one doesn't. The join will
   537	        # be variadic, since fixed tuple is a subtype of variadic, but not vice versa.
   538	        unpack = variadic.items[unpack_index]
   539	        assert isinstance(unpack, UnpackType)
   540	        unpacked = get_proper_type(unpack.type)
   541	        if not isinstance(unpacked, Instance):
   542	            return None
   543	        if fixed.length() < variadic.length() - 1:
   544	            # There are no non-trivial types that are supertype of both.
   545	            return None
   546	        prefix_len = unpack_index
   547	        suffix_len = variadic.length() - prefix_len - 1
   548	        prefix, middle, suffix = split_with_prefix_and_suffix(
   549	            tuple(fixed.items), prefix_len, suffix_len
   550	        )
   551	        items = []
   552	        for fi, vi in zip(prefix, variadic.items[:prefix_len]):
   553	            items.append(join_types(fi, vi))
   554	        mid_joined = join_type_list(list(middle))
   555	        mid_joined = join_types(mid_joined, unpacked.args[0])
   556	        items.append(UnpackType(unpacked.copy_modified(args=[mid_joined])))
   557	        if suffix_len:
   558	            for fi, vi in zip(suffix, variadic.items[-suffix_len:]):
   559	                items.append(join_types(fi, vi))
   560	        return items

   562	    def visit_tuple_type(self, t: TupleType) -> ProperType:
   563	        # When given two fixed-length tuples:
   564	        # * If they have the same length, join their subtypes item-wise:
   565	        #   Tuple[int, bool] + Tuple[bool, bool] becomes Tuple[int, bool]
   566	        # * If lengths do not match, return a variadic tuple:
   567	        #   Tuple[bool, int] + Tuple[bool] becomes Tuple[int, ...]
   568	        #
   569	        # Otherwise, `t` is a fixed-length tuple but `self.s` is NOT:
   570	        # * Joining with a variadic tuple returns variadic tuple:
   571	        #   Tuple[int, bool] + Tuple[bool, ...] becomes Tuple[int, ...]
   572	        # * Joining with any Sequence also returns a Sequence:
   573	        #   Tuple[int, bool] + List[bool] becomes Sequence[int]
   574	        if isinstance(self.s, TupleType):
   575	            if self.instance_joiner is None:
   576	                self.instance_joiner = InstanceJoiner()
   577	            fallback = self.instance_joiner.join_instances(
   578	                mypy.typeops.tuple_fallback(self.s), mypy.typeops.tuple_fallback(t)
   579	            )
   580	            assert isinstance(fallback, Instance)
   581	            items = self.join_tuples(self.s, t)
   582	            if items is not None:
   583	                if len(items) == 1 and isinstance(item := items[0], UnpackType):
   584	                    if isinstance(unpacked := get_proper_type(item.type), Instance):
   585	                        # Avoid double-wrapping tuple[*tuple[X, ...]]
   586	                        return unpacked
   587	                return TupleType(items, fallback)
   588	            else:
   589	                # TODO: should this be a default fallback behaviour like for meet?
   590	                if is_proper_subtype(self.s, t):
   591	                    return t
   592	                if is_proper_subtype(t, self.s):
   593	                    return self.s
   594	                return fallback
   595	        else:
   596	            return join_types(self.s, mypy.typeops.tuple_fallback(t))

   598	    def visit_typeddict_type(self, t: TypedDictType) -> ProperType:
   599	        if isinstance(self.s, TypedDictType):
   600	            items = {
   601	                item_name: s_item_type
   602	                for (item_name, s_item_type, t_item_type) in self.s.zip(t)
   603	                if (
   604	                    is_equivalent(s_item_type, t_item_type)
   605	                    and (item_name in t.required_keys) == (item_name in self.s.required_keys)
   606	                )
   607	            }
   608	            fallback = self.s.create_anonymous_fallback()
   609	            all_keys = set(items.keys())
   610	            # We need to filter by items.keys() since some required keys present in both t and
   611	            # self.s might be missing from the join if the types are incompatible.
   612	            required_keys = all_keys & t.required_keys & self.s.required_keys
   613	            # If one type has a key as readonly, we mark it as readonly for both:
   614	            readonly_keys = (t.readonly_keys | t.readonly_keys) & all_keys
   615	            return TypedDictType(items, required_keys, readonly_keys, fallback)
   616	        elif isinstance(self.s, Instance):
   617	            return join_types(self.s, t.fallback)
   618	        else:
   619	            return self.default(self.s)

   622	        if isinstance(self.s, LiteralType):
   623	            if t == self.s:
   624	                return t
   625	            if self.s.fallback.type.is_enum and t.fallback.type.is_enum:
   626	                return mypy.typeops.make_simplified_union([self.s, t])
   627	            return join_types(self.s.fallback, t.fallback)
   628	        else:
   629	            return join_types(self.s, t.fallback)

   647	    def default(self, typ: Type) -> ProperType:
   648	        typ = get_proper_type(typ)
   649	        if isinstance(typ, Instance):
   650	            return object_from_instance(typ)
   651	        elif isinstance(typ, TypeType):
   652	            return self.default(typ.item)
   653	        elif isinstance(typ, UnboundType):
   654	            return AnyType(TypeOfAny.special_form)
   655	        elif isinstance(typ, TupleType):
   656	            return self.default(mypy.typeops.tuple_fallback(typ))
   657	        elif isinstance(typ, TypedDictType):
   658	            return self.default(typ.fallback)
   659	        elif isinstance(typ, FunctionLike):
   660	            return self.default(typ.fallback)
   661	        elif isinstance(typ, TypeVarType):
   662	            return self.default(typ.upper_bound)
   663	        elif isinstance(typ, ParamSpecType):
   664	            return self.default(typ.upper_bound)
   665	        else:
   666	            return AnyType(TypeOfAny.special_form)
