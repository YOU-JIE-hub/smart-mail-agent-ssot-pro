   876	    split it into frag words that fit in lines of length
   877	    maxWidth
   878	    maxWidths[lineno+1]
   879	    .....
   880	    maxWidths[lineno+n]

   882	    return the new word list which is either

   894	        newLineWidth = lineWidth+cw
   895	        tooLong = newLineWidth>maxWidth

   897	            f = f.clone()
   898	            if hasattr(f,'text'):
   899	                f.text = fragText

   914	    W.append((f,fragText))
   915	    W = (_SplitFragHS if isinstance(w,_HSFrag) else _SplitFragH)([wordWidth]+W)

   925	def _slash_parts(uri,scheme,slash):
   926	    tail = u''
   927	    while uri.endswith(slash):
   928	        tail += slash
   929	        uri = uri[:-1]
   930
   931	    i = 2
   932	    while True:
   933	        i = uri.find(slash,i)
   934	        if i<0: break
   935	        i += 1
   936	        yield scheme+uri[:i],uri[i:]+tail

   938	def _uri_split_pairs(uri):
   939	    if isBytes(uri): uri = uri.decode('utf8')
   940	    m = uri_pat.match(uri)
   941	    if not m: return None
   942	    scheme = m.group(1)
   943	    uri = uri[len(scheme):]
   944
   945	    slash = (u'\\' if not scheme and u'/' not in uri #might be a microsoft pattern
   946	            else u'/')
   947	    R = ([(scheme, uri)] if scheme and uri else []) + list(_slash_parts(uri,scheme,slash))
   948	    R.reverse()
   949	    return R

   978	            sfx = m.group(0)
   979	            s = s[:-len(sfx)]
   980	        else:
   981	            sfx = u''
   982	    if len(s) < hymwl: return
   983

   985	    R = _uri_split_pairs(s)

  1030	    if a rise is seen or a non-zerowidth cbdefn then we return
  1031	    None. Otherwise we return (uword,([i1,c1],[i2,c2],...])
  1032	    where each ii is the index of the word fragment in the word

  1034	    cc = plen = 0

  1036	    eX = X.extend
  1037	    U = []

  1039	    for i in range(1,len(FW)):

  1044	        if isBytes(t): t = t.decode('utf8')
