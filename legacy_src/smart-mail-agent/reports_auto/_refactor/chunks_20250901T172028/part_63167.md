   412	def _pipepager(
   413	    generator: cabc.Iterable[str], cmd_parts: list[str], color: bool | None
   414	) -> bool:
   415	    """Page through text by feeding it to another program. Invoking a
   416	    pager through this might support colors.
   417
   418	    Returns `True` if the command was found, `False` otherwise and thus another
   419	    pager should be attempted.
   420	    """
   421	    # Split the command into the invoked CLI and its parameters.
   422	    if not cmd_parts:
   423	        return False
   424	    cmd = cmd_parts[0]
   425	    cmd_params = cmd_parts[1:]
   426
   427	    cmd_filepath = which(cmd)
   428	    if not cmd_filepath:
   429	        return False
   430	    # Resolves symlinks and produces a normalized absolute path string.
   431	    cmd_path = Path(cmd_filepath).resolve()
   432	    cmd_name = cmd_path.name
   433
   434	    import subprocess
   435
   436	    # Make a local copy of the environment to not affect the global one.
   437	    env = dict(os.environ)
   438
   439	    # If we're piping to less and the user hasn't decided on colors, we enable
   440	    # them by default we find the -R flag in the command line arguments.
   441	    if color is None and cmd_name == "less":
   442	        less_flags = f"{os.environ.get('LESS', '')}{' '.join(cmd_params)}"
   443	        if not less_flags:
   444	            env["LESS"] = "-R"
   445	            color = True
   446	        elif "r" in less_flags or "R" in less_flags:
   447	            color = True
   448
   449	    c = subprocess.Popen(
   450	        [str(cmd_path)] + cmd_params,
   451	        shell=True,
   452	        stdin=subprocess.PIPE,
   453	        env=env,
   454	        errors="replace",
   455	        text=True,
   456	    )
   457	    assert c.stdin is not None
   458	    try:
   459	        for text in generator:
   460	            if not color:
   461	                text = strip_ansi(text)
   462
   463	            c.stdin.write(text)
   464	    except BrokenPipeError:
   465	        # In case the pager exited unexpectedly, ignore the broken pipe error.
   466	        pass
   467	    except Exception as e:
   468	        # In case there is an exception we want to close the pager immediately
   469	        # and let the caller handle it.
   470	        # Otherwise the pager will keep running, and the user may not notice
   471	        # the error message, or worse yet it may leave the terminal in a broken state.
   472	        c.terminate()
   473	        raise e
   474	    finally:
   475	        # We must close stdin and wait for the pager to exit before we continue
   476	        try:
   477	            c.stdin.close()
   478	        # Close implies flush, so it might throw a BrokenPipeError if the pager
   479	        # process exited already.
   480	        except BrokenPipeError:
   481	            pass
   482
   483	        # Less doesn't respect ^C, but catches it for its own UI purposes (aborting
   484	        # search or other commands inside less).
   485	        #
   486	        # That means when the user hits ^C, the parent process (click) terminates,
   487	        # but less is still alive, paging the output and messing up the terminal.
   488	        #
   489	        # If the user wants to make the pager exit on ^C, they should set
   490	        # `LESS='-K'`. It's not our decision to make.
   491	        while True:
   492	            try:
   493	                c.wait()
   494	            except KeyboardInterrupt:
   495	                pass
   496	            else:
   497	                break
   498
   499	    return True

   754	    def getchar(echo: bool) -> str:
   755	        # The function `getch` will return a bytes object corresponding to
   756	        # the pressed character. Since Windows 10 build 1803, it will also
   757	        # return \x00 when called a second time after pressing a regular key.
   758	        #
   759	        # `getwch` does not share this probably-bugged behavior. Moreover, it
   760	        # returns a Unicode object by default, which is what we want.
   761	        #
   762	        # Either of these functions will return \x00 or \xe0 to indicate
   763	        # a special key, and you need to call the same function again to get
   764	        # the "rest" of the code. The fun part is that \u00e0 is
   765	        # "latin small letter a with grave", so if you type that on a French
   766	        # keyboard, you _also_ get a \xe0.
   767	        # E.g., consider the Up arrow. This returns \xe0 and then \x48. The
   768	        # resulting Unicode string reads as "a with grave" + "capital H".
   769	        # This is indistinguishable from when the user actually types
   770	        # "a with grave" and then "capital H".
   771	        #
   772	        # When \xe0 is returned, we assume it's part of a special-key sequence
   773	        # and call `getwch` again, but that means that when the user types
   774	        # the \u00e0 character, `getchar` doesn't return until a second
   775	        # character is typed.
   776	        # The alternative is returning immediately, but that would mess up
   777	        # cross-platform handling of arrow keys and others that start with
   778	        # \xe0. Another option is using `getch`, but then we can't reliably
   779	        # read non-ASCII characters, because return values of `getch` are
   780	        # limited to the current 8-bit codepage.
   781	        #
   782	        # Anyway, Click doesn't claim to do this Right(tm), and using `getwch`
   783	        # is doing the right thing in more situations than with `getch`.
   784
   785	        if echo:
   786	            func = t.cast(t.Callable[[], str], msvcrt.getwche)
   787	        else:
   788	            func = t.cast(t.Callable[[], str], msvcrt.getwch)

   793	            # \x00 and \xe0 are control characters that indicate special key,
   794	            # see above.
   795	            rv += func()
