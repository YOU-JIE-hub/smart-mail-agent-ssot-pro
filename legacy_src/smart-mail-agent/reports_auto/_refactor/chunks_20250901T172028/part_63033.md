   309	    if min(float(i) for i in n_d) < min_val:
   310	        n_d = _limit_rational(val, abs(min_val))

   313	    if max(n_d_float) > max_val:
   314	        n_d = _limit_rational(n_d_float[0] / n_d_float[1], max_val)
   315
   316	    return n_d
