   124	class PydanticPlugin(Plugin):
   125	    """The Pydantic mypy plugin."""
   126
   127	    def __init__(self, options: Options) -> None:
   128	        self.plugin_config = PydanticPluginConfig(options)
   129	        self._plugin_data = self.plugin_config.to_data()
   130	        super().__init__(options)
   131
   132	    def get_base_class_hook(self, fullname: str) -> Callable[[ClassDefContext], None] | None:
   133	        """Update Pydantic model class."""
   134	        sym = self.lookup_fully_qualified(fullname)
   135	        if sym and isinstance(sym.node, TypeInfo):  # pragma: no branch
   136	            # No branching may occur if the mypy cache has not been cleared
   137	            if sym.node.has_base(BASEMODEL_FULLNAME):
   138	                return self._pydantic_model_class_maker_callback
   139	        return None
   140
   141	    def get_metaclass_hook(self, fullname: str) -> Callable[[ClassDefContext], None] | None:
   142	        """Update Pydantic `ModelMetaclass` definition."""
   143	        if fullname == MODEL_METACLASS_FULLNAME:
   144	            return self._pydantic_model_metaclass_marker_callback
   145	        return None
   146
   147	    def get_method_hook(self, fullname: str) -> Callable[[MethodContext], Type] | None:
   148	        """Adjust return type of `from_orm` method call."""
   149	        if fullname.endswith('.from_orm'):
   150	            return from_attributes_callback
   151	        return None
   152
   153	    def report_config_data(self, ctx: ReportConfigContext) -> dict[str, Any]:
   154	        """Return all plugin config data.
   155
   156	        Used by mypy to determine if cache needs to be discarded.
   157	        """
   158	        return self._plugin_data
   159
   160	    def _pydantic_model_class_maker_callback(self, ctx: ClassDefContext) -> None:
   161	        transformer = PydanticModelTransformer(ctx.cls, ctx.reason, ctx.api, self.plugin_config)
   162	        transformer.transform()
   163
   164	    def _pydantic_model_metaclass_marker_callback(self, ctx: ClassDefContext) -> None:
   165	        """Reset dataclass_transform_spec attribute of ModelMetaclass.
   166
   167	        Let the plugin handle it. This behavior can be disabled
   168	        if 'debug_dataclass_transform' is set to True', for testing purposes.
   169	        """
   170	        if self.plugin_config.debug_dataclass_transform:
   171	            return
   172	        info_metaclass = ctx.cls.info.declared_metaclass
   173	        assert info_metaclass, "callback not passed from 'get_metaclass_hook'"
   174	        if getattr(info_metaclass.type, 'dataclass_transform_spec', None):
   175	            info_metaclass.type.dataclass_transform_spec = None

   178	class PydanticPluginConfig:
   179	    """A Pydantic mypy plugin config holder.
   180
   181	    Attributes:
   182	        init_forbid_extra: Whether to add a `**kwargs` at the end of the generated `__init__` signature.
   183	        init_typed: Whether to annotate fields in the generated `__init__`.
   184	        warn_required_dynamic_aliases: Whether to raise required dynamic aliases error.
   185	        debug_dataclass_transform: Whether to not reset `dataclass_transform_spec` attribute
   186	            of `ModelMetaclass` for testing purposes.
   187	    """
   188
   189	    __slots__ = (
   190	        'init_forbid_extra',
   191	        'init_typed',
   192	        'warn_required_dynamic_aliases',
   193	        'debug_dataclass_transform',
   194	    )
   195	    init_forbid_extra: bool
   196	    init_typed: bool
   197	    warn_required_dynamic_aliases: bool
   198	    debug_dataclass_transform: bool  # undocumented
   199
   200	    def __init__(self, options: Options) -> None:
   201	        if options.config_file is None:  # pragma: no cover
   202	            return
   203
   204	        toml_config = parse_toml(options.config_file)
   205	        if toml_config is not None:
   206	            config = toml_config.get('tool', {}).get('pydantic-mypy', {})
   207	            for key in self.__slots__:
   208	                setting = config.get(key, False)
   209	                if not isinstance(setting, bool):
   210	                    raise ValueError(f'Configuration value must be a boolean for key: {key}')
   211	                setattr(self, key, setting)
   212	        else:
   213	            plugin_config = ConfigParser()
   214	            plugin_config.read(options.config_file)
   215	            for key in self.__slots__:
   216	                setting = plugin_config.getboolean(CONFIGFILE_KEY, key, fallback=False)
   217	                setattr(self, key, setting)
   218
   219	    def to_data(self) -> dict[str, Any]:
   220	        """Returns a dict of config names to their values."""
   221	        return {key: getattr(self, key) for key in self.__slots__}

   224	def from_attributes_callback(ctx: MethodContext) -> Type:
   225	    """Raise an error if from_attributes is not enabled."""
   226	    model_type: Instance
   227	    ctx_type = ctx.type
   228	    if isinstance(ctx_type, TypeType):
   229	        ctx_type = ctx_type.item
   230	    if isinstance(ctx_type, CallableType) and isinstance(ctx_type.ret_type, Instance):
   231	        model_type = ctx_type.ret_type  # called on the class
   232	    elif isinstance(ctx_type, Instance):
   233	        model_type = ctx_type  # called on an instance (unusual, but still valid)
   234	    else:  # pragma: no cover
   235	        detail = f'ctx.type: {ctx_type} (of type {ctx_type.__class__.__name__})'
   236	        error_unexpected_behavior(detail, ctx.api, ctx.context)
   237	        return ctx.default_return_type
   238	    pydantic_metadata = model_type.type.metadata.get(METADATA_KEY)
   239	    if pydantic_metadata is None:
   240	        return ctx.default_return_type
   241	    if not model_type.type.has_base(BASEMODEL_FULLNAME):
   242	        # not a Pydantic v2 model
   243	        return ctx.default_return_type
   244	    from_attributes = pydantic_metadata.get('config', {}).get('from_attributes')
   245	    if from_attributes is not True:
   246	        error_from_attributes(model_type.type.name, ctx.api, ctx.context)
   247	    return ctx.default_return_type

   250	class PydanticModelField:
   251	    """Based on mypy.plugins.dataclasses.DataclassAttribute."""
   252
   253	    def __init__(
   254	        self,
   255	        name: str,
   256	        alias: str | None,
   257	        is_frozen: bool,
   258	        has_dynamic_alias: bool,
   259	        has_default: bool,
   260	        strict: bool | None,
   261	        line: int,
   262	        column: int,
   263	        type: Type | None,
   264	        info: TypeInfo,
   265	    ):
   266	        self.name = name
   267	        self.alias = alias
   268	        self.is_frozen = is_frozen
   269	        self.has_dynamic_alias = has_dynamic_alias
   270	        self.has_default = has_default
   271	        self.strict = strict
   272	        self.line = line
   273	        self.column = column
   274	        self.type = type
   275	        self.info = info
   276
   277	    def to_argument(
   278	        self,
   279	        current_info: TypeInfo,
   280	        typed: bool,
   281	        model_strict: bool,
   282	        force_optional: bool,
   283	        use_alias: bool,
   284	        api: SemanticAnalyzerPluginInterface,
   285	        force_typevars_invariant: bool,
   286	        is_root_model_root: bool,
   287	    ) -> Argument:
   288	        """Based on mypy.plugins.dataclasses.DataclassAttribute.to_argument."""
   289	        variable = self.to_var(current_info, api, use_alias, force_typevars_invariant)
   290
   291	        strict = model_strict if self.strict is None else self.strict
   292	        if typed or strict:
   293	            type_annotation = self.expand_type(current_info, api, include_root_type=True)
   294	        else:
   295	            type_annotation = AnyType(TypeOfAny.explicit)
   296
   297	        return Argument(
   298	            variable=variable,
   299	            type_annotation=type_annotation,
   300	            initializer=None,
   301	            kind=ARG_OPT
   302	            if is_root_model_root
   303	            else (ARG_NAMED_OPT if force_optional or self.has_default else ARG_NAMED),
   304	        )
   305
   306	    def expand_type(
   307	        self,
   308	        current_info: TypeInfo,
   309	        api: SemanticAnalyzerPluginInterface,
   310	        force_typevars_invariant: bool = False,
   311	        include_root_type: bool = False,
   312	    ) -> Type | None:
   313	        """Based on mypy.plugins.dataclasses.DataclassAttribute.expand_type."""
   314	        if force_typevars_invariant:
   315	            # In some cases, mypy will emit an error "Cannot use a covariant type variable as a parameter"
   316	            # To prevent that, we add an option to replace typevars with invariant ones while building certain
   317	            # method signatures (in particular, `__init__`). There may be a better way to do this, if this causes
   318	            # us problems in the future, we should look into why the dataclasses plugin doesn't have this issue.
   319	            if isinstance(self.type, TypeVarType):
   320	                modified_type = self.type.copy_modified()
   321	                modified_type.variance = INVARIANT
   322	                self.type = modified_type
   323
   324	        if self.type is not None and self.info.self_type is not None:
   325	            # In general, it is not safe to call `expand_type()` during semantic analysis,
   326	            # however this plugin is called very late, so all types should be fully ready.
   327	            # Also, it is tricky to avoid eager expansion of Self types here (e.g. because
   328	            # we serialize attributes).
   329	            with state.strict_optional_set(api.options.strict_optional):
   330	                filled_with_typevars = fill_typevars(current_info)
   331	                # Cannot be TupleType as current_info represents a Pydantic model:
   332	                assert isinstance(filled_with_typevars, Instance)
   333	                if force_typevars_invariant:
   334	                    for arg in filled_with_typevars.args:
   335	                        if isinstance(arg, TypeVarType):
   336	                            arg.variance = INVARIANT
   337
   338	                expanded_type = expand_type(self.type, {self.info.self_type.id: filled_with_typevars})
   339	                if include_root_type and isinstance(expanded_type, Instance) and is_root_model(expanded_type.type):
   340	                    # When a root model is used as a field, Pydantic allows both an instance of the root model
   341	                    # as well as instances of the `root` field type:
   342	                    root_type = expanded_type.type['root'].type
   343	                    if root_type is None:
   344	                        # Happens if the hint for 'root' has unsolved forward references
   345	                        return expanded_type
   346	                    expanded_root_type = expand_type_by_instance(root_type, expanded_type)
   347	                    expanded_type = UnionType([expanded_type, expanded_root_type])
   348	                return expanded_type
   349	        return self.type
   350
   351	    def to_var(
   352	        self,
   353	        current_info: TypeInfo,
   354	        api: SemanticAnalyzerPluginInterface,
   355	        use_alias: bool,
   356	        force_typevars_invariant: bool = False,
   357	    ) -> Var:
   358	        """Based on mypy.plugins.dataclasses.DataclassAttribute.to_var."""
   359	        if use_alias and self.alias is not None:
   360	            name = self.alias
   361	        else:
   362	            name = self.name
   363
   364	        return Var(name, self.expand_type(current_info, api, force_typevars_invariant))

   366	    def serialize(self) -> JsonDict:
   367	        """Based on mypy.plugins.dataclasses.DataclassAttribute.serialize."""
   368	        assert self.type
   369	        return {
   370	            'name': self.name,
   371	            'alias': self.alias,
   372	            'is_frozen': self.is_frozen,
   373	            'has_dynamic_alias': self.has_dynamic_alias,
   374	            'has_default': self.has_default,
   375	            'strict': self.strict,
   376	            'line': self.line,
   377	            'column': self.column,
   378	            'type': self.type.serialize(),
   379	        }
   380
   381	    @classmethod
   382	    def deserialize(cls, info: TypeInfo, data: JsonDict, api: SemanticAnalyzerPluginInterface) -> PydanticModelField:
   383	        """Based on mypy.plugins.dataclasses.DataclassAttribute.deserialize."""
   384	        data = data.copy()
   385	        typ = deserialize_and_fixup_type(data.pop('type'), api)
   386	        return cls(type=typ, info=info, **data)

   388	    def expand_typevar_from_subtype(self, sub_type: TypeInfo, api: SemanticAnalyzerPluginInterface) -> None:
   389	        """Expands type vars in the context of a subtype when an attribute is inherited
   390	        from a generic super type.

   392	        if self.type is not None:

   422	class PydanticModelTransformer:
   423	    """Transform the BaseModel subclass according to the plugin settings.
   424
   425	    Attributes:
   426	        tracked_config_fields: A set of field configs that the plugin has to track their value.
   427	    """
   428
   429	    tracked_config_fields: set[str] = {
   430	        'extra',
   431	        'frozen',
   432	        'from_attributes',
   433	        'populate_by_name',
   434	        'validate_by_alias',
   435	        'validate_by_name',
   436	        'alias_generator',
   437	        'strict',
   438	    }
   439
   440	    def __init__(
   441	        self,
   442	        cls: ClassDef,
   443	        reason: Expression | Statement,
   444	        api: SemanticAnalyzerPluginInterface,
   445	        plugin_config: PydanticPluginConfig,
   446	    ) -> None:
   447	        self._cls = cls
   448	        self._reason = reason
   449	        self._api = api
   450
   451	        self.plugin_config = plugin_config
   452
   453	    def transform(self) -> bool:
   454	        """Configures the BaseModel subclass according to the plugin settings.
   455
   456	        In particular:
   457
   458	        * determines the model config and fields,
   459	        * adds a fields-aware signature for the initializer and construct methods
   460	        * freezes the class if frozen = True
   461	        * stores the fields, config, and if the class is settings in the mypy metadata for access by subclasses
   462	        """
   463	        info = self._cls.info
   464	        is_a_root_model = is_root_model(info)
   465	        config = self.collect_config()
   466	        fields, class_vars = self.collect_fields_and_class_vars(config, is_a_root_model)
   467	        if fields is None or class_vars is None:
   468	            # Some definitions are not ready. We need another pass.
   469	            return False
   470	        for field in fields:
   471	            if field.type is None:
   472	                return False
   473
   474	        is_settings = info.has_base(BASESETTINGS_FULLNAME)
   475	        self.add_initializer(fields, config, is_settings, is_a_root_model)
   476	        self.add_model_construct_method(fields, config, is_settings, is_a_root_model)
   477	        self.set_frozen(fields, self._api, frozen=config.frozen is True)
   478
   479	        self.adjust_decorator_signatures()
   480
   481	        in
