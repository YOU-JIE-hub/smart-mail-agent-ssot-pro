    25	class StateTracker:
    26	    """Keeps a stack of transforms and state
    27	    properties.  It can contain any properties you
    28	    want, but the keys 'transform' and 'ctm' have
    29	    special meanings.  The getCTM()
    30	    method returns the current transformation
    31	    matrix at any point, without needing to
    32	    invert matrixes when you pop."""
    33	    def __init__(self, defaults=None, defaultObj=None):
    34	        # one stack to keep track of what changes...
    35	        self._deltas = []
    36
    37	        # and another to keep track of cumulative effects.  Last one in
    38	        # list is the current graphics state.  We put one in to simplify
    39	        # loops below.
    40	        self._combined = []
    41	        if defaults is None:
    42	            defaults = STATE_DEFAULTS.copy()
    43	        if defaultObj:
    44	            for k in STATE_DEFAULTS.keys():
    45	                a = 'initial'+k[:1].upper()+k[1:]
    46	                if hasattr(defaultObj,a):
    47	                    defaults[k] = getattr(defaultObj,a)
    48	        #ensure  that if we have a transform, we have a CTM
    49	        if 'transform' in defaults:
    50	            defaults['ctm'] = defaults['transform']
    51	        self._combined.append(defaults)

    56	    def push(self,delta):
    57	        """Take a new state dictionary of changes and push it onto
    58	        the stack.  After doing this, the combined state is accessible
    59	        through getState()"""
    60
    61	        newstate = self._combined[-1].copy()
    62	        for key, value in delta.items():
    63	            if key == 'transform':  #do cumulative matrix
    64	                newstate['transform'] = delta['transform']
    65	                newstate['ctm'] = mmult(self._combined[-1]['ctm'], delta['transform'])
    66	                #print 'statetracker transform = (%0.2f, %0.2f, %0.2f, %0.2f, %0.2f, %0.2f)' % tuple(newstate['transform'])
    67	                #print 'statetracker ctm = (%0.2f, %0.2f, %0.2f, %0.2f, %0.2f, %0.2f)' % tuple(newstate['ctm'])
    68
    69	            else:  #just overwrite it
    70	                newstate[key] = value
    71
    72	        self._combined.append(newstate)
    73	        self._deltas.append(delta)

    75	    def pop(self):
    76	        """steps back one, and returns a state dictionary with the
    77	        deltas to reverse out of wherever you are.  Depending
    78	        on your back end, you may not need the return value,
    79	        since you can get the complete state afterwards with getState()"""
    80	        del self._combined[-1]
    81	        newState = self._combined[-1]
    82	        lastDelta = self._deltas[-1]
    83	        del  self._deltas[-1]
    84	        #need to diff this against the last one in the state
    85	        reverseDelta = {}
    86	        #print 'pop()...'
    87	        for key, curValue in lastDelta.items():
    88	            #print '   key=%s, value=%s' % (key, curValue)
    89	            prevValue = newState[key]
    90	            if prevValue != curValue:
    91	                #print '    state popping "%s"="%s"' % (key, curValue)
    92	                if key == 'transform':
    93	                    reverseDelta[key] = inverse(lastDelta['transform'])
    94	                else:  #just return to previous state
    95	                    reverseDelta[key] = prevValue
    96	        return reverseDelta
