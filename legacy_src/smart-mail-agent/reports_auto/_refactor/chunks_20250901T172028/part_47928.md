  1550	        common_prefix = min(template_prefix, actual_prefix)
  1551	        common_suffix = min(template_suffix, actual_suffix)

  1554	            # (note however partial overlap is OK for variadic tuples, it is handled below).
  1555	            t_unpack = template_args[template_unpack]
  1556
  1557	    # Handle constraints from prefixes/suffixes first.
  1558	    start, middle, end = split_with_prefix_and_suffix(
  1559	        tuple(actual_args), common_prefix, common_suffix
  1560	    )
  1561	    for t, a in zip(template_args[:common_prefix], start):
  1562	        res.extend(infer_constraints(t, a, direction))
  1563	    if common_suffix:
  1564	        for t, a in zip(template_args[-common_suffix:], end):
  1565	            res.extend(infer_constraints(t, a, direction))
  1566
  1567	    if t_unpack is not None:
  1568	        # Add constraint(s) for variadic item when possible.
  1569	        assert isinstance(t_unpack, UnpackType)
  1570	        tp = get_proper_type(t_unpack.type)
  1571	        if isinstance(tp, Instance) and tp.type.fullname == "builtins.tuple":
  1572	            # Homogeneous case *tuple[T, ...] <: [X, Y, Z, ...].
  1573	            for a in middle:

  1575	                if not isinstance(a, UnpackType):
  1576	                    res.extend(infer_constraints(tp.args[0], a, direction))
  1577	                else:

  1581	                        res.extend(infer_constraints(tp.args[0], a_tp.args[0], direction))
  1582	        elif isinstance(tp, TypeVarTupleType):
  1583	            res.append(Constraint(tp, direction, TupleType(list(middle), tp.tuple_fallback)))
  1584	    elif actual_unpack is not None:

  1586	        # Tuple[..., *tuple[T, ...], ...] <: Tuple[..., *tuple[X, ...], ...].

  1589	        a_unpacked = get_proper_type(actual_unpack_type.type)

  1592	            assert isinstance(t_unpack, UnpackType)
  1593	            tp = get_proper_type(t_unpack.type)
  1594	            if isinstance(tp, Instance) and tp.type.fullname == "builtins.tuple":

  1599	def infer_directed_arg_constraints(left: Type, right: Type, direction: int) -> list[Constraint]:
  1600	    """Infer constraints between two arguments using direction between original callables."""
  1601	    if isinstance(left, (ParamSpecType, UnpackType)) or isinstance(
  1602	        right, (ParamSpecType, UnpackType)
  1603	    ):
  1604	        # This avoids bogus constraints like T <: P.args
  1605	        # TODO: can we infer something useful for *T vs P?
  1606	        return []
  1607	    if direction == SUBTYPE_OF:
  1608	        # We invert direction to account for argument contravariance.
  1609	        return infer_constraints(left, right, neg_op(direction))
  1610	    else:
  1611	        return infer_constraints(right, left, neg_op(direction))

  1614	def infer_callable_arguments_constraints(
  1615	    template: NormalizedCallableType | Parameters,
  1616	    actual: NormalizedCallableType | Parameters,
  1617	    direction: int,
  1618	) -> list[Constraint]:
  1619	    """Infer constraints between argument types of two callables.
  1620
  1621	    This function essentially extracts four steps from are_parameters_compatible() in
  1622	    subtypes.py that involve subtype checks between argument types. We keep the argument
  1623	    matching logic, but ignore various strictness flags present there, and checks that
  1624	    do not involve subtyping. Then in place of every subtype check we put an infer_constraints()
  1625	    call for the same types.
  1626	    """
  1627	    res = []
  1628	    if direction == SUBTYPE_OF:
  1629	        left, right = template, actual
  1630	    else:
  1631	        left, right = actual, template
  1632	    left_star = left.var_arg()
  1633	    left_star2 = left.kw_arg()
  1634	    right_star = right.var_arg()
  1635	    right_star2 = right.kw_arg()
  1636
  1637	    # Numbering of steps below matches the one in are_parameters_compatible() for convenience.
  1638	    # Phase 1a: compare star vs star arguments.
  1639	    if left_star is not None and right_star is not None:
  1640	        res.extend(infer_directed_arg_constraints(left_star.typ, right_star.typ, direction))
  1641	    if left_star2 is not None and right_star2 is not None:
  1642	        res.extend(infer_directed_arg_constraints(left_star2.typ, right_star2.typ, direction))
  1643
  1644	    # Phase 1b: compare left args with corresponding non-star right arguments.
  1645	    for right_arg in right.formal_arguments():
  1646	        left_arg = mypy.typeops.callable_corresponding_argument(left, right_arg)
  1647	        if left_arg is None:
  1648	            continue
  1649	        res.extend(infer_directed_arg_constraints(left_arg.typ, right_arg.typ, direction))
  1650
  1651	    # Phase 1c: compare left args with right *args.
  1652	    if right_star is not None:
  1653	        right_by_position = right.try_synthesizing_arg_from_vararg(None)
  1654	        assert right_by_position is not None
  1655	        i = right_star.pos
  1656	        assert i is not None
  1657	        while i < len(left.arg_kinds) and left.arg_kinds[i].is_positional():
  1658	            left_by_position = left.argument_by_position(i)
  1659	            assert left_by_position is not None
  1660	            res.extend(
  1661	                infer_directed_arg_constraints(
  1662	                    left_by_position.typ, right_by_position.typ, direction
  1663	                )
  1664	            )
  1665	            i += 1
  1666
  1667	    # Phase 1d: compare left args with right **kwargs.
  1668	    if right_star2 is not None:
  1669	        right_names = {name for name in right.arg_names if name is not None}
  1670	        left_only_names = set()
  1671	        for name, kind in zip(left.arg_names, left.arg_kinds):
  1672	            if name is None or kind.is_star() or name in right_names:
  1673	                continue
  1674	            left_only_names.add(name)
  1675
  1676	        right_by_name = right.try_synthesizing_arg_from_kwarg(None)
  1677	        assert right_by_name is not None
  1678	        for name in left_only_names:
  1679	            left_by_name = left.argument_by_name(name)
  1680	            assert left_by_name is not None
  1681	            res.extend(
  1682	                infer_directed_arg_constraints(left_by_name.typ, right_by_name.typ, direction)
  1683	            )
  1684	    return res

  1688	    """For each ParamSpec remove all imprecise constraints, if at least one precise available."""
  1689	    have_precise = set()
  1690	    for c in cs:
  1691	        if not isinstance(c.origin_type_var, ParamSpecType):
  1692	            continue
  1693	        if (
