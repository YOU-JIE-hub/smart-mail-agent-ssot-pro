  472	        # supporting explicitly attached values.  In any case, we will try
   473	        # to long match the option first.
   474	        if "=" in arg:
   475	            long_opt, explicit_value = arg.split("=", 1)
   476	        else:

   478	        norm_long_opt = _normalize_opt(long_opt, self.ctx)

   481	        # the long option matching code.  Note that this allows options
   482	        # like "-foo" to be matched as long options.
   483	        try:
   484	            self._match_long_opt(norm_long_opt, explicit_value, state)
   485	        except NoSuchOption:

   487	            # to try with short options.  However there is a special rule
   488	            # which says, that if we have a two character options prefix

   490	            # short option code and will instead raise the no option
   491	            # error.
   492	            if arg[:2] not in self._opt_prefixes:
   493	                self._match_short_opt(arg, state)
   494	                return
   495
   496	            if not self.ignore_unknown_options:
   497	                raise
   498
   499	            state.largs.append(arg)

   444	def split_arg_string(string: str) -> list[str]:
   445	    """Split an argument string as with :func:`shlex.split`, but don't
   446	    fail if the string is incomplete. Ignores a missing closing quote or
   447	    incomplete escape sequence and uses the partial token as-is.
   448
   449	    .. code-block:: python
   450
   451	        split_arg_string("example 'my file")
   452	        ["example", "my file"]
   453
   454	        split_arg_string("example my\\")
   455	        ["example", "my"]
   456
   457	    :param string: String to split.
   458
   459	    .. versionchanged:: 8.2
   460	        Moved to ``shell_completion`` from ``parser``.
   461	    """
   462	    import shlex
   463
   464	    lex = shlex.shlex(string, posix=True)
   465	    lex.whitespace_split = True
   466	    lex.commenters = ""
   467	    out = []
   468
   469	    try:
   470	        for token in lex:
   471	            out.append(token)
   472	    except ValueError:
   473	        # Raised when end-of-string is reached in an invalid state. Use
   474	        # the partial token as-is. The quote or escape character is in
   475	        # lex.state, not lex.token.
   476	        out.append(lex.token)
   477
   478	    return out
