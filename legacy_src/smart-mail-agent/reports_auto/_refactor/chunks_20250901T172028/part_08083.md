dle, left_suffix = split_with_prefix_and_suffix(
   580	                left.args, prefix, suffix
   581	            )
   582	            right_prefix, right_middle, right_suffix = split_with_prefix_and_suffix(
   583	                right.args, prefix, suffix
   584	            )
   585	            left_args = left_prefix + (TupleType(list(left_middle), fallback),) + left_suffix
   586	            right_args = right_prefix + (TupleType(list(right_middle), fallback),) + right_suffix
   587	        else:
   588	            left_args = left.args
   589	            right_args = right.args
   590	        if len(left_args) == len(right_args):
   591	            # Note: we don't really care about variance here, since the overlapping check
   592	            # is symmetric and since we want to return 'True' even for partial overlaps.
   593	            #
   594	            # For example, suppose we have two types Wrapper[Parent] and Wrapper[Child].
   595	            # It doesn't matter whether Wrapper is covariant or contravariant since
   596	            # either way, one of the two types will overlap with the other.
   597	            #
   598	            # Similarly, if Wrapper was invariant, the two types could still be partially
   599	            # overlapping -- what if Wrapper[Parent] happened to contain only instances of
   600	            # specifically Child?
   601	            #
   602	            # Or, to use a more concrete example, List[Union[A, B]] and List[Union[B, C]]
   603	            # would be considered partially overlapping since it's possible for both lists
   604	            # to contain only instances of B at runtime.
   605	            if all(
   606	                _is_overlapping_types(left_arg, right_arg)
   607	                for left_arg, right_arg in zip(left_args, right_args)
   608	            ):
   609	                return True
   610
   611	        return False
   612
   613	    # We ought to have handled every case by now: we conclude the
   614	    # two types are not overlapping, either completely or partially.
   615	    #
   616	    # Note: it's unclear however, whether returning False is the right thing
   617	    # to do when inferring reachability -- see  https://github.com/python/mypy/issues/5529
   618
   619	    assert type(left) != type(right), f"{type(left)} vs {type(right)}"
   620	    return False
   621

   635	def are_typed_dicts_overlapping(
   636	    left: TypedDictType, right: TypedDictType, is_overlapping: Callable[[Type, Type], bool]
   637	) -> bool:
   638	    """Returns 'true' if left and right are overlapping TypeDictTypes."""
   639	    # All required keys in left are present and overlapping with something in right
   640	    for key in left.required_keys:
   641	        if key not in right.items:
   642	            return False
   643	        if not is_overlapping(left.items[key], right.items[key]):
   644	            return False
   645
   646	    # Repeat check in the other direction
   647	    for key in right.required_keys:
   648	        if key not in left.items:
   649	            return False
   650	        if not is_overlapping(left.items[key], right.items[key]):
   651	            return False
   652
   653	    # The presence of any additional optional keys does not affect whether the two
   654	    # TypedDicts are partially overlapping: the dicts would be overlapping if the
   655	    # keys happened to be missing.
   656	    return True

   659	def are_tuples_overlapping(
   660	    left: Type, right: Type, is_overlapping: Callable[[Type, Type], bool]
   661	) -> bool:
   662	    """Returns true if left and right are overlapping tuples."""
   663	    left, right = get_proper_types((left, right))
   664	    left = adjust_tuple(left, right) or left
   665	    right = adjust_tuple(right, left) or right
   666	    assert isinstance(left, TupleType), f"Type {left} is not a tuple"
   667	    assert isinstance(right, TupleType), f"Type {right} is not a tuple"
   668
   669	    # This algorithm works well if only one tuple is variadic, if both are
   670	    # variadic we may get rare false negatives for overlapping prefix/suffix.
   671	    # Also, this ignores empty unpack case, but it is probably consistent with
   672	    # how we handle e.g. empty lists in overload overlaps.
   673	    # TODO: write a more robust algorithm for cases where both types are variadic.
   674	    left_unpack = find_unpack_in_list(left.items)
   675	    right_unpack = find_unpack_in_list(right.items)
   676	    if left_unpack is not None:
   677	        left = expand_tuple_if_possible(left, len(right.items))
   678	    if right_unpack is not None:
   679	        right = expand_tuple_if_possible(right, len(left.items))
   680
   681	    if len(left.items) != len(right.items):
   682	        return False
   683	    if not all(is_overlapping(l, r) for l, r in zip(left.items, right.items)):
   684	        return False
   685
   686	    # Check that the tuples aren't from e.g. different NamedTuples.
   687	    if is_named_instance(right.partial_fallback, "builtins.tuple") or is_named_instance(
   688	        left.partial_fallback, "builtins.tuple"
   689	    ):
   690	        return True
   691	    else:
   692	        return is_overlapping(left.partial_fallback, right.partial_fallback)
   693

   695	def expand_tuple_if_possible(tup: TupleType, target: int) -> TupleType:
   696	    if len(tup.items) > target + 1:
   697	        return tup
   698	    extra = target + 1 - len(tup.items)
   699	    new_items = []
   700	    for it in tup.items:
   701	        if not isinstance(it, UnpackType):
   702	            new_items.append(it)
   703	            continue
   704	        unpacked = get_proper_type(it.type)
   705	        if isinstance(unpacked, TypeVarTupleType):
   706	            instance = unpacked.tuple_fallback
   707	        else:
   708	            # Nested non-variadic tuples should be normalized at this point.
   709	            assert isinstance(unpacked, Instance)
   710	            instance = unpacked
   711	        assert instance.type.fullname == "builtins.tuple"
   712	        new_items.extend([instance.args[0]] * extra)
   713	    return tup.copy_modified(items=new_items)

   716	def adjust_tuple(left: ProperType, r: ProperType) -> TupleType | None:
   717	    """Find out if `left` is a Tuple[A, ...], and adjust its length to `right`"""
   718	    if isinstance(left, Instance) and left.type.fullname == "builtins.tuple":
   719	        n = r.length() if isinstance(r, TupleType) else 1
   720	        return TupleType([left.args[0]] * n, left)
   721	    return None

   724	def is_tuple(typ: Type) -> bool:
   725	    typ = get_proper_type(typ)
   726	    return isinstance(typ, TupleType) or (
   727	        isinstance(typ, Instance) and typ.type.fullname == "builtins.tuple"
   728	    )

   731	class TypeMeetVisitor(TypeVisitor[ProperType]):
   732	    def __init__(self, s: ProperType) -> None:
   733	        self.s = s
   734
   735	    def visit_unbound_type(self, t: UnboundType) -> ProperType:
   736	        if isinstance(self.s, NoneType):
   737	            if state.strict_optional:
   738	                return UninhabitedType()
   739	            else:
   740	                return self.s
   741	        elif isinstance(self.s, UninhabitedType):
   742	            return self.s
   743	        else:
   744	            return AnyType(TypeOfAny.special_form)
   745
   746	    def visit_any(self, t: AnyType) -> ProperType:
   747	        return self.s
   748
   749	    def visit_union_type(self, t: UnionType) -> ProperType:
   750	        if isinstance(self.s, UnionType):
   751	            meets: list[Type] = []
   752	            for x in t.items:
   753	                for y in self.s.items:
   754	                    meets.append(meet_types(x, y))
   755	        else:
   756	            meets = [meet_types(x, self.s) for x in t.items]
   757	        return make_simplified_union(meets)
   758
   759	    def visit_none_type(self, t: NoneType) -> ProperType:
   760	        if state.strict_optional:
   761	            if isinstance(self.s, NoneType) or (
   762	                isinstance(self.s, Instance) and self.s.type.fullname == "builtins.object"
   763	            ):
   764	                return t
   765	            else:
   766	                return UninhabitedType()
   767	        else:
   768	            return t
   769
   770	    def visit_uninhabited_type(self, t: UninhabitedType) -> ProperType:
   771	        return t
   772
   773	    def visit_deleted_type(self, t: DeletedType) -> ProperType:
   774	        if isinstance(self.s, NoneType):
   775	            if state.strict_optional:
   776	                return t
   777	            else:
   778	                return self.s
   779	        elif isinstance(self.s, UninhabitedType):
   780	            return self.s
   781	        else:
   782	            return t
   783
   784	    def visit_erased_type(self, t: ErasedType) -> ProperType:
   785	        return self.s
   786
   787	    def visit_type_var(self, t: TypeVarType) -> ProperType:
   788	        if isinstance(self.s, TypeVarType) and self.s.id == t.id:
   789	            if self.s.upper_bound == t.upper_bound:
   790	                return self.s
   791	            return self.s.copy_modified(upper_bound=self.meet(self.s.upper_bound, t.upper_bound))
   792	        else:
   793	            return self.default(self.s)
   794
   795	    def visit_param_spec(self, t: ParamSpecType) -> ProperType:
   796	        if self.s == t:
   797	            return self.s
   798	        else:
   799	            return self.default(self.s)
   800
   801	    def visit_type_var_tuple(self, t: TypeVarTupleType) -> ProperType:
   802	        if isinstance(self.s, TypeVarTupleType) and self.s.id == t.id:
   803	            return self.s if self.s.min_len > t.min_len else t
   804	        else:
   805	            return self.default(self.s)
   806
   807	    def visit_unpack_type(self, t: UnpackType) -> ProperType:
   808	        raise NotImplementedError
   809
   810	    def visit_parameters(self, t: Parameters) -> ProperType:
   811	        if isinstance(self.s, Parameters):
   812	            if len(t.arg_types) != len(self.s.arg_types):
   813	                return self.default(self.s)
   814	            from mypy.join import join_types
   815
   816	            return t.copy_modified(
   817	                arg_types=[join_types(s_a, t_a) for s_a, t_a in zip(self.s.arg_types, t.arg_types)]
   818	            )
   819	        else:
   820	            return self.default(self.s)
   821
   822	    def visit_instance(self, t: Instance) -> ProperType:
   823	        if isinstance(self.s, Instance):
   824	            if t.type == self.s.type:
   825	                if is_subtype(t, self.s) or is_subtype(self.s, t):
   826	                    # Combine type arguments. We could have used join below
   827	                    # equivalently.
   828	                    args: list[Type] = []
   829	                    # N.B: We use zip instead of indexing because the lengths might have
   830	                    # mismatches during daemon reprocessing.
   831	                    if t.type.has_type_var_tuple_type:
   832	                        # We handle meet of variadic instances by simply creating correct mapping
   833	                        # for type arguments and compute the individual meets same as for regular
   834	                        # instances. All the heavy lifting is done in the meet of tuple types.
   835	                        s = self.s
   836	                        assert s.type.type_var_tuple_prefix is not None
   837	                        assert s.type.type_var_tuple_suffix is not None
   838	                        prefix = s.type.type_var_tuple_prefix
   839	                        suffix = s.type.type_var_tuple_suffix
   840	                        tvt = s.type.defn.type_vars[prefix]
   841	                        assert isinstance(tvt, TypeVarTupleType)
   842	                        fallback = tvt.tuple_fallback
   843	                        s_prefix, s_middle, s_suffix = split_with_prefix_and_suffix(
   844	                            s.args, prefix, suffix
   845	                        )
   846	                        t_prefix, t_middle, t_suffix = split_with_prefix_and_suffix(
   847	                            t.args, prefix, suffix
   848	                        )
   849	                        s_args = s_prefix + (TupleType(list(s_middle), fallback),) + s_suffix
   850	                        t_args = t_prefix + (TupleType(list(t_middle), fallback),) + t_suffix
   851	                    else:
   852	                        t_args = t.args
   853	                        s_args = self.s.args
   854	                    for ta, sa, tv in zip(t_args, s_args, t.type.defn.type_vars):
   855	                        meet = self.meet(ta, sa)
   856	                        if isinstance(tv, TypeVarTupleType):
   857	                            # Correctly unpack possible outcomes of meets of tuples: it can be
   858	                            # either another tuple type or Never (normalized as *tuple[Never, ...])
   859	                            if isinstance(meet, TupleType):
   860	                                args.extend(meet.items)
   861	                                continue
   862	                            else:
   863	                                assert isinstance(meet, UninhabitedType)
   864	                                meet = UnpackType(tv.tuple_fallback.copy_modified(args=[meet]))
   865	                        args.append(meet)
   866	                    return Instance(t.type, args)
   867	                else:
   868	                    if state.strict_optional:
   869	                        return UninhabitedType()
   870	                    else:
   871	                        return NoneType()
   872	            else:
   873	                alt_promote = t.type.alt_promote
   874	                if alt_promote and alt_promote.type is self.s.type:
   875	                    return t
   876	                alt_promote = self.s.type.alt_promote
   877	                if alt_promote and alt_promote.type is t.type:
   878	                    return self.s
   879	                if is_subtype(t, self.s):
   880	                    return t
   881	                elif is_subtype(self.s, t):
   882	                    # See also above comment.
   883	                    return self.s
   884	                else:
   885	                    if state.strict_optional:
   886	                        return UninhabitedType()
   887	                    else:
   888	                        return NoneType()
   889	        elif isinstance(self.s, FunctionLike) and t.type.is_protocol:
   890	            call = join.unpack_callback_protocol(t)
   891	            if call:
   892	                return meet_types(call, self.s)
   893	        elif isinstance(self.s, FunctionLike) and self.s.is_type_obj() and t.type.is_metaclass():
   894	            if is_subtype(self.s.fallback, t):
   895	                return self.s
   896	            return self.default(self.s)
   897	        elif isinstance(self.s, TypeType):
   898	            return meet_types(t, self.s)
   899	        elif isinstance(self.s, TupleType):
   900	            return meet_types(t, self.s)
   901	        elif isinstance(self.s, LiteralType):
   902	            return meet_types(t, self.s)
   903	        elif isinstance(self.s, TypedDictType):
   904	            return meet_types(t, self.s)
   905	        return self.default(self.s)
   906
   907	    def visit_callable_type(self, t: CallableType) -> ProperType:
   908	        if isinstance(self.s, CallableType) and join.is_similar_callables(t, self.s):
   909	            if is_equivalent(t, self.s):
   910	                return join.combine_similar_callables(t, self.s)
   911	            result = meet_similar_callables(t, self.s)
   912	            # We set the from_type_type flag to suppress error when a collection of
   913	            # concrete class objects gets inferred as their common abstract superclass.
   914	            if not (
   915	                (t.is_type_obj() and t.type_object().is_abstract)
   916	                or (self.s.is_type_obj() and self.s.type_object().is_abstract)
   917	            ):
   918	                result.from_type_type = True
   919	            if isinstance(get_proper_type(result.ret_type), UninhabitedType):
   920	                # Return a plain None or <uninhabited> instead of a weird function.
   921	                return self.default(self.s)
   922	            return result
   923	        elif isinstance(self.s, TypeType) and t.is_type_obj() and not t.is_generic():
   924	            # In this case we are able to potentially produce a better meet.
   925	            res = meet_types(self.s.item, t.ret_type)
   926	            if not isinstance(res, (NoneType, UninhabitedType)):
   927	                return TypeType.make_normalized(res)
   928	            return self.default(self.s)
   929	        elif isinstance(self.s, Instance) and self.s.type.is_protocol:
   930	            call = join.unpack_callback_protocol(self.s)
   931	            if call:
   932	                return meet_types(t, call)
   933	        return self.default(self.s)

   935	    def visit_overloaded(self, t: Overloaded) -> ProperType:
   936	        # TODO: Implement a better algorithm that covers at least the same cases
   937	        # as TypeJoinVisitor.visit_overloaded().
   938	        s = self.s
   939	        if isinstance(s, FunctionLike):
   940	            if s.items == t.items:
   941	                return Overloaded(t.items)
   942	            elif is_subtype(s, t):
   943	                return s
   944	            elif is_subtype(t, s):
   945	                return t
   946	            else:
   947	                return meet_types(t.fallback, s.fallback)
   948	        elif isinstance(self.s, Instance) and self.s.type.is_protocol:
   949	            call = join.unpack_callback_protocol(self.s)
   950	            if call:
   951	                return meet_types(t, call)
   952	        return meet_types(t.fallback, s)

   954	    def meet_tuples(self, s: TupleType, t: TupleType) -> list[Type] | None:
   955	        """Meet two tuple types while handling variadic entries.
   956
   957	        This is surprisingly tricky, and we don't handle some tricky corner cases.
   958	        Most of the trickiness comes from the variadic tuple items like *tuple[X, ...]
   959	        since they can have arbitrary partial overlaps (while *Ts can't be split). This
   960	        function is roughly a mirror of join_tuples() w.r.t. to the fact that fixed
   961	        tuples are subtypes of variadic ones but not vice versa.
   962	        """
   963	        s_unpack_index = find_unpack_in_list(s.items)
   964	        t_unpack_index = find_unpack_in_list(t.items)
   965	        if s_unpack_index is None and t_unpack_index is None:
   966	            if s.length() == t.length():
   967	                items: list[Type] = []
   968	                for i in range(t.length()):
   969	                    items.append(self.meet(t.items[i], s.items[i]))
   970	                return items
   971	            return None
   972	        if s_unpack_index is not None and t_unpack_index is not None:
   973	            # The only simple case we can handle if both tuples are variadic
   974	            # is when their structure fully matches. Other cases are tricky because
   975	            # a variadic item is effectively a union of tuples of all length, thus
   976	            # potentially causing overlap between a suffix in `s` and a prefix
   977	            # in `t` (see how this is handled in is_subtype() for details).
