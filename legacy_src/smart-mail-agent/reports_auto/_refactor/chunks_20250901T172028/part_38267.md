   944	        self,

   946	        args: Sequence[Type] | None = None,
   947	        line: int = -1,
   948	        column: int = -1,

   950	        empty_tuple_index: bool = False,

   952	        original_str_fallback: str | None = None,

   975	        # so we don't have to try and recompute it later

   977	        self.original_str_fallback = original_str_fallback

   981	            args = self.args
   982	        return UnboundType(
   983	            name=self.name,
   984	            args=args,
   985	            line=self.line,
   986	            column=self.column,
   987	            optional=self.optional,
   988	            empty_tuple_index=self.empty_tuple_index,

   990	            original_str_fallback=self.original_str_fallback,
   991	        )

   994	        return visitor.visit_unbound_type(self)
   995
   996	    def __hash__(self) -> int:

  1001	            return NotImplemented
  1002	        return (
  1003	            self.name == other.name
  1004	            and self.optional == other.optional
  1005	            and self.args == other.args

  1012	            ".class": "UnboundType",
