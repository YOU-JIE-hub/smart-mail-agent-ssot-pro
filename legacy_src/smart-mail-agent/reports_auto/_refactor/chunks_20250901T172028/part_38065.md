  4562	    def try_getting_int_literals(self, index: Expression) -> list[int] | None:
  4563	        """If the given expression or type corresponds to an int literal
  4564	        or a union of int literals, returns a list of the underlying ints.
  4565	        Otherwise, returns None.
  4566
  4567	        Specifically, this function is guaranteed to return a list with
  4568	        one or more ints if one the following is true:
  4569
  4570	        1. 'expr' is a IntExpr or a UnaryExpr backed by an IntExpr
  4571	        2. 'typ' is a LiteralType containing an int
  4572	        3. 'typ' is a UnionType containing only LiteralType of ints
  4573	        """
  4574	        if isinstance(index, IntExpr):
  4575	            return [index.value]
  4576	        elif isinstance(index, UnaryExpr):
  4577	            if index.op == "-":
  4578	                operand = index.expr
  4579	                if isinstance(operand, IntExpr):
  4580	                    return [-1 * operand.value]
  4581	            if index.op == "+":
  4582	                operand = index.expr
  4583	                if isinstance(operand, IntExpr):
  4584	                    return [operand.value]
  4585	        typ = get_proper_type(self.accept(index))
  4586	        if isinstance(typ, Instance) and typ.last_known_value is not None:
  4587	            typ = typ.last_known_value
  4588	        if isinstance(typ, LiteralType) and isinstance(typ.value, int):
  4589	            return [typ.value]
  4590	        if isinstance(typ, UnionType):
  4591	            out = []
  4592	            for item in get_proper_types(typ.items):
  4593	                if isinstance(item, LiteralType) and isinstance(item.value, int):
  4594	                    out.append(item.value)
  4595	                else:
  4596	                    return None
  4597	            return out
  4598	        return None
