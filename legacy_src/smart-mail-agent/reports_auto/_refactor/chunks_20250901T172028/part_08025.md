  1304	    def apply_signature_hook(
  1305	        self,
  1306	        callee: FunctionLike,
  1307	        args: list[Expression],
  1308	        arg_kinds: list[ArgKind],
  1309	        arg_names: Sequence[str | None] | None,
  1310	        hook: Callable[[list[list[Expression]], CallableType], FunctionLike],
  1311	    ) -> FunctionLike:
  1312	        """Helper to apply a signature hook for either a function or method"""
  1313	        if isinstance(callee, CallableType):
  1314	            num_formals = len(callee.arg_kinds)
  1315	            formal_to_actual = map_actuals_to_formals(
  1316	                arg_kinds,
  1317	                arg_names,
  1318	                callee.arg_kinds,
  1319	                callee.arg_names,
  1320	                lambda i: self.accept(args[i]),
  1321	            )
  1322	            formal_arg_exprs: list[list[Expression]] = [[] for _ in range(num_formals)]
  1323	            for formal, actuals in enumerate(formal_to_actual):
  1324	                for actual in actuals:
  1325	                    formal_arg_exprs[formal].append(args[actual])
  1326	            return hook(formal_arg_exprs, callee)
  1327	        else:
  1328	            assert isinstance(callee, Overloaded)
  1329	            items = []
  1330	            for item in callee.items:
  1331	                adjusted = self.apply_signature_hook(item, args, arg_kinds, arg_names, hook)
  1332	                assert isinstance(adjusted, CallableType)
  1333	                items.append(adjusted)
  1334	            return Overloaded(items)

  1378	    def transform_callee_type(
  1379	        self,
  1380	        callable_name: str | None,
  1381	        callee: Type,
  1382	        args: list[Expression],
  1383	        arg_kinds: list[ArgKind],
  1384	        context: Context,
  1385	        arg_names: Sequence[str | None] | None = None,
  1386	        object_type: Type | None = None,
  1387	    ) -> Type:
  1388	        """Attempt to determine a more accurate signature for a method call.
  1389
  1390	        This is done by looking up and applying a method signature hook (if one exists for the
  1391	        given method name).
  1392
  1393	        If no matching method signature hook is found, callee is returned unmodified. The same
  1394	        happens if the arguments refer to a non-method callable (this is allowed so that the code
  1395	        calling transform_callee_type needs to perform fewer boilerplate checks).
  1396
  1397	        Note: this method is *not* called automatically as part of check_call, because in some
  1398	        cases check_call is called multiple times while checking a single call (for example when
  1399	        dealing with overloads). Instead, this method needs to be called explicitly
  1400	        (if appropriate) before the signature is passed to check_call.
  1401	        """
  1402	        callee = get_proper_type(callee)
  1403	        if callable_name is not None and isinstance(callee, FunctionLike):
  1404	            if object_type is not None:
  1405	                method_sig_hook = self.plugin.get_method_signature_hook(callable_name)
  1406	                if method_sig_hook:
  1407	                    return self.apply_method_signature_hook(
  1408	                        callee, args, arg_kinds, context, arg_names, object_type, method_sig_hook
  1409	                    )
  1410	            else:
  1411	                function_sig_hook = self.plugin.get_function_signature_hook(callable_name)
  1412	                if function_sig_hook:
  1413	                    return self.apply_function_signature_hook(
  1414	                        callee, args, arg_kinds, context, arg_names, function_sig_hook
  1415	                    )
  1416
  1417	        return callee

  1419	    def is_generic_decorator_overload_call(
  1420	        self, callee_type: CallableType, args: list[Expression]
  1421	    ) -> Overloaded | None:
  1422	        """Check if this looks like an application of a generic function to overload argument."""
  1423	        assert callee_type.variables
  1424	        if len(callee_type.arg_types) != 1 or len(args) != 1:
  1425	            # TODO: can we handle more general cases?
  1426	            return None
  1427	        if not isinstance(get_proper_type(callee_type.arg_types[0]), CallableType):
  1428	            return None
  1429	        if not isinstance(get_proper_type(callee_type.ret_type), CallableType):
  1430	            return None
  1431	        with self.chk.local_type_map():
  1432	            with self.msg.filter_errors():
  1433	                arg_type = get_proper_type(self.accept(args[0], type_context=None))
  1434	        if isinstance(arg_type, Overloaded):
  1435	            return arg_type
  1436	        return None

  1438	    def handle_decorator_overload_call(
  1439	        self, callee_type: CallableType, overloaded: Overloaded, ctx: Context
  1440	    ) -> tuple[Type, Type] | None:
  1441	        """Type-check application of a generic callable to an overload.
  1442
  1443	        We check call on each individual overload item, and then combine results into a new
  1444	        overload. This function should be only used if callee_type takes and returns a Callable.
  1445	        """
  1446	        result = []
  1447	        inferred_args = []
  1448	        for item in overloaded.items:
  1449	            arg = TempNode(typ=item)
  1450	            with self.msg.filter_errors() as err:
  1451	                item_result, inferred_arg = self.check_call(callee_type, [arg], [ARG_POS], ctx)
  1452	            if err.has_new_errors():
  1453	                # This overload doesn't match.
  1454	                continue
  1455	            p_item_result = get_proper_type(item_result)
  1456	            if not isinstance(p_item_result, CallableType):
  1457	                continue
  1458	            p_inferred_arg = get_proper_type(inferred_arg)
  1459	            if not isinstance(p_inferred_arg, CallableType):
  1460	                continue
  1461	            inferred_args.append(p_inferred_arg)
  1462	            result.append(p_item_result)
  1463	        if not result or not inferred_args:
  1464	            # None of the overload matched (or overload was initially malformed).
  1465	            return None
  1466	        return Overloaded(result), Overloaded(inferred_args)

  1483	        If 'callable_name' is None but 'member' is not None (member call), try constructing
  1484	        'callable_name' using 'object_type' (the base type on which the method is called),
  1485	        for example 'typing.Mapping.get'.

  1487	        if callable_name is None and member is not None:
  1488	            assert object_type is not None
  1489	            callable_name = self.method_fullname(object_type, member)
  1490	        object_type = get_proper_type(object_type)

  1492	            # Try to refine the call signature using plugin hooks before checking the call.
  1493	            callee_type = self.transform_callee_type(

  1495	            )

  1497	        elif member is not None and isinstance(object_type, UnionType):
  1498	            return self.check_union_call_expr(e, object_type, member)

  1500	            callee_type,
  1501	            e.args,
  1502	            e.arg_kinds,
  1503	            e,
  1504	            e.arg_names,
  1505	            callable_node=e.callee,
  1506	            callable_name=callable_name,
  1507	            object_type=object_type,
  1508	        )
  1509	        proper_callee = get_proper_type(callee_type)
  1510	        if isinstance(e.callee, (NameExpr, MemberExpr)):
  1511	            node = e.callee.node
  1512	            if node is None and member is not None and isinstance(object_type, Instance):
  1513	                if (symbol := object_type.type.get(member)) is not None:
  1514	                    node = symbol.node
  1515	            self.chk.check_deprecated(node, e)
  1516	            self.chk.warn_deprecated_overload_item(
  1517	                node, e, target=callee_type, selftype=object_type
  1518	            )
  1519	        if isinstance(e.callee, RefExpr) and isinstance(proper_callee, CallableType):
  1520	            # Cache it for find_isinstance_check()
  1521	            if proper_callee.type_guard is not None:
  1522	                e.callee.type_guard = proper_callee.type_guard
  1523	            if proper_callee.type_is is not None:
  1524	                e.callee.type_is = proper_callee.type_is
