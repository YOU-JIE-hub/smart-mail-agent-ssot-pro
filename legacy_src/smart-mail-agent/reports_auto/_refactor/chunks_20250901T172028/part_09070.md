    19	        out.write_bytes(b"%PDF-FAKE\n" + text.encode("utf-8"))
    20	        return str(out)
    21	    except Exception:
    22	        alt = out.with_suffix(".txt")
    23	        alt.write_text(text, encoding="utf-8")
    24	        return str(alt)
