   281	class CheckerScope:
   282	    # We keep two stacks combined, to maintain the relative order
   283	    stack: list[TypeInfo | FuncItem | MypyFile]
   284
   285	    def __init__(self, module: MypyFile) -> None:
   286	        self.stack = [module]
   287
   288	    def current_function(self) -> FuncItem | None:
   289	        for e in reversed(self.stack):
   290	            if isinstance(e, FuncItem):
   291	                return e
   292	        return None
   293
   294	    def top_level_function(self) -> FuncItem | None:
   295	        """Return top-level non-lambda function."""
   296	        for e in self.stack:
   297	            if isinstance(e, FuncItem) and not isinstance(e, LambdaExpr):
   298	                return e
   299	        return None
   300
   301	    def active_class(self) -> TypeInfo | None:
   302	        if isinstance(self.stack[-1], TypeInfo):
   303	            return self.stack[-1]
   304	        return None
   305
   306	    def enclosing_class(self, func: FuncItem | None = None) -> TypeInfo | None:
   307	        """Is there a class *directly* enclosing this function?"""
   308	        func = func or self.current_function()
   309	        assert func, "This method must be called from inside a function"
   310	        index = self.stack.index(func)
   311	        assert index, "CheckerScope stack must always start with a module"
   312	        enclosing = self.stack[index - 1]
   313	        if isinstance(enclosing, TypeInfo):
   314	            return enclosing
   315	        return None
   316
   317	    def active_self_type(self) -> Instance | TupleType | None:
   318	        """An instance or tuple type representing the current class.
   319
   320	        This returns None unless we are in class body or in a method.
   321	        In particular, inside a function nested in method this returns None.
   322	        """
   323	        info = self.active_class()
   324	        if not info and self.current_function():
   325	            info = self.enclosing_class()
   326	        if info:
   327	            return fill_typevars(info)
   328	        return None
   329
   330	    def current_self_type(self) -> Instance | TupleType | None:
   331	        """Same as active_self_type() but handle functions nested in methods."""
   332	        for item in reversed(self.stack):
   333	            if isinstance(item, TypeInfo):
   334	                return fill_typevars(item)
   335	        return None
   336
   337	    @contextmanager
   338	    def push_function(self, item: FuncItem) -> Iterator[None]:
   339	        self.stack.append(item)
   340	        yield
   341	        self.stack.pop()
   342
   343	    @contextmanager
   344	    def push_class(self, info: TypeInfo) -> Iterator[None]:
   345	        self.stack.append(info)
   346	        yield
   347	        self.stack.pop()

   253	    """Recursively extracts all module references from a reference expression.

   255	    Note that currently, the only two subclasses of RefExpr are NameExpr and

   257	    output: set[str] = set()
   258	    while isinstance(expr.node, MypyFile) or expr.fullname:
