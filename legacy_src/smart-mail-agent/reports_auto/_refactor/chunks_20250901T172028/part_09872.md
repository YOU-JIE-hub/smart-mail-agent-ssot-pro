   595	    def txAP(self,
   596	                key,                    #N/D/R
   597	                value,
   598	                iFontName,
   599	                rFontName,
   600	                fontSize,
   601	                shape='square',
   602	                fillColor=None,
   603	                borderColor=None,
   604	                textColor=None,
   605	                borderWidth=1,
   606	                borderStyle='solid',
   607	                width=120,
   608	                height=36,
   609	                dashLen=3,
   610	                wkind='textfield',
   611	                labels=[],
   612	                I=[],
   613	                sel_bg='0.600006 0.756866 0.854904 rg',
   614	                sel_fg='0 g',
   615	                ):
   616	        stream = [].append
   617	        if opaqueColor(fillColor):
   618	            streamFill = self.streamFillColor(fillColor)
   619	            stream('%(streamFill)s\n0 0 %(width)s %(height)s re\nf')
   620	        if borderWidth!=None and borderWidth>0 and opaqueColor(borderColor):
   621	            hbw = borderWidth*0.5
   622	            bww = width - borderWidth
   623	            bwh = height - borderWidth
   624	            _2bw = 2*borderWidth
   625	            if borderStyle in ('bevelled','inset'):
   626	                bw2w = width - _2bw
   627	                bw2h = height - _2bw
   628	                if borderStyle == 'bevelled':
   629	                    bbs0 = '1 g'
   630	                    if fillColor or borderColor:
   631	                        bbs1 = '-0.250977 0.749023 -0.250977 rg'
   632	                    else:
   633	                        bbs1 = '.75293 g'
   634	                else:
   635	                    bbs0 = '.501953 g'
   636	                    bbs1 = '.75293 g'
   637	                stream('%(bbs0)s\n%(borderWidth)s %(borderWidth)s m\n%(borderWidth)s %(bwh)s l\n%(bww)s %(bwh)s l\n%(bw2w)s %(bw2h)s l\n%(_2bw)s %(bw2h)s l\n%(_2bw)s %(_2bw)s l\nf\n%(bbs1)s\n%(bww)s %(bwh)s m\n%(bww)s %(borderWidth)s l\n%(borderWidth)s %(borderWidth)s l\n%(_2bw)s %(_2bw)s l\n%(bw2w)s %(_2bw)s l\n%(bw2w)s %(bw2h)s l\nf')
   638	        else:
   639	            hbw = _2bw = borderWidth = 0
   640	            bww = width
   641	            bwh = height
   642	        undash = ''
   643	        if opaqueColor(borderColor) and borderWidth:
   644	            streamStroke = self.streamStrokeColor(borderColor)
   645	            if borderStyle=='underlined':
   646	                stream('%(streamStroke)s %(borderWidth)s w 0 %(hbw)s m %(width)s %(hbw)s l s')
   647	            elif borderStyle in ('dashed','inset','bevelled','solid'):
   648	                if borderStyle=='dashed':
   649	                    dash = '\n[%s ] 0 d\n' % fp_str(dashLen)
   650	                    undash = '[] 0 d'
   651	                else:
   652	                    dash = '\n%s w' % borderWidth
   653	                stream('%(streamStroke)s\n%(dash)s\n%(hbw)s %(hbw)s %(bww)s %(bwh)s re\ns')
   654	        _4bw = 4*borderWidth
   655	        w4bw = width - _4bw
   656	        h4bw = height - _4bw
   657	        textFill = self.streamFillColor(textColor)
   658	        stream('/Tx BMC \nq\n%(_2bw)s %(_2bw)s %(w4bw)s %(h4bw)s re\nW\nn')
   659	        leading = 1.2 * fontSize
   660	        if wkind=='listbox':
   661	            nopts = int(h4bw/leading)
   662	            leading = h4bw/float(nopts)
   663	            if nopts>len(labels):
   664	                i0 = 0
   665	                nopts = len(labels)
   666	            elif len(I)<=1:
   667	                i0 = I[0] if I else 0
   668	                if i0:
   669	                    if i0<nopts:
   670	                        i0 = 0
   671	                    else:
   672	                        i = len(labels) - nopts
   673	                        if i0>=i:
   674	                            i0 = i
   675	            else:   #|I|>1
   676	                if I[1]<nopts:
   677	                    i0 = 0
   678	                else:
   679	                    i0 = I[0]
   680	            y = len(labels)
   681	            i = i0 + nopts
   682	            if i>y: i0 = i - y
   683	            ilim = min(y,i0+nopts)
   684	            if I:
   685	                i = i0
   686	                y = height - _2bw - leading
   687	                stream(sel_bg)
   688	                while i<ilim:
   689	                    if i in I:
   690	                        #draw selected bg
   691	                        stream('%%(_2bw)s %s %%(w4bw)s %%(leading)s re\nf' % fp_str(y))
   692	                    y -= leading
   693	                    i += 1
   694	            i = i0
   695	            y = height - _2bw - fontSize
   696	            stream('0 g\n0 G\n%(undash)s')
   697	            while i<ilim:
   698	                stream('BT')
   699	                if i==i0:
   700	                    stream('/%(iFontName)s %(fontSize)s Tf')
   701	                stream(sel_fg if i in I else '%(textFill)s')
   702	                stream('%%(_4bw)s %s Td\n(%s) Tj' % (fp_str(y),escPDF(labels[i])))
   703	                y -= leading
   704	                i += 1
   705	                stream('ET')
   706	        else:
   707	            stream('0 g\n0 G\n%(undash)s')
   708	            y = height - fontSize - _2bw
   709	            stream('BT\n/%(iFontName)s %(fontSize)s Tf\n%(textFill)s')
   710	            for line in value.split('\n'):
   711	                stream('%%(_4bw)s %s Td\n(%s) Tj' % (y,escPDF(line)))
   712	                y -= leading
   713	            stream('ET')
   714	        leading = fp_str(leading)
   715	        stream('Q\nEMC\n')
   716	        stream = ('\n'.join(stream.__self__) % vars()).replace('  ',' ').replace('\n\n','\n')
   717	        return self.makeStream(
   718	                width, height, stream,
   719	                Resources = PDFFromString('<< /ProcSet [/PDF /Text] /Font %(rFontName)s >>' % vars()),
   720	                )
