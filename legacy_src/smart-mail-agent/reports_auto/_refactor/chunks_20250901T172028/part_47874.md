   452	    def check_first_pass(self) -> None:
   453	        """Type check the entire file, but defer functions with unresolved references.
   454
   455	        Unresolved references are forward references to variables
   456	        whose types haven't been inferred yet.  They may occur later
   457	        in the same file or in a different file that's being processed
   458	        later (usually due to an import cycle).
   459
   460	        Deferred functions will be processed by check_second_pass().
   461	        """
   462	        self.recurse_into_functions = True
   463	        with state.strict_optional_set(self.options.strict_optional), checker_state.set(self):
   464	            self.errors.set_file(
   465	                self.path, self.tree.fullname, scope=self.tscope, options=self.options
   466	            )
   467	            with self.tscope.module_scope(self.tree.fullname):
   468	                with self.enter_partial_types(), self.binder.top_frame_context():
   469	                    for d in self.tree.defs:
   470	                        if self.binder.is_unreachable():
   471	                            if not self.should_report_unreachable_issues():
   472	                                break
   473	                            if not self.is_noop_for_reachability(d):
   474	                                self.msg.unreachable_statement(d)
   475	                                break
   476	                        else:
   477	                            self.accept(d)
   478
   479	                assert not self.current_node_deferred
   480
   481	                all_ = self.globals.get("__all__")
   482	                if all_ is not None and all_.type is not None:
   483	                    all_node = all_.node
   484	                    assert all_node is not None
   485	                    seq_str = self.named_generic_type(
   486	                        "typing.Sequence", [self.named_type("builtins.str")]
   487	                    )
   488	                    if not is_subtype(all_.type, seq_str):
   489	                        str_seq_s, all_s = format_type_distinctly(
   490	                            seq_str, all_.type, options=self.options
   491	                        )
   492	                        self.fail(
   493	                            message_registry.ALL_MUST_BE_SEQ_STR.format(str_seq_s, all_s), all_node
   494	                        )

   496	    def check_second_pass(
   497	        self, todo: Sequence[DeferredNode | FineGrainedDeferredNode] | None = None
   498	    ) -> bool:
   499	        """Run second or following pass of type checking.
   500
   501	        This goes through deferred nodes, returning True if there were any.
   502	        """
   503	        self.recurse_into_functions = True
   504	        with state.strict_optional_set(self.options.strict_optional), checker_state.set(self):
   505	            if not todo and not self.deferred_nodes:
   506	                return False
   507	            self.errors.set_file(
   508	                self.path, self.tree.fullname, scope=self.tscope, options=self.options
   509	            )
   510	            with self.tscope.module_scope(self.tree.fullname):
   511	                self.pass_num += 1
   512	                if not todo:
   513	                    todo = self.deferred_nodes
   514	                else:
   515	                    assert not self.deferred_nodes
   516	                self.deferred_nodes = []
   517	                done: set[DeferredNodeType | FineGrainedDeferredNodeType] = set()
   518	                for node, active_typeinfo in todo:
   519	                    if node in done:
   520	                        continue
   521	                    # This is useful for debugging:
   522	                    # print("XXX in pass %d, class %s, function %s" %
   523	                    #       (self.pass_num, type_name, node.fullname or node.name))
   524	                    done.add(node)
   525	                    with ExitStack() as stack:
   526	                        if active_typeinfo:
   527	                            stack.enter_context(self.tscope.class_scope(active_typeinfo))
   528	                            stack.enter_context(self.scope.push_class(active_typeinfo))
   529	                        self.check_partial(node)
   530	            return True
