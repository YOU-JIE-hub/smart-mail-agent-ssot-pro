  1197	    are considered non-overlapping despite empty list belongs to both. However, List[int]

  1207
  1208	    if isinstance(left, TypedDictType):
  1209	        assert isinstance(right, Instance)
  1210	        typed, other = left, right
  1211	    else:
  1212	        assert isinstance(left, Instance)
  1213	        assert isinstance(right, TypedDictType)
  1214	        typed, other = right, left

  1216	    mutable_mapping = next(

  1218	    )
  1219	    if mutable_mapping is not None and typed.readonly_keys:
  1220	        return False

  1222	    mapping = next(base for base in other.type.mro if base.fullname == "typing.Mapping")
  1223	    other = map_instance_to_supertype(other, mapping)
  1224	    key_type, value_type = get_proper_types(other.args)

  1228	    str_type = fallback.type.bases[0].args[0]  # typing._TypedDict inherits Mapping[str, object]
  1229
  1230	    # Special case: a TypedDict with no required keys overlaps with an empty dict.
  1231	    if isinstance(key_type, UninhabitedType) and isinstance(value_type, UninhabitedType):
  1232	        return not typed.required_keys

  1234	    if typed.required_keys:
