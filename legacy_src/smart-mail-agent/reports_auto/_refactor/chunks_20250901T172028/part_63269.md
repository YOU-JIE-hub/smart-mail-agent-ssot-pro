  6421	def arg_approximate_similarity(actual: Type, formal: Type) -> bool:
  6422	    """Return if caller argument (actual) is roughly compatible with signature arg (formal).
  6423
  6424	    This function is deliberately loose and will report two types are similar
  6425	    as long as their "shapes" are plausibly the same.
  6426
  6427	    This is useful when we're doing error reporting: for example, if we're trying
  6428	    to select an overload alternative and there's no exact match, we can use
  6429	    this function to help us identify which alternative the user might have
  6430	    *meant* to match.
  6431	    """
  6432	    actual = get_proper_type(actual)
  6433	    formal = get_proper_type(formal)
  6434
  6435	    # Erase typevars: we'll consider them all to have the same "shape".
  6436	    if isinstance(actual, TypeVarType):
  6437	        actual = erase_to_union_or_bound(actual)
  6438	    if isinstance(formal, TypeVarType):
  6439	        formal = erase_to_union_or_bound(formal)
  6440
  6441	    # Callable or Type[...]-ish types
  6442	    def is_typetype_like(typ: ProperType) -> bool:
  6443	        return (
  6444	            isinstance(typ, TypeType)
  6445	            or (isinstance(typ, FunctionLike) and typ.is_type_obj())
  6446	            or (isinstance(typ, Instance) and typ.type.fullname == "builtins.type")
  6447	        )
  6448
  6449	    if isinstance(formal, CallableType):
  6450	        if isinstance(actual, (CallableType, Overloaded, TypeType)):
  6451	            return True
  6452	    if is_typetype_like(actual) and is_typetype_like(formal):
  6453	        return True
  6454
  6455	    # Unions
  6456	    if isinstance(actual, UnionType):
  6457	        return any(arg_approximate_similarity(item, formal) for item in actual.relevant_items())
  6458	    if isinstance(formal, UnionType):
  6459	        return any(arg_approximate_similarity(actual, item) for item in formal.relevant_items())
  6460
  6461	    # TypedDicts
  6462	    if isinstance(actual, TypedDictType):
  6463	        if isinstance(formal, TypedDictType):
  6464	            return True
  6465	        return arg_approximate_similarity(actual.fallback, formal)
  6466
  6467	    # Instances
  6468	    # For instances, we mostly defer to the existing is_subtype check.
  6469	    if isinstance(formal, Instance):
  6470	        if isinstance(actual, CallableType):
  6471	            actual = actual.fallback
  6472	        if isinstance(actual, Overloaded):
  6473	            actual = actual.items[0].fallback
  6474	        if isinstance(actual, TupleType):
  6475	            actual = tuple_fallback(actual)
  6476	        if isinstance(actual, Instance) and formal.type in actual.type.mro:
  6477	            # Try performing a quick check as an optimization
  6478	            return True
  6479
  6480	    # Fall back to a standard subtype check for the remaining kinds of type.
  6481	    return is_subtype(erasetype.erase_type(actual), erasetype.erase_type(formal))

  6484	def any_causes_overload_ambiguity(
  6485	    items: list[CallableType],
  6486	    return_types: list[Type],
  6487	    arg_types: list[Type],
  6488	    arg_kinds: list[ArgKind],
  6489	    arg_names: Sequence[str | None] | None,
  6490	) -> bool:
  6491	    """May an argument containing 'Any' cause ambiguous result type on call to overloaded function?
  6492
  6493	    Note that this sometimes returns True even if there is no ambiguity, since a correct
  6494	    implementation would be complex (and the call would be imprecisely typed due to Any
  6495	    types anyway).
  6496
  6497	    Args:
  6498	        items: Overload items matching the actual arguments
  6499	        arg_types: Actual argument types
  6500	        arg_kinds: Actual argument kinds
  6501	        arg_names: Actual argument names
  6502	    """
  6503	    if all_same_types(return_types):
  6504	        return False
  6505
  6506	    actual_to_formal = [
  6507	        map_formals_to_actuals(
  6508	            arg_kinds, arg_names, item.arg_kinds, item.arg_names, lambda i: arg_types[i]
  6509	        )
  6510	        for item in items
  6511	    ]
  6512
  6513	    for arg_idx, arg_type in enumerate(arg_types):
  6514	        # We ignore Anys in type object callables as ambiguity
  6515	        # creators, since that can lead to falsely claiming ambiguity
  6516	        # for overloads between Type and Callable.
  6517	        if has_any_type(arg_type, ignore_in_type_obj=True):
  6518	            matching_formals_unfiltered = [
  6519	                (item_idx, lookup[arg_idx])
  6520	                for item_idx, lookup in enumerate(actual_to_formal)
  6521	                if lookup[arg_idx]
  6522	            ]
  6523
  6524	            matching_returns = []
  6525	            matching_formals = []
  6526	            for item_idx, formals in matching_formals_unfiltered:
  6527	                matched_callable = items[item_idx]
  6528	                matching_returns.append(matched_callable.ret_type)
  6529
  6530	                # Note: if an actual maps to multiple formals of differing types within
  6531	                # a single callable, then we know at least one of those formals must be
  6532	                # a different type then the formal(s) in some other callable.
  6533	                # So it's safe to just append everything to the same list.
  6534	                for formal in formals:
  6535	                    matching_formals.append(matched_callable.arg_types[formal])
  6536	            if not all_same_types(matching_formals) and not all_same_types(matching_returns):
  6537	                # Any maps to multiple different types, and the return types of these items differ.
  6538	                return True
  6539	    return False

  6572	        if target.is_generic():
