   560	class SingledispatchVisitor(TraverserVisitor):
   561	    current_path: str
   562
   563	    def __init__(self, errors: Errors) -> None:
   564	        super().__init__()
   565
   566	        # Map of main singledispatch function to list of registered implementations
   567	        self.singledispatch_impls: defaultdict[FuncDef, list[RegisterImplInfo]] = defaultdict(list)
   568
   569	        # Map of decorated function to the indices of any decorators to remove
   570	        self.decorators_to_remove: dict[FuncDef, list[int]] = {}
   571
   572	        self.errors: Errors = errors
   573
   574	    def visit_decorator(self, dec: Decorator) -> None:
   575	        if dec.decorators:
   576	            decorators_to_store = dec.decorators.copy()
   577	            decorators_to_remove: list[int] = []
   578	            # the index of the last non-register decorator before finding a register decorator
   579	            # when going through decorators from top to bottom
   580	            last_non_register: int | None = None
   581	            for i, d in enumerate(decorators_to_store):
   582	                impl = get_singledispatch_register_call_info(d, dec.func)
   583	                if impl is not None:
   584	                    self.singledispatch_impls[impl.singledispatch_func].append(
   585	                        (impl.dispatch_type, dec.func)
   586	                    )
   587	                    decorators_to_remove.append(i)
   588	                    if last_non_register is not None:
   589	                        # found a register decorator after a non-register decorator, which we
   590	                        # don't support because we'd have to make a copy of the function before
   591	                        # calling the decorator so that we can call it later, which complicates
   592	                        # the implementation for something that is probably not commonly used
   593	                        self.errors.error(
   594	                            "Calling decorator after registering function not supported",
   595	                            self.current_path,
   596	                            decorators_to_store[last_non_register].line,
   597	                        )
   598	                else:
   599	                    if refers_to_fullname(d, "functools.singledispatch"):
   600	                        decorators_to_remove.append(i)
   601	                        # make sure that we still treat the function as a singledispatch function
   602	                        # even if we don't find any registered implementations (which might happen
   603	                        # if all registered implementations are registered dynamically)
   604	                        self.singledispatch_impls.setdefault(dec.func, [])
   605	                    last_non_register = i
   606
   607	            if decorators_to_remove:
   608	                # calling register on a function that tries to dispatch based on type annotations
   609	                # raises a TypeError because compiled functions don't have an __annotations__
   610	                # attribute
   611	                self.decorators_to_remove[dec.func] = decorators_to_remove
   612
   613	        super().visit_decorator(dec)

   621	def get_singledispatch_register_call_info(
   622	    decorator: Expression, func: FuncDef
   623	) -> RegisteredImpl | None:
   624	    # @fun.register(complex)
   625	    # def g(arg): ...
   626	    if (
   627	        isinstance(decorator, CallExpr)
   628	        and len(decorator.args) == 1
   629	        and isinstance(decorator.args[0], RefExpr)
   630	    ):
   631	        callee = decorator.callee
   632	        dispatch_type = decorator.args[0].node
   633	        if not isinstance(dispatch_type, TypeInfo):
   634	            return None
   635
   636	        if isinstance(callee, MemberExpr):
   637	            return registered_impl_from_possible_register_call(callee, dispatch_type)
   638	    # @fun.register
   639	    # def g(arg: int): ...
   640	    elif isinstance(decorator, MemberExpr):
   641	        # we don't know if this is a register call yet, so we can't be sure that the function
   642	        # actually has arguments
   643	        if not func.arguments:
   644	            return None
   645	        arg_type = get_proper_type(func.arguments[0].variable.type)
   646	        if not isinstance(arg_type, Instance):
   647	            return None
   648	        info = arg_type.type
   649	        return registered_impl_from_possible_register_call(decorator, info)
   650	    return None

   656	    if expr.name == "register" and isinstance(expr.expr, NameExpr):

   116	def _apply_specialization(
   117	    builder: IRBuilder, expr: CallExpr, callee: RefExpr, name: str | None, typ: RType | None = None
   118	) -> Value | None:
   119	    # TODO: Allow special cases to have default args or named args. Currently they don't since

   122	    # If there is a specializer for this function, try calling it.

   125	        for specializer in specializers[name, typ]:
