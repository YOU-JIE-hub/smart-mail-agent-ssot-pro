def send_email_with_attachment(*, recipient: str, subject: str, body_html: str, attachment_path: str) -> bool:
    p = pathlib.Path(attachment_path)
    if not p.exists():
        raise FileNotFoundError(str(p))

    msg["To"] = recipient
    msg["Subject"] = subject

    with smtplib.SMTP_SSL(host, port) as s:
