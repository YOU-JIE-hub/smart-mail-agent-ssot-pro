    94	def run_setup(script_name: str, script_args: list[str]) -> bool:
    95	    """Run a setup script in a somewhat controlled environment.
    96
    97	    This is adapted from code in distutils and our goal here is that is
    98	    faster to not need to spin up a python interpreter to run it.
    99
   100	    We had to fork it because the real run_setup swallows errors
   101	    and KeyboardInterrupt with no way to recover them (!).
   102	    The real version has some extra features that we removed since
   103	    we weren't using them.
   104
   105	    Returns whether the setup succeeded.
   106	    """
   107	    save_argv = sys.argv.copy()
   108	    g = {"__file__": script_name}
   109	    try:
   110	        try:
   111	            sys.argv[0] = script_name
   112	            sys.argv[1:] = script_args
   113	            with open(script_name, "rb") as f:
   114	                exec(f.read(), g)
   115	        finally:
   116	            sys.argv = save_argv
   117	    except SystemExit as e:
   118	        # distutils converts KeyboardInterrupt into a SystemExit with
   119	        # "interrupted" as the argument. Convert it back so that
   120	        # pytest will exit instead of just failing the test.
   121	        if e.code == "interrupted":
   122	            raise KeyboardInterrupt from e
   123
   124	        return e.code == 0 or e.code is None
   125
   126	    return True
