  1333	    def resetTransforms(self):
  1334	        """I want to draw something (eg, string underlines) w.r.t. the default user space.
  1335	           Reset the matrix! This should be used usually as follows::
  1336
  1337	              canv.saveState()
  1338	              canv.resetTransforms()
  1339	              #...draw some stuff in default space coords...
  1340	              canv.restoreState() # go back!
  1341	        """
  1342	        # we have to adjoin the inverse, since reset is not a basic operation (without save/restore)
  1343	        (selfa, selfb, selfc, selfd, selfe, selff) = self._currentMatrix
  1344	        det = selfa*selfd - selfc*selfb
  1345	        resulta = selfd/det
  1346	        resultc = -selfc/det
  1347	        resulte = (selfc*selff - selfd*selfe)/det
  1348	        resultd = selfa/det
  1349	        resultb = -selfb/det
  1350	        resultf = (selfe*selfb - selff*selfa)/det
  1351	        self.transform(resulta, resultb, resultc, resultd, resulte, resultf)

  1353	    def transform(self, a,b,c,d,e,f):
  1354	        """adjoin a mathematical transform to the current graphics state matrix.
  1355	           Not recommended for beginners."""
  1356	        #How can Python track this?
  1357	        if ENABLE_TRACKING:
  1358	            a0,b0,c0,d0,e0,f0 = self._currentMatrix
  1359	            self._currentMatrix = (a0*a+c0*b,    b0*a+d0*b,
  1360	                                   a0*c+c0*d,    b0*c+d0*d,
  1361	                                   a0*e+c0*f+e0, b0*e+d0*f+f0)
  1362	        if self._code and self._code[-1][-3:]==' cm':
  1363	            L = self._code[-1].split()
  1364	            a0, b0, c0, d0, e0, f0 = list(map(float,L[-7:-1]))
  1365	            s = len(L)>7 and join(L)+ ' %s cm' or '%s cm'
  1366	            self._code[-1] = s % fp_str(a0*a+c0*b,b0*a+d0*b,a0*c+c0*d,b0*c+d0*d,a0*e+c0*f+e0,b0*e+d0*f+f0)
  1367	        else:
  1368	            self._code.append('%s cm' % fp_str(a,b,c,d,e,f))

  1641	    def drawAlignedString(self, x, y, text, pivotChar=rl_config.decimalSymbol, mode=None, charSpace=0, direction=None, wordSpace=None):
  1642	        """Draws a string aligned on the first '.' (or other pivot character).
  1643
  1644	        The centre position of the pivot character will be used as x.
  1645	        So, you could draw a straight line down through all the decimals in a
  1646	        column of numbers, and anything without a decimal should be
  1647	        optically aligned with those that have.
  1648
  1649	        There is one special rule to help with accounting formatting.  Here's
  1650	        how normal numbers should be aligned on the 'dot'. Look at the
  1651	        LAST two::
  1652
  1653	           12,345,67
  1654	              987.15
  1655	               42
  1656	           -1,234.56
  1657	             (456.78)
  1658	             (456)
  1659	               27 inches
  1660	               13cm
  1661
  1662	        Since the last three do not contain a dot, a crude dot-finding
  1663	        rule would place them wrong. So we test for the special case
  1664	        where no pivot is found, digits are present, but the last character
  1665	        is not a digit.  We then work back from the end of the string
  1666	        This case is a tad slower but hopefully rare.
  1667
  1668	        """
  1669	        parts = text.split(pivotChar,1)
  1670	        pivW = self.stringWidth(pivotChar, self._fontname, self._fontsize)
  1671
  1672	        if len(parts) == 1 and digitPat.search(text) is not None and text[-1] not in digits:
  1673	            #we have no decimal but it ends in a bracket, or 'in' or something.
  1674	            #the cut should be after the last digit.
  1675	            leftText = parts[0][0:-1]
  1676	            rightText = parts[0][-1]
  1677	            #any more?
  1678	            while leftText[-1] not in digits:
  1679	                rightText = leftText[-1] + rightText
  1680	                leftText = leftText[0:-1]
  1681
  1682	            self.drawRightString(x-0.5*pivW, y, leftText, mode=mode, charSpace=charSpace,
  1683	                    direction=direction, wordSpace=wordSpace)
  1684	            self.drawString(x-0.5*pivW, y, rightText, mode=mode, charSpace=charSpace,
  1685	                    direction=direction, wordSpace=wordSpace)
  1686
  1687	        else:
  1688	            #normal
