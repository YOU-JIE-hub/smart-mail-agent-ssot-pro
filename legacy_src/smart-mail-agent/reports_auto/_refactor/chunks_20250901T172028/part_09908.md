   554	    def shrinkWrap(self, line):
   555	        # for non justified text, collapse adjacent text/shift's into single operations
   556	        result = []
   557	        index = 0
   558	        maxindex = len(line)
   559	        while index<maxindex:
   560	            e = line[index]
   561	            if isinstance(e,str) and index<maxindex-1:
   562	                # collect strings and floats
   563	                thestrings = [e]
   564	                thefloats = 0.0
   565	                index += 1
   566	                nexte = line[index]
   567	                while index<maxindex and isinstance(nexte,(float,str)):
   568	                    # switch to expandable space if appropriate
   569	                    if isinstance(nexte,float):
   570	                        if thefloats<0 and nexte>0:
   571	                            thefloats = -thefloats
   572	                        if nexte<0 and thefloats>0:
   573	                            nexte = -nexte
   574	                        thefloats += nexte
   575	                    elif isinstance(nexte,str):
   576	                        thestrings.append(nexte)
   577	                    index += 1
   578	                    if index<maxindex:
   579	                        nexte = line[index]
   580	                # wrap up the result
   581	                s = ' '.join(thestrings)
   582	                result.append(s)
   583	                result.append(float(thefloats))
   584	                # back up for unhandled element
   585	                index -= 1
   586	            else:
   587	                result.append(e)
   588	            index += 1
   589
   590	        return result
