]:
   376	        """
   377	        Contracts a list of types in a sequence pattern depending on the position of a starred
   378	        capture pattern.
   379
   380	        For example if the sequence pattern [a, *b, c] is matched against types [bool, int, str,
   381	        bytes] the contracted types are [bool, Union[int, str], bytes].
   382
   383	        If star_pos in None the types are returned unchanged.
   384	        """
   385	        unpack_index = find_unpack_in_list(types)
   386	        if unpack_index is not None:
   387	            # Variadic tuples require "re-shaping" to match the requested pattern.
   388	            unpack = types[unpack_index]
   389	            assert isinstance(unpack, UnpackType)
   390	            unpacked = get_proper_type(unpack.type)
   391	            # This should be guaranteed by the normalization in the caller.
   392	            assert isinstance(unpacked, Instance) and unpacked.type.fullname == "builtins.tuple"
   393	            if star_pos is None:
   394	                missing = num_patterns - len(types) + 1
   395	                new_types = types[:unpack_index]
   396	                new_types += [unpacked.args[0]] * missing
   397	                new_types += types[unpack_index + 1 :]
   398	                return new_types
   399	            prefix, middle, suffix = split_with_prefix_and_suffix(
   400	                tuple([UnpackType(unpacked) if isinstance(t, UnpackType) else t for t in types]),
   401	                star_pos,
   402	                num_patterns - star_pos,
   403	            )
   404	            new_middle = []
   405	            for m in middle:
   406	                # The existing code expects the star item type, rather than the type of
   407	                # the whole tuple "slice".
   408	                if isinstance(m, UnpackType):
   409	                    new_middle.append(unpacked.args[0])
   410	                else:
   411	                    new_middle.append(m)
   412	            return list(prefix) + [make_simplified_union(new_middle)] + list(suffix)
   413	        else:
   414	            if star_pos is None:
   415	                return types
   416	            new_types = types[:star_pos]
   417	            star_length = len(types) - num_patterns
   418	            new_types.append(make_simplified_union(types[star_pos : star_pos + star_length]))
   419	            new_types += types[star_pos + star_length :]
   420	            return new_types
   421
   422	    def expand_starred_pattern_types(
   423	        self, types: list[Type], star_pos: int | None, num_types: int, original_unpack: bool
   424	    ) -> list[Type]:
   425	        """Undoes the contraction done by contract_starred_pattern_types.
   426
   427	        For example if the sequence pattern is [a, *b, c] and types [bool, int, str] are extended
   428	        to length 4 the result is [bool, int, int, str].
   429	        """
   430	        if star_pos is None:
   431	            return types
   432	        if original_unpack:
   433	            # In the case where original tuple type has an unpack item, it is not practical
   434	            # to coerce pattern type back to the original shape (and may not even be possible),
   435	            # so we only restore the type of the star item.
   436	            res = []
   437	            for i, t in enumerate(types):
   438	                if i != star_pos:
   439	                    res.append(t)
   440	                else:
   441	                    res.append(UnpackType(self.chk.named_generic_type("builtins.tuple", [t])))
   442	            return res
   443	        new_types = types[:star_pos]
   444	        star_length = num_types - len(types) + 1
   445	        new_types += [types[star_pos]] * star_length
   446	        new_types += types[star_pos + 1 :]
   447
   448	        return new_types

   450	    def narrow_sequence_child(self, outer_type: Type, inner_type: Type, ctx: Context) -> Type:
   451	        new_type = self.construct_sequence_child(outer_type, inner_type)
   452	        if is_subtype(new_type, outer_type):
   453	            new_type, _ = self.chk.conditional_types_with_intersection(
   454	                outer_type, [get_type_range(new_type)], ctx, default=outer_type
   455	            )
   456	        else:
   457	            new_type = outer_type
   458	        return new_type

   460	    def visit_starred_pattern(self, o: StarredPattern) -> PatternType:
   461	        captures: dict[Expression, Type] = {}
   462	        if o.capture is not None:

   464	            captures[o.capture] = list_type
   465	        return PatternType(self.type_context[-1], UninhabitedType(), captures)

   467	    def visit_mapping_pattern(self, o: MappingPattern) -> PatternType:
   468	        current_type = get_proper_type(self.type_context[-1])
   469	        can_match = True
   470	        captures: dict[Expression, Type] = {}
   471	        for key, value in zip(o.keys, o.values):
   472	            inner_type = self.get_mapping_item_type(o, current_type, key)
   473	            if inner_type is None:
   474	                can_match = False
   475	                inner_type = self.chk.named_type("builtins.object")
   476	            pattern_type = self.accept(value, inner_type)
   477	            if is_uninhabited(pattern_type.type):
   478	                can_match = False
   479	            else:
   480	                self.update_type_map(captures, pattern_type.captures)
   481
   482	        if o.rest is not None:
   483	            mapping = self.chk.named_type("typing.Mapping")
   484	            if is_subtype(current_type, mapping) and isinstance(current_type, Instance):
   485	                mapping_inst = map_instance_to_supertype(current_type, mapping.type)
   486	                dict_typeinfo = self.chk.lookup_typeinfo("builtins.dict")
   487	                rest_type = Instance(dict_typeinfo, mapping_inst.args)
   488	            else:
   489	                object_type = self.chk.named_type("builtins.object")
   490	                rest_type = self.chk.named_generic_type(
   491	                    "builtins.dict", [object_type, object_type]
   492	                )
   493
   494	            captures[o.rest] = rest_type
   495
   496	        if can_match:
   497	            # We can't narrow the type here, as Mapping key is invariant.
   498	            new_type = self.type_context[-1]
   499	        else:
   500	            new_type = UninhabitedType()
   501	        return PatternType(new_type, current_type, captures)

   503	    def get_mapping_item_type(
   504	        self, pattern: MappingPattern, mapping_type: Type, key: Expression
   505	    ) -> Type | None:
   506	        mapping_type = get_proper_type(mapping_type)
   507	        if isinstance(mapping_type, TypedDictType):
   508	            with self.msg.filter_errors() as local_errors:
   509	                result: Type | None = self.chk.expr_checker.visit_typeddict_index_expr(
   510	                    mapping_type, key
   511	                )[0]
   512	                has_local_errors = local_errors.has_new_errors()
   513	            # If we can't determine the type statically fall back to treating it as a normal
   514	            # mapping
   515	            if has_local_errors:
   516	                with self.msg.filter_errors() as local_errors:
   517	                    result = self.get_simple_mapping_item_type(pattern, mapping_type, key)
   518
   519	                    if local_errors.has_new_errors():
   520	                        result = None
   521	        else:
   522	            with self.msg.filter_errors():
   523	                result = self.get_simple_mapping_item_type(pattern, mapping_type, key)
   524	        return result
   525
   526	    def get_simple_mapping_item_type(
   527	        self, pattern: MappingPattern, mapping_type: Type, key: Expression
   528	    ) -> Type:
   529	        result, _ = self.chk.expr_checker.check_method_call_by_name(
   530	            "__getitem__", mapping_type, [key], [ARG_POS], pattern
   531	        )
   532	        return result

   534	    def visit_class_pattern(self, o: ClassPattern) -> PatternType:
   535	        current_type = get_proper_type(self.type_context[-1])
   536
   537	        #
   538	        # Check class type
   539	        #
   540	        type_info = o.class_ref.node
   541	        if type_info is None:
   542	            return PatternType(AnyType(TypeOfAny.from_error), AnyType(TypeOfAny.from_error), {})
   543	        if isinstance(type_info, TypeAlias) and not type_info.no_args:
   544	            self.msg.fail(message_registry.CLASS_PATTERN_GENERIC_TYPE_ALIAS, o)
   545	            return self.early_non_match()
   546	        if isinstance(type_info, TypeInfo):
   547	            typ: Type = fill_typevars_with_any(type_info)
   548	        elif isinstance(type_info, TypeAlias):
   549	            typ = type_info.target
   550	        elif (
   551	            isinstance(type_info, Var)
   552	            and type_info.type is not None
   553	            and isinstance(get_proper_type(type_info.type), AnyType)
   554	        ):
   555	            typ = type_info.type
   556	        else:
   557	            if isinstance(type_info, Var) and type_info.type is not None:
   558	                name = type_info.type.str_with_options(self.options)
   559	            else:
   560	                name = type_info.name
   561	            self.msg.fail(message_registry.CLASS_PATTERN_TYPE_REQUIRED.format(name), o)
   562	            return self.early_non_match()
   563
   564	        new_type, rest_type = self.chk.conditional_types_with_intersection(
   565	            current_type, [get_type_range(typ)], o, default=current_type
   566	        )
   567	        if is_uninhabited(new_type):
   568	            return self.early_non_match()
   569	        # TODO: Do I need this?
   570	        narrowed_type = narrow_declared_type(current_type, new_type)
   571
   572	        #
   573	        # Convert positional to keyword patterns
   574	        #
   575	        keyword_pairs: list[tuple[str | None, Pattern]] = []
   576	        match_arg_set: set[str] = set()
   577
   578	        captures: dict[Expression, Type] = {}
   579
   580	        if len(o.positionals) != 0:
   581	            if self.should_self_match(typ):
   582	                if len(o.positionals) > 1:
   583	                    self.msg.fail(message_registry.CLASS_PATTERN_TOO_MANY_POSITIONAL_ARGS, o)
   584	                pattern_type = self.accept(o.positionals[0], narrowed_type)
   585	                if not is_uninhabited(pattern_type.type):
   586	                    return PatternType(
   587	                        pattern_type.type,
   588	                        join_types(rest_type, pattern_type.rest_type),
   589	                        pattern_type.captures,
   590	                    )
   591	                captures = pattern_type.captures
   592	            else:
   593	                with self.msg.filter_errors() as local_errors:
   594	                    match_args_type = analyze_member_access(
   595	                        "__match_args__",
   596	                        typ,
   597	                        o,
   598	                        is_lvalue=False,
   599	                        is_super=False,
   600	                        is_operator=False,
   601	                        original_type=typ,
   602	                        chk=self.chk,
   603	                    )
   604	                    has_local_errors = local_errors.has_new_errors()
   605	                if has_local_errors:
   606	                    self.msg.fail(
   607	                        message_registry.MISSING_MATCH_ARGS.format(
   608	                            typ.str_with_options(self.options)
   609	                        ),
   610	                        o,
   611	                    )
   612	                    return self.early_non_match()
   613
   614	                proper_match_args_type = get_proper_type(match_args_type)
   615	                if isinstance(proper_match_args_type, TupleType):
   616	                    match_arg_names = get_match_arg_names(proper_match_args_type)
   617
   618	                    if len(o.positionals) > len(match_arg_names):
   619	                        self.msg.fail(message_registry.CLASS_PATTERN_TOO_MANY_POSITIONAL_ARGS, o)
   620	                        return self.early_non_match()
   621	                else:
   622	                    match_arg_names = [None] * len(o.positionals)
   623
   624	                for arg_name, pos in zip(match_arg_names, o.positionals):
   625	                    keyword_pairs.append((arg_name, pos))
   626	                    if arg_name is not None:
   627	                        match_arg_set.add(arg_name)
   628
   629	        #
   630	        # Check for duplicate patterns
   631	        #
   632	        keyword_arg_set = set()
   633	        has_duplicates = False
   634	        for key, value in zip(o.keyword_keys, o.keyword_values):
   635	            keyword_pairs.append((key, value))
   636	            if key in match_arg_set:
   637	                self.msg.fail(
   638	                    message_registry.CLASS_PATTERN_KEYWORD_MATCHES_POSITIONAL.format(key), value
   639	                )
   640	                has_duplicates = True
   641	            elif key in keyword_arg_set:
   642	                self.msg.fail(
   643	                    message_registry.CLASS_PATTERN_DUPLICATE_KEYWORD_PATTERN.format(key), value
   644	                )
   645	                has_duplicates = True
   646	            keyword_arg_set.add(key)
   647
   648	        if has_duplicates:
   649	            return self.early_non_match()
   650
   651	        #
   652	        # Check keyword patterns
   653	        #
   654	        can_match = True
   655	        for keyword, pattern in keyword_pairs:
   656	            key_type: Type | None = None
   657	            with self.msg.filter_errors() as local_errors:
   658	                if keyword is not None:
   659	                    key_type = analyze_member_access(
   660	                        keyword,
   661	                        narrowed_type,
   662	                        pattern,
   663	                        is_lvalue=False,
   664	                        is_super=False,
   665	                        is_operator=False,
   666	                        original_type=new_type,
   667	                        chk=self.chk,
   668	                    )
   669	                else:
   670	                    key_type = AnyType(TypeOfAny.from_error)
   671	                has_local_errors = local_errors.has_new_errors()
   672	            if has_local_errors or key_type is None:
   673	                key_type = AnyType(TypeOfAny.from_error)
   674	                self.msg.fail(
   675	                    message_registry.CLASS_PATTERN_UNKNOWN_KEYWORD.format(
   676	                        typ.str_with_options(self.options), keyword
   677	                    ),
   678	                    pattern,
   679	                )
   680
   681	            inner_type, inner_rest_type, inner_captures = self.accept(pattern, key_type)
   682	            if is_uninhabited(inner_type):
   683	                can_match = False
   684	            else:
   685	                self.update_type_map(captures, inner_captures)
   686	                if not is_uninhabited(inner_rest_type):
   687	                    rest_type = current_type
   688
   689	        if not can_match:
   690	            new_type = UninhabitedType()
   691	        return PatternType(new_type, rest_type, captures)

   693	    def should_self_match(self, typ: Type) -> bool:
   694	        typ = get_proper_type(typ)
   695	        if isinstance(typ, TupleType):
   696	            typ = typ.partial_fallback
   697	        if isinstance(typ, Instance) and typ.type.get("__match_args__") is not None:
   698	            # Named tuples and other subtypes of builtins that define __match_args__
   699	            # should not self match.
   700	            return False
   701	        for other in self.self_match_types:
   702	            if is_subtype(typ, other):
   703	                return True
   704	        return False
   705
   706	    def can_match_sequence(self, typ: ProperType) -> bool:
   707	        if isinstance(typ, AnyType):
   708	            return True
   709	        if isinstance(typ, UnionType):
   710	            return any(self.can_match_sequence(get_proper_type(item)) for item in typ.items)
   711	        for other in self.non_sequence_match_types:
   712	            # We have to ignore promotions, as memoryview should match, but bytes,
   713	            # which it can be promoted to, shouldn't
   714	            if is_subtype(typ, other, ignore_promotions=True):
   715	                return False
   716	        sequence = self.chk.named_type("typing.Sequence")
   717	        # If the static type is more general than sequence the actual type could still match
   718	        return is_subtype(typ, sequence) or is_subtype(sequence, typ)

   720	    def generate_types_from_names(self, type_names: list[str]) -> list[Type]:
   721	        types: list[Type] = []
   722	        for name in type_names:
   723	            try:
   724	                types.append(self.chk.named_type(name))
   725	            except KeyError as e:
   726	                # Some built in types are not defined in all test cases
   727	                if not name.startswith("builtins."):
   728	                    raise e
   729	        return types

   731	    def update_type_map(
   732	        self, original_type_map: dict[Expression, Type], extra_type_map: dict[Expression, Type]
   733	    ) -> None:
   734	        # Calculating this would not be needed if TypeMap directly used literal hashes instead of
   735	        # expressions, as suggested in the TODO above it's definition
   736	        already_captured = {literal_hash(expr) for expr in original_type_map}
   737	        for expr, typ in extra_type_map.items():
   738	            if literal_hash(expr) in already_captured:
   739	                node = get_var(expr)
   740	                self.msg.fail(
   741	                    message_registry.MULTIPLE_ASSIGNMENTS_IN_PATTERN.format(node.name), expr
   742	                )
   743	            else:
   744	                original_type_map[expr] = typ

   746	    def construct_sequence_child(self, outer_type: Type, inner_type: Type) -> Type:
   747	        """
   748	        If outer_type is a child class of typing.Sequence returns a new instance of
   749	        outer_type, that is a Sequence of inner_type. If outer_type is not a child class of
   750	        typing.Sequence just returns a Sequence of inner_type
   751
   752	        For example:
   753	        construct_sequence_child(List[int], str) = List[str]
   754
   755	        TODO: this doesn't make sense. For example if one has class S(Sequence[int], Generic[T])
   756	        or class T(Sequence[Tuple[T, T]]), there is no way any of those can map to Sequence[str].
   757	        """
   758	        proper_type = get_proper_type(outer_type)
   759	        if isinstance(proper_type, AnyType):
   760	            return outer_type
   761	        if isinstance(proper_type, UnionType):
   762	            types = [
   763	                self.construct_sequence_child(item, inner_type)
   764	                for item in proper_type.items
   765	                if self.can_match_sequence(get_proper_type(item))
   766	            ]
   767	            return make_simplified_union(types)
   768	        sequence = self.chk.named_generic_type("typing.Sequence", [inner_type])
   769	        if is_subtype(outer_type, self.chk.named_type("typing.Sequence")):
   770	            if isinstance(proper_type, TupleType):
   771	                proper_type = tuple_fallback(proper_type)
   772	            assert isinstance(proper_type, Instance)
   773	            empty_type = fill_typevars(proper_type.type)
   774	            partial_type = expand_type_by_instance(empty_type, sequence)
   775	            return expand_type_by_instance(partial_type, proper_type)
