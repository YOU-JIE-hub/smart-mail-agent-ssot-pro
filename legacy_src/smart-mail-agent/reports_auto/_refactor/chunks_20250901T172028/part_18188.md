 = False
   366	        for fr, analysis in get_analysis_to_report(self.coverage, morfs):
   367	            have_data = True
   368	            ftr = FileToReport(fr, analysis)
   369	            if self.should_report(analysis, self.index_pages["file"]):
   370	                files_to_report.append(ftr)
   371	            else:
   372	                file_be_gone(os.path.join(self.directory, ftr.html_filename))
   373
   374	        if not have_data:
   375	            raise NoDataError("No data to report.")
   376
   377	        self.make_directory()
   378	        self.make_local_static_report_files()
   379
   380	        if files_to_report:
   381	            for ftr1, ftr2 in zip(files_to_report[:-1], files_to_report[1:]):
   382	                ftr1.next_html = ftr2.html_filename
   383	                ftr2.prev_html = ftr1.html_filename
   384	            files_to_report[0].prev_html = "index.html"
   385	            files_to_report[-1].next_html = "index.html"
   386
   387	        for ftr in files_to_report:
   388	            self.write_html_page(ftr)
   389	            for noun, plural_noun in ftr.fr.code_region_kinds():
   390	                if noun not in self.index_pages:
   391	                    self.index_pages[noun] = self.new_index_page(noun, plural_noun)
   392
   393	        # Write the index page.
   394	        if files_to_report:
   395	            first_html = files_to_report[0].html_filename
   396	            final_html = files_to_report[-1].html_filename
   397	        else:
   398	            first_html = final_html = "index.html"
   399	        self.write_file_index_page(first_html, final_html)
   400
   401	        # Write function and class index pages.
   402	        self.write_region_index_pages(files_to_report)
   403
   404	        return (
   405	            self.index_pages["file"].totals.n_statements
   406	            and self.index_pages["file"].totals.pc_covered
   407	        )

   441	    def should_report(self, analysis: Analysis, index_page: IndexPage) -> bool:
   442	        """Determine if we'll report this file or region."""
   443	        # Get the numbers for this file.
   444	        nums = analysis.numbers
   445	        index_page.totals += nums
   446
   447	        if self.skip_covered:
   448	            # Don't report on 100% files.
   449	            no_missing_lines = (nums.n_missing == 0)
   450	            no_missing_branches = (nums.n_partial_branches == 0)
   451	            if no_missing_lines and no_missing_branches:
   452	                index_page.skipped_covered_count += 1
   453	                return False
   454
   455	        if self.skip_empty:
   456	            # Don't report on empty files.
   457	            if nums.n_statements == 0:
   458	                index_page.skipped_empty_count += 1
   459	                return False
   460
   461	        return True

   463	    def write_html_page(self, ftr: FileToReport) -> None:
   464	        """Generate an HTML page for one source file.
   465
   466	        If the page on disk is already correct based on our incremental status
   467	        checking, then the page doesn't have to be generated, and this function
   468	        only does page summary bookkeeping.
   469
   470	        """
   471	        # Find out if the page on disk is already correct.
   472	        if self.incr.can_skip_file(self.data, ftr.fr, ftr.rootname):
   473	            self.index_pages["file"].summaries.append(self.incr.index_info(ftr.rootname))
   474	            return
   475
   476	        # Write the HTML page for this source file.
   477	        file_data = self.datagen.data_for_file(ftr.fr, ftr.analysis)
   478
   479	        contexts = collections.Counter(c for cline in file_data.lines for c in cline.contexts)
   480	        context_codes = {y: i for (i, y) in enumerate(x[0] for x in contexts.most_common())}
   481	        if context_codes:
   482	            contexts_json = json.dumps(
   483	                {encode_int(v): k for (k, v) in context_codes.items()},
   484	                indent=2,
   485	            )
   486	        else:
   487	            contexts_json = None
   488
   489	        for ldata in file_data.lines:
   490	            # Build the HTML for the line.
   491	            html_parts = []
   492	            for tok_type, tok_text in ldata.tokens:
   493	                if tok_type == "ws":
   494	                    html_parts.append(escape(tok_text))
   495	                else:
   496	                    tok_html = escape(tok_text) or "&nbsp;"
   497	                    html_parts.append(f'<span class="{tok_type}">{tok_html}</span>')
   498	            ldata.html = "".join(html_parts)
   499	            if ldata.context_list:
   500	                encoded_contexts = [
   501	                    encode_int(context_codes[c_context]) for c_context in ldata.context_list
   502	                ]
   503	                code_width = max(len(ec) for ec in encoded_contexts)
   504	                ldata.context_str = (
   505	                    str(code_width)
   506	                    + "".join(ec.ljust(code_width) for ec in encoded_contexts)
   507	                )
   508	            else:
   509	                ldata.context_str = ""
   510
   511	            if ldata.short_annotations:
   512	                # 202F is NARROW NO-BREAK SPACE.
   513	                # 219B is RIGHTWARDS ARROW WITH STROKE.
   514	                ldata.annotate = ",&nbsp;&nbsp; ".join(
   515	                    f"{ldata.number}&#x202F;&#x219B;&#x202F;{d}"
   516	                    for d in ldata.short_annotations
   517	                )
   518	            else:
   519	                ldata.annotate = None
   520
   521	            if ldata.long_annotations:
   522	                longs = ldata.long_annotations
   523	                # A line can only have two branch destinations. If there were
   524	                # two missing, we would have written one as "always raised."
   525	                assert len(longs) == 1, (
   526	                    f"Had long annotations in {ftr.fr.relative_filename()}: {longs}"
   527	                )
   528	                ldata.annotate_long = longs[0]
   529	            else:
   530	                ldata.annotate_long = None
   531
   532	            css_classes = []
   533	            if ldata.category:
   534	                css_classes.append(
   535	                    self.template_globals["category"][ldata.category],   # type: ignore[index]
   536	                )
   537	            ldata.css_class = " ".join(css_classes) or "pln"
   538
   539	        html_path = os.path.join(self.directory, ftr.html_filename)
   540	        html = self.source_tmpl.render({
   541	            **file_data.__dict__,
   542	            "contexts_json": contexts_json,
   543	            "prev_html": ftr.prev_html,
   544	            "next_html": ftr.next_html,
   545	        })
   546	        write_html(html_path, html)
   547
   548	        # Save this file's information for the index page.
   549	        index_info = IndexItem(
   550	            url = ftr.html_filename,
   551	            file = escape(ftr.fr.relative_filename()),
   552	            nums = ftr.analysis.numbers,
   553	        )
   554	        self.index_pages["file"].summaries.append(index_info)
   555	        self.incr.set_index_info(ftr.rootname, index_info)
   556
   557	    def write_file_index_page(self, first_html: str, final_html: str) -> None:
   558	        """Write the file index page for this report."""
   559	        index_file = self.write_index_page(
   560	            self.index_pages["file"],
   561	            first_html=first_html,
   562	            final_html=final_html,
   563	        )
   564
   565	        print_href = stdout_link(index_file, f"file://{os.path.abspath(index_file)}")
   566	        self.coverage._message(f"Wrote HTML report to {print_href}")
   567
   568	        # Write the latest hashes for next time.
   569	        self.incr.write()

   571	    def write_region_index_pages(self, files_to_report: Iterable[FileToReport]) -> None:
   572	        """Write the other index pages for this report."""
   573	        for ftr in files_to_report:
   574	            region_nouns = [pair[0] for pair in ftr.fr.code_region_kinds()]
   575	            num_lines = len(ftr.fr.source().splitlines())
   576	            regions = ftr.fr.code_regions()
   577
   578	            for noun in region_nouns:
   579	                page_data = self.index_pages[noun]
   580	                outside_lines = set(range(1, num_lines + 1))
   581
   582	                for region in regions:
   583	                    if region.kind != noun:
   584	                        continue
   585	                    outside_lines -= region.lines
   586	                    analysis = ftr.analysis.narrow(region.lines)
   587	                    if not self.should_report(analysis, page_data):
   588	                        continue
   589	                    sorting_name = region.name.rpartition(".")[-1].lstrip("_")
   590	                    page_data.summaries.append(IndexItem(
   591	                        url=f"{ftr.html_filename}#t{region.start}",
   592	                        file=escape(ftr.fr.relative_filename()),
   593	                        description=(
   594	                            f"<data value='{escape(sorting_name)}'>"
   595	                            + escape(region.name)
   596	                            + "</data>"
   597	                        ),
   598	                        nums=analysis.numbers,
   599	                    ))
   600
   601	                analysis = ftr.analysis.narrow(outside_lines)
   602	                if self.should_report(analysis, page_data):
   603	                    page_data.summaries.append(IndexItem(
   604	                        url=ftr.html_filename,
   605	                        file=escape(ftr.fr.relative_filename()),
   606	                        description=(
   607	                            "<data value=''>"
   608	                            + f"<span class='no-noun'>(no {escape(noun)})</span>"
   609	                            + "</data>"
   610	                        ),
   611	                        nums=analysis.numbers,
   612	                    ))
   613
   614	        for noun, index_page in self.index_pages.items():
   615	            if noun != "file":
   616	                self.write_index_page(index_page)

   618	    def write_index_page(self, index_page: IndexPage, **kwargs: str) -> str:
   619	        """Write an index page specified by `index_page`.
   620
   621	        Returns the filename created.
   622	        """
   623	        skipped_covered_msg = skipped_empty_msg = ""
   624	        if n := index_page.skipped_covered_count:
   625	            word = plural(n, index_page.noun, index_page.plural)
   626	            skipped_covered_msg = f"{n} {word} skipped due to complete coverage."
   627	        if n := index_page.skipped_empty_count:
   628	            word = plural(n, index_page.noun, index_page.plural)
   629	            skipped_empty_msg = f"{n} empty {word} skipped."
   630
   631	        index_buttons = [
   632	            {
   633	                "label": ip.plural.title(),
   634	                "url": ip.filename if ip.noun != index_page.noun else "",
   635	                "current": ip.noun == index_page.noun,
   636	            }
   637	            for ip in self.index_pages.values()
   638	        ]
   639	        render_data = {
   640	            "regions": index_page.summaries,
   641	            "totals": index_page.totals,
   642	            "noun": index_page.noun,
   643	            "region_noun": index_page.noun if index_page.noun != "file" else "",
   644	            "skip_covered": self.skip_covered,
   645	            "skipped_covered_msg": skipped_covered_msg,
   646	            "skipped_empty_msg": skipped_empty_msg,
   647	            "first_html": "",
   648	            "final_html": "",
   649	            "index_buttons": index_buttons,
   650	        }
   651	        render_data.update(kwargs)
   652	        html = self.index_tmpl.render(render_data)
   653
   654	        index_file = os.path.join(self.directory, index_page.filename)
   655	        write_html(index_file, html)
   656	        return index_file

   666	class IncrementalChecker:
   667	    """Logic and data to support incremental reporting.
   668
   669	    When generating an HTML report, often only a few of the source files have
   670	    changed since the last time we made the HTML report.  This means previously
   671	    created HTML pages can be reused without generating them again, speeding
   672	    the command.
   673
   674	    This class manages a JSON data file that captures enough information to
   675	    know whether an HTML page for a .py file needs to be regenerated or not.
   676	    The data file also needs to store all the information needed to create the
   677	    entry for the file on the index page so that if the HTML page is reused,
   678	    the index page can still be created to refer to it.
   679
   680	    The data looks like::
   681
   682	        {
   683	            "note": "This file is an internal implementation detail ...",
   684	            // A fixed number indicating the data format.  STATUS_FORMAT
   685	            "format": 5,
   686	            // The version of coverage.py
   687	            "version": "7.4.4",
   688	            // A hash of a number of global things, including the configuration
   689	            // settings and the pyfile.html template itself.
   690	            "globals": "540ee119c15d52a68a53fe6f0897346d",
   691	            "files": {
   692	                // An entry for each source file keyed by the flat_rootname().
   693	                "z_7b071bdc2a35fa80___init___py": {
   694	                    // Hash of the source, the text of the .py file.
   695	                    "hash": "e45581a5b48f879f301c0f30bf77a50c",
   696	                    // Information for the index.html file.
   697	                    "index": {
   698	                        "url": "z_7b071bdc2a35fa80___init___py.html",
   699	                        "file": "cogapp/__init__.py",
   700	                        "description": "",
   701	                        // The Numbers for this file.
   702	                        "nums": { "precision": 2, "n_files": 1, "n_statements": 43, ... }
   703	                    }
   704	                },
   705	                ...
   706	            }
   707	        }
   708
   709	    """
   710
   711	    STATUS_FILE = "status.json"
   712	    STATUS_FORMAT = 5
   713	    NOTE = (
   714	        "This file is an internal implementation detail to speed up HTML report"
   715	        + " generation. Its format can change at any time. You might be looking"
   716	        + " for the JSON report: https://coverage.rtfd.io/cmd.html#cmd-json"
   717	    )
   718
   719	    def __init__(self, directory: str) -> None:
   720	        self.directory = directory
   721	        self._reset()
   722
   723	    def _reset(self) -> None:
   724	        """Initialize to empty. Causes all files to be reported."""
   725	        self.globals = ""
   726	        self.files: dict[str, FileInfo] = {}
   727
   728	    def read(self) -> None:
   729	        """Read the information we stored last time."""
   730	        try:
   731	            status_file = os.path.join(self.directory, self.STATUS_FILE)
   732	            with open(status_file, encoding="utf-8") as fstatus:
   733	                status = json.load(fstatus)
   734	        except (OSError, ValueError):
   735	            # Status file is missing or malformed.
   736	            usable = False
   737	        else:
   738	            if status["format"] != self.STATUS_FORMAT:
   739	                usable = False
   740	            elif status["version"] != coverage.__version__:
   741	                usable = False
   742	            else:
   743	                usable = True
   744
   745	        if usable:
   746	            self.files = {}
   747	            for filename, filedict in status["files"].items():
   748	                indexdict = filedict["index"]
   749	                index_item = IndexItem(**indexdict)
   750	                index_item.nums = Numbers(**indexdict["nums"])
   751	                fileinfo = FileInfo(
   752	                    hash=filedict["hash"],
   753	                    index=index_item,
   754	                )
   755	                self.files[filename] = fileinfo
   756	            self.globals = status["globals"]
   757	        else:
   758	            self._reset()
   759
   760	    def write(self) -> None:
   761	        """Write the current status."""
   762	        status_file = os.path.join(self.directory, self.STATUS_FILE)
   763	        status_data = {
   764	            "note": self.NOTE,
   765	            "format": self.STATUS_FORMAT,
   766	            "version": coverage.__version__,
   767	            "globals": self.globals,
   768	            "files": {
   769	                fname: dataclasses.asdict(finfo)
   770	                for fname, finfo in self.files.items()
   771	            },
   772	        }
   773	        with open(status_file, "w", encoding="utf-8") as fout:
   774	            json.dump(status_data, fout, separators=(",", ":"))
   775
   776	    def check_global_data(self, *data: Any) -> None:
   777	        """Check the global data that can affect incremental reporting.
   778
   779	        Pass in whatever global information could affect the content of the
   780	        HTML pages.  If the global data has changed since last time, this will
   781	        clear the data so that all files are regenerated.
   782
   783	        """
   784	        m = Hasher()
   785	        for d in data:
   786	            m.update(d)
   787	        these_globals = m.hexdigest()
   788	        if self.globals != these_globals:
   789	            self._reset()
   790	            self.globals = these_globals

   792	    def can_skip_file(self, data: CoverageData, fr: FileReporter, rootname: str) -> bool:
   793	        """Can we skip reporting this file?
   794
   795	        `data` is a CoverageData object, `fr` is a `FileReporter`, and
   796	        `rootname` is the name being used for the file.
   797
   798	        Returns True if the HTML
