    class Stub:
        def predict(self, text: str):
            return {"label": label, "score": score}

    return Stub()


def _mk_stub_rules(spam: bool):
    mod = types.SimpleNamespace()
    mod.contains_keywords = lambda s: spam
    mod.link_ratio = lambda s: 0.9 if spam else 0.0
    return mod
