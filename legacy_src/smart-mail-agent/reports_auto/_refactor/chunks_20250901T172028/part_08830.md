 SH[k].idDelta
   818	                        if glyph!=0 and glyph<self.numGlyphs:
   819	                            enc = (unichar<<8)|(j+SH[k].firstCode)
   820	                            charToGlyph[enc] = glyph
   821	                            glyphToChar.setdefault(glyph,[]).append(enc)
   822	                    if last==-1:
   823	                        last = unichar
   824	        else:
   825	            raise ValueError('Unsupported cmap encoding format %d' % fmt)
   826
   827	        # hmtx - Horizontal metrics table
   828	        # (needs data from hhea, maxp, and cmap tables)
   829	        self.seek_table("hmtx")
   830	        aw = None
   831	        self.charWidths = charWidths = {}
   832	        self.hmetrics = []
   833	        for glyph in range(numberOfHMetrics):
   834	            # advance width and left side bearing.  lsb is actually signed
   835	            # short, but we don't need it anyway (except for subsetting)
   836	            aw, lsb = self.read_ushort(), self.read_ushort()
   837	            self.hmetrics.append((aw, lsb))
   838	            aw = scale(aw)
   839	            if glyph == 0:
   840	                self.defaultWidth = aw
   841	            if glyph in glyphToChar:
   842	                for char in glyphToChar[glyph]:
   843	                    charWidths[char] = aw
   844	        for glyph in range(numberOfHMetrics, numGlyphs):
   845	            # the rest of the table only lists advance left side bearings.
   846	            # so we reuse aw set by the last iteration of the previous loop
   847	            lsb = self.read_ushort()
   848	            self.hmetrics.append((aw, lsb))
   849	            if glyph in glyphToChar:
   850	                for char in glyphToChar[glyph]:
   851	                    charWidths[char] = aw
   852
   853	        # loca - Index to location
   854	        if 'loca' not in self.table: raise TTFError('missing location table')
   855	        self.seek_table('loca')
   856	        self.glyphPos = []
   857	        if indexToLocFormat == 0:
   858	            for n in range(numGlyphs + 1):
   859	                self.glyphPos.append(self.read_ushort() << 1)
   860	        elif indexToLocFormat == 1:
   861	            for n in range(numGlyphs + 1):
   862	                self.glyphPos.append(self.read_ulong())
   863	        else:
   864	            raise TTFError('Unknown location table format (%d)' % indexToLocFormat)
   865	        if 0x20 in charToGlyph:
   866	            charToGlyph[0xa0] = charToGlyph[0x20]
   867	            charWidths[0xa0] = charWidths[0x20]
   868	        elif 0xa0 in charToGlyph:
   869	            charToGlyph[0x20] = charToGlyph[0xa0]
   870	            charWidths[0x20] = charWidths[0xa0]
   871

   874	    def makeSubset(self, subset):
   875	        """Create a subset of a TrueType font"""
   876	        output = TTFontMaker()
   877
   878	        # Build a mapping of glyphs in the subset to glyph numbers in
   879	        # the original font.  Also build a mapping of UCS codes to
   880	        # glyph values in the new font.
   881
   882	        # Start with 0 -> 0: "missing character"
   883	        glyphMap = [0]                  # new glyph index -> old glyph index
   884	        glyphSet = {0:0}                # old glyph index -> new glyph index
   885	        codeToGlyph = {}                # unicode -> new glyph index
   886	        for code in subset:
   887	            if code in self.charToGlyph:
   888	                originalGlyphIdx = self.charToGlyph[code]
   889	            else:
   890	                originalGlyphIdx = 0
   891	            if originalGlyphIdx not in glyphSet:
   892	                glyphSet[originalGlyphIdx] = len(glyphMap)
   893	                glyphMap.append(originalGlyphIdx)
   894	            codeToGlyph[code] = glyphSet[originalGlyphIdx]
   895
   896	        # Also include glyphs that are parts of composite glyphs
   897	        start = self.get_table_pos('glyf')[0]
   898	        n = 0
   899	        while n < len(glyphMap):
   900	            originalGlyphIdx = glyphMap[n]
   901	            glyphPos = self.glyphPos[originalGlyphIdx]
   902	            glyphLen = self.glyphPos[originalGlyphIdx + 1] - glyphPos
   903	            n += 1
   904	            if not glyphLen: continue
   905	            self.seek(start + glyphPos)
   906	            numberOfContours = self.read_short()
   907	            if numberOfContours < 0:
   908	                # composite glyph
   909	                self.skip(8)
   910	                flags = GF_MORE_COMPONENTS
   911	                while flags & GF_MORE_COMPONENTS:
   912	                    flags = self.read_ushort()
   913	                    glyphIdx = self.read_ushort()
   914	                    if glyphIdx not in glyphSet:
   915	                        glyphSet[glyphIdx] = len(glyphMap)
   916	                        glyphMap.append(glyphIdx)
   917	                    if flags & GF_ARG_1_AND_2_ARE_WORDS:
   918	                        self.skip(4)
   919	                    else:
   920	                        self.skip(2)
   921	                    if flags & GF_WE_HAVE_A_SCALE:
   922	                        self.skip(2)
   923	                    elif flags & GF_WE_HAVE_AN_X_AND_Y_SCALE:
   924	                        self.skip(4)
   925	                    elif flags & GF_WE_HAVE_A_TWO_BY_TWO:
   926	                        self.skip(8)
   927
   928
   929	        # The following tables are simply copied from the original
   930	        for tag in ('name', 'OS/2', 'cvt ', 'fpgm', 'prep'):
   931	            try:
   932	                output.add(tag, self.get_table(tag))
   933	            except KeyError:
   934	                # Apparently some of the tables are optional (cvt, fpgm, prep).
   935	                # The lack of the required ones (name, OS/2) would have already
   936	                # been caught before.
   937	                pass
   938
   939	        # post - PostScript
   940	        post = b"\x00\x03\x00\x00" + self.get_table('post')[4:16] + b"\x00" * 16
   941	        output.add('post', post)
   942
   943	        numGlyphs = len(glyphMap)
   944
   945	        # hmtx - Horizontal Metrics
   946	        hmtx = []
   947	        for n in range(numGlyphs):
   948	            aw, lsb = self.hmetrics[glyphMap[n]]
   949	            hmtx.append(int(aw))
   950	            hmtx.append(int(lsb))
   951
   952	        #work out n as 0 or first aw that's the start of a run
   953	        n = len(hmtx)-2
   954	        while n and hmtx[n]==hmtx[n-2]:
   955	            n -= 2
   956	        if not n: n = 2                 #need at least one pair
   957	        numberOfHMetrics = n>>1         #number of full H Metric pairs
   958	        hmtx = hmtx[:n] + hmtx[n+1::2]  #full pairs + all the trailing lsb's
   959
   960	        hmtx = pack(*([">%dH" % len(hmtx)] + hmtx))
   961	        output.add('hmtx', hmtx)
   962
   963	        # hhea - Horizontal Header
   964	        hhea = self.get_table('hhea')
   965	        hhea = _set_ushort(hhea, 34, numberOfHMetrics)
   966	        output.add('hhea', hhea)
   967
   968	        # maxp - Maximum Profile
   969	        maxp = self.get_table('maxp')
   970	        maxp = _set_ushort(maxp, 4, numGlyphs)
   971	        output.add('maxp', maxp)
   972
   973	        # cmap - Character to glyph mapping
   974	        # XXX maybe use format 0 if possible, not 6?
   975	        entryCount = len(subset)
   976	        length = 10 + entryCount * 2
   977	        cmap = [0, 1,           # version, number of tables
   978	                1, 0, 0,12,     # platform, encoding, offset (hi,lo)
   979	                6, length, 0,   # format, length, language
   980	                0,
   981	                entryCount] + \
   982	               list(map(codeToGlyph.get, subset))
   983	        cmap = pack(*([">%dH" % len(cmap)] + cmap))
   984	        output.add('cmap', cmap)
   985
   986	        # glyf - Glyph data
   987	        glyphData = self.get_table('glyf')
   988	        offsets = []
   989	        glyf = []
   990	        pos = 0
   991	        for n in range(numGlyphs):
   992	            offsets.append(pos)
   993	            originalGlyphIdx = glyphMap[n]
   994	            glyphPos = self.glyphPos[originalGlyphIdx]
   995	            glyphLen = self.glyphPos[originalGlyphIdx + 1] - glyphPos
   996	            data = glyphData[glyphPos:glyphPos+glyphLen]
   997	            # Fix references in composite glyphs
   998	            if glyphLen > 2 and unpack(">h", data[:2])[0] < 0:
   999	                # composite glyph
  1000	                pos_in_glyph = 10
  1001	                flags = GF_MORE_COMPONENTS
  1002	                while flags & GF_MORE_COMPONENTS:
  1003	                    flags = unpack(">H", data[pos_in_glyph:pos_in_glyph+2])[0]
  1004	                    glyphIdx = unpack(">H", data[pos_in_glyph+2:pos_in_glyph+4])[0]
  1005	                    data = _set_ushort(data, pos_in_glyph + 2, glyphSet[glyphIdx])
  1006	                    pos_in_glyph = pos_in_glyph + 4
  1007	                    if flags & GF_ARG_1_AND_2_ARE_WORDS:
  1008	                        pos_in_glyph = pos_in_glyph + 4
  1009	                    else:
  1010	                        pos_in_glyph = pos_in_glyph + 2
  1011	                    if flags & GF_WE_HAVE_A_SCALE:
  1012	                        pos_in_glyph = pos_in_glyph + 2
  1013	                    elif flags & GF_WE_HAVE_AN_X_AND_Y_SCALE:
  1014	                        pos_in_glyph = pos_in_glyph + 4
  1015	                    elif flags & GF_WE_HAVE_A_TWO_BY_TWO:
  1016	                        pos_in_glyph = pos_in_glyph + 8
  1017	            glyf.append(data)
  1018	            pos = pos + glyphLen
  1019	            if pos % 4 != 0:
  1020	                padding = 4 - pos % 4
  1021	                glyf.append(b'\0' * padding)
  1022	                pos = pos + padding
  1023	        offsets.append(pos)
  1024	        output.add('glyf', b''.join(glyf))
  1025
  1026	        # loca - Index to location
  1027	        loca = []
  1028	        if (pos + 1) >> 1 > 0xFFFF:
  1029	            indexToLocFormat = 1        # long format
  1030	            for offset in offsets:
  1031	                loca.append(offset)
  1032	            loca = pack(*([">%dL" % len(loca)] + loca))
  1033	        else:
  1034	            indexToLocFormat = 0        # short format
  1035	            for offset in offsets:
  1036	                loca.append(offset >> 1)
  1037	            loca = pack(*([">%dH" % len(loca)] + loca))
  1038	        output.add('loca', loca)
  1039
  1040	        # head - Font header
  1041	        head = self.get_table('head')
  1042	        head = _set_ushort(head, 50, indexToLocFormat)
  1043	        output.add('head', head)
  1044
  1045	        return output.makeStream()
