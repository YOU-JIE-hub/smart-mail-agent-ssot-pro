ould("dataop"):
   491	            self._debug.write("Adding lines: %d files, %d lines total" % (
   492	                len(line_data), sum(len(lines) for lines in line_data.values()),
   493	            ))
   494	            if self._debug.should("dataop2"):
   495	                for filename, linenos in sorted(line_data.items()):
   496	                    self._debug.write(f"  {filename}: {linenos}")
   497	        self._start_using()
   498	        self._choose_lines_or_arcs(lines=True)
   499	        if not line_data:
   500	            return
   501	        with self._connect() as con:
   502	            self._set_context_id()
   503	            for filename, linenos in line_data.items():
   504	                line_bits = nums_to_numbits(linenos)
   505	                file_id = self._file_id(filename, add=True)
   506	                query = "select numbits from line_bits where file_id = ? and context_id = ?"
   507	                with con.execute(query, (file_id, self._current_context_id)) as cur:
   508	                    existing = list(cur)
   509	                if existing:
   510	                    line_bits = numbits_union(line_bits, existing[0][0])
   511
   512	                con.execute_void(
   513	                    "insert or replace into line_bits " +
   514	                    " (file_id, context_id, numbits) values (?, ?, ?)",
   515	                    (file_id, self._current_context_id, line_bits),
   516	                )

   519	    def add_arcs(self, arc_data: Mapping[str, Collection[TArc]]) -> None:
   520	        """Add measured arc data.
   521
   522	        `arc_data` is a dictionary mapping file names to iterables of pairs of
   523	        ints::
   524
   525	            { filename: { (l1,l2), (l1,l2), ... }, ...}
   526
   527	        """
   528	        if self._debug.should("dataop"):
   529	            self._debug.write("Adding arcs: %d files, %d arcs total" % (
   530	                len(arc_data), sum(len(arcs) for arcs in arc_data.values()),
   531	            ))
   532	            if self._debug.should("dataop2"):
   533	                for filename, arcs in sorted(arc_data.items()):
   534	                    self._debug.write(f"  {filename}: {arcs}")
   535	        self._start_using()
   536	        self._choose_lines_or_arcs(arcs=True)
   537	        if not arc_data:
   538	            return

   540	            self._set_context_id()
   541	            for filename, arcs in arc_data.items():
   542	                if not arcs:
   543	                    continue
   544	                file_id = self._file_id(filename, add=True)
   545	                data = [(file_id, self._current_context_id, fromno, tono) for fromno, tono in arcs]
   546	                con.executemany_void(
   547	                    "insert or ignore into arc " +
   548	                    "(file_id, context_id, fromno, tono) values (?, ?, ?, ?)",
   549	                    data,
   550	                )

   552	    def _choose_lines_or_arcs(self, lines: bool = False, arcs: bool = False) -> None:
   553	        """Force the data file to choose between lines and arcs."""
   554	        assert lines or arcs
   555	        assert not (lines and arcs)
   556	        if lines and self._has_arcs:
   557	            if self._debug.should("dataop"):
   558	                self._debug.write("Error: Can't add line measurements to existing branch data")
   559	            raise DataError("Can't add line measurements to existing branch data")
   560	        if arcs and self._has_lines:
   561	            if self._debug.should("dataop"):
   562	                self._debug.write("Error: Can't add branch measurements to existing line data")
   563	            raise DataError("Can't add branch measurements to existing line data")
   564	        if not self._has_arcs and not self._has_lines:
   565	            self._has_lines = lines
   566	            self._has_arcs = arcs
   567	            with self._connect() as con:
   568	                con.execute_void(
   569	                    "insert or ignore into meta (key, value) values (?, ?)",
   570	                    ("has_arcs", str(int(arcs))),
   571	                )

   574	    def add_file_tracers(self, file_tracers: Mapping[str, str]) -> None:
   575	        """Add per-file plugin information.
   576
   577	        `file_tracers` is { filename: plugin_name, ... }
   578
   579	        """
   580	        if self._debug.should("dataop"):
   581	            self._debug.write("Adding file tracers: %d files" % (len(file_tracers),))
   582	        if not file_tracers:
   583	            return
   584	        self._start_using()
   585	        with self._connect() as con:
   586	            for filename, plugin_name in file_tracers.items():
   587	                file_id = self._file_id(filename, add=True)
   588	                existing_plugin = self.file_tracer(filename)
   589	                if existing_plugin:
   590	                    if existing_plugin != plugin_name:
   591	                        raise DataError(
   592	                            "Conflicting file tracer name for '{}': {!r} vs {!r}".format(
   593	                                filename, existing_plugin, plugin_name,
   594	                            ),
   595	                        )
   596	                elif plugin_name:
   597	                    con.execute_void(
   598	                        "insert into tracer (file_id, tracer) values (?, ?)",
   599	                        (file_id, plugin_name),
   600	                    )

   653	    def update(
   654	        self,
   655	        other_data: CoverageData,
   656	        map_path: Callable[[str], str] | None = None,
   657	    ) -> None:
   658	        """Update this data with data from another :class:`CoverageData`.
   659
   660	        If `map_path` is provided, it's a function that re-map paths to match
   661	        the local machine's.  Note: `map_path` is None only when called
   662	        directly from the test suite.
   663
   664	        """
   665	        if self._debug.should("dataop"):
   666	            self._debug.write("Updating with data from {!r}".format(
   667	                getattr(other_data, "_filename", "???"),
   668	            ))
   669	        if self._has_lines and other_data._has_arcs:
   670	            raise DataError("Can't combine branch coverage data with statement data")
   671	        if self._has_arcs and other_data._has_lines:
   672	            raise DataError("Can't combine statement coverage data with branch data")
   673
   674	        map_path = map_path or (lambda p: p)
   675
   676	        # Force the database we're writing to to exist before we start nesting contexts.
   677	        self._start_using()
   678
   679	        # Collector for all arcs, lines and tracers
   680	        other_data.read()
   681	        with other_data._connect() as con:
   682	            # Get files data.
   683	            with con.execute("select path from file") as cur:
   684	                files = {path: map_path(path) for (path,) in cur}
   685
   686	            # Get contexts data.
   687	            with con.execute("select context from context") as cur:
   688	                contexts = [context for (context,) in cur]
   689
   690	            # Get arc data.
   691	            with con.execute(
   692	                "select file.path, context.context, arc.fromno, arc.tono " +
   693	                "from arc " +
   694	                "inner join file on file.id = arc.file_id " +
   695	                "inner join context on context.id = arc.context_id",
   696	            ) as cur:
   697	                arcs = [
   698	                    (files[path], context, fromno, tono)
   699	                    for (path, context, fromno, tono) in cur
   700	                ]
   701
   702	            # Get line data.
   703	            with con.execute(
   704	                "select file.path, context.context, line_bits.numbits " +
   705	                "from line_bits " +
   706	                "inner join file on file.id = line_bits.file_id " +
   707	                "inner join context on context.id = line_bits.context_id",
   708	            ) as cur:
   709	                lines: dict[tuple[str, str], bytes] = {}
   710	                for path, context, numbits in cur:
   711	                    key = (files[path], context)
   712	                    if key in lines:
   713	                        numbits = numbits_union(lines[key], numbits)
   714	                    lines[key] = numbits
   715
   716	            # Get tracer data.
   717	            with con.execute(
   718	                "select file.path, tracer " +
   719	                "from tracer " +
   720	                "inner join file on file.id = tracer.file_id",
   721	            ) as cur:
   722	                tracers = {files[path]: tracer for (path, tracer) in cur}
   723
   724	        with self._connect() as con:
   725	            assert con.con is not None
   726	            con.con.isolation_level = "IMMEDIATE"
   727
   728	            # Get all tracers in the DB. Files not in the tracers are assumed
   729	            # to have an empty string tracer. Since Sqlite does not support
   730	            # full outer joins, we have to make two queries to fill the
   731	            # dictionary.
   732	            with con.execute("select path from file") as cur:
   733	                this_tracers = {path: "" for path, in cur}
   734	            with con.execute(
   735	                "select file.path, tracer from tracer " +
   736	                "inner join file on file.id = tracer.file_id",
   737	            ) as cur:
   738	                this_tracers.update({
   739	                    map_path(path): tracer
   740	                    for path, tracer in cur
   741	                })
   742
   743	            # Create all file and context rows in the DB.
   744	            con.executemany_void(
   745	                "insert or ignore into file (path) values (?)",
   746	                ((file,) for file in files.values()),
   747	            )
   748	            with con.execute("select id, path from file") as cur:
   749	                file_ids = {path: id for id, path in cur}
   750	            self._file_map.update(file_ids)
   751	            con.executemany_void(
   752	                "insert or ignore into context (context) values (?)",
   753	                ((context,) for context in contexts),
   754	            )
   755	            with con.execute("select id, context from context") as cur:
   756	                context_ids = {context: id for id, context in cur}
   757
   758	            # Prepare tracers and fail, if a conflict is found.
   759	            # tracer_paths is used to ensure consistency over the tracer data
   760	            # and tracer_map tracks the tracers to be inserted.
   761	            tracer_map = {}
   762	            for path in files.values():
   763	                this_tracer = this_tracers.get(path)
   764	                other_tracer = tracers.get(path, "")
   765	                # If there is no tracer, there is always the None tracer.
   766	                if this_tracer is not None and this_tracer != other_tracer:
   767	                    raise DataError(
   768	                        "Conflicting file tracer name for '{}': {!r} vs {!r}".format(
   769	                            path, this_tracer, other_tracer,
   770	                        ),
   771	                    )
   772	                tracer_map[path] = other_tracer
   773
   774	            # Prepare arc and line rows to be inserted by converting the file
   775	            # and context strings with integer ids. Then use the efficient
   776	            # `executemany()` to insert all rows at once.
   777
   778	            if arcs:
   779	                self._choose_lines_or_arcs(arcs=True)
   780
   781	                arc_rows = (
   782	                    (file_ids[file], context_ids[context], fromno, tono)
   783	                    for file, context, fromno, tono in arcs
   784	                )
   785
   786	                # Write the combined data.
   787	                con.executemany_void(
   788	                    "insert or ignore into arc " +
   789	                    "(file_id, context_id, fromno, tono) values (?, ?, ?, ?)",
   790	                    arc_rows,
   791	                )
   792
   793	            if lines:
   794	                self._choose_lines_or_arcs(lines=True)
   795
   796	                for (file, context), numbits in lines.items():
   797	                    with con.execute(
   798	                        "select numbits from line_bits where file_id = ? and context_id = ?",
   799	                        (file_ids[file], context_ids[context]),
   800	                    ) as cur:
   801	                        existing = list(cur)
   802	                    if existing:
   803	                        lines[(file, context)] = numbits_union(numbits, existing[0][0])
   804
   805	                con.executemany_void(
   806	                    "insert or replace into line_bits " +
   807	                    "(file_id, context_id, numbits) values (?, ?, ?)",
   808	                    [
   809	                        (file_ids[file], context_ids[context], numbits)
   810	                        for (file, context), numbits in lines.items()
   811	                    ],
   812	                )
   813
   814	            con.executemany_void(
   815	                "insert or ignore into tracer (file_id, tracer) values (?, ?)",
   816	                ((file_ids[filename], tracer) for filename, tracer in tracer_map.items()),
   817	            )
   818
   819	        if not self._no_disk:
   820	            # Update all internal cache data.
   821	            self._reset()
   822	            self.read()
   823

   825	        """Erase the data in this object.
   826
   827	        If `parallel` is true, then also deletes data files created from the
   828	        basename by parallel-mode.

   830	        """
   831	        self._reset()
   832	        if self._no_disk:
   833	            return
   834	        self._debug_dataio("Erasing data file", self._filename)

   839	            pattern = glob.escape(local_abs_path) + ".*"
   840	            for filename in glob.glob(pattern):
   841	                self._debug_dataio("Erasing parallel data file", filename)
   842	                file_be_gone(filename)

   892
   893	        Returns the name of the plugin that handles this file.  If the file was
   894	        measured, but didn't use a plugin, then "" is returned.  If the file

   908	    def set_query_context(self, context: str) -> None:

   911	        The next :meth:`lines`, :meth:`arcs`, or :meth:`contexts_by_lineno`
   912	        calls will be limited to only one context.  `context` is a string which
