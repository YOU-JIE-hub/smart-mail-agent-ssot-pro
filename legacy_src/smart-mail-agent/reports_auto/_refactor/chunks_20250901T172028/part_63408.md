   209	            # This could have the additional effect of loading not-yet-loaded submodules
   210	            # mentioned in __all__

   215	def test_module(module_name: str) -> Iterator[Error]:
   216	    """Tests a given module's stub against introspecting it at runtime.
   217
   218	    Requires the stub to have been built already, accomplished by a call to ``build_stubs``.
   219
   220	    :param module_name: The module to test
   221
   222	    """
   223	    stub = get_stub(module_name)
   224	    if stub is None:
   225	        if not is_probably_private(module_name.split(".")[-1]):
   226	            runtime_desc = repr(sys.modules[module_name]) if module_name in sys.modules else "N/A"
   227	            yield Error(
   228	                [module_name], "failed to find stubs", MISSING, None, runtime_desc=runtime_desc
   229	            )
   230	        return
   231
   232	    try:
   233	        runtime = silent_import_module(module_name)
   234	    except KeyboardInterrupt:
   235	        raise
   236	    except BaseException as e:
   237	        note = ""
   238	        if isinstance(e, ModuleNotFoundError):
   239	            note = " Maybe install the runtime package or alter PYTHONPATH?"
   240	        yield Error(
   241	            [module_name], f"failed to import.{note} {type(e).__name__}: {e}", stub, MISSING
   242	        )
   243	        return
   244
   245	    with warnings.catch_warnings():
   246	        warnings.simplefilter("ignore")
   247	        try:
   248	            yield from verify(stub, runtime, [module_name])
   249	        except Exception as e:
   250	            bottom_frame = list(traceback.walk_tb(e.__traceback__))[-1][0]
   251	            bottom_module = bottom_frame.f_globals.get("__name__", "")
   252	            # Pass on any errors originating from stubtest or mypy
   253	            # These can occur expectedly, e.g. StubtestFailure
   254	            if bottom_module == "__main__" or bottom_module.split(".")[0] == "mypy":
   255	                raise
   256	            yield Error(
   257	                [module_name],
   258	                f"encountered unexpected error, {type(e).__name__}: {e}",
   259	                stub,
   260	                runtime,
   261	                stub_desc="N/A",
   262	                runtime_desc=(
   263	                    "This is most likely the fault of something very dynamic in your library. "
   264	                    "It's also possible this is a bug in stubtest.\nIf in doubt, please "
   265	                    "open an issue at https://github.com/python/mypy\n\n"
   266	                    + traceback.format_exc().strip()
   267	                ),
   268	            )

   286	def _verify_exported_names(
   287	    object_path: list[str], stub: nodes.MypyFile, runtime_all_as_set: set[str]
   288	) -> Iterator[Error]:
   289	    # note that this includes the case the stub simply defines `__all__: list[str]`
   290	    assert "__all__" in stub.names
   291	    public_names_in_stub = {m for m, o in stub.names.items() if o.module_public}
   292	    names_in_stub_not_runtime = sorted(public_names_in_stub - runtime_all_as_set)
   293	    names_in_runtime_not_stub = sorted(runtime_all_as_set - public_names_in_stub)
   294	    if not (names_in_runtime_not_stub or names_in_stub_not_runtime):
   295	        return
   296	    yield Error(
   297	        object_path + ["__all__"],
   298	        (
   299	            "names exported from the stub do not correspond to the names exported at runtime. "
   300	            "This is probably due to things being missing from the stub or an inaccurate `__all__` in the stub"
   301	        ),
   302	        # Pass in MISSING instead of the stub and runtime objects, as the line numbers aren't very
   303	        # relevant here, and it makes for a prettier error message
   304	        # This means this error will be ignored when using `--ignore-missing-stub`, which is
   305	        # desirable in at least the `names_in_runtime_not_stub` case
   306	        stub_object=MISSING,
   307	        runtime_object=MISSING,
   308	        stub_desc=(f"Names exported in the stub but not at runtime: {names_in_stub_not_runtime}"),
   309	        runtime_desc=(
   310	            f"Names exported at runtime but not in the stub: {names_in_runtime_not_stub}"
   311	        ),
   312	    )

   315	@functools.lru_cache
   316	def _module_symbol_table(runtime: types.ModuleType) -> symtable.SymbolTable | None:
   317	    """Retrieve the symbol table for the module (or None on failure).
   318
   319	    1) Use inspect to retrieve the source code of the module
   320	    2) Use symtable to parse the source (and use what symtable knows for its purposes)
   321	    """
   322	    try:
   323	        source = inspect.getsource(runtime)
   324	    except (OSError, TypeError, SyntaxError):
   325	        return None
   326
   327	    try:
   328	        return symtable.symtable(source, runtime.__name__, "exec")
   329	    except SyntaxError:
   330	        return None

   333	@verify.register(nodes.MypyFile)
   334	def verify_mypyfile(
   335	    stub: nodes.MypyFile, runtime: MaybeMissing[types.ModuleType], object_path: list[str]
   336	) -> Iterator[Error]:
   337	    if isinstance(runtime, Missing):
   338	        yield Error(object_path, "is not present at runtime", stub, runtime)
   339	        return
   340	    if not isinstance(runtime, types.ModuleType):
   341	        # Can possibly happen:
   342	        yield Error(object_path, "is not a module", stub, runtime)  # type: ignore[unreachable]
   343	        return
   344
   345	    runtime_all_as_set: set[str] | None
   346
   347	    if hasattr(runtime, "__all__"):
   348	        runtime_all_as_set = set(runtime.__all__)
   349	        if "__all__" in stub.names:
   350	            # Only verify the contents of the stub's __all__
   351	            # if the stub actually defines __all__
   352	            yield from _verify_exported_names(object_path, stub, runtime_all_as_set)
   353	        else:
   354	            yield Error(object_path + ["__all__"], "is not present in stub", MISSING, runtime)
   355	    else:
   356	        runtime_all_as_set = None
   357
   358	    # Check things in the stub
   359	    to_check = {
   360	        m
   361	        for m, o in stub.names.items()
   362	        if not o.module_hidden and (not is_probably_private(m) or hasattr(runtime, m))
   363	    }
   364
   365	    def _belongs_to_runtime(r: types.ModuleType, attr: str) -> bool:
   366	        """Heuristics to determine whether a name originates from another module."""
   367	        obj = getattr(r, attr)
   368	        if isinstance(obj, types.ModuleType):
   369	            return False
   370
   371	        symbol_table = _module_symbol_table(r)
   372	        if symbol_table is not None:
   373	            try:
   374	                symbol = symbol_table.lookup(attr)
   375	            except KeyError:
   376	                pass
   377	            else:
   378	                if symbol.is_imported():
   379	                    # symtable says we got this from another module
   380	                    return False
   381	                # But we can't just return True here, because symtable doesn't know about symbols
   382	                # that come from `from module import *`
   383	                if symbol.is_assigned():
   384	                    # symtable knows we assigned this symbol in the module
   385	                    return True
   386
   387	        # The __module__ attribute is unreliable for anything except functions and classes,
   388	        # but it's our best guess at this point
   389	        try:
   390	            obj_mod = obj.__module__
   391	        except Exception:
   392	            pass
   393	        else:
   394	            if isinstance(obj_mod, str):
   395	                return bool(obj_mod == r.__name__)
   396	        return True
   397
   398	    runtime_public_contents = (
   399	        runtime_all_as_set
   400	        if runtime_all_as_set is not None
   401	        else {
   402	            m
   403	            for m in dir(runtime)
   404	            if not is_probably_private(m)
   405	            # Filter out objects that originate from other modules (best effort). Note that in the
   406	            # absence of __all__, we don't have a way to detect explicit / intentional re-exports
   407	            # at runtime
   408	            and _belongs_to_runtime(runtime, m)
   409	        }
   410	    )
   411	    # Check all things declared in module's __all__, falling back to our best guess
   412	    to_check.update(runtime_public_contents)
   413	    to_check.difference_update(IGNORED_MODULE_DUNDERS)
   414
   415	    for entry in sorted(to_check):
   416	        stub_entry = stub.names[entry].node if entry in stub.names else MISSING
   417	        if isinstance(stub_entry, nodes.MypyFile):
   418	            # Don't recursively check exported modules, since that leads to infinite recursion
   419	            continue
   420	        assert stub_entry is not None
   421	        try:
   422	            runtime_entry = getattr(runtime, entry, MISSING)
   423	        except Exception:
   424	            # Catch all exceptions in case the runtime raises an unexpected exception
   425	            # from __getattr__ or similar.
   426	            continue
   427	        yield from verify(stub_entry, runtime_entry, object_path + [entry])

   430	def _verify_final(
   431	    stub: nodes.TypeInfo, runtime: type[Any], object_path: list[str]
   432	) -> Iterator[Error]:
   433	    try:
   434
   435	        class SubClass(runtime):  # type: ignore[misc]
   436	            pass
   437
   438	    except TypeError:
   439	        # Enum classes are implicitly @final
   440	        if not stub.is_final and not issubclass(runtime, enum.Enum):
   441	            yield Error(
   442	                object_path,
   443	                "cannot be subclassed at runtime, but isn't marked with @final in the stub",
   444	                stub,
   445	                runtime,
   446	                stub_desc=repr(stub),
   447	            )
   448	    except Exception:
   449	        # The class probably wants its subclasses to do something special.
   450	        # Examples: ctypes.Array, ctypes._SimpleCData
   451	        pass
   452
   453	    # Runtime class might be annotated with `@final`:
   454	    try:
   455	        runtime_final = getattr(runtime, "__final__", False)
   456	    except Exception:
   457	        runtime_final = False
   458
   459	    if runtime_final and not stub.is_final:
   460	        yield Error(
   461	            object_path,
   462	            "has `__final__` attribute, but isn't marked with @final in the stub",
   463	            stub,
   464	            runtime,
   465	            stub_desc=repr(stub),
   466	        )

   469	def _verify_metaclass(
   470	    stub: nodes.TypeInfo, runtime: type[Any], object_path: list[str], *, is_runtime_typeddict: bool
   471	) -> Iterator[Error]:
   472	    # We exclude protocols, because of how complex their implementation is in different versions of
   473	    # python. Enums are also hard, as are runtime TypedDicts; ignoring.
   474	    # TODO: check that metaclasses are identical?
   475	    if not stub.is_protocol and not stub.is_enum and not is_runtime_typeddict:
   476	        runtime_metaclass = type(runtime)
   477	        if runtime_metaclass is not type and stub.metaclass_type is None:
   478	            # This means that runtime has a custom metaclass, but a stub does not.
   479	            yield Error(
   480	                object_path,
   481	                "is inconsistent, metaclass differs",
   482	                stub,
   483	                runtime,
   484	                stub_desc="N/A",
   485	                runtime_desc=f"{runtime_metaclass}",
   486	            )
   487	        elif (
   488	            runtime_metaclass is type
   489	            and stub.metaclass_type is not None
   490	            # We ignore extra `ABCMeta` metaclass on stubs, this might be typing hack.
   491	            # We also ignore `builtins.type` metaclass as an implementation detail in mypy.
   492	            and not mypy.types.is_named_instance(
   493	                stub.metaclass_type, ("abc.ABCMeta", "builtins.type")
   494	            )
   495	        ):
   496	            # This means that our stub has a metaclass that is not present at runtime.
   497	            yield Error(
   498	                object_path,
   499	                "metaclass mismatch",
   500	                stub,
   501	                runtime,
   502	                stub_desc=f"{stub.metaclass_type.type.fullname}",
   503	                runtime_desc="N/A",
   504	            )

   507	@verify.register(nodes.TypeInfo)
   508	def verify_typeinfo(
   509	    stub: nodes.TypeInfo, runtime: MaybeMissing[type[Any]], object_path: list[str]
   510	) -> Iterator[Error]:
   511	    if stub.is_type_check_only:
   512	        # This type only exists in stubs, we only check that the runtime part
   513	        # is missing. Other checks are not required.
   514	        if not isinstance(runtime, Missing):
   515	            yield Error(
   516	                object_path,
   517	                'is marked as "@type_check_only", but also exists at runtime',
   518	                stub,
   519	                runtime,
   520	                stub_desc=repr(stub),
   521	            )
   522	        return
   523
   524	    if isinstance(runtime, Missing):
   525	        yield Error(object_path, "is not present at runtime", stub, runtime, stub_desc=repr(stub))
   526	        return
   527	    if not isinstance(runtime, type):
   528	        # Yes, some runtime objects can be not types, no way to tell mypy about that.
   529	        yield Error(object_path, "is not a type", stub, runtime, stub_desc=repr(stub))  # type: ignore[unreachable]
   530	        return
   531
   532	    yield from _verify_final(stub, runtime, object_path)
   533	    is_runtime_typeddict = stub.typeddict_type is not None and is_typeddict(runtime)
   534	    yield from _verify_metaclass(
   535	        stub, runtime, object_path, is_runtime_typeddict=is_runtime_typeddict
   536	    )
   537
   538	    # Check everything already defined on the stub class itself (i.e. not inherited)
   539	    #
   540	    # Filter out non-identifier names, as these are (hopefully always?) whacky/fictional things
   541	    # (like __mypy-replace or __mypy-post_init, etc.) that don't exist at runtime,
   542	    # and exist purely for internal mypy reasons
   543	    to_check = {name for name in stub.names if name.isidentifier()}
   544	    # Check all public things on the runtime class
   545	    to_check.update(
   546	        m for m in vars(runtime) if not is_probably_private(m) and m not in IGNORABLE_CLASS_DUNDERS
   547	    )
   548	    # Special-case the __init__ method for Protocols and the __new__ method for TypedDicts
   549	    #
   550	    # TODO: On Python <3.11, __init__ methods on Protocol classes
   551	    # are silently discarded and replaced.
   552	    # However, this is not the case on Python 3.11+.
   553	    # Ideally, we'd figure out a good way of validating Protocol __init__ methods on 3.11+.
   554	    if stub.is_protocol:
   555	        to_check.discard("__init__")
   556	    if is_runtime_typeddict:
   557	        to_check.discard("__new__")
   558
   559	    for entry in sorted(to_check):
   560	        mangled_entry = entry
   561	        if entry.startswith("__") and not entry.endswith("__"):
   562	            mangled_entry = f"_{stub.name.lstrip('_')}{entry}"
   563	        stub_to_verify = next((t.names[entry].node for t in stub.mro if entry in t.names), MISSING)
   564	        assert stub_to_verify is not None
   565	        try:
   566	            try:
   567	                runtime_attr = getattr(runtime, mangled_entry)
   568	            except AttributeError:
   569	                runtime_attr = inspect.getattr_static(runtime, mangled_entry, MISSING)
   570	        except Exception:
   571	            # Catch all exceptions in case the runtime raises an unexpected exception
   572	            # from __getattr__ or similar.
   573	            continue
   574
   575	        # If it came from the metaclass, consider the runtime_attr to be MISSING
   576	        # for a more accurate message
   577	        if (
   578	            runtime_attr is not MISSING
   579	            and type(runtime) is not runtime
   580	            and getattr(runtime_attr, "__objclass__", None) is type(runtime)
   581	        ):
   582	            runtime_attr = MISSING
   583
   584	        # __setattr__ and __delattr__ on object are a special case,
   585	        # so if we only have these methods inherited from there, pretend that
   586	        # we don't have them. See python/typeshed#7385.
   587	        if (
   588	            entry in ("__setattr__", "__delattr__")
   589	            and runtime_attr is not MISSING
   590	            and runtime is not object
   591	            and getattr(runtime_attr, "__objclass__", None) is object
   592	        ):
   593	            runtime_attr = MISSING
   594
   595	        # Do not error for an object missing from the stub
   596	        # If the runtime object is a types.WrapperDescriptorType object
   597	        # and has a non-special dunder name.
   598	        # The vast majority of these are false positives.
   599	        if not (
   600	            isinstance(stub_to_verify, Missing)
   601	            and isinstance(runtime_attr, types.WrapperDescriptorType)
   602	            and is_dunder(mangled_entry, exclude_special=True)
   603	        ):
   604	            yield from verify(stub_to_verify, runtime_attr, object_path + [entry])
   605
