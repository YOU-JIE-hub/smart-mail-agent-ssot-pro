   715	            for resp in r.history:

   719
   720	        # Resolve redirects if allowed.
   721	        if allow_redirects:
   722	            # Redirect resolving generator.
   723	            gen = self.resolve_redirects(r, request, **kwargs)
   724	            history = [resp for resp in gen]
