  6701	    def add_symbol_table_node(
  6702	        self,
  6703	        name: str,
  6704	        symbol: SymbolTableNode,
  6705	        context: Context | None = None,
  6706	        can_defer: bool = True,
  6707	        escape_comprehensions: bool = False,
  6708	        no_progress: bool = False,
  6709	        type_param: bool = False,
  6710	    ) -> bool:
  6711	        """Add symbol table node to the currently active symbol table.
  6712
  6713	        Return True if we actually added the symbol, or False if we refused
  6714	        to do so (because something is not ready or it was a no-op).
  6715
  6716	        Generate an error if there is an invalid redefinition.
  6717
  6718	        If context is None, unconditionally add node, since we can't report
  6719	        an error. Note that this is used by plugins to forcibly replace nodes!
  6720
  6721	        TODO: Prevent plugins from replacing nodes, as it could cause problems?
  6722
  6723	        Args:
  6724	            name: short name of symbol
  6725	            symbol: Node to add
  6726	            can_defer: if True, defer current target if adding a placeholder
  6727	            context: error context (see above about None value)
  6728	        """
  6729	        names = self.current_symbol_table(
  6730	            escape_comprehensions=escape_comprehensions, type_param=type_param
  6731	        )
  6732	        existing = names.get(name)
  6733	        if isinstance(symbol.node, PlaceholderNode) and can_defer:
  6734	            if context is not None:
  6735	                self.process_placeholder(name, "name", context)
  6736	            else:
  6737	                # see note in docstring describing None contexts
  6738	                self.defer()
  6739	        if (
  6740	            existing is not None
  6741	            and context is not None
  6742	            and not is_valid_replacement(existing, symbol)
  6743	        ):
  6744	            # There is an existing node, so this may be a redefinition.
  6745	            # If the new node points to the same node as the old one,
  6746	            # or if both old and new nodes are placeholders, we don't
  6747	            # need to do anything.
  6748	            old = existing.node
  6749	            new = symbol.node
  6750	            if isinstance(new, PlaceholderNode):
  6751	                # We don't know whether this is okay. Let's wait until the next iteration.
  6752	                return False
  6753	            if not is_same_symbol(old, new):
  6754	                if isinstance(new, (FuncDef, Decorator, OverloadedFuncDef, TypeInfo)):
  6755	                    self.add_redefinition(names, name, symbol)
  6756	                if not (isinstance(new, (FuncDef, Decorator)) and self.set_original_def(old, new)):
  6757	                    self.name_already_defined(name, context, existing)
  6758	        elif name not in self.missing_names[-1] and "*" not in self.missing_names[-1]:
  6759	            names[name] = symbol
  6760	            if not no_progress:
  6761	                self.progress = True
  6762	            return True
  6763	        return False

  6765	    def add_redefinition(self, names: SymbolTable, name: str, symbol: SymbolTableNode) -> None:
  6766	        """Add a symbol table node that reflects a redefinition as a function or a class.
  6767
  6768	        Redefinitions need to be added to the symbol table so that they can be found
  6769	        through AST traversal, but they have dummy names of form 'name-redefinition[N]',
  6770	        where N ranges over 2, 3, ... (omitted for the first redefinition).
  6771
  6772	        Note: we always store redefinitions independently of whether they are valid or not
  6773	        (so they will be semantically analyzed), the caller should give an error for invalid
  6774	        redefinitions (such as e.g. variable redefined as a class).
  6775	        """
  6776	        i = 1
  6777	        # Don't serialize redefined nodes. They are likely to have
  6778	        # busted internal references which can cause problems with
  6779	        # serialization and they can't have any external references to
  6780	        # them.
  6781	        symbol.no_serialize = True
  6782	        while True:
  6783	            if i == 1:
  6784	                new_name = f"{name}-redefinition"
  6785	            else:
  6786	                new_name = f"{name}-redefinition{i}"
  6787	            existing = names.get(new_name)
  6788	            if existing is None:
  6789	                names[new_name] = symbol
  6790	                return
  6791	            elif existing.node is symbol.node:
  6792	                # Already there
  6793	                return
  6794	            i += 1

  6926	    def defer(self, debug_context: Context | None = None, force_progress: bool = False) -> None:
  6927	        """Defer current analysis target to be analyzed again.
  6928
  6929	        This must be called if something in the current target is
  6930	        incomplete or has a placeholder node. However, this must *not*
  6931	        be called during the final analysis iteration! Instead, an error
  6932	        should be generated. Often 'process_placeholder' is a good
  6933	        way to either defer or generate an error.
  6934
  6935	        NOTE: Some methods, such as 'anal_type', 'mark_incomplete' and
  6936	              'record_incomplete_ref', call this implicitly, or when needed.
  6937	              They are usually preferable to a direct defer() call.
  6938	        """
  6939	        assert not self.final_iteration, "Must not defer during final iteration"
  6940	        if force_progress:
  6941	            # Usually, we report progress if we have replaced a placeholder node
  6942	            # with an actual valid node. However, sometimes we need to update an
  6943	            # existing node *in-place*. For example, this is used by type aliases
  6944	            # in context of forward references and/or recursive aliases, and in
  6945	            # similar situations (recursive named tuples etc).
  6946	            self.progress = True
  6947	        self.deferred = True
  6948	        # Store debug info for this deferral.
  6949	        line = (
  6950	            debug_context.line if debug_context else self.statement.line if self.statement else -1
  6951	        )
  6952	        self.deferral_debug_context.append((self.cur_mod_id, line))
