    51	    """IR basic block.
    52
    53	    Contains a sequence of Ops and ends with a ControlOp (Goto,
    54	    Branch, Return or Unreachable). Only the last op can be a
    55	    ControlOp.
    56
    57	    All generated Ops live in basic blocks. Basic blocks determine the
    58	    order of evaluation and control flow within a function. A basic
    59	    block is always associated with a single function/method (FuncIR).
    60
    61	    When building the IR, ops that raise exceptions can be included in
    62	    the middle of a basic block, but the exceptions aren't checked.
    63	    Afterwards we perform a transform that inserts explicit checks for
    64	    all error conditions and splits basic blocks accordingly to preserve
    65	    the invariant that a jump, branch or return can only ever appear
    66	    as the final op in a block. Manually inserting error checking ops
    67	    would be boring and error-prone.
    68
    69	    BasicBlocks have an error_handler attribute that determines where
    70	    to jump if an error occurs. If none is specified, an error will
    71	    propagate up out of the function. This is compiled away by the
    72	    `exceptions` module.
    73
    74	    Block labels are used for pretty printing and emitting C code, and
    75	    get filled in by those passes.
    76
    77	    Ops that may terminate the program aren't treated as exits.
    78	    """
    79
    80	    def __init__(self, label: int = -1) -> None:
    81	        self.label = label
    82	        self.ops: list[Op] = []
    83	        self.error_handler: BasicBlock | None = None
    84	        self.referenced = False
    85
    86	    @property
    87	    def terminated(self) -> bool:
    88	        """Does the block end with a jump, branch or return?
    89
    90	        This should always be true after the basic block has been fully built, but
    91	        this is false during construction.
    92	        """
    93	        return bool(self.ops) and isinstance(self.ops[-1], ControlOp)
    94
    95	    @property
    96	    def terminator(self) -> ControlOp:
    97	        """The terminator operation of the block."""
    98	        assert bool(self.ops) and isinstance(self.ops[-1], ControlOp)
    99	        return self.ops[-1]

   103	ERR_NEVER: Final = 0
   104	# Generates magic value (c_error_value) based on target RType on exception
   105	ERR_MAGIC: Final = 1
   106	# Generates false (bool) on exception
   107	ERR_FALSE: Final = 2
   108	# Always fails
   109	ERR_ALWAYS: Final = 3

   111	# an extra PyErr_Occurred() check is also required

   118	class Value:
   119	    """Abstract base class for all IR values.
   120
   121	    These include references to registers, literals, and all
   122	    operations (Ops), such as assignments, calls and branches.

   124	    Values are often used as inputs of Ops. Register can be used as an
   125	    assignment target.

   130	    See also: Op is a subclass of Value that is the base class of all
   131	    operations.
   132	    """

   135	    line = -1
   136	    # Type of the value or the result of the operation
   137	    type: RType = void_rtype

   140	    @property
   141	    def is_void(self) -> bool:
   142	        return isinstance(self.type, RVoid)

   171	class Integer(Value):
   172	    """Short integer literal.
   173
   174	    Integer literals are treated as constant values and are generally
   175	    not included in data flow analyses and such, unlike Register and
   176	    Op subclasses.
   177
   178	    Integer can represent multiple types:
   179
   180	     * Short tagged integers (short_int_primitive type; the tag bit is clear)
   181	     * Ordinary fixed-width integers (e.g., int32_rprimitive)
   182	     * Values of other unboxed primitive types that are represented as integers
   183	       (none_rprimitive, bool_rprimitive)
   184	     * Null pointers (value 0) of various types, including object_rprimitive
   185	    """
   186
   187	    def __init__(self, value: int, rtype: RType = short_int_rprimitive, line: int = -1) -> None:
   188	        if is_short_int_rprimitive(rtype) or is_int_rprimitive(rtype):
   189	            self.value = value * 2
   190	        else:
   191	            self.value = value
   192	        self.type = rtype
   193	        self.line = line
   194
   195	    def numeric_value(self) -> int:
   196	        if is_short_int_rprimitive(self.type) or is_int_rprimitive(self.type):
   197	            return self.value // 2
   198	        return self.value

   215	class Op(Value):
   216	    """Abstract base class for all IR operations.
   217
   218	    Each operation must be stored in a BasicBlock (in 'ops') to be
   219	    active in the IR. This is different from non-Op values, including
   220	    Register and Integer, where a reference from an active Op is
   221	    sufficient to be considered active.
   222
   223	    In well-formed IR an active Op has no references to inactive ops
   224	    or ops used in another function.
   225	    """
   226
   227	    def __init__(self, line: int) -> None:
   228	        self.line = line
   229
   230	    def can_raise(self) -> bool:
   231	        # Override this is if Op may raise an exception. Note that currently the fact that
   232	        # only RegisterOps may raise an exception in hard coded in some places.
   233	        return False
   234
   235	    @abstractmethod
   236	    def sources(self) -> list[Value]:
   237	        """All the values the op may read."""
   238
   239	    @abstractmethod
   240	    def set_sources(self, new: list[Value]) -> None:
   241	        """Rewrite the sources of an op"""
   242
   243	    def stolen(self) -> list[Value]:
   244	        """Return arguments that have a reference count stolen by this op"""
   245	        return []
   246
   247	    def unique_sources(self) -> list[Value]:
   248	        result: list[Value] = []
   249	        for reg in self.sources():
   250	            if reg not in result:
   251	                result.append(reg)
   252	        return result
   253
   254	    @abstractmethod
   255	    def accept(self, visitor: OpVisitor[T]) -> T:
   256	        pass

   363	class Branch(ControlOp):
   364	    """Branch based on a value.
   365
   366	    If op is BOOL, branch based on a bit/bool value:
   367	       if [not] r1 goto L1 else goto L2
   368
   369	    If op is IS_ERROR, branch based on whether there is an error value:
   370	       if [not] is_error(r1) goto L1 else goto L2
   371	    """
   372
   373	    # Branch ops never raise an exception.
   374	    error_kind = ERR_NEVER
   375
   376	    BOOL: Final = 100
   377	    IS_ERROR: Final = 101
   378
   379	    def __init__(
   380	        self,
   381	        value: Value,
   382	        true_label: BasicBlock,
   383	        false_label: BasicBlock,
   384	        op: int,
   385	        line: int = -1,
   386	        *,
   387	        rare: bool = False,
   388	    ) -> None:
   389	        super().__init__(line)
   390	        # Target value being checked
   391	        self.value = value
   392	        # Branch here if the condition is true
   393	        self.true = true_label
   394	        # Branch here if the condition is false
   395	        self.false = false_label
   396	        # Branch.BOOL (boolean check) or Branch.IS_ERROR (error value check)
   397	        self.op = op
   398	        # If True, the condition is negated
   399	        self.negated = False
   400	        # If not None, the true label should generate a traceback entry (func name, line number)
   401	        self.traceback_entry: tuple[str, int] | None = None
   402	        # If True, we expect to usually take the false branch (for optimization purposes);
   403	        # this is implicitly treated as true if there is a traceback entry
   404	        self.rare = rare
   405
   406	    def targets(self) -> Sequence[BasicBlock]:
   407	        return (self.true, self.false)
   408
   409	    def set_target(self, i: int, new: BasicBlock) -> None:
   410	        assert i == 0 or i == 1
   411	        if i == 0:
   412	            self.true = new
   413	        else:
   414	            self.false = new
   415
   416	    def sources(self) -> list[Value]:
   417	        return [self.value]
   418
   419	    def set_sources(self, new: list[Value]) -> None:
   420	        (self.value,) = new
   421
   422	    def invert(self) -> None:
   423	        self.negated = not self.negated
   424
   425	    def accept(self, visitor: OpVisitor[T]) -> T:
   426	        return visitor.visit_branch(self)
