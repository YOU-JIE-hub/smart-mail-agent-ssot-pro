   303	def cmyk2rgb(cmyk,density=1):
   304	    "Convert from a CMYK color tuple to an RGB color tuple"
   305	    c,m,y,k = cmyk
   306	    # From the Adobe Postscript Ref. Manual 2nd ed.
   307	    r = 1.0 - min(1.0, c + k)
   308	    g = 1.0 - min(1.0, m + k)
   309	    b = 1.0 - min(1.0, y + k)
   310	    return (r,g,b)

   312	def rgb2cmyk(r,g,b):
   313	    '''one way to get cmyk from rgb'''
   314	    c = 1 - r
   315	    m = 1 - g
   316	    y = 1 - b
   317	    k = min(c,m,y)
   318	    c = min(1,max(0,c-k))
   319	    m = min(1,max(0,m-k))
   320	    y = min(1,max(0,y-k))
   321	    k = min(1,max(0,k))
   322	    return (c,m,y,k)

   392	def linearlyInterpolatedColor(c0, c1, x0, x1, x):
   393	    """
   394	    Linearly interpolates colors. Can handle RGB, CMYK and PCMYK
   395	    colors - give ValueError if colours aren't the same.
   396	    Doesn't currently handle 'Spot Color Interpolation'.
   397	    """
   398
   399	    if c0.__class__ != c1.__class__:
   400	        raise ValueError("Color classes must be the same for interpolation!\nGot %r and %r'"%(c0,c1))
   401	    if x1<x0:
   402	        x0,x1,c0,c1 = x1,x0,c1,c0 # normalized so x1>x0
   403	    if x<x0-1e-8 or x>x1+1e-8: # fudge factor for numerical problems
   404	        raise ValueError("Can't interpolate: x=%f is not between %f and %f!" % (x,x0,x1))
   405	    if x<=x0:

   452	            m = c0.magenta+x*(c1.magenta - c0.magenta)/dx
