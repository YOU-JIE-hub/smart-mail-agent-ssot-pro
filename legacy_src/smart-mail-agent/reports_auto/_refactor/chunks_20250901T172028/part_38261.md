  2171	        fill_typevars = act_len != max_tv_count
  2172	        correct = min_tv_count <= act_len <= max_tv_count
  2173	    else:

  2196	                        f" expected at least {min_tv_count}, given {act_len}"

  2199	                    msg = (
  2200	                        "Bad number of arguments for type alias,"
  2201	                        f" expected between {min_tv_count} and {max_tv_count}, given {act_len}"
  2202	                    )
  2203	                else:
  2204	                    msg = (
  2205	                        "Bad number of arguments for type alias,"
  2206	                        f" expected {min_tv_count}, given {act_len}"
  2207	                    )
  2208	            fail(msg, ctx, code=codes.TYPE_ARG)
  2209	            args = []
  2210	        return set_any_tvars(node, args, ctx.line, ctx.column, options, from_error=True)
  2211	    elif node.tvar_tuple_index is not None:
  2212	        # We also need to check if we are not performing a type variable tuple split.
  2213	        unpack = find_unpack_in_list(args)

  2216	            assert isinstance(unpack_arg, UnpackType)

  2225	    # TODO: we need to check args validity w.r.t alias.alias_tvars.
  2226	    # Otherwise invalid instantiations will be allowed in runtime context.
  2227	    # Note: in type context, these will be still caught by semanal_typeargs.
  2228	    typ = TypeAliasType(node, args, ctx.line, ctx.column)
  2229	    assert typ.alias is not None
  2230	    # HACK: Implement FlexibleAlias[T, typ] by expanding it to typ here.
  2231	    if (
  2232	        isinstance(typ.alias.target, Instance)  # type: ignore[misc]
  2233	        and typ.alias.target.type.fullname == "mypy_extensions.FlexibleAlias"

  2343	def detect_diverging_alias(
  2344	    node: TypeAlias,
  2345	    target: Type,
  2346	    lookup: Callable[[str, Context], SymbolTableNode | None],
  2347	    scope: TypeVarLikeScope,

  2349	    """This detects type aliases that will diverge during type checking.
  2350
  2351	    For example F = Something[..., F[List[T]]]. At each expansion step this will produce
  2352	    *new* type aliases: e.g. F[List[int]], F[List[List[int]]], etc. So we can't detect
  2353	    recursion. It is a known problem in the literature, recursive aliases and generic types

  2358	    Nested = Union[T, Nested[List[T]]] ~> Union[T, List[T], List[List[T]], ...]
  2359	    """
  2360	    visitor = DivergingAliasDetector({node}, lookup, scope)
  2361	    _ = target.accept(visitor)
  2362	    return visitor.diverging
