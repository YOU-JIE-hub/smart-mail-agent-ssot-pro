ss(shell)
    40
    41	    if comp_cls is None:
    42	        return 1
    43
    44	    comp = comp_cls(cli, ctx_args, prog_name, complete_var)
    45
    46	    if instruction == "source":
    47	        echo(comp.source())
    48	        return 0
    49
    50	    if instruction == "complete":
    51	        echo(comp.complete())
    52	        return 0
    53
    54	    return 1

    57	class CompletionItem:
    58	    """Represents a completion value and metadata about the value. The
    59	    default metadata is ``type`` to indicate special shell handling,
    60	    and ``help`` if a shell supports showing a help string next to the
    61	    value.
    62
    63	    Arbitrary parameters can be passed when creating the object, and
    64	    accessed using ``item.attr``. If an attribute wasn't passed,
    65	    accessing it returns ``None``.
    66
    67	    :param value: The completion suggestion.
    68	    :param type: Tells the shell script to provide special completion
    69	        support for the type. Click uses ``"dir"`` and ``"file"``.
    70	    :param help: String shown next to the value if supported.
    71	    :param kwargs: Arbitrary metadata. The built-in implementations
    72	        don't use this, but custom type completions paired with custom
    73	        shell support could use it.
    74	    """
    75
    76	    __slots__ = ("value", "type", "help", "_info")
    77
    78	    def __init__(
    79	        self,
    80	        value: t.Any,
    81	        type: str = "plain",
    82	        help: str | None = None,
    83	        **kwargs: t.Any,
    84	    ) -> None:
    85	        self.value: t.Any = value
    86	        self.type: str = type
    87	        self.help: str | None = help
    88	        self._info = kwargs
    89
    90	    def __getattr__(self, name: str) -> t.Any:
    91	        return self._info.get(name)

   194	class ShellComplete:
   195	    """Base class for providing shell completion support. A subclass for
   196	    a given shell will override attributes and methods to implement the
   197	    completion instructions (``source`` and ``complete``).
   198
   199	    :param cli: Command being called.
   200	    :param prog_name: Name of the executable in the shell.
   201	    :param complete_var: Name of the environment variable that holds
   202	        the completion instruction.
   203
   204	    .. versionadded:: 8.0
   205	    """
   206
   207	    name: t.ClassVar[str]
   208	    """Name to register the shell as with :func:`add_completion_class`.
   209	    This is used in completion instructions (``{name}_source`` and
   210	    ``{name}_complete``).
   211	    """
   212
   213	    source_template: t.ClassVar[str]
   214	    """Completion script template formatted by :meth:`source`. This must
   215	    be provided by subclasses.
   216	    """
   217
   218	    def __init__(
   219	        self,
   220	        cli: Command,
   221	        ctx_args: cabc.MutableMapping[str, t.Any],
   222	        prog_name: str,
   223	        complete_var: str,
   224	    ) -> None:
   225	        self.cli = cli
   226	        self.ctx_args = ctx_args
   227	        self.prog_name = prog_name
   228	        self.complete_var = complete_var
   229
   230	    @property
   231	    def func_name(self) -> str:
   232	        """The name of the shell function defined by the completion
   233	        script.
   234	        """
   235	        safe_name = re.sub(r"\W*", "", self.prog_name.replace("-", "_"), flags=re.ASCII)
   236	        return f"_{safe_name}_completion"
   237
   238	    def source_vars(self) -> dict[str, t.Any]:
   239	        """Vars for formatting :attr:`source_template`.
   240
   241	        By default this provides ``complete_func``, ``complete_var``,
   242	        and ``prog_name``.
   243	        """
   244	        return {
   245	            "complete_func": self.func_name,
   246	            "complete_var": self.complete_var,
   247	            "prog_name": self.prog_name,
   248	        }
   249
   250	    def source(self) -> str:
   251	        """Produce the shell script that defines the completion
   252	        function. By default this ``%``-style formats
   253	        :attr:`source_template` with the dict returned by
   254	        :meth:`source_vars`.
   255	        """
   256	        return self.source_template % self.source_vars()
   257
   258	    def get_completion_args(self) -> tuple[list[str], str]:
   259	        """Use the env vars defined by the shell script to return a
   260	        tuple of ``args, incomplete``. This must be implemented by
   261	        subclasses.
   262	        """
   263	        raise NotImplementedError
   264
   265	    def get_completions(self, args: list[str], incomplete: str) -> list[CompletionItem]:
   266	        """Determine the context and last complete command or parameter
   267	        from the complete args. Call that object's ``shell_complete``
   268	        method to get the completions for the incomplete value.
   269
   270	        :param args: List of complete args before the incomplete value.
   271	        :param incomplete: Value being completed. May be empty.
   272	        """
   273	        ctx = _resolve_context(self.cli, self.ctx_args, self.prog_name, args)
   274	        obj, incomplete = _resolve_incomplete(ctx, args, incomplete)
   275	        return obj.shell_complete(ctx, incomplete)
   276
   277	    def format_completion(self, item: CompletionItem) -> str:
   278	        """Format a completion item into the form recognized by the
   279	        shell script. This must be implemented by subclasses.
   280
   281	        :param item: Completion item to format.
   282	        """
   283	        raise NotImplementedError
   284
   285	    def complete(self) -> str:
   286	        """Produce the completion data to send back to the shell.
   287
   288	        By default this calls :meth:`get_completion_args`, gets the
   289	        completions, then calls :meth:`format_completion` for each
   290	        completion.
   291	        """
   292	        args, incomplete = self.get_completion_args()
   293	        completions = self.get_completions(args, incomplete)
   294	        out = [self.format_completion(item) for item in completions]
   295	        return "\n".join(out)

   444	def split_arg_string(string: str) -> list[str]:
   445	    """Split an argument string as with :func:`shlex.split`, but don't
   446	    fail if the string is incomplete. Ignores a missing closing quote or
   447	    incomplete escape sequence and uses the partial token as-is.
   448
   449	    .. code-block:: python
   450
   451	        split_arg_string("example 'my file")
   452	        ["example", "my file"]
   453
   454	        split_arg_string("example my\\")
   455	        ["example", "my"]
   456
   457	    :param string: String to split.
   458
   459	    .. versionchanged:: 8.2
   460	        Moved to ``shell_completion`` from ``parser``.
   461	    """
   462	    import shlex
   463
   464	    lex = shlex.shlex(string, posix=True)
   465	    lex.whitespace_split = True
   466	    lex.commenters = ""
   467	    out = []
   468
   469	    try:
   470	        for token in lex:
   471	            out.append(token)
   472	    except ValueError:
   473	        # Raised when end-of-string is reached in an invalid state. Use
   474	        # the partial token as-is. The quote or escape character is in
   475	        # lex.state, not lex.token.
   476	        out.append(lex.token)
   477
   478	    return out

   482	    """Determine if the given parameter is an argument that can still
   483	    accept values.
   484
   485	    :param ctx: Invocation context for the command represented by the

   498	        or (

   500	            and isinstance(value, (tuple, list))
   501	            and len(value) < param.nargs

   515	def _is_incomplete_option(ctx: Context, args: list[str], param: Parameter) -> bool:
   516	    """Determine if the given parameter is an option that needs a value.
   517
   518	    :param args: List of complete args before the incomplete value.
   519	    :param param: Option object being checked.
   520	    """
   521	    if not isinstance(param, Option):
   522	        return False
   523
   524	    if param.is_flag or param.count:
   525	        return False
   526
   527	    last_option = None
   528
   529	    for index, arg in enumerate(reversed(args)):
   530	        if index + 1 > param.nargs:
   531	            break
   532
   533	        if _start_of_option(ctx, arg):
   534	            last_option = arg
   535
   536	    return last_option is not None and last_option in param.opts

   539	def _resolve_context(
   540	    cli: Command,
   541	    ctx_args: cabc.MutableMapping[str, t.Any],
   542	    prog_name: str,
   543	    args: list[str],
   544	) -> Context:
   545	    """Produce the context hierarchy starting with the command and
   546	    traversing the complete arguments. This only follows the commands,
   547	    it doesn't trigger input prompts or callbacks.
   548
   549	    :param cli: Command being called.
   550	    :param prog_name: Name of the executable in the shell.
   551	    :param args: List of complete args before the incomplete value.
   552	    """
   553	    ctx_args["resilient_parsing"] = True
   554	    with cli.make_context(prog_name, args.copy(), **ctx_args) as ctx:
   555	        args = ctx._protected_args + ctx.args
   556
   557	        while args:
   558	            command = ctx.command
   559
   560	            if isinstance(command, Group):
   561	                if not command.chain:
   562	                    name, cmd, args = command.resolve_command(ctx, args)
   563
   564	                    if cmd is None:
   565	                        return ctx
   566
   567	                    with cmd.make_context(
   568	                        name, args, parent=ctx, resilient_parsing=True
   569	                    ) as sub_ctx:
   570	                        ctx = sub_ctx
   571	                        args = ctx._protected_args + ctx.args
   572	                else:
   573	                    sub_ctx = ctx
   574
   575	                    while args:
   576	                        name, cmd, args = command.resolve_command(ctx, args)
   577
   578	                        if cmd is None:
   579	                            return ctx
   580
   581	                        with cmd.make_context(
   582	                            name,
   583	                            args,
   584	                            parent=ctx,
   585	                            allow_extra_args=True,
   586	                            allow_interspersed_args=False,
   587	                            resilient_parsing=True,
   588	                        ) as sub_sub_ctx:
   589	                            sub_ctx = sub_sub_ctx
   590	                            args = sub_ctx.args
   591
   592	                    ctx = sub_ctx
   593	                    args = [*sub_ctx._protected_args, *sub_ctx.args]
   594	            else:
   59
