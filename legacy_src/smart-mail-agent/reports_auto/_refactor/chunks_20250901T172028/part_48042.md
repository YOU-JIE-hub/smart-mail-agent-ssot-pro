   394	    def pretty_name(
   395	        self,
   396	        name: str,
   397	        kind: int | None,
   398	        fullname: str | None,
   399	        is_inferred_def: bool,
   400	        target_node: mypy.nodes.Node | None = None,
   401	    ) -> str:
   402	        n = name
   403	        if is_inferred_def:
   404	            n += "*"
   405	        if target_node:
   406	            id = self.format_id(target_node)
   407	        else:
   408	            id = ""
   409	        if isinstance(target_node, mypy.nodes.MypyFile) and name == fullname:
   410	            n += id
   411	        elif kind == mypy.nodes.GDEF or (fullname != name and fullname):
   412	            # Append fully qualified name for global references.
   413	            n += f" [{fullname}{id}]"
   414	        elif kind == mypy.nodes.LDEF:
   415	            # Add tag to signify a local reference.
   416	            n += f" [l{id}]"
   417	        elif kind == mypy.nodes.MDEF:
   418	            # Add tag to signify a member reference.
   419	            n += f" [m{id}]"
   420	        else:
   421	            n += id
   422	        return n
