nd self.is_rewrite_disabled(doc):
   703	            return
   704	        pos = 0
   705	        item = None
   706	        for item in mod.body:
   707	            if (
   708	                expect_docstring
   709	                and isinstance(item, ast.Expr)
   710	                and isinstance(item.value, ast.Constant)
   711	                and isinstance(item.value.value, str)
   712	            ):
   713	                doc = item.value.value
   714	                if self.is_rewrite_disabled(doc):
   715	                    return
   716	                expect_docstring = False
   717	            elif (
   718	                isinstance(item, ast.ImportFrom)
   719	                and item.level == 0
   720	                and item.module == "__future__"
   721	            ):
   722	                pass
   723	            else:
   724	                break
   725	            pos += 1
   726	        # Special case: for a decorated function, set the lineno to that of the
   727	        # first decorator, not the `def`. Issue #4984.
   728	        if isinstance(item, ast.FunctionDef) and item.decorator_list:
   729	            lineno = item.decorator_list[0].lineno
   730	        else:
   731	            lineno = item.lineno
   732	        # Now actually insert the special imports.
   733	        if sys.version_info >= (3, 10):
   734	            aliases = [
   735	                ast.alias("builtins", "@py_builtins", lineno=lineno, col_offset=0),
   736	                ast.alias(
   737	                    "_pytest.assertion.rewrite",
   738	                    "@pytest_ar",
   739	                    lineno=lineno,
   740	                    col_offset=0,
   741	                ),
   742	            ]
   743	        else:
   744	            aliases = [
   745	                ast.alias("builtins", "@py_builtins"),
   746	                ast.alias("_pytest.assertion.rewrite", "@pytest_ar"),
   747	            ]
   748	        imports = [
   749	            ast.Import([alias], lineno=lineno, col_offset=0) for alias in aliases
   750	        ]
   751	        mod.body[pos:pos] = imports
   752
   753	        # Collect asserts.
   754	        self.scope = (mod,)
   755	        nodes: list[ast.AST | Sentinel] = [mod]
   756	        while nodes:
   757	            node = nodes.pop()
   758	            if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
   759	                self.scope = tuple((*self.scope, node))
   760	                nodes.append(_SCOPE_END_MARKER)
   761	            if node == _SCOPE_END_MARKER:
   762	                self.scope = self.scope[:-1]
   763	                continue
   764	            assert isinstance(node, ast.AST)
   765	            for name, field in ast.iter_fields(node):
   766	                if isinstance(field, list):
   767	                    new: list[ast.AST] = []
   768	                    for i, child in enumerate(field):
   769	                        if isinstance(child, ast.Assert):
   770	                            # Transform assert.
   771	                            new.extend(self.visit(child))
   772	                        else:
   773	                            new.append(child)
   774	                            if isinstance(child, ast.AST):
   775	                                nodes.append(child)
   776	                    setattr(node, name, new)
   777	                elif (
   778	                    isinstance(field, ast.AST)
   779	                    # Don't recurse into expressions as they can't contain
   780	                    # asserts.
   781	                    and not isinstance(field, ast.expr)
   782	                ):
   783	                    nodes.append(field)

   817	    def explanation_param(self, expr: ast.expr) -> str:
   818	        """Return a new named %-formatting placeholder for expr.
   819
   820	        This creates a %-formatting placeholder for expr in the
   821	        current formatting context, e.g. ``%(py0)s``.  The placeholder
   822	        and expr are placed in the current format context so that it
   823	        can be used on the next call to .pop_format_context().
   824	        """
   825	        specifier = "py" + str(next(self.variable_counter))
   826	        self.explanation_specifiers[specifier] = expr
   827	        return "%(" + specifier + ")s"

   829	    def push_format_context(self) -> None:
   830	        """Create a new formatting context.
   831
   832	        The format context is used for when an explanation wants to
   833	        have a variable value formatted in the assertion message.  In
   834	        this case the value required can be added using
   835	        .explanation_param().  Finally .pop_format_context() is used
   836	        to format a string of %-formatted values as added by
   837	        .explanation_param().
   838	        """
   839	        self.explanation_specifiers: dict[str, ast.expr] = {}
   840	        self.stack.append(self.explanation_specifiers)
   841
   842	    def pop_format_context(self, expl_expr: ast.expr) -> ast.Name:
   843	        """Format the %-formatted string with current format context.
   844
   845	        The expl_expr should be an str ast.expr instance constructed from
   846	        the %-placeholders created by .explanation_param().  This will
   847	        add the required code to format said string to .expl_stmts and
   848	        return the ast.Name instance of the formatted string.
   849	        """
   850	        current = self.stack.pop()
   851	        if self.stack:
   852	            self.explanation_specifiers = self.stack[-1]
   853	        keys: list[ast.expr | None] = [ast.Constant(key) for key in current.keys()]
   854	        format_dict = ast.Dict(keys, list(current.values()))
   855	        form = ast.BinOp(expl_expr, ast.Mod(), format_dict)
   856	        name = "@py_format" + str(next(self.variable_counter))
   857	        if self.enable_assertion_pass_hook:
   858	            self.format_variables.append(name)
   859	        self.expl_stmts.append(ast.Assign([ast.Name(name, ast.Store())], form))
   860	        return ast.Name(name, ast.Load())

   864	        assert isinstance(node, ast.expr)

   866	        return res, self.explanation_param(self.display(res))

   868	    def visit_Assert(self, assert_: ast.Assert) -> list[ast.stmt]:
   869	        """Return the AST statements to replace the ast.Assert instance.
   870
   871	        This rewrites the test of an assertion to provide
   872	        intermediate values and replace it with an if statement which
   873	        raises an assertion error with a detailed explanation in case
   874	        the expression is false.
   875	        """
   876	        if isinstance(assert_.test, ast.Tuple) and len(assert_.test.elts) >= 1:
   877	            import warnings
   878
   879	            from _pytest.warning_types import PytestAssertRewriteWarning
   880
   881	            # TODO: This assert should not be needed.
   882	            assert self.module_path is not None
   883	            warnings.warn_explicit(
   884	                PytestAssertRewriteWarning(
   885	                    "assertion is always true, perhaps remove parentheses?"
   886	                ),
   887	                category=None,
   888	                filename=self.module_path,
   889	                lineno=assert_.lineno,
   890	            )
   891
   892	        self.statements: list[ast.stmt] = []
   893	        self.variables: list[str] = []
   894	        self.variable_counter = itertools.count()
   895
   896	        if self.enable_assertion_pass_hook:
   897	            self.format_variables: list[str] = []
   898
   899	        self.stack: list[dict[str, ast.expr]] = []
   900	        self.expl_stmts: list[ast.stmt] = []
   901	        self.push_format_context()
   902	        # Rewrite assert into a bunch of statements.
   903	        top_condition, explanation = self.visit(assert_.test)
   904
   905	        negation = ast.UnaryOp(ast.Not(), top_condition)
   906
   907	        if self.enable_assertion_pass_hook:  # Experimental pytest_assertion_pass hook
   908	            msg = self.pop_format_context(ast.Constant(explanation))
   909
   910	            # Failed
   911	            if assert_.msg:
   912	                assertmsg = self.helper("_format_assertmsg", assert_.msg)
   913	                gluestr = "\n>assert "
   914	            else:
   915	                assertmsg = ast.Constant("")
   916	                gluestr = "assert "
   917	            err_explanation = ast.BinOp(ast.Constant(gluestr), ast.Add(), msg)
   918	            err_msg = ast.BinOp(assertmsg, ast.Add(), err_explanation)
   919	            err_name = ast.Name("AssertionError", ast.Load())
   920	            fmt = self.helper("_format_explanation", err_msg)
   921	            exc = ast.Call(err_name, [fmt], [])
   922	            raise_ = ast.Raise(exc, None)
   923	            statements_fail = []
   924	            statements_fail.extend(self.expl_stmts)
   925	            statements_fail.append(raise_)
   926
   927	            # Passed
   928	            fmt_pass = self.helper("_format_explanation", msg)
   929	            orig = _get_assertion_exprs(self.source)[assert_.lineno]
   930	            hook_call_pass = ast.Expr(
   931	                self.helper(
   932	                    "_call_assertion_pass",
   933	                    ast.Constant(assert_.lineno),
   934	                    ast.Constant(orig),
   935	                    fmt_pass,
   936	                )
   937	            )
   938	            # If any hooks implement assert_pass hook
   939	            hook_impl_test = ast.If(
   940	                self.helper("_check_if_assertion_pass_impl"),
   941	                [*self.expl_stmts, hook_call_pass],
   942	                [],
   943	            )
   944	            statements_pass: list[ast.stmt] = [hook_impl_test]
   945
   946	            # Test for assertion condition
   947	            main_test = ast.If(negation, statements_fail, statements_pass)
   948	            self.statements.append(main_test)
   949	            if self.format_variables:
   950	                variables: list[ast.expr] = [
   951	                    ast.Name(name, ast.Store()) for name in self.format_variables
   952	                ]
   953	                clear_format = ast.Assign(variables, ast.Constant(None))
   954	                self.statements.append(clear_format)
   955
   956	        else:  # Original assertion rewriting
   957	            # Create failure message.
   958	            body = self.expl_stmts
   959	            self.statements.append(ast.If(negation, body, []))
   960	            if assert_.msg:
   961	                assertmsg = self.helper("_format_assertmsg", assert_.msg)
   962	                explanation = "\n>assert " + explanation
   963	            else:
   964	                assertmsg = ast.Constant("")
   965	                explanation = "assert " + explanation
   966	            template = ast.BinOp(assertmsg, ast.Add(), ast.Constant(explanation))
   967	            msg = self.pop_format_context(template)
   968	            fmt = self.helper("_format_explanation", msg)
   969	            err_name = ast.Name("AssertionError", ast.Load())
   970	            exc = ast.Call(err_name, [fmt], [])
   971	            raise_ = ast.Raise(exc, None)
   972
   973	            body.append(raise_)
   974
   975	        # Clear temporary variables by setting them to None.
   976	        if self.variables:
   977	            variables = [ast.Name(name, ast.Store()) for name in self.variables]
   978	            clear = ast.Assign(variables, ast.Constant(None))
   979	            self.statements.append(clear)
   980	        # Fix locations (line numbers/column offsets).
   981	        for stmt in self.statements:
   982	            for node in traverse_node(stmt):
   983	                if getattr(node, "lineno", None) is None:
   984	                    # apply the assertion location to all generated ast nodes without source location
   985	                    # and preserve the location of existing nodes or generated nodes with an correct location.
   986	                    ast.copy_location(node, assert_)
   987	        return self.statements

   989	    def visit_NamedExpr(self, name: ast.NamedExpr) -> tuple[ast.NamedExpr, str]:
   990	        # This method handles the 'walrus operator' repr of the target
   991	        # name if it's a local variable or _should_repr_global_name()
   992	        # thinks it's acceptable.
   993	        locs = ast.Call(self.builtin("locals"), [], [])
   994	        target_id = name.target.id
   995	        inlocs = ast.Compare(ast.Constant(target_id), [ast.In()], [locs])
   996	        dorepr = self.helper("_should_repr_global_name", name)
   997	        test = ast.BoolOp(ast.Or(), [inlocs, dorepr])
   998	        expr = ast.IfExp(test, self.display(name), ast.Constant(target_id))
   999	        return name, self.explanation_param(expr)

  1011	    def visit_BoolOp(self, boolop: ast.BoolOp) -> tuple[ast.Name, str]:
  1012	        res_var = self.variable()
  1013	        expl_list = self.assign(ast.List([], ast.Load()))
  1014	        app = ast.Attribute(expl_list, "append", ast.Load())
  1015	        is_or = int(isinstance(boolop.op, ast.Or))
  1016	        body = save = self.statements
  1017	        fail_save = self.expl_stmts
  1018	        levels = len(boolop.values) - 1
  1019	        self.push_format_context()
  1020	        # Process each operand, short-circuiting if needed.
  1021	        for i, v in enumerate(boolop.values):
  1022	            if i:
  1023	                fail_inner: list[ast.stmt] = []
  1024	                # cond is set in a prior loop iteration below
  1025	                self.expl_stmts.append(ast.If(cond, fail_inner, []))  # noqa: F821
  1026	                self.expl_stmts = fail_inner
  1027	                # Check if the left operand is a ast.NamedExpr and the value has already been visited
  1028	                if (
  1029	                    isinstance(v, ast.Compare)
  1030	                    and isinstance(v.left, ast.NamedExpr)
  1031	                    and v.left.target.id
  1032	                    in [
  1033	                        ast_expr.id
  1034	                        for ast_expr in boolop.values[:i]
  1035	                        if hasattr(ast_expr, "id")
  1036	                    ]
  1037	                ):
  1038	                    pytest_temp = self.variable()
  1039	                    self.variables_overwrite[self.scope][v.left.target.id] = v.left  # type:ignore[assignment]
  1040	                    v.left.target.id = pytest_temp
  1041	            self.push_format_context()
  1042	            res, expl = self.visit(v)
  1043	            body.append(ast.Assign([ast.Name(res_var, ast.Store())], res))
  1044	            expl_format = self.pop_format_context(ast.Constant(expl))
  1045	            call = ast.Call(app, [expl_format], [])
  1046	            self.expl_stmts.append(ast.Expr(call))
  1047	            if i < levels:
  1048	                cond: ast.expr = res
  1049	                if is_or:
  1050	                    cond = ast.UnaryOp(ast.Not(), cond)
  1051	                inner: list[ast.stmt] = []
  1052	                self.statements.append(ast.If(cond, inner, []))
  1053	                self.statements = body = inner
  1054	        self.statements = save
  1055	        self.expl_stmts = fail_save
  1056	        expl_template = self.helper("_format_boolop", expl_list, ast.Constant(is_or))
  1057	        expl = self.pop_format_context(expl_template)
  1058	        return ast.Name(res_var, ast.Load()), self.explanation_param(expl)
