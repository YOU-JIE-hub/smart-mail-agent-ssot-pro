   502	def _repr(self,I=None):
   503	    '''return a repr style string with named fixed args first, then keywords'''
   504	    if isinstance(self,float):
   505	        return fp_str(self)
   506	    elif isSeq(self):
   507	        s = ''
   508	        for v in self:
   509	            s = s + '%s,' % _repr(v,I)
   510	        if isinstance(self,list):
   511	            return '[%s]' % s[:-1]
   512	        else:
   513	            return '(%s%s)' % (s[:-1],len(self)==1 and ',' or '')
   514	    elif self is EmptyClipPath:
   515	        if I: _addObjImport(self,I,'EmptyClipPath')
   516	        return 'EmptyClipPath'
   517	    elif isinstance(self,Shape):
   518	        if I: _addObjImport(self,I)
   519	        from inspect import getfullargspec
   520	        args, varargs, varkw, defaults, kwonlyargs, kwonlydefaults, annotations = getfullargspec(self.__init__)
   521	        if defaults:
   522	            kargs = args[-len(defaults):]
   523	            del args[-len(defaults):]
   524	        else:
   525	            kargs = []
   526	        P = self.getProperties()
   527	        s = self.__class__.__name__+'('
   528	        for n in args[1:]:
   529	            v = P.pop(n,None)
   530	            s += '%s,' % _repr(v,I)
   531	        for n in kargs:
   532	            v = P.pop(n,None)
   533	            s += '%s=%s,' % (n,_repr(v,I))
   534	        for n,v in P.items():
   535	            v = P[n]
   536	            s += '%s=%s,' % (n, _repr(v,I))
   537	        return s[:-1]+')'
   538	    else:
   539	        return repr(self)
