   423	def any_all_helper(
   424	    builder: IRBuilder,
   425	    gen: GeneratorExpr,
   426	    initial_value: Callable[[], Value],
   427	    modify: Callable[[Value], Value],
   428	    new_value: Callable[[], Value],
   429	) -> Value:
   430	    retval = Register(bool_rprimitive)
   431	    builder.assign(retval, initial_value(), -1)
   432	    loop_params = list(zip(gen.indices, gen.sequences, gen.condlists, gen.is_async))
   433	    true_block, false_block, exit_block = BasicBlock(), BasicBlock(), BasicBlock()
   434
   435	    def gen_inner_stmts() -> None:
   436	        comparison = modify(builder.accept(gen.left_expr))
   437	        builder.add_bool_branch(comparison, true_block, false_block)

   440	        builder.goto(exit_block)
   441	        builder.activate_block(false_block)
   442
   443	    comprehension_helper(builder, loop_params, gen_inner_stmts, gen.line)
   444	    builder.goto_and_activate(exit_block)
   445
   446	    return retval

   451	    # specialized implementation is used if:
   452	    # - only one or two arguments given (if not, sum() has been given invalid arguments)
   453	    # - first argument is a Generator (there is no benefit to optimizing the performance of eg.
   454	    #   sum([1, 2, 3]), so non-Generator Iterables are not handled)

   463	    if len(expr.args) == 2:

   465	        if expr.arg_kinds[1] not in (ARG_POS, ARG_NAMED):

   468	    else:
   469	        start_expr = IntExpr(0)

   473	    retval = Register(target_type)

   505	@specialize_function("builtins.next")
   506	def translate_next_call(builder: IRBuilder, expr: CallExpr, callee: RefExpr) -> Value | None:
   507	    """Special case for calling next() on a generator expression, an
   508	    idiom that shows up some in mypy.
   509
   510	    For example, next(x for x in l if x.id == 12, None) will
   511	    generate code that searches l for an element where x.id == 12
   512	    and produce the first such object, or None if no such element
   513	    exists.
   514	    """
   515	    if not (
   516	        expr.arg_kinds in ([ARG_POS], [ARG_POS, ARG_POS])
   517	        and isinstance(expr.args[0], GeneratorExpr)
   518	    ):
   519	        return None
   520
   521	    gen = expr.args[0]
   522	    retval = Register(builder.node_type(expr))
   523	    default_val = builder.accept(expr.args[1]) if len(expr.args) > 1 else None
   524	    exit_block = BasicBlock()
   525
   526	    def gen_inner_stmts() -> None:
   527	        # next takes the first element of the generator, so if
   528	        # something gets produced, we are done.
   529	        builder.assign(retval, builder.accept(gen.left_expr), gen.left_expr.line)
   530	        builder.goto(exit_block)
   531
   532	    loop_params = list(zip(gen.indices, gen.sequences, gen.condlists, gen.is_async))
   533	    comprehension_helper(builder, loop_params, gen_inner_stmts, gen.line)
   534
   535	    # Now we need the case for when nothing got hit. If there was
   536	    # a default value, we produce it, and otherwise we raise
   537	    # StopIteration.
   538	    if default_val:
   539	        builder.assign(retval, default_val, gen.left_expr.line)
   540	        builder.goto(exit_block)
   541	    else:
   542	        builder.add(RaiseStandardError(RaiseStandardError.STOP_ITERATION, None, expr.line))
   543	        builder.add(Unreachable())
   544
   545	    builder.activate_block(exit_block)
   546	    return retval

   642	def translate_fstring(builder: IRBuilder, expr: CallExpr, callee: RefExpr) -> Value | None:
   643	    """Special case for f-string, which is translated into str.join()
   644	    in mypy AST.
   645
   646	    This specializer optimizes simplest f-strings which don't contain
   647	    any format operation.
   648	    """
   649	    if (
   650	        isinstance(callee, MemberExpr)
   651	        and isinstance(callee.expr, StrExpr)
   652	        and callee.expr.value == ""
   653	        and expr.arg_kinds == [ARG_POS]
   654	        and isinstance(expr.args[0], ListExpr)
   655	    ):

   657	            if isinstance(item, StrExpr):
   658	                continue
   659	            elif isinstance(item, CallExpr):
   660	                if not isinstance(item.callee, MemberExpr) or item.callee.name != "format":
   661	                    return None
   662	                elif (
   663	                    not isinstance(item.callee.expr, StrExpr) or item.callee.expr.value != "{:{}}"
   664	                ):
   665	                    return None
   666
   667	                if not isinstance(item.args[1], StrExpr) or item.args[1].value != "":
   668	                    return None
   669	            else:
   670	                return None
   671
   672	        format_ops = []
   673	        exprs: list[Expression] = []
   674
   675	        for item in expr.args[0].items:
   676	            if isinstance(item, StrExpr) and item.value != "":

   678	                exprs.append(item)

   680	                format_ops.append(FormatOp.STR)
   681	                exprs.append(item.args[0])
   682
   683	        substitutions = convert_format_expr_to_str(builder, format_ops, exprs, expr.line)
   684	        if substitutions is None:
   685	            return None
   686
   687	        return join_formatted_strings(builder, None, substitutions, expr.line)
   688	    return None
