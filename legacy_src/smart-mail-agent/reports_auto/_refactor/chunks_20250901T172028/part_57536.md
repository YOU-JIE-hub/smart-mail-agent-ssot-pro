   591	    @classmethod
   592	    def construct(cls: Type['Model'], _fields_set: Optional['SetStr'] = None, **values: Any) -> 'Model':
   593	        """
   594	        Creates a new model setting __dict__ and __fields_set__ from trusted or pre-validated data.
   595	        Default values are respected, but no other validation is performed.
   596	        Behaves as if `Config.extra = 'allow'` was set since it adds all passed values
   597	        """
   598	        m = cls.__new__(cls)
   599	        fields_values: Dict[str, Any] = {}
   600	        for name, field in cls.__fields__.items():
   601	            if field.alt_alias and field.alias in values:
   602	                fields_values[name] = values[field.alias]
   603	            elif name in values:
   604	                fields_values[name] = values[name]
   605	            elif not field.required:
   606	                fields_values[name] = field.get_default()
   607	        fields_values.update(values)
   608	        object_setattr(m, '__dict__', fields_values)
   609	        if _fields_set is None:
   610	            _fields_set = set(values.keys())
   611	        object_setattr(m, '__fields_set__', _fields_set)
   612	        m._init_private_attributes()
   613	        return m

   615	    def _copy_and_set_values(self: 'Model', values: 'DictStrAny', fields_set: 'SetStr', *, deep: bool) -> 'Model':
   616	        if deep:
   617	            # chances of having empty dict here are quite low for using smart_deepcopy
   618	            values = deepcopy(values)
   619
   620	        cls = self.__class__
   621	        m = cls.__new__(cls)
   622	        object_setattr(m, '__dict__', values)
   623	        object_setattr(m, '__fields_set__', fields_set)
   624	        for name in self.__private_attributes__:
   625	            value = getattr(self, name, Undefined)
   626	            if value is not Undefined:
   627	                if deep:
   628	                    value = deepcopy(value)
   629	                object_setattr(m, name, value)
   630
   631	        return m

   633	    def copy(
   634	        self: 'Model',
   635	        *,
   636	        include: Optional[Union['AbstractSetIntStr', 'MappingIntStrAny']] = None,
   637	        exclude: Optional[Union['AbstractSetIntStr', 'MappingIntStrAny']] = None,
   638	        update: Optional['DictStrAny'] = None,
   639	        deep: bool = False,
   640	    ) -> 'Model':
   641	        """
   642	        Duplicate a model, optionally choose which fields to include, exclude and change.
   643
   644	        :param include: fields to include in new model
   645	        :param exclude: fields to exclude from new model, as with values this takes precedence over include
   646	        :param update: values to change/add in the new model. Note: the data is not validated before creating
   647	            the new model: you should trust this data
   648	        :param deep: set to `True` to make a deep copy of the model
   649	        :return: new model instance
   650	        """
   651
   652	        values = dict(
   653	            self._iter(to_dict=False, by_alias=False, include=include, exclude=exclude, exclude_unset=False),
   654	            **(update or {}),
   655	        )
   656
   657	        # new `__fields_set__` can have unset optional fields with a set value in `update` kwarg
   658	        if update:
   659	            fields_set = self.__fields_set__ | update.keys()
   660	        else:
   661	            fields_set = set(self.__fields_set__)
   662
   663	        return self._copy_and_set_values(values, fields_set, deep=deep)

   688	    @classmethod
   689	    def validate(cls: Type['Model'], value: Any) -> 'Model':
   690	        if isinstance(value, cls):
   691	            copy_on_model_validation = cls.__config__.copy_on_model_validation
   692	            # whether to deep or shallow copy the model on validation, None means do not copy
   693	            deep_copy: Optional[bool] = None
   694	            if copy_on_model_validation not in {'deep', 'shallow', 'none'}:
   695	                # Warn about deprecated behavior
   696	                warnings.warn(
   697	                    "`copy_on_model_validation` should be a string: 'deep', 'shallow' or 'none'", DeprecationWarning
   698	                )
   699	                if copy_on_model_validation:
   700	                    deep_copy = False
   701
   702	            if copy_on_model_validation == 'shallow':
   703	                # shallow copy
   704	                deep_copy = False
   705	            elif copy_on_model_validation == 'deep':
   706	                # deep copy
   707	                deep_copy = True
   708
   709	            if deep_copy is None:
   710	                return value
   711	            else:
   712	                return value._copy_and_set_values(value.__dict__, value.__fields_set__, deep=deep_copy)
   713
   714	        value = cls._enforce_dict_if_root(value)
   715
   716	        if isinstance(value, dict):
   717	            return cls(**value)
   718	        elif cls.__config__.orm_mode:
   719	            return cls.from_orm(value)
   720	        else:
   721	            try:
   722	                value_as_dict = dict(value)
   723	            except (TypeError, ValueError) as e:
   724	                raise DictError() from e
   725	            return cls(**value_as_dict)

   733	    @classmethod
   734	    @no_type_check
   735	    def _get_value(
   736	        cls,
   737	        v: Any,
   738	        to_dict: bool,
   739	        by_alias: bool,
   740	        include: Optional[Union['AbstractSetIntStr', 'MappingIntStrAny']],
   741	        exclude: Optional[Union['AbstractSetIntStr', 'MappingIntStrAny']],
   742	        exclude_unset: bool,
   743	        exclude_defaults: bool,
   744	        exclude_none: bool,
   745	    ) -> Any:
   746	        if isinstance(v, BaseModel):
   747	            if to_dict:
   748	                v_dict = v.dict(
   749	                    by_alias=by_alias,
   750	                    exclude_unset=exclude_unset,
   751	                    exclude_defaults=exclude_defaults,
   752	                    include=include,
   753	                    exclude=exclude,
   754	                    exclude_none=exclude_none,
   755	                )
   756	                if ROOT_KEY in v_dict:
   757	                    return v_dict[ROOT_KEY]
   758	                return v_dict
   759	            else:
   760	                return v.copy(include=include, exclude=exclude)
   761
   762	        value_exclude = ValueItems(v, exclude) if exclude else None
   763	        value_include = ValueItems(v, include) if include else None
   764
   765	        if isinstance(v, dict):
   766	            return {
   767	                k_: cls._get_value(
   768	                    v_,
   769	                    to_dict=to_dict,
   770	                    by_alias=by_alias,
   771	                    exclude_unset=exclude_unset,
   772	                    exclude_defaults=exclude_defaults,
   773	                    include=value_include and value_include.for_element(k_),
   774	                    exclude=value_exclude and value_exclude.for_element(k_),
   775	                    exclude_none=exclude_none,
   776	                )
   777	                for k_, v_ in v.items()
   778	                if (not value_exclude or not value_exclude.is_excluded(k_))
   779	                and (not value_include or value_include.is_included(k_))
   780	            }
   781
   782	        elif sequence_like(v):
   783	            seq_args = (
   784	                cls._get_value(
   785	                    v_,
   786	                    to_dict=to_dict,
   787	                    by_alias=by_alias,
   788	                    exclude_unset=exclude_unset,
   789	                    exclude_defaults=exclude_defaults,
   790	                    include=value_include and value_include.for_element(i),
   791	                    exclude=value_exclude and value_exclude.for_element(i),
   792	                    exclude_none=exclude_none,
   793	                )
   794	                for i, v_ in enumerate(v)
   795	                if (not value_exclude or not value_exclude.is_excluded(i))
   796	                and (not value_include or value_include.is_included(i))
   797	            )
   798
   799	            return v.__class__(*seq_args) if is_namedtuple(v.__class__) else v.__class__(seq_args)
   800
   801	        elif isinstance(v, Enum) and getattr(cls.Config, 'use_enum_values', False):
   802	            return v.value
   803
   804	        else:
   805	            return v

   828	    def _iter(
   829	        self,
   830	        to_dict: bool = False,
   831	        by_alias: bool = False,
   832	        include: Optional[Union['AbstractSetIntStr', 'MappingIntStrAny']] = None,
   833	        exclude: Optional[Union['AbstractSetIntStr', 'MappingIntStrAny']] = None,
   834	        exclude_unset: bool = False,
   835	        exclude_defaults: bool = False,
   836	        exclude_none: bool = False,
   837	    ) -> 'TupleGenerator':
   838	        # Merge field set excludes with explicit exclude parameter with explicit overriding field set options.
   839	        # The extra "is not None" guards are not logically necessary but optimizes performance for the simple case.
   840	        if exclude is not None or self.__exclude_fields__ is not None:
   841	            exclude = ValueItems.merge(self.__exclude_fields__, exclude)
   842
   843	        if include is not None or self.__include_fields__ is not None:
   844	            include = ValueItems.merge(self.__include_fields__, include, intersect=True)
   845
   846	        allowed_keys = self._calculate_keys(
   847	            include=include, exclude=exclude, exclude_unset=exclude_unset  # type: ignore
   848	        )
   849	        if allowed_keys is None and not (to_dict or by_alias or exclude_unset or exclude_defaults or exclude_none):
   850	            # huge boost for plain _iter()
   851	            yield from self.__dict__.items()
   852	            return
   853
   854	        value_exclude = ValueItems(self, exclude) if exclude is not None else None
   855	        value_include = ValueItems(self, include) if include is not None else None
   856
   857	        for field_key, v in self.__dict__.items():
   858	            if (allowed_keys is not None and field_key not in allowed_keys) or (exclude_none and v is None):
   859	                continue
   860
   861	            if exclude_defaults:
   862	                model_field = self.__fields__.get(field_key)
   863	                if not getattr(model_field, 'required', True) and getattr(model_field, 'default', _missing) == v:
   864	                    continue
   865
   866	            if by_alias and field_key in self.__fields__:
   867	                dict_key = self.__fields__[field_key].alias
   868	            else:
   869	                dict_key = field_key
   870
   871	            if to_dict or value_include or value_exclude:
   872	                v = self._get_value(
   873	                    v,
   874	                    to_dict=to_dict,
   875	                    by_alias=by_alias,
   876	                    include=value_include and value_include.for_element(field_key),
   877	                    exclude=value_exclude and value_exclude.for_element(field_key),
   878	                    exclude_unset=exclude_unset,
   879	                    exclude_defaults=exclude_defaults,
   880	                    exclude_none=exclude_none,
   881	                )
   882	            yield dict_key, v

   884	    def _calculate_keys(
   885	        self,
   886	        include: Optional['MappingIntStrAny'],
   887	        exclude: Optional['MappingIntStrAny'],
   888	        exclude_unset: bool,
   889	        update: Optional['DictStrAny'] = None,
   890	    ) -> Optional[AbstractSet[str]]:
   891	        if include is None and exclude is None and exclude_unset is False:
   892	            return None
   893
   894	        keys: AbstractSet[str]
   895	        if exclude_unset:
   896	            keys = self.__fields_set__.copy()
   897	        else:
   898	            keys = self.__dict__.keys()
   899
   900	        if include is not None:
   901	            keys &= include.keys()
   902
   903	        if update:
   904	            keys -= update.keys()
   905
   906	        if exclude:
   907	            keys -= {k for k, v in exclude.items() if ValueItems.is_true(v)}
   908
   909	        return keys
