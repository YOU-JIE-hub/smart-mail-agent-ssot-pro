    12	def frange(start, end=None, inc=None):
    13	    "A range function, that does accept float increments..."
    14
    15	    if end == None:
    16	        end = start + 0.0
    17	        start = 0.0
    18
    19	    if inc == None:
    20	        inc = 1.0
    21
    22	    L = []
    23	    end = end - inc*0.0001  #to avoid numrical problems
    24	    while 1:
    25	        next = start + len(L) * inc
    26	        if inc > 0 and next >= end:
    27	            break
    28	        elif inc < 0 and next <= end:
    29	            break
    30	        L.append(next)
    31
    32	    return L
