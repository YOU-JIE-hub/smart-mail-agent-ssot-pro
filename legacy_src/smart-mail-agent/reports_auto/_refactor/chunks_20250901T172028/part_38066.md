  5153	        # number of items than e.  In that case we use those context

  5166	        # that's not a Tuple.

  5169	        for i in range(len(e.items)):

  5171	            if isinstance(item, StarExpr):
  5172	                # Special handling for star expressions.
  5173	                # TODO: If there's a context, and item.expr is a
  5174	                # TupleExpr, flatten it, so we can benefit from the
  5175	                # context?  Counterargument: Why would anyone write

  5192	                            return self.check_lst_expr(e, "builtins.tuple", "<tuple>")

  5194	                            seen_unpack_in_items = True
  5195	                    items.extend(tt.items)
  5196	                    # Note: this logic depends on full structure match in tuple_context_matches().
  5197	                    if unpack_in_context:
  5198	                        j += 1
  5199	                    else:

  5202	                        j += len(tt.items)
