   927	def write_deps_cache(
   928	    rdeps: dict[str, dict[str, set[str]]], manager: BuildManager, graph: Graph
   929	) -> None:
   930	    """Write cache files for fine-grained dependencies.
   931
   932	    Serialize fine-grained dependencies map for fine grained mode.
   933
   934	    Dependencies on some module 'm' is stored in the dependency cache
   935	    file m.deps.json.  This entails some spooky action at a distance:
   936	    if module 'n' depends on 'm', that produces entries in m.deps.json.
   937	    When there is a dependency on a module that does not exist in the
   938	    build, it is stored with its first existing parent module. If no
   939	    such module exists, it is stored with the fake module FAKE_ROOT_MODULE.
   940
   941	    This means that the validity of the fine-grained dependency caches
   942	    are a global property, so we store validity checking information for
   943	    fine-grained dependencies in a global cache file:
   944	     * We take a snapshot of current sources to later check consistency
   945	       between the fine-grained dependency cache and module cache metadata
   946	     * We store the mtime of all of the dependency files to verify they
   947	       haven't changed
   948	    """
   949	    metastore = manager.metastore
   950
   951	    error = False
   952
   953	    fg_deps_meta = manager.fg_deps_meta.copy()
   954
   955	    for id in rdeps:
   956	        if id != FAKE_ROOT_MODULE:
   957	            _, _, deps_json = get_cache_names(id, graph[id].xpath, manager.options)
   958	        else:
   959	            deps_json = DEPS_ROOT_FILE
   960	        assert deps_json
   961	        manager.log("Writing deps cache", deps_json)
   962	        if not manager.metastore.write(deps_json, deps_to_json(rdeps[id])):
   963	            manager.log(f"Error writing fine-grained deps JSON file {deps_json}")
   964	            error = True
   965	        else:
   966	            fg_deps_meta[id] = {"path": deps_json, "mtime": manager.getmtime(deps_json)}
   967
   968	    meta_snapshot: dict[str, str] = {}
   969	    for id, st in graph.items():
   970	        # If we didn't parse a file (so it doesn't have a
   971	        # source_hash), then it must be a module with a fresh cache,
   972	        # so use the hash from that.
   973	        if st.source_hash:
   974	            hash = st.source_hash
   975	        else:
   976	            if st.meta:
   977	                hash = st.meta.hash
   978	            else:
   979	                hash = ""
   980	        meta_snapshot[id] = hash
   981
   982	    meta = {"snapshot": meta_snapshot, "deps_meta": fg_deps_meta}
   983
   984	    if not metastore.write(DEPS_META_FILE, json_dumps(meta)):
   985	        manager.log(f"Error writing fine-grained deps meta JSON file {DEPS_META_FILE}")
   986	        error = True
   987
   988	    if error:
   989	        manager.errors.set_file(_cache_dir_prefix(manager.options), None, manager.options)
   990	        manager.errors.report(0, 0, "Error writing fine-grained dependencies cache", blocker=True)

   993	def invert_deps(deps: dict[str, set[str]], graph: Graph) -> dict[str, dict[str, set[str]]]:
   994	    """Splits fine-grained dependencies based on the module of the trigger.
   995
   996	    Returns a dictionary from module ids to all dependencies on that
   997	    module. Dependencies not associated with a module in the build will be
   998	    associated with the nearest parent module that is in the build, or the
   999	    fake module FAKE_ROOT_MODULE if none are.
  1000	    """
  1001	    # Lazy import to speed up startup
  1002	    from mypy.server.target import trigger_to_target
  1003
  1004	    # Prepopulate the map for all the modules that have been processed,
  1005	    # so that we always generate files for processed modules (even if
  1006	    # there aren't any dependencies to them.)
  1007	    rdeps: dict[str, dict[str, set[str]]] = {id: {} for id, st in graph.items() if st.tree}
  1008	    for trigger, targets in deps.items():
  1009	        module = module_prefix(graph, trigger_to_target(trigger))
  1010	        if not module or not graph[module].tree:
  1011	            module = FAKE_ROOT_MODULE
  1012
  1013	        mod_rdeps = rdeps.setdefault(module, {})
  1014	        mod_rdeps.setdefault(trigger, set()).update(targets)
  1015
  1016	    return rdeps

  1019	def generate_deps_for_cache(manager: BuildManager, graph: Graph) -> dict[str, dict[str, set[str]]]:
  1020	    """Generate fine-grained dependencies into a form suitable for serializing.
  1021
  1022	    This does a couple things:
  1023	    1. Splits fine-grained deps based on the module of the trigger
  1024	    2. For each module we generated fine-grained deps for, load any previous
  1025	       deps and merge them in.
  1026
  1027	    Returns a dictionary from module ids to all dependencies on that
  1028	    module. Dependencies not associated with a module in the build will be
  1029	    associated with the nearest parent module that is in the build, or the
  1030	    fake module FAKE_ROOT_MODULE if none are.
  1031	    """
  1032	    from mypy.server.deps import merge_dependencies  # Lazy import to speed up startup
  1033
  1034	    # Split the dependencies out into based on the module that is depended on.
  1035	    rdeps = invert_deps(manager.fg_deps, graph)
  1036
  1037	    # We can't just clobber existing dependency information, so we
  1038	    # load the deps for every module we've generated new dependencies
  1039	    # to and merge the new deps into them.
  1040	    for module, mdeps in rdeps.items():
  1041	        old_deps = manager.load_fine_grained_deps(module)
  1042	        merge_dependencies(old_deps, mdeps)
  1043
  1044	    return rdeps

  1097	def read_deps_cache(manager: BuildManager, graph: Graph) -> dict[str, FgDepMeta] | None:
  1098	    """Read and validate the fine-grained dependencies cache.
  1099
  1100	    See the write_deps_cache documentation for more information on
  1101	    the details of the cache.
  1102
  1103	    Returns None if the cache was invalid in some way.
  1104	    """
  1105	    deps_meta = _load_json_file(
  1106	        DEPS_META_FILE,
  1107	        manager,
  1108	        log_success="Deps meta ",
  1109	        log_error="Could not load fine-grained dependency metadata: ",
  1110	    )
  1111	    if deps_meta is None:
  1112	        return None
  1113	    meta_snapshot = deps_meta["snapshot"]
  1114	    # Take a snapshot of the source hashes from all of the metas we found.
  1115	    # (Including the ones we rejected because they were out of date.)
  1116	    # We use this to verify that they match up with the proto_deps.
  1117	    current_meta_snapshot = {
  1118	        id: st.meta_source_hash for id, st in graph.items() if st.meta_source_hash is not None
  1119	    }
  1120
  1121	    common = set(meta_snapshot.keys()) & set(current_meta_snapshot.keys())
  1122	    if any(meta_snapshot[id] != current_meta_snapshot[id] for id in common):
  1123	        # TODO: invalidate also if options changed (like --strict-optional)?
  1124	        manager.log("Fine-grained dependencies cache inconsistent, ignoring")
  1125	        return None
  1126
  1127	    module_deps_metas = deps_meta["deps_meta"]
  1128	    assert isinstance(module_deps_metas, dict)
  1129	    if not manager.options.skip_cache_mtime_checks:
  1130	        for meta in module_deps_metas.values():
  1131	            try:
  1132	                matched = manager.getmtime(meta["path"]) == meta["mtime"]
  1133	            except FileNotFoundError:
  1134	                matched = False
  1135	            if not matched:
  1136	                manager.log(f"Invalid or missing fine-grained deps cache: {meta['path']}")
  1137	                return None
  1138
  1139	    return module_deps_metas

  1229	def get_cache_names(id: str, path: str, options: Options) -> tuple[str, str, str | None]:
  1230	    """Return the file names for the cache files.
  1231
  1232	    Args:
  1233	      id: module ID
  1234	      path: module path
  1235	      cache_dir: cache directory
  1236	      pyversion: Python version (major, minor)
  1237
  1238	    Returns:
  1239	      A tuple with the file names to be used for the meta JSON, the
  1240	      data JSON, and the fine-grained deps JSON, respectively.
  1241	    """
  1242	    if options.cache_map:
  1243	        pair = options.cache_map.get(normpath(path, options))
  1244	    else:
  1245	        pair = None
  1246	    if pair is not None:
  1247	        # The cache map paths were specified relative to the base directory,
  1248	        # but the filesystem metastore APIs operates relative to the cache
  1249	        # prefix directory.
  1250	        # Solve this by rewriting the paths as relative to the root dir.
  1251	        # This only makes sense when using the filesystem backed cache.
  1252	        root = _cache_dir_prefix(options)
  1253	        return (os.path.relpath(pair[0], root), os.path.relpath(pair[1], root), None)
  1254	    prefix = os.path.join(*id.split("."))
  1255	    is_package = os.path.basename(path).startswith("__init__.py")
  1256	    if is_package:
  1257	        prefix = os.path.join(prefix, "__init__")
  1258
  1259	    deps_json = None
  1260	    if options.cache_fine_grained:
  1261	        deps_json = prefix + ".deps.json"
  1262	    return (prefix + ".meta.json", prefix + ".data.json", deps_json)

  1265	def find_cache_meta(id: str, path: str, manager: BuildManager) -> CacheMeta | None:
  1266	    """Find cache data for a module.
  1267
  1268	    Args:
  1269	      id: module ID
  1270	      path: module path
  1271	      manager: the build manager (for pyversion, log/trace, and build options)
  1272
  1273	    Returns:
  1274	      A CacheMeta instance if the cache data was found and appears
  1275	      valid; otherwise None.
  1276	    """
  1277	    # TODO: May need to take more build options into account
  1278	    meta_json, data_json, _ = get_cache_names(id, path, manager.options)
  1279	    manager.trace(f"Looking for {id} at {meta_json}")
  1280	    t0 = time.time()
  1281	    meta = _load_json_file(
  1282	        meta_json, manager, log_success=f"Meta {id} ", log_error=f"Could not load cache for {id}: "
  1283	    )
  1284	    t1 = time.time()
  1285	    if meta is None:
  1286	        return None
  1287	    if not isinstance(meta, dict):
  1288	        manager.log(f"Could not load cache for {id}: meta cache is not a dict: {repr(meta)}")  # type: ignore[unreachable]
  1289	        return None
  1290	    m = cache_meta_from_dict(meta, data_json)
  1291	    t2 = time.time()
  1292	    manager.add_stats(
  1293	        load_meta_time=t2 - t0, load_meta_load_time=t1 - t0, load_meta_from_dict_time=t2 - t1
  1294	    )
  1295
  1296	    # Don't check for path match, that is dealt with in validate_meta().
  1297	    #
  1298	    # TODO: these `type: ignore`s wouldn't be necessary
  1299	    # if the type annotations for CacheMeta were more accurate
  1300	    # (all of these attributes can be `None`)
  1301	    if (
  1302	        m.id != id
  1303	        or m.mtime is None  # type: ignore[redundant-expr]
  1304	        or m.size is None  # type: ignore[redundant-expr]
  1305	        or m.dependencies is None  # type: ignore[redundant-expr]
  1306	        or m.data_mtime is None
  1307	    ):
  1308	        manager.log(f"Metadata abandoned for {id}: attributes are missing")
  1309	        return None
  1310
  1311	    # Ignore cache if generated by an older mypy version.
  1312	    if (
  1313	        (m.version_id != manager.version_id and not manager.options.skip_version_check)
  1314	        or m.options is None
  1315	        or len(m.dependencies) + len(m.suppressed) != len(m.dep_prios)
  1316	        or len(m.dependencies) + len(m.suppressed) != len(m.dep_lines)
  1317	    ):
  1318	        manager.log(f"Metadata abandoned for {id}: new attributes are missing")
  1319	        return None
  1320
  1321	    # Ignore cache if (relevant) options aren't the same.
  1322	    # Note that it's fine to mutilate cached_options since it's only used here.
  1323	    cached_options = m.options
  1324	    current_options = manager.options.clone_for_module(id).select_options_affecting_cache()
  1325	    if manager.options.skip_version_check:
  1326	        # When we're lax about version we're also lax about platform.
  1327	        cached_options["platform"] = current_options["platform"]
  1328	    if "debug_cache" in cached_options:
  1329	        # Older versions included debug_cache, but it's silly to compare it.
  1330	        del cached_options["debug_cache"]
  1331	    if cached_options != current_options:
  1332	        manager.log(f"Metadata abandoned for {id}: options differ")
  1333	        if manager.options.verbosity >= 2:
  1334	            for key in sorted(set(cached_options) | set(current_options)):
  1335	                if cached_options.get(key) != current_options.get(key):
  1336	                    manager.trace(
  1337	                        "    {}: {} != {}".format(
  1338	                            key, cached_options.get(key), current
