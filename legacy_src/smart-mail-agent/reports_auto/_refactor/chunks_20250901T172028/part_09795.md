   333	                            shf1 = shadingAmount
   334	                        else:
   335	                            dsh = (shadingAmount-1)/float(nshades-1)
   336	                            shf1 = 1
   337	                        shda = (a2-a1)/float(nshades)

   339	                        theSector.fillColor = None
   340	                        for ish in range(nshades):
   341	                            sha1 = a1 + ish*shda
   342	                            sha2 = a1 + (ish+1)*shda
