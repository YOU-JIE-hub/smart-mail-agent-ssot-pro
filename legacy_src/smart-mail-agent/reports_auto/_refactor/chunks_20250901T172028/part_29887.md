entTarget:
   618	        if line == -1:
   619	            line = lvalue.line
   620	        if isinstance(lvalue, NameExpr):
   621	            # If we are visiting a decorator, then the SymbolNode we really want to be looking at
   622	            # is the function that is decorated, not the entire Decorator node itself.
   623	            symbol = lvalue.node
   624	            if isinstance(symbol, Decorator):
   625	                symbol = symbol.func
   626	            if symbol is None:
   627	                # Semantic analyzer doesn't create ad-hoc Vars for special forms.
   628	                assert lvalue.is_special_form
   629	                symbol = Var(lvalue.name)
   630	            if not for_read and isinstance(symbol, Var) and symbol.is_cls:
   631	                self.error("Cannot assign to the first argument of classmethod", line)
   632	            if lvalue.kind == LDEF:
   633	                if symbol not in self.symtables[-1]:
   634	                    if isinstance(symbol, Var) and not isinstance(symbol.type, DeletedType):
   635	                        reg_type = self.type_to_rtype(symbol.type)
   636	                    else:
   637	                        reg_type = self.node_type(lvalue)
   638	                    # If the function is a generator function, then first define a new variable
   639	                    # in the current function's environment class. Next, define a target that
   640	                    # refers to the newly defined variable in that environment class. Add the
   641	                    # target to the table containing class environment variables, as well as the
   642	                    # current environment.
   643	                    if self.fn_info.is_generator:
   644	                        return self.add_var_to_env_class(
   645	                            symbol, reg_type, self.fn_info.generator_class, reassign=False
   646	                        )
   647
   648	                    # Otherwise define a new local variable.
   649	                    return self.add_local_reg(symbol, reg_type)
   650	                else:
   651	                    # Assign to a previously defined variable.
   652	                    return self.lookup(symbol)
   653	            elif lvalue.kind == GDEF:
   654	                globals_dict = self.load_globals_dict()
   655	                name = self.load_str(lvalue.name)
   656	                return AssignmentTargetIndex(globals_dict, name)
   657	            else:
   658	                assert False, lvalue.kind
   659	        elif isinstance(lvalue, IndexExpr):
   660	            # Indexed assignment x[y] = e
   661	            base = self.accept(lvalue.base)
   662	            index = self.accept(lvalue.index)
   663	            return AssignmentTargetIndex(base, index)
   664	        elif isinstance(lvalue, MemberExpr):
   665	            # Attribute assignment x.y = e
   666	            can_borrow = self.is_native_attr_ref(lvalue)
   667	            obj = self.accept(lvalue.expr, can_borrow=can_borrow)
   668	            return AssignmentTargetAttr(obj, lvalue.name, can_borrow=can_borrow)
   669	        elif isinstance(lvalue, TupleExpr):
   670	            # Multiple assignment a, ..., b = e
   671	            star_idx: int | None = None
   672	            lvalues = []
   673	            for idx, item in enumerate(lvalue.items):
   674	                targ = self.get_assignment_target(item)
   675	                lvalues.append(targ)
   676	                if isinstance(item, StarExpr):
   677	                    if star_idx is not None:
   678	                        self.error("Two starred expressions in assignment", line)
   679	                    star_idx = idx
   680
   681	            return AssignmentTargetTuple(lvalues, star_idx)
   682
   683	        elif isinstance(lvalue, StarExpr):
   684	            return self.get_assignment_target(lvalue.expr)
   685
   686	        assert False, "Unsupported lvalue: %r" % lvalue

   688	    def read(
   689	        self, target: Value | AssignmentTarget, line: int = -1, can_borrow: bool = False
   690	    ) -> Value:

   692	            return target
   693	        if isinstance(target, AssignmentTargetRegister):
   694	            return target.register

   697	                target.base, "__getitem__", [target.index], target.type, line

   700	                return reg

   702	        if isinstance(target, AssignmentTargetAttr):
   703	            if isinstance(target.obj.type, RInstance) and target.obj.type.class_ir.is_ext_class:
   704	                borrow = can_borrow and target.can_borrow
   705	                return self.add(GetAttr(target.obj, target.attr, line, borrow=borrow))
   706	            else:
   707	                return self.py_get_attr(target.obj, target.attr, line)

   709	        assert False, "Unsupported lvalue: %r" % target

   720	    def assign(self, target: Register | AssignmentTarget, rvalue_reg: Value, line: int) -> None:
   721	        if isinstance(target, Register):
   722	            self.add(Assign(target, self.coerce_rvalue(rvalue_reg, target.type, line)))
   723	        elif isinstance(target, AssignmentTargetRegister):
   724	            rvalue_reg = self.coerce_rvalue(rvalue_reg, target.type, line)
   725	            self.add(Assign(target.register, rvalue_reg))
   726	        elif isinstance(target, AssignmentTargetAttr):
   727	            if isinstance(target.obj_type, RInstance):
   728	                rvalue_reg = self.coerce_rvalue(rvalue_reg, target.type, line)
   729	                self.add(SetAttr(target.obj, target.attr, rvalue_reg, line))
   730	            else:
   731	                key = self.load_str(target.attr)
   732	                boxed_reg = self.builder.box(rvalue_reg)
   733	                self.primitive_op(py_setattr_op, [target.obj, key, boxed_reg], line)
   734	        elif isinstance(target, AssignmentTargetIndex):
   735	            target_reg2 = self.gen_method_call(
   736	                target.base, "__setitem__", [target.index, rvalue_reg], None, line
   737	            )
   738	            assert target_reg2 is not None, target.base.type
   739	        elif isinstance(target, AssignmentTargetTuple):
   740	            if isinstance(rvalue_reg.type, RTuple) and target.star_idx is None:
   741	                rtypes = rvalue_reg.type.types
   742	                assert len(rtypes) == len(target.items)
   743	                for i in range(len(rtypes)):
   744	                    item_value = self.add(TupleGet(rvalue_reg, i, line))
   745	                    self.assign(target.items[i], item_value, line)
   746	            elif (
   747	                is_list_rprimitive(rvalue_reg.type) or is_tuple_rprimitive(rvalue_reg.type)
   748	            ) and target.star_idx is None:
   749	                self.process_sequence_assignment(target, rvalue_reg, line)
   750	            else:
   751	                self.process_iterator_tuple_assignment(target, rvalue_reg, line)
   752	        else:
   753	            assert False, "Unsupported assignment target"

   755	    def coerce_rvalue(self, rvalue: Value, rtype: RType, line: int) -> Value:
   756	        if is_float_rprimitive(rtype) and is_tagged(rvalue.type):
   757	            typename = rvalue.type.short_name()
   758	            if typename == "short_int":
   759	                typename = "int"
   760	            self.error(
   761	                "Incompatible value representations in assignment "
   762	                + f'(expression has type "{typename}", variable has type "float")',
   763	                line,
   764	            )
   765	        return self.coerce(rvalue, rtype, line)

   767	    def process_sequence_assignment(
   768	        self, target: AssignmentTargetTuple, rvalue: Value, line: int
   769	    ) -> None:
   770	        """Process assignment like 'x, y = s', where s is a variable-length list or tuple."""
   771	        # Check the length of sequence.
   772	        expected_len = Integer(len(target.items), c_pyssize_t_rprimitive)
   773	        self.builder.call_c(check_unpack_count_op, [rvalue, expected_len], line)
   774
   775	        # Read sequence items.
   776	        values = []
   777	        for i in range(len(target.items)):
   778	            item = target.items[i]
   779	            index = self.builder.load_int(i)
   780	            if is_list_rprimitive(rvalue.type):
   781	                item_value = self.primitive_op(list_get_item_unsafe_op, [rvalue, index], line)
   782	            else:
   783	                item_value = self.builder.gen_method_call(
   784	                    rvalue, "__getitem__", [index], item.type, line
   785	                )
   786	            values.append(item_value)
   787
   788	        # Assign sequence items to the target lvalues.
   789	        for lvalue, value in zip(target.items, values):
   790	            self.assign(lvalue, value, line)

   792	    def process_iterator_tuple_assignment_helper(
   793	        self, litem: AssignmentTarget, ritem: Value, line: int
   794	    ) -> None:
   795	        error_block, ok_block = BasicBlock(), BasicBlock()
   796	        self.add(Branch(ritem, error_block, ok_block, Branch.IS_ERROR))
   797
   798	        self.activate_block(error_block)
   799	        self.add(
   800	            RaiseStandardError(RaiseStandardError.VALUE_ERROR, "not enough values to unpack", line)
   801	        )
   802	        self.add(Unreachable())
   803
   804	        self.activate_block(ok_block)
   805	        self.assign(litem, ritem, line)
   806
   807	    def process_iterator_tuple_assignment(
   808	        self, target: AssignmentTargetTuple, rvalue_reg: Value, line: int
   809	    ) -> None:
   810	        iterator = self.primitive_op(iter_op, [rvalue_reg], line)
   811
   812	        # This may be the whole lvalue list if there is no starred value
   813	        split_idx = target.star_idx if target.star_idx is not None else len(target.items)
   814
   815	        # Assign values before the first starred value
   816	        for litem in target.items[:split_idx]:
   817	            ritem = self.call_c(next_op, [iterator], line)
   818	            error_block, ok_block = BasicBlock(), BasicBlock()
   819	            self.add(Branch(ritem, error_block, ok_block, Branch.IS_ERROR))
   820
   821	            self.activate_block(error_block)
   822	            self.add(
   823	                RaiseStandardError(
   824	                    RaiseStandardError.VALUE_ERROR, "not enough values to unpack", line
   825	                )
   826	            )
   827	            self.add(Unreachable())
   828
   829	            self.activate_block(ok_block)
   830
   831	            self.assign(litem, ritem, line)
   832
   833	        # Assign the starred value and all values after it
   834	        if target.star_idx is not None:
   835	            post_star_vals = target.items[split_idx + 1 :]
   836	            iter_list = self.primitive_op(to_list, [iterator], line)
   837	            iter_list_len = self.builtin_len(iter_list, line)
   838	            post_star_len = Integer(len(post_star_vals))
   839	            condition = self.binary_op(post_star_len, iter_list_len, "<=", line)
   840
   841	            error_block, ok_block = BasicBlock(), BasicBlock()
   842	            self.add(Branch(condition, ok_block, error_block, Branch.BOOL))
   843
   844	            self.activate_block(error_block)
   845	            self.add(
   846	                RaiseStandardError(
   847	                    RaiseStandardError.VALUE_ERROR, "not enough values to unpack", line
   848	                )
   849	            )
   850	            self.add(Unreachable())
   851
   852	            self.activate_block(ok_block)
   853
   854	            for litem in reversed(post_star_vals):
   855	                ritem = self.primitive_op(list_pop_last, [iter_list], line)
   856	                self.assign(litem, ritem, line)
   857
   858	            # Assign the starred value
   859	            self.assign(target.items[target.star_idx], iter_list, line)
   860
   861	        # There is no starred value, so check if there are extra values in rhs that
   862	        # have not been assigned.
   863	        else:
   864	            extra = self.call_c(next_op, [iterator], line)
   865	            error_block, ok_block = BasicBlock(), BasicBlock()
   866	            self.add(Branch(extra, ok_block, error_block, Branch.IS_ERROR))
   867
   868	            self.activate_block(error_block)
   869	            self.add(
   870	                RaiseStandardError(
   871	                    RaiseStandardError.VALUE_ERROR, "too many values to unpack", line
   872	                )
   873	            )
   874	            self.add(Unreachable())
   875
   876	            self.activate_block(ok_block)
   877
