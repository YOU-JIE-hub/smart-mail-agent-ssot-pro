    12	def frange(start, end=None, inc=None):
    13	    "A range function, that does accept float increments..."
    14
    15	    if end == None:
    16	        end = start + 0.0
    17	        start = 0.0
    18
    19	    if inc == None:
    20	        inc = 1.0
    21
    22	    L = []
    23	    end = end - inc*0.0001  #to avoid numrical problems
    24	    while 1:
    25	        next = start + len(L) * inc
    26	        if inc > 0 and next >= end:
    27	            break
    28	        elif inc < 0 and next <= end:
    29	            break
    30	        L.append(next)
    31
    32	    return L

    35	def makeDistancesList(list):
    36	    """Returns a list of distances between adjacent numbers in some input list.
    37
    38	    E.g. [1, 1, 2, 3, 5, 7] -> [0, 1, 1, 2, 2]
    39	    """
    40
    41	    d = []
    42	    for i in range(len(list[:-1])):
    43	        d.append(list[i+1] - list[i])
    44
    45	    return d

    48	class Grid(Widget):
    49	    """This makes a rectangular grid of equidistant stripes.
    50
    51	    The grid contains an outer border rectangle, and stripes
    52	    inside which can be drawn with lines and/or as solid tiles.
    53	    The drawing order is: outer rectangle, then lines and tiles.
    54
    55	    The stripes' width is indicated as 'delta'. The sequence of
    56	    stripes can have an offset named 'delta0'. Both values need
    57	    to be positive!
    58	    """
    59
    60	    _attrMap = AttrMap(
    61	        x = AttrMapValue(isNumber, desc="The grid's lower-left x position."),
    62	        y = AttrMapValue(isNumber, desc="The grid's lower-left y position."),
    63	        width = AttrMapValue(isNumber, desc="The grid's width."),
    64	        height = AttrMapValue(isNumber, desc="The grid's height."),
    65	        orientation = AttrMapValue(OneOf(('vertical', 'horizontal')),
    66	            desc='Determines if stripes are vertical or horizontal.'),
    67	        useLines = AttrMapValue(OneOf((0, 1)),
    68	            desc='Determines if stripes are drawn with lines.'),
    69	        useRects = AttrMapValue(OneOf((0, 1)),
    70	            desc='Determines if stripes are drawn with solid rectangles.'),
    71	        delta = AttrMapValue(isNumber,
    72	            desc='Determines the width/height of the stripes.'),
    73	        delta0 = AttrMapValue(isNumber,
    74	            desc='Determines the stripes initial width/height offset.'),
    75	        deltaSteps = AttrMapValue(isListOfNumbers,
    76	            desc='List of deltas to be used cyclically.'),
    77	        stripeColors = AttrMapValue(isListOfColors,
    78	            desc='Colors applied cyclically in the right or upper direction.'),
    79	        fillColor = AttrMapValue(isColorOrNone,
    80	            desc='Background color for entire rectangle.'),
    81	        strokeColor = AttrMapValue(isColorOrNone,
    82	            desc='Color used for lines.'),
    83	        strokeWidth = AttrMapValue(isNumber,
    84	            desc='Width used for lines.'),
    85	        rectStrokeColor = AttrMapValue(isColorOrNone, desc='Color for outer rect stroke.'),
    86	        rectStrokeWidth = AttrMapValue(isNumberOrNone, desc='Width for outer rect stroke.'),

   114	    def makeOuterRect(self):
   115	        strokeColor = getattr(self,'rectStrokeColor',self.strokeColor)
   116	        strokeWidth = getattr(self,'rectStrokeWidth',self.strokeWidth)
   117	        if self.fillColor or (strokeColor and strokeWidth):
   118	            rect = Rect(self.x, self.y, self.width, self.height)
   119	            rect.fillColor = self.fillColor
   120	            rect.strokeColor = strokeColor
   121	            rect.strokeWidth = strokeWidth
   122	            return rect
   123	        else:
   124	            return None

   126	    def makeLinePosList(self, start, isX=0):
   127	        "Returns a list of positions where to place lines."
   128
   129	        w, h = self.width, self.height
   130	        if isX:
   131	            length = w
   132	        else:
   133	            length = h
   134	        if self.deltaSteps:
   135	            r = [start + self.delta0]
   136	            i = 0
   137	            while 1:
   138	                if r[-1] > start + length:
   139	                    del r[-1]
   140	                    break
   141	                r.append(r[-1] + self.deltaSteps[i % len(self.deltaSteps)])
   142	                i = i + 1
   143	        else:
   144	            r = frange(start + self.delta0, start + length, self.delta)
   145
   146	        r.append(start + length)
   147	        if self.delta0 != 0:
   148	            r.insert(0, start)
   149	        #print 'Grid.makeLinePosList() -> %s' % r
   150	        return r

   153	    def makeInnerLines(self):
   154	        # inner grid lines
   155	        group = Group()
   156
   157	        w, h = self.width, self.height
   158
   159	        if self.useLines == 1:
   160	            if self.orientation == 'vertical':
   161	                r = self.makeLinePosList(self.x, isX=1)
   162	                for x in r:
   163	                    line = Line(x, self.y, x, self.y + h)
   164	                    line.strokeColor = self.strokeColor
   165	                    line.strokeWidth = self.strokeWidth
   166	                    group.add(line)
   167	            elif self.orientation == 'horizontal':
   168	                r = self.makeLinePosList(self.y, isX=0)
   169	                for y in r:
   170	                    line = Line(self.x, y, self.x + w, y)
   171	                    line.strokeColor = self.strokeColor
   172	                    line.strokeWidth = self.strokeWidth
   173	                    group.add(line)
   174
   175	        return group
   176
   177
   178	    def makeInnerTiles(self):
   179	        # inner grid lines
   180	        group = Group()
   181
   182	        w, h = self.width, self.height
   183
   184	        # inner grid stripes (solid rectangles)
   185	        if self.useRects == 1:
   186	            cols = self.stripeColors
   187
   188	            if self.orientation == 'vertical':
   189	                r = self.makeLinePosList(self.x, isX=1)
   190	            elif self.orientation == 'horizontal':
   191	                r = self.makeLinePosList(self.y, isX=0)
   192
   193	            dist = makeDistancesList(r)
   194
   195	            i = 0
   196	            for j in range(len(dist)):
   197	                if self.orientation == 'vertical':
   198	                    x = r[j]
   199	                    stripe = Rect(x, self.y, dist[j], h)
   200	                elif self.orientation == 'horizontal':
   201	                    y = r[j]
   202	                    stripe = Rect(self.x, y, w, dist[j])
   203	                stripe.fillColor = cols[i % len(cols)]
   204	                stripe.strokeColor = None
   205	                group.add(stripe)
   206	                i = i + 1
   207
   208	        return group
   209
   210
   211	    def draw(self):
   212	        # general widget bits
   213	        group = Group()
   214
   215	        group.add(self.makeOuterRect())
   216	        group.add(self.makeInnerTiles())
   217	        group.add(self.makeInnerLines(),name='_gridLines')
   218
   219	        return group

   339	        self.fillColorEnd = colors.black

   352	    def _flipRectCorners(self):
   353	        "Flip rectangle's corners if width or height is negative."
   354	        x, y, width, height, fillColorStart, fillColorEnd = self.x, self.y, self.width, self.height, self.fillColorStart, self.fillColorEnd
   355	        if width < 0 and height > 0:
   356	            x = x + width
   357	            width = -width
   358	            if self.orientation=='vertical': fillColorStart, fillColorEnd = fillColorEnd, fillColorStart
   359	        elif height<0 and width>0:
   360	            y = y + height
   361	            height = -height
   362	            if self.orientation=='horizontal': fillColorStart, fillColorEnd = fillColorEnd, fillColorStart
   363	        elif height < 0 and height < 0:
   364	            x = x + width
   365	            width = -width
   366	            y = y + height
   367	            height = -height
   368	        return x, y, width, height, fillColorStart, fillColorEnd
   369
   370	    def draw(self):
   371	        # general widget bits
   372	        group = Group()
   373	        x, y, w, h, c0, c1 = self._flipRectCorners()
   374	        numShades = self.numShades
   375	        if self.cylinderMode:
   376	            if not numShades%2: numShades = numShades+1
   377	            halfNumShades = int((numShades-1)/2) + 1
   378	        num = float(numShades) # must make it float!
   379	        vertical = self.orientation == 'vertical'
   380	        if vertical:
   381	            if numShades == 1:
   382	                V = [x]
   383	            else:
   384	                V = frange(x, x + w, w/num)
   385	        else:
   386	            if numShades == 1:
   387	                V = [y]
   388	            else:
   389	                V = frange(y, y + h, h/num)
   390
   391	        for v in V:
   392	            stripe = vertical and Rect(v, y, w/num, h) or Rect(x, v, w, h/num)
   393	            if self.cylinderMode:
   394	                if V.index(v)>=halfNumShades:
   395	                    col = colors.linearlyInterpolatedColor(c1,c0,V[halfNumShades],V[-1], v)
   396	                else:
   397	                    col = colors.linearlyInterpolatedColor(c0,c1,V[0],V[halfNumShades], v)
   398	            else:
   399	                col = colors.linearlyInterpolatedColor(c0,c1,V[0],V[-1], v)
   400	            stripe.fillColor = col
   401	            stripe.strokeColor = col
   402	            stripe.strokeWidth = 1
   403	            group.add(stripe)
   404	        if self.strokeColor and self.strokeWidth>=0:
   405	            rect = Rect(x, y, w, h)
   406	            rect.strokeColor = self.strokeColor
   407	            rect.strokeWidth = self.strokeWidth
   408	            rect.fillColor = None
   409	            group.add(rect)
   410	        return group

   425	def centroid(P):
   426	    '''compute average point of a set of points'''
   427	    cx = 0
   428	    cy = 0
   429	    for x,y in P:
   430	        cx+=x
   431	        cy+=y
   432	    n = float(len(P))
   433	    return cx/n, cy/n

   435	def rotatedEnclosingRect(P, angle, rect):
   436	    '''
   437	    given P a sequence P of x,y coordinate pairs and an angle in degrees
   438	    find the centroid of P and the axis at angle theta through it
   439	    find the extreme points of P wrt axis parallel distance and axis
   440	    orthogonal distance. Then compute the least rectangle that will still
   441	    enclose P when rotated by angle.
   442
   443	    The class R
   444	    '''
   445	    from math import pi, cos, sin
   446	    x0, y0 = centroid(P)
   447	    theta = (angle/180.)*pi
   448	    s,c=sin(theta),cos(theta)
   449	    def parallelAxisDist(xy,s=s,c=c,x0=x0,y0=y0):
   450	        x,y = xy
   451	        return (s*(y-y0)+c*(x-x0))
   452	    def orthogonalAxisDist(xy,s=s,c=c,x0=x0,y0=y0):
   453	        x,y = xy
   454	        return (c*(y-y0)+s*(x-x0))
   455	    L = list(map(parallelAxisDist,P))
   456	    L.sort()
   457	    a0, a1 = L[0], L[-1]
   458	    L = list(map(orthogonalAxisDist,P))
   459	    L.sort()
   460	    b0, b1 = L[0], L[-1]
   461	    rect.x, rect.width = a0, a1-a0
   462	    rect.y, rect.height = b0, b1-b0
   463	    g = Group(transform=(c,s,-s,c,x0,y0))
   464	    g.add(rect)
   465	    return g
