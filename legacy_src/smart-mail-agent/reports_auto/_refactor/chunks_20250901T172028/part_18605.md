   327	def all_values(args: list[Register], blocks: list[BasicBlock]) -> list[Value]:
   328	    """Return the set of all values that may be initialized in the blocks.
   329
   330	    This omits registers that are only read.
   331	    """
   332	    values: list[Value] = list(args)
   333	    seen_registers = set(args)
   334
   335	    for block in blocks:
   336	        for op in block.ops:
   337	            if not isinstance(op, ControlOp):
   338	                if isinstance(op, (Assign, AssignMulti)):
   339	                    if op.dest not in seen_registers:
   340	                        values.append(op.dest)
   341	                        seen_registers.add(op.dest)
   342	                elif op.is_void:
   343	                    continue
   344	                else:
   345	                    # If we take the address of a register, it might get initialized.
   346	                    if (
   347	                        isinstance(op, LoadAddress)
   348	                        and isinstance(op.src, Register)
   349	                        and op.src not in seen_registers
   350	                    ):
   351	                        values.append(op.src)
   352	                        seen_registers.add(op.src)
   353	                    values.append(op)
   354
   355	    return values

   358	def all_values_full(args: list[Register], blocks: list[BasicBlock]) -> list[Value]:
   359	    """Return set of all values that are initialized or accessed."""
   360	    values: list[Value] = list(args)
   361	    seen_registers = set(args)
   362
   363	    for block in blocks:
   364	        for op in block.ops:
   365	            for source in op.sources():
   366	                # Look for uninitialized registers that are accessed. Ignore
   367	                # non-registers since we don't allow ops outside basic blocks.
   368	                if isinstance(source, Register) and source not in seen_registers:
   369	                    values.append(source)
   370	                    seen_registers.add(source)
   371	            if not isinstance(op, ControlOp):
   372	                if isinstance(op, (Assign, AssignMulti)):
   373	                    if op.dest not in seen_registers:
   374	                        values.append(op.dest)
   375	                        seen_registers.add(op.dest)
   376	                elif op.is_void:
   377	                    continue
   378	                else:
   379	                    values.append(op)
   380
   381	    return values

    50	class BasicBlock:
    51	    """IR basic block.
    52
    53	    Contains a sequence of Ops and ends with a ControlOp (Goto,
    54	    Branch, Return or Unreachable). Only the last op can be a
    55	    ControlOp.
    56
    57	    All generated Ops live in basic blocks. Basic blocks determine the
    58	    order of evaluation and control flow within a function. A basic
    59	    block is always associated with a single function/method (FuncIR).
    60
    61	    When building the IR, ops that raise exceptions can be included in
    62	    the middle of a basic block, but the exceptions aren't checked.
    63	    Afterwards we perform a transform that inserts explicit checks for
    64	    all error conditions and splits basic blocks accordingly to preserve
    65	    the invariant that a jump, branch or return can only ever appear
    66	    as the final op in a block. Manually inserting error checking ops
    67	    would be boring and error-prone.
    68
    69	    BasicBlocks have an error_handler attribute that determines where
    70	    to jump if an error occurs. If none is specified, an error will
    71	    propagate up out of the function. This is compiled away by the
    72	    `exceptions` module.
    73
    74	    Block labels are used for pretty printing and emitting C code, and
    75	    get filled in by those passes.
    76
    77	    Ops that may terminate the program aren't treated as exits.
    78	    """
    79
    80	    def __init__(self, label: int = -1) -> None:
    81	        self.label = label
    82	        self.ops: list[Op] = []
    83	        self.error_handler: BasicBlock | None = None
    84	        self.referenced = False
    85
    86	    @property
    87	    def terminated(self) -> bool:
    88	        """Does the block end with a jump, branch or return?
    89
    90	        This should always be true after the basic block has been fully built, but
    91	        this is false during construction.
    92	        """
    93	        return bool(self.ops) and isinstance(self.ops[-1], ControlOp)
    94
    95	    @property
    96	    def terminator(self) -> ControlOp:
    97	        """The terminator operation of the block."""
    98	        assert bool(self.ops) and isinstance(self.ops[-1], ControlOp)
    99	        return self.ops[-1]
