   148	def enum_value_callback(ctx: mypy.plugin.AttributeContext) -> Type:
   149	    """This plugin refines the 'value' attribute in enums to refer to
   150	    the original underlying value. For example, suppose we have the
   151	    following:
   152
   153	        class SomeEnum:
   154	            FOO = A()
   155	            BAR = B()
   156
   157	    By default, mypy will infer that 'SomeEnum.FOO.value' and
   158	    'SomeEnum.BAR.value' both are of type 'Any'. This plugin refines
   159	    this inference so that mypy understands the expressions are
   160	    actually of types 'A' and 'B' respectively. This better reflects
   161	    the actual runtime behavior.
   162
   163	    This plugin works simply by looking up the original value assigned
   164	    to the enum. For example, when this plugin sees 'SomeEnum.BAR.value',
   165	    it will look up whatever type 'BAR' had in the SomeEnum TypeInfo and
   166	    use that as the inferred type of the overall expression.
   167
   168	    This plugin assumes that the provided context is an attribute access
   169	    matching one of the strings found in 'ENUM_VALUE_ACCESS'.

   171	    enum_field_name = _extract_underlying_field_name(ctx.type)
   172	    if enum_field_name is None:
   173	        # We do not know the enum field name (perhaps it was passed to a
   174	        # function and we only know that it _is_ a member).  All is not lost
   175	        # however, if we can prove that the all of the enum members have the
   176	        # same value-type, then it doesn't matter which member was passed in.
   177	        # The value-type is still known.

   201	                or (not isinstance(t, CallableType) and not is_named_instance(t, "enum.nonmember"))
   202	            ]
   203	            underlying_type = _first(proper_types)
   204	            if underlying_type is None:
   205	                return ctx.default_attr_type

   207	            # At first we try to predict future `value` type if all other items
   208	            # have the same type. For example, `int`.
   209	            # If this is the case, we simply return this type.
   210	            # See https://github.com/python/mypy/pull/9443
   211	            all_same_value_type = all(
   212	                proper_type is not None and proper_type == underlying_type
   213	                for proper_type in proper_types
   214	            )
   215	            if all_same_value_type:
   216	                if underlying_type is not None:
   217	                    return underlying_type
   218
   219	            # But, after we started treating all `Enum` values as `Final`,
   220	            # we start to infer types in
   221	            # `item = 1` as `Literal[1]`, not just `int`.
   222	            # So, for example types in this `Enum` will all be different:
   223	            #
   224	            #  class Ordering(IntEnum):

   226	            #      two = 2
   227	            #      three = 3
   228	            #
   229	            # We will infer three `Literal` types here.
   230	            # They are not the same, but they are equivalent.
   231	            # So, we unify them to make sure `.value` prediction still works.
   232	            # Result will be `Literal[1] | Literal[2] | Literal[3]` for this case.
   233	            all_equivalent_types = all(
   234	                proper_type is not None and is_equivalent(proper_type, underlying_type)
   235	                for proper_type in proper_types
   236	            )
   237	            if all_equivalent_types:
   238	                return make_simplified_union(cast(Sequence[Type], proper_types))
   239	        return ctx.default_attr_type

   241	    assert isinstance(ctx.type, Instance)
   242	    info = ctx.type.type
   243
   244	    # As long as mypy doesn't understand attribute creation in __new__,
   245	    # there is no way to predict the value type if the enum class has a
   246	    # custom implementation
   247	    if _implements_new(info):
   248	        return ctx.default_attr_type
   249
   250	    stnode = info.get(enum_field_name)
   251	    if stnode is None:
   252	        return ctx.default_attr_type
   253
   254	    underlying_type = _infer_value_type_with_auto_fallback(ctx, get_proper_type(stnode.type))
   255	    if underlying_type is None:
   256	        return ctx.default_attr_type
   257
   258	    return underlying_type
