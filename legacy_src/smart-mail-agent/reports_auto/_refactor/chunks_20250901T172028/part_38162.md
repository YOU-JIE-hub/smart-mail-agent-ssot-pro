  1716	            elif (deprecated := self.get_deprecated(d)) is not None:

  1718	            elif not dec.var.is_property:
  1719	                # We have seen a "non-trivial" decorator before seeing @property, if
  1720	                # we will see a @property later, give an error, as we don't support this.

  1726	            dec.var.is_initialized_in_class = True
  1727	        if not no_type_check and self.recurse_into_functions:
  1728	            dec.func.accept(self)
  1729	        if could_be_decorated_property and dec.decorators and dec.var.is_property:
  1730	            self.fail(
  1731	                "Decorators on top of @property are not supported", dec, code=PROPERTY_DECORATOR
  1732	            )
  1733	        if (dec.func.is_static or dec.func.is_class) and dec.var.is_property:
  1734	            self.fail("Only instance methods can be decorated with @property", dec)
  1735	        if dec.func.abstract_status == IS_ABSTRACT and dec.func.is_final:
  1736	            self.fail(f"Method {dec.func.name} is both abstract and final", dec)
  1737	        if dec.func.is_static and dec.func.is_class:
  1738	            self.fail(message_registry.CLASS_PATTERN_CLASS_OR_STATIC_METHOD, dec)

  1761	    def push_type_args(
  1762	        self, type_args: list[TypeParam] | None, context: Context
  1763	    ) -> list[tuple[str, TypeVarLikeExpr]] | None:
  1764	        if not type_args:
  1765	            return []
  1766	        self.locals.append(SymbolTable())
  1767	        self.scope_stack.append(SCOPE_ANNOTATION)
  1768	        tvs: list[tuple[str, TypeVarLikeExpr]] = []
  1769	        for p in type_args:
  1770	            tv = self.analyze_type_param(p, context)
  1771	            if tv is None:
  1772	                return None
  1773	            tvs.append((p.name, tv))
  1774
  1775	        for name, tv in tvs:
  1776	            if self.is_defined_type_param(name):
  1777	                self.fail(f'"{name}" already defined as a type parameter', context)
  1778	            else:
  1779	                self.add_symbol(name, tv, context, no_progress=True, type_param=True)
  1780
  1781	        return tvs

  1783	    def is_defined_type_param(self, name: str) -> bool:
  1784	        for names in self.locals:
  1785	            if names is None:
  1786	                continue
  1787	            if name in names:
  1788	                node = names[name].node
  1789	                if isinstance(node, TypeVarLikeExpr):
  1790	                    return True
  1791	        return False

  1793	    def analyze_type_param(
  1794	        self, type_param: TypeParam, context: Context
  1795	    ) -> TypeVarLikeExpr | None:
  1796	        fullname = self.qualified_name(type_param.name)
  1797	        if type_param.upper_bound:
  1798	            upper_bound = self.anal_type(type_param.upper_bound, allow_placeholder=True)
  1799	            # TODO: we should validate the upper bound is valid for a given kind.
  1800	            if upper_bound is None:
  1801	                # This and below copies special-casing for old-style type variables, that
  1802	                # is equally necessary for new-style classes to break a vicious circle.
  1803	                upper_bound = PlaceholderType(None, [], context.line)
  1804	        else:
  1805	            if type_param.kind == TYPE_VAR_TUPLE_KIND:
  1806	                upper_bound = self.named_type("builtins.tuple", [self.object_type()])
  1807	            else:
  1808	                upper_bound = self.object_type()
  1809	        if type_param.default:
  1810	            default = self.anal_type(
  1811	                type_param.default,
  1812	                allow_placeholder=True,
  1813	                allow_unbound_tvars=True,
  1814	                report_invalid_types=False,
  1815	                allow_param_spec_literals=type_param.kind == PARAM_SPEC_KIND,
  1816	                allow_tuple_literal=type_param.kind == PARAM_SPEC_KIND,
  1817	                allow_unpack=type_param.kind == TYPE_VAR_TUPLE_KIND,
  1818	            )
  1819	            if default is None:
  1820	                default = PlaceholderType(None, [], context.line)
  1821	            elif type_param.kind == TYPE_VAR_KIND:
  1822	                default = self.check_typevar_default(default, type_param.default)
  1823	            elif type_param.kind == PARAM_SPEC_KIND:
  1824	                default = self.check_paramspec_default(default, type_param.default)
  1825	            elif type_param.kind == TYPE_VAR_TUPLE_KIND:
  1826	                default = self.check_typevartuple_default(default, type_param.default)
  1827	        else:
  1828	            default = AnyType(TypeOfAny.from_omitted_generics)
  1829	        if type_param.kind == TYPE_VAR_KIND:
  1830	            values: list[Type] = []
  1831	            if type_param.values:
  1832	                for value in type_param.values:
  1833	                    analyzed = self.anal_type(value, allow_placeholder=True)
  1834	                    if analyzed is None:
  1835	                        analyzed = PlaceholderType(None, [], context.line)
  1836	                    if has_type_vars(analyzed):
  1837	                        self.fail(message_registry.TYPE_VAR_GENERIC_CONSTRAINT_TYPE, context)
  1838	                        values.append(AnyType(TypeOfAny.from_error))
  1839	                    else:
  1840	                        values.append(analyzed)
  1841	            return TypeVarExpr(
  1842	                name=type_param.name,
  1843	                fullname=fullname,
  1844	                values=values,
  1845	                upper_bound=upper_bound,
  1846	                default=default,
  1847	                variance=VARIANCE_NOT_READY,
  1848	                is_new_style=True,
  1849	                line=context.line,
  1850	            )
  1851	        elif type_param.kind == PARAM_SPEC_KIND:
  1852	            return ParamSpecExpr(
  1853	                name=type_param.name,
  1854	                fullname=fullname,
  1855	                upper_bound=upper_bound,
  1856	                default=default,
  1857	                is_new_style=True,
  1858	                line=context.line,
  1859	            )
  1860	        else:
  1861	            assert type_param.kind == TYPE_VAR_TUPLE_KIND
  1862	            tuple_fallback = self.named_type("builtins.tuple", [self.object_type()])
  1863	            return TypeVarTupleExpr(
  1864	                name=type_param.name,
  1865	                fullname=fullname,
  1866	                upper_bound=upper_bound,
  1867	                tuple_fallback=tuple_fallback,
  1868	                default=default,
  1869	                is_new_style=True,
  1870	                line=context.line,
  1871	            )

  1879	    def analyze_class(self, defn: ClassDef) -> None:
  1880	        fullname = self.qualified_name(defn.name)
  1881	        if not defn.info and not self.is_core_builtin_class(defn):
  1882	            # Add placeholder so that self-references in base classes can be
  1883	            # resolved.  We don't want this to cause a deferral, since if there
  1884	            # are no incomplete references, we'll replace this with a TypeInfo
  1885	            # before returning.
  1886	            placeholder = PlaceholderNode(fullname, defn, defn.line, becomes_typeinfo=True)
  1887	            self.add_symbol(defn.name, placeholder, defn, can_defer=False)
  1888
  1889	        tag = self.track_incomplete_refs()
  1890
  1891	        # Restore base classes after previous iteration (things like Generic[T] might be removed).
  1892	        defn.base_type_exprs.extend(defn.removed_base_type_exprs)
  1893	        defn.removed_base_type_exprs.clear()
  1894
  1895	        self.infer_metaclass_and_bases_from_compat_helpers(defn)
  1896
  1897	        bases = defn.base_type_exprs
  1898	        bases, tvar_defs, is_protocol = self.clean_up_bases_and_infer_type_variables(
  1899	            defn, bases, context=defn
  1900	        )
  1901
  1902	        self.check_type_alias_bases(bases)
  1903
  1904	        for tvd in tvar_defs:
  1905	            if isinstance(tvd, TypeVarType) and any(
  1906	                has_placeholder(t) for t in [tvd.upper_bound] + tvd.values
  1907	            ):
  1908	                # Some type variable bounds or values are not ready, we need
  1909	                # to re-analyze this class.
  1910	                self.defer()
  1911	            if has_placeholder(tvd.default):
  1912	                # Placeholder values in TypeVarLikeTypes may get substituted in.
  1913	                # Defer current target until they are ready.
  1914	                self.mark_incomplete(defn.name, defn)
  1915	                return
  1916
  1917	        self.analyze_class_keywords(defn)
  1918	        bases_result = self.analyze_base_classes(bases)
  1919	        if bases_result is None or self.found_incomplete_ref(tag):
  1920	            # Something was incomplete. Defer current target.
  1921	            self.mark_incomplete(defn.name, defn)
  1922	            return
  1923
  1924	        base_types, base_error = bases_result
  1925	        if any(isinstance(base, PlaceholderType) for base, _ in base_types):
  1926	            # We need to know the TypeInfo of each base to construct the MRO. Placeholder types
  1927	            # are okay in nested positions, since they can't affect the MRO.
  1928	            self.mark_incomplete(defn.name, defn)
  1929	            return
  1930
  1931	        declared_metaclass, should_defer, any_meta = self.get_declared_metaclass(
  1932	            defn.name, defn.metaclass
  1933	        )
  1934	        if should_defer or self.found_incomplete_ref(tag):
  1935	            # Metaclass was not ready. Defer current target.
  1936	            self.mark_incomplete(defn.name, defn)
  1937	            return
  1938
  1939	        if self.analyze_typeddict_classdef(defn):
  1940	            if defn.info:
  1941	                self.setup_type_vars(defn, tvar_defs)
  1942	                self.setup_alias_type_vars(defn)
  1943	            return
  1944
  1945	        if self.analyze_namedtuple_classdef(defn, tvar_defs):
  1946	            return
  1947
  1948	        # Create TypeInfo for class now that base classes and the MRO can be calculated.
  1949	        self.prepare_class_def(defn)
  1950	        self.setup_type_vars(defn, tvar_defs)
  1951	        if base_error:
  1952	            defn.info.fallback_to_any = True
  1953	        if any_meta:
  1954	            defn.info.meta_fallback_to_any = True
  1955
  1956	        with self.scope.class_scope(defn.info):
  1957	            self.configure_base_classes(defn, base_types)
  1958	            defn.info.is_protocol = is_protocol
  1959	            self.recalculate_metaclass(defn, declared_metaclass)
  1960	            defn.info.runtime_protocol = False
  1961
  1962	            if defn.type_args:
  1963	                # PEP 695 type parameters are not in scope in class decorators, so
  1964	                # temporarily disable type parameter namespace.
  1965	                type_params_names = self.locals.pop()
  1966	                self.scope_stack.pop()
  1967	            for decorator in defn.decorators:
  1968	                self.analyze_class_decorator(defn, decorator)
  1969	            if defn.type_args:
  1970	                self.locals.append(type_params_names)
  1971	                self.scope_stack.append(SCOPE_ANNOTATION)
  1972
  1973	            self.analyze_class_body_common(defn)
