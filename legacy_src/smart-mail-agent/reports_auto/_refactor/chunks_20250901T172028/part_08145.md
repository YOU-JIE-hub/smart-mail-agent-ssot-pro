  7767	def is_trivial_body(block: Block) -> bool:
  7768	    """Returns 'true' if the given body is "trivial" -- if it contains just a "pass",
  7769	    "..." (ellipsis), or "raise NotImplementedError()". A trivial body may also
  7770	    start with a statement containing just a string (e.g. a docstring).
  7771
  7772	    Note: Functions that raise other kinds of exceptions do not count as
  7773	    "trivial". We use this function to help us determine when it's ok to
  7774	    relax certain checks on body, but functions that raise arbitrary exceptions
  7775	    are more likely to do non-trivial work. For example:
  7776
  7777	       def halt(self, reason: str = ...) -> NoReturn:
  7778	           raise MyCustomError("Fatal error: " + reason, self.line, self.context)
  7779
  7780	    A function that raises just NotImplementedError is much less likely to be
  7781	    this complex.
  7782
  7783	    Note: If you update this, you may also need to update
  7784	    mypy.fastparse.is_possible_trivial_body!
  7785	    """
  7786	    body = block.body
  7787	    if not body:
  7788	        # Functions have empty bodies only if the body is stripped or the function is
  7789	        # generated or deserialized. In these cases the body is unknown.
  7790	        return False
  7791
  7792	    # Skip a docstring
  7793	    if isinstance(body[0], ExpressionStmt) and isinstance(body[0].expr, StrExpr):
  7794	        body = block.body[1:]
  7795
  7796	    if len(body) == 0:
  7797	        # There's only a docstring (or no body at all).
  7798	        return True
  7799	    elif len(body) > 1:
  7800	        return False
  7801
  7802	    stmt = body[0]
  7803
  7804	    if isinstance(stmt, RaiseStmt):
  7805	        expr = stmt.expr
  7806	        if expr is None:
  7807	            return False
  7808	        if isinstance(expr, CallExpr):
  7809	            expr = expr.callee
  7810
  7811	        return isinstance(expr, NameExpr) and expr.fullname == "builtins.NotImplementedError"
  7812
  7813	    return isinstance(stmt, PassStmt) or (
  7814	        isinstance(stmt, ExpressionStmt) and isinstance(stmt.expr, EllipsisExpr)
  7815	    )

    42	def calculate_class_abstract_status(typ: TypeInfo, is_stub_file: bool, errors: Errors) -> None:
    43	    """Calculate abstract status of a class.
    44
    45	    Set is_abstract of the type to True if the type has an unimplemented
    46	    abstract attribute.  Also compute a list of abstract attributes.
    47	    Report error is required ABCMeta metaclass is missing.
    48	    """
    49	    typ.is_abstract = False
    50	    typ.abstract_attributes = []
    51	    if typ.typeddict_type:
    52	        return  # TypedDict can't be abstract
    53	    concrete: set[str] = set()
    54	    # List of abstract attributes together with their abstract status
    55	    abstract: list[tuple[str, int]] = []
    56	    abstract_in_this_class: list[str] = []
    57	    if typ.is_newtype:
    58	        # Special case: NewTypes are considered as always non-abstract, so they can be used as:
    59	        #     Config = NewType('Config', Mapping[str, str])
    60	        #     default = Config({'cannot': 'modify'})  # OK
    61	        return
    62	    for base in typ.mro:
    63	        for name, symnode in base.names.items():
    64	            node = symnode.node
    65	            if isinstance(node, OverloadedFuncDef):
    66	                # Unwrap an overloaded function definition. We can just
    67	                # check arbitrarily the first overload item. If the
    68	                # different items have a different abstract status, there
    69	                # should be an error reported elsewhere.
    70	                if node.items:  # can be empty for invalid overloads
    71	                    func: Node | None = node.items[0]
    72	                else:
    73	                    func = None
    74	            else:
    75	                func = node
    76	            if isinstance(func, Decorator):
    77	                func = func.func
    78	            if isinstance(func, FuncDef):
    79	                if (
    80	                    func.abstract_status in (IS_ABSTRACT, IMPLICITLY_ABSTRACT)
    81	                    and name not in concrete
    82	                ):
    83	                    typ.is_abstract = True
    84	                    abstract.append((name, func.abstract_status))
    85	                    if base is typ:
    86	                        abstract_in_this_class.append(name)
    87	            elif isinstance(node, Var):
    88	                if node.is_abstract_var and name not in concrete:
    89	                    typ.is_abstract = True
    90	                    abstract.append((name, IS_ABSTRACT))
    91	                    if base is typ:
    92	                        abstract_in_this_class.append(name)
    93	            concrete.add(name)

    95	    # easy to accidentally leave a concrete class abstract by forgetting to
    96	    # implement some methods.

    99	        if typ.declared_metaclass and typ.declared_metaclass.type.has_base("abc.ABCMeta"):

   102	            return
