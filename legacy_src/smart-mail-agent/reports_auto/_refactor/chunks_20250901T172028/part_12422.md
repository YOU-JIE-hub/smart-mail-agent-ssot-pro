   115	    def mktemp(self, basename: str, numbered: bool = True) -> Path:
   116	        """Create a new temporary directory managed by the factory.
   117
   118	        :param basename:
   119	            Directory base name, must be a relative path.

   121	        :param numbered:
   122	            If ``True``, ensure the directory is unique by adding a numbered
   123	            suffix greater than any existing one: ``basename="foo-"`` and ``numbered=True``
   124	            means that this function will create directories named ``"foo-0"``,
   125	            ``"foo-1"``, ``"foo-2"`` and so on.

   127	        :returns:
   128	            The path to the new directory.
   129	        """
   130	        basename = self._ensure_relative_to_basetemp(basename)
   131	        if not numbered:
   132	            p = self.getbasetemp().joinpath(basename)
   133	            p.mkdir(mode=0o700)
   134	        else:

   139	    def getbasetemp(self) -> Path:
   140	        """Return the base temporary directory, creating it if needed.
   141
   142	        :returns:
   143	            The base temporary directory.
   144	        """
   145	        if self._basetemp is not None:
   146	            return self._basetemp
   147
   148	        if self._given_basetemp is not None:
   149	            basetemp = self._given_basetemp
   150	            if basetemp.exists():
   151	                rm_rf(basetemp)
   152	            basetemp.mkdir(mode=0o700)
   153	            basetemp = basetemp.resolve()
   154	        else:
   155	            from_env = os.environ.get("PYTEST_DEBUG_TEMPROOT")
   156	            temproot = Path(from_env or tempfile.gettempdir()).resolve()
   157	            user = get_user() or "unknown"
   158	            # use a sub-directory in the temproot to speed-up
   159	            # make_numbered_dir() call
   160	            rootdir = temproot.joinpath(f"pytest-of-{user}")
   161	            try:
   162	                rootdir.mkdir(mode=0o700, exist_ok=True)
   163	            except OSError:
   164	                # getuser() likely returned illegal characters for the platform, use unknown back off mechanism
   165	                rootdir = temproot.joinpath("pytest-of-unknown")
   166	                rootdir.mkdir(mode=0o700, exist_ok=True)
   167	            # Because we use exist_ok=True with a predictable name, make sure
   168	            # we are the owners, to prevent any funny business (on unix, where
   169	            # temproot is usually shared).
   170	            # Also, to keep things private, fixup any world-readable temp
   171	            # rootdir's permissions. Historically 0o755 was used, so we can't
   172	            # just error out on this, at least for a while.
   173	            uid = get_user_id()
   174	            if uid is not None:
   175	                rootdir_stat = rootdir.stat()
   176	                if rootdir_stat.st_uid != uid:
   177	                    raise OSError(
   178	                        f"The temporary directory {rootdir} is not owned by the current user. "
   179	                        "Fix this and try again."
   180	                    )
   181	                if (rootdir_stat.st_mode & 0o077) != 0:
   182	                    os.chmod(rootdir, rootdir_stat.st_mode & ~0o077)
   183	            keep = self._retention_count
   184	            if self._retention_policy == "none":
   185	                keep = 0
   186	            basetemp = make_numbered_dir_with_cleanup(
   187	                prefix="pytest-",
   188	                root=rootdir,
   189	                keep=keep,
   190	                lock_timeout=LOCK_TIMEOUT,
   191	                mode=0o700,
   192	            )
   193	        assert basetemp is not None, basetemp
   194	        self._basetemp = basetemp
   195	        self._trace("new basetemp", basetemp)
   196	        return basetemp
