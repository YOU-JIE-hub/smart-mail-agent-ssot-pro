er, extra
   552	                        ),

   564	                )
   565	                return codes.ATTR_DEFINED
   566	        return None

   619	    def incompatible_argument(
   620	        self,
   621	        n: int,
   622	        m: int,
   623	        callee: CallableType,
   624	        arg_type: Type,
   625	        arg_kind: ArgKind,
   626	        object_type: Type | None,
   627	        context: Context,
   628	        outer_context: Context,
   629	    ) -> ErrorInfo:
   630	        """Report an error about an incompatible argument type.
   631
   632	        The argument type is arg_type, argument number is n and the
   633	        callee type is 'callee'. If the callee represents a method
   634	        that corresponds to an operator, use the corresponding
   635	        operator name in the messages.
   636
   637	        Return the error code that used for the argument (multiple error
   638	        codes are possible).
   639	        """
   640	        arg_type = get_proper_type(arg_type)
   641
   642	        target = ""
   643	        callee_name = callable_name(callee)
   644	        if callee_name is not None:
   645	            name = callee_name
   646	            if object_type is not None:
   647	                base = format_type(object_type, self.options)
   648	            else:
   649	                base = extract_type(name)
   650
   651	            for method, op in op_methods_to_symbols.items():
   652	                for variant in method, "__r" + method[2:]:
   653	                    # FIX: do not rely on textual formatting
   654	                    if name.startswith(f'"{variant}" of'):
   655	                        if op == "in" or variant != method:
   656	                            # Reversed order of base/argument.
   657	                            return self.unsupported_operand_types(
   658	                                op, arg_type, base, context, code=codes.OPERATOR
   659	                            )
   660	                        else:
   661	                            return self.unsupported_operand_types(
   662	                                op, base, arg_type, context, code=codes.OPERATOR
   663	                            )
   664
   665	            if name.startswith('"__getitem__" of'):
   666	                return self.invalid_index_type(
   667	                    arg_type, callee.arg_types[n - 1], base, context, code=codes.INDEX
   668	                )
   669
   670	            if name.startswith('"__setitem__" of'):
   671	                if n == 1:
   672	                    return self.invalid_index_type(
   673	                        arg_type, callee.arg_types[n - 1], base, context, code=codes.INDEX
   674	                    )
   675	                else:
   676	                    arg_type_str, callee_type_str = format_type_distinctly(
   677	                        arg_type, callee.arg_types[n - 1], options=self.options
   678	                    )
   679	                    info = (
   680	                        f" (expression has type {arg_type_str}, target has type {callee_type_str})"
   681	                    )
   682	                    error_msg = (
   683	                        message_registry.INCOMPATIBLE_TYPES_IN_ASSIGNMENT.with_additional_msg(info)
   684	                    )
   685	                    return self.fail(error_msg.value, context, code=error_msg.code)
   686
   687	            target = f"to {name} "
   688
   689	        msg = ""
   690	        code = codes.MISC
   691	        notes: list[str] = []
   692	        if callee_name == "<list>":
   693	            name = callee_name[1:-1]
   694	            n -= 1
   695	            actual_type_str, expected_type_str = format_type_distinctly(
   696	                arg_type, callee.arg_types[0], options=self.options
   697	            )
   698	            msg = "{} item {} has incompatible type {}; expected {}".format(
   699	                name.title(), n, actual_type_str, expected_type_str
   700	            )
   701	            code = codes.LIST_ITEM
   702	        elif callee_name == "<dict>" and isinstance(
   703	            get_proper_type(callee.arg_types[n - 1]), TupleType
   704	        ):
   705	            name = callee_name[1:-1]
   706	            n -= 1
   707	            key_type, value_type = cast(TupleType, arg_type).items
   708	            expected_key_type, expected_value_type = cast(TupleType, callee.arg_types[n]).items
   709
   710	            # don't increase verbosity unless there is need to do so
   711	            if is_subtype(key_type, expected_key_type):
   712	                key_type_str = format_type(key_type, self.options)
   713	                expected_key_type_str = format_type(expected_key_type, self.options)
   714	            else:
   715	                key_type_str, expected_key_type_str = format_type_distinctly(
   716	                    key_type, expected_key_type, options=self.options
   717	                )
   718	            if is_subtype(value_type, expected_value_type):
   719	                value_type_str = format_type(value_type, self.options)
   720	                expected_value_type_str = format_type(expected_value_type, self.options)
   721	            else:
   722	                value_type_str, expected_value_type_str = format_type_distinctly(
   723	                    value_type, expected_value_type, options=self.options
   724	                )
   725
   726	            msg = "{} entry {} has incompatible type {}: {}; expected {}: {}".format(
   727	                name.title(),
   728	                n,
   729	                key_type_str,
   730	                value_type_str,
   731	                expected_key_type_str,
   732	                expected_value_type_str,
   733	            )
   734	            code = codes.DICT_ITEM
   735	        elif callee_name == "<dict>":
   736	            value_type_str, expected_value_type_str = format_type_distinctly(
   737	                arg_type, callee.arg_types[n - 1], options=self.options
   738	            )
   739	            msg = "Unpacked dict entry {} has incompatible type {}; expected {}".format(
   740	                n - 1, value_type_str, expected_value_type_str
   741	            )
   742	            code = codes.DICT_ITEM
   743	        elif callee_name == "<list-comprehension>":
   744	            actual_type_str, expected_type_str = map(
   745	                strip_quotes,
   746	                format_type_distinctly(arg_type, callee.arg_types[0], options=self.options),
   747	            )
   748	            msg = "List comprehension has incompatible type List[{}]; expected List[{}]".format(
   749	                actual_type_str, expected_type_str
   750	            )
   751	        elif callee_name == "<set-comprehension>":
   752	            actual_type_str, expected_type_str = map(
   753	                strip_quotes,
   754	                format_type_distinctly(arg_type, callee.arg_types[0], options=self.options),
   755	            )
   756	            msg = "Set comprehension has incompatible type Set[{}]; expected Set[{}]".format(
   757	                actual_type_str, expected_type_str
   758	            )
   759	        elif callee_name == "<dictionary-comprehension>":
   760	            actual_type_str, expected_type_str = format_type_distinctly(
   761	                arg_type, callee.arg_types[n - 1], options=self.options
   762	            )
   763	            msg = (
   764	                "{} expression in dictionary comprehension has incompatible type {}; "
   765	                "expected type {}"
   766	            ).format("Key" if n == 1 else "Value", actual_type_str, expected_type_str)
   767	        elif callee_name == "<generator>":
   768	            actual_type_str, expected_type_str = format_type_distinctly(
   769	                arg_type, callee.arg_types[0], options=self.options
   770	            )
   771	            msg = "Generator has incompatible item type {}; expected {}".format(
   772	                actual_type_str, expected_type_str
   773	            )
   774	        else:
   775	            if self.prefer_simple_messages():
   776	                msg = "Argument has incompatible type"
   777	            else:
   778	                try:
   779	                    expected_type = callee.arg_types[m - 1]
   780	                except IndexError:  # Varargs callees
   781	                    expected_type = callee.arg_types[-1]
   782	                arg_type_str, expected_type_str = format_type_distinctly(
   783	                    arg_type, expected_type, bare=True, options=self.options
   784	                )
   785	                if arg_kind == ARG_STAR:
   786	                    arg_type_str = "*" + arg_type_str
   787	                elif arg_kind == ARG_STAR2:
   788	                    arg_type_str = "**" + arg_type_str
   789
   790	                # For function calls with keyword arguments, display the argument name rather
   791	                # than the number.
   792	                arg_label = str(n)
   793	                if isinstance(outer_context, CallExpr) and len(outer_context.arg_names) >= n:
   794	                    arg_name = outer_context.arg_names[n - 1]
   795	                    if arg_name is not None:
   796	                        arg_label = f'"{arg_name}"'
   797	                if (
   798	                    arg_kind == ARG_STAR2
   799	                    and isinstance(arg_type, TypedDictType)
   800	                    and m <= len(callee.arg_names)
   801	                    and callee.arg_names[m - 1] is not None
   802	                    and callee.arg_kinds[m - 1] != ARG_STAR2
   803	                ):
   804	                    arg_name = callee.arg_names[m - 1]
   805	                    assert arg_name is not None
   806	                    arg_type_str, expected_type_str = format_type_distinctly(
   807	                        arg_type.items[arg_name], expected_type, bare=True, options=self.options
   808	                    )
   809	                    arg_label = f'"{arg_name}"'
   810	                if isinstance(outer_context, IndexExpr) and isinstance(
   811	                    outer_context.index, StrExpr
   812	                ):
   813	                    msg = 'Value of "{}" has incompatible type {}; expected {}'.format(
   814	                        outer_context.index.value,
   815	                        quote_type_string(arg_type_str),
   816	                        quote_type_string(expected_type_str),
   817	                    )
   818	                else:
   819	                    msg = "Argument {} {}has incompatible type {}; expected {}".format(
   820	                        arg_label,
   821	                        target,
   822	                        quote_type_string(arg_type_str),
   823	                        quote_type_string(expected_type_str),
   824	                    )
   825	                expected_type = get_proper_type(expected_type)
   826	                if isinstance(expected_type, UnionType):
   827	                    expected_types = list(expected_type.items)
   828	                else:
   829	                    expected_types = [expected_type]
   830	                for type in get_proper_types(expected_types):
   831	                    if isinstance(arg_type, Instance) and isinstance(type, Instance):
   832	                        notes = append_invariance_notes(notes, arg_type, type)
   833	                        notes = append_numbers_notes(notes, arg_type, type)
   834	            object_type = get_proper_type(object_type)
   835	            if isinstance(object_type, TypedDictType):
   836	                code = codes.TYPEDDICT_ITEM
   837	            else:
   838	                code = codes.ARG_TYPE
   839	        error = self.fail(msg, context, code=code)
   840	        if notes:
   841	            for note_msg in notes:
   842	                self.note(note_msg, context, code=code)
   843	        return error

   845	    def incompatible_argument_note(
   846	        self,
   847	        original_caller_type: ProperType,
   848	        callee_type: ProperType,
   849	        context: Context,
   850	        parent_error: ErrorInfo,
   851	    ) -> None:
   852	        if self.prefer_simple_messages():
   853	            return
   854	        if isinstance(
   855	            original_caller_type, (Instance, TupleType, TypedDictType, TypeType, CallableType)
   856	        ):
   857	            if isinstance(callee_type, Instance) and callee_type.type.is_protocol:
   858	                self.report_protocol_problems(
   859	                    original_caller_type, callee_type, context, parent_error=parent_error
   860	                )
   861	            if isinstance(callee_type, UnionType):
   862	                for item in callee_type.items:
   863	                    item = get_proper_type(item)
   864	                    if isinstance(item, Instance) and item.type.is_protocol:
   865	                        self.report_protocol_problems(
   866	                            original_caller_type, item, context, parent_error=parent_error
   867	                        )
   868	        if isinstance(callee_type, CallableType) and isinstance(original_caller_type, Instance):
   869	            call = find_member(
   870	                "__call__", original_caller_type, original_caller_type, is_operator=True
   871	            )
   872	            if call:
   873	                self.note_call(original_caller_type, call, context, code=parent_error.code)
   874	        if isinstance(callee_type, Instance) and callee_type.type.is_protocol:
   875	            call = find_member("__call__", callee_type, callee_type, is_operator=True)
   876	            if call:
   877	                self.note_call(callee_type, call, context, code=parent_error.code)
   878	        self.maybe_note_concatenate_pos_args(
   879	            original_caller_type, callee_type, context, parent_error.code
   880	        )

   882	    def maybe_note_concatenate_pos_args(
   883	        self,
   884	        original_caller_type: ProperType,
   885	        callee_type: ProperType,
   886	        context: Context,
   887	        code: ErrorCode | None = None,
   888	    ) -> None:
   889	        # pos-only vs positional can be confusing, with Concatenate
   890	        if (
   891	            isinstance(callee_type, CallableType)
   892	            and isinstance(original_caller_type, CallableType)
   893	            and (original_caller_type.from_concatenate or callee_type.from_concatenate)
   894	        ):
   895	            names: list[str] = []
   896	            for c, o in zip(
   897	                callee_type.formal_arguments(), original_caller_type.formal_arguments()
   898	            ):
   899	                if None in (c.pos, o.pos):
   900	                    # non-positional
   901	                    continue
   902	                if c.name != o.name and c.name is None and o.name is not None:
   903	                    names.append(o.name)
   904
   905	            if names:
   906	                missing_arguments = '"' + '", "'.join(names) + '"'
   907	                self.note(
   908	                    f'This is likely because "{original_caller_type.name}" has named arguments: '
   909	                    f"{missing_arguments}. Consider marking them positional-only",
   910	                    context,
   911	                    code=code,
   912	                )
