    96	def getfuncargnames(
    97	    function: Callable[..., object],
    98	    *,
    99	    name: str = "",
   100	    cls: type | None = None,
   101	) -> tuple[str, ...]:
   102	    """Return the names of a function's mandatory arguments.
   103
   104	    Should return the names of all function arguments that:
   105	    * Aren't bound to an instance or type as in instance or class methods.
   106	    * Don't have default values.
   107	    * Aren't bound with functools.partial.
   108	    * Aren't replaced with mocks.
   109
   110	    The cls arguments indicate that the function should be treated as a bound
   111	    method even though it's not unless the function is a static method.
   112
   113	    The name parameter should be the original name in which the function was collected.
   114	    """
   115	    # TODO(RonnyPfannschmidt): This function should be refactored when we
   116	    # revisit fixtures. The fixture mechanism should ask the node for
   117	    # the fixture names, and not try to obtain directly from the
   118	    # function object well after collection has occurred.
   119
   120	    # The parameters attribute of a Signature object contains an
   121	    # ordered mapping of parameter names to Parameter instances.  This
   122	    # creates a tuple of the names of the parameters that don't have
   123	    # defaults.
   124	    try:
   125	        parameters = signature(function).parameters.values()
   126	    except (ValueError, TypeError) as e:
   127	        from _pytest.outcomes import fail
   128
   129	        fail(
   130	            f"Could not determine arguments of {function!r}: {e}",
   131	            pytrace=False,
   132	        )
   133
   134	    arg_names = tuple(
   135	        p.name
   136	        for p in parameters
   137	        if (
   138	            p.kind is Parameter.POSITIONAL_OR_KEYWORD
   139	            or p.kind is Parameter.KEYWORD_ONLY
   140	        )
   141	        and p.default is Parameter.empty
   142	    )
   143	    if not name:
   144	        name = function.__name__
   145
   146	    # If this function should be treated as a bound method even though
   147	    # it's passed as an unbound method or function, and its first parameter
   148	    # wasn't defined as positional only, remove the first parameter name.
   149	    if not any(p.kind is Parameter.POSITIONAL_ONLY for p in parameters) and (
   150	        # Not using `getattr` because we don't want to resolve the staticmethod.
   151	        # Not using `cls.__dict__` because we want to check the entire MRO.
   152	        cls
   153	        and not isinstance(
   154	            inspect.getattr_static(cls, name, default=None), staticmethod
   155	        )
   156	    ):
   157	        arg_names = arg_names[1:]
   158	    # Remove any names that will be replaced with mocks.
   159	    if hasattr(function, "__wrapped__"):
   160	        arg_names = arg_names[num_mock_patch_args(function) :]
   161	    return arg_names

   184	def ascii_escaped(val: bytes | str) -> str:
   185	    r"""If val is pure ASCII, return it as an str, otherwise, escape
   186	    bytes objects into a sequence of escaped bytes:
   187
   188	    b'\xc3\xb4\xc5\xd6' -> r'\xc3\xb4\xc5\xd6'
   189
   190	    and escapes strings into a sequence of escaped unicode ids, e.g.:
   191
   192	    r'4\nV\U00043efa\x0eMXWB\x1e\u3028\u15fd\xcd\U0007d944'
   193
   194	    Note:
   195	       The obvious "v.decode('unicode-escape')" will return
   196	       valid UTF-8 unicode if it finds them in bytes, but we
   197	       want to return escaped bytes for any byte, even if they match
   198	       a UTF-8 string.
   199	    """
   200	    if isinstance(val, bytes):
   201	        ret = val.decode("ascii", "backslashreplace")
   202	    else:
   203	        ret = val.encode("unicode_escape").decode("ascii")
   204	    return ret.translate(_non_printable_ascii_translate_table)

   250	    """Return the current process's real user id or None if it could not be
   251	    determined.
   252
   253	    :return: The user id or None if it could not be determined.
   254	    """
   255	    # mypy follows the version and platform checking expectation of PEP 484:
   256	    # https://mypy.readthedocs.io/en/stable/common_issues.html?highlight=platform#python-version-and-system-platform-checks

   270	# Perform exhaustiveness checking.
   271	#
   272	# Consider this example:
   273	#
   274	#     MyUnion = Union[int, str]
   275	#
   276	#     def handle(x: MyUnion) -> int {
   277	#         if isinstance(x, int):
   278	#             return 1
   279	#         elif isinstance(x, str):
   280	#             return 2
   281	#         else:
   282	#             raise Exception('unreachable')
   283	#
   284	# Now suppose we add a new variant:
   285	#
   286	#     MyUnion = Union[int, str, bytes]
   287	#
   288	# After doing this, we must remember ourselves to go and update the handle
   289	# function to handle the new variant.
   290	#
   291	# With `assert_never` we can do better:
   292	#
   293	#     // raise Exception('unreachable')
   294	#     return assert_never(x)
   295	#
   296	# Now, if we forget to handle the new variant, the type-checker will emit a
   297	# compile-time error, instead of the runtime error we would have gotten
   298	# previously.
   299	#
   300	# This also work for Enums (if you use `is` to compare) and Literals.
   301	def assert_never(value: NoReturn) -> NoReturn:
   302	    assert False, f"Unhandled value: {value} ({type(value).__name__})"
