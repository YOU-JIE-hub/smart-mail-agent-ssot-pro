   238	class CoverageOptionParser(optparse.OptionParser):
   239	    """Base OptionParser for coverage.py.
   240
   241	    Problems don't exit the program.
   242	    Defaults are initialized for all options.
   243
   244	    """
   245
   246	    def __init__(self, *args: Any, **kwargs: Any) -> None:
   247	        kwargs["add_help_option"] = False
   248	        super().__init__(*args, **kwargs)
   249	        self.set_defaults(
   250	            # Keep these arguments alphabetized by their names.
   251	            action=None,
   252	            append=None,
   253	            branch=None,
   254	            concurrency=None,
   255	            context=None,
   256	            contexts=None,
   257	            data_file=None,
   258	            debug=None,
   259	            directory=None,
   260	            fail_under=None,
   261	            format=None,
   262	            help=None,
   263	            ignore_errors=None,
   264	            include=None,
   265	            keep=None,
   266	            module=None,
   267	            omit=None,
   268	            parallel_mode=None,
   269	            precision=None,
   270	            pylib=None,
   271	            quiet=None,
   272	            rcfile=True,
   273	            save_signal=None,
   274	            show_contexts=None,
   275	            show_missing=None,
   276	            skip_covered=None,
   277	            skip_empty=None,
   278	            sort=None,
   279	            source=None,
   280	            timid=None,
   281	            title=None,
   282	            version=None,
   283	        )
   284
   285	        self.disable_interspersed_args()
   286
   287	    class OptionParserError(Exception):
   288	        """Used to stop the optparse error handler ending the process."""
   289	        pass
   290
   291	    def parse_args_ok(self, args: list[str]) -> tuple[bool, optparse.Values | None, list[str]]:
   292	        """Call optparse.parse_args, but return a triple:
   293
   294	        (ok, options, args)
   295
   296	        """
   297	        try:
   298	            options, args = super().parse_args(args)
   299	        except self.OptionParserError:
   300	            return False, None, []
   301	        return True, options, args
   302
   303	    def error(self, msg: str) -> NoReturn:
   304	        """Override optparse.error so sys.exit doesn't get called."""
   305	        show_help(msg)
   306	        raise self.OptionParserError

   321	class CmdOptionParser(CoverageOptionParser):
   322	    """Parse one of the new-style commands for coverage.py."""
   323
   324	    def __init__(
   325	        self,
   326	        action: str,
   327	        options: list[optparse.Option],
   328	        description: str,
   329	        usage: str | None = None,
   330	    ):
   331	        """Create an OptionParser for a coverage.py command.
   332
   333	        `action` is the slug to put into `options.action`.
   334	        `options` is a list of Option's for the command.
   335	        `description` is the description of the command, for the help text.
   336	        `usage` is the usage string to display in help.
   337
   338	        """
   339	        if usage:
   340	            usage = "%prog " + usage
   341	        super().__init__(
   342	            usage=usage,
   343	            description=description,
   344	        )
   345	        self.set_defaults(action=action)
   346	        self.add_options(options)
   347	        self.cmd = action
   348
   349	    def __eq__(self, other: str) -> bool:       # type: ignore[override]
   350	        # A convenience equality, so that I can put strings in unit test
   351	        # results, and they will compare equal to objects.
   352	        return (other == f"<CmdOptionParser:{self.cmd}>")
   353
   354	    __hash__ = None         # type: ignore[assignment]
   355
   356	    def get_prog_name(self) -> str:
   357	        """Override of an undocumented function in optparse.OptionParser."""
   358	        program_name = super().get_prog_name()
   359
   360	        # Include the sub-command for this parser as part of the command.
   361	        return f"{program_name} {self.cmd}"
