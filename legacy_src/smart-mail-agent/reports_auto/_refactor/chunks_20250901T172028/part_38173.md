  6245	    def lookup(
  6246	        self, name: str, ctx: Context, suppress_errors: bool = False
  6247	    ) -> SymbolTableNode | None:
  6248	        """Look up an unqualified (no dots) name in all active namespaces.
  6249
  6250	        Note that the result may contain a PlaceholderNode. The caller may
  6251	        want to defer in that case.
  6252
  6253	        Generate an error if the name is not defined unless suppress_errors
  6254	        is true or the current namespace is incomplete. In the latter case
  6255	        defer.
  6256	        """
  6257	        implicit_name = False
  6258	        # 1a. Name declared using 'global x' takes precedence
  6259	        if name in self.global_decls[-1]:
  6260	            if name in self.globals:
  6261	                return self.globals[name]
  6262	            if not suppress_errors:
  6263	                self.name_not_defined(name, ctx)
  6264	            return None
  6265	        # 1b. Name declared using 'nonlocal x' takes precedence
  6266	        if name in self.nonlocal_decls[-1]:
  6267	            for table in reversed(self.locals[:-1]):
  6268	                if table is not None and name in table:
  6269	                    return table[name]
  6270	            if not suppress_errors:
  6271	                self.name_not_defined(name, ctx)
  6272	            return None
  6273	        # 2a. Class attributes (if within class definition)
  6274	        if self.type and not self.is_func_scope() and name in self.type.names:
  6275	            node = self.type.names[name]
  6276	            if not node.implicit:
  6277	                if self.is_active_symbol_in_class_body(node.node):
  6278	                    return node
  6279	            else:
  6280	                # Defined through self.x assignment
  6281	                implicit_name = True
  6282	                implicit_node = node
  6283	        # 2b. Class attributes __qualname__ and __module__
  6284	        if self.type and not self.is_func_scope() and name in {"__qualname__", "__module__"}:
  6285	            return SymbolTableNode(MDEF, Var(name, self.str_type()))
  6286	        # 3. Local (function) scopes
  6287	        for table in reversed(self.locals):
  6288	            if table is not None and name in table:
  6289	                return table[name]
  6290
  6291	        # 4. Current file global scope
  6292	        if name in self.globals:
  6293	            return self.globals[name]
  6294	        # 5. Builtins
  6295	        b = self.globals.get("__builtins__", None)
  6296	        if b:
  6297	            assert isinstance(b.node, MypyFile)
  6298	            table = b.node.names
  6299	            if name in table:
  6300	                if len(name) > 1 and name[0] == "_" and name[1] != "_":
  6301	                    if not suppress_errors:
  6302	                        self.name_not_defined(name, ctx)
  6303	                    return None
  6304	                node = table[name]
  6305	                return node
  6306	        # Give up.
  6307	        if not implicit_name and not suppress_errors:
  6308	            self.name_not_defined(name, ctx)
  6309	        else:
  6310	            if implicit_name:
  6311	                return implicit_node
  6312	        return None

  6314	    def is_active_symbol_in_class_body(self, node: SymbolNode | None) -> bool:
  6315	        """Can a symbol defined in class body accessed at current statement?
  6316
  6317	        Only allow access to class attributes textually after
  6318	        the definition, so that it's possible to fall back to the
  6319	        outer scope. Example:
  6320
  6321	            class X: ...
  6322
  6323	            class C:
  6324	                X = X  # Initializer refers to outer scope
  6325
  6326	        Nested classes are an exception, since we want to support
  6327	        arbitrary forward references in type annotations. Also, we
  6328	        allow forward references to type aliases to support recursive
  6329	        types.
  6330	        """
  6331	        # TODO: Forward reference to name imported in class body is not
  6332	        #       caught.
  6333	        if self.statement is None:
  6334	            # Assume it's fine -- don't have enough context to check
  6335	            return True
  6336	        if (
  6337	            node is None
  6338	            or self.is_textually_before_statement(node)
  6339	            or not self.is_defined_in_current_module(node.fullname)
  6340	        ):
  6341	            return True
  6342	        if self.is_type_like(node):
  6343	            # Allow forward references to classes/type aliases (see docstring), but
  6344	            # a forward reference should never shadow an existing regular reference.
  6345	            if node.name not in self.globals:
  6346	                return True
  6347	            global_node = self.globals[node.name]
  6348	            if not self.is_textually_before_class(global_node.node):
  6349	                return True
  6350	            return not self.is_type_like(global_node.node)
  6351	        return False
  6352

  6355	            isinstance(node, PlaceholderNode) and node.becomes_typeinfo

  6358	    def is_textually_before_statement(self, node: SymbolNode) -> bool:
  6359	        """Check if a node is defined textually before the current statement
  6360
  6361	        Note that decorated functions' line number are the same as
  6362	        the top decorator.
  6363	        """
  6364	        assert self.statement
  6365	        line_diff = self.statement.line - node.line
  6366
  6367	        # The first branch handles reference an overloaded function variant inside itself,
  6368	        # this is a corner case where mypy technically deviates from runtime name resolution,
  6369	        # but it is fine because we want an overloaded function to be treated as a single unit.
  6370	        if self.is_overloaded_item(node, self.statement):
  6371	            return False
  6372	        elif isinstance(node, Decorator) and not node.is_overload:
  6373	            return line_diff > len(node.original_decorators)
  6374	        else:
  6375	            return line_diff > 0

  6379	        assert self.type is not None
  6380	        if node is None:

  6402	    def lookup_qualified(
  6403	        self, name: str, ctx: Context, suppress_errors: bool = False
  6404	    ) -> SymbolTableNode | None:
  6405	        """Lookup a qualified name in all activate namespaces.
  6406
  6407	        Note that the result may contain a PlaceholderNode. The caller may
  6408	        want to defer in that case.
  6409
  6410	        Generate an error if the name is not defined unless suppress_errors
  6411	        is true or the current namespace is incomplete. In the latter case
  6412	        defer.
  6413	        """
  6414	        if "." not in name:
  6415	            # Simple case: look up a short name.
  6416	            return self.lookup(name, ctx, suppress_errors=suppress_errors)
  6417	        parts = name.split(".")
  6418	        namespace = self.cur_mod_id
  6419	        sym = self.lookup(parts[0], ctx, suppress_errors=suppress_errors)
  6420	        if sym:
  6421	            for i in range(1, len(parts)):
  6422	                node = sym.node
  6423	                part = parts[i]
  6424	                if isinstance(node, TypeInfo):
  6425	                    nextsym = node.get(part)
  6426	                elif isinstance(node, MypyFile):
  6427	                    nextsym = self.get_module_symbol(node, part)
  6428	                    namespace = node.fullname
  6429	                elif isinstance(node, PlaceholderNode):
  6430	                    return sym
  6431	                elif isinstance(node, TypeAlias) and node.no_args:
  6432	                    assert isinstance(node.target, ProperType)
  6433	                    if isinstance(node.target, Instance):
  6434	                        nextsym = node.target.type.get(part)
  6435	                    else:
  6436	                        nextsym = None
  6437	                else:
  6438	                    if isinstance(node, Var):
  6439	                        typ = get_proper_type(node.type)
  6440	                        if isinstance(typ, AnyType):
  6441	                            # Allow access through Var with Any type without error.
  6442	                            return self.implicit_symbol(sym, name, parts[i:], typ)
  6443	                    # This might be something like valid `P.args` or invalid `P.__bound__` access.
  6444	                    # Important note that `ParamSpecExpr` is also ignored in other places.
  6445	                    # See https://github.com/python/mypy/pull/13468
  6446	                    if isinstance(node, ParamSpecExpr) and part in ("args", "kwargs"):
  6447	                        return None
  6448	                    # Lookup through invalid node, such as variable or function
  6449	                    nextsym = None
  6450	                if not nextsym or nextsym.module_hidden:
  6451	                    if not suppress_errors:
  6452	                        self.name_not_defined(name, ctx, namespace=namespace)
  6453	                    return None
  6454	                sym = nextsym
  6455	        return sym
