  1675	                # between Python versions.

  1678	        join_method.set_line(empty_string)

  1926	        if node is None:

  1958	    def visit_Call(self, e: Call) -> Type:

  1960	        f = e.func

  1967	            return self.invalid_type(e, note=note)

  1969	            self.fail(message_registry.ARG_CONSTRUCTOR_NAME_EXPECTED, e.lineno, e.col_offset)

  1972	        default_type = AnyType(TypeOfAny.special_form)
  1973	        typ: Type = default_type

  1975	            if i == 0:

  1981	            else:

  1985	            if k.arg == "name":

  1988	                        message_registry.MULTIPLE_VALUES_FOR_NAME_KWARG.format(constructor),
  1989	                        f.lineno,
  1990	                        f.col_offset,
  1991	                    )
  1992	                name = self._extract_argument_name(value)
  1993	            elif k.arg == "type":
  1994	                if typ is not default_type:
  1995	                    self.fail(
  1996	                        message_registry.MULTIPLE_VAL
