   290	    def from_file(self, filename: str, warn: Callable[[str], None], our_file: bool) -> bool:
   291	        """Read configuration from a .rc file.
   292
   293	        `filename` is a file name to read.
   294
   295	        `our_file` is True if this config file is specifically for coverage,
   296	        False if we are examining another config file (tox.ini, setup.cfg)
   297	        for possible settings.
   298
   299	        Returns True or False, whether the file could be read, and it had some
   300	        coverage.py settings in it.
   301
   302	        """
   303	        _, ext = os.path.splitext(filename)
   304	        cp: TConfigParser
   305	        if ext == ".toml":
   306	            cp = TomlConfigParser(our_file)
   307	        else:
   308	            cp = HandyConfigParser(our_file)
   309
   310	        self.config_files_attempted.append(os.path.abspath(filename))
   311
   312	        try:
   313	            files_read = cp.read(filename)
   314	        except (configparser.Error, TomlDecodeError) as err:
   315	            raise ConfigError(f"Couldn't read config file {filename}: {err}") from err
   316	        if not files_read:
   317	            return False
   318
   319	        self.config_files_read.extend(map(os.path.abspath, files_read))
   320
   321	        any_set = False
   322	        try:
   323	            for option_spec in self.CONFIG_FILE_OPTIONS:
   324	                was_set = self._set_attr_from_config_option(cp, *option_spec)
   325	                if was_set:
   326	                    any_set = True
   327	        except ValueError as err:
   328	            raise ConfigError(f"Couldn't read config file {filename}: {err}") from err
   329
   330	        # Check that there are no unrecognized options.
   331	        all_options = collections.defaultdict(set)
   332	        for option_spec in self.CONFIG_FILE_OPTIONS:
   333	            section, option = option_spec[1].split(":")
   334	            all_options[section].add(option)
   335
   336	        for section, options in all_options.items():
   337	            real_section = cp.real_section(section)
   338	            if real_section:
   339	                for unknown in set(cp.options(section)) - options:
   340	                    warn(
   341	                        "Unrecognized option '[{}] {}=' in config file {}".format(
   342	                            real_section, unknown, filename,
   343	                        ),
   344	                    )
   345
   346	        # [paths] is special
   347	        if cp.has_section("paths"):
   348	            for option in cp.options("paths"):
   349	                self.paths[option] = cp.getlist("paths", option)
   350	                any_set = True
   351
   352	        # plugins can have options
   353	        for plugin in self.plugins:
   354	            if cp.has_section(plugin):
   355	                self.plugin_options[plugin] = cp.get_section(plugin)
   356	                any_set = True
   357
   358	        # Was this file used as a config file? If it's specifically our file,
   359	        # then it was used.  If we're piggybacking on someone else's file,
   360	        # then it was only used if we found some settings in it.
   361	        if our_file:
   362	            used = True
   363	        else:
   364	            used = any_set
   365
   366	        if used:
   367	            self.config_file = os.path.abspath(filename)
   368	            with open(filename, "rb") as f:
   369	                self._config_contents = f.read()
   370
   371	        return used

   456	    def _set_attr_from_config_option(
   457	        self,
   458	        cp: TConfigParser,
   459	        attr: str,
   460	        where: str,
   461	        type_: str = "",
   462	    ) -> bool:
   463	        """Set an attribute on self if it exists in the ConfigParser.
   464
   465	        Returns True if the attribute was set.
   466
   467	        """
   468	        section, option = where.split(":")
   469	        if cp.has_option(section, option):
   470	            method = getattr(cp, f"get{type_}")
   471	            setattr(self, attr, method(section, option))
   472	            return True
   473	        return False

   479	    def set_option(self, option_name: str, value: TConfigValueIn | TConfigSectionIn) -> None:
   480	        """Set an option in the configuration.
   481
   482	        `option_name` is a colon-separated string indicating the section and
   483	        option name.  For example, the ``branch`` option in the ``[run]``
   484	        section of the config file would be indicated with `"run:branch"`.
   485
   486	        `value` is the new value for the option.
   487
   488	        """
   489	        # Special-cased options.
   490	        if option_name == "paths":
   491	            # This is ugly, but type-checks and ensures the values are close
   492	            # to right.
   493	            self.paths = {}
   494	            assert isinstance(value, Mapping)
   495	            for k, v in value.items():
   496	                assert isinstance(v, Iterable)
   497	                self.paths[k] = list(v)
   498	            return
   499
   500	        # Check all the hard-coded options.
   501	        for option_spec in self.CONFIG_FILE_OPTIONS:
   502	            attr, where = option_spec[:2]
   503	            if where == option_name:
   504	                setattr(self, attr, value)
   505	                return
   506
   507	        # See if it's a plugin option.
   508	        plugin_name, _, key = option_name.partition(":")
   509	        if key and plugin_name in self.plugins:
   510	            self.plugin_options.setdefault(plugin_name, {})[key] = value # type: ignore[index]
   511	            return
   512
   513	        # If we get here, we didn't find the option.
   514	        raise ConfigError(f"No such option: {option_name!r}")
   515
   516	    def get_option(self, option_name: str) -> TConfigValueOut | None:
   517	        """Get an option from the configuration.
   518
   519	        `option_name` is a colon-separated string indicating the section and
   520	        option name.  For example, the ``branch`` option in the ``[run]``
   521	        section of the config file would be indicated with `"run:branch"`.
   522
   523	        Returns the value of the option.
   524
   525	        """
   526	        # Special-cased options.
   527	        if option_name == "paths":
   528	            return self.paths
   529
   530	        # Check all the hard-coded options.
   531	        for option_spec in self.CONFIG_FILE_OPTIONS:
   532	            attr, where = option_spec[:2]
   533	            if where == option_name:
   534	                return getattr(self, attr)  # type: ignore[no-any-return]
   535
   536	        # See if it's a plugin option.
   537	        plugin_name, _, key = option_name.partition(":")
   538	        if key and plugin_name in self.plugins:
   539	            return self.plugin_options.get(plugin_name, {}).get(key)
   540
   541	        # If we get here, we didn't find the option.
   542	        raise ConfigError(f"No such option: {option_name!r}")

   563	    def serialize(self) -> str:
   564	        """Convert to a string that can be ingested with `deserialize_config`.
   565
   566	        File paths used by `coverage run` are made absolute to ensure the
   567	        deserialized config will refer to the same files.
   568	        """
   569	        data = {k:v for k, v in self.__dict__.items() if not k.startswith("_")}
   570	        for k in self.SERIALIZE_ABSPATH:
   571	            v = data[k]
   572	            if isinstance(v, list):
   573	                v = list(map(os.path.abspath, v))
   574	            elif isinstance(v, str):
   575	                v = os.path.abspath(v)
   576	            data[k] = v
   577	        return base64.b64encode(json.dumps(data).encode()).decode()

   585	def process_regexlist(name: str, option: str, values: list[str]) -> list[str]:
   586	    """Check the values in a regex list and keep the non-blank ones."""
   587	    value_list = []
   588	    for value in values:
   589	        value = value.strip()
   590	        try:
   591	            re.compile(value)
   592	        except re.error as e:

   594	        if value:
   595	            value_list.append(value)
   596	    return value_list

   599	def config_files_to_try(config_file: bool | str) -> list[tuple[str, bool, bool]]:
   600	    """What config files should we try to read?
   601
   602	    Returns a list of tuples:
   603	        (filename, is_our_file, was_file_specified)
   604	    """
   605
   606	    # Some API users were specifying ".coveragerc" to mean the same as
   607	    # True, so make it so.
   608	    if config_file == ".coveragerc":
   609	        config_file = True
   610	    specified_file = (config_file is not True)
   611	    if not specified_file:
   612	        # No file was specified. Check COVERAGE_RCFILE.
   613	        rcfile = os.getenv("COVERAGE_RCFILE")
   614	        if rcfile:
   615	            config_file = rcfile
   616	            specified_file = True
   617	    if not specified_file:
   618	        # Still no file specified. Default to .coveragerc
   619	        config_file = ".coveragerc"
   620	    assert isinstance(config_file, str)
   621	    files_to_try = [
   622	        (config_file, True, specified_file),
   623	        ("setup.cfg", False, False),
   624	        ("tox.ini", False, False),
   625	        ("pyproject.toml", False, False),
   626	    ]
   627	    return files_to_try
