  4780	    def visit_type_application(self, tapp: TypeApplication) -> Type:
  4781	        """Type check a type application (expr[type, ...]).
  4782
  4783	        There are two different options here, depending on whether expr refers
  4784	        to a type alias or directly to a generic class. In the first case we need
  4785	        to use a dedicated function typeanal.instantiate_type_alias(). This
  4786	        is due to slight differences in how type arguments are applied and checked.
  4787	        """
  4788	        if isinstance(tapp.expr, RefExpr) and isinstance(tapp.expr.node, TypeAlias):
  4789	            if tapp.expr.node.python_3_12_type_alias:
  4790	                return self.type_alias_type_type()
  4791	            # Subscription of a (generic) alias in runtime context, expand the alias.
  4792	            item = instantiate_type_alias(
  4793	                tapp.expr.node,
  4794	                tapp.types,
  4795	                self.chk.fail,
  4796	                tapp.expr.node.no_args,
  4797	                tapp,
  4798	                self.chk.options,
  4799	            )
  4800	            item = get_proper_type(item)
  4801	            if isinstance(item, Instance):
  4802	                tp = type_object_type(item.type, self.named_type)

  4804	            elif isinstance(item, TupleType) and item.partial_fallback.type.is_named_tuple:
  4805	                tp = type_object_type(item.partial_fallback.type, self.named_type)
  4806	                return self.apply_type_arguments_to_callable(tp, item.partial_fallback.args, tapp)
  4807	            elif isinstance(item, TypedDictType):
  4808	                return self.typeddict_callable_from_context(item)
  4809	            else:
  4810	                self.chk.fail(message_registry.ONLY_CLASS_APPLICATION, tapp)
  4811	                return AnyType(TypeOfAny.from_error)

  4813	        # This is typically used as `x = G[int]()`.
  4814	        tp = get_proper_type(self.accept(tapp.expr))
  4815	        if isinstance(tp, (CallableType, Overloaded)):
  4816	            if not tp.is_type_obj():
  4817	                self.chk.fail(message_registry.ONLY_CLASS_APPLICATION, tapp)
  4818	            return self.apply_type_arguments_to_callable(tp, tapp.types, tapp)
  4819	        if isinstance(tp, AnyType):

  4837	    def alias_type_in_runtime_context(
  4838	        self, alias: TypeAlias, *, ctx: Context, alias_definition: bool = False
  4839	    ) -> Type:
  4840	        """Get type of a type alias (could be generic) in a runtime expression.
  4841
  4842	        Note that this function can be called only if the alias appears _not_
  4843	        as a target of type application, which is treated separately in the
  4844	        visit_type_application method. Some examples where this method is called are
  4845	        casts and instantiation:
  4846
  4847	            class LongName(Generic[T]): ...
  4848	            A = LongName[int]
  4849
  4850	            x = A()
  4851	            y = cast(A, ...)
  4852	        """
  4853	        if alias.python_3_12_type_alias:
  4854	            return self.type_alias_type_type()
  4855	        if isinstance(alias.target, Instance) and alias.target.invalid:  # type: ignore[misc]
  4856	            # An invalid alias, error already has been reported
  4857	            return AnyType(TypeOfAny.from_error)
  4858	        # If this is a generic alias, we set all variables to `Any`.
  4859	        # For example:
  4860	        #     A = List[Tuple[T, T]]
  4861	        #     x = A() <- same as List[Tuple[Any, Any]], see PEP 484.
  4862	        disallow_any = self.chk.options.disallow_any_generics and self.is_callee
  4863	        item = get_proper_type(
  4864	            set_any_tvars(
  4865	                alias,
  4866	                [],
  4867	                ctx.line,
  4868	                ctx.column,
  4869	                self.chk.options,
  4870	                disallow_any=disallow_any,
  4871	                fail=self.msg.fail,
  4872	            )
  4873	        )
  4874	        if isinstance(item, Instance):
  4875	            # Normally we get a callable type (or overloaded) with .is_type_obj() true
  4876	            # representing the class's constructor
  4877	            tp = type_object_type(item.type, self.named_type)
  4878	            if alias.no_args:
  4879	                return tp
  4880	            return self.apply_type_arguments_to_callable(tp, item.args, ctx)
  4881	        elif (
  4882	            isinstance(item, TupleType)
  4883	            and
  4884	            # Tuple[str, int]() fails at runtime, only named tuples and subclasses work.
  4885	            tuple_fallback(item).type.fullname != "builtins.tuple"
  4886	        ):
  4887	            return type_object_type(tuple_fallback(item).type, self.named_type)
  4888	        elif isinstance(item, TypedDictType):
  4889	            return self.typeddict_callable_from_context(item)
  4890	        elif isinstance(item, NoneType):
  4891	            return TypeType(item, line=item.line, column=item.column)
  4892	        elif isinstance(item, AnyType):
  4893	            return AnyType(TypeOfAny.from_another_any, source_any=item)
  4894	        elif (
  4895	            isinstance(item, UnionType)
  4896	            and item.uses_pep604_syntax
  4897	            and self.chk.options.python_version >= (3, 10)
  4898	        ):
  4899	            return self.chk.named_generic_type("types.UnionType", item.items)
  4900	        else:
  4901	            if alias_definition:
  4902	                return AnyType(TypeOfAny.special_form)
  4903	            # The _SpecialForm type can be used in some runtime contexts (e.g. it may have __or__).
  4904	            return self.named_type("typing._SpecialForm")

  4906	    def split_for_callable(
  4907	        self, t: CallableType, args: Sequence[Type], ctx: Context
  4908	    ) -> list[Type]:
  4909	        """Handle directly applying type arguments to a variadic Callable.
  4910
  4911	        This is needed in situations where e.g. variadic class object appears in
  4912	        runtime context. For example:
  4913	            class C(Generic[T, Unpack[Ts]]): ...
  4914	            x = C[int, str]()
  4915
  4916	        We simply group the arguments that need to go into Ts variable into a TupleType,
  4917	        similar to how it is done in other places using split_with_prefix_and_suffix().
  4918	        """
  4919	        if t.is_type_obj():
  4920	            # Type arguments must map to class type variables, ignoring constructor vars.
  4921	            vars = t.type_object().defn.type_vars
  4922	        else:
  4923	            vars = list(t.variables)
  4924	        args = flatten_nested_tuples(args)
  4925
  4926	        # TODO: this logic is duplicated with semanal_typeargs.
  4927	        for tv, arg in zip(t.variables, args):
  4928	            if isinstance(tv, ParamSpecType):
  4929	                if not isinstance(
  4930	                    get_proper_type(arg), (Parameters, ParamSpecType, AnyType, UnboundType)
  4931	                ):
  4932	                    self.chk.fail(
  4933	                        "Can only replace ParamSpec with a parameter types list or"
  4934	                        f" another ParamSpec, got {format_type(arg, self.chk.options)}",
  4935	                        ctx,
  4936	                    )
  4937	                    return [AnyType(TypeOfAny.from_error)] * len(vars)
  4938
  4939	        if not vars or not any(isinstance(v, TypeVarTupleType) for v in vars):
  4940	            return list(args)
  4941	        # TODO: in future we may want to support type application to variadic functions.
  4942	        assert t.is_type_obj()
  4943	        info = t.type_object()
  4944	        # We reuse the logic from semanal phase to reduce code duplication.
  4945	        fake = Instance(info, args, line=ctx.line, column=ctx.column)
  4946	        # This code can be only called either from checking a type application, or from
  4947	        # checking a type alias (after the caller handles no_args aliases), so we know it
  4948	        # was initially an IndexExpr, and we allow empty tuple type arguments.
  4949	        if not validate_instance(fake, self.chk.fail, empty_tuple_index=True):
  4950	            fix_instance(
  4951	                fake, self.chk.fail, self.chk.note, disallow_any=False, options=self.chk.options
  4952	            )
  4953	            args = list(fake.args)
  4954
  4955	        prefix = next(i for (i, v) in enumerate(vars) if isinstance(v, TypeVarTupleType))
  4956	        suffix = len(vars) - prefix - 1
  4957	        tvt = vars[prefix]
  4958	        assert isinstance(tvt, TypeVarTupleType)
  4959	        start, middle, end = split_with_prefix_and_suffix(tuple(args), prefix, suffix)
  4960	        return list(start) + [TupleType(list(middle), tvt.tuple_fallback)] + list(end)

  4962	    def apply_type_arguments_to_callable(
  4963	        self, tp: Type, args: Sequence[Type], ctx: Context
  4964	    ) -> Type:
  4965	        """Apply type arguments to a generic callable type coming from a type object.
  4966
  4967	        This will first perform type arguments count checks, report the
  4968	        error as needed, and return the correct kind of Any. As a special
  4969	        case this returns Any for non-callable types, because if type object type
  4970	        is not callable, then an error should be already reported.
  4971	        """
  4972	        tp = get_proper_type(tp)
  4973
  4974	        if isinstance(tp, CallableType):
  4975	            if tp.is_type_obj():
  4976	                # If we have a class object in runtime context, then the available type
  4977	                # variables are those of the class, we don't include additional variables
  4978	                # of the constructor. So that with
  4979	                #     class C(Generic[T]):
  4980	                #         def __init__(self, f: Callable[[S], T], x: S) -> None
  4981	                # C[int] is valid
  4982	                # C[int, str] is invalid (although C as a callable has 2 type variables)
  4983	                # Note: various logic below and in applytype.py relies on the fact that
  4984	                # class type variables appear *before* constructor variables.
  4985	                type_vars = tp.type_object().defn.type_vars
  4986	            else:
  4987	                type_vars = list(tp.variables)
  4988	            min_arg_count = sum(not v.has_default() for v in type_vars)
  4989	            has_type_var_tuple = any(isinstance(v, TypeVarTupleType) for v in type_vars)
  4990	            if (
  4991	                len(args) < min_arg_count or len(args) > len(type_vars)
  4992	            ) and not has_type_var_tuple:
  4993	                if tp.is_type_obj() and tp.type_object().fullname == "builtins.tuple":
  4994	                    # e.g. expression tuple[X, Y]
  4995	                    # - want the type of the expression i.e. a function with that as its return type
  4996	                    # - tp is type of tuple (note it won't have params as we are only called
  4997	                    #   with generic callable type)
  4998	                    # - tuple[X, Y]() takes a single arg that is a tuple containing an X and a Y
  4999	                    return CallableType(
  5000	                        [TupleType(list(args), self.chk.named_type("tuple"))],
  5001	                        [ARG_POS],
  5002	                        [None],
  5003	                        TupleType(list(args), self.chk.named_type("tuple")),
  5004	                        tp.fallback,
  5005	                        name="tuple",
  5006	                        definition=tp.definition,
  5007	                        is_bound=tp.is_bound,
  5008	                    )
  5009	                self.msg.incompatible_type_application(
  5010	                    min_arg_count, len(type_vars), len(args), ctx
  5011	                )
  5012	                return AnyType(TypeOfAny.from_error)
  5013	            return self.apply_generic_arguments(tp, self.split_for_callable(tp, args, ctx), ctx)
  5014	        if isinstance(tp, Overloaded):
  5015	            for it in tp.items:
  5016	                if tp.is_type_obj():
  5017	                    # Same as above.
  5018	                    type_vars = tp.type_object().defn.type_vars
  5019	                else:
  5020	                    type_vars = list(it.variables)
  5021	                min_arg_count = sum(not v.has_default() for v in type_vars)
  5022	                has_type_var_tuple = any(isinstance(v, TypeVarTupleType) for v in type_vars)
  5023	                if (
  5024	                    len(args) < min_arg_count or len(args) > len(type_vars)
  5025	                ) and not has_type_var_tuple:
  5026	                    self.msg.incompatible_type_application(
  5027	                        min_arg_count, len(type_vars), len(args), ctx
  5028	                    )
  5029	                    return AnyType(TypeOfAny.from_error)
  5030	            return Overloaded(
  5031	                [
  5032	                    self.apply_generic_arguments(it, self.split_for_callable(it, args, ctx), ctx)
  5033	                    for it in tp.items
  5034	                ]
  5035	            )
  5036	        return AnyType(TypeOfAny.special_form)
