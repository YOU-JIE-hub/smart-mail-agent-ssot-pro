   550	        'Compute x, y for a label'
   551	        nudge = label.nudge
   552	        bt = getattr(label,'boxTarget','normal')
   553	        anti = bt=='anti'
   554	        if anti: nudge = -nudge
   555	        pm = value = height
   556	        if anti: value = 0
   557	        a = x + 0.5*width

   559	        if bt=='mid':
   560	            b = y+height*0.5
   561	        elif bt=='hi':
   562	            if value>=0:
   563	                b = y + value + nudge
   564	            else:
   565	                b = y - nudge
   566	                pm = -pm
   567	        elif bt=='lo':
   568	            if value<=0:
   569	                b = y + value + nudge
   570	            else:
   571	                b = y - nudge
   572	                pm = -pm
   573	        else:
   574	            b = y + value + nudge
   575	        label._pmv = pm #the plus minus val
