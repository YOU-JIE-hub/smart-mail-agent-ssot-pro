 reversed(bases):
   137	            if _is_base_model_class_defined and issubclass(base, BaseModel) and base != BaseModel:
   138	                fields.update(smart_deepcopy(base.__fields__))
   139	                config = inherit_config(base.__config__, config)
   140	                validators = inherit_validators(base.__validators__, validators)
   141	                pre_root_validators += base.__pre_root_validators__
   142	                post_root_validators += base.__post_root_validators__
   143	                base_private_attributes.update(base.__private_attributes__)
   144	                class_vars.update(base.__class_vars__)
   145	                hash_func = base.__hash__
   146
   147	        resolve_forward_refs = kwargs.pop('__resolve_forward_refs__', True)
   148	        allowed_config_kwargs: SetStr = {
   149	            key
   150	            for key in dir(config)
   151	            if not (key.startswith('__') and key.endswith('__'))  # skip dunder methods and attributes
   152	        }
   153	        config_kwargs = {key: kwargs.pop(key) for key in kwargs.keys() & allowed_config_kwargs}
   154	        config_from_namespace = namespace.get('Config')
   155	        if config_kwargs and config_from_namespace:
   156	            raise TypeError('Specifying config in two places is ambiguous, use either Config attribute or class kwargs')
   157	        config = inherit_config(config_from_namespace, config, **config_kwargs)
   158
   159	        validators = inherit_validators(extract_validators(namespace), validators)
   160	        vg = ValidatorGroup(validators)
   161
   162	        for f in fields.values():
   163	            f.set_config(config)
   164	            extra_validators = vg.get_validators(f.name)
   165	            if extra_validators:
   166	                f.class_validators.update(extra_validators)
   167	                # re-run prepare to add extra validators
   168	                f.populate_validators()
   169
   170	        prepare_config(config, name)
   171
   172	        untouched_types = ANNOTATED_FIELD_UNTOUCHED_TYPES
   173
   174	        def is_untouched(v: Any) -> bool:
   175	            return isinstance(v, untouched_types) or v.__class__.__name__ == 'cython_function_or_method'
   176
   177	        if (namespace.get('__module__'), namespace.get('__qualname__')) != ('pydantic.main', 'BaseModel'):
   178	            annotations = resolve_annotations(namespace.get('__annotations__', {}), namespace.get('__module__', None))
   179	            # annotation only fields need to come first in fields
   180	            for ann_name, ann_type in annotations.items():
   181	                if is_classvar(ann_type):
   182	                    class_vars.add(ann_name)
   183	                elif is_finalvar_with_default_val(ann_type, namespace.get(ann_name, Undefined)):
   184	                    class_vars.add(ann_name)
   185	                elif is_valid_field(ann_name):
   186	                    validate_field_name(bases, ann_name)
   187	                    value = namespace.get(ann_name, Undefined)
   188	                    allowed_types = get_args(ann_type) if is_union(get_origin(ann_type)) else (ann_type,)
   189	                    if (
   190	                        is_untouched(value)
   191	                        and ann_type != PyObject
   192	                        and not any(
   193	                            lenient_issubclass(get_origin(allowed_type), Type) for allowed_type in allowed_types
   194	                        )
   195	                    ):
   196	                        continue
   197	                    fields[ann_name] = ModelField.infer(
   198	                        name=ann_name,
   199	                        value=value,
   200	                        annotation=ann_type,
   201	                        class_validators=vg.get_validators(ann_name),
   202	                        config=config,
   203	                    )
   204	                elif ann_name not in namespace and config.underscore_attrs_are_private:
   205	                    private_attributes[ann_name] = PrivateAttr()
   206
   207	            untouched_types = UNTOUCHED_TYPES + config.keep_untouched
   208	            for var_name, value in namespace.items():
   209	                can_be_changed = var_name not in class_vars and not is_untouched(value)
   210	                if isinstance(value, ModelPrivateAttr):
   211	                    if not is_valid_private_name(var_name):
   212	                        raise NameError(
   213	                            f'Private attributes "{var_name}" must not be a valid field name; '
   214	                            f'Use sunder or dunder names, e. g. "_{var_name}" or "__{var_name}__"'
   215	                        )
   216	                    private_attributes[var_name] = value
   217	                elif config.underscore_attrs_are_private and is_valid_private_name(var_name) and can_be_changed:
   218	                    private_attributes[var_name] = PrivateAttr(default=value)
   219	                elif is_valid_field(var_name) and var_name not in annotations and can_be_changed:
   220	                    validate_field_name(bases, var_name)
   221	                    inferred = ModelField.infer(
   222	                        name=var_name,
   223	                        value=value,
   224	                        annotation=annotations.get(var_name, Undefined),
   225	                        class_validators=vg.get_validators(var_name),
   226	                        config=config,
   227	                    )
   228	                    if var_name in fields:
   229	                        if lenient_issubclass(inferred.type_, fields[var_name].type_):
   230	                            inferred.type_ = fields[var_name].type_
   231	                        else:
   232	                            raise TypeError(
   233	                                f'The type of {name}.{var_name} differs from the new default value; '
   234	                                f'if you wish to change the type of this field, please use a type annotation'
   235	                            )
   236	                    fields[var_name] = inferred
   237
   238	        _custom_root_type = ROOT_KEY in fields
   239	        if _custom_root_type:
   240	            validate_custom_root_type(fields)
   241	        vg.check_for_unused()
   242	        if config.json_encoders:
   243	            json_encoder = partial(custom_pydantic_encoder, config.json_encoders)
   244	        else:
   245	            json_encoder = pydantic_encoder
   246	        pre_rv_new, post_rv_new = extract_root_validators(namespace)
   247
   248	        if hash_func is None:
   249	            hash_func = generate_hash_function(config.frozen)
   250
   251	        exclude_from_namespace = fields | private_attributes.keys() | {'__slots__'}
   252	        new_namespace = {
   253	            '__config__': config,
   254	            '__fields__': fields,
   255	            '__exclude_fields__': {
   256	                name: field.field_info.exclude for name, field in fields.items() if field.field_info.exclude is not None
   257	            }
   258	            or None,
   259	            '__include_fields__': {
   260	                name: field.field_info.include for name, field in fields.items() if field.field_info.include is not None
   261	            }
   262	            or None,
   263	            '__validators__': vg.validators,
   264	            '__pre_root_validators__': unique_list(
   265	                pre_root_validators + pre_rv_new,
   266	                name_factory=lambda v: v.__name__,
   267	            ),
   268	            '__post_root_validators__': unique_list(
   269	                post_root_validators + post_rv_new,
   270	                name_factory=lambda skip_on_failure_and_v: skip_on_failure_and_v[1].__name__,
   271	            ),
   272	            '__schema_cache__': {},
   273	            '__json_encoder__': staticmethod(json_encoder),
   274	            '__custom_root_type__': _custom_root_type,
   275	            '__private_attributes__': {**base_private_attributes, **private_attributes},
   276	            '__slots__': slots | private_attributes.keys(),
   277	            '__hash__': hash_func,
   278	            '__class_vars__': class_vars,
   279	            **{n: v for n, v in namespace.items() if n not in exclude_from_namespace},
   280	        }
   281
   282	        cls = super().__new__(mcs, name, bases, new_namespace, **kwargs)
   283	        # set __signature__ attr only for model class, but not for its instances
   284	        cls.__signature__ = ClassAttribute('__signature__', generate_model_signature(cls.__init__, fields, config))
   285
   286	        if not _is_base_model_class_defined:
   287	            # Cython does not understand the `if TYPE_CHECKING:` condition in the
   288	            # BaseModel's body (where annotations are set), so clear them manually:
   289	            getattr(cls, '__annotations__', {}).clear()
   290
   291	        if resolve_forward_refs:
   292	            cls.__try_update_forward_refs__()
   293
   294	        # preserve `__set_name__` protocol defined in https://peps.python.org/pep-0487
   295	        # for attributes not in `new_namespace` (e.g. private attributes)
   296	        for name, obj in namespace.items():
   297	            if name not in new_namespace:
   298	                set_name = getattr(obj, '__set_name__', None)
   299	                if callable(set_name):
   300	                    set_name(cls, name)
   301
   302	        return cls

   338	    def __init__(__pydantic_self__, **data: Any) -> None:
   339	        """
   340	        Create a new model by parsing and validating input data from keyword arguments.
   341
   342	        Raises ValidationError if the input data cannot be parsed to form a valid model.
   343	        """
   344	        # Uses something other than `self` the first arg to allow "self" as a settable attribute
   345	        values, fields_set, validation_error = validate_model(__pydantic_self__.__class__, data)
   346	        if validation_error:
   347	            raise validation_error
   348	        try:
   349	            object_setattr(__pydantic_self__, '__dict__', values)
   350	        except TypeError as e:
   351	            raise TypeError(
   352	                'Model values must be a dict; you may not have returned a dictionary from a root validator'

   357	    @no_type_check
   358	    def __setattr__(self, name, value):  # noqa: C901 (ignore complexity)
   359	        if name in self.__private_attributes__ or name in DUNDER_ATTRIBUTES:
   360	            return object_setattr(self, name, value)
   361
   362	        if self.__config__.extra is not Extra.allow and name not in self.__fields__:
   363	            raise ValueError(f'"{self.__class__.__name__}" object has no field "{name}"')
   364	        elif not self.__config__.allow_mutation or self.__config__.frozen:
   365	            raise TypeError(f'"{self.__class__.__name__}" is immutable and does not support item assignment')
   366	        elif name in self.__fields__ and self.__fields__[name].final:
   367	            raise TypeError(
   368	                f'"{self.__class__.__name__}" object "{name}" field is final and does not support reassignment'
   369	            )
   370	        elif self.__config__.validate_assignment:
   371	            new_values = {**self.__dict__, name: value}
   372
   373	            for validator in self.__pre_root_validators__:
   374	                try:
   375	                    new_values = validator(self.__class__, new_values)
   376	                except (ValueError, TypeError, AssertionError) as exc:
   377	                    raise ValidationError([ErrorWrapper(exc, loc=ROOT_KEY)], self.__class__)
   378
   379	            known_field = self.__fields__.get(name, None)
   380	            if known_field:
   381	                # We want to
   382	                # - make sure validators are called without the current value for this field inside `values`
   383	                # - keep other values (e.g. submodels) untouched (using `BaseModel.dict()` will change them into dicts)
   384	                # - keep the order of the fields
   385	                if not known_field.field_info.allow_mutation:
   386	                    raise TypeError(f'"{known_field.name}" has allow_mutation set to False and cannot be assigned')
   387	                dict_without_original_value = {k: v for k, v in self.__dict__.items() if k != name}
   388	                value, error_ = known_field.validate(value, dict_without_original_value, loc=name, cls=self.__class__)
   389	                if error_:
   390	                    raise ValidationError([error_], self.__class__)
   391	                else:
   392	                    new_values[name] = value
   393
   394	            errors = []
   395	            for skip_on_failure, validator in self.__post_root_validators__:
   396	                if skip_on_failure and errors:
   397	                    continue
   398	                try:
   399	                    new_values = validator(self.__class__, new_values)
   400	                except (ValueError, TypeError, AssertionError) as exc:
   401	                    errors.append(ErrorWrapper(exc, loc=ROOT_KEY))
   402	            if errors:
   403	                raise ValidationError(errors, self.__class__)
   404
   405	            # update the whole __dict__ as other values than just `value`
   406	            # may be changed (e.g. with `root_validator`)
   407	            object_setattr(self, '__dict__', new_values)
   408	        else:
   409	            self.__dict__[name] = value
   410
   411	        self.__fields_set__.add(name)

   467	    def json(
   468	        self,
   469	        *,
   470	        include: Optional[Union['AbstractSetIntStr', 'MappingIntStrAny']] = None,
   471	        exclude: Optional[Union['AbstractSetIntStr', 'MappingIntStrAny']] = None,
   472	        by_alias: bool = False,
   473	        skip_defaults: Optional[bool] = None,
   474	        exclude_unset: bool = False,
   475	        exclude_defaults: bool = False,
   476	        exclude_none: bool = False,
   477	        encoder: Optional[Callable[[Any], Any]] = None,
   478	        models_as_dict: bool = True,
   479	        **dumps_kwargs: Any,
   480	    ) -> str:
   481	        """
   482	        Generate a JSON representation of the model, `include` and `exclude` arguments as per `dict()`.
   483
   484	        `encoder` is an optional function to supply as `default` to json.dumps(), other arguments as per `json.dumps()`.
   485	        """
   486	        if skip_defaults is not None:
   487	            warnings.warn(
   488	                f'{self.__class__.__name__}.json(): "skip_defaults" is deprecated and replaced by "exclude_unset"',
   489	                DeprecationWarning,
   490	            )
   491	            exclude_unset = skip_defaults
   492	        encoder = cast(Callable[[Any], Any], encoder or self.__json_encoder__)
   493
   494	        # We don't directly call `self.dict()`, which does exactly this with `to_dict=True`
   495	        # because we want to be able to keep raw `BaseModel` instances and not as `dict`.
   496	        # This allows users to write custom JSON encoders for given `BaseModel` classes.
   497	        data = dict(
   498	            self._iter(
   499	                to_dict=models_as_dict,
   500	                by_alias=by_alias,
   501	                include=include,
   502	                exclude=exclude,
   503	                exclude_unset=exclude_unset,
   504	                exclude_defaults=exclude_defaults,
   505	                exclude_none=exclude_none,
   506	            )
   507	        )
   508	        if self.__custom_root_type__:
   509	            data = data[ROOT_KEY]
   510	        return self.__config__.json_dumps(data, default=encoder, **dumps_kwargs)
