   452	    def _dataPosIterator(self):
   453	        cols = itertools.chain(range(self.moduleCount - 1, 6, -2),
   454	                               range(5, 0, -2))
   455	        rows = (list(range(9, self.moduleCount - 8)),
   456	                list(itertools.chain(range(6), range(7, self.moduleCount))),
   457	                list(range(9, self.moduleCount)))
   458	        rrows = tuple( list(reversed(r)) for r in rows)
   459
   460	        ppos = QRUtil.getPatternPosition(self.version)
   461	        ppos = set(itertools.chain.from_iterable(
   462	            (p-2, p-1, p, p+1, p+2) for p in ppos))
   463	        maxpos = self.moduleCount - 11
   464
   465	        for col in cols:
   466	            rows, rrows = rrows, rows
   467	            if col <= 8: rowidx = 0
   468	            elif col >= self.moduleCount - 8: rowidx = 2
   469	            else: rowidx = 1
   470	            for row in rows[rowidx]:
   471	                for c in range(2):
   472	                    c = col - c
   473	                    if self.version >= 7:
   474	                        if row < 6 and c >= self.moduleCount - 11:
   475	                            continue
   476	                        elif col < 6 and row >= self.moduleCount - 11:
   477	                            continue

   479	                        if not (row < 11 and (c < 11 or c > maxpos) or
   480	                            c < 11 and (row < 11 or row > maxpos)):
   481	                            continue

   515	    @staticmethod
   516	    def createData(version, errorCorrectLevel, dataList):
   517	        rsBlocks = QRRSBlock.getRSBlocks(version, errorCorrectLevel)
   518	        buffer = QRBitBuffer();
   519	        for data in dataList:
   520	            data.write(buffer, version)
   521	        # calc num max data.
   522	        totalDataCount = 0;
   523	        for block in rsBlocks:
   524	            totalDataCount += block.dataCount
   525	        if (buffer.getLengthInBits() > totalDataCount * 8):
   526	            raise Exception("code length overflow. (%d > %d)" %
   527	                            (buffer.getLengthInBits(), totalDataCount * 8))
   528	        # end code
   529	        if (buffer.getLengthInBits() + 4 <= totalDataCount * 8):
   530	            buffer.put(0, 4)
   531	        # padding
   532	        while (buffer.getLengthInBits() % 8 != 0):
   533	            buffer.putBit(False)
   534	        # padding
   535	        while (True):
   536	            if (buffer.getLengthInBits() >= totalDataCount * 8):
   537	                break
   538	            buffer.put(QRCode.PAD0, 8)
   539	            if (buffer.getLengthInBits() >= totalDataCount * 8):
   540	                break
   541	            buffer.put(QRCode.PAD1, 8)
   542	        return QRCode.createBytes(buffer, rsBlocks)

   544	    @staticmethod
   545	    def createBytes(buffer, rsBlocks):
   546	        offset = 0
   547	        maxDcCount = 0
   548	        maxEcCount = 0
   549	        totalCodeCount = 0
   550	        dcdata = []
   551	        ecdata = []
   552	        for block in rsBlocks:
   553	            totalCodeCount += block.totalCount
   554	            dcCount = block.dataCount
   555	            ecCount = block.totalCount - dcCount
   556	            maxDcCount = max(maxDcCount, dcCount)
   557	            maxEcCount = max(maxEcCount, ecCount)
   558	            dcdata.append(buffer.buffer[offset:offset+dcCount])
   559	            offset += dcCount
   560	            rsPoly = QRUtil.getErrorCorrectPolynomial(ecCount)
   561	            rawPoly = QRPolynomial(dcdata[-1], rsPoly.getLength() - 1)
   562	            modPoly = rawPoly.mod(rsPoly)
   563	            rLen = rsPoly.getLength() - 1
   564	            mLen = modPoly.getLength()
   565	            ecdata.append([ (modPoly.get(i) if i >= 0 else 0)
   566	                          for i in range(mLen - rLen, mLen) ])
   567
   568	        data = [ d for dd in itertools.chain(
   569	                zip_longest(*dcdata), zip_longest(*ecdata))
   570	                 for d in dd if d is not None]
   571	        return data

   691	    def maskScoreRule1vert(cls, modules):
   692	        score = 0
   693	        lastCount = [0]
   694	        lastRow = None
   695	        for row in modules:
   696	            # Vertical patterns
   697	            if lastRow:
   698	                changed = [a ^ b for a,b in zip(row, lastRow)]
   699	                scores = [a and (b-4+3) for a,b in
   700	                          zip_longest(changed, lastCount, fillvalue=0)
   701	                          if b >= 4]
   702	                score += sum(scores)

   704	                             for a,b in zip_longest(changed, lastCount,
   705	                                                    fillvalue=0)]
   706	            lastRow = row

   708	        score += sum([b-4+3 for b in lastCount if b >= 4])  # final counts
   709	        return score
