   234	def _dedenter(text,dedent=0):
   235	    '''
   236	    tidy up text - carefully, it is probably code.  If people want to
   237	    indent code within a source script, you can supply an arg to dedent
   238	    and it will chop off that many character, otherwise it leaves

   271	def splitLine(line_to_split, lines_splitted, maximum_length, \
   272	split_characters, new_line_characters):
   273	    # Used to implement the characters added
   274	    #at the beginning of each new line created
   275	    first_line = True
   276
   277	    # Check if the text can be splitted
   278	    while line_to_split and len(line_to_split)>0:
   279
   280	        # Index of the character where we can split
   281	        split_index = 0
   282
   283	        # Check if the line length still exceeds the maximum length
   284	        if len(line_to_split) <= maximum_length:
   285	            # Return the remaining of the line
   286	            split_index = len(line_to_split)
   287	        else:
   288	            # Iterate for each character of the line
   289	            for line_index in range(maximum_length):
   290	                # Check if the character is in the list
   291	                # of allowed characters to split on
   292	                if line_to_split[line_index] in split_characters:
   293	                    split_index = line_index + 1
   294
   295	        # If the end of the line was reached
   296	        # with no character to split on
   297	        if split_index==0:
   298	            split_index = line_index + 1
   299
   300	        if first_line:
   301	            lines_splitted.append(line_to_split[0:split_index])
   302	            first_line = False
   303	            maximum_length -= len(new_line_characters)
   304	        else:
   305	            lines_splitted.append(new_line_characters + \
   306	            line_to_split[0:split_index])
   307
   308	        # Remaining text to split
   309	        line_to_split = line_to_split[split_index:]

   312	    """This is like the HTML <PRE> tag.
   313	    It attempts to display text exactly as you typed it in a fixed width "typewriter" font.
   314	    By default the line breaks are exactly where you put them, and it will not be wrapped.
   315	    You can optionally define a maximum line length and the code will be wrapped; and

   319	        """text is the text to display. If dedent is set then common leading space
   320	        will be chopped off the front (for example if the entire text is indented
   321	        6 spaces or more then each line will have 6 spaces removed from the front).
   322	        """

   324	        self.bulletText = bulletText
   325	        self.lines = _dedenter(text,dedent)
   326	        if text and maxLineLength:

   328	                                self.lines,

   332	                        )

   337	        if bT is not None:

   339	        return "%s'''\\ \n%s''')" % (H, '\n'.join(self.lines))

   353	        #returns two Preformatted objects
   354
   355	        #not sure why they can be called with a negative height
   356	        if availHeight < self.style.leading:
   357	            return []
   358
   359	        linesThatFit = int(availHeight * 1.0 / self.style.leading)
   360
   361	        text1 = '\n'.join(self.lines[0:linesThatFit])
   362	        text2 = '\n'.join(self.lines[linesThatFit:])
   363	        style = self.style
   364	        if style.firstLineIndent != 0:
   365	            style = deepcopy(style)
   366	            style.firstLineIndent = 0
   367	        return [Preformatted(text1, self.style), Preformatted(text2, style)]

   374	        cur_x = self.style.leftIndent

   402	                 mask="auto", lazy=1, hAlign='CENTER', useDPI=False):
