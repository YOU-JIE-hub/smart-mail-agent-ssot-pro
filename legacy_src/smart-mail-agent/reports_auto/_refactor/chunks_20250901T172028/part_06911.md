
    29	    has_arcs = data.has_arcs()
    30	    statements = file_reporter.lines()
    31	    excluded = file_reporter.excluded_lines()
    32	    executed = file_reporter.translate_lines(data.lines(filename) or [])
    33
    34	    if has_arcs:
    35	        arc_possibilities_set = file_reporter.arcs()
    36	        arcs: Iterable[TArc] = data.arcs(filename) or []
    37	        arcs = file_reporter.translate_arcs(arcs)

    39	        # Reduce the set of arcs to the ones that could be branches.
    40	        dests = collections.defaultdict(set)
    41	        for fromno, tono in arc_possibilities_set:
    42	            dests[fromno].add(tono)
    43	        single_dests = {
    44	            fromno: list(tonos)[0]

    50	            if fromno != tono:
    51	                new_arcs.add((fromno, tono))
    52	            else:
    53	                if fromno in single_dests:
    54	                    new_arcs.add((fromno, single_dests[fromno]))
    55
    56	        arcs_executed_set = file_reporter.translate_arcs(new_arcs)
    57	        exit_counts = file_reporter.exit_counts()
    58	        no_branch = file_reporter.no_branch_lines()
    59	    else:
    60	        arc_possibilities_set = set()
    61	        arcs_executed_set = set()
    62	        exit_counts = {}
    63	        no_branch = set()
    64
    65	    return Analysis(
    66	        precision=precision,
    67	        filename=filename,
    68	        has_arcs=has_arcs,
    69	        statements=statements,
    70	        excluded=excluded,
    71	        executed=executed,
    72	        arc_possibilities_set=arc_possibilities_set,
    73	        arcs_executed_set=arcs_executed_set,

   319	def display_covered(pc: float, precision: int) -> str:
   320	    """Return a displayable total percentage, as a string.
   321
   322	    Note that "0" is only returned when the value is truly zero, and "100"
   323	    is only returned when the value is truly 100.  Rounding can never
   324	    result in either "0" or "100".

   326	    """
   327	    near0 = 1.0 / 10 ** precision
   328	    if 0 < pc < near0:
   329	        pc = near0
   330	    elif (100.0 - near0) < pc < 100:

   337	def _line_ranges(
   338	    statements: Iterable[TLineNo],
   339	    lines: Iterable[TLineNo],
   340	) -> list[tuple[TLineNo, TLineNo]]:
   341	    """Produce a list of ranges for `format_lines`."""
   342	    statements = sorted(statements)
   343	    lines = sorted(lines)
   344
   345	    pairs = []
   346	    start: TLineNo | None = None
   347	    lidx = 0
   348	    for stmt in statements:
   349	        if lidx >= len(lines):
   350	            break
   351	        if stmt == lines[lidx]:
   352	            lidx += 1
   353	            if not start:
   354	                start = stmt
   355	            end = stmt
   356	        elif start:
   357	            pairs.append((start, end))
   358	            start = None
   359	    if start:
   360	        pairs.append((start, end))
   361	    return pairs

   364	def format_lines(
   365	    statements: Iterable[TLineNo],
   366	    lines: Iterable[TLineNo],
   367	    arcs: Iterable[tuple[TLineNo, list[TLineNo]]] | None = None,
   368	) -> str:
   369	    """Nicely format a list of line numbers.
   370
   371	    Format a list of line numbers for printing by coalescing groups of lines as
   372	    long as the lines represent consecutive statements.  This will coalesce
   373	    even if there are gaps between statements.
   374
   375	    For example, if `statements` is [1,2,3,4,5,10,11,12,13,14] and
   376	    `lines` is [1,2,5,10,11,13,14] then the result will be "1-2, 5-11, 13-14".
   377
   378	    Both `lines` and `statements` can be any iterable. All of the elements of
   379	    `lines` must be in `statements`, and all of the values must be positive
   380	    integers.
   381
   382	    If `arcs` is provided, they are (start,[end,end,end]) pairs that will be
   383	    included in the output as long as start isn't in `lines`.
   384
   385	    """
   386	    line_items = [(pair[0], nice_pair(pair)) for pair in _line_ranges(statements, lines)]
   387	    if arcs is not None:
   388	        line_exits = sorted(arcs)
   389	        for line, exits in line_exits:
   390	            for ex in sorted(exits):
   391	                if line not in lines and ex not in lines:
   392	                    dest = (ex if ex > 0 else "exit")
   393	                    line_items.append((line, f"{line}->{dest}"))
   394
   395	    ret = ", ".join(t[-1] for t in sorted(line_items))
   396	    return ret

   400	    """Determine if a total should fail due to fail-under.
   401
   402	    `total` is a float, the coverage measurement total. `fail_under` is the
   403	    fail_under setting to compare with. `precision` is the number of digits
   404	    to consider after the decimal point.
   405
   406	    Returns True if the total should fail.
   407
   408	    """
   409	    # We can never achieve higher than 100% coverage, or less than zero.
   410	    if not (0 <= fail_under <= 100.0):
   411	        msg = f"fail_under={fail_under} is invalid. Must be between 0 and 100."
   412	        raise ConfigError(msg)
   413
   414	    # Special case for fail_under=100, it must really be 100.
   415	    if fail_under == 100.0 and total != 100.0:
   416	        return True
   417
   418	    return round(total, precision) < fail_under
