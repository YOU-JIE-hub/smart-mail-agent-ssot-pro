   124	def compare_symbol_table_snapshots(
   125	    name_prefix: str, snapshot1: dict[str, SymbolSnapshot], snapshot2: dict[str, SymbolSnapshot]
   126	) -> set[str]:
   127	    """Return names that are different in two snapshots of a symbol table.
   128
   129	    Only shallow (intra-module) differences are considered. References to things defined
   130	    outside the module are compared based on the name of the target only.
   131
   132	    Recurse into class symbol tables (if the class is defined in the target module).
   133
   134	    Return a set of fully-qualified names (e.g., 'mod.func' or 'mod.Class.method').
   135	    """
   136	    # Find names only defined only in one version.
   137	    names1 = {f"{name_prefix}.{name}" for name in snapshot1}
   138	    names2 = {f"{name_prefix}.{name}" for name in snapshot2}
   139	    triggers = names1 ^ names2
   140
   141	    # Look for names defined in both versions that are different.
   142	    for name in set(snapshot1.keys()) & set(snapshot2.keys()):
   143	        item1 = snapshot1[name]
   144	        item2 = snapshot2[name]
   145	        kind1 = item1[0]
   146	        kind2 = item2[0]
   147	        item_name = f"{name_prefix}.{name}"
   148	        if kind1 != kind2:
   149	            # Different kind of node in two snapshots -> trivially different.
   150	            triggers.add(item_name)
   151	        elif kind1 == "TypeInfo":
   152	            if item1[:-1] != item2[:-1]:
   153	                # Record major difference (outside class symbol tables).
   154	                triggers.add(item_name)
   155	            # Look for differences in nested class symbol table entries.
   156	            assert isinstance(item1[-1], dict)
   157	            assert isinstance(item2[-1], dict)
   158	            triggers |= compare_symbol_table_snapshots(item_name, item1[-1], item2[-1])
   159	        else:
   160	            # Shallow node (no interesting internal structure). Just use equality.
   161	            if snapshot1[name] != snapshot2[name]:
   162	                triggers.add(item_name)
   163
   164	    return triggers
