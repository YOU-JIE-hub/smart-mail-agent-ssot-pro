 not contribute into the min_len.
   172			# For example a, b, *c = g
   173			# g must have at least 2 elements, not 3. 'c' is empyt if g has only 2.
   174			min_len = len([ob for ob in tpl.elts if not self.is_starred(ob)])
   175			offset = 0
   176
   177			for idx, val in enumerate(tpl.elts):
   178				# After a starred element specify the child index from the back.
   179				# Since it is unknown how many elements from the sequence are
   180				# consumed by the starred element.
   181				# For example a, *b, (c, d) = g
   182				# Then (c, d) has the index '-1'
   183				if self.is_starred(val):
   184					offset = min_len + 1
   185
   186				elif isinstance(val, ast.Tuple):
   187					el = ast.Tuple([], ast.Load())
   188					el.elts.append(ast.Num(idx - offset))
   189					el.elts.append(self.gen_unpack_spec(val))
   190					spec.values[0].elts.append(el)
   191
   192			spec.keys.append(ast.Str('min_len'))
   193			spec.values.append(ast.Num(min_len))
   194
   195			return spec
   196

   199			return ast.Call(
   200				func=ast.Name('__rl_unpack_sequence__', ast.Load()),
   201				args=[value, spec, ast.Name('__rl_getiter__', ast.Load())],
   202				keywords=[])

   204		def gen_unpack_wrapper(self, node, target, ctx='store'):
   205			"""Helper function to protect tuple unpacks.
   206
   207			node: used to copy the locations for the new nodes.
   208			target: is the tuple which must be protected.
   209			ctx: Defines the context of the returned temporary node.
   210
   211			It returns a tuple with two element.
   212
   213			Element 1: Is a temporary name node which must be used to
   214					   replace the target.
   215					   The context (store, param) is defined
   216					   by the 'ctx' parameter..
   217
   218			Element 2: Is a try .. finally where the body performs the
   219					   protected tuple unpack of the temporary variable
   220					   into the original target.
   221			"""
   222
   223			# Generate a tmp name to replace the tuple with.
   224			tnam = self.tmpName
   225
   226			# Generates an expressions which protects the unpack.
   227			# converter looks like 'wrapper(tnam)'.
   228			# 'wrapper' takes care to protect sequence unpacking with __rl_getiter__.
   229			converter = self.protect_unpack_sequence(
   230				target,
   231				ast.Name(tnam, ast.Load()))
   232
   233			# Assign the expression to the original names.
   234			# Cleanup the temporary variable.
   235			# Generates:
   236			# try:
   237			#	  # converter is 'wrapper(tnam)'
   238			#	  arg = converter
   239			# finally:
   240			#	  del tmp_arg
   241			try_body = [ast.Assign(targets=[target], value=converter)]
   242			finalbody = [self.gen_del_stmt(tnam)]
   243
   244			cleanup = ast.Try(
   245				body=try_body, finalbody=finalbody, handlers=[], orelse=[])
   246
   247			if ctx == 'store':
   248				ctx = ast.Store()
   249			elif ctx == 'param':
   250				ctx = ast.Param()
   251			else:  # pragma: no cover
   252				# Only store and param are defined ctx.
   253				raise NotImplementedError('bad ctx "%s"' % type(ctx))
   254
   255			# This node is used to catch the tuple in a tmp variable.
   256			tmp_target = ast.Name(tnam, ctx)
   257
   258			copy_locations(tmp_target, node)
   259			copy_locations(cleanup, node)
   260
   261			return (tmp_target, cleanup)

   276			"""Transform slices into function parameters.
   277
   278			ast.Slice nodes are only allowed within a ast.Subscript node.
   279			To use a slice as an argument of ast.Call it has to be converted.
   280			Conversion is done by calling the 'slice' function from builtins
   281			"""

   283			if isinstance(slice_, ast.Index):

   286			elif isinstance(slice_, ast.Slice):

   406		def visit_Call(self, node):
   407			"""Checks calls with '*args' and '**kwargs'.
   408
   409			Note: The following happens only if '*args' or '**kwargs' is used.
   410
   411			Transfroms 'foo(<all the possible ways of args>)' into
   412			__rl_apply__(foo, <all the possible ways for args>)
   413
   414			The thing is that '__rl_apply__' has only '*args', '**kwargs', so it gets
   415			Python to collapse all the myriad ways to call functions
   416			into one manageable from.
   417
   418			From there, '__rl_apply__()' wraps args and kws in guarded accessors,
   419			then calls the function, returning the value.
   420			"""
   421
   422			if isinstance(node.func, ast.Name):
   423				if node.func.id == 'exec':
   424					self.error(node, 'Exec calls are not allowed.')
   425				elif node.func.id == 'eval':
   426					self.error(node, 'Eval calls are not allowed.')
   427
   428			needs_wrap = False
   429
   430			for pos_arg in node.args:
   431				if isinstance(pos_arg, ast.Starred):
   432					needs_wrap = True
   433
   434			for keyword_arg in node.keywords:
   435				if keyword_arg.arg is None:
   436					needs_wrap = True
   437
   438			node = self.visit_children(node)
   439
   440			#if not needs_wrap:
   441			#	return node
   442
   443			node.args.insert(0, node.func)
   444			node.func = ast.Name('__rl_apply__', ast.Load())
   445			copy_locations(node.func, node.args[0])
   446			return node

   525		def visit_Assign(self, node):
   526			node = self.visit_children(node)
   527
   528			if not any(isinstance(t, ast.Tuple) for t in node.targets):
   529				return node
   530
   531			# Handle sequence unpacking.
   532			# For briefness this example omits cleanup of the temporary variables.
   533			# Check 'transform_tuple_assign' how its done.
   534			#
   535			# - Single target (with nested support)
   536			# (a, (b, (c, d))) = <exp>
   537			# is converted to
   538			# (a, t1) = __rl_getiter__(<exp>)
   539			# (b, t2) = __rl_getiter__(t1)
   540			# (c, d) = __rl_getiter__(t2)
   541			#
   542			# - Multi targets
   543			# (a, b) = (c, d) = <exp>
   544			# is converted to
   545			# (c, d) = __rl_getiter__(<exp>)
   546			# (a, b) = __rl_getiter__(<exp>)
   547			# Why is this valid ? The original bytecode for this multi targets
   548			# behaves the same way.
   549
   550			# ast.NodeTransformer works with list results.
   551			# He injects it at the rightplace of the node's parent statements.
   552			new_nodes = []
   553
   554			# python fills the right most target first.
   555			for target in reversed(node.targets):
   556				if isinstance(target, ast.Tuple):
   557					wrapper = ast.Assign(
   558						targets=[target],
   559						value=self.protect_unpack_sequence(target, node.value))
   560					new_nodes.append(wrapper)
   561				else:
   562					new_node = ast.Assign(targets=[target], value=node.value)
   563					new_nodes.append(new_node)
   564
   565			for new_node in new_nodes:
   566				copy_locations(new_node, node)
   567
   568			return new_nodes
