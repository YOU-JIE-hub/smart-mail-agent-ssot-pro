   134	def get_scope_node(node: nodes.Node, scope: Scope) -> nodes.Node | None:
   135	    """Get the closest parent node (including self) which matches the given
   136	    scope.
   137
   138	    If there is no parent node for the scope (e.g. asking for class scope on a
   139	    Module, or on a Function when not defined in a class), returns None.
   140	    """
   141	    import _pytest.python
   142
   143	    if scope is Scope.Function:
   144	        # Type ignored because this is actually safe, see:
   145	        # https://github.com/python/mypy/issues/4717
   146	        return node.getparent(nodes.Item)  # type: ignore[type-abstract]
   147	    elif scope is Scope.Class:
   148	        return node.getparent(_pytest.python.Class)
   149	    elif scope is Scope.Module:
   150	        return node.getparent(_pytest.python.Module)
   151	    elif scope is Scope.Package:
   152	        return node.getparent(_pytest.python.Package)
   153	    elif scope is Scope.Session:
   154	        return node.getparent(_pytest.main.Session)
   155	    else:
   156	        assert_never(scope)

   173	@dataclasses.dataclass(frozen=True)
   174	class ParamArgKey:
   175	    """A key for a high-scoped parameter used by an item.
   176
   177	    For use as a hashable key in `reorder_items`. The combination of fields
   178	    is meant to uniquely identify a particular "instance" of a param,
   179	    potentially shared by multiple items in a scope.
   180	    """
   181
   182	    #: The param name.
   183	    argname: str
   184	    param_index: int
   185	    #: For scopes Package, Module, Class, the path to the file (directory in
   186	    #: Package's case) of the package/module/class where the item is defined.
   187	    scoped_item_path: Path | None
   188	    #: For Class scope, the class where the item is defined.
   189	    item_cls: type | None
   190
   191
   192	_V = TypeVar("_V")
   193	OrderedSet = dict[_V, None]
   194
   195
   196	def get_param_argkeys(item: nodes.Item, scope: Scope) -> Iterator[ParamArgKey]:
   197	    """Return all ParamArgKeys for item matching the specified high scope."""
   198	    assert scope is not Scope.Function
   199
   200	    try:
   201	        callspec: CallSpec2 = item.callspec  # type: ignore[attr-defined]
   202	    except AttributeError:
   203	        return
   204
   205	    item_cls = None
   206	    if scope is Scope.Session:
   207	        scoped_item_path = None
   208	    elif scope is Scope.Package:
   209	        # Package key = module's directory.
   210	        scoped_item_path = item.path.parent
   211	    elif scope is Scope.Module:
   212	        scoped_item_path = item.path
   213	    elif scope is Scope.Class:
   214	        scoped_item_path = item.path
   215	        item_cls = item.cls  # type: ignore[attr-defined]
   216	    else:
   217	        assert_never(scope)
   218
   219	    for argname in callspec.indices:
   220	        if callspec._arg2scope[argname] != scope:
   221	            continue
   222	        param_index = callspec.indices[argname]
   223	        yield ParamArgKey(argname, param_index, scoped_item_path, item_cls)

   226	def reorder_items(items: Sequence[nodes.Item]) -> list[nodes.Item]:
   227	    argkeys_by_item: dict[Scope, dict[nodes.Item, OrderedSet[ParamArgKey]]] = {}
   228	    items_by_argkey: dict[Scope, dict[ParamArgKey, OrderedDict[nodes.Item, None]]] = {}
   229	    for scope in HIGH_SCOPES:
   230	        scoped_argkeys_by_item = argkeys_by_item[scope] = {}
   231	        scoped_items_by_argkey = items_by_argkey[scope] = defaultdict(OrderedDict)
   232	        for item in items:
   233	            argkeys = dict.fromkeys(get_param_argkeys(item, scope))
   234	            if argkeys:
   235	                scoped_argkeys_by_item[item] = argkeys
   236	                for argkey in argkeys:
   237	                    scoped_items_by_argkey[argkey][item] = None
   238
   239	    items_set = dict.fromkeys(items)
   240	    return list(
   241	        reorder_items_atscope(
   242	            items_set, argkeys_by_item, items_by_argkey, Scope.Session
   243	        )
   244	    )

   247	def reorder_items_atscope(
   248	    items: OrderedSet[nodes.Item],
   249	    argkeys_by_item: Mapping[Scope, Mapping[nodes.Item, OrderedSet[ParamArgKey]]],
   250	    items_by_argkey: Mapping[
   251	        Scope, Mapping[ParamArgKey, OrderedDict[nodes.Item, None]]
   252	    ],
   253	    scope: Scope,
   254	) -> OrderedSet[nodes.Item]:
   255	    if scope is Scope.Function or len(items) < 3:
   256	        return items
   257
   258	    scoped_items_by_argkey = items_by_argkey[scope]
   259	    scoped_argkeys_by_item = argkeys_by_item[scope]
   260
   261	    ignore: set[ParamArgKey] = set()
   262	    items_deque = deque(items)
   263	    items_done: OrderedSet[nodes.Item] = {}
   264	    while items_deque:
   265	        no_argkey_items: OrderedSet[nodes.Item] = {}
   266	        slicing_argkey = None
   267	        while items_deque:
   268	            item = items_deque.popleft()
   269	            if item in items_done or item in no_argkey_items:
   270	                continue
   271	            argkeys = dict.fromkeys(
   272	                k for k in scoped_argkeys_by_item.get(item, ()) if k not in ignore
   273	            )
   274	            if not argkeys:
   275	                no_argkey_items[item] = None
   276	            else:
   277	                slicing_argkey, _ = argkeys.popitem()
   278	                # We don't have to remove relevant items from later in the
   279	                # deque because they'll just be ignored.
   280	                matching_items = [
   281	                    i for i in scoped_items_by_argkey[slicing_argkey] if i in items
   282	                ]
   283	                for i in reversed(matching_items):
   284	                    items_deque.appendleft(i)
   285	                    # Fix items_by_argkey order.
   286	                    for other_scope in HIGH_SCOPES:
   287	                        other_scoped_items_by_argkey = items_by_argkey[other_scope]
   288	                        for argkey in argkeys_by_item[other_scope].get(i, ()):
   289	                            argkey_dict = other_scoped_items_by_argkey[argkey]
   290	                            if not hasattr(sys, "pypy_version_info"):
   291	                                argkey_dict[i] = None
   292	                                argkey_dict.move_to_end(i, last=False)
   293	                            else:
   294	                                # Work around a bug in PyPy:
   295	                                # https://github.com/pypy/pypy/issues/5257
   296	                                # https://github.com/pytest-dev/pytest/issues/13312
   297	                                bkp = argkey_dict.copy()
   298	                                argkey_dict.clear()
   299	                                argkey_dict[i] = None
   300	                                argkey_dict.update(bkp)
   301	                break
   302	        if no_argkey_items:
   303	            reordered_no_argkey_items = reorder_items_atscope(
   304	                no_argkey_items, argkeys_by_item, items_by_argkey, scope.next_lower()
   305	            )
   306	            items_done.update(reordered_no_argkey_items)
   307	        if slicing_argkey is not None:
   308	            ignore.add(slicing_argkey)
   309	    return items_done

   312	@dataclasses.dataclass(frozen=True)
   313	class FuncFixtureInfo:
   314	    """Fixture-related information for a fixture-requesting item (e.g. test
   315	    function).
   316
   317	    This is used to examine the fixtures which an item requests statically
   318	    (known during collection). This includes autouse fixtures, fixtures
   319	    requested by the `usefixtures` marker, fixtures requested in the function
   320	    parameters, and the transitive closure of these.
   321
   322	    An item may also request fixtures dynamically (using `request.getfixturevalue`);
   323	    these are not reflected here.
   324	    """
   325
   326	    __slots__ = ("argnames", "initialnames", "name2fixturedefs", "names_closure")
   327
   328	    # Fixture names that the item requests directly by function parameters.
   329	    argnames: tuple[str, ...]
   330	    # Fixture names that the item immediately requires. These include
   331	    # argnames + fixture names specified via usefixtures and via autouse=True in
   332	    # fixture definitions.
   333	    initialnames: tuple[str, ...]
   334	    # The transitive closure of the fixture names that the item requires.
   335	    # Note: can't include dynamic dependencies (`request.getfixturevalue` calls).
   336	    names_closure: list[str]
   337	    # A map from a fixture name in the transitive closure to the FixtureDefs
   338	    # matching the name which are applicable to this function.
   339	    # There may be multiple overriding fixtures with the same name. The
   340	    # sequence is ordered from furthest to closes to the function.
   341	    name2fixturedefs: dict[str, Sequence[FixtureDef[Any]]]
   342
   343	    def prune_dependency_tree(self) -> None:
   344	        """Recompute names_closure from initialnames and name2fixturedefs.
   345
   346	        Can only reduce names_closure, which means that the new closure will
   347	        always be a subset of the old one. The order is preserved.
   348
   349	        This method is needed because direct parametrization may shadow some
   350	        of the fixtures that were included in the originally built dependency
   351	        tree. In this way the dependency tree can get pruned, and the closure
   352	        of argnames may get reduced.
   353	        """
   354	        closure: set[str] = set()
   355	        working_set = set(self.initialnames)
   356	        while working_set:
   357	            argname = working_set.pop()
   358	            # Argname may be something not included in the original names_closure,
   359	            # in which case we ignore it. This currently happens with pseudo
   360	            # FixtureDefs which wrap 'get_direct_param_fixture_func(request)'.
   361	            # So they introduce the new dependency 'request' which might have
   362	            # been missing in the original tree (closure).
   363	            if argname not in closure and argname in self.names_closure:
   364	                closure.add(argname)
   365	                if argname in self.name2fixturedefs:
   366	                    working_set.update(self.name2fixturedefs[argname][-1].argnames)
   367
   368	        self.names_closure[:] = sorted(closure, key=self.names_closure.index)

   371	class FixtureRequest(abc.ABC):
   372	    """The type of the ``request`` fixture.
   373
   374	    A request object gives access to the requesting test context and has a
   375	    ``param`` attribute in case the fixture is parametrized.
   376	    """
   377
   378	    def __init__(
   379	        self,
   380	        pyfuncitem: Function,
   381	        fixturename: str | None,
   382	        arg2fixturedefs: dict[str, Sequence[FixtureDef[Any]]],
   383	        fixture_defs: dict[str, FixtureDef[Any]],
   384	        *,
   385	        _ispytest: bool = False,
   386	    ) -> None:
   387	        check_ispytest(_ispytest)
   388	        #: Fixture for which this request is being performed.
   389	        self.fixturename: Final = fixturename
   390	        self._pyfuncitem: Final = pyfuncitem
   391	        # The FixtureDefs for each fixture name requested by this item.
   392	        # Starts from the statically-known fixturedefs resolved during
   393	        # collection. Dynamically requested fixtures (using
   394	        # `request.getfixturevalue("foo")`) are added dynamically.
   395	        self._arg2fixturedefs: Final = arg2fixturedefs
   396	        # The evaluated argnames so far, mapping to the FixtureDef they resolved
   397	        # to.
   398	        self._fixture_defs: Final = fixture_defs
   399	        # Notes on the type of `param`:
   400	        # -`request.param` is only defined in parametrized fixtures, and will raise
   401	        #   AttributeError otherwise. Python typing has no notion of "undefined", so
   402	        #   this cannot be reflected in the type.
   403	        # - Technically `param` is only (possibly) defined on SubRequest, not
   404	        #   FixtureRequest, but the typing of that is still in flux so this cheats.
   405	        # - In the future we might consider using a generic for the param type, but
   406	        #   for now just using Any.
   407	        self.param: Any
