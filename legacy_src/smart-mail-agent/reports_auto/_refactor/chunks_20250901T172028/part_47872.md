  2882	def dispatch(sources: list[BuildSource], manager: BuildManager, stdout: TextIO) -> Graph:
  2883	    log_configuration(manager, sources)
  2884
  2885	    t0 = time.time()
  2886	    graph = load_graph(sources, manager)
  2887
  2888	    # This is a kind of unfortunate hack to work around some of fine-grained's
  2889	    # fragility: if we have loaded less than 50% of the specified files from
  2890	    # cache in fine-grained cache mode, load the graph again honestly.
  2891	    # In this case, we just turn the cache off entirely, so we don't need
  2892	    # to worry about some files being loaded and some from cache and so
  2893	    # that fine-grained mode never *writes* to the cache.
  2894	    if manager.use_fine_grained_cache() and len(graph) < 0.50 * len(sources):
  2895	        manager.log("Redoing load_graph without cache because too much was missing")
  2896	        manager.cache_enabled = False
  2897	        graph = load_graph(sources, manager)
  2898
  2899	    t1 = time.time()
  2900	    manager.add_stats(
  2901	        graph_size=len(graph),
  2902	        stubs_found=sum(g.path is not None and g.path.endswith(".pyi") for g in graph.values()),
  2903	        graph_load_time=(t1 - t0),
  2904	        fm_cache_size=len(manager.find_module_cache.results),
  2905	    )
  2906	    if not graph:
  2907	        print("Nothing to do?!", file=stdout)
  2908	        return graph
  2909	    manager.log(f"Loaded graph with {len(graph)} nodes ({t1 - t0:.3f} sec)")
  2910	    if manager.options.dump_graph:
  2911	        dump_graph(graph, stdout)
  2912	        return graph
  2913
  2914	    # Fine grained dependencies that didn't have an associated module in the build
  2915	    # are serialized separately, so we read them after we load the graph.
  2916	    # We need to read them both for running in daemon mode and if we are generating
  2917	    # a fine-grained cache (so that we can properly update them incrementally).
  2918	    # The `read_deps_cache` will also validate
  2919	    # the deps cache against the loaded individual cache files.
  2920	    if manager.options.cache_fine_grained or manager.use_fine_grained_cache():
  2921	        t2 = time.time()
  2922	        fg_deps_meta = read_deps_cache(manager, graph)
  2923	        manager.add_stats(load_fg_deps_time=time.time() - t2)
  2924	        if fg_deps_meta is not None:
  2925	            manager.fg_deps_meta = fg_deps_meta
  2926	        elif manager.stats.get("fresh_metas", 0) > 0:
  2927	            # Clear the stats so we don't infinite loop because of positive fresh_metas
  2928	            manager.stats.clear()
  2929	            # There were some cache files read, but no fine-grained dependencies loaded.
  2930	            manager.log("Error reading fine-grained dependencies cache -- aborting cache load")
  2931	            manager.cache_enabled = False
  2932	            manager.log("Falling back to full run -- reloading graph...")
  2933	            return dispatch(sources, manager, stdout)
  2934
  2935	    # If we are loading a fine-grained incremental mode cache, we
  2936	    # don't want to do a real incremental reprocess of the
  2937	    # graph---we'll handle it all later.
  2938	    if not manager.use_fine_grained_cache():
  2939	        process_graph(graph, manager)
  2940	        # Update plugins snapshot.
  2941	        write_plugins_snapshot(manager)
  2942	        manager.old_plugins_snapshot = manager.plugins_snapshot
  2943	        if manager.options.cache_fine_grained or manager.options.fine_grained_incremental:
  2944	            # If we are running a daemon or are going to write cache for further fine grained use,
  2945	            # then we need to collect fine grained protocol dependencies.
  2946	            # Since these are a global property of the program, they are calculated after we
  2947	            # processed the whole graph.
  2948	            type_state.add_all_protocol_deps(manager.fg_deps)
  2949	            if not manager.options.fine_grained_incremental:
  2950	                rdeps = generate_deps_for_cache(manager, graph)
  2951	                write_deps_cache(rdeps, manager, graph)
  2952
  2953	    if manager.options.dump_deps:
  2954	        # This speeds up startup a little when not using the daemon mode.
  2955	        from mypy.server.deps import dump_all_dependencies
  2956
  2957	        dump_all_dependencies(
  2958	            manager.modules, manager.all_types, manager.options.python_version, manager.options
  2959	        )
  2960
  2961	    return graph

  3004	def dump_graph(graph: Graph, stdout: TextIO | None = None) -> None:
  3005	    """Dump the graph as a JSON string to stdout.
  3006
  3007	    This copies some of the work by process_graph()
  3008	    (sorted_components() and order_ascc()).
  3009	    """
  3010	    stdout = stdout or sys.stdout
  3011	    nodes = []
  3012	    sccs = sorted_components(graph)
  3013	    for i, ascc in enumerate(sccs):
  3014	        scc = order_ascc(graph, ascc)
  3015	        node = NodeInfo(i, scc)
  3016	        nodes.append(node)
  3017	    inv_nodes = {}  # module -> node_id
  3018	    for node in nodes:
  3019	        for mod in node.scc:
  3020	            inv_nodes[mod] = node.node_id
  3021	    for node in nodes:
  3022	        for mod in node.scc:
  3023	            state = graph[mod]
  3024	            size = 0
  3025	            if state.path:
  3026	                try:
  3027	                    size = os.path.getsize(state.path)
  3028	                except OSError:
  3029	                    pass
  3030	            node.sizes[mod] = size
  3031	            for dep in state.dependencies:
  3032	                if dep in state.priorities:
  3033	                    pri = state.priorities[dep]
  3034	                    if dep in inv_nodes:
  3035	                        dep_id = inv_nodes[dep]
  3036	                        if dep_id != node.node_id and (
  3037	                            dep_id not in node.deps or pri < node.deps[dep_id]
  3038	                        ):
  3039	                            node.deps[dep_id] = pri
  3040	    print("[" + ",\n ".join(node.dumps() for node in nodes) + "\n]", file=stdout)

  3043	def load_graph(
  3044	    sources: list[BuildSource],
  3045	    manager: BuildManager,
  3046	    old_graph: Graph | None = None,
  3047	    new_modules: list[State] | None = None,
  3048	) -> Graph:
  3049	    """Given some source files, load the full dependency graph.
  3050
  3051	    If an old_graph is passed in, it is used as the starting point and
  3052	    modified during graph loading.
  3053
  3054	    If a new_modules is passed in, any modules that are loaded are
  3055	    added to the list. This is an argument and not a return value
  3056	    so that the caller can access it even if load_graph fails.
  3057
  3058	    As this may need to parse files, this can raise CompileError in case
  3059	    there are syntax errors.
  3060	    """
  3061
  3062	    graph: Graph = old_graph if old_graph is not None else {}
  3063
  3064	    # The deque is used to implement breadth-first traversal.
  3065	    # TODO: Consider whether to go depth-first instead.  This may
  3066	    # affect the order in which we process files within import cycles.
  3067	    new = new_modules if new_modules is not None else []
  3068	    entry_points: set[str] = set()
  3069	    # Seed the graph with the initial root sources.
  3070	    for bs in sources:
  3071	        try:
  3072	            st = State(
  3073	                id=bs.module,
  3074	                path=bs.path,
  3075	                source=bs.text,
  3076	                manager=manager,
  3077	                root_source=not bs.followed,
  3078	            )
  3079	        except ModuleNotFound:
  3080	            continue
  3081	        if st.id in graph:
  3082	            manager.errors.set_file(st.xpath, st.id, manager.options)
  3083	            manager.errors.report(
  3084	                -1,
  3085	                -1,
  3086	                f'Duplicate module named "{st.id}" (also at "{graph[st.id].xpath}")',
  3087	                blocker=True,
  3088	            )
  3089	            manager.errors.report(
  3090	                -1,
  3091	                -1,
  3092	                "See https://mypy.readthedocs.io/en/stable/running_mypy.html#mapping-file-paths-to-modules "
  3093	                "for more info",
  3094	                severity="note",
  3095	            )
  3096	            manager.errors.report(
  3097	                -1,
  3098	                -1,
  3099	                "Common resolutions include: a) using `--exclude` to avoid checking one of them, "
  3100	                "b) adding `__init__.py` somewhere, c) using `--explicit-package-bases` or "
  3101	                "adjusting MYPYPATH",
  3102	                severity="note",
  3103	            )
  3104
  3105	            manager.errors.raise_error()
  3106	        graph[st.id] = st
  3107	        new.append(st)
  3108	        entry_points.add(bs.module)
  3109
  3110	    # Note: Running this each time could be slow in the daemon. If it's a problem, we
  3111	    # can do more work to maintain this incrementally.
  3112	    seen_files = {st.abspath: st for st in graph.values() if st.path}
  3113
  3114	    # Collect dependencies.  We go breadth-first.
  3115	    # More nodes might get added to new as we go, but that's fine.
  3116	    for st in new:
  3117	        assert st.ancestors is not None
  3118	        # Strip out indirect dependencies.  These will be dealt with
  3119	        # when they show up as direct dependencies, and there's a
  3120	        # scenario where they hurt:
  3121	        # - Suppose A imports B and B imports C.
  3122	        # - Suppose on the next round:
  3123	        #   - C is deleted;
  3124	        #   - B is updated to remove the dependency on C;
  3125	        #   - A is unchanged.
  3126	        # - In this case A's cached *direct* dependencies are still valid
  3127	        #   (since direct dependencies reflect the imports found in the source)
  3128	        #   but A's cached *indirect* dependency on C is wrong.
  3129	        dependencies = [dep for dep in st.dependencies if st.priorities.get(dep) != PRI_INDIRECT]
  3130	        if not manager.use_fine_grained_cache():

  3132	            # because they are not in build with `follow-imports=skip`.
  3133	            # This way we could avoid overhead of cloning options in `State.__init__()`
  3134	            # below to get the option value. This is quite minor performance loss however.
  3135	            added = [dep for dep in st.suppressed if find_module_simple(dep, manager)]
  3136	        else:
  3137	            # During initial loading we don't care about newly added modules,
  3138	            # they will be taken care of during fine grained update. See also
  3139	            # comment about this in `State.__init__()`.
  3140	            added = []
  3141	        for dep in st.ancestors + dependencies + st.suppressed:
  3142	            ignored = dep in st.suppressed_set and dep not in entry_points
  3143	            if ignored and dep not in added:
  3144	                manager.missing_modules.add(dep)
  3145	            elif dep not in graph:
  3146	                try:
  3147	                    if dep in st.ancestors:
  3148	                        # TODO: Why not 'if dep not in st.dependencies' ?
  3149	                        # Ancestors don't have import context.
  3150	                        newst = State(
  3151	                            id=dep, path=None, source=None, manager=manager, ancestor_for=st
  3152	                        )
  3153	                    else:
  3154	                        newst = State(
  3155	                            id=dep,
  3156	                            path=None,
  3157	                            source=None,
  3158	                            manager=manager,
  3159	                            caller_state=st,
  3160	                            caller_line=st.dep_line_map.get(dep, 1),
  3161	                        )
  3162	                except ModuleNotFound:
  3163	                    if dep in st.dependencies_set:
  3164	                        st.suppress_dependency(dep)
  3165	                else:
  3166	                    if newst.path:
  3167	                        newst_path = os.path.abspath(newst.path)
  3168
  3169	                        if newst_path in seen_files:
  3170	                            manager.errors.report(
  3171	                                -1,
  3172	                                0,
  3173	                                "Source file found twice under different module names: "
  3174	                                '"{}" and "{}"'.format(seen_files[newst_path].id, newst.id),
  3175	                                blocker=True,
  3176	                            )
  3177	                            manager.errors.report(
  3178	                                -1,
  3179	                                0,
  3180	                                "See https://mypy.readthedocs.io/en/stable/running_mypy.html#mapping-file-paths-to-modules "
  3181	                                "for more info",
  3182	                                severity="note",
  3183	                            )
  3184	                            manager.errors.report(
  3185	                                -1,
  3186	                                0,
  3187	                                "Common resolutions include: a) adding `__init__.py` somewhere, "
  3188	                                "b) using `--explicit-package-bases` or adjusting MYPYPATH",
  3189	                                severity="note",
  3190	                            )
  3191	                            manager.errors.raise_error()
  3192
  3193	                        seen_files[newst_path] = newst
  3194
  3195	                    assert newst.id not in graph, newst.id
  3196
