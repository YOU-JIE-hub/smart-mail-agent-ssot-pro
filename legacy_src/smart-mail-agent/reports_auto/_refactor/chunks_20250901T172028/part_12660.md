  1706	        self, defn: FuncItem, reverse_type: CallableType, reverse_name: str, context: Context
  1707	    ) -> None:
  1708	        """Check a reverse operator method such as __radd__."""
  1709	        # Decides whether it's worth calling check_overlapping_op_methods().
  1710
  1711	        # This used to check for some very obscure scenario.  It now
  1712	        # just decides whether it's worth calling
  1713	        # check_overlapping_op_methods().
  1714
  1715	        assert defn.info
  1716
  1717	        # First check for a valid signature
  1718	        method_type = CallableType(
  1719	            [AnyType(TypeOfAny.special_form), AnyType(TypeOfAny.special_form)],
  1720	            [nodes.ARG_POS, nodes.ARG_POS],
  1721	            [None, None],
  1722	            AnyType(TypeOfAny.special_form),
  1723	            self.named_type("builtins.function"),
  1724	        )
  1725	        if not is_subtype(reverse_type, method_type):
  1726	            self.msg.invalid_signature(reverse_type, context)
  1727	            return
  1728
  1729	        if reverse_name in ("__eq__", "__ne__"):
  1730	            # These are defined for all objects => can't cause trouble.
  1731	            return
  1732
  1733	        # With 'Any' or 'object' return type we are happy, since any possible
  1734	        # return value is valid.
  1735	        ret_type = get_proper_type(reverse_type.ret_type)
  1736	        if isinstance(ret_type, AnyType):
  1737	            return
  1738	        if isinstance(ret_type, Instance):
  1739	            if ret_type.type.fullname == "builtins.object":
  1740	                return
  1741	        if reverse_type.arg_kinds[0] == ARG_STAR:
  1742	            reverse_type = reverse_type.copy_modified(
  1743	                arg_types=[reverse_type.arg_types[0]] * 2,
  1744	                arg_kinds=[ARG_POS] * 2,
  1745	                arg_names=[reverse_type.arg_names[0], "_"],
  1746	            )
  1747	        assert len(reverse_type.arg_types) >= 2
  1748
  1749	        forward_name = operators.normal_from_reverse_op[reverse_name]
  1750	        forward_inst = get_proper_type(reverse_type.arg_types[1])
  1751	        if isinstance(forward_inst, TypeVarType):
  1752	            forward_inst = get_proper_type(forward_inst.upper_bound)
  1753	        elif isinstance(forward_inst, TupleType):
  1754	            forward_inst = tuple_fallback(forward_inst)
  1755	        elif isinstance(forward_inst, (FunctionLike, TypedDictType, LiteralType)):
  1756	            forward_inst = forward_inst.fallback
  1757	        if isinstance(forward_inst, TypeType):
  1758	            item = forward_inst.item
  1759	            if isinstance(item, Instance):
  1760	                opt_meta = item.type.metaclass_type
  1761	                if opt_meta is not None:
  1762	                    forward_inst = opt_meta
  1763

  1765	            # TODO: Deal with attributes of TupleType etc.
  1766	            if isinstance(typ, Instance):
  1767	                return typ.type.has_readable_member(name)
  1768	            return all(
  1769	                (isinstance(x, UnionType) and has_readable_member(x, name))
  1770	                or (isinstance(x, Instance) and x.type.has_readable_member(name))
  1771	                for x in get_proper_types(typ.relevant_items())
  1772	            )
  1773
  1774	        if not (
  1775	            isinstance(forward_inst, (Instance, UnionType))
  1776	            and has_readable_member(forward_inst, forward_name)
  1777	        ):
  1778	            return
  1779	        forward_base = reverse_type.arg_types[1]
  1780	        forward_type = self.expr_checker.analyze_external_member_access(
  1781	            forward_name, forward_base, context=defn
  1782	        )
  1783	        self.check_overlapping_op_methods(
  1784	            reverse_type,
  1785	            reverse_name,
  1786	            defn.info,

  1789	            forward_base,

  1793	    def check_overlapping_op_methods(
  1794	        self,
  1795	        reverse_type: CallableType,
  1796	        reverse_name: str,
  1797	        reverse_class: TypeInfo,
  1798	        forward_type: Type,
  1799	        forward_name: str,
  1800	        forward_base: Type,
  1801	        context: Context,
  1802	    ) -> None:
  1803	        """Check for overlapping method and reverse method signatures.
  1804
  1805	        This function assumes that:
  1806
  1807	        -   The reverse method has valid argument count and kinds.
  1808	        -   If the reverse operator method accepts some argument of type
  1809	            X, the forward operator method also belong to class X.
  1810
  1811	            For example, if we have the reverse operator `A.__radd__(B)`, then the
  1812	            corresponding forward operator must have the type `B.__add__(...)`.
  1813	        """
  1814
  1815	        # Note: Suppose we have two operator methods "A.__rOP__(B) -> R1" and
  1816	        # "B.__OP__(C) -> R2". We check if these two methods are unsafely overlapping
  1817	        # by using the following algorithm:
  1818	        #
  1819	        # 1. Rewrite "B.__OP__(C) -> R1"  to "temp1(B, C) -> R1"
  1820	        #
  1821	        # 2. Rewrite "A.__rOP__(B) -> R2" to "temp2(B, A) -> R2"
  1822	        #
  1823	        # 3. Treat temp1 and temp2 as if they were both variants in the same
  1824	        #    overloaded function. (This mirrors how the Python runtime calls
  1825	        #    operator methods: we first try __OP__, then __rOP__.)
  1826	        #
  1827	        #    If the first signature is unsafely overlapping with the second,
  1828	        #    report an error.
  1829	        #
  1830	        # 4. However, if temp1 shadows temp2 (e.g. the __rOP__ method can never
  1831	        #    be called), do NOT report an error.
  1832	        #
  1833	        #    This behavior deviates from how we handle overloads -- many of the
  1834	        #    modules in typeshed seem to define __OP__ methods that shadow the
  1835	        #    corresponding __rOP__ method.
  1836	        #
  1837	        # Note: we do not attempt to handle unsafe overlaps related to multiple
  1838	        # inheritance. (This is consistent with how we handle overloads: we also
  1839	        # do not try checking unsafe overlaps due to multiple inheritance there.)
  1840
  1841	        for forward_item in flatten_nested_unions([forward_type]):
  1842	            forward_item = get_proper_type(forward_item)
  1843	            if isinstance(forward_item, CallableType):
  1844	                if self.is_unsafe_overlapping_op(forward_item, forward_base, reverse_type):
  1845	                    self.msg.operator_method_signatures_overlap(
  1846	                        reverse_class, reverse_name, forward_base, forward_name, context
  1847	                    )
  1848	            elif isinstance(forward_item, Overloaded):
  1849	                for item in forward_item.items:
  1850	                    if self.is_unsafe_overlapping_op(item, forward_base, reverse_type):
  1851	                        self.msg.operator_method_signatures_overlap(
  1852	                            reverse_class, reverse_name, forward_base, forward_name, context
  1853	                        )
  1854	            elif not isinstance(forward_item, AnyType):
  1855	                self.msg.forward_operator_not_callable(forward_name, context)
  1856
  1857	    def is_unsafe_overlapping_op(
  1858	        self, forward_item: CallableType, forward_base: Type, reverse_type: CallableType
  1859	    )
