   511	class TypeVarId:
   512	    # A type variable is uniquely identified by its raw id and meta level.
   513
   514	    # For plain variables (type parameters of generic classes and
   515	    # functions) raw ids are allocated by semantic analysis, using
   516	    # positive ids 1, 2, ... for generic class parameters and negative
   517	    # ids -1, ... for generic function type arguments. A special value 0
   518	    # is reserved for Self type variable (autogenerated). This convention
   519	    # is only used to keep type variable ids distinct when allocating
   520	    # them; the type checker makes no distinction between class and
   521	    # function type variables.
   522
   523	    # Metavariables are allocated unique ids starting from 1.
   524	    raw_id: int
   525
   526	    # Level of the variable in type inference. Currently either 0 for
   527	    # declared types, or 1 for type inference metavariables.
   528	    meta_level: int = 0
   529
   530	    # Class variable used for allocating fresh ids for metavariables.
   531	    next_raw_id: ClassVar[int] = 1
   532
   533	    # Fullname of class or function/method which declares this type
   534	    # variable (not the fullname of the TypeVar definition!), or ''
   535	    namespace: str
   536
   537	    def __init__(self, raw_id: int, meta_level: int = 0, *, namespace: str = "") -> None:
   538	        self.raw_id = raw_id
   539	        self.meta_level = meta_level
   540	        self.namespace = namespace
   541
   542	    @staticmethod
   543	    def new(meta_level: int) -> TypeVarId:
   544	        raw_id = TypeVarId.next_raw_id
   545	        TypeVarId.next_raw_id += 1
   546	        return TypeVarId(raw_id, meta_level)
   547
   548	    def __repr__(self) -> str:
   549	        return self.raw_id.__repr__()
   550
   551	    def __eq__(self, other: object) -> bool:
   552	        return (
   553	            isinstance(other, TypeVarId)
   554	            and self.raw_id == other.raw_id
   555	            and self.meta_level == other.meta_level
   556	            and self.namespace == other.namespace
   557	        )
   558
   559	    def __ne__(self, other: object) -> bool:
   560	        return not (self == other)
   561
   562	    def __hash__(self) -> int:
   563	        return hash((self.raw_id, self.meta_level, self.namespace))
   564
   565	    def is_meta_var(self) -> bool:
   566	        return self.meta_level > 0
   567
   568	    def is_self(self) -> bool:
   569	        # This is a special value indicating typing.Self variable.
   570	        return self.raw_id == 0

   724	class ParamSpecType(TypeVarLikeType):
   725	    """Type that refers to a ParamSpec.
   726
   727	    A ParamSpec is a type variable that represents the parameter
   728	    types, names and kinds of a callable (i.e., the signature without
   729	    the return type).
   730
   731	    This can be one of these forms
   732	     * P (ParamSpecFlavor.BARE)
   733	     * P.args (ParamSpecFlavor.ARGS)
   734	     * P.kwargs (ParamSpecFLavor.KWARGS)
   735
   736	    The upper_bound is really used as a fallback type -- it's shared
   737	    with TypeVarType for simplicity. It can't be specified by the user
   738	    and the value is directly derived from the flavor (currently
   739	    always just 'object').
   740	    """
   741
   742	    __slots__ = ("flavor", "prefix")
   743
   744	    flavor: int
   745	    prefix: Parameters
   746

   748	        self,
   749	        name: str,
   750	        fullname: str,
   751	        id: TypeVarId,
   752	        flavor: int,
   753	        upper_bound: Type,
   754	        default: Type,
   755	        *,
   756	        line: int = -1,
   757	        column: int = -1,
   758	        prefix: Parameters | None = None,
   759	    ) -> None:
   760	        super().__init__(name, fullname, id, upper_bound, default, line=line, column=column)
   761	        self.flavor = flavor
   762	        self.prefix = prefix or Parameters([], [], [])

   764	    def with_flavor(self, flavor: int) -> ParamSpecType:
   765	        return ParamSpecType(
   766	            self.name,
   767	            self.fullname,
   768	            self.id,
   769	            flavor,
   770	            upper_bound=self.upper_bound,
   771	            default=self.default,
   772	            prefix=self.prefix,
   773	        )
   774
   775	    def copy_modified(
   776	        self,
   777	        *,
   778	        id: Bogus[TypeVarId] = _dummy,
   779	        flavor: int = _dummy_int,
   780	        prefix: Bogus[Parameters] = _dummy,
   781	        default: Bogus[Type] = _dummy,
   782	        **kwargs: Any,
   783	    ) -> ParamSpecType:

   785	            self.name,
   786	            self.fullname,
   787	            id if id is not _dummy else self.id,
   788	            flavor if flavor != _dummy_int else self.flavor,

   808	        return hash((self.id, self.flavor, self.prefix))

   811	        if not isinstance(other, ParamSpecType):

   813	        # Upper bound can be ignored, since it's determined by flavor.
   814	        return self.id == other.id and self.flavor == other.flavor and self.prefix == other.prefix

   832	        assert data[".class"] == "ParamSpecType"
   833	        return ParamSpecType(
