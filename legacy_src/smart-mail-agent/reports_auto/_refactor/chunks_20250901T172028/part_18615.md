  1211	        self,
  1212	        class_ir: ClassIR,
  1213	        name: str,
  1214	        ret_type: RType,
  1215	        fn_info: FuncInfo | str = "",
  1216	        self_type: RType | None = None,
  1217	    ) -> Iterator[None]:
  1218	        """Generate IR for a method.
  1219
  1220	        If the method takes arguments, you should immediately afterwards call
  1221	        add_argument() for each non-self argument (self is created implicitly).
  1222
  1223	        Args:
  1224	            class_ir: Add method to this class
  1225	            name: Short name of the method
  1226	            ret_type: Return type of the method
  1227	            fn_info: Optionally, additional information about the method

  1229	                argument (by default, derive type from class_ir)

  1235	            self_type = RInstance(class_ir)

  1238	            yield

  1240	            arg_regs, args, blocks, ret_type, fn_info = self.leave()

  1413	def gen_arg_defaults(builder: IRBuilder) -> None:

  1421	    for arg in fitem.arguments:

  1430	                    return builder.accept(arg.initializer)
  1431
  1432	                # Because gen_arg_defaults runs before calculate_arg_defaults, we
