    91	        sattr =  str(attr)

    95	            attrVal = node.getAttribute(sattr)
    96	            newNode.setAttribute(newAttr, attrVal or sattr)

   103	    either unicode or utf8 encoded bytes. it will accumulate

   106	    BOMS =  {
   107	        'utf-32':codecs.BOM_UTF32,
   108	        'utf-32-be':codecs.BOM_UTF32_BE,

   121	        if isBytes(u):
   122	            try:
   123	                 u = u.decode('utf-8')
   124	            except:
   125	                et, ev, tb = sys.exc_info()
   126	                ev = str(ev)
   127	                del et, tb
   128	                raise ValueError("String %r not encoded as 'utf-8'\nerror=%s" % (u,ev))
   129	        elif not isUnicode(u):
   130	            raise ValueError("EncodedWriter.write(%s) argument should be 'utf-8' bytes or str" % ascii(u))
   131	        self.append(u)
