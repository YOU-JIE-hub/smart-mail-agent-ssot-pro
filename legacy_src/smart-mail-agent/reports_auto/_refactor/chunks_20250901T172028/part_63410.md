  1061	    static_runtime = _static_lookup_runtime(object_path)

  1073	    runtime_is_coroutine = inspect.iscoroutinefunction(runtime)

  1321	def _resolve_funcitem_from_decorator(dec: nodes.OverloadPart) -> nodes.FuncItem | None:
  1322	    """Returns a FuncItem that corresponds to the output of the decorator.
  1323
  1324	    Returns None if we can't figure out what that would be. For convenience, this function also
  1325	    accepts FuncItems.
  1326	    """
  1327	    if isinstance(dec, nodes.FuncItem):
  1328	        return dec
  1329	    if dec.func.is_property:
  1330	        return None
  1331
  1332	    def apply_decorator_to_funcitem(
  1333	        decorator: nodes.Expression, func: nodes.FuncItem
  1334	    ) -> nodes.FuncItem | None:
  1335	        if (
  1336	            isinstance(decorator, nodes.CallExpr)
  1337	            and isinstance(decorator.callee, nodes.RefExpr)
  1338	            and decorator.callee.fullname in mypy.types.DEPRECATED_TYPE_NAMES
  1339	        ):
  1340	            return func
  1341	        if not isinstance(decorator, nodes.RefExpr):
  1342	            return None
  1343	        if not decorator.fullname:
  1344	            # Happens with namedtuple
  1345	            return None
  1346	        if (
  1347	            decorator.fullname in ("builtins.staticmethod", "abc.abstractmethod")
  1348	            or decorator.fullname in mypy.types.OVERLOAD_NAMES
  1349	            or decorator.fullname in mypy.types.OVERRIDE_DECORATOR_NAMES
  1350	            or decorator.fullname in mypy.types.FINAL_DECORATOR_NAMES
  1351	        ):
  1352	            return func
  1353	        if decorator.fullname == "builtins.classmethod":
  1354	            if func.arguments[0].variable.name not in ("cls", "mcs", "metacls"):
  1355	                raise StubtestFailure(
  1356	                    f"unexpected class argument name {func.arguments[0].variable.name!r} "
  1357	                    f"in {dec.fullname}"
  1358	                )
  1359	            # FuncItem is written so that copy.copy() actually works, even when compiled
  1360	            ret = copy.copy(func)
  1361	            # Remove the cls argument, since it's not present in inspect.signature of classmethods
  1362	            ret.arguments = ret.arguments[1:]
  1363	            return ret
  1364	        # Just give up on any other decorators. After excluding properties, we don't run into
  1365	        # anything else when running on typeshed's stdlib.
  1366	        return None
  1367
  1368	    func: nodes.FuncItem = dec.func
  1369	    for decorator in dec.original_decorators:
  1370	        resulting_func = apply_decorator_to_funcitem(decorator, func)
  1371	        if resulting_func is None:
  1372	            return None
  1373	        func = resulting_func
  1374	    return func
