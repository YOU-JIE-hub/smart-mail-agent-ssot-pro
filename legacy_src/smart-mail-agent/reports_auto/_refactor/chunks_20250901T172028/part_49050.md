   555	        # for non justified text, collapse adjacent text/shift's into single operations
   556	        result = []
   557	        index = 0
   558	        maxindex = len(line)

   562	                # collect strings and floats

   570	                        if thefloats<0 and nexte>0:

   885	def readLength(text):

   899	        try:

   904	                'pt':1

   906	        except KeyError:
   907	            raise ValueError("invalid length attribute '%s'" % text)
   908
   909	        return number * multiplier
