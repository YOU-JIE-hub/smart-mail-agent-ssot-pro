8	        """Update this data with data from another :class:`CoverageData`.
   659
   660	        If `map_path` is provided, it's a function that re-map paths to match
   661	        the local machine's.  Note: `map_path` is None only when called
   662	        directly from the test suite.
   663
   664	        """
   665	        if self._debug.should("dataop"):
   666	            self._debug.write("Updating with data from {!r}".format(
   667	                getattr(other_data, "_filename", "???"),
   668	            ))
   669	        if self._has_lines and other_data._has_arcs:
   670	            raise DataError("Can't combine branch coverage data with statement data")
   671	        if self._has_arcs and other_data._has_lines:
   672	            raise DataError("Can't combine statement coverage data with branch data")
   673
   674	        map_path = map_path or (lambda p: p)
   675
   676	        # Force the database we're writing to to exist before we start nesting contexts.
   677	        self._start_using()
   678
   679	        # Collector for all arcs, lines and tracers
   680	        other_data.read()
   681	        with other_data._connect() as con:
   682	            # Get files data.
   683	            with con.execute("select path from file") as cur:
   684	                files = {path: map_path(path) for (path,) in cur}
   685
   686	            # Get contexts data.
   687	            with con.execute("select context from context") as cur:
   688	                contexts = [context for (context,) in cur]
   689
   690	            # Get arc data.
   691	            with con.execute(
   692	                "select file.path, context.context, arc.fromno, arc.tono " +
   693	                "from arc " +
   694	                "inner join file on file.id = arc.file_id " +
   695	                "inner join context on context.id = arc.context_id",
   696	            ) as cur:
   697	                arcs = [
   698	                    (files[path], context, fromno, tono)
   699	                    for (path, context, fromno, tono) in cur
   700	                ]
   701
   702	            # Get line data.
   703	            with con.execute(
   704	                "select file.path, context.context, line_bits.numbits " +
   705	                "from line_bits " +
   706	                "inner join file on file.id = line_bits.file_id " +
   707	                "inner join context on context.id = line_bits.context_id",
   708	            ) as cur:
   709	                lines: dict[tuple[str, str], bytes] = {}
   710	                for path, context, numbits in cur:
   711	                    key = (files[path], context)
   712	                    if key in lines:
   713	                        numbits = numbits_union(lines[key], numbits)
   714	                    lines[key] = numbits
   715
   716	            # Get tracer data.
   717	            with con.execute(
   718	                "select file.path, tracer " +
   719	                "from tracer " +
   720	                "inner join file on file.id = tracer.file_id",
   721	            ) as cur:
   722	                tracers = {files[path]: tracer for (path, tracer) in cur}
   723
   724	        with self._connect() as con:
   725	            assert con.con is not None
   726	            con.con.isolation_level = "IMMEDIATE"
   727
   728	            # Get all tracers in the DB. Files not in the tracers are assumed
   729	            # to have an empty string tracer. Since Sqlite does not support
   730	            # full outer joins, we have to make two queries to fill the
   731	            # dictionary.
   732	            with con.execute("select path from file") as cur:
   733	                this_tracers = {path: "" for path, in cur}
   734	            with con.execute(
   735	                "select file.path, tracer from tracer " +
   736	                "inner join file on file.id = tracer.file_id",
   737	            ) as cur:
   738	                this_tracers.update({
   739	                    map_path(path): tracer
   740	                    for path, tracer in cur
   741	                })
   742
   743	            # Create all file and context rows in the DB.
   744	            con.executemany_void(
   745	                "insert or ignore into file (path) values (?)",
   746	                ((file,) for file in files.values()),
   747	            )
   748	            with con.execute("select id, path from file") as cur:
   749	                file_ids = {path: id for id, path in cur}
   750	            self._file_map.update(file_ids)
   751	            con.executemany_void(
   752	                "insert or ignore into context (context) values (?)",
   753	                ((context,) for context in contexts),
   754	            )
   755	            with con.execute("select id, context from context") as cur:
   756	                context_ids = {context: id for id, context in cur}
   757
   758	            # Prepare tracers and fail, if a conflict is found.
   759	            # tracer_paths is used to ensure consistency over the tracer data
   760	            # and tracer_map tracks the tracers to be inserted.
   761	            tracer_map = {}
   762	            for path in files.values():
   763	                this_tracer = this_tracers.get(path)
   764	                other_tracer = tracers.get(path, "")
   765	                # If there is no tracer, there is always the None tracer.
   766	                if this_tracer is not None and this_tracer != other_tracer:
   767	                    raise DataError(
   768	                        "Conflicting file tracer name for '{}': {!r} vs {!r}".format(
   769	                            path, this_tracer, other_tracer,
   770	                        ),
   771	                    )
   772	                tracer_map[path] = other_tracer
   773
   774	            # Prepare arc and line rows to be inserted by converting the file
   775	            # and context strings with integer ids. Then use the efficient
   776	            # `executemany()` to insert all rows at once.
   777
   778	            if arcs:
   779	                self._choose_lines_or_arcs(arcs=True)
   780
   781	                arc_rows = (
   782	                    (file_ids[file], context_ids[context], fromno, tono)
   783	                    for file, context, fromno, tono in arcs
   784	                )
   785
   786	                # Write the combined data.
   787	                con.executemany_void(
   788	                    "insert or ignore into arc " +
   789	                    "(file_id, context_id, fromno, tono) values (?, ?, ?, ?)",
   790	                    arc_rows,
   791	                )
   792
   793	            if lines:
   794	                self._choose_lines_or_arcs(lines=True)
   795
   796	                for (file, context), numbits in lines.items():
   797	                    with con.execute(
   798	                        "select numbits from line_bits where file_id = ? and context_id = ?",
   799	                        (file_ids[file], context_ids[context]),
   800	                    ) as cur:
   801	                        existing = list(cur)
   802	                    if existing:
   803	                        lines[(file, context)] = numbits_union(numbits, existing[0][0])
   804
   805	                con.executemany_void(
   806	                    "insert or replace into line_bits " +
   807	                    "(file_id, context_id, numbits) values (?, ?, ?)",
   808	                    [
   809	                        (file_ids[file], context_ids[context], numbits)
   810	                        for (file, context), numbits in lines.items()
   811	                    ],
   812	                )
   813
   814	            con.executemany_void(
   815	                "insert or ignore into tracer (file_id, tracer) values (?, ?)",
   816	                ((file_ids[filename], tracer) for filename, tracer in tracer_map.items()),
   817	            )
   818
   819	        if not self._no_disk:
   820	            # Update all internal cache data.
   821	            self._reset()
   822	            self.read()
   823
   824	    def erase(self, parallel: bool = False) -> None:
   825	        """Erase the data in this object.
   826
   827	        If `parallel` is true, then also deletes data files created from the
   828	        basename by parallel-mode.
   829
   830	        """
   831	        self._reset()
   832	        if self._no_disk:
   833	            return
   834	        self._debug_dataio("Erasing data file", self._filename)
   835	        file_be_gone(self._filename)
   836	        if parallel:
   837	            data_dir, local = os.path.split(self._filename)
   838	            local_abs_path = os.path.join(os.path.abspath(data_dir), local)
   839	            pattern = glob.escape(local_abs_path) + ".*"
   840	            for filename in glob.glob(pattern):
   841	                self._debug_dataio("Erasing parallel data file", filename)
   842	                file_be_gone(filename)

   854	    def _start_using(self) -> None:
   855	        """Call this before using the database at all."""
   856	        if self._pid != os.getpid():
   857	            # Looks like we forked! Have to start a new data file.
   858	            self._reset()
   859	            self._choose_filename()
   860	            self._pid = os.getpid()
   861	        if not self._have_used:
   862	            self.erase()
   863	        self._have_used = True

   908	    def set_query_context(self, context: str) -> None:
   909	        """Set a context for subsequent querying.
   910
   911	        The next :meth:`lines`, :meth:`arcs`, or :meth:`contexts_by_lineno`
   912	        calls will be limited to only one context.  `context` is a string which
   913	        must match a context exactly.  If it does not, no exception is raised,
   914	        but queries will return no data.
   915
   916	        .. versionadded:: 5.0
   917
   918	        """
   919	        self._start_using()
   920	        with self._connect() as con:
   921	            with con.execute("select id from context where context = ?", (context,)) as cur:
   922	                self._query_context_ids = [row[0] for row in cur.fetchall()]
   923
   924	    def set_query_contexts(self, contexts: Sequence[str] | None) -> None:
   925	        """Set a number of contexts for subsequent querying.
   926
   927	        The next :meth:`lines`, :meth:`arcs`, or :meth:`contexts_by_lineno`
   928	        calls will be limited to the specified contexts.  `contexts` is a list
   929	        of Python regular expressions.  Contexts will be matched using
   930	        :func:`re.search <python:re.search>`.  Data will be included in query
   931	        results if they are part of any of the contexts matched.
   932
   933	        .. versionadded:: 5.0
   934
   935	        """
   936	        self._start_using()
   937	        if contexts:
   938	            with self._connect() as con:
   939	                context_clause = " or ".join(["context regexp ?"] * len(contexts))
   940	                with con.execute("select id from context where " + context_clause, contexts) as cur:
   941	                    self._query_context_ids = [row[0] for row in cur.fetchall()]
   942	        else:
   943	            self._query_context_ids = None

   945	    def lines(self, filename: str) -> list[TLineNo] | None:
   946	        """Get the list of lines executed for a source file.
   947
   948	        If the file was not measured, returns None.  A file might be measured,
   949	        and have no lines executed, in which case an empty list is returned.
   950
   951	        If the file was executed, returns a list of integers, the line numbers
   952	        executed in the file. The list is in no particular order.
   953
   954	        """
   955	        self._start_using()
   956	        if self.has_arcs():
   957	            arcs = self.arcs(filename)
   958	            if arcs is not None:
   959	                all_lines = itertools.chain.from_iterable(arcs)
   960	                return list({l for l in all_lines if l > 0})
   961
   962	        with self._connect() as con:
   963	            file_id = self._file_id(filename)
   964	            if file_id is None:
   965	                return None
   966	            else:
   967	                query = "select numbits from line_bits where file_id = ?"
   968	                data = [file_id]
   969	                if self._query_context_ids is not None:
   970	                    ids_array = ", ".join("?" * len(self._query_context_ids))
   971	                    query += " and context_id in (" + ids_array + ")"
   972	                    data += self._query_context_ids
   973	                with con.execute(query, data) as cur:
   974	                    bitmaps = list(cur)
   975	                nums = set()
   976	                for row in bitmaps:
   977	                    nums.update(numbits_to_nums(row[0]))
   978	                return list(nums)

   980	    def arcs(self, filename: str) -> list[TArc] | None:
   981	        """Get the list of arcs executed for a file.
   982
   983	        If the file was not measured, returns None.  A file might be measured,
   984	        and have no arcs executed, in which case an empty list is returned.
   985
   986	        If the file was executed, returns a list of 2-tuples of integers. Each
   987	        pair is a starting line number and an ending line number for a
   988	        transition from one line to another. The list is in no particular
   989	        order.
   990
   991	        Negative numbers have special meaning.  If the starting line number is
   992	        -N, it represents an entry to the code object that starts at line N.
   993	        If the ending ling number is -N, it's an exit from the code object that
   994	        starts at line N.
   995
   996	        """
   997	        self._start_using()
   998	        with self._connect() as con:
   999	            file_id = self._file_id(filename)
  1000	            if file_id is None:
  1001	                return None
  1002	            else:
  1003	                query = "select distinct fromno, tono from arc where file_id = ?"
  1004	                data = [file_id]
  1005	                if self._query_context_ids is not None:
  1006	                    ids_array = ", ".join("?" * len(self._query_context_ids))
  1007	                    query += " and context_id in (" + ids_array + ")"
  1008	                    data += self._query_context_ids
  1009	                with con.execute(query, data) as cur:
  1010	                    return list(cur)

  1012	    def contexts_by_lineno(self, filename: str) -> dict[TLineNo, list[str]]:
  1013	        """Get the contexts for each line in a file.
  1014
  1015	        Returns:
  1016	            A dict mapping line numbers to a list of context names.
  1017
  1018	        .. versionadded:: 5.0
  1019
  1020	        """
  1021	        self._start_using()
  1022	        with self._connect() as con:
  1023	            file_id = self._file_id(filename)
  1024	            if file_id is None:
  1025	                return {}
  1026
  1027	            lineno_contexts_map = collections.defaultdict(set)
  1028	            if self.has_arcs():
  1029	                query = (
  1030	                    "select arc.fromno, arc.tono, context.context " +
  1031	                    "from arc, context " +
  1032	                    "where arc.file_id = ? and arc.context_id = context.id"
  1033	                )
  1034	                data = [file_id]
  1035	                if self._query_context_ids is not None:
  1036	                    ids_array = ", ".join("?" * len(self._query_context_ids))
  1037	                    query += " and arc.context_id in (" + ids_array + ")"
  1038	                    data += self._query_context_ids
  1039	                with con.execute(query, data) as cur:
  1040	                    for fromno, tono, context in cur:
  1041	                        if fromno > 0:
  1042	                            lineno_contexts_map[fromno].add(context)
  1043	                        if tono > 0:
  1044	                            lineno_contexts_map[tono].add(context)
  1045	            else:
  1046	                query = (
  1047	                    "select l.numbits, c.context from line_bits l, context c " +
  1048	                    "where l.context_id = c.id " +
  1049	                    "and file_id = ?"
  1050	                )
  1051	                data = [file_id]
  1052	                if self._query_context_ids is not None:
  1053	                    ids_array = ", ".join("?" * len(self._query_context_ids))
  1054	                    query += " and l.context_id in (" + ids_array + ")"
  1055	                    data += self._query_context_ids
  1056	                with con.execute(query, data) as cur:
  1057	                    for numbits, context in cur:
  1058	                        for lineno in numbits_to_nums(numbits):
  1059	                            lineno_contexts_map[lineno].add(context)
  1060
  1061	        return {lineno: list(contexts) for lineno, contexts in lineno_contexts_map.items()}

  1079	            ("sqlite3_temp_store", temp_store),
  1080	            ("sqlite3_compile_options", copts),
  1081	        ]
