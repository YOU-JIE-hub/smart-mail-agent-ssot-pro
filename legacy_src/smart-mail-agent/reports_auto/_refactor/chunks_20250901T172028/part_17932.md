    52	def smart_truncate(text: str, max_chars: int = 1000) -> str:
    53	    """智慧截斷長文本，避免超過模型長度限制"""

    58	    mid_end = int(len(text) / 2 + max_chars * 0.15)
    59	    middle = text[mid_start:mid_end]
    60	    tail = text[-int(max_chars * 0.3) :]
