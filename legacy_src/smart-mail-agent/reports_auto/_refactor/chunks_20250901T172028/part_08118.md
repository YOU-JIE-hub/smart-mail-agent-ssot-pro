   192	    """Try to infer a better signature type for TypedDict.get.
   193
   194	    This is used to get better type context for the second argument that
   195	    depends on a TypedDict value type.
   196	    """
   197	    signature = ctx.default_signature
   198	    if (
   199	        isinstance(ctx.type, TypedDictType)
   200	        and len(ctx.args) == 2
   201	        and len(ctx.args[0]) == 1
   202	        and isinstance(ctx.args[0][0], StrExpr)
   203	        and len(signature.arg_types) == 2
   204	        and len(signature.variables) == 1
   205	        and len(ctx.args[1]) == 1
   206	    ):
   207	        key = ctx.args[0][0].value
   208	        value_type = get_proper_type(ctx.type.items.get(key))

   218	                value_type = value_type.copy_modified(required_keys=set())
