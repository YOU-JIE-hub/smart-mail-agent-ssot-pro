   437	def _pfbCheck(p,d,m,fn):
   438	    if chr(d[p])!=PFB_MARKER or chr(d[p+1])!=m:
   439	        raise ValueError('Bad pfb file\'%s\' expected chr(%d)chr(%d) at char %d, got chr(%d)chr(%d)' % (fn,ord(PFB_MARKER),ord(m),p,d[p],d[p+1]))
   440	    if m==PFB_EOF: return
   441	    p = p + 2
   442	    l = (((((d[p+3])<<8)|(d[p+2])<<8)|(d[p+1]))<<8)|(d[p])
   443	    p = p + 4
   444	    if p+l>len(d):
   445	        raise ValueError('Bad pfb file\'%s\' needed %d+%d bytes have only %d!' % (fn,p,l,len(d)))
   446	    return p, p+l

   636	    """Lazily construct known typefaces if not found"""
   637	    try:

   639	    except KeyError:
   640	        # not found, construct it if known
   641	        if faceName in standardFonts:
   642	            face = TypeFace(faceName)
   643	            (face.familyName, face.bold, face.italic) = _fontdata.standardFontAttributes[faceName]

   646	##                face.name, face.familyName, face.bold, face.italic)
   647	            return face
   648	        else:
   649	            #try a brute force search
   650	            afm = bruteForceSearchForAFM(faceName)
   651	            if afm:
   652	                for e in ('.pfb', '.PFB'):
