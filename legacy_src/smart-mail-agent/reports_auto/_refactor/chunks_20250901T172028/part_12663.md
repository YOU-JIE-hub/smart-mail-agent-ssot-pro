  2819	    def check_multiple_inheritance(self, typ: TypeInfo) -> None:
  2820	        """Check for multiple inheritance related errors."""
  2821	        if len(typ.bases) <= 1:
  2822	            # No multiple inheritance.
  2823	            return
  2824	        # Verify that inherited attributes are compatible.
  2825	        mro = typ.mro[1:]
  2826	        all_names = {name for base in mro for name in base.names}
  2827	        for name in sorted(all_names - typ.names.keys()):
  2828	            # Sort for reproducible message order.
  2829	            # Attributes defined in both the type and base are skipped.
  2830	            # Normal checks for attribute compatibility should catch any problems elsewhere.
  2831	            if is_private(name):
  2832	                continue
  2833	            # Compare the first base defining a name with the rest.
  2834	            # Remaining bases may not be pairwise compatible as the first base provides
  2835	            # the used definition.

  2841	    def check_compatibility(
  2842	        self, name: str, base1: TypeInfo, base2: TypeInfo, ctx: TypeInfo
  2843	    ) -> None:
  2844	        """Check if attribute name in base1 is compatible with base2 in multiple inheritance.
  2845
  2846	        Assume base1 comes before base2 in the MRO, and that base1 and base2 don't have
  2847	        a direct subclass relationship (i.e., the compatibility requirement only derives from
  2848	        multiple inheritance).
  2849
  2850	        This check verifies that a definition taken from base1 (and mapped to the current
  2851	        class ctx), is type compatible with the definition taken from base2 (also mapped), so
  2852	        that unsafe subclassing like this can be detected:
  2853	            class A(Generic[T]):
  2854	                def foo(self, x: T) -> None: ...
  2855
  2856	            class B:
  2857	                def foo(self, x: str) -> None: ...
  2858
  2859	            class C(B, A[int]): ...  # this is unsafe because...
  2860
  2861	            x: A[int] = C()
  2862	            x.foo  # ...runtime type is (str) -> None, while static type is (int) -> None
  2863	        """
  2864	        if name in ("__init__", "__new__", "__init_subclass__"):
  2865	            # __init__ and friends can be incompatible -- it's a special case.
  2866	            return
  2867	        first = base1.names[name]
  2868	        second = base2.names[name]
  2869	        # Specify current_class explicitly as this function is called after leaving the class.
  2870	        first_type, _ = self.node_type_from_base(name, base1, ctx, current_class=ctx)
  2871	        second_type, _ = self.node_type_from_base(name, base2, ctx, current_class=ctx)
  2872
  2873	        # TODO: use more principled logic to decide is_subtype() vs is_equivalent().
  2874	        # We should rely on mutability of superclass node, not on types being Callable.
  2875	        # (in particular handle settable properties with setter type different from getter).
  2876
  2877	        p_first_type = get_proper_type(first_type)
  2878	        p_second_type = get_proper_type(second_type)
  2879	        if isinstance(p_first_type, FunctionLike) and isinstance(p_second_type, FunctionLike):
  2880	            if p_first_type.is_type_obj() and p_second_type.is_type_obj():
  2881	                # For class objects only check the subtype relationship of the classes,
  2882	                # since we allow incompatible overrides of '__init__'/'__new__'
  2883	                ok = is_subtype(
  2884	                    left=fill_typevars_with_any(p_first_type.type_object()),
  2885	                    right=fill_typevars_with_any(p_second_type.type_object()),
  2886	                )
  2887	            else:
  2888	                assert first_type and second_type
  2889	                ok = is_subtype(first_type, second_type, ignore_pos_arg_names=True)
  2890	        elif first_type and second_type:
  2891	            if second.node is not None and not self.is_writable_attribute(second.node):
  2892	                ok = is_subtype(first_type, second_type)
  2893	            else:
  2894	                ok = is_equivalent(first_type, second_type)
  2895	            if ok:
  2896	                if (
  2897	                    first.node
  2898	                    and second.node
  2899	                    and self.is_writable_attribute(second.node)
  2900	                    and is_property(first.node)
  2901	                    and isinstance(first.node, Decorator)
  2902	                    and not isinstance(p_second_type, AnyType)
  2903	                ):
  2904	                    self.msg.fail(
  2905	                        f'Cannot override writeable attribute "{name}" in base "{base2.name}"'
  2906	                        f' with read-only property in base "{base1.name}"',
  2907	                        ctx,
  2908	                        code=codes.OVERRIDE,
  2909	                    )
  2910	        else:
  2911	            if first_type is None:
  2912	                self.msg.cannot_determine_type_in_base(name, base1.name, ctx)
  2913	            if second_type is None:
  2914	                self.msg.cannot_determine_type_in_base(name, base2.name, ctx)
  2915	            ok = True
  2916	        # Final attributes can never be overridden, but can override
  2917	        # non-final read-only attributes.
  2918	        if is_final_node(second.node) and not is_private(name):
  2919	            self.msg.cant_override_final(name, base2.name, ctx)
  2920	        if is_final_node(first.node):
  2921	            self.check_if_final_var_override_writable(name, second.node, ctx)
  2922	        # Some attributes like __slots__ and __deletable__ are special, and the type can
  2923	        # vary across class hierarchy.
  2924	        if isinstance(second.node, Var) and second.node.allow_incompatible_override:
  2925	            ok = True
  2926	        if not ok:
  2927	            self.msg.base_class_definitions_incompatible(name, base1, base2, ctx)
