   769	            )

   771	                return builder.unary_op(result, "not", e.line)

   780	                return translate_is_none(builder, e.operands[0], negated=first_op != "is")

   783	            right_expr = e.operands[1]
   784	            if is_int_rprimitive(builder.node_type(right_expr)):

   787	                    left = builder.accept(left_expr, can_borrow=borrow_left)

   792	    # All of the trickiness here is due to support for chained conditionals
   793	    # (`e1 < e2 > e3`, etc). `e1 < e2 > e3` is approximately equivalent to
   794	    # `e1 < e2 and e2 > e3` except that `e2` is only evaluated once.
   795	    expr_type = builder.node_type(e)

   797	    # go(i, prev) generates code for `ei opi e{i+1} op{i+1} ... en`,
   798	    # assuming that prev contains the value of `ei`.
   799	    def go(i: int, prev: Value) -> Value:
   800	        if i == len(e.operators) - 1:
   801	            return transform_basic_comparison(
   802	                builder, e.operators[i], prev, builder.accept(e.operands[i + 1]), e.line
   803	            )

   805	        next = builder.accept(e.operands[i + 1])

   807	            "and",
   808	            expr_type,
   809	            lambda: transform_basic_comparison(builder, e.operators[i], prev, next, e.line),
   810	            lambda: go(i + 1, next),
   811	            e.line,
   812	        )

   814	    return go(0, builder.accept(e.operands[0]))
