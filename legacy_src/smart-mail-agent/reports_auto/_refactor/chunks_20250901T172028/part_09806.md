   183	    def normalizeData(self, outer = 0.0):
   184	        """Turns data into normalized ones where each datum is < 1.0,
   185	        and 1.0 = maximum radius.  Adds 10% at outside edge by default"""
   186	        data = self.data
   187	        assert min(list(map(min,data))) >=0, "Cannot do spider plots of negative numbers!"

   189	        norm *= (1.0+outer)

   191	        self._norm = norm

   195	        "Draw a label for a given item in the list."
   196	        fmt = sty.format
   197	        value = radius*self._norm

   199	            text = None
   200	        elif isinstance(fmt,str):

   202	                text = sty._text

   220	    def draw(self):
   221	        # normalize slice data
   222	        g = self.makeBackground() or Group()
   223
   224	        xradius = self.width/2.0
   225	        yradius = self.height/2.0
   226	        self._radius = radius = min(xradius, yradius)
   227	        cx = self.x + xradius
   228	        cy = self.y + yradius
   229
   230	        data = self.normalizeData()
   231
   232	        self._seriesCount = len(data)
   233	        n = len(data[0])
   234
   235	        #labels
   236	        if self.labels is None:
   237	            labels = [''] * n
   238	        else:
   239	            labels = self.labels
   240	            #there's no point in raising errors for less than enough errors if
   241	            #we silently create all for the extreme case of no labels.
   242	            i = n-len(labels)
   243	            if i>0:
   244	                labels = labels + ['']*i
   245
   246	        S = []
   247	        STRANDS = []
   248	        STRANDAREAS = []
   249	        syms = []
   250	        labs = []
   251	        csa = []
   252	        angle = self.startAngle*pi/180
   253	        direction = self.direction == "clockwise" and -1 or 1
   254	        angleBetween = direction*(2 * pi)/float(n)
   255	        spokes = self.spokes
   256	        spokeLabels = self.spokeLabels
   257	        for i in range(n):
   258	            car = cos(angle)*radius
   259	            sar = sin(angle)*radius
   260	            csa.append((car,sar,angle))
   261	            si = self.spokes[i]
   262	            if si.visible:
   263	                spoke = Line(cx, cy, cx + car, cy + sar, strokeWidth = si.strokeWidth, strokeColor=si.strokeColor, strokeDashArray=si.strokeDashArray)
   264	            S.append(spoke)
   265	            sli = spokeLabels[i]
   266	            text = sli._text
   267	            if not text: text = labels[i]
   268	            if text:
   269	                S.append(_setupLabel(WedgeLabel, text, si.labelRadius, cx, cy, angle, car, sar, sli))
   270	            angle += angleBetween
   271
   272	        # now plot the polygons
   273	        rowIdx = 0
   274	        strands = self.strands
   275	        strandLabels = self.strandLabels
   276	        for row in data:
   277	            # series plot
   278	            rsty = strands[rowIdx]
   279	            points = []
   280	            car, sar = csa[-1][:2]
   281	            r = row[-1]
   282	            points.append(cx+car*r)
   283	            points.append(cy+sar*r)
   284	            for i in range(n):
   285	                car, sar, angle = csa[i]
   286	                r = row[i]
   287	                points.append(cx+car*r)
   288	                points.append(cy+sar*r)
   289	                L = self._innerDrawLabel(strandLabels[(rowIdx,i)], r, cx, cy, angle, car, sar, labelClass=StrandLabel)
   290	                if L: labs.append(L)
   291	                sty = strands[(rowIdx,i)]
   292	                uSymbol = sty.symbol
   293
   294	                # put in a marker, if it needs one
   295	                if uSymbol:
   296	                    s_x =  cx+car*r
   297	                    s_y = cy+sar*r
   298	                    s_fillColor = sty.fillColor
   299	                    s_strokeColor = sty.strokeColor
   300	                    s_strokeWidth = sty.strokeWidth
   301	                    s_angle = 0
   302	                    s_size = sty.symbolSize
   303	                    if type(uSymbol) is type(''):
   304	                        symbol = makeMarker(uSymbol,
   305	                                    size = s_size,
   306	                                    x =  s_x,
   307	                                    y = s_y,
   308	                                    fillColor = s_fillColor,
   309	                                    strokeColor = s_strokeColor,
   310	                                    strokeWidth = s_strokeWidth,
   311	                                    angle = s_angle,
   312	                                    )
   313	                    else:
   314	                        symbol = uSymbol2Symbol(uSymbol,s_x,s_y,s_fillColor)
   315	                        for k,v in (('size', s_size), ('fillColor', s_fillColor),
   316	                                    ('x', s_x), ('y', s_y),
   317	                                    ('strokeColor',s_strokeColor), ('strokeWidth',s_strokeWidth),
   318	                                    ('angle',s_angle),):
   319	                            if getattr(symbol,k,None) is None:
   320	                                try:
   321	                                    setattr(symbol,k,v)
   322	                                except:
   323	                                    pass
   324	                    syms.append(symbol)
   325
   326	            # make up the 'strand'
   327	            if rsty.fillColor:
   328	                strand = Polygon(points)
   329	                strand.fillColor = rsty.fillColor
   330	                strand.strokeColor = None
   331	                strand.strokeWidth = 0
   332	                STRANDAREAS.append(strand)
   333	            if rsty.strokeColor and rsty.strokeWidth:
   334	                strand = PolyLine(points)
   335	                strand.strokeColor = rsty.strokeColor
   336	                strand.strokeWidth = rsty.strokeWidth
   337	                strand.strokeDashArray = rsty.strokeDashArray
   338	                STRANDS.append(strand)
   339	            rowIdx += 1
   340
   341	        for s in (STRANDAREAS+STRANDS+syms+S+labs): g.add(s)
   342	        return g
