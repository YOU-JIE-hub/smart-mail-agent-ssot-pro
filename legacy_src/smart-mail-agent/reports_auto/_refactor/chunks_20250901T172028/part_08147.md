   179	def process_top_levels(graph: Graph, scc: list[str], patches: Patches) -> None:
   180	    # Process top levels until everything has been bound.
   181
   182	    # Reverse order of the scc so the first modules in the original list will be
   183	    # be processed first. This helps with performance.
   184	    scc = list(reversed(scc))  # noqa: FURB187 intentional copy
   185
   186	    # Initialize ASTs and symbol tables.
   187	    for id in scc:
   188	        state = graph[id]
   189	        assert state.tree is not None
   190	        state.manager.semantic_analyzer.prepare_file(state.tree)
   191
   192	    # Initially all namespaces in the SCC are incomplete (well they are empty).
   193	    state.manager.incomplete_namespaces.update(scc)
   194
   195	    worklist = scc.copy()
   196	    # HACK: process core stuff first. This is mostly needed to support defining
   197	    # named tuples in builtin SCC.
   198	    if all(m in worklist for m in core_modules):
   199	        worklist += list(reversed(core_modules)) * CORE_WARMUP
   200	    final_iteration = False
   201	    iteration = 0
   202	    analyzer = state.manager.semantic_analyzer
   203	    analyzer.deferral_debug_context.clear()
   204
   205	    while worklist:
   206	        iteration += 1
   207	        if iteration > MAX_ITERATIONS:
   208	            # Just pick some module inside the current SCC for error context.
   209	            assert state.tree is not None
   210	            with analyzer.file_context(state.tree, state.options):
   211	                analyzer.report_hang()
   212	            break
   213	        if final_iteration:
   214	            # Give up. It's impossible to bind all names.
   215	            state.manager.incomplete_namespaces.clear()
   216	        all_deferred: list[str] = []
   217	        any_progress = False
   218	        while worklist:
   219	            next_id = worklist.pop()
   220	            state = graph[next_id]
   221	            assert state.tree is not None
   222	            deferred, incomplete, progress = semantic_analyze_target(
   223	                next_id, next_id, state, state.tree, None, final_iteration, patches
   224	            )
   225	            all_deferred += deferred
   226	            any_progress = any_progress or progress
   227	            if not incomplete:
   228	                state.manager.incomplete_namespaces.discard(next_id)
   229	        if final_iteration:
   230	            assert not all_deferred, "Must not defer during final iteration"
   231	        # Reverse to process the targets in the same order on every iteration. This avoids
   232	        # processing the same target twice in a row, which is inefficient.
   233	        worklist = list(reversed(all_deferred))
   234	        final_iteration = not any_progress

   237	def order_by_subclassing(targets: list[FullTargetInfo]) -> Iterator[FullTargetInfo]:
   238	    """Make sure that superclass methods are always processed before subclass methods.
   239
   240	    This algorithm is not very optimal, but it is simple and should work well for lists
   241	    that are already almost correctly ordered.
   242	    """
   243
   244	    # First, group the targets by their TypeInfo (since targets are sorted by line,
   245	    # we know that each TypeInfo will appear as group key only once).
   246	    grouped = [(k, list(g)) for k, g in groupby(targets, key=lambda x: x[3])]
   247	    remaining_infos = {info for info, _ in grouped if info is not None}
   248
   249	    next_group = 0
   250	    while grouped:
   251	        if next_group >= len(grouped):
   252	            # This should never happen, if there is an MRO cycle, it should be reported
   253	            # and fixed during top-level processing.
   254	            raise ValueError("Cannot order method targets by MRO")
   255	        next_info, group = grouped[next_group]
   256	        if next_info is None:
   257	            # Trivial case, not methods but functions, process them straight away.
   258	            yield from group
   259	            grouped.pop(next_group)
   260	            continue
   261	        if any(parent in remaining_infos for parent in next_info.mro[1:]):
   262	            # We cannot process this method group yet, try a next one.
   263	            next_group += 1
   264	            continue
   265	        yield from group
   266	        grouped.pop(next_group)
   267	        remaining_infos.discard(next_info)
   268	        # Each time after processing a method group we should retry from start,
   269	        # since there may be some groups that are not blocked on parents anymore.
   270	        next_group = 0

   273	def process_functions(graph: Graph, scc: list[str], patches: Patches) -> None:
   274	    # Process functions.
   275	    all_targets = []
   276	    for module in scc:
   277	        tree = graph[module].tree
   278	        assert tree is not None
   279	        # In principle, functions can be processed in arbitrary order,
   280	        # but _methods_ must be processed in the order they are defined,
   281	        # because some features (most notably partial types) depend on
   282	        # order of definitions on self.
   283	        #
   284	        # There can be multiple generated methods per line. Use target
   285	        # name as the second sort key to get a repeatable sort order.
   286	        targets = sorted(get_all_leaf_targets(tree), key=lambda x: (x[1].line, x[0]))
   287	        all_targets.extend(
   288	            [(module, target, node, active_type) for target, node, active_type in targets]
   289	        )
   290
   291	    for module, target, node, active_type in order_by_subclassing(all_targets):
   292	        analyzer = graph[module].manager.semantic_analyzer
   293	        assert isinstance(node, (FuncDef, OverloadedFuncDef, Decorator))
   294	        process_top_level_function(
   295	            analyzer, graph[module], module, target, node, active_type, patches
   296	        )

   299	def process_top_level_function(
   300	    analyzer: SemanticAnalyzer,
   301	    state: State,
   302	    module: str,
   303	    target: str,
   304	    node: FuncDef | OverloadedFuncDef | Decorator,
   305	    active_type: TypeInfo | None,
   306	    patches: Patches,
   307	) -> None:
   308	    """Analyze single top-level function or method.
   309
   310	    Process the body of the function (including nested functions) again and again,
   311	    until all names have been resolved (or iteration limit reached).
   312	    """
   313	    # We need one more iteration after incomplete is False (e.g. to report errors, if any).
   314	    final_iteration = False
   315	    incomplete = True
   316	    # Start in the incomplete state (no missing names will be reported on first pass).
   317	    # Note that we use module name, since functions don't create qualified names.
   318	    deferred = [module]
   319	    analyzer.deferral_debug_context.clear()
   320	    analyzer.incomplete_namespaces.add(module)
   321	    iteration = 0
   322	    while deferred:
   323	        iteration += 1
   324	        if iteration == MAX_ITERATIONS:
   325	            # Just pick some module inside the current SCC for error context.
   326	            assert state.tree is not None
   327	            with analyzer.file_context(state.tree, state.options):
   328	                analyzer.report_hang()
   329	            break
   330	        if not (deferred or incomplete) or final_iteration:
   331	            # OK, this is one last pass, now missing names will be reported.
   332	            analyzer.incomplete_namespaces.discard(module)
   333	        deferred, incomplete, progress = semantic_analyze_target(
   334	            target, module, state, node, active_type, final_iteration, patches
   335	        )
   336	        if not incomplete:
   337	            state.manager.incomplete_namespaces.discard(module)
   338	        if final_iteration:
   339	            assert not deferred, "Must not defer during final iteration"
   340	        if not progress:
   341	            final_iteration = True
   342
   343	    analyzer.incomplete_namespaces.discard(module)
   344	    # After semantic analysis is done, discard local namespaces
   345	    # to avoid memory hoarding.
   346	    analyzer.saved_locals.clear()

   368	def semantic_analyze_target(
   369	    target: str,
   370	    module: str,
   371	    state: State,
   372	    node: MypyFile | FuncDef | OverloadedFuncDef | Decorator,
   373	    active_type: TypeInfo | None,
   374	    final_iteration: bool,
   375	    patches: Patches,
   376	) -> tuple[list[str], bool, bool]:
   377	    """Semantically analyze a single target.
   378
   379	    Return tuple with these items:
   380	    - list of deferred targets
   381	    - was some definition incomplete (need to run another pass)
   382	    - were any new names defined (or placeholders replaced)
   383	    """
   384	    state.manager.processed_targets.append((module, target))
   385	    tree = state.tree
   386	    assert tree is not None
   387	    analyzer = state.manager.semantic_analyzer
   388	    # TODO: Move initialization to somewhere else
   389	    analyzer.global_decls = [set()]
   390	    analyzer.nonlocal_decls = [set()]
   391	    analyzer.globals = tree.names
   392	    analyzer.imports = set()
   393	    analyzer.progress = False
   394	    with state.wrap_context(check_blockers=False):
   395	        refresh_node = node
   396	        if isinstance(refresh_node, Decorator):
   397	            # Decorator expressions will be processed as part of the module top level.
   398	            refresh_node = refresh_node.func
   399	        analyzer.refresh_partial(
   400	            refresh_node,
   401	            patches,
   402	            final_iteration,
   403	            file_node=tree,
   404	            options=state.options,
   405	            active_type=active_type,
   406	        )
   407	        if isinstance(node, Decorator):
   408	            infer_decorator_signature_if_simple(node, analyzer)
   409	    for dep in analyzer.imports:
   410	        state.add_dependency(dep)
   411	        priority = mypy.build.PRI_LOW
   412	        if priority <= state.priorities.get(dep, priority):
   413	            state.priorities[dep] = priority
   414
   415	    # Clear out some stale data to avoid memory leaks and astmerge
   416	    # validity check confusion
   417	    analyzer.statement = None
   418	    del analyzer.cur_mod_node
   419
   420	    if analyzer.deferred:
   421	        return [target], analyzer.incomplete, analyzer.progress
   422	    else:
   423	        return [], analyzer.incomplete, analyzer.progress
