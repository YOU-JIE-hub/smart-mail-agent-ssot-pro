   263	    """Exception raised when there is something wrong with the status file.
   264
   265	    For example:
   266	    - No status file found
   267	    - Status file malformed
   268	    - Process whose pid is in the status file does not exist

   272	def main(argv: list[str]) -> None:
   273	    """The code is top-down."""
   274	    check_python_version("dmypy")
   275
   276	    # set recursion limit consistent with mypy/main.py
   277	    sys.setrecursionlimit(RECURSION_LIMIT)
   278
   279	    args = parser.parse_args(argv)
   280	    if not args.action:
   281	        parser.print_usage()
   282	    else:
   283	        try:
   284	            args.action(args)
   285	        except BadStatus as err:
   286	            fail(err.args[0])
   287	        except Exception:
   288	            # We do this explicitly to avoid exceptions percolating up
   289	            # through mypy.api invocations
   290	            traceback.print_exc()
   291	            sys.exit(2)
   292

   294	def fail(msg: str) -> NoReturn:
   295	    print(msg, file=sys.stderr)
   296	    sys.exit(2)

   302	def action(subparser: argparse.ArgumentParser) -> Callable[[ActionFunction], ActionFunction]:
   303	    """Decorator to tie an action function to a subparser."""

   315	@action(start_parser)
   316	def do_start(args: argparse.Namespace) -> None:
   317	    """Start daemon (it must not already be running).
   318
   319	    This is where mypy flags are set from the command line.
   320
   321	    Setting flags is a bit awkward; you have to use e.g.:
   322
   323	      dmypy start -- --strict

   325	    since we don't want to duplicate mypy's huge list of flags.
   326	    """
   327	    try:
   328	        get_status(args.status_file)
   329	    except BadStatus:
   330	        # Bad or missing status file or dead process; good to start.
   331	        pass
   332	    else:
   333	        fail("Daemon is still alive")
   334	    start_server(args)

   357	def start_server(args: argparse.Namespace, allow_sources: bool = False) -> None:
   358	    """Start the server from command arguments and wait for it."""
   359	    # Lazy import so this import doesn't slow down other commands.
   360	    from mypy.dmypy_server import daemonize, process_start_options
   361
   362	    start_options = process_start_options(args.flags, allow_sources)
   363	    if daemonize(start_options, args.status_file, timeout=args.timeout, log_file=args.log_file):
   364	        sys.exit(2)
   365	    wait_for_server(args.status_file)

   368	def wait_for_server(status_file: str, timeout: float = 5.0) -> None:
   369	    """Wait until the server is up.
   370
   371	    Exit if it doesn't happen within the timeout.
   372	    """
   373	    endtime = time.time() + timeout
   374	    while time.time() < endtime:
   375	        try:
   376	            data = read_status(status_file)
   377	        except BadStatus:
   378	            # If the file isn't there yet, retry later.
   379	            time.sleep(0.1)
   380	            continue
   381	        # If the file's content is bogus or the process is dead, fail.
   382	        check_status(data)
   383	        print("Daemon started")
   384	        return
   385	    fail("Timed out waiting for daemon to start")

   388	@action(run_parser)
   389	def do_run(args: argparse.Namespace) -> None:
   390	    """Do a check, starting (or restarting) the daemon as necessary
   391
   392	    Restarts the daemon if the running daemon reports that it is
   393	    required (due to a configuration change, for example).
   394
   395	    Setting flags is a bit awkward; you have to use e.g.:
   396
   397	      dmypy run -- --strict a.py b.py ...
   398
   399	    since we don't want to duplicate mypy's huge list of flags.
   400	    (The -- is only necessary if flags are specified.)
   401	    """
   402	    if not is_running(args.status_file):
   403	        # Bad or missing status file or dead process; good to start.
   404	        start_server(args, allow_sources=True)
   405	    t0 = time.time()
   406	    response = request(
   407	        args.status_file,
   408	        "run",
   409	        version=__version__,
   410	        args=args.flags,
   411	        export_types=args.export_types,
   412	    )
   413	    # If the daemon signals that a restart is necessary, do it
   414	    if "restart" in response:
   415	        print(f"Restarting: {response['restart']}")
   416	        restart_server(args, allow_sources=True)
   417	        response = request(
   418	            args.status_file,
   419	            "run",
   420	            version=__version__,
   421	            args=args.flags,
   422	            export_types=args.export_types,
   423	        )
   424
   425	    t1 = time.time()
   426	    response["roundtrip_time"] = t1 - t0
   427	    check_output(response, args.verbose, args.junit_xml, args.perf_stats_file)

   430	@action(status_parser)
   431	def do_status(args: argparse.Namespace) -> None:
   432	    """Print daemon status.
   433
   434	    This verifies that it is responsive to requests.
   435	    """
   436	    status = read_status(args.status_file)
   437	    if args.verbose:
   438	        show_stats(status)
   439	    # Both check_status() and request() may raise BadStatus,
   440	    # which will be handled by main().
   441	    check_status(status)
   442	    response = request(
   443	        args.status_file, "status", fswatcher_dump_file=args.fswatcher_dump_file, timeout=5
   444	    )
   445	    if args.verbose or "error" in response:
   446	        show_stats(response)
   447	    if "error" in response:
   448	        fail(f"Daemon may be busy processing; if this persists, consider {sys.argv[0]} kill")
   449	    print("Daemon is up and running")

   558	def check_output(
   559	    response: dict[str, Any], verbose: bool, junit_xml: str | None, perf_stats_file: str | None
   560	) -> None:
   561	    """Print the output from a check or recheck command.
   562
   563	    Call sys.exit() unless the status code is zero.
   564	    """
   565	    if os.name == "nt":
   566	        # Enable ANSI color codes for Windows cmd using this strange workaround
   567	        # ( see https://github.com/python/cpython/issues/74261 )
   568	        os.system("")
   569	    if "error" in response:
   570	        fail(response["error"])
   571	    try:
   572	        out, err, status_code = response["out"], response["err"], response["status"]
   573	    except KeyError:
   574	        fail(f"Response: {str(response)}")
   575	    sys.stdout.write(out)
   576	    sys.stdout.flush()
   577	    sys.stderr.write(err)
   578	    sys.stderr.flush()
   579	    if verbose:
   580	        show_stats(response)
   581	    if junit_xml:
   582	        # Lazy import so this import doesn't slow things down when not writing junit
   583	        from mypy.util import write_junit_xml
   584
   585	        messages = (out + err).splitlines()
   586	        write_junit_xml(
   587	            response["roundtrip_time"],
   588	            bool(err),
   589	            {None: messages} if messages else {},
   590	            junit_xml,
   591	            response["python_version"],
   592	            response["platform"],
   593	        )
   594	    if perf_stats_file:
   595	        telemetry = response.get("stats", {})
   596	        with open(perf_stats_file, "w") as f:
   597	            json.dump(telemetry, f)
   598
   599	    if status_code:
   600	        sys.exit(status_code)

   603	def show_stats(response: Mapping[str, object]) -> None:
   604	    for key, value in sorted(response.items()):
   605	        if key in ("out", "err", "stdout", "stderr"):
   606	            # Special case text output to display just 40 characters of text
   607	            value = repr(value)[1:-1]
   608	            if len(value) > 50:
   609	                value = f"{value[:40]} ... {len(value)-40} more characters"
   610	            print("%-24s: %s" % (key, value))
   611	            continue
   612	        print("%-24s: %10s" % (key, "%.3f" % value if isinstance(value, float) else value))
