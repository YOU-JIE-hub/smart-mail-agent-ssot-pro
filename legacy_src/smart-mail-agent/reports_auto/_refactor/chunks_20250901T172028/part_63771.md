    21	        lines = []
    22	        for line in textwrap.dedent(input).splitlines():
    23	            if line.strip() == '':
    24	                pass
    25	            elif (line.startswith(' ') or
    26	                  line.startswith('except:') or
    27	                  line.startswith('except ') or
    28	                  line.startswith('finally:') or
    29	                  line.startswith('else:') or

    40	                """
    41	            ''')
