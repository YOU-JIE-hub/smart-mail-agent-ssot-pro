    19	def setup_callable_class(builder: IRBuilder) -> None:
    20	    """Generate an (incomplete) callable class representing a function.
    21
    22	    This can be a nested function or a function within a non-extension
    23	    class.  Also set up the 'self' variable for that class.
    24
    25	    This takes the most recently visited function and returns a
    26	    ClassIR to represent that function. Each callable class contains
    27	    an environment attribute which points to another ClassIR
    28	    representing the environment class where some of its variables can
    29	    be accessed.
    30
    31	    Note that some methods, such as '__call__', are not yet
    32	    created here. Use additional functions, such as
    33	    add_call_to_callable_class(), to add them.
    34
    35	    Return a newly constructed ClassIR representing the callable
    36	    class for the nested function.
    37	    """
    38	    # Check to see that the name has not already been taken. If so,
    39	    # rename the class. We allow multiple uses of the same function
    40	    # name because this is valid in if-else blocks. Example:
    41	    #
    42	    #     if True:
    43	    #         def foo():          ---->    foo_obj()
    44	    #             return True
    45	    #     else:
    46	    #         def foo():          ---->    foo_obj_0()
    47	    #             return False
    48	    name = base_name = f"{builder.fn_info.namespaced_name()}_obj"
    49	    count = 0
    50	    while name in builder.callable_class_names:
    51	        name = base_name + "_" + str(count)
    52	        count += 1
    53	    builder.callable_class_names.add(name)
    54
    55	    # Define the actual callable class ClassIR, and set its
    56	    # environment to point at the previously defined environment
    57	    # class.
    58	    callable_class_ir = ClassIR(name, builder.module_name, is_generated=True, is_final_class=True)
    59
    60	    # The functools @wraps decorator attempts to call setattr on
    61	    # nested functions, so we create a dict for these nested
    62	    # functions.
    63	    # https://github.com/python/cpython/blob/3.7/Lib/functools.py#L58
    64	    if builder.fn_info.is_nested:
    65	        callable_class_ir.has_dict = True
    66
    67	    # If the enclosing class doesn't contain nested (which will happen if
    68	    # this is a toplevel lambda), don't set up an environment.
    69	    if builder.fn_infos[-2].contains_nested:
    70	        callable_class_ir.attributes[ENV_ATTR_NAME] = RInstance(builder.fn_infos[-2].env_class)
    71	    callable_class_ir.mro = [callable_class_ir]
    72	    builder.fn_info.callable_class = ImplicitClass(callable_class_ir)
    73	    builder.classes.append(callable_class_ir)
    74
    75	    # Add a 'self' variable to the environment of the callable class,
    76	    # and store that variable in a register to be accessed later.
    77	    self_target = builder.add_self_to_env(callable_class_ir)
    78	    builder.fn_info.callable_class.self_reg = builder.read(self_target, builder.fn_info.fitem.line)

    81	def add_call_to_callable_class(
    82	    builder: IRBuilder,
    83	    args: list[Register],
    84	    blocks: list[BasicBlock],
    85	    sig: FuncSignature,
    86	    fn_info: FuncInfo,
    87	) -> FuncIR:
    88	    """Generate a '__call__' method for a callable class representing a nested function.
    89
    90	    This takes the blocks and signature associated with a function
    91	    definition and uses those to build the '__call__' method of a
    92	    given callable class, used to represent that function.
    93	    """
    94	    # Since we create a method, we also add a 'self' parameter.
    95	    nargs = len(sig.args) - sig.num_bitmap_args
    96	    sig = FuncSignature(
    97	        (RuntimeArg(SELF_NAME, object_rprimitive),) + sig.args[:nargs], sig.ret_type
    98	    )
    99	    call_fn_decl = FuncDecl("__call__", fn_info.callable_class.ir.name, builder.module_name, sig)
   100	    call_fn_ir = FuncIR(
   101	        call_fn_decl, args, blocks, fn_info.fitem.line, traceback_name=fn_info.fitem.name
   102	    )
   103	    fn_info.callable_class.ir.methods["__call__"] = call_fn_ir
   104	    fn_info.callable_class.ir.method_decls["__call__"] = call_fn_decl
   105	    return call_fn_ir

   108	def add_get_to_callable_class(builder: IRBuilder, fn_info: FuncInfo) -> None:
   109	    """Generate the '__get__' method for a callable class."""
   110	    line = fn_info.fitem.line
   111	    with builder.enter_method(
   112	        fn_info.callable_class.ir,
   113	        "__get__",
   114	        object_rprimitive,
   115	        fn_info,
   116	        self_type=object_rprimitive,
   117	    ):
   118	        instance = builder.add_argument("instance", object_rprimitive)
   119	        builder.add_argument("owner", object_rprimitive)
   120
   121	        # If accessed through the class, just return the callable
   122	        # object. If accessed through an object, create a new bound
   123	        # instance method object.
   124	        instance_block, class_block = BasicBlock(), BasicBlock()
   125	        comparison = builder.translate_is_op(
   126	            builder.read(instance), builder.none_object(), "is", line
   127	        )
   128	        builder.add_bool_branch(comparison, class_block, instance_block)
   129
   130	        builder.activate_block(class_block)
   131	        builder.add(Return(builder.self()))
   132
   133	        builder.activate_block(instance_block)
   134	        builder.add(
   135	            Return(builder.call_c(method_new_op, [builder.self(), builder.read(instance)], line))
   136	        )

   139	def instantiate_callable_class(builder: IRBuilder, fn_info: FuncInfo) -> Value:
   140	    """Create an instance of a callable class for a function.
   141
   142	    Calls to the function will actually call this instance.
   143
   144	    Note that fn_info refers to the function being assigned, whereas
   145	    builder.fn_info refers to the function encapsulating the function
   146	    being turned into a callable class.
   147	    """
   148	    fitem = fn_info.fitem
   149	    func_reg = builder.add(Call(fn_info.callable_class.ir.ctor, [], fitem.line))
   150
   151	    # Set the environment attribute of the callable class to point at
   152	    # the environment class defined in the callable class' immediate
   153	    # outer scope. Note that there are three possible environment
   154	    # class registers we may use. This depends on what the encapsulating
   155	    # (parent) function is:
   156	    #
   157	    # - A nested function: the callable class is instantiated
   158	    #   from the current callable class' '__call__' function, and hence
   159	    #   the callable class' environment register is used.
   160	    # - A generator function: the callable class is instantiated
   161	    #   from the '__next__' method of the generator class, and hence the
   162	    #   environment of the generator class is used.
   163	    # - Regular function: we use the environment of the original function.
   164	    curr_env_reg = None
   165	    if builder.fn_info.is_generator:
   166	        curr_env_reg = builder.fn_info.generator_class.curr_env_reg
   167	    elif builder.fn_info.is_nested:
   168	        curr_env_reg = builder.fn_info.callable_class.curr_env_reg
   169	    elif builder.fn_info.contains_nested:
   170	        curr_env_reg = builder.fn_info.curr_env_reg
   171	    if curr_env_reg:
   172	        builder.add(SetAttr(func_reg, ENV_ATTR_NAME, curr_env_reg, fitem.line))
   173	    return func_reg

    87	def transform_class_def(builder: IRBuilder, cdef: ClassDef) -> None:
    88	    """Create IR for a class definition.
    89
    90	    This can generate both extension (native) and non-extension
    91	    classes.  These are generated in very different ways. In the
    92	    latter case we construct a Python type object at runtime by doing
    93	    the equivalent of "type(name, bases, dict)" in IR. Extension
    94	    classes are defined via C structs that are generated later in
    95	    mypyc.codegen.emitclass.
    96
    97	    This is the main entry point to this module.
    98	    """

   100	        builder.error("Nested class definitions not supported", cdef.line)
   101	        return
   102
   103	    ir = builder.mapper.type_to_ir[cdef.info]
   104

   106	    # classes aren't necessarily populated yet at
   107	    # prepare_class_def time.
   108	    if any(ir.base_mro[i].base != ir.base_mro[i + 1] for i in range(len(ir.base_mro) - 1)):
   109	        builder.error("Multiple inheritance is not supported (except for traits)", cdef.line)
   110
   111	    if ir.allow_interpreted_subclasses:
   112	        for parent in ir.mro:
   113	            if not parent.allow_interpreted_subclasses:
   114	                builder.error(
   115	                    'Base class "{}" does not allow interpreted subclasses'.format(
   116	                        parent.fullname
   117	                    ),
   118	                    cdef.line,
   119	                )

   121	    # Currently, we only create non-extension classes for classes that are
   122	    # decorated or inherit from Enum. Classes decorated with @trait do not
   123	    # apply here, and are handled in a different way.

   129	            cls_builder = DataClassBuilder(builder, cdef)
   130	        elif cls_type == "attr":
   131	            cls_builder = AttrsClassBuilder(builder, cdef)
   132	        else:

   135	        cls_builder = NonExtClassBuilder(builder, cdef)

   137	    for stmt in cdef.defs.body:
   138	        if isinstance(stmt, OverloadedFuncDef) and stmt.is_property:

   155	            isinstance(stmt, ExpressionStmt) and isinstance(stmt.expr, EllipsisExpr)
   156	        ):

   161	                continue

   163	            if not isinstance(lvalue, NameExpr):
   164	                builder.error(
   165	                    "Only assignment to variables is supported in class bodies", stmt.line
   166	                )
   167	                continue

   169	            # non-extension classes and fake dataclass ones.
   170	            cls_builder.add_attr(lvalue, stmt)

   174	            pass
   175	        else:

   196	            # TODO: Do we need interpreted glue methods? Maybe not?
