   107	def align8to32(bytes: bytes, width: int, mode: str) -> bytes:
   108	    """
   109	    converts each scanline of data from 8 bit to 32 bit aligned
   110	    """
   111
   112	    bits_per_pixel = {"1": 1, "L": 8, "P": 8, "I;16": 16}[mode]
   113
   114	    # calculate bytes per line and the extra padding if needed
   115	    bits_per_line = bits_per_pixel * width
   116	    full_bytes_per_line, remaining_bits_per_line = divmod(bits_per_line, 8)
   117	    bytes_per_line = full_bytes_per_line + (1 if remaining_bits_per_line else 0)
   118
   119	    extra_padding = -bytes_per_line % 4
   120
   121	    # already 32 bit aligned by luck
   122	    if not extra_padding:
   123	        return bytes
   124
   125	    new_data = [
   126	        bytes[i * bytes_per_line : (i + 1) * bytes_per_line] + b"\x00" * extra_padding
   127	        for i in range(len(bytes) // bytes_per_line)
   128	    ]
   129
   130	    return b"".join(new_data)

   134	    data = None
   135	    colortable = None
   136	    exclusive_fp = False

   138	    # handle filename, if given instead of image name
