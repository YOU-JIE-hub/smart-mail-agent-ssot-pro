pe = self.init_type or self.info[self.name].type
   156
   157	        unannotated = False
   158	        if init_type is None:
   159	            unannotated = True
   160	            # Convert type not set to Any.
   161	            init_type = AnyType(TypeOfAny.unannotated)
   162	        else:
   163	            proper_type = get_proper_type(init_type)
   164	            if isinstance(proper_type, AnyType):
   165	                if proper_type.type_of_any == TypeOfAny.unannotated:
   166	                    unannotated = True
   167
   168	        if unannotated and ctx.api.options.disallow_untyped_defs:
   169	            # This is a compromise.  If you don't have a type here then the
   170	            # __init__ will be untyped. But since the __init__ is added it's
   171	            # pointing at the decorator. So instead we also show the error in the
   172	            # assignment, which is where you would fix the issue.
   173	            node = self.info[self.name].node
   174	            assert node is not None
   175	            ctx.api.msg.need_annotation_for_var(node, self.context)
   176
   177	        if self.kw_only:
   178	            arg_kind = ARG_NAMED_OPT if self.has_default else ARG_NAMED
   179	        else:
   180	            arg_kind = ARG_OPT if self.has_default else ARG_POS
   181
   182	        # Attrs removes leading underscores when creating the __init__ arguments.
   183	        name = self.alias or self.name.lstrip("_")
   184	        return Argument(Var(name, init_type), init_type, None, arg_kind)
   185
   186	    def serialize(self) -> JsonDict:
   187	        """Serialize this object so it can be saved and restored."""
   188	        return {
   189	            "name": self.name,
   190	            "alias": self.alias,
   191	            "has_default": self.has_default,
   192	            "init": self.init,
   193	            "kw_only": self.kw_only,
   194	            "has_converter": self.converter is not None,
   195	            "converter_init_type": (
   196	                self.converter.init_type.serialize()
   197	                if self.converter and self.converter.init_type
   198	                else None
   199	            ),
   200	            "context_line": self.context.line,
   201	            "context_column": self.context.column,
   202	            "init_type": self.init_type.serialize() if self.init_type else None,
   203	        }
   204
   205	    @classmethod
   206	    def deserialize(
   207	        cls, info: TypeInfo, data: JsonDict, api: SemanticAnalyzerPluginInterface
   208	    ) -> Attribute:
   209	        """Return the Attribute that was serialized."""
   210	        raw_init_type = data["init_type"]
   211	        init_type = deserialize_and_fixup_type(raw_init_type, api) if raw_init_type else None
   212	        raw_converter_init_type = data["converter_init_type"]
   213	        converter_init_type = (
   214	            deserialize_and_fixup_type(raw_converter_init_type, api)
   215	            if raw_converter_init_type
   216	            else None
   217	        )
   218
   219	        return Attribute(
   220	            data["name"],
   221	            data["alias"],
   222	            info,
   223	            data["has_default"],
   224	            data["init"],
   225	            data["kw_only"],
   226	            Converter(converter_init_type) if data["has_converter"] else None,
   227	            Context(line=data["context_line"], column=data["context_column"]),
   228	            init_type,
   229	        )

   231	    def expand_typevar_from_subtype(self, sub_type: TypeInfo) -> None:
   232	        """Expands type vars in the context of a subtype when an attribute is inherited
   233	        from a generic super type."""
   234	        if self.init_type:
   235	            self.init_type = map_type_from_supertype(self.init_type, sub_type, self.info)
   236	        else:
   237	            self.init_type = None

   240	def _determine_eq_order(ctx: mypy.plugin.ClassDefContext) -> bool:
   241	    """
   242	    Validate the combination of *cmp*, *eq*, and *order*. Derive the effective
   243	    value of order.
   244	    """
   245	    cmp = _get_decorator_optional_bool_argument(ctx, "cmp")
   246	    eq = _get_decorator_optional_bool_argument(ctx, "eq")
   247	    order = _get_decorator_optional_bool_argument(ctx, "order")
   248
   249	    if cmp is not None and any((eq is not None, order is not None)):
   250	        ctx.api.fail('Don\'t mix "cmp" with "eq" and "order"', ctx.reason)
   251
   252	    # cmp takes precedence due to bw-compatibility.
   253	    if cmp is not None:
   254	        return cmp
   255
   256	    # If left None, equality is on and ordering mirrors equality.
   257	    if eq is None:
   258	        eq = True
   259
   260	    if order is None:
   261	        order = eq
   262
   263	    if eq is False and order is True:
   264	        ctx.api.fail("eq must be True if order is True", ctx.reason)
   265
   266	    return order
   267

   269	def _get_decorator_optional_bool_argument(
   270	    ctx: mypy.plugin.ClassDefContext, name: str, default: bool | None = None
   271	) -> bool | None:
   272	    """Return the Optional[bool] argument for the decorator.
   273
   274	    This handles both @decorator(...) and @decorator.
   275	    """
   276	    if isinstance(ctx.reason, CallExpr):
   277	        attr_value = _get_argument(ctx.reason, name)
   278	        if attr_value:
   279	            if isinstance(attr_value, NameExpr):
   280	                if attr_value.fullname == "builtins.True":
   281	                    return True
   282	                if attr_value.fullname == "builtins.False":
   283	                    return False
   284	                if attr_value.fullname == "builtins.None":
   285	                    return None
   286	            ctx.api.fail(
   287	                f'"{name}" argument must be a True, False, or None literal',
   288	                ctx.reason,
   289	                code=LITERAL_REQ,
   290	            )
   291	            return default
   292	        return default
   293	    else:
   294	        return default

   307	def attr_class_maker_callback(
   308	    ctx: mypy.plugin.ClassDefContext,
   309	    auto_attribs_default: bool | None = False,
   310	    frozen_default: bool = False,
   311	    slots_default: bool = False,
   312	) -> bool:
   313	    """Add necessary dunder methods to classes decorated with attr.s.
   314
   315	    attrs is a package that lets you define classes without writing dull boilerplate code.
   316
   317	    At a quick glance, the decorator searches the class body for assignments of `attr.ib`s (or
   318	    annotated variables if auto_attribs=True), then depending on how the decorator is called,
   319	    it will add an __init__ or all the compare methods.
   320	    For frozen=True it will turn the attrs into properties.
   321
   322	    Hashability will be set according to https://www.attrs.org/en/stable/hashing.html.
   323
   324	    See https://www.attrs.org/en/stable/how-does-it-work.html for information on how attrs works.
   325
   326	    If this returns False, some required metadata was not ready yet, and we need another
   327	    pass.
   328	    """
   329	    with state.strict_optional_set(ctx.api.options.strict_optional):
   330	        # This hook is called during semantic analysis, but it uses a bunch of
   331	        # type-checking ops, so it needs the strict optional set properly.
   332	        return attr_class_maker_callback_impl(
   333	            ctx, auto_attribs_default, frozen_default, slots_default
   334	        )
   335

   337	def attr_class_maker_callback_impl(
   338	    ctx: mypy.plugin.ClassDefContext,
   339	    auto_attribs_default: bool | None,
   340	    frozen_default: bool,
   341	    slots_default: bool,
   342	) -> bool:
   343	    info = ctx.cls.info
   344
   345	    init = _get_decorator_bool_argument(ctx, "init", True)
   346	    frozen = _get_frozen(ctx, frozen_default)
   347	    order = _determine_eq_order(ctx)
   348	    slots = _get_decorator_bool_argument(ctx, "slots", slots_default)
   349
   350	    auto_attribs = _get_decorator_optional_bool_argument(ctx, "auto_attribs", auto_attribs_default)
   351	    kw_only = _get_decorator_bool_argument(ctx, "kw_only", False)
   352	    match_args = _get_decorator_bool_argument(ctx, "match_args", True)
   353
   354	    for super_info in ctx.cls.info.mro[1:-1]:
   355	        if "attrs_tag" in super_info.metadata and "attrs" not in super_info.metadata:
   356	            # Super class is not ready yet. Request another pass.
   357	            return False
   358
   359	    attributes = _analyze_class(ctx, auto_attribs, kw_only)
   360
   361	    # Check if attribute types are ready.
   362	    for attr in attributes:
   363	        node = info.get(attr.name)
   364	        if node is None:
   365	            # This name is likely blocked by some semantic analysis error that
   366	            # should have been reported already.
   367	            _add_empty_metadata(info)
   368	            return True
   369
   370	    _add_attrs_magic_attribute(ctx, [(attr.name, info[attr.name].type) for attr in attributes])
   371	    if slots:
   372	        _add_slots(ctx, attributes)
   373	    if match_args and ctx.api.options.python_version[:2] >= (3, 10):
   374	        # `.__match_args__` is only added for python3.10+, but the argument
   375	        # exists for earlier versions as well.
   376	        _add_match_args(ctx, attributes)
   377
   378	    # Save the attributes so that subclasses can reuse them.
   379	    ctx.cls.info.metadata["attrs"] = {
   380	        "attributes": [attr.serialize() for attr in attributes],
   381	        "frozen": frozen,
   382	    }
   383
   384	    adder = MethodAdder(ctx)
   385	    # If  __init__ is not being generated, attrs still generates it as __attrs_init__ instead.
   386	    _add_init(ctx, attributes, adder, "__init__" if init else ATTRS_INIT_NAME)
   387
   388	    if order:
   389	        _add_order(ctx, adder)
   390	    if frozen:
   391	        _make_frozen(ctx, attributes)
   392	        # Frozen classes are hashable by default, even if inheriting from non-frozen ones.
   393	        hashable: bool | None = _get_decorator_bool_argument(
   394	            ctx, "hash", True
   395	        ) and _get_decorator_bool_argument(ctx, "unsafe_hash", True)
   396	    else:
   397	        hashable = _get_decorator_optional_bool_argument(ctx, "unsafe_hash")
   398	        if hashable is None:  # unspecified
   399	            hashable = _get_decorator_optional_bool_argument(ctx, "hash")
   400
   401	    eq = _get_decorator_optional_bool_argument(ctx, "eq")
   402	    has_own_hash = "__hash__" in ctx.cls.info.names
   403
   404	    if has_own_hash or (hashable is None and eq is False):
   405	        pass  # Do nothing.
   406	    elif hashable:
   407	        # We copy the `__hash__` signature from `object` to make them hashable.
   408	        ctx.cls.info.names["__hash__"] = ctx.cls.info.mro[-1].names["__hash__"]
   409	    else:
   410	        _remove_hashability(ctx)
   411
   412	    return True
   413

   415	def _get_frozen(ctx: mypy.plugin.ClassDefContext, frozen_default: bool) -> bool:
   416	    """Return whether this class is frozen."""
   417	    if _get_decorator_bool_argument(ctx, "frozen", frozen_default):
   418	        return True
   419	    # Subclasses of frozen classes are frozen so check that.
   420	    for super_info in ctx.cls.info.mro[1:-1]:

   426	def _analyze_class(
   427	    ctx: mypy.plugin.ClassDefContext, auto_attribs: bool | None, kw_only: bool
   428	) -> list[Attribute]:
   429	    """Analyze the class body of an attr maker, its parents, and return the Attributes found.
   430
   431	    auto_attribs=True means we'll generate attributes from type annotations also.
   432	    auto_attribs=None means we'll detect which mode to use.
   433	    kw_only=True means that all attributes created here will be keyword only args in __init__.
   434	    """
   435	    own_attrs: dict[str, Attribute] = {}
   436	    if auto_attribs is None:
   437	        auto_attribs = _detect_auto_attribs(ctx)
   438
   439	    # Walk the body looking for assignments and decorators.
   440	    for stmt in ctx.cls.defs.body:
   441	        if isinstance(stmt, AssignmentStmt):
   442	            for attr in _attributes_from_assignment(ctx, stmt, auto_attribs, kw_only):
   443	                # When attrs are defined twice in the same body we want to use the 2nd definition
   444	                # in the 2nd location. So remove it from the OrderedDict.
   445	                # Unless it's auto_attribs in which case we want the 2nd definition in the
   446	                # 1st location.
   447	                if not auto_attribs and attr.name in own_attrs:
   448	                    del own_attrs[attr.name]
   449	                own_attrs[attr.name] = attr
   450	        elif isinstance(stmt, Decorator):
   451	            _cleanup_decorator(stmt, own_attrs)
   452
   453	    for attribute in own_attrs.values():
   454	        # Even though these look like class level assignments we want them to look like
   455	        # instance level assignments.
   456	        if attribute.name in ctx.cls.info.names:
   457	            node = ctx.cls.info.names[attribute.name].node
   458	            if isinstance(node, PlaceholderNode):
   459	                # This node is not ready yet.
   460	                continue
   461	            assert isinstance(node, Var)
   462	            node.is_initialized_in_class = False
   463
   464	    # Traverse the MRO and collect attributes from the parents.
   465	    taken_attr_names = set(own_attrs)
   466	    super_attrs = []
   467	    for super_info in ctx.cls.info.mro[1:-1]:
   468	        if "attrs" in super_info.metadata:
   469	            # Each class depends on the set of attributes in its attrs ancestors.
   470	            ctx.api.add_plugin_dependency(make_wildcard_trigger(super_info.fullname))
   471
   472	            for data in super_info.metadata["attrs"]["attributes"]:
   473	                # Only add an attribute if it hasn't been defined before.  This
   474	                # allows for overwriting attribute definitions by subclassing.
   475	                if data["name"] not in taken_attr_names:
   476	                    a = Attribute.deserialize(super_info, data, ctx.api)
   477	                    a.expand_typevar_from_subtype(ctx.cls.info)
   478	                    super_attrs.append(a)
   479	                    taken_attr_names.add(a.name)
   480	    attributes = super_attrs + list(own_attrs.values())
   481
   482	    # Check the init args for correct default-ness.  Note: This has to be done after all the
   483	    # attributes for all classes have been read, because subclasses can override parents.
   484	    last_default = False
   485
   486	    for i, attribute in enumerate(attributes):
   487	        if not attribute.init:
   488	            continue
   489
   490	        if attribute.kw_only:
   491	            # Keyword-only attributes don't care whether they are default or not.
   492	            continue
   493
   494	        # If the issue comes from merging different classes, report it
   495	        # at the class definition point.
   496	        context = attribute.context if i >= len(super_attrs) else ctx.cls
   497
   498	        if not attribute.has_default and last_default:
   499	            ctx.api.fail("Non-default attributes not allowed after default attributes.", context)
   500	        last_default |= attribute.has_default
   501
   502	    return attributes
   503

   510	def _detect_auto_attribs(ctx: mypy.plugin.ClassDefContext) -> bool:
   511	    """Return whether auto_attribs should be enabled or disabled.
   512
   513	    It's disabled if there are any unannotated attribs()
   514	    """
   515	    for stmt in ctx.cls.defs.body:
   516	        if isinstance(stmt, AssignmentStmt):
   517	            for lvalue in stmt.lvalues:
   518	                lvalues, rvalues = _parse_assignments(lvalue, stmt)
   519
   520	                if len(lvalues) != len(rvalues):
   521	                    # This means we have some assignment that isn't 1 to 1.
   522	                    # It can't be an attrib.
   523	                    continue
   524
   525	                for lhs, rvalue in zip(lvalues, rvalues):
   526	                    # Check if the right hand side is a call to an attribute maker.
   527	                    if (
   528	                        isinstance(rvalue, CallExpr)
   529	                        and isinstance(rvalue.callee, RefExpr)
   530	                        and rvalue.callee.fullname in attr_attrib_makers
   531	                        and not stmt.new_syntax
   532	                    ):
   533	                        # This means we have an attrib without an annotation and so
   534	                        # we can't do auto_attribs=True
   535	                        return False
   536	    return True

   539	def _attributes_from_assignment(
   540	    ctx: mypy.plugin.ClassDefContext, stmt: AssignmentStmt, auto_attribs: bool, kw_only: bool
   541	) -> Iterable[Attribute]:
   542	    """Return Attribute objects that are created by this assignment.
   543
   544	    The assignments can look like this:
   545	        x = attr.ib()
   546	        x = y = attr.ib()
   547	        x, y = attr.ib(), attr.ib()
   548	    or if auto_attribs is enabled also like this:
   549	        x: type
   550	        x: type = default_value
   551	        x: type = attr.ib(...)
   552	    """
   553	    for lvalue in stmt.lvalues:
   554	        lvalues, rvalues = _parse_assignments(lvalue, stmt)
   555
   556	        if len(lvalues) != len(rvalues):
   557	            # This means we have some assignment that isn't 1 to 1.
   558	            # It can't be an attrib.
   559	            continue
   560
   561	        for lhs, rvalue in zip(lvalues, rvalues):
   562	            # Check if the right hand side is a call to an attribute maker.
   563	            if (
   564	                isinstance(rvalue, CallExpr)
   565	                and isinstance(rvalue.callee, RefExpr)
   566	                and rvalue.callee.fullname in attr_attrib_makers
   567	            ):
   568	                attr = _attribute_from_attrib_maker(ctx, auto_attribs, kw_only, lhs, rvalue, stmt)
   569	                if attr:
   570	                    yield attr
   571	            elif auto_attribs and stmt.type and stmt.new_syntax and not is_class_var(lhs):
   572	                yield _attribute_from_auto_attrib(ctx, kw_only, lhs, rvalue, stmt)

   575	def _cleanup_decorator(stmt: Decorator, attr_map: dict[str, Attribute]) -> None:
   576	    """Handle decorators in class bodies.
   577
   578	    `x.default` will set a default value on x
   579	    `x.validator` and `x.default` will get removed to avoid throwing a type error.
   580	    """
   581	    remove_me = []
   582	    for func_decorator in stmt.decorators:
   583	        if (
   584	            isinstance(func_decorator, MemberExpr)
   585	            and isinstance(func_decorator.expr, NameExpr)
   586	            and func_decorator.expr.name in attr_map
   587	        ):
   588	            if func_decorator.name == "default":
   589	                attr_map[func_decorator.expr.name].has_default = True
   590
   591
