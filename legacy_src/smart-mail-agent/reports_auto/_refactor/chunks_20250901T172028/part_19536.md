   158	def _quadrantDef(xpos, ypos, corner, r, kind=0, direction='left-right', m=0.4472):
   159	    t = m*r
   160	    if xpos=='right' and ypos=='bottom': #bottom right
   161	        xhi,ylo = corner
   162	        P = [(xhi - r, ylo),(xhi-t, ylo), (xhi, ylo + t), (xhi, ylo + r)]
   163	    elif xpos=='right' and ypos=='top': #top right
   164	        xhi,yhi = corner
   165	        P = [(xhi, yhi - r),(xhi, yhi - t), (xhi - t, yhi), (xhi - r, yhi)]
   166	    elif xpos=='left' and ypos=='top': #top left
   167	        xlo,yhi = corner
   168	        P = [(xlo + r, yhi),(xlo + t, yhi), (xlo, yhi - t), (xlo, yhi - r)]
   169	    elif xpos=='left' and ypos=='bottom': #bottom left
   170	        xlo,ylo = corner
   171	        P = [(xlo, ylo + r),(xlo, ylo + t), (xlo + t, ylo), (xlo + r, ylo)]
   172	    else:
   173	        raise ValueError(f'Unknown quadrant position (xpos,ypos)={(xpos,ypos)!r}')
   174	    if direction=='left-right' and P[0][0]>P[-1][0] or direction=='bottom-top' and P[0][1]>P[-1][1]:
   175	        P.reverse()
   176	    P = _calcBezierPoints(P, kind)
   177	    return P

   226	def spanFixDim(V0,V,spanCons,lim=None,FUZZ=rl_config._FUZZ):
   227	    #assign required space to variable rows equally to existing calculated values
   228	    M = {}
   229	    if not lim: lim = len(V0)   #in longtables the row calcs may be truncated
   230
   231	    #we assign the largest spaces first hoping to get a smaller result
   232	    for v,(x0,x1) in reversed(sorted(((iv,ik) for ik,iv in spanCons.items()))):
   233	        if x0>=lim: continue
   234	        x1 += 1
   235	        t = sum([V[x]+M.get(x,0) for x in range(x0,x1)])
   236	        if t>=v-FUZZ: continue      #already good enough
   237	        X = [x for x in range(x0,x1) if V0[x] is None] #variable candidates
   238	        if not X: continue          #something wrong here mate
   239	        v -= t
   240	        v /= float(len(X))
   241	        for x in X:
   242	            M[x] = M.get(x,0)+v
   243	    for x,v in M.items():
   244	        V[x] += v
