   902	    def parse_flags(self, program_text: str, extra: list[str]) -> Options:
   903	        flags = re.search("# flags: (.*)$", program_text, flags=re.MULTILINE)
   904	        pyversion = None
   905	        if flags:
   906	            flag_list = flags.group(1).split()
   907	            for i, flag in enumerate(flag_list):
   908	                if flag.startswith("--python-version="):
   909	                    pyversion = flag.split("=", 1)[1]
   910	                    del flag_list[i]
   911	                    break
   912	        else:
   913	            flag_list = []
   914	        options = parse_options(flag_list + extra)
   915	        if pyversion:
   916	            # A hack to allow testing old python versions with new language constructs
   917	            # This should be rarely used in general as stubgen output should not be version-specific
   918	            major, minor = pyversion.split(".", 1)
   919	            options.pyversion = (int(major), int(minor))
   920	        if "--verbose" not in flag_list:
   921	            options.quiet = True
   922	        else:
   923	            options.verbose = True
   924	        return options
