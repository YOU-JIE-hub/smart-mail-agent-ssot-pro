ndant_condition_in_if(True, e.cond)
  5834
  5835	        if_type = self.analyze_cond_branch(
  5836	            if_map, e.if_expr, context=ctx, allow_none_return=allow_none_return
  5837	        )
  5838
  5839	        # we want to keep the narrowest value of if_type for union'ing the branches
  5840	        # however, it would be silly to pass a literal as a type context. Pass the
  5841	        # underlying fallback type instead.
  5842	        if_type_fallback = simple_literal_type(get_proper_type(if_type)) or if_type
  5843
  5844	        # Analyze the right branch using full type context and store the type
  5845	        full_context_else_type = self.analyze_cond_branch(
  5846	            else_map, e.else_expr, context=ctx, allow_none_return=allow_none_return
  5847	        )
  5848
  5849	        if not mypy.checker.is_valid_inferred_type(if_type, self.chk.options):
  5850	            # Analyze the right branch disregarding the left branch.
  5851	            else_type = full_context_else_type
  5852	            # we want to keep the narrowest value of else_type for union'ing the branches
  5853	            # however, it would be silly to pass a literal as a type context. Pass the
  5854	            # underlying fallback type instead.
  5855	            else_type_fallback = simple_literal_type(get_proper_type(else_type)) or else_type
  5856
  5857	            # If it would make a difference, re-analyze the left
  5858	            # branch using the right branch's type as context.
  5859	            if ctx is None or not is_equivalent(else_type_fallback, ctx):
  5860	                # TODO: If it's possible that the previous analysis of
  5861	                # the left branch produced errors that are avoided
  5862	                # using this context, suppress those errors.
  5863	                if_type = self.analyze_cond_branch(
  5864	                    if_map,
  5865	                    e.if_expr,
  5866	                    context=else_type_fallback,
  5867	                    allow_none_return=allow_none_return,
  5868	                )
  5869
  5870	        elif if_type_fallback == ctx:
  5871	            # There is no point re-running the analysis if if_type is equal to ctx.
  5872	            # That would  be an exact duplicate of the work we just did.
  5873	            # This optimization is particularly important to avoid exponential blowup with nested
  5874	            # if/else expressions: https://github.com/python/mypy/issues/9591
  5875	            # TODO: would checking for is_proper_subtype also work and cover more cases?
  5876	            else_type = full_context_else_type
  5877	        else:
  5878	            # Analyze the right branch in the context of the left
  5879	            # branch's type.
  5880	            else_type = self.analyze_cond_branch(
  5881	                else_map,
  5882	                e.else_expr,
  5883	                context=if_type_fallback,
  5884	                allow_none_return=allow_none_return,
  5885	            )
  5886
  5887	        # In most cases using if_type as a context for right branch gives better inferred types.
  5888	        # This is however not the case for literal types, so use the full context instead.
  5889	        if is_literal_type_like(full_context_else_type) and not is_literal_type_like(else_type):
  5890	            else_type = full_context_else_type
  5891
  5892	        res: Type = make_simplified_union([if_type, else_type])
  5893	        if has_uninhabited_component(res) and not isinstance(
  5894	            get_proper_type(self.type_context[-1]), UnionType
  5895	        ):
  5896	            # In rare cases with empty collections join may give a better result.
  5897	            alternative = join.join_types(if_type, else_type)
  5898	            p_alt = get_proper_type(alternative)
  5899	            if not isinstance(p_alt, Instance) or p_alt.type.fullname != "builtins.object":
  5900	                res = alternative
  5901	        return res

  5903	    def analyze_cond_branch(
  5904	        self,
  5905	        map: dict[Expression, Type] | None,
  5906	        node: Expression,
  5907	        context: Type | None,
  5908	        allow_none_return: bool = False,
  5909	        suppress_unreachable_errors: bool = True,
  5910	    ) -> Type:
  5911	        with self.chk.binder.frame_context(can_skip=True, fall_through=0):
  5912	            if map is None:
  5913	                # We still need to type check node, in case we want to
  5914	                # process it for isinstance checks later. Since the branch was
  5915	                # determined to be unreachable, any errors should be suppressed.
  5916	                with self.msg.filter_errors(filter_errors=suppress_unreachable_errors):
  5917	                    self.accept(node, type_context=context, allow_none_return=allow_none_return)
  5918	                return UninhabitedType()
  5919	            self.chk.push_type_map(map)

  5926	    def accept(
  5927	        self,
  5928	        node: Expression,
  5929	        type_context: Type | None = None,
  5930	        allow_none_return: bool = False,
  5931	        always_allow_any: bool = False,
  5932	        is_callee: bool = False,
  5933	    ) -> Type:
  5934	        """Type check a node in the given type context.  If allow_none_return
  5935	        is True and this expression is a call, allow it to return None.  This
  5936	        applies only to this expression and not any subexpressions.
  5937	        """
  5938	        if node in self.type_overrides:
  5939	            # This branch is very fast, there is no point timing it.
  5940	            return self.type_overrides[node]
  5941	        # We don't use context manager here to get most precise data (and avoid overhead).
  5942	        record_time = False
  5943	        if self.collect_line_checking_stats and not self.in_expression:
  5944	            t0 = time.perf_counter_ns()
  5945	            self.in_expression = True
  5946	            record_time = True
  5947	        self.type_context.append(type_context)
  5948	        old_is_callee = self.is_callee
  5949	        self.is_callee = is_callee
  5950	        try:
  5951	            if allow_none_return and isinstance(node, CallExpr):
  5952	                typ = self.visit_call_expr(node, allow_none_return=True)
  5953	            elif allow_none_return and isinstance(node, YieldFromExpr):
  5954	                typ = self.visit_yield_from_expr(node, allow_none_return=True)
  5955	            elif allow_none_return and isinstance(node, ConditionalExpr):
  5956	                typ = self.visit_conditional_expr(node, allow_none_return=True)
  5957	            elif allow_none_return and isinstance(node, AwaitExpr):
  5958	                typ = self.visit_await_expr(node, allow_none_return=True)
  5959	            else:
  5960	                typ = node.accept(self)
  5961	        except Exception as err:
  5962	            report_internal_error(
  5963	                err, self.chk.errors.file, node.line, self.chk.errors, self.chk.options
  5964	            )
  5965	        self.is_callee = old_is_callee
  5966	        self.type_context.pop()
  5967	        assert typ is not None
  5968	        self.chk.store_type(node, typ)
  5969
  5970	        if (
  5971	            self.chk.options.disallow_any_expr
  5972	            and not always_allow_any
  5973	            and not self.chk.is_stub
  5974	            and self.chk.in_checked_function()
  5975	            and has_any_type(typ)
  5976	            and not self.chk.current_node_deferred
  5977	        ):
  5978	            self.msg.disallowed_any_type(typ, node)
  5979
  5980	        if not self.chk.in_checked_function() or self.chk.current_node_deferred:
  5981	            result: Type = AnyType(TypeOfAny.unannotated)
  5982	        else:
  5983	            result = typ
  5984	        if record_time:
  5985	            self.per_line_checking_time_ns[node.line] += time.perf_counter_ns() - t0
  5986	            self.in_expression = False
  5987	        return result
