             if module_name.startswith(self.module_name + "."):
   420	                    # from {.rel_name} import {mod_name} as {name}
   421	                    pkg_name, mod_name = module_name.rsplit(".", 1)
   422	                    rel_module = pkg_name[len(self.module_name) :] or "."
   423	                    self.import_tracker.add_import_from(rel_module, [(mod_name, name)])
   424	                    self.import_tracker.reexport(name)
   425	                else:
   426	                    # import {module_name} as {name}
   427	                    self.import_tracker.add_import(module_name, name)
   428	                    self.import_tracker.reexport(name)
   429	            elif self.is_defined_in_module(obj) and not inspect.ismodule(obj):
   430	                # process this below
   431	                items.append((name, obj))
   432	            else:
   433	                # from {obj_module} import {obj_name}
   434	                obj_module_name = self.get_obj_module(obj)
   435	                if obj_module_name:
   436	                    self.import_tracker.add_import_from(obj_module_name, [(name, None)])
   437	                    if self.should_reexport(name, obj_module_name, name_is_alias=False):
   438	                        self.import_tracker.reexport(name)
   439
   440	        self.set_defined_names({name for name, obj in all_items if not inspect.ismodule(obj)})
   441
   442	        if self.resort_members:
   443	            functions: list[str] = []
   444	            types: list[str] = []
   445	            variables: list[str] = []
   446	        else:
   447	            output: list[str] = []
   448	            functions = types = variables = output
   449
   450	        for name, obj in items:
   451	            if self.is_function(obj):
   452	                self.generate_function_stub(name, obj, output=functions)
   453	            elif inspect.isclass(obj):
   454	                self.generate_class_stub(name, obj, output=types)
   455	            else:
   456	                self.generate_variable_stub(name, obj, output=variables)
   457
   458	        self._output = []
   459
   460	        if self.resort_members:
   461	            for line in variables:
   462	                self._output.append(line + "\n")
   463	            for line in types:
   464	                if line.startswith("class") and self._output and self._output[-1]:
   465	                    self._output.append("\n")
   466	                self._output.append(line + "\n")
   467	            if self._output and functions:
   468	                self._output.append("\n")
   469	            for line in functions:
   470	                self._output.append(line + "\n")
   471	        else:
   472	            for i, line in enumerate(output):
   473	                if (
   474	                    self._output
   475	                    and line.startswith("class")
   476	                    and (
   477	                        not self._output[-1].startswith("class")
   478	                        or (len(output) > i + 1 and output[i + 1].startswith("    "))
   479	                    )
   480	                ) or (
   481	                    self._output
   482	                    and self._output[-1].startswith("def")
   483	                    and not line.startswith("def")
   484	                ):
   485	                    self._output.append("\n")
   486	                self._output.append(line + "\n")
   487	        self.check_undefined_names()
   488
   489	    def is_skipped_attribute(self, attr: str) -> bool:
   490	        return (
   491	            attr
   492	            in (
   493	                "__class__",
   494	                "__getattribute__",
   495	                "__str__",
   496	                "__repr__",
   497	                "__doc__",
   498	                "__dict__",
   499	                "__module__",
   500	                "__weakref__",
   501	                "__annotations__",
   502	                "__firstlineno__",
   503	                "__static_attributes__",
   504	                "__annotate__",
   505	            )
   506	            or attr in self.IGNORED_DUNDERS
   507	            or is_pybind_skipped_attribute(attr)  # For pickling

   511	    def get_members(self, obj: object) -> list[tuple[str, Any]]:
   512	        obj_dict: Mapping[str, Any] = getattr(obj, "__dict__")  # noqa: B009
   513	        results = []
   514	        for name in obj_dict:
   515	            if self.is_skipped_attribute(name):
   516	                continue
   517	            # Try to get the value via getattr
   518	            try:
   519	                value = getattr(obj, name)
   520	            except AttributeError:
   521	                continue
   522	            else:
   523	                results.append((name, value))
   524	        return results

   526	    def get_type_annotation(self, obj: object) -> str:
   527	        """
   528	        Given an instance, return a string representation of its type that is valid
   529	        to use as a type annotation.
   530	        """
   531	        if obj is None or obj is type(None):
   532	            return "None"
   533	        elif inspect.isclass(obj):

   537	        elif isinstance(obj, ModuleType):
   538	            return self.add_name("types.ModuleType", require=False)
   539	        else:
   540	            return self.get_type_fullname(type(obj))

   568	    def is_staticmethod(self, class_info: ClassInfo | None, name: str, obj: object) -> bool:

   598	    def process_inferred_sigs(self, inferred: list[FunctionSig]) -> None:
   599	        for i, sig in enumerate(inferred):

   602	                    arg.type = self.strip_or_import(arg.type)

   606	    def generate_function_stub(
   607	        self, name: str, obj: object, *, output: list[str], class_info: ClassInfo | None = None
   608	    ) -> None:
   609	        """Generate stub for a single function or method.
   610
   611	        The result (always a single line) will be appended to 'output'.
   612	        If necessary, any required names will be added to 'imports'.
   613	        The 'class_name' is used to find signature of __init__ or __new__ in
   614	        'class_sigs'.
   615	        """
   616	        docstring: Any = getattr(obj, "__doc__", None)
   617	        if not isinstance(docstring, str):
   618	            docstring = None
   619
   620	        ctx = FunctionContext(
   621	            self.module_name,
   622	            name,
   623	            docstring=docstring,
   624	            is_abstract=self.is_abstract_method(obj),
   625	            class_info=class_info,
   626	        )
   627	        if self.is_private_name(name, ctx.fullname) or self.is_not_in_all(name):
   628	            return
   629
   630	        self.record_name(ctx.name)
   631	        default_sig = self.get_default_function_sig(obj, ctx)
   632	        inferred = self.get_signatures(default_sig, self.sig_generators, ctx)
   633	        self.process_inferred_sigs(inferred)
   634
   635	        decorators = []
   636	        if len(inferred) > 1:
   637	            decorators.append("@{}".format(self.add_name("typing.overload")))
   638
   639	        if ctx.is_abstract:
   640	            decorators.append("@{}".format(self.add_name("abc.abstractmethod")))
   641
   642	        if class_info is not None:
   643	            if self.is_staticmethod(class_info, name, obj):
   644	                decorators.append("@staticmethod")
   645	            else:
   646	                for sig in inferred:
   647	                    if not sig.args or sig.args[0].name not in ("self", "cls"):
   648	                        sig.args.insert(0, ArgSig(name=class_info.self_var))
   649	                # a sig generator indicates @classmethod by specifying the cls arg.
   650	                if inferred[0].args and inferred[0].args[0].name == "cls":
   651	                    decorators.append("@classmethod")
   652
   653	        docstring = self._indent_docstring(ctx.docstring) if ctx.docstring else None
   654	        output.extend(self.format_func_def(inferred, decorators=decorators, docstring=docstring))
   655	        self._fix_iter(ctx, inferred, output)
   656
   657	    def _indent_docstring(self, docstring: str) -> str:
   658	        """Fix indentation of docstring extracted from pybind11 or other binding generators."""
   659	        lines = docstring.splitlines(keepends=True)
   660	        indent = self._indent + "    "
   661	        if len(lines) > 1:
   662	            if not all(line.startswith(indent) or not line.strip() for line in lines):
   663	                # if the docstring is not indented, then indent all but the first line
   664	                for i, line in enumerate(lines[1:]):
   665	                    if line.strip():
   666	                        lines[i + 1] = indent + line
   667	        # if there's a trailing newline, add a final line to visually indent the quoted docstring
   668	        if lines[-1].endswith("\n"):
   669	            if len(lines) > 1:
   670	                lines.append(indent)
   671	            else:
   672	                lines[-1] = lines[-1][:-1]
   673	        return "".join(lines)

   675	    def _fix_iter(
   676	        self, ctx: FunctionContext, inferred: list[FunctionSig], output: list[str]
   677	    ) -> None:
   678	        """Ensure that objects which implement old-style iteration via __getitem__
   679	        are considered iterable.
   680	        """
   681	        if (
   682	            ctx.class_info
   683	            and ctx.class_info.cls is not None
   684	            and ctx.name == "__getitem__"
   685	            and "__iter__" not in ctx.class_info.cls.__dict__
   686	        ):
   687	            item_type: str | None = None
   688	            for sig in inferred:
   689	                if sig.args and sig.args[-1].type == "int":
   690	                    item_type = sig.ret_type
   691	                    break
   692	            if item_type is None:
   693	                return
   694	            obj = CFunctionStub(
   695	                "__iter__", f"def __iter__(self) -> typing.Iterator[{item_type}]\n"
   696	            )
   697	            self.generate_function_stub("__iter__", obj, output=output, class_info=ctx.class_info)

   699	    def generate_property_stub(
   700	        self,
   701	        name: str,
   702	        raw_obj: object,
   703	        obj: object,
   704	        static_properties: list[str],
   705	        rw_properties: list[str],
   706	        ro_properties: list[str],
   707	        class_info: ClassInfo | None = None,
   708	    ) -> None:
   709	        """Generate property stub using introspection of 'obj'.
   710
   711	        Try to infer type from docstring, append resulting lines to 'output'.
   712
   713	        raw_obj : object before evaluation of descriptor (if any)
   714	        obj : object after evaluation of descriptor
   715	        """
   716
   717	        docstring = getattr(raw_obj, "__doc__", None)
   718	        fget = getattr(raw_obj, "fget", None)
   719	        if fget:
   720	            alt_docstr = getattr(fget, "__doc__", None)
   721	            if alt_docstr and docstring:
   722	                docstring += "\n" + alt_docstr
   723	            elif alt_docstr:
   724	                docstring = alt_docstr
   725
   726	        ctx = FunctionContext(
   727	            self.module_name, name, docstring=docstring, is_abstract=False, class_info=class_info
   728	        )
   729
   730	        if self.is_private_name(name, ctx.fullname) or self.is_not_in_all(name):
   731	            return
   732
   733	        self.record_name(ctx.name)
   734	        static = self.is_static_property(raw_obj)
   735	        readonly = self.is_property_readonly(raw_obj)
   736	        if static:
   737	            ret_type: str | None = self.strip_or_import(self.get_type_annotation(obj))
   738	        else:
   739	            default_sig = self.get_default_function_sig(raw_obj, ctx)
   740	            ret_type = default_sig.ret_type
   741
   742	        inferred_type = self.get_property_type(ret_type, self.sig_generators, ctx)
   743	        if inferred_type is not None:
   744	            inferred_type = self.strip_or_import(inferred_type)
   745
   746	        if static:
   747	            classvar = self.add_name("typing.ClassVar")
   748	            trailing_comment = "  # read-only" if readonly else ""
   749	            if inferred_type is None:
   750	                inferred_type = self.add_name("_typeshed.Incomplete")
   751
   752	            static_properties.append(
   753	                f"{self._indent}{name}: {classvar}[{inferred_type}] = ...{trailing_comment}"
   754	            )
   755	        else:  # regular property
   756	            if readonly:
   757	                docstring = self._indent_docstring(ctx.docstring) if ctx.docstring else None
   758	                ro_properties.append(f"{self._indent}@property")
   759	                sig = FunctionSig(name, [ArgSig("self")], inferred_type, docstring=docstring)
   760	                ro_properties.append(
   761	                    sig.format_sig(
   762	                        indent=self._indent, include_docstrings=self._include_docstrings
   763	                    )
   764	                )
   765	            else:
   766	                if inferred_type is None:
   767	                    inferred_type = self.add_name("_typeshed.Incomplete")
   768
   769	                rw_properties.append(f"{self._indent}{name}: {inferred_type}")
   770
   771	    def get_type_fullname(self, typ: type) -> str:
   772	        """Given a type, return a string representation"""
   773	        if typ is Any:
   774	            return "Any"
   775	        typename = getattr(typ, "__qualname__", typ.__name__)
   776	        module_name = self.get_obj_module(typ)
   777	        if module_name is None:
   778	            # This should not normally happen, but some types may resist our
   779	            # introspection attempts too hard. See
   780	            # https://github.com/python/mypy/issues/19031
   781	            return "_typeshed.Incomplete"
   782	        if module_name != "builtins":
   783	            typename = f"{module_name}.{typename}"
   784	        return typename

   786	    def get_base_types(self, obj: type) -> list[str]:
   787	        all_bases = type.mro(obj)
   788	        if all_bases[-1] is object:
   789	            # TODO: Is this always object?
   790	            del all_bases[-1]
   791	        # remove pybind11_object. All classes generated by pybind11 have pybind11_object in their MRO,
   792	        # which only overrides a few functions in object type
   793	        if all_bases and all_bases[-1].__name__ == "pybind11_object":
   794	            del all_bases[-1]
   795	        # remove the class itself
   796	        all_bases = all_bases[1:]
   797	        # Remove base classes of other bases as redundant.
   798	        bases: list[type] = []
   799	        for base in all_bases:
   800	            if not any(issubclass(b, base) for b in bases):
   801	                bases.append(base)
   802	        return [self.strip_or_import(self.get_type_fullname(base)) for base in bases]

   810	        required names will be added to 'imports'.

   821	        ro_properties: list[str] = []

   833	            raw_value = raw_lookup.get(attr, value)
   834	            if self.is_method(class_info, attr, value) or self.is_classmethod(

   842	                        # better signature than __init__() ?
   843	                        continue
   844	                    attr = "__init__"
   845	                # FIXME: make this nicer
   846	                if self.is_staticmethod(class_info, attr, value
