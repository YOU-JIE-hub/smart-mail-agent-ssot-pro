   248	def _findOverlapRun(B,i,wrap):
   249	    '''find overlap run containing B[i]'''
   250	    n = len(B)
   251	    R = [i]
   252	    while 1:
   253	        i = R[-1]
   254	        j = (i+1)%n
   255	        if j in R or not boundsOverlap(B[i],B[j]): break
   256	        R.append(j)
   257	    while 1:
   258	        i = R[0]
   259	        j = (i-1)%n
   260	        if j in R or not boundsOverlap(B[i],B[j]): break
   261	        R.insert(0,j)
   262	    return R

   264	def findOverlapRun(B,wrap=1):
   265	    '''determine a set of overlaps in bounding boxes B or return None'''
   266	    n = len(B)
   267	    if n>1:
   268	        for i in range(n-1):
   269	            R = _findOverlapRun(B,i,wrap)
   270	            if len(R)>1: return R
   271	    return None

   376	def _fPLSide(l,width,side=None):
   377	    data = l._origdata
   378	    if side is None:
   379	        li = data['li']
   380	        ri = data['ri']
   381	        if li is None:
   382	            side = 1
   383	            i = ri
   384	        elif ri is None:
   385	            side = 0
   386	            i = li
   387	        elif li[1]-li[0]>ri[1]-ri[0]:
   388	            side = 0
   389	            i = li
   390	        else:
   391	            side = 1
   392	            i = ri
   393	    w = data['width']
   394	    edgePad = data['edgePad']
   395	    if not side:    #on left
   396	        l._pmv = 180
   397	        l.x = edgePad+w
   398	        i = data['li']
   399	    else:
   400	        l._pmv = 0
   401	        l.x = width - w - edgePad
   402	        i = data['ri']

   404	    data['smid'] = sin(mid/_180_pi)
   405	    data['cmid'] = cos(mid/_180_pi)
   406	    data['side'] = side
