   532	        """Handle a setup/call/teardown report, generating the appropriate
   533	        XML tags as necessary.
   534
   535	        Note: due to plugins like xdist, this hook may be called in interlaced
   536	        order with reports from other nodes. For example:
   537
   538	        Usual call order:
   539	            -> setup node1
   540	            -> call node1
   541	            -> teardown node1
   542	            -> setup node2
   543	            -> call node2
   544	            -> teardown node2

   547	            -> setup node1
   548	            -> call node1
   549	            -> setup node2
   550	            -> call node2
   551	            -> teardown node2
   552	            -> teardown node1
   553	        """

   555	        if report.passed:
   556	            if report.when == "call":  # ignore setup/teardown
   557	                reporter = self._opentestcase(report)
   558	                reporter.append_pass(report)
   559	        elif report.failed:
   560	            if report.when == "teardown":

   562	                report_wid = getattr(report, "worker_id", None)

   572	                        )
   573	                    ),
   574	                    None,
   575	                )
   576	                if close_report:
   577	                    # We need to open new testcase in case we have failure in
   578	                    # call and error in teardown in order to follow junit
   579	                    # schema.
   580	                    self.finalize(close_report)
   581	                    self.cnt_double_fail_tests += 1
   582	            reporter = self._opentestcase(report)
   583	            if report.when == "call":
   584	                reporter.append_failure(report)

   611	                None,
