   129	def solve_with_dependent(
   130	    vars: list[TypeVarId],
   131	    constraints: list[Constraint],
   132	    original_vars: list[TypeVarId],
   133	    originals: dict[TypeVarId, TypeVarLikeType],
   134	) -> tuple[Solutions, list[TypeVarLikeType]]:
   135	    """Solve set of constraints that may depend on each other, like T <: List[S].
   136
   137	    The whole algorithm consists of five steps:
   138	      * Propagate via linear constraints and use secondary constraints to get transitive closure
   139	      * Find dependencies between type variables, group them in SCCs, and sort topologically
   140	      * Check that all SCC are intrinsically linear, we can't solve (express) T <: List[T]
   141	      * Variables in leaf SCCs that don't have constant bounds are free (choose one per SCC)
   142	      * Solve constraints iteratively starting from leaves, updating bounds after each step.
   143	    """
   144	    graph, lowers, uppers = transitive_closure(vars, constraints)
   145
   146	    dmap = compute_dependencies(vars, graph, lowers, uppers)
   147	    sccs = list(strongly_connected_components(set(vars), dmap))
   148	    if not all(check_linear(scc, lowers, uppers) for scc in sccs):
   149	        return {}, []
   150	    raw_batches = list(topsort(prepare_sccs(sccs, dmap)))
   151
   152	    free_vars = []
   153	    free_solutions = {}
   154	    for scc in raw_batches[0]:
   155	        # If there are no bounds on this SCC, then the only meaningful solution we can
   156	        # express, is that each variable is equal to a new free variable. For example,
   157	        # if we have T <: S, S <: U, we deduce: T = S = U = <free>.
   158	        if all(not lowers[tv] and not uppers[tv] for tv in scc):
   159	            best_free = choose_free([originals[tv] for tv in scc], original_vars)
   160	            if best_free:
   161	                # TODO: failing to choose may cause leaking type variables,
   162	                # we need to fail gracefully instead.
   163	                free_vars.append(best_free.id)
   164	                free_solutions[best_free.id] = best_free
   165
   166	    # Update lowers/uppers with free vars, so these can now be used
   167	    # as valid solutions.
   168	    for l, u in graph:
   169	        if l in free_vars:
   170	            lowers[u].add(free_solutions[l])
   171	        if u in free_vars:
   172	            uppers[l].add(free_solutions[u])
   173
   174	    # Flatten the SCCs that are independent, we can solve them together,
   175	    # since we don't need to update any targets in between.
   176	    batches = []
   177	    for batch in raw_batches:
   178	        next_bc = []
   179	        for scc in batch:
   180	            next_bc.extend(list(scc))
   181	        batches.append(next_bc)
   182
   183	    solutions: dict[TypeVarId, Type | None] = {}
   184	    for flat_batch in batches:
   185	        res = solve_iteratively(flat_batch, graph, lowers, uppers)
   186	        solutions.update(res)
   187	    return solutions, [free_solutions[tv] for tv in free_vars]

   190	def solve_iteratively(
   191	    batch: list[TypeVarId], graph: Graph, lowers: Bounds, uppers: Bounds
   192	) -> Solutions:
   193	    """Solve transitive closure sequentially, updating upper/lower bounds after each step.
   194
   195	    Transitive closure is represented as a linear graph plus lower/upper bounds for each
   196	    type variable, see transitive_closure() docstring for details.
   197
   198	    We solve for type variables that appear in `batch`. If a bound is not constant (i.e. it
   199	    looks like T :> F[S, ...]), we substitute solutions found so far in the target F[S, ...]
   200	    after solving the batch.
   201
   202	    Importantly, after solving each variable in a batch, we move it from linear graph to
   203	    upper/lower bounds, this way we can guarantee consistency of solutions (see comment below
   204	    for an example when this is important).
   205	    """
   206	    solutions = {}
   207	    s_batch = set(batch)
   208	    while s_batch:
   209	        for tv in sorted(s_batch, key=lambda x: x.raw_id):
   210	            if lowers[tv] or uppers[tv]:
   211	                solvable_tv = tv
