   264	    def accept(self, node: Statement | Expression, *, can_borrow: bool = False) -> Value | None:
   265	        """Transform an expression or a statement.
   266
   267	        If can_borrow is true, prefer to generate a borrowed reference.
   268	        Borrowed references are faster since they don't require reference count
   269	        manipulation, but they are only safe to use in specific contexts.
   270	        """
   271	        with self.catch_errors(node.line):
   272	            if isinstance(node, Expression):
   273	                old_can_borrow = self.can_borrow
   274	                self.can_borrow = can_borrow
   275	                try:
   276	                    res = node.accept(self.visitor)
   277	                    res = self.coerce(res, self.node_type(node), node.line)
   278	                # If we hit an error during compilation, we want to
   279	                # keep trying, so we can produce more error
   280	                # messages. Generate a temp of the right type to keep
   281	                # from causing more downstream trouble.
   282	                except UnsupportedException:
   283	                    res = Register(self.node_type(node))
   284	                self.can_borrow = old_can_borrow
   285	                if not can_borrow:
   286	                    self.flush_keep_alives()
   287	                return res
   288	            else:
   289	                try:
   290	                    node.accept(self.visitor)
   291	                except UnsupportedException:
   292	                    pass
   293	                return None

   619	            line = lvalue.line

   621	            # If we are visiting a decorator, then the SymbolNode we really want to be looking at
   622	            # is the function that is decorated, not the entire Decorator node itself.
   623	            symbol = lvalue.node
   624	            if isinstance(symbol, Decorator):
   625	                symbol = symbol.func
   626	            if symbol is None:

   628	                assert lvalue.is_special_form

   639	                    # in the current function's environment class. Next, define a target that
   640	                    # refers to the newly defined variable in that environment class. Add the

   642	                    # current environment.

   644	                        return self.add_var_to_env_class(
   645	                            symbol, reg_type, self.fn_info.generator_class, reassign=False
   646	                        )

   649	                    return self.add_local_reg(symbol, reg_type)

   663	            return AssignmentTargetIndex(base, index)

   668	            return AssignmentTargetAttr(obj, lvalue.name, can_borrow=can_borrow)
