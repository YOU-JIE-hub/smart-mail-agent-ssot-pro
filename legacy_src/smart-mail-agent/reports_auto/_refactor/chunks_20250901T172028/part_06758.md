   103
   104	    Should return the names of all function arguments that:
   105	    * Aren't bound to an instance or type as in instance or class methods.
   106	    * Don't have default values.
   107	    * Aren't bound with functools.partial.
   108	    * Aren't replaced with mocks.

   110	    The cls arguments indicate that the function should be treated as a bound
   111	    method even though it's not unless the function is a static method.
   112
   113	    The name parameter should be the original name in which the function was collected.

   116	    # revisit fixtures. The fixture mechanism should ask the node for
   117	    # the fixture names, and not try to obtain directly from the
   118	    # function object well after collection has occurred.

   121	    # ordered mapping of parameter names to Parameter instances.  This
   122	    # creates a tuple of the names of the parameters that don't have
   123	    # defaults.
   124	    try:
   125	        parameters = signature(function).parameters.values()

   135	        p.name

   185	    r"""If val is pure ASCII, return it as an str, otherwise, escape
   186	    bytes objects into a sequence of escaped bytes:

   271	#
   272	# Consider this example:
   273	#
   274	#     MyUnion = Union[int, str]
   275	#
   276	#     def handle(x: MyUnion) -> int {
   277	#         if isinstance(x, int):
   278	#             return 1
   279	#         elif isinstance(x, str):
   280	#             return 2
   281	#         else:
   282	#             raise Exception('unreachable')
   283	#
   284	# Now suppose we add a new variant:
   285	#
   286	#     MyUnion = Union[int, str, bytes]
   287	#
   288	# After doing this, we must remember ourselves to go and update the handle
   289	# function to handle the new variant.
   290	#
   291	# With `assert_never` we can do better:
   292	#
   293	#     // raise Exception('unreachable')
   294	#     return assert_never(x)
   295	#
   296	# Now, if we forget to handle the new variant, the type-checker will emit a
   297	# compile-time error, instead of the runtime error we would have gotten
   298	# previously.
   299	#
   300	# This also work for Enums (if you use `is` to compare) and Literals.
   301	def assert_never(value: NoReturn) -> NoReturn:
   302	    assert False, f"Unhandled value: {value} ({type(value).__name__})"
